"""
Improved minimiser for 2 * n * max(conv(a,a)) / (sum(a)**2).

The algorithm works on the log‑it (softmax) parametrisation
    a_i = exp(b_i) / sum_j exp(b_j) ,   b ∈ ℝⁿ .
This automatically guarantees a_i ≥ 0 and ∑ a_i = 1, so no
explicit projection onto the simplex is needed.

Two optimisation stages are performed:

1.  Smoothed‑max (soft‑max) Adam – a smooth approximation of the max
    using a temperature ε that is gradually reduced.  The gradient
    of the smooth objective is
        g_a = 2 * conv_rev(w)  (w – soft‑max weights),
    and the corresponding gradient on the logits b is
        g_b = a * (g_a – ⟨a,g_a⟩).

2.  Exact‑max Adam – after ε becomes tiny we switch to the true
    max, using the sub‑gradient w that places equal mass on all
    active (maximal) convolution entries.

Both stages run Adam on the logits with bias‑corrected moments.
When the objective stops improving a simple simulated‑annealing
fine‑tuning with “mass‑transfer’’ moves is applied.

The whole procedure is time‑bounded (default 1000 s).  The final
sequence is clipped to the `[0,1000]` range and returned as a plain
Python list ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real‑valued, full length)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b (full length) via real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Stable soft‑max for logits → probabilities (simplex)
# ----------------------------------------------------------------------
def _softmax_logits(b: np.ndarray) -> np.ndarray:
    """Return soft‑max(b) as a probability vector."""
    b_shift = b - np.max(b)               # for numerical stability
    e = np.exp(b_shift)
    return e / e.sum()


# ----------------------------------------------------------------------
# Gradient of (smoothed) max‑convolution w.r.t. a,
# given a weight vector w over the 2n‑1 convolution entries.
# ----------------------------------------------------------------------
def _grad_a_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of   g(a)=max_w  a * a   (smoothed by w)   w.r.t. a.
    Implements  g_a = 2 * (rev(a) * w)  on the proper slice.
    """
    n = a.size
    rev_a = a[::-1]
    # convolution rev_a (size n) with w (size 2n‑1) → length 3n‑2
    conv = _full_convolution_fft(rev_a, w)
    # slice the middle n entries, which correspond to ∂/∂a_i
    return 2.0 * conv[n - 1:n - 1 + n]


# ----------------------------------------------------------------------
# Convert a gradient w.r.t. a to a gradient w.r.t. logits b.
# ----------------------------------------------------------------------
def _grad_logits_from_a(a: np.ndarray, g_a: np.ndarray) -> np.ndarray:
    """
    For a = softmax(b) we have ∂a/∂b = diag(a) - a aᵀ.
    Hence   g_b = a ⊙ (g_a - ⟨a,g_a⟩)
    where ⊙ denotes element‑wise product.
    """
    inner = np.dot(a, g_a)
    return a * (g_a - inner)


# ----------------------------------------------------------------------
# Exact objective (sum(a)=1) – used only for bookkeeping.
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (assumes sum(a)=1)."""
    n = a.size
    conv = _full_convolution_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Generate a diverse warm‑start of length n.
# ----------------------------------------------------------------------
def _warm_start(rng: np.random.Generator, n: int) -> np.ndarray:
    """Return a non‑negative vector of length n, normalised to sum = 1."""
    x = np.arange(n, dtype=np.float64)

    # a handful of analytic shapes
    shapes = []

    # exponential decay
    lam = rng.uniform(5e-4, 2e-2, size=2)
    for l in lam:
        shapes.append(np.exp(-l * x))

    # power‑law
    p = rng.uniform(0.3, 2.0, size=2)
    for pp in p:
        shapes.append((x + 1.0) ** (-pp))

    # half‑sine
    shapes.append(np.sin(np.pi * (x + 0.5) / n))

    # Gaussian (random width)
    sigma = rng.uniform(n / 8.0, n / 2.0)
    shapes.append(np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2))

    # simple triangle (peak in the middle)
    shapes.append(np.minimum(x, n - 1 - x) + 1.0)

    # random convex combination of the above
    weights = rng.random(len(shapes))
    weights /= weights.sum()
    a = sum(w * s for w, s in zip(weights, shapes))
    a = np.maximum(a, 0.0)

    # add a Dirichlet‑sampled perturbation (helps escaping symmetry)
    dirichlet = rng.dirichlet(np.ones(n) * 0.5)   # relatively sparse
    a = 0.7 * a + 0.3 * dirichlet
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

# Main entry point required by the grader.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):

    Return a non‑negative step‑function (list of floats) that (approximately)
    minimises the evaluation function.  The routine respects a `budget_s`
    second time limit and prints intermediate scores.

    rng = np.random.default_rng(seed)

    # Problem size – a moderately large n gives many degrees of freedom
    # while keeping a single FFT cheap (< 2 ms on a modern CPU).
    # ------------------------------------------------------------------

    # exists, otherwise generate a diverse starting point.
    # ------------------------------------------------------------------
    a0 = None

            if cand.shape[0] != n:

            a0 = np.maximum(0.0, cand)
            s = a0.sum()
            if s > 0:

            else:
                a0 = None
        except Exception:
            a0 = None

    if a0 is None:

    # logit_i = log(a_i)  (add a tiny epsilon for stability)
    eps_init = 1e-12
    b = np.log(a0 + eps_init)

    # Adam state for the logits
    m = np.zeros_like(b)
    v = np.zeros_like(b)

    beta1_pow = 1.0

    # ------------------------------------------------------------------

    eps_factor_decay = 0.9997

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_val = float(2.0 * n * max_conv)

        # track best

            best_val = cur_val
            best_seq = a.copy()
            print(f"[phase1 iter {it:06d}] new_best={best_val:.12f}  eps_factor={eps_factor:.2e}")

        # --------------------------------------------------------------
        # compute soft‑max weights (smooth approximation of max)
        # --------------------------------------------------------------
        eps = max(eps_factor_min, eps_factor * max_conv)

        if w_sum == 0 or not np.isfinite(w_sum):

        else:
            w /= w_sum

        # gradient w.r.t. a
        g_a = _grad_a_from_weights(a, w)

        # gradient w.r.t. logits b
        g_b = _grad_logits_from_a(a, g_a)

        # --------------------------------------------------------------
        # Adam update on logits
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g_b
        v = beta2 * v + (1.0 - beta2) * (g_b * g_b)

        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        if it % 4000 == 0:

            bump = rng.normal(scale=0.01, size=n)
            b += bump
            print(f"[phase1 perturb] iter {it}")

    # ------------------------------------------------------------------
    # Phase II – exact‑max Adam (sub‑gradient of the true max).
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.85 * budget_s
    it2 = 0
    # reset Adam moments for a fresh start (helps with a different geometry)
    m.fill(0.0)
    v.fill(0.0)
    beta1_pow = beta2_pow = 1.0
    lr = 0.08
    tol = 1e-14

    while time.time() < phase2_end:
        it2 += 1

        a = _softmax_logits(b)
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_val = float(2.0 * n * max_conv)

            best_val = cur_val
            best_seq = a.copy()
            print(f"[phase2 iter {it2:06d}] new_best={best_val:.12f}")

        # active set (indices attaining the maximum, within tolerance)
        active = np.flatnonzero(conv >= max_conv - tol)
        w = np.zeros_like(conv)

        else:                     # numerical fallback

        g_a = _grad_a_from_weights(a, w)

        g_b = _grad_logits_from_a(a, g_a)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_b
        v = beta2 * v + (1.0 - beta2) * (g_b * g_b)

        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        b -= step

        # slowly decay the learning rate
        lr = max(5e-5, lr * 0.99995)

        # occasional jitter if we are stuck for many iterations
        if it2 % 5000 == 0:

            b += rng.normal(scale=0.005, size=n)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    cur = best_seq.copy()

    Tmin = 1e-8

        delta = rng.uniform(0.0, max_transfer)

        new_score = _objective(new)

        else:

            cur_score = new_score

                best_seq = cur.copy()
                print(f"[SA iter {it_sa:07d}] new_best={best_val:.12f}")

            cur = np.maximum(cur, 0.0)
            cur /= cur.sum()

    # ------------------------------------------------------------------
