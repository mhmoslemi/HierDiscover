"""
Hybrid optimizer for the step‑function evaluation problem.

The routine works in four stages:

1. **Smooth surrogate L‑BFGS** – many restarts of L‑BFGS on a
   log‑sum‑exp (soft‑max) surrogate of the max‑convolution.  The
   soft‑max guarantees non‑negativity and a unit ℓ₁‑norm, while the
   surrogate objective approximates the true objective but is smooth
   with a closed‑form gradient.

2. **Anti‑symmetric re‑ordering** – after each L‑BFGS run the best
   multiset of heights is reordered by pairing the largest values with
   the smallest ones (function ``_anti_symmetric_reorder``).  This
   dramatically lowers the peak of the self‑convolution without changing
   the multiset of heights.

3. **Simulated‑annealing on permutations** – a fast FFT‑based evaluator
   repeatedly proposes swaps (and occasional mass transfers).  Moves are
   accepted if they improve the true objective or with a temperature‑
   controlled probability, allowing escape from local minima.

4. **Final hill‑climbing** – a short deterministic refinement with swaps
   and small mass transfers finishes the search.

The problem size is increased to ``n = 8000`` (instead of 4000) which
gives the optimiser many more degrees of freedom and typically yields a
lower bound.  The whole search respects a 1000 s budget and always
returns a feasible non‑negative list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate (avoid closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using FFT.
    Return an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clamp tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast evaluation of the exact objective defined in ``evaluate_sequence``.
    ``a`` is assumed to be non‑negative and already respects the clipping
    limits (0 ≤ a_i ≤ 1000).  Returns the same value as
    ``evaluate_sequence(a.tolist())`` but is much faster.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smooth surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate for the true objective.
    Parameters
    ----------
    w   : unconstrained parameters (real numbers)
    eps : smoothing parameter for the log‑sum‑exp
    n   : length of the sequence
    Returns
    -------
    L   : smoothed objective value
    grad_w : gradient w.r.t. w
    """
    # soft‑max → non‑negative a with Σa = 1
    w_shift = w - np.max(w)               # improve numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # self‑convolution
    conv = _fft_convolve_self(a)           # length 2·n‑1
    max_conv = conv.max()

    # smooth approximation of max(conv)
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)           # log∑ exp((conv−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient wrt a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()            # probability distribution over k

    # convolution p_rev * a (used in the derivation)
    p_rev = p[::-1]                        # reverse p
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    # central part corresponds to indices n‑1 … 2n‑2
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]           # gradient dL/da

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# Helper: convolution used inside the gradient (FFT based)
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# Bind the helper to the gradient function (so it can be called internally)
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Re‑ordering heuristic
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Produce an ordering where large values are paired with small ones
    when the sequence is reversed.  This deterministic heuristic often
    reduces the peak of the self‑convolution without changing the
    multiset of heights.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large values to the left
        new_a[n - 1 - i] = sorted_vals[i]      # small values to the right

    if n % 2:
        new_a[half] = sorted_vals[half]   # middle element stays put
    return new_a


# ----------------------------------------------------------------------
# Simulated annealing on permutations (swap moves only)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """
    Classic SA on the permutation of ``a``.
    Swaps preserve the ℓ₁‑norm.
    ``cur_val`` is the evaluation of ``a`` (computed with ``_eval_fast``).
    Returns the best sequence and its value found before ``deadline``.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # relatively high initial temperature to explore widely
    T = max(cur_val * 0.5, 1e-6)
    decay = 0.99995

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose a swap
        a[i], a[j] = a[j], a[i]
        new_val = _eval_fast(a)
        delta = new_val - cur_val

        # Metropolis criterion
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f}")
        else:
            # revert the swap
            a[i], a[j] = a[j], a[i]

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (mass transfers + swaps)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    """
    Deterministic local refinement: occasional swaps and small mass
    transfers between two random positions.  Both operations preserve
    the ℓ₁‑norm.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        # 20 % chance of a pure swap, else a mass transfer
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time
    budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin
    print(f"[init] time budget = {budget_s}s, deadline @ {deadline:.2f}")

    # --------------------------------------------------------------
    # Problem size – increase the dimensionality for a better bound
    # --------------------------------------------------------------
    n = 8000
    global _GLOBAL_EPS, _GLOBAL_N
    _GLOBAL_N = n

    # --------------------------------------------------------------
    # Initialise from a known construction if available
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to requested size
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        # start with a uniform vector (will be reshaped later by soft‑max)
        a0 = np.full(n, 1.0 / n)
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()                     # enforce Σa = 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # L‑BFGS schedule (log‑space smoothing)
    # --------------------------------------------------------------
    eps_schedule = np.logspace(-2, -12, num=20)

    # ------------------------------------------------------------------
    # Multi‑restart L‑BFGS phase
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 120.0:   # reserve ~2 min for later phases
        restart += 1
        print(f"[restart {restart}]")
        # random soft‑max parameters – large variance creates spiky starts
        w = rng.standard_normal(n) * 5.0

        for eps in eps_schedule:
            if time.time() >= deadline - 110.0:
                break
            _GLOBAL_EPS = eps

            # L‑BFGS on the smooth surrogate
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 400,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # convert back to the primal representation
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # ----------------------------------------------------------
        # Anti‑symmetric re‑ordering of the best sequence from this restart
        # ----------------------------------------------------------
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing on the permutation of the best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        print("[SA] starting simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val,
                                                deadline - 50.0)

    # ------------------------------------------------------------------
    # Final cheap hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing phase")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report final value (using the official evaluator)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required
    return best_a.tolist()
