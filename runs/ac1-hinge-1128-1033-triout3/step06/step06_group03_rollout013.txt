"""
A fast stochastic–gradient optimiser for the step‑function problem.

The objective to minimise (up to a constant 2·n) is
    f(a) = max_k (a * a)[k] / (sum a)^2 .
We keep the vector on the probability simplex (sum=1, a_i ≥ 0) – the scaling
factor disappears.  The algorithm works on a fixed length `n` (chosen as 5000
for a good trade‑off between quality and speed) and consists of three
phases:

1.  Simulated‑annealing style updates driven by the gradient of a smooth
    approximation of the max (soft‑max with a large λ).  The gradient is a
    correlation of the current sequence with the soft‑max weights and is
    computed by a single FFT pair.  After each step the vector is projected
    back onto the probability simplex.

2.  If the search stagnates we randomly re‑initialise the current point
    (a Dirichlet draw) to escape local minima.

3.  A short guided hill‑climbing refinement moves a tiny amount of mass away
    from the most offending convolution lag – the same routine used in the
    previous state‑of‑the‑art code.

The whole routine respects the 1000 s budget and prints occasional progress
messages.  The best sequence found is finally evaluated with the provided
``evaluate_sequence`` and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)           # eliminate tiny negatives
    return conv


def _corr_fft(a_rev: np.ndarray, w: np.ndarray, out_len: int) -> np.ndarray:
    """
    Correlation a⋆w = conv(a_rev, w) truncated to `out_len` entries.
    Returns the central `out_len` entries (indices n‑1 … n‑1+out_len‑1).
    """
    size = _next_pow2(a_rev.shape[0] + w.shape[0] - 1)
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(w, size)
    conv_full = np.fft.irfft(fa * fw, size)
    start = a_rev.shape[0] - 1                # = n‑1
    return conv_full[start:start + out_len]


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Small local search that repeatedly moves a little mass away from the
    lag where the autocorrelation attains its maximum.
    """
    n = a.shape[0]
    cur_a = a.copy()
    step_frac = 0.05

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Helper: objective for an internal vector (no clipping)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum(a)^2 for a non‑negative vector."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a)
    M = conv.max()
    return float(2.0 * n * M / (s * s))

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a step‑function by stochastic gradient descent on a smooth max,
    occasional random restarts and a final guided hill‑climb.  Returns the
    best list of non‑negative heights found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5           # safety margin

    # problem size – 5000 is a comfortable trade‑off between quality and speed
    n = 5000

    # ----- initialise with a random Dirichlet draw --------------------
    alpha = 0.7                                 # Dirichlet concentration
    a = rng.gamma(alpha, size=n)
    a /= a.sum()                               # normalise to sum = 1
    cur_val = _objective(a)

    best_a = a.copy()
    best_val = cur_val
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # ----- hyper‑parameters for the smoothed max gradient descent -------
    lam = 20000.0                               # λ for soft‑max (very sharp)
    step0 = 0.04                                 # initial step magnitude
    min_step = 1e-6
    T0 = 0.05 * best_val + 1e-12                # initial temperature for SA
    stagnation_counter = 0
    max_stagnation = 4000                       # after this we re‑initialise

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # full autocorrelation of current a
        conv = _full_convolution(a)
        maxc = conv.max()

        # soft‑max weights over the lag axis (numerically stable)
        w = np.exp(lam * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:                # extreme case – fall back to uniform
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2 * correlation(a, w)
        a_rev = a[::-1]
        grad = 2.0 * _corr_fft(a_rev, w, n)

        # temperature and step size follow a linear cooling schedule
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        temp = T0 * max(1.0 - frac, 0.0)
        step = step0 * max(1.0 - frac, 0.0) + min_step

        # propose a new point and project onto the simplex
        cand = a - step * grad
        cand = _simplex_projection(cand)

        cand_val = _objective(cand)

        # simulated‑annealing acceptance rule
        if cand_val < cur_val:
            a = cand
            cur_val = cand_val
            improved = True
        else:
            if temp > 1e-12:
                prob = np.exp(-(cand_val - cur_val) / temp)
                if rng.random() < prob:
                    a = cand
                    cur_val = cand_val
                    improved = True
                else:
                    improved = False
            else:
                improved = False

        # track the global best
        if cand_val < best_val - 1e-12:
            best_val = cand_val
            best_a = cand.copy()
            print(f"[iter {iteration}] new best = {best_val:.6f}")
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        # if we have been stuck for a long time, re‑initialise around the best
        if stagnation_counter >= max_stagnation:
            # small Gaussian perturbation around the best known point
            pert = rng.normal(scale=0.01, size=n)
            a = _simplex_projection(best_a + pert)
            cur_val = _objective(a)
            stagnation_counter = 0
            print(f"[restart] after {iteration} iters – current obj {cur_val:.6f}")

    # ----- final guided hill‑climbing refinement -----------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")

    return best_a.tolist()
