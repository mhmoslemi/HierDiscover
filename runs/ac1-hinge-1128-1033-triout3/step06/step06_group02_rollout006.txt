"""
Propose a non‑negative step‐function minimizing evaluate_sequence.

The algorithm consists of:
  1. Multiple random initialisations (uniform, Dirichlet, sinusoid,
     previous best) and keeps the best.
  2. A long L‑BFGS optimisation on a smooth log‑sum‑exp surrogate of
     the max‑convolution, with a decreasing smoothing parameter.
  3. A projected gradient descent on the exact max‑convolution (mirror
     descent) that exploits the analytic gradient.
  4. Simulated‑annealing style random mass‑transfer moves that can
     escape shallow local minima.
  5. A final cheap hill‑climbing pass (swap / transfer) to polish the
     result.

All operations stay inside the feasible set (a_i ≥ 0, Σ a_i = 1) and
fit comfortably inside the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# global placeholders used by the scipy‑minimize wrapper
_GLOBAL_EPS = None
_GLOBAL_N = None

def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater or equal to x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # Eliminate tiny negative values caused by round‑off.
    np.maximum(conv, 0, out=conv)
    return conv

def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed objective L_ε and its gradient w.r.t. the soft‑max parameters w.

    The actual decision variables are a_i = exp(w_i) / Σ_j exp(w_j).
    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N
    # stable soft‑max
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = a.sum()
    a = a / a_sum

    conv = _full_convolution(a)          # length 2·n‑1
    max_c = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_c) / eps
    # log‑sum‑exp in a numerically stable way
    log_sum = logsumexp(shifted)

    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # gradient w.r.t a
    # p_k = exp((conv_k - max_c)/ε) / Σ_t exp((conv_t - max_c)/ε)
    weights = np.exp(shifted)
    p = weights / weights.sum()

    # derivative of max_c term: for each a_i, d(max_c)/da_i = a_{k*-i}
    # where k* is the index of the maximum. The smoothed term adds a
    # convolution with p.
    # The overall gradient w.r.t a can be written as:
    #   grad_a = 4·n·(a * p_rev) where * denotes linear convolution.
    # The expression below computes a * p_rev efficiently via FFT.
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # The segment conv_ap[n-1 : 2·n-1] corresponds to the required part.
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]    # reverse to line up with a_i ordering

    # chain rule through the soft‑max (see e.g. “softmax‑gradient”)
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w

def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation of the algorithm from
    # Wang & Carreira‑Perpiñán (2013), “Projection onto the probability simplex”.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback (should not happen for our use‑case)
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _fast_evaluate(a: np.ndarray) -> float:
    """Fast version of evaluate_sequence for non‑negative, sum‑one vectors."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    # sum_a is 1 after normalisation, but keep the formula for safety.
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))

def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Heuristic that places large values on one side and small on the other."""
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a

def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %). The vector is kept on
    the simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            # move at most 5 % of the mass at i
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Re‑project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.

    The pipeline is:
      * several random initialisations (including the provided
        ``height_sequence_1`` if present);
      * a long L‑BFGS optimisation on a smooth surrogate of the
        max‑convolution;
      * projected mirror‑descent on the exact max‑convolution;
      * a simulated‑annealing phase of random mass transfers;
      * a final cheap hill‑climb.
    The function respects the time budget and returns the best sequence
    found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # keep a small safety margin

    # --------------------------------------------------------------
    # Problem size – use the length of the known construction if
    # available, otherwise fall back to 4000.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # --------------------------------------------------------------
    # Generate a handful of start points.
    # --------------------------------------------------------------
    init_candidates = []

    # 1. Uniform distribution.
    init_candidates.append(np.full(n, 1.0 / n))

    # 2. The previous best construction (if any).
    if has_old:
        init_candidates.append(old / old.sum())

    # 3. Random Dirichlet (α = 1 – roughly exponential).
    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))
    # 4. Random Dirichlet (α = 5 – smoother).
    init_candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # 5. Sinusoidal shape (helps to spread mass).
    x = np.linspace(0, np.pi, n)
    sinusoid = np.sin(x) + 0.1   # avoid zeros
    init_candidates.append(sinusoid)

    # Normalise all candidates onto the simplex.
    init_candidates = [c / c.sum() for c in init_candidates]

    # Keep the best of the initial batch.
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best among seeds = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on a smooth log‑sum‑exp surrogate.
    # ------------------------------------------------------------------
    # The surrogate is  L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k e^{(conv_k‑max)/ε}.
    # We start with a relatively large ε and decrease it geometrically.
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=12)

    # Soft‑max parameters (log‑space) – initialise from the current best.
    w = np.log(best_a + 1e-12)

    # Perform several independent restarts (different random seeds)
    # to increase the chance of escaping local minima.
    restart = 0
    max_restarts = 12
    while time.time() < deadline - 30.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        if restart % 4 == 0:
            w = np.random.randn(n) * 2.0   # fresh random start

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 500,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            # Convert back to a (soft‑max) probability vector.
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # Occasionally test the anti‑symmetric reordering.
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Mirror‑descent on the exact max‑convolution.
    # ------------------------------------------------------------------
    a = best_a.copy()
    alpha0 = 0.5                     # initial step length
    decay = 0.9995                  # geometric decay per iteration
    min_alpha = 0.005
    it = 0
    while time.time() < deadline - 20.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())
        # Gradient of the (non‑smooth) max wrt a:
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = a[jdx]                     # g_i = a_{k*-i}
        # Exponential mirror‑descent update.
        alpha = max(alpha0 * (decay ** (it // 200)), min_alpha)
        a = a * np.exp(-alpha * g)

        # Re‑project onto the simplex.
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        if it % 100 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[grad it {it}] new best = {best_val:.6f} (max conv {conv[max_k]:.8f})")
            # anti‑symmetric re‑ordering from time to time
            if it % 500 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[grad reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing style random mass transfers.
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T0 = 0.02
    T = T0
    # Cooling schedule: multiply by factor < 1 each iteration.
    cooling = 0.99997
    rng = np.random.default_rng(seed + 12345)
    while time.time() < deadline - 10.0:
        # Choose move type.
        if rng.random() < 0.5:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            # transfer up to 8 % of the mass at i
            delta = cur_a[i] * 0.08 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        # Project onto the simplex (the operations already preserve sum,
        # but rounding errors may accumulate).
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion.
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))
        if accept:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        # Cool down.

    # ------------------------------------------------------------------

    try:

        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()

    """
    eps = _GLOBAL_EPS
    n = _GLOBAL_N

    # ---------- soft‑max ----------
    w_shift = w - np.max(w)          # numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ---------- convolution ----------
    conv = _full_convolution(a)      # length 2·n‑1
    max_c = conv.max()

    # ---------- smooth max ----------
    shifted = (conv - max_c) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_c + 2.0 * n * eps * log_sum

    # ---------- gradient w.r.t. a ----------
    weights = np.exp(shifted)
    p = weights / weights.sum()      # distribution over lag indices

    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # segment that corresponds to the derivative
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]    # reverse to align with a_i order

    # ---------- chain rule through the soft‑max ----------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray) -> tuple[float, np.ndarray]:
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Implementation of the algorithm from Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _fast_evaluate(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` for non‑negative vectors that sum to 1.
    Returns 2·n·max(conv) (the denominator Σ a_i = 1, so the formula simplifies).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that places large values on one side and small values on the
    opposite side – often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that only accepts strict improvements.
    Random swaps (20 %) and mass transfers (80 %). The vector is kept on the
    simplex after each move.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # Project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _fast_evaluate(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.

    The procedure follows the four‑phase outline described in the module
    docstring and respects the allotted time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – use the length of the known construction if available.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
        has_old = True
    except Exception:
        n = 4000
        has_old = False

    # ------------------------------------------------------------------
    # Generate a few diverse starting points.
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # previous construction (if any)
    if has_old:
        init_candidates.append(old / old.sum())

    # random Dirichlet (α = 1 – exponential‑like)
    init_candidates.append(rng.gamma(shape=1.0, scale=1.0, size=n))

    # random Dirichlet (α = 5 – smoother)
    init_candidates.append(rng.gamma(shape=5.0, scale=1.0, size=n))

    # simple sinusoidal shape
    x = np.linspace(0, np.pi, n)
    init_candidates.append(np.sin(x) + 0.1)

    # normalise everything onto the simplex
    init_candidates = [c / c.sum() for c in init_candidates]

    # ------------------------------------------------------------------
    # Pick the best of the cheap initial candidates.
    # ------------------------------------------------------------------
    best_a = None
    best_val = float('inf')
    for idx, cand in enumerate(init_candidates):
        val = _fast_evaluate(cand)
        print(f"[init {idx}] evaluation = {val:.6f}")
        if val < best_val:
            best_val = val
            best_a = cand.copy()
    print(f"[init] best initial value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on a smooth surrogate of the max‑convolution.
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=12)
    w = np.log(best_a + 1e-12)           # initialise soft‑max parameters

    max_restarts = 12
    restart = 0
    while time.time() < deadline - 30.0 and restart < max_restarts:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        if restart % 4 == 0:
            w = np.random.randn(n) * 2.0   # fresh random start

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 500,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            # convert back to probabilities via soft‑max
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _fast_evaluate(a_candidate)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # try the anti‑symmetric reorder after each restart
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _fast_evaluate(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent on the exact max‑convolution.
    # ------------------------------------------------------------------
    a = best_a.copy()
    alpha0 = 0.5          # initial step size
    decay = 0.9995        # geometric decay
    min_alpha = 0.005
    it = 0
    while time.time() < deadline - 20.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # gradient of max(conv) w.r.t. a:
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = a[jdx]                     # g_i = a_{k*-i}

        # exponential mirror‑descent update
        alpha = max(alpha0 * (decay ** (it // 200)), min_alpha)
        a = a * np.exp(-alpha * g)

        # project onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        if it % 100 == 0:
            cur_val = _fast_evaluate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[grad it {it}] new best = {best_val:.6f} (max conv {conv[max_k]:.8f})")
            if it % 500 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _fast_evaluate(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[grad reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing style random mass transfers.
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    T = 0.02                # initial temperature
    cooling = 0.99997
    rng = np.random.default_rng(seed + 12345)

    while time.time() < deadline - 10.0:
        if rng.random() < 0.5:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.08 * rng.random()   # up to 8 % of a_i
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:
            # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]

        # re‑normalize (floating‑point drift)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_evaluate(new_a)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12))

        if accept:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[SA] new best = {best_val:.6f}")

        T = max(T * cooling, 1e-6)

    # ------------------------------------------------------------------
    # Phase 4 – final cheap hill‑climbing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Final safety check (use the official evaluator) and return.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
