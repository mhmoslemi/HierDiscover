        A_ub[i:i+n, i] = a_old

    x = result.x

    t_new = x[-1]  # new estimate of max_b

    return a_new, t_new

def full_convolution(x):
    n = len(x)
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def objective(a):
    n = len(a)
    b = full_convolution(a)
    return float(2.0 * n * b.max())

    a = a_init.copy()
    n = len(a)
    best_a = a.copy()
    best_val = objective(a)
    iteration = 0
    while time.time() < deadline:
        if iteration >= max_iter:
            break

        b = full_convolution(a)
        cur_max = b.max()
        cur_obj = 2.0 * n * cur_max
        if cur_obj < best_val - 1e-12:
            best_val = cur_obj
            best_a = a.copy()
            print(f"[LP {iteration:04d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        rows = 2 * n - 1

        c[-1] = 1.0  # minimize t

            print("[LP] linprog exception:", exc)
            break

            print("[LP] linprog failed:", res.message)
            break

        x = res.x
        a_new = x[:n]
        # t_new = x[-1]  # predicted bound, not needed for next iteration

        # numerical cleaning: set tiny negatives to zero (should be none)
        a_new = np.maximum(a_new, 0.0)

            break

def hillclimb_optimize(a_init, deadline, target_val=1.503, stagnation_limit=200000):
    a = a_init.copy()
    n = len(a)
    # compute conv once
    b = full_convolution(a)
    cur_val = objective(a)
    best_a = a.copy()
    best_val = cur_val
    iteration = 0
    no_improve = 0
    start_time = time.time()
    while time.time() < deadline:
        # progress-controlled max transfer fraction
        elapsed = time.time() - start_time
        progress = elapsed / (deadline - start_time)
        max_frac = 0.60 * (1 - progress) + 0.02 * progress

        # locate current max conv entry
        k_max = int(np.argmax(b))

        # compute contributions at k_max
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source selection
        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = np.random.randint(n)

        if a[i] <= 1e-12:
            continue

        # target selection
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + 1e-12)
        target_weights[i] = 0.0
        sum_tw = target_weights.sum()
        if sum_tw > 0:
            j = np.random.choice(n, p=target_weights / sum_tw)
        else:
            j = np.random.randint(n-1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0:
            continue

        # incremental update of convolution
        # b_new = b copy
        b_new = b.copy()
        # update contributions from moving delta from i to j.
        # When we subtract delta from a[i] and add to a[j], the convolution changes:
        # For each k where a[i] appears (i.e., b[k] contains a[i]*a[?]), we need to adjust.
        # We can use vectorized update as in original code: b_new[j:j+n] += 2*delta*a
        # and b_new[i:i+n] -= 2*delta*a
        # plus adjusting the quadratic delta^2 terms at positions 2*j, 2*i, i+j

        b_new[j:j+n] += 2.0 * delta * a
        b_new[i:i+n] -= 2.0 * delta * a
        b_new[2*j] += delta * delta
        b_new[2*i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = float(2.0 * n * new_max)

        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:07d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > stagnation_limit:
            # perturb
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = full_convolution(a)
            cur_val = objective(a)
            no_improve = 0
            print(f"[Hill {iteration:07d}] perturb")
        iteration += 1
    return best_a, best_val

"""
Step‑function optimiser.

* Initialise a non‑negative vector on the probability simplex using a
  one‑parameter “U‑shaped’’ power‑law family.
* If scipy is available, perform a few iterations of a *linearised
  majorisation‑minimisation* scheme:
    – minimise a new variable t (an upper bound on max conv) while
      keeping the convolution of the current iterate with the new
      candidate ≤ t (linearised constraints).
    – each iteration solves a dense linear programme (≈4000 constraints,
      2000 variables) with the HiGHS solver; the LP guarantees a
      non‑increasing bound.
* Finish the optimisation with a fast hill‑climbing phase that repeatedly
  transfers a small amount of mass from indices that contribute most to
  the current peak of the autocorrelation to indices that contribute
  little.  Moves are accepted only when the exact objective improves;
  after long stagnation a small random perturbation is applied.
* The best vector found is returned as a plain Python list; it is
  automatically normalised (sum = 1) and therefore compatible with the
  provided ``evaluate_sequence`` wrapper.
"""

import time
import numpy as np

# Try to import the linear‑programming routine; fall back to hill‑climb only
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Phase 0 – cheap parametric sweep to obtain a good deterministic seed
# ----------------------------------------------------------------------
def _initial_shape(n: int):
    """Search a one‑parameter U‑shape family a_i ∝ ((i+½)(n‑i‑½))^{-γ}."""
    best_a = None
    best_val = np.inf
    x = np.arange(n, dtype=np.float64)
    for gamma in np.linspace(0.0, 4.0, 41):
        denom = (x + 0.5) * (n - x - 0.5)
        a = np.power(denom, -gamma)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – Linearised majorisation‑minimisation (LP)
# ----------------------------------------------------------------------
def _lp_optimize(a_start: np.ndarray,
                 deadline: float,
                 target_val: float = 1.503,
                 max_iter: int = 200) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a linear programme that linearises the max‑convolution
    constraints at the current iterate.  Returns the best vector found.
    """
    a = a_start.copy()
    n = a.shape[0]
    rows = 2 * n - 1
    cols = n + 1                     # extra column for the auxiliary variable t

    # constant part of the constraint matrix (the column for t)
    A_ub = np.zeros((rows, cols), dtype=np.float64)
    A_ub[:, -1] = -1.0               # coefficient of t

    # equality: Σ a_i = 1
    A_eq = np.zeros((1, cols), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # variable bounds (a_i ≥ 0, t ≥ 0)
    bounds = [(0, None)] * n + [(0, None)]

    # objective: minimise t
    c = np.zeros(cols, dtype=np.float64)
    c[-1] = 1.0

    best_a = a.copy()
    best_val = _objective(a)

    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        # current full convolution and maximal entry
        b = _full_convolution(a)
        cur_obj = _objective(a)

        if cur_obj < best_val - 1e-12:
            best_val = cur_obj
            best_a = a.copy()
            print(f"[LP {iteration:04d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        # ----- build the linearised constraints for the current a -----
        # reset the part that depends on a
        A_ub[:, :n] = 0.0
        for i in range(n):
            A_ub[i:i + n, i] = a           # places a_i on the appropriate rows
        A_ub[:, :n] *= 2.0                  # factor 2 * a_{k-i}

        # RHS of the inequalities: b_k (the exact convolution values)
        b_ub = b.copy()

        # ----- solve the LP -------------------------------------------------
        try:
            res = linprog(c,
                          A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds,
                          method='highs')
        except Exception as exc:            # pragma: no cover
            print("[LP] linprog raised exception:", exc)
            break

        if not res.success:
            print("[LP] linprog failed:", res.message)
            break

        x = res.x
        a_new = x[:n]

        # numerical safety
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            print("[LP] zero solution, abort")
            break
        a_new /= s

        # evaluate improvement on the *exact* objective
        new_obj = _objective(a_new)
        if new_obj < cur_obj - 1e-12:
            a = a_new
        else:
            # no progress – stop to avoid endless looping
            print("[LP] no improvement after LP step")
            break

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – fast hill‑climbing (same spirit as the original code)
# ----------------------------------------------------------------------
def _hillclimb_optimize(a_start: np.ndarray,
                        deadline: float,
                        target_val: float = 1.503,
                        stagnation_limit: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Targeted hill‑climbing that moves a small amount of mass from a source
    index (large contribution to the current maximal convolution entry) to
    a target index (small contribution).  Stagnation triggers a tiny random
    perturbation.
    """
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = _objective(a)

    best_a = a.copy()
    best_val = cur_val

    max_frac_start, max_frac_end = 0.60, 0.02
    it = 0
    no_improve = 0
    start = time.time()
    eps = 1e-12

    while time.time() < deadline:
        # schedule for maximal transferable fraction
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ----- choose source (biased by contribution) -----
        if total_contrib > 0.0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ----- choose target (bias towards *small* contribution) -----
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0                     # avoid picking the source itself
        w_sum = target_weights.sum()
        if w_sum > 0.0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept only strictly improving moves
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:07d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ----- stagnation handling -----
        if no_improve > stagnation_limit:
            # small random perturbation around the best seen so far
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = _objective(a)
            no_improve = 0
            print(f"[Hill {it:07d}] stagnation – perturb")
        it += 1

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimises a non‑negative step function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin of 0.5 s for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Phase 0 – deterministic parametric sweep for a good seed.
    # ------------------------------------------------------------------
    a, cur_val = _initial_shape(n)
    best_a = a.copy()
    best_val = cur_val
    print(f"[init] parametric sweep gave value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – linearised majorisation‑minimisation (requires scipy).
    # ------------------------------------------------------------------
    if _SCIPY_AVAILABLE:
        # allocate roughly 60 % of the remaining budget to the LP phase
        lp_deadline = time.time() + 0.60 * (deadline - time.time())
        best_a, best_val = _lp_optimize(best_a,
                                         deadline=lp_deadline,
                                         target_val=1.503,
                                         max_iter=200)
        print(f"[LP] finished with best_val = {best_val:.6f}")
    else:
        print("[LP] scipy not available – skipping LP phase")

    # ------------------------------------------------------------------
    # Phase 2 – fast targeted hill‑climbing to polish the result.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _hillclimb_optimize(best_a,
                                               deadline=deadline,
                                               target_val=1.503,
                                               stagnation_limit=150_000)
        print(f"[Hill] final best_val = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
