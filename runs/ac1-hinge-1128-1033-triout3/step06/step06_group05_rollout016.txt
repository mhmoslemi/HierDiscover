"""
Propose a step‑function (non‑negative heights) that minimizes
`evaluate_sequence`.  The algorithm consists of three phases:

1. **Initialisation** – start from the known construction `height_sequence_1`
   if it exists (interpolated to the chosen length) or from a random
   point on the probability simplex.

2. **Smooth‑max “soft‑max” descent** – minimise a smooth approximation of
   the L∞‑norm of the auto‑convolution.
   For a vector a we set b = a∗a and define
       L(a) = (1/α)·log( Σ_k exp(α·b_k) ).
   When α → ∞, L → max(b).  The gradient of L is obtained by a single
   FFT‑based cross‑correlation and the update is performed with the Adam
   optimiser.  α is increased from 1 to 30 during the phase, while the
   learning‑rate decays from 0.5 to 0.01.  After each step we project back
   onto the simplex (Σa = 1, a≥0).

3. **Deterministic greedy hill‑climbing** – starting from the best point
   of phase 2 we repeatedly move a small amount of mass from the index
   that contributes most to the current worst convolution entry to the
   index that contributes least.  The full convolution is updated
   incrementally in O(n) time, allowing tens of millions of cheap moves.
   When progress stalls we re‑inject a tiny Gaussian perturbation around
   the best solution found so far.

The whole routine respects the 1000 s wall‑time limit, prints intermediate
best scores and finally returns the best sequence (as a plain Python list)
found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length = 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _cross_corr(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation (x ⋆ y) of real vectors.
    Result length = len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    size = 1 << (L - 1).bit_length()
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    corr = np.fft.irfft(X * np.conj(Y), size)[:L]
    return corr


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    s = w.sum()
    if s > 0:
        w /= s
    else:                     # numerical safeguard
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w

# Phase 2 – Smoothed max minimisation (soft‑max) with Adam
# ----------------------------------------------------------------------
def _smooth_max_phase(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimisation of the soft‑max surrogate of max(b)."""
    start = time.time()
    phase_len = deadline - start

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    it = 0
    while time.time() < deadline:
        # current convolution and objective
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth {it:7d}] new best = {best_val:.6f}")

        # --- soft‑max weights ----------------------------------------
        # α ramps from 1 up to 30 linearly
        progress = (time.time() - start) / phase_len
        alpha = 1.0 + 29.0 * progress

        # shift by max to avoid overflow
        exp_arg = alpha * (b - cur_max)
        # exp may under‑flow, but that just gives zero weight – okay
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(b, 1.0 / b.size)
        else:
            w /= w_sum

        # grad_i = 2 * Σ_k w_k * a_{k-i}  = 2 * (w ⋆ a)[i]
        grad = 2.0 * _cross_corr(w, a)[:n]

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate linearly decays 0.5 → 0.01
        lr = 0.5 * (1.0 - progress) + 0.01 * progress

        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a_candidate)

        # occasional tiny perturbation to escape plateaus
        if it % 15000 == 0:
            a += np.random.normal(scale=0.001, size=n)
            a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – Greedy hill‑climbing on the exact max‑objective
# ----------------------------------------------------------------------
def _greedy_hillclimb(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> np.ndarray:
    """
    Local search that repeatedly transfers a small amount of mass from the
    index that contributes most to the current worst convolution entry to the
    index that contributes least.  The full convolution is updated
    incrementally in O(n) time.
    """
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # contribution of each a_i to the presently worst entry b[max_idx]
        k = max_idx
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]          # proportional to ∂b_k/∂a_i (ignores factor 2)

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        # if degenerate, pick random distinct pair
        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        # Ensure source has some mass
        if a[source] <= 1e-14:
            nz = np.where(a > 1e-14)[0]
            if nz.size == 0:
                break
            source = int(np.random.choice(nz))

        # Move a random fraction (up to 50 %) of the source mass
        delta = a[source] * 0.5 * np.random.rand()
        if delta <= 0.0:
            continue

        # Incremental update of the full convolution
        b_new = b.copy()
        # linear part (2·δ·a) – safe slicing because source/target ≤ n‑1
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            # accept
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # If we get stuck for a long stretch, perturb around the best solution
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.003, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")
        it += 1

    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative step heights that (hopefully) minimises
    `evaluate_sequence`.  The routine respects the 1000 s wall‑time limit
    and prints intermediate progress.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size – a little larger than the previous 8000 gives more
    # degrees of freedom while still keeping FFTs cheap.
    # --------------------------------------------------------------
    n = 10000

    # --------------------------------------------------------------
    # Initialise a – reuse the known construction if it exists,
    # otherwise start from a random simplex point.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # pylint: disable=undefined-variable
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)                # enforce Σa = 1, a≥0
    best_a = a.copy()
    best_val = float('inf')

    # --------------------------------------------------------------
    # Split the remaining budget:
    #   60 % → smooth‑max Adam phase
    #   40 % → greedy hill‑climbing
    # --------------------------------------------------------------
    total_time = deadline - start_time
    t_smooth_end = start_time + 0.60 * total_time
    t_hill_end = deadline

    # ----------------------------
    # Phase 2 – smooth‑max descent
    # ----------------------------
    best_a, best_val = _smooth_max_phase(
        a, n, t_smooth_end, best_a, best_val
    )
    a = best_a.copy()   # continue from the best point found

    # ----------------------------
    # Phase 3 – greedy hill‑climb
    # ----------------------------
    best_a = _greedy_hillclimb(
        a, n, t_hill_end, best_a, best_val
    )

    # ------------------------------------------------------------------
    # Final evaluation (the wrapper may clamp values)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:                 # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
