"""
Optimising a step‑function to minimise ``evaluate_sequence``.

Idea
----
The objective is  f(a)=2·n·max(conv(a,a)) / (∑a)² .
Because the expression is homogeneous we can restrict ourselves to the
probability simplex (∑a=1, a_i≥0).  The quantity
    max(conv(a,a)) = max_k  Σ_i a_i·a_{k-i}
is the ℓ∞‑norm of the self‑convolution of a probability mass function.
The exact optimum is the uniform distribution, which yields a value of 2,
but the true lower bound is 2·n/(2n−1)≈1.0.  To approach this bound we
 minimise a smooth approximation of the max‑operator:

    Φ_β(a)= (1/β)·log Σ_k exp(β·(a∗a)_k)   (β>0).

For β→∞, Φ_β approaches the exact max.  Its gradient is simple:
    ∂Φ/∂a_j = 2· Σ_k w_k·a_{k‑j},
where w_k = exp(β·(a∗a)_k) / Σ_t exp(β·(a∗a)_t)   (a probability over lags).

We perform projected gradient descent on the simplex:
*   each iteration computes the full convolution (FFT based),
*   builds the soft‑max weights w,
*   obtains the gradient via another FFT convolution,
*   takes a diminishing step and projects back onto the simplex
    (Duchi et al. 2008).
The temperature β is slowly increased, turning the smooth surrogate
into the true max.  A final short guided hill‑climb (mass‑transfer
moves) refines the best solution.

The whole routine respects the 1000 s budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)                 # eliminate tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (FFT based)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Small stochastic hill‑climb that moves a tiny fraction of mass away
    from the most offending lag.  The routine runs until the time budget
    expires and returns the best sequence found.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.03            # how much mass to move each try

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # indices (i,j) that actually contribute to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # choose two distinct low‑mass destinations
        low = np.where(cur_a < cur_a.mean())[0]
        if low.size >= 2:
            p, q = rng.choice(low, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        # avoid recreating the same lag
        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # feasible mass transfer
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max (a diagonal term)
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] improved = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val

    """
    Returns a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – 4000 gives a good trade‑off between quality and speed
    n = 4000

    # ------------------------------------------------------------------
    # Initial distribution – uniform on the simplex
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=float)
    cur_val = evaluate_sequence(a.tolist())
    best_a = a.copy()
    best_val = cur_val
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Parameters for the smooth‑max optimisation
    # ------------------------------------------------------------------
    beta = 5.0                # start temperature (smoothness)
    beta_inc = 1.10           # increase factor every few iterations
    beta_max = 200.0
    step0 = 0.5               # initial step size (will be damped)
    it = 0
    no_improve = 0

    while time.time() < deadline - 5.0:
        it += 1
        # full convolution of the current a
        conv = _full_convolution(a)
        max_conv = conv.max()
        # exact objective (sum(a)=1 after projection)
        cur_obj = 2.0 * n * max_conv
        if cur_obj < best_val - 1e-12:
            best_val = cur_obj
            best_a = a.copy()
            print(f"[iter {it}] new best = {best_val:.6f}")

        # ----- smooth‑max (log‑sum‑exp) -----
        # w_k = softmax over β·conv_k
        scaled = beta * conv
        # subtract max for numerical stability
        scaled -= np.max(scaled)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            # should not happen, but fall back to uniform weights
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # gradient g_i = 2 Σ_k w_k a_{k-i}
        a_rev = a[::-1]                     # reversed a
        conv_w = _conv_fft(w, a_rev)        # length = (2n‑1)+n‑1 = 3n‑2
        # extract the central n entries (indices n‑1 … 2n‑2)

        # diminishing step size

        cand = a - step * g
        cand = _simplex_projection(cand)

        # evaluate candidate (fast estimate using the already computed conv)
        cand_conv = _full_convolution(cand)
        cand_obj = 2.0 * n * cand_conv.max()

        # keep the best global solution
        if cand_obj < best_val - 1e-12:
            best_val = cand_obj
            best_a = cand.copy()
            print(f"[iter {it}] improved candidate = {best_val:.6f}")

        # accept if we made progress, otherwise shrink step
        if cand_obj < cur_obj - 1e-14:
            a = cand
            cur_val = cand_obj
            no_improve = 0
        else:
            no_improve += 1
            step0 *= 0.7
            if no_improve >= 300:
                # escape stagnation: small random perturbation around the best
                perturb = rng.normal(scale=0.005, size=n)
                a = _simplex_projection(best_a + perturb)
                cur_val = 2.0 * n * _full_convolution(a).max()
                no_improve = 0
                step0 = 0.5

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")

    return best_a.tolist()
