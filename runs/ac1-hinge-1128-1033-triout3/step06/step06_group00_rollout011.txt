def _full_convolution_fft(a,b):
    size = len(a) + len(b) - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, nfft)
    Fb = np.fft.rfft(b, nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, nfft)[:size]
    return conv

def full_convolution_fft(a,b):
    size = len(a) + len(b) - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, nfft)
    Fb = np.fft.rfft(b, nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, nfft)[:size]
    return conv

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 5.0
    rng = np.random.default_rng(seed)

    n = 10000   # may adjust

    # Warm start shapes
    def initial_shape():
        shapes = []
        x = np.arange(n, dtype=np.float64)

        # arcsine-like shape
        arcsine = 1.0 / np.sqrt((x + 0.5)*(n - 0.5 - x))
        shapes.append(arcsine)

        # exponential decays
        lam = rng.uniform(1e-4, 5e-2)
        shapes.append(np.exp(-lam * x))
        lam2 = rng.uniform(1e-4, 5e-2)
        shapes.append(np.exp(-lam2 * (n-1 - x)))

        # power law
        p = rng.uniform(0.2, 2.5)
        shapes.append((x + 1) ** (-p))

        # half-sine
        shapes.append(np.sin(np.pi * (x + 0.5) / n))

        # Gaussian
        mu = n/2
        sigma = rng.uniform(n/10, n/2)
        shapes.append(np.exp(-0.5 * ((x - mu)/sigma) ** 2))

        # random convex combo of these shapes
        weights = rng.random(len(shapes))
        weights /= weights.sum()
        a = sum(w * s for w, s in zip(weights, shapes))
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        return a

    a = None
    # try using existing height_sequence_1 if present
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if base.shape[0] != n:
                if base.shape[0] > n:
                    base = base[:n]
                else:
                    base = np.pad(base, (0, n - base.shape[0]))
            a = np.maximum(0.0, base)
            if a.sum() > 0:
                a = a / a.sum()
        except Exception:
            a = None

    if a is None:
        a = initial_shape()

    # compute initial best
    conv = _full_convolution_fft(a, a)
    best_val = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"[init] n={n} score={best_val:.12f}")

    # Phase 1: EG with softmax
    phase1_end = start_time + 0.45 * budget_s
    eta0 = 1.0   # starting step size
    eps = max(conv.max() * 0.2, 1e-8)
    eps_decay = 0.9995

    iter1 = 0
    stagn = 0
    while time.time() < phase1_end:
        iter1 += 1

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[phase1 iter {iter1:6d}] new_best={best_val:.12f} eps={eps:.2e}")
        else:
            stagn += 1

        # softmax weights
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient
        rev_a = a[::-1]
        grad = 2.0 * _full_convolution_fft(rev_a, w)[n-1:n-1+n]

        eta = eta0 / np.sqrt(iter1)
        mult = np.exp(np.clip(-eta * grad, -50, 50))
        a = a * mult
        a = a / a.sum()

        eps = max(1e-10, eps * eps_decay)

        if stagn > 2000:
            # jitter
            jitter = rng.normal(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            stagn = 0
            print(f"[phase1 jitter] iter {iter1}")

    # Phase 2: EG with exact active set
    phase2_end = start_time + 0.85 * budget_s
    eta0_2 = 0.5
    iter2 = 0
    stagn = 0
    while time.time() < phase2_end:
        iter2 += 1

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[phase2 iter {iter2:6d}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # active set
        tol = 1e-14
        active = np.flatnonzero(conv >= max_conv - tol)
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size  # average weight
        else:
            w[np.argmax(conv)] = 1.0

        rev_a = a[::-1]
        grad = 2.0 * _full_convolution_fft(rev_a, w)[n-1:n-1+n]

        eta = eta0_2 / np.sqrt(iter2)
        mult = np.exp(np.clip(-eta * grad, -50, 50))
        a = a * mult
        a = a / a.sum()

        if stagn > 3000:
            jitter = rng.normal(scale=0.002, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            stagn = 0
            print(f"[phase2 jitter] iter {iter2}")

    # Phase 3: Simulated annealing
    cur = a.copy()
    cur_score = 2.0 * n * _full_convolution_fft(cur, cur).max()
    if cur_score < best_val:
        best_val = cur_score
        best_seq = cur.copy()
        print(f"[SA init] cur_score={cur_score:.12f}")

    T0 = 1e-3
    Tmin = 1e-7
    it_sa = 0
    while time.time() < deadline:
        it_sa += 1
        T = max(Tmin, T0 * (1.0 - it_sa / 300_000))

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if cur[i] <= 0.0:
            continue
        max_transfer = min(cur[i], 0.01)
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # ensure non-neg
        new = np.clip(new, 0.0, None)
        # normalize sum (should stay 1 but due to rounding may drift)
        new_sum = new.sum()
        if new_sum == 0:
            continue
        new /= new_sum

        new_score = 2.0 * n * _full_convolution_fft(new, new).max()
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            cur = new
            cur_score = new_score
            if cur_score < best_val - 1e-12:
                best_val = cur_score
                best_seq = cur.copy()
                print(f"[SA iter {it_sa:6d}] new_best={best_val:.12f}")

        if it_sa % 50000 == 0:
            # occasional global jitter
            cur = cur + rng.normal(scale=0.001, size=n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_score = 2.0 * n * _full_convolution_fft(cur, cur).max()
            print(f"[SA jitter] iter {it_sa}")

    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception as e:
        print("[result] evaluate_sequence not available:", e)

    return final_seq

"""
Propose a step‑function that (approximately) minimizes

      2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three stages under the given time budget:

1. **Warm‑start** – a mixture of several analytic shapes (exponential,
   power‑law, half‑sine, Gaussian, triangular) together with a
   deterministic “arcsine‑like’’ shape is generated; the best of these
   is taken as the initial probability vector (sum = 1).

2. **Exponentiated‑gradient descent** – we minimise the piecewise
   linear objective on the simplex using an exponentiated‑gradient
   update.  In the first part a smooth‑max (soft‑max) is used; later the
   exact active set of the maximal convolution entries is employed.
   The step size decays like 1/√t and occasional random jitter
   prevents stagnation.

3. **Greedy mass‑transfer + Simulated annealing** – after the
   gradient phase we repeatedly move a small amount of probability mass
   from the index with the largest sub‑gradient to the one with the
   smallest (greedy improvement).  When this phase plateaus a final
   simulated‑annealing fine‑tuning with a Metropolis rule is performed.

All vectors are kept on the probability simplex, guaranteeing a
well‑defined evaluation score (the sum of the sequence is 1, so the
objective equals 2·n·max (conv)).  The best sequence found before the
time limit is returned as a plain list of floats clipped to the
interval [0, 1000].
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast and memory‑efficient)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Objective for a vector that already satisfies sum(a)=1
# ----------------------------------------------------------------------
def _objective_on_simplex(a: np.ndarray) -> float:
    """2 * n * max(conv(a,a))   (assumes sum(a)==1)."""
    n = a.size
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution (exact version)
# ----------------------------------------------------------------------
def _max_conv_subgradient(a: np.ndarray, eps: float = 1e-14) -> tuple:
    """
    Return (conv, max_val, grad) where
    * conv  – full convolution a*a,
    * max_val – its maximum,
    * grad  – sub‑gradient of max_val w.r.t. a (size = len(a)).
    """
    n = a.size
    conv = _full_convolution_fft(a, a)
    max_val = conv.max()
    # indices where the maximum is attained (active set)
    active = np.flatnonzero(conv >= max_val - eps)
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0    # each active shift gets unit weight
    else:                  # fallback, should never happen
        w[np.argmax(conv)] = 1.0
    rev_a = a[::-1]
    # convolution rev_a * w, then take the middle n entries
    grad = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]
    return conv, max_val, grad


# ----------------------------------------------------------------------
# Soft‑max (smooth max) weights – used in the first EG phase
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, max_val: float, eps: float) -> np.ndarray:
    """Return a probability vector proportional to exp((conv‑max)/eps)."""
    w = np.exp((conv - max_val) / eps)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        return np.full_like(conv, 1.0 / conv.size)
    return w / w_sum


# ----------------------------------------------------------------------
# Exponentiated‑gradient update (keeps positivity automatically)
# ----------------------------------------------------------------------
def _eg_update(a: np.ndarray, grad: np.ndarray, step: float) -> np.ndarray:
    """
    a <-- a * exp(-step * grad)  , then re‑normalise to sum = 1.
    The exponent is clipped to avoid overflow/underflow.
    """
    mult = np.exp(np.clip(-step * grad, -50.0, 50.0))
    a_new = a * mult
    s = a_new.sum()
    if s == 0.0:
        # numerical disaster – fall back to the old vector
        return a.copy()
    return a_new / s


# ----------------------------------------------------------------------
# Greedy mass‑transfer move (high‑gradient → low‑gradient)
# ----------------------------------------------------------------------
def _greedy_transfer(a: np.ndarray, grad: np.ndarray, rng: np.random.Generator,
                     max_frac: float = 0.25) -> np.ndarray:
    """
    Move a small amount of mass from the index with the largest
    gradient to the one with the smallest.  The transferred amount is
    at most ``max_frac`` of the source mass.
    """
    i_high = int(np.argmax(grad))
    i_low = int(np.argmin(grad))
    if a[i_high] <= 0.0:
        return a.copy()
    delta = min(a[i_high] * max_frac, rng.random() * a[i_high])
    if delta <= 0.0:
        return a.copy()
    a_new = a.copy()
    a_new[i_high] -= delta
    a_new[i_low] += delta
    # Numerical drift – re‑normalise (preserves total mass)
    a_new = np.clip(a_new, 0.0, None)
    a_new /= a_new.sum()
    return a_new

# Main entry point required by the grader.

    deadline = start_time + budget_s - 5.0

    # Problem size – a size that fits well in memory and gives enough
    # degrees of freedom to approach the target bound.
    # ------------------------------------------------------------------
    n = 10000                     # you may tune this if time permits

    # ------------------------------------------------------------------
    # Warm start – generate several classic shapes, mix them, and keep the
    # best (according to the objective on the probability simplex).
    # ------------------------------------------------------------------
    a = None
    if 'height_sequence_1' in globals():
        try:

            if a.sum() > 0.0:

            else:

        except Exception:

    if a is None:
        # build a pool of deterministic shapes
        x = np.arange(n, dtype=np.float64)

        shapes = []

        mu = (n - 1) / 2.0

        # simple triangle
        shapes.append(np.minimum(x, n - 1 - x) + 1.0)

        eps = 1e-12

        a = sum(weight * shape for weight, shape in zip(w, shapes))

        a = a / a.sum()

    # ------------------------------------------------------------------
    # Keep the best solution seen so far.

    # ------------------------------------------------------------------
