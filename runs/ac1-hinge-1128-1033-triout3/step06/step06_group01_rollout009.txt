def proj_simplex(v):
    """Project onto probability simplex (nonnegative, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _subgradient_descent(a, max_iter, deadline):
    n = len(a)
    best_a = a.copy()
    conv = _fft_convolve_self(a)
    maxb = conv.max()
    best_val = 2 * n * maxb
    cur_a = a.copy()
    step0 = 0.5
    t = 0
    while time.time() < deadline and t < max_iter:
        conv = _fft_convolve_self(cur_a)
        maxb = conv.max()
        active = np.where(conv >= maxb - 1e-12)[0]
        # compute gradient
        g = np.zeros_like(cur_a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1)
            j_idx = k - i_idx
            # contributions
            g[i_idx] += 2 * cur_a[j_idx]
            # correct diagonal double-count
            if k % 2 == 0:
                i_mid = k // 2
                if i_mid >= 0 and i_mid < n:
                    g[i_mid] -= cur_a[i_mid]
        # step size
        step = step0 / np.sqrt(t + 1)
        new_a = cur_a - step * g
        # project onto simplex
        new_a = proj_simplex(new_a)
        # evaluate if improvement
        conv_new = _fft_convolve_self(new_a)
        maxb_new = conv_new.max()
        val_new = 2 * n * maxb_new
        if val_new < best_val:
            best_val = val_new
            best_a = new_a.copy()
            print(f"[subgrad] iter {t}, new best = {best_val:.6f}")
        cur_a = new_a
        t += 1
    return best_a, best_val

"""
propose_candidate
-----------------
A multi‑phase optimizer for the step‑function evaluation problem.

The algorithm runs under a 1000 s budget and consists of the following
stages:

1. **Smooth surrogate optimisation** – several L‑BFGS restarts on a
   log‑sum‑exp (soft‑max) smoothing of the max‑convolution.  The smoothing
   parameter ε is decreased geometrically, letting the optimizer get
   close to the true nondifferentiable objective.

2. **Projected sub‑gradient descent** – direct optimisation of the true
   objective `max(a*a)` via a sub‑gradient that considers only the indices
   where the convolution attains its maximum.  After each step the vector is
   projected onto the probability simplex (non‑negative, ℓ₁‑norm = 1).

3. **Anti‑symmetric re‑ordering** – a deterministic permutation that pairs
   large values with small ones, which is known to reduce the peak of the
   self‑convolution.

4. **Refined simulated annealing** – a permutation‑only meta‑heuristic
   with swaps and occasional segment reversals.  The temperature schedule
   is very gentle, allowing the search to explore many permutations.

5. **Hill‑climbing polish** – fast mass‑transfer and swap moves that exploit
   the exact FFT‑based evaluator.

If time permits a final short L‑BFGS run with an extremely small ε
fine‑tunes the real values.

The whole pipeline obeys the time limit, always returns a feasible
non‑negative list of floats and prints progress information.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate (avoids closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence

# ----------------------------------------------------------------------
# Basic FFT utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)      # clamp tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast version of the official `evaluate_sequence`.
    The array a is assumed to be non‑negative and already respects the
    clipping limits (0 ≤ a_i ≤ 1000).  Returns the same float value.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) objective used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns the smoothed objective L_ε and its gradient w.r.t. the
    soft‑max parameters w.
    """
    # ---- soft‑max (ensures positivity and Σa = 1) ----
    w_shift = w - np.max(w)                # numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # ---- self‑convolution of a ----
    conv = _fft_convolve_self(a)           # length 2·n‑1
    max_conv = conv.max()

    # ---- smooth approximation of the max (log‑sum‑exp) ----
    shifted = (conv - max_conv) / eps
    log_sum = np.log(np.sum(np.exp(shifted)))
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a (closed form) ----
    weights = np.exp(shifted)
    p = weights / weights.sum()            # probability distribution over k

    # convolution a * p_rev  (helper defined below)
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)

    # middle n entries of the full convolution correspond to indices
    # [n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]           # reverse to match index order

    # ---- chain rule through the soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by `scipy.optimize.minimize` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# helper convolution used inside the gradient (FFT based)
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# bind the helper so the gradient routine can call it
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Simple re‑ordering heuristic (large ↔ small pairing)
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Returns a deterministic permutation where the largest values are placed
    at the leftmost positions and the smallest at the rightmost ones.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left
        new_a[n - 1 - i] = sorted_vals[i]          # small → right

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# Project a vector onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex.
    Reference: Duchi et al., “Efficient Projections onto the L1-Ball for
    Learning in High Dimensions”, ICML 2008.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         max_iter: int,
                         deadline: float) -> (np.ndarray, float):
    """
    Projected sub‑gradient method for   f(a) = max(a * a)   (ℓ₁‑norm = 1).
    Returns the best vector and its objective value encountered.
    """
    n = a.shape[0]
    cur_a = a.copy()
    # initial objective (sum = 1 ⇒ objective = 2·n·max_b)
    conv = _fft_convolve_self(cur_a)
    best_val = 2.0 * n * conv.max()
    best_a = cur_a.copy()
    step0 = 0.5
    t = 0
    while time.time() < deadline and t < max_iter:
        conv = _fft_convolve_self(cur_a)
        maxb = conv.max()
        active = np.where(conv >= maxb - 1e-12)[0]

        # gradient of max(b) w.r.t. a (sub‑gradient)
        g = np.zeros_like(cur_a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_idx = np.arange(i_start, i_end + 1)
            j_idx = k - i_idx
            g[i_idx] += 2.0 * cur_a[j_idx]          # double count
            if k % 2 == 0:                          # correct the diagonal term
                i_mid = k // 2
                if 0 <= i_mid < n:
                    g[i_mid] -= cur_a[i_mid]

        # step size (diminishing)
        step = step0 / np.sqrt(t + 1.0)
        new_a = cur_a - step * g
        new_a = _proj_simplex(new_a)

        # new objective
        conv_new = _fft_convolve_self(new_a)
        val_new = 2.0 * n * conv_new.max()
        if val_new < best_val:
            best_val = val_new
            best_a = new_a.copy()
            print(f"[subgrad] iter {t:5d} -> new best {best_val:.6f}")

        cur_a = new_a
        t += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing on the permutation of a fixed multiset of heights
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """
    Classic SA that only permutes the entries of a.
    Supports swaps and occasional segment reversals.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val
    cur_seq = a.copy()
    cur_score = cur_val

    # generous initial temperature
    T = max(cur_score * 0.15, 1e-6)
    decay = 0.99996

    it = 0
    while time.time() < deadline:
        it += 1
        r = rng.random()
        # ----- move type -----
        if r < 0.85:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
            undone = ("swap", i, j)
        else:                            # block reversal
            i = rng.integers(0, n - 1)
            j = rng.integers(i + 1, n)    # j is exclusive
            cur_seq[i:j] = cur_seq[i:j][::-1]
            undone = ("rev", i, j)

        new_score = _eval_fast(cur_seq)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / T):
            # accept
            cur_score = new_score
            if cur_score < best_val:
                best_val = cur_score
                best_seq = cur_seq.copy()
                print(f"[SA] iter {it:6d} new best = {best_val:.6f}")
        else:
            # reject → revert
            typ, i, j = undone
            if typ == "swap":
                cur_seq[i], cur_seq[j] = cur_seq[j], cur_seq[i]
            else:  # reversal
                cur_seq[i:j] = cur_seq[i:j][::-1]

        T *= decay
        if T < 1e-12:
            T = max(cur_score * 0.05, 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (mass transfers + swaps)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float) -> (np.ndarray, float):
    """
    Very fast local search: occasional swaps and small mass transfers
    between random indices.  The ℓ₁‑norm is preserved by the transfer.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.20:                # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                   # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.04 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val + 1e-12 < cur_val:               # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

def _final_l_bfgs(a: np.ndarray,

    """
    One short L‑BFGS minimisation of the smoothed surrogate with

    """
    n = a.shape[0]
    eps = 1e-13
    global _GLOBAL_EPS
    _GLOBAL_EPS = eps

    res = minimize(_smooth_obj_and_grad_wrapper,

    a_opt = np.exp(w_shift)
    a_opt /= a_opt.sum()
    val_opt = _eval_fast(a_opt)

        w = np.random.randn(n) * 2.0                 # random logits

        for eps in eps_schedule:
            if time.time() >= deadline - 240.0:
                break
            _GLOBAL_EPS = eps

            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 400,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # evaluate current iterate
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # one cheap anti‑symmetric re‑ordering attempt
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # -------------------- Phase 2: projected sub‑gradient descent -----------------
    if time.time() < deadline - 150.0:
        print("[subgrad] starting projected sub‑gradient descent")
        sub_deadline = deadline - 150.0
        best_a, best_val = _subgradient_descent(best_a,
                                                max_iter=5000,
                                                deadline=sub_deadline)
        print(f"[subgrad] after descent best = {best_val:.6f}")

        # re‑order once more (often helps after a value‑only update)
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after subgrad] improved to {best_val:.6f}")

    # -------------------- Phase 3: simulated annealing ---------------------------
    if time.time() < deadline - 50.0:
        print("[SA] starting refined simulated annealing")
        sa_deadline = deadline - 20.0
        best_a, best_val = _simulated_annealing(best_a,
                                                best_val,
                                                sa_deadline)

    # -------------------- Phase 4: final hill‑climbing -------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing phase")
        best_a, best_val = _refine_hill_climb(best_a,
                                              best_val,
                                              deadline)

    # -------------------- Optional short L‑BFGS fine‑tune -----------------------
    if time.time() < deadline - 5.0:
        print("[final‑L‑BFGS] short fine‑tune")
        best_a, best_val = _final_l_bfgs(best_a, deadline)

    # -------------------- report using the official evaluator ------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:                # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return plain Python list
    return best_a.tolist()
