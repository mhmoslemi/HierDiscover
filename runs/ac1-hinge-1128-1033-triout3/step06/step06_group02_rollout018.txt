def mirror_descent_step(a, g, alpha):
    a_new = a * np.exp(-alpha * g)

"""
Propose a step function that minimizes ``evaluate_sequence``.

Algorithm outline
-----------------
1. **Initialisation** – load a previously discovered sequence
   (``height_sequence_1``) if it exists, otherwise start from the uniform
   distribution.  The vector is projected onto the simplex
   (non‑negative, sum = 1).

2. **Smooth surrogate optimisation** – minimise a smoothed version of the
   objective using L‑BFGS on the log‑soft‑max parametrisation.
   The surrogate replaces the non‑smooth max by a log‑sum‑exp with a
   decreasing smoothing parameter ``eps``.

3. **Mirror‑descent on the true max‑convolution** – repeatedly
   compute the full self‑convolution, identify *all* (within a tiny
   tolerance) lag positions attaining the maximal value, form the exact
   sub‑gradient of the max, and perform an exponential mirror update
   ``a ← a·exp(−α·g)`` followed by projection onto the simplex.
   The step size ``α`` decays geometrically.

4. **Linear‑programming refinement** – with the current vector ``f``,
   solve a linear programme that maximises ``sum(g)`` under the
   constraints ``(f*g)[k] ≤ M`` for every *active* lag ``k`` where
   ``M = max(f*f)``.  The optimiser returns a vector ``g`` that can be
   normalised to unit sum; mixing ``f`` and the normalised ``g`` (or
   using ``g`` directly) strictly reduces the maximal convolution
   value.

5. **Local hill‑climbing** – random swaps and small mass transfers
   (always re‑projected onto the simplex) are performed until the time
   budget expires.  Only strict improvements are kept.

The whole procedure respects the 1000 s budget, prints progress
information and finally returns the best sequence found as a plain
``list[float]``.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (updated for each L‑BFGS run)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter
_GLOBAL_N = None     # sequence length


# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution ``a*a`` using an FFT of size next power of two."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    np.maximum(conv, 0, out=conv)                # clip tiny negatives
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Heuristic: pair the largest values with the smallest ones
    (places them at opposite ends)."""
    n = a.shape[0]
    s = np.sort(a)               # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[n - 1 - i]       # large → left side
        b[n - 1 - i] = s[i]       # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective ``L_ε`` and its gradient w.r.t. the log‑soft‑max
    parameters ``w``.
        a_i = exp(w_i) / Σ_j exp(w_j)   (≥0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k−max)/ε)
    """
    # soft‑max (numerically stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)          # length 2·n−1
    max_conv = conv.max()

    # smooth approximation of the max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()          # distribution over lag indices

    # ∂max/∂a = 4·n * (conv of a with reversed p) on the central n entries
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]
    seg = conv_ap[n - 1: 2 * n - 1]               # central n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _max_conv_gradient(a: np.ndarray, tol_rel: float = 1e-12):
    """
    Compute the (sub‑)gradient of ``max(conv)``.
    Returns the full convolution, the maximal value, the gradient vector,
    and the list of *active* lag positions (where the max is attained
    within the relative tolerance).
    """
    conv = _full_convolution(a)
    max_val = conv.max()
    # active set – all lags whose value is within tol of the maximum
    if max_val == 0:
        active = np.arange(conv.shape[0])
    else:
        tol = max(tol_rel * max_val, 1e-15)
        active = np.where(conv >= max_val - tol)[0]

    n = a.shape[0]
    g = np.zeros_like(a)

    # g_i = Σ_{k∈active} a_{k‑i}   (zero when index out of range)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        g[i_range] += a[j_range]

    if len(active) > 1:               # average when several ties exist
        g /= len(active)
    return conv, max_val, g, active


def _mirror_descent_step(a: np.ndarray, g: np.ndarray, alpha: float) -> np.ndarray:
    """Perform one exponential mirror‑descent update and re‑project onto the simplex."""
    a_new = a * np.exp(-alpha * g)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new /= s
    return a_new


def _solve_lp_max_sum_under_bound(f: np.ndarray, M: float, rel_tol: float = 1e-6):
    """
    Linear programme:
        maximise   Σ_j g_j
        subject to  g_j ≥ 0
                    (f * g)[k] ≤ M   for every *active* lag k where
                    f*f attains its maximum (within ``rel_tol``).

    Returns the optimal ``g`` as a dense ``np.ndarray`` or ``None`` on failure.
    """
    n = f.shape[0]
    conv_f = _full_convolution(f)
    active_k = np.where(conv_f >= M * (1.0 - rel_tol))[0]
    if active_k.size == 0:
        return None

    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active_k):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = k - i_vals                     # indices of g
        fi_vals = f[i_vals]
        rows.extend([row_idx] * len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(fi_vals.tolist())

    A = sp.csr_matrix((data, (rows, cols)), shape=(active_k.size, n))
    c = -np.ones(n)                     # maximise sum(g)  → minimise –sum(g)
    bounds = [(0, None)] * n
    b = np.full(active_k.size, M, dtype=np.float64)

    # use HiGHS (fast, stable)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30.0})
    if res.success:
        return res.x
    return None


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Simple hill‑climbing: random swaps (20 %) and mass transfers (80 %).
    Only strict improvements are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * (0.05 + 0.10 * rng.random())   # up to 15 % of source mass
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val
