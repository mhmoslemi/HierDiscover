def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # decide n (maybe 800 or 1200). Let's choose 800 for speed.
    n = 800

    # Phase 0: find good initial shape by gamma sweep similar to earlier but with more resolution
    best_a, best_val = None, np.inf
    for gamma in np.linspace(0.0, 6.0, 61):
        a = compute shape using denominator power -gamma; also allow offset param eps.
        a /= a.sum()
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val, best_a = val, a.copy()
    # Keep best shape.

    # iterate:
    t = 0.25  # mixing factor
    iteration = 0
    while time.time() < deadline:
        # compute convolution and M
        a = best_a
        conv = np.convolve(a, a)  # full convolution
        M = conv.max()
        # compute A
        # Build Toeplitz matrix to exclude unnecessary parts
        # Use vectorized approach
        idx = np.arange(2*n-1)[:,None] - np.arange(n)[None,:]
        mask = (idx >= 0) & (idx < n)
        A = np.where(mask, a[np.newaxis,:], 0.0) // shape (2n-1, n)
        # Solve LP
        import cvxpy as cp
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)
        # solve with ECOS
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
        except Exception as e:
            print("LP solve error:", e)
            break
        if g.value is None:
            print("LP solver failed")
            break
        g_sol = np.maximum(g.value, 0.0)  # ensure nonnegative
        sum_g = g_sol.sum()
        if sum_g == 0:
            # No improvement, break
            print("LP gave zero sum")
            break
        g_norm = g_sol / sum_g
        # mixing update
        a_new = (1 - t) * a + t * g_norm
        a_new /= a_new.sum()  # ensure sum=1
        # compute new val
        new_val = evaluate_sequence(a_new.tolist())
        # Accept if improvement else maybe reduce t or break.
        if new_val < best_val - 1e-12:
            best_a, best_val = a_new, new_val
            print(f"[Iter {iteration}] improved to {best_val:.6f}")
        else:
            # reduce t maybe
            t *= 0.95
            # optionally revert to a (no change)
        iteration +=1
        # stop early if best_val <= target
        if best_val <= 1.503:
            break
    # final return best_a.tolist()
    return best_a.tolist()

"""
Propose a step‑function with a very low value of
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

0️⃣  Fast U‑shape sweep to obtain a decent deterministic seed.
1️⃣  Iterative linear‑programming refinement:
    – For the current distribution f we compute M = max(conv(f,f)).
    – Solve the LP   max Σ g  subject to   f * g ≤ M  (component‑wise)
      and g ≥ 0.  The LP is written with an explicit convolution matrix
      and solved with SciPy’s HiGHS LP solver.
    – Normalise the solution g, blend it with f (mixing factor t) and
      repeat while time permits.  Each blend never increases the
      objective, and in practice drives M down.
2️⃣  Targeted hill‑climbing on the exact objective (same idea as the
    reference code but with a simpler update).  Small amounts of mass
    are moved from indices that contribute most to the current maximum
    convolution entry towards indices that contribute little; moves are
    accepted only if the exact objective improves.

The whole procedure respects the 1000 s time budget and returns a
non‑negative list whose sum is 1 (scale‑invariant for the objective).
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective value for a (sum(a) == 1)."""
    n = a.shape[0]
    b = np.convolve(a, a)          # full linear convolution
    return float(2.0 * n * b.max())

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix A such that (A @ g)[k] = (a * g)[k].
    For each column i we place a[i] on rows i … i+n‑1.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a[i]
    return A

def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ g   s.t.   f * g ≤ M , g ≥ 0.
    Returns the normalised solution g_norm (sum = 1).
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)                # shape (2n‑1, n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Maximise Σ g  ⇔ minimise -Σ g
    c = -np.ones(n, dtype=np.float64)

    # HiGHS is fast for dense LPs; give it a tiny time budget per solve.
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=(0, None),
        method="highs",
        options={"presolve": True, "time_limit": 5.0},
    )
    if not res.success or res.x is None:
        raise RuntimeError("LP failed: " + (res.message if hasattr(res, "message") else ""))
    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s <= 0:
        raise RuntimeError("LP returned zero sum")
    return g / s

def _hill_climb(a: np.ndarray,
                deadline: float,
                target_val: float = 1.503,
                max_frac_start: float = 0.60,
                max_frac_end: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Small‑scale local search.
    Moves a random amount of mass from a ‘source’ index (which
    contributes heavily to the current maximal convolution entry) to a
    ‘target’ index (which contributes little).  The exact objective is
    recomputed each trial and the move is kept only if it strictly
    improves the score.
    """
    n = a.shape[0]
    cur_a = a.copy()
    b = np.convolve(cur_a, cur_a)
    cur_val = 2.0 * n * b.max()
    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    eps = 1e-12

    while time.time() < deadline:
        # progressive bound on how much mass we may shift
        progress = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - time.time()))
        # the above expression is a dummy – we just linearly decay max_frac over the run
        remaining = max(deadline - time.time(), 0.0)
        total_runtime = deadline - (deadline - remaining)
        frac = max_frac_start * (1.0 - total_runtime / (deadline - (deadline - remaining))) + max_frac_end * (total_runtime / (deadline - (deadline - remaining)))
        max_frac = max_frac_start * (1.0 - (time.time() - (deadline - remaining)) / (deadline - (deadline - remaining))) + max_frac_end * ((time.time() - (deadline - remaining)) / (deadline - (deadline - remaining)))

        # locate the index k where the convolution attains its maximum
        k_max = int(np.argmax(b))

        # contribution of each a[i] to that entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(cur_a)
        contrib[mask] = cur_a[mask] * cur_a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – proportional to contribution
        if total_contrib > 0:
            src = np.random.choice(n, p=contrib / total_contrib)
        else:
            src = int(np.argmax(cur_a))

        if cur_a[src] <= eps:
            continue

        # target index – biased towards small contribution
        target_w = np.zeros_like(cur_a)
        target_w[mask] = 1.0 / (contrib[mask] + eps)
        target_w[src] = 0.0
        wsum = target_w.sum()
        if wsum > 0:
            tgt = np.random.choice(n, p=target_w / wsum)
        else:
            # fall back to uniformly random different index
            tgt = np.random.randint(n - 1)
            if tgt >= src:
                tgt += 1

        # move a random fraction of the source mass
        delta = cur_a[src] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incrementally update the convolution (O(n))
        b_new = b.copy()
        # contributions of moved mass to all convolution entries
        b_new[tgt:tgt + n] += 2.0 * delta * cur_a
        b_new[src:src + n] -= 2.0 * delta * cur_a
        # quadratic self‑terms
        b_new[2 * tgt] += delta * delta
        b_new[2 * src] += delta * delta
        b_new[src + tgt] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-14:
            # accept move
            cur_a[src] -= delta
            cur_a[tgt] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < target_val:
                break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb and restart from the best known point
            cur_a = _project_simplex(cur_a + np.random.normal(scale=0.02, size=n))
            b = np.convolve(cur_a, cur_a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return cur_a, cur_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list (sum = 1) that minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a moderate size – large enough for a good asymptotic bound,
    # small enough for fast LP solves.
    # ------------------------------------------------------------------
    n = 1000

    # ------------------------------------------------------------------
    # Phase 0 – parametric sweep of a simple U‑shape family.
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf
    gamma_vals = np.linspace(0.0, 6.0, 61)           # 61 candidates
    x = np.arange(n, dtype=np.float64)
    eps = 1e-12
    for gamma in gamma_vals:
        denom = (x + 0.5) * (n - x - 0.5)
        a = np.power(denom + eps, -gamma)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        val = _objective(a)                       # exact internal objective
        if val < best_val:
            best_val, best_a = val, a.copy()
    print(f"[Phase0] best value after sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP based improvement.
    # ------------------------------------------------------------------
    # Mixing factor (how much of the LP solution we incorporate each round)
    t = 0.35
    it = 0
    while time.time() < deadline - 10.0:           # leave some time for phase 2
        a = best_a
        # current max convolution value
        conv = np.convolve(a, a)
        M = conv.max()

        # solve LP and obtain a normalised g
        try:
            g_norm = _lp_step(a, M)
        except Exception as e:
            print("[LP] solve failed, breaking:", e)
            break

        # blend current a with g_norm
        a_new = (1.0 - t) * a + t * g_norm
        # projection is not needed (both sum to 1) but we keep it for safety
        a_new = _project_simplex(a_new)

        new_val = _objective(a_new)
        if new_val < best_val - 1e-12:
            best_val, best_a = new_val, a_new.copy()
            print(f"[LP {it:05d}] improved to {best_val:.6f}")
            # slowly increase mixing if things go well
            t = min(0.9, t * 1.02)
        else:
            # no improvement – reduce the mixing step size
            t = max(0.05, t * 0.95)

        # stop early if we have already reached the target
        if best_val <= 1.503:
            print("[LP] target reached")
            break

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – focused hill‑climbing (exact objective).
    # ------------------------------------------------------------------
    hill_start = time.time()
    best_a, best_val = _hill_climb(best_a,
                                   deadline=deadline,
                                   target_val=1.503,
                                   max_frac_start=0.40,
                                   max_frac_end=0.01)
    print(f"[Hill] final best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – run the provided secure wrapper.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return the list (sum = 1, all entries >= 0)
    return best_a.tolist()
