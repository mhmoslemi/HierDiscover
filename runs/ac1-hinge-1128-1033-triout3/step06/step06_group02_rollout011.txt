"""
Propose a step‐function (non‑negative heights) that minimizes
``evaluate_sequence``.

Algorithm outline
-----------------
* Initialise from the historic best construction ``height_sequence_1`` (if
  available) and up‑sample it to the chosen length ``n``.
* Run a few short L‑BFGS optimisations on a smooth surrogate
  ``L_ε = 2·n·max(conv) + 2·n·ε·log∑exp((conv‑max)/ε)``.
* Perform a long mirror–descent phase that uses a *multi‑active* sub‑gradient:
  instead of the single lag achieving the maximum, we sum the gradients of the
  top‑K lags (K≈5).  The update is
        a ← a·exp(‑α·g)   (α decays slowly)
  and a is projected back onto the probability simplex.
* Occasionally apply cheap re‑orderings that tend to flatten the convolution:
  – anti‑symmetric (large ↔ small on opposite sides)  
  – high‑low interleaving (largest values placed alternately at the left and
    right ends)
* Finish with a fast hill‑climbing refinement that performs random swaps and
  small mass transfers, accepting only strict improvements.
* The internal objective ``2·n·max(conv)/(sum(a)**2)`` is used throughout
  (identical to ``evaluate_sequence`` for feasible vectors).  The expensive
  ``evaluate_sequence`` call is made only at the very end (and a few time‑stamped
  checkpoints) to keep the total runtime well below the 1000 s budget.
"""

import time

from scipy.special import logsumexp

# Global placeholders for the smooth‑surrogate L‑BFGS routine
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (useful for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Clip tiny negative values that may appear due to rounding
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Pair large values with small ones when the sequence is reversed."""
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _interleave_high_low(a: np.ndarray) -> np.ndarray:
    """Place the sorted values alternately at the left and right ends."""
    n = a.shape[0]
    sorted_desc = np.sort(a)[::-1]          # descending
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    toggle = True
    for v in sorted_desc:
        if toggle:
            new_a[left] = v
            left += 1
        else:
            new_a[right] = v
            right -= 1
        toggle = not toggle
    return new_a


def _geometric_decay_sequence(n: int, decay: float) -> np.ndarray:
    """
    Produce a symmetric ‘bow‑tie’ shape: a_i = decay**i + decay**(n‑1‑i),
    then normalise to sum 1.
    """
    idx = np.arange(n, dtype=float)
    a = decay ** idx + decay ** (n - 1 - idx)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _objective_score(a: np.ndarray) -> float:
    """
    Internal objective identical to ``evaluate_sequence`` for feasible a.
    a is assumed non‑negative; sum may be arbitrary.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Smooth surrogate for L‑BFGS (log‑sum‑exp smoothing of the max)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (numerically stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)                # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()                # probability distribution over k

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # extract the part that corresponds to dL/da
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (reads globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

        alpha = alpha0 * (0.995 ** (it // 500))

        # exponential (mirror) update preserving non‑negativity
        a = a * np.exp(-alpha * g)

        # re‑project onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        # occasional bookkeeping
        if it % 150 == 0:
            cur_score = _objective_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[grad iter {it}] new best internal score = {best_score:.6f}")

            # try the two cheap re‑orderings
            for reorder_fun, name in [( _anti_symmetric_reorder, "anti‑sym"),
                                      ( _interleave_high_low,     "interleave")]:
                cand = reorder_fun(a)
                sc = _objective_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_a = cand.copy()
                    a = cand.copy()
                    print(f"[{name} reorder @ iter {it}] improved internal score to {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing refinement (swap / mass transfer moves)
    # ------------------------------------------------------------------
    a = best_a.copy()
    best_score = _objective_score(a)

    while time.time() < deadline:
        # randomly choose an operation
        if rng.random() < 0.2:            # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                             # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 1e-15:
                continue
            delta = a[i] * 0.08 * rng.random()    # up to 8 % of a[i]
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project to guard against numerical drift
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_score = _objective_score(new_a)
        if new_score + 1e-12 < best_score:   # strict improvement
            a = new_a
            best_a = new_a.copy()
            best_score = new_score
            print(f"[hill climb] new best internal score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final check & return (use the official evaluator once)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] true score reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
