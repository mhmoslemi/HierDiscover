"""
Propose a non‑negative step function that minimises the evaluation function.

Algorithm (≈ 1000 s time budget)

0️⃣  Choose a problem size.  Larger n gives a finer discretisation of the
    (continuous) optimal shape.  Here n=12000 works well with the FFT
    helpers used below.

1️⃣  **Parametric sweep** – generate a handful of hand‑crafted shapes
    (uniform, linear decay, quadratic decay, half‑sine,
    exponential decay, symmetric V‑shapes and their inverse powers).
    Keep the best one (lowest exact objective).

2️⃣  **Adam minimisation on the exact max** – we work on the simplex by
    keeping soft‑max logits `w`.  At each iteration the true maximal
    convolution entry `b_k` is found, the exact sub‑gradient
    ∂max/∂a_i = 2·a_{k‑i} (if the index is in range) is built and
    an Adam update is performed on the logits.  The learning‑rate and
    Adam‐ε decay linearly; the phase uses ≈ 40 % of the total budget.

3️⃣  **Targeted hill‑climbing** – a fast local search that moves a tiny
    amount of mass from an index that contributes most to the current
    maximal convolution entry to a low‑contribution index.
    The move is accepted only if the exact objective improves.
    This phase consumes ≈ 45 % of the budget.

4️⃣  **Short simulated‑annealing** – occasional worsening moves are
    accepted to escape plateaus (≈ 15 % of the budget).

All phases respect the time budget, keep the vector on the simplex,
and monitor the best sequence seen.  The final sequence is returned as a
list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Parametric initialisation – explore a few useful families
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return the best of several hand‑crafted shapes and its objective."""
    best_a = np.full(n, 1.0 / n)   # uniform start
    best_val = _objective_value(best_a)

    # helper to evaluate and keep the best
    def _consider(a):
        nonlocal best_a, best_val
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            return
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    # monotone decreasing (linear)
    a = np.arange(n, 0, -1, dtype=float)
    _consider(a)

    # monotone decreasing (quadratic)
    a = (np.arange(n, 0, -1, dtype=float) ** 2)
    _consider(a)

    # half‑sine
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    _consider(a)

    # exponential decay
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        _consider(a)

    # symmetric V‑shape (ends high, centre low)
    idx = np.arange(n)
    a = np.maximum(idx + 1, n - idx)                     # linear V
    _consider(a)

    # power V‑shape (|i‑center|+1)^p
    centre = (n - 1) / 2.0
    for p in (0.5, 1.0, 2.0):
        a = (np.abs(idx - centre) + 1) ** p
        _consider(a)

    # inverse‑power (U‑shape) a_i = (i+1)^-p + (n-i)^-p
    for p in (0.5, 1.0, 2.0):
        a = (idx + 1) ** (-p) + (n - idx) ** (-p)
        _consider(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on the *exact* max (sub‑gradient of max)
# ----------------------------------------------------------------------
def _adam_max_phase(a0: np.ndarray,
                    n: int,
                    start: float,
                    deadline: float,
                    best_a: np.ndarray,
                    best_val: float,
                    target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the exact max of the self‑convolution.
    The vector is kept on the simplex via the soft‑max parametrisation.
    """
    # initialise logits
    w = np.log(a0 + 1e-12)                     # Σa = 1 automatically via soft‑max

    # Adam hyper‑parameters (tuned for stability)
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.15, 0.001
    eps_start, eps_end = 1e-2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 180_000

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max to obtain a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # exact convolution and max index
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))
        max_b = conv[k_max]

        cur_val = 2.0 * n * max_b                     # exact objective

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[AdamMax {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedules
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # sub‑gradient of max w.r.t a: g_i = 2 * a_{k_max - i}
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad_a = np.zeros_like(a)
        # factor 2 (derivative of b_k) times 2*n (objective scaling) = 4*n
        grad_a[mask] = 4.0 * n * a[idx[mask]]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.003:
            w += np.random.normal(scale=0.01, size=n)

        # restart logic
        if no_improve > IMPROVE_LIMIT:
            print(f"[AdamMax {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Move tiny amounts of mass from high‑contribution indices to low‑contribution ones."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each i to the current maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – bias towards large contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – 30 % chance to deliberately pick a low‑contrib position
        if np.random.rand() < 0.30:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            # favour indices with *small* rev_contrib
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = np.random.choice(n, p=probs_j)
            else:
                # fallback to uniform distinct index
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta          # self‑term at j
        b_new[2 * i] += delta * delta          # self‑term at i (removed mass)
        b_new[i + j] -= 2.0 * delta * delta    # cross term

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-14:          # strict improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb the current best and restart from there
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")
        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase C – short simulated‑annealing refinement
# ----------------------------------------------------------------------
def _annealing_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Metropolis‑style refinement that sometimes accepts worsening moves."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # temperature schedule (linear decay)
    T_start, T_end = 5e-4, 5e-8
    IMPROVE_LIMIT = 100_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        k_max = int(np.argmax(b))

        # source index – same bias as hill‑climber
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # simple uniform target distinct from i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # tiny transfer (≤ 0.5 % of source mass)
        delta = a[i] * np.random.rand() * 0.005
        if delta <= 0.0:
            continue

        # incremental convolution update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # small random shake
            noise = np.random.normal(scale=0.003, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function under the given time budget.
    Returns a list of non‑negative floats (normalised to sum = 1)
    that minimises `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a half‑second safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few ten‑thousands balances fidelity and speed.
    # ------------------------------------------------------------------
    n = 12000

    # ------------------------------------------------------------------
    # Initialise a – try to interpolate from a previous construction if
    # available, otherwise fall back to the parametric sweep.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0:
            raise ValueError
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    TARGET = 1.503

    # ------------------------------------------------------------------
    # Phase A – Adam on the exact max
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + (deadline - start_time) * 0.40
    best_a, best_val = _adam_max_phase(a0, n,
                                       start=start_time,
                                       deadline=phase_a_deadline,
                                       best_a=best_a,
                                       best_val=best_val,
                                       target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase B – targeted hill‑climbing on the exact objective
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + (deadline - start_time) * 0.85
    best_a = _hillclimb_targeted(best_a, n,
                                 start=phase_a_deadline,
                                 deadline=phase_b_deadline,
                                 best_a=best_a,
                                 best_val=best_val,
                                 target_val=TARGET)

    # ------------------------------------------------------------------
    # Phase C – short simulated‑annealing refinement
    # ------------------------------------------------------------------
    best_a, best_val = _annealing_phase(best_a, n,
                                        start=phase_b_deadline,
                                        deadline=deadline,
                                        best_a=best_a,
                                        best_val=best_val,
                                        target_val=TARGET)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
