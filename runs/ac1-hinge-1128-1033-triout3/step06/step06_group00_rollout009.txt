def subgrad_descent(a, end_time):
    while time.time() < end_time:
        conv = _full_convolution_fft(a, a)
        maxc = conv.max()
        val = 2.0 * n * maxc

        # compute active

        active = np.where(conv >= maxc - tol)[0]

        w = np.zeros_like(conv)

        else:
            w[np.argmax(conv)] = 1.0
        # gradient
        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n-1:n-1+n]
        # step size search
        step = 0.5  # initial maybe high
        improvement = False

            a_new = _project_simplex(a - step * g)

                a = a_new
                val = val_new
                improvement = True
                break
            step *= 0.5
        if not improvement:

            a = a + rng.normal(scale=0.001, size=n)

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Algorithm
---------

1. **Warm‑start generation**  
   Build a large pool of candidate sequences:
   * a handful of classic analytic shapes (exponential, power‑law,
     half‑sine, Gaussian, triangle, Catalan‑type).
   * many random convex mixtures of those shapes (different random
     weights each time).
   * a few Dirichlet‑distributed random vectors with random
     concentration parameters.
   * optionally reuse the previously‑found global `height_sequence_1`.

   All candidates are projected onto the probability simplex
   (non‑negative, sum = 1) and the best one (lowest objective) is kept.

2. **Phase 1 – Smoothed max‑gradient descent**  
   Optimise the objective using a soft‑max approximation of the max
   (weights `w_i ∝ exp(conv_i/ε)`).  Nesterov momentum is applied and
   both the learning rate `η` and the temperature `ε` are slowly
   decayed.  The iterate is always projected onto the simplex.

3. **Phase 2 – Exact sub‑gradient descent with back‑tracking line‑search**  
   The true sub‑gradient of `max(conv)` (uniform weight on all active
   indices where the max is attained) is computed.  A simple back‑track
   line‑search along the projected descent direction guarantees a
   monotone decrease.

4. **Phase 3 – Deterministic mass‑transfer refinement**  
   Using the current sub‑gradient we repeatedly move a small amount of
   probability mass from the index with the largest gradient (most
   harmful) to the one with the smallest gradient.  When this deterministic
   move does not improve the score we fall back to a tiny random mass
   transfer (simulated‑annealing style).

The whole procedure respects a 1000 s wall‑clock budget.  The best
sequence discovered is finally clipped to the interval `[0, 1000]` and
returned.  Intermediate scores are printed for monitoring.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex using the algorithm of Duchi et al. (2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution using a real FFT (fast and accurate)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b (length len(a)+len(b)-1)."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a, a)) (assumes sum(a) == 1)."""
    n = a.size
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * conv.max()

            print(f"[phase1 iter {it:6d}] new_best={best_val:.12f}")

        # soft‑max weights (smooth approximation of the max)
        w = np.exp(conv / eps)
        w_sum = w.sum()
        if not np.isfinite(w_sum) or w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # Nesterov update
        vel = momentum * vel - eta * g
        a_new = a + vel
        a = _project_simplex(a_new)

        # schedules
        eta = max(eta_min, eta * eta_decay)
        eps = max(eps_min, eps * eps_decay)

        # occasional tiny random jitter to avoid stagnation
        if rng.random() < 0.001:
            a = a + rng.normal(scale=0.004, size=n)
            a = _project_simplex(a)

    # --------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent with back‑tracking line‑search.
    # --------------------------------------------------------------
    phase2_end = start_time + 0.80 * budget_s
    a = best_seq.copy()
    it = 0
    while time.time() < phase2_end:
        it += 1
        conv = _full_convolution_fft(a, a)
        maxc = conv.max()
        cur_val = 2.0 * n * maxc
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[phase2 iter {it:6d}] new_best={best_val:.12f}")

        # active set (indices where the maximum is attained)
        tol = 1e-12
        active = np.flatnonzero(conv >= maxc - tol)
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        else:
            w[np.argmax(conv)] = 1.0

        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # simple back‑tracking line‑search
        step = 0.5
        improved = False
        while step > 1e-12:
            a_candidate = _project_simplex(a - step * g)
            conv_cand = _full_convolution_fft(a_candidate, a_candidate)
            val_cand = 2.0 * n * conv_cand.max()
            if val_cand < cur_val - 1e-14:
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            # small stochastic perturbation if we get stuck
            a = a + rng.normal(scale=0.001, size=n)
            a = _project_simplex(a)

    # --------------------------------------------------------------
    # Phase 3 – deterministic mass‑transfer + occasional random moves.
    # --------------------------------------------------------------
    cur = best_seq.copy()
    cur_val = _objective(cur)
    if cur_val < best_val - 1e-12:
        best_val = cur_val
        best_seq = cur.copy()
        print(f"[phase3 start] new_best={best_val:.12f}")

    while time.time() < deadline:
        conv = _full_convolution_fft(cur, cur)
        maxc = conv.max()
        cur_val = 2.0 * n * maxc

        # sub‑gradient for the active constraints
        tol = 1e-12
        active = np.flatnonzero(conv >= maxc - tol)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size
        rev_cur = cur[::-1]
        g = 2.0 * _full_convolution_fft(rev_cur, w)[n - 1:n - 1 + n]

        # indices with largest and smallest gradient components
        i_high = np.argmax(g)
        i_low = np.argmin(g)

        # try a deterministic mass transfer from i_high -> i_low
        if cur[i_high] > 0:
            delta = min(cur[i_high] * 0.10, 0.02)  # up to 10 % or 0.02 mass
            new = cur.copy()
            new[i_high] -= delta
            new[i_low] += delta
            new = _project_simplex(new)
            new_val = _objective(new)
            if new_val < cur_val - 1e-12:
                cur = new
                cur_val = new_val
                if new_val < best_val - 1e-12:
                    best_val = new_val
                    best_seq = cur.copy()
                    print(f"[transfer] new_best={best_val:.12f}")
                continue  # go to next iteration

        # fallback: small random mass‑transfer (simulated‑annealing style)
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] <= 0:
            continue
        max_transfer = min(cur[i], 0.01)
        delta = rng.uniform(0.0, max_transfer)
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new = _project_simplex(new)
        new_val = _objective(new)
        if new_val < cur_val - 1e-12:
            cur = new
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_seq = cur.copy()
                print(f"[random transfer] new_best={best_val:.12f}")

    # ------------------------------------------------------------------
    # Return the best sequence in the format expected by `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # Optional logging of the final official score (if evaluate_sequence is available)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] evaluate_sequence not available")

    return final_seq
