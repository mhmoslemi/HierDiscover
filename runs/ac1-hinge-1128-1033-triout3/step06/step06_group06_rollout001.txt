def best_exponential():
    lo=0.9
    hi=0.999999
    for _ in range(30):
        m1=lo+(hi-lo)/3
        m2=hi-(hi-lo)/3
        a1 = np.power(m1, np.arange(n, dtype=float))
        a1 /= a1.sum()
        val1 = objective(a1)
        a2 = np.power(m2, np.arange(n, dtype=float))
        a2 /= a2.sum()
        val2 = objective(a2)
        if val1 < val2:
            hi=m2
        else:
            lo=m1
    r_opt = (lo+hi)/2
    a_opt = np.power(r_opt, np.arange(n, dtype=float))
    a_opt /= a_opt.sum()
    return a_opt, objective(a_opt), r_opt

  parameter(s) w.r.t. the exact objective 2·n·max(conv(a,a)).
  The best candidate from this sweep seeds the next stage.

    * Work on logits w with a soft‑max to enforce Σa=1 and a≥0.
    * A smooth approximation of max via a temperature ε (log‑sum‑exp)

      learning rate (0.1→0.002) and ε (0.1→1e‑12).  Restarts are triggered

# ----------------------------------------------------------------------
# Convolution helpers

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) is assumed to be 1)
# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())

# Parameteric families
# ----------------------------------------------------------------------
def _make_exponential(n: int, r: float, L: int = None) -> np.ndarray:
    """Exponential a_i ∝ r**i, optionally truncated after L entries."""
    if L is None:
        L = n
    idx = np.arange(L, dtype=np.float64)
    a = r ** idx
    a_full = np.zeros(n, dtype=np.float64)
    a_full[:L] = a
    a_full /= a_full.sum()
    return a_full


def _make_powerlaw(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)**(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Simple golden‑section search for a single scalar parameter
# ----------------------------------------------------------------------
def _golden_section(func, lo, hi, iters=30):
    """Minimise `func` on [lo,hi] using golden‑section."""
    phi = (1 + 5 ** 0.5) / 2
    invphi = 1 / phi
    a, b = lo, hi
    fa = func(a)
    fb = func(b)
    for _ in range(iters):
        c = b - invphi * (b - a)
        d = a + invphi * (b - a)
        fc = func(c)
        fd = func(d)
        if fc < fd:
            b, fb = d, fd
        else:
            a, fa = c, fc
    # return the best of the final interval
    if fa < fb:
        return a, fa
    else:
        return b, fb

    # 5) a half‑wave sine shape (already part of the earlier code)
    a_sin = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a_sin = np.maximum(a_sin, 0.0)
    a_sin /= a_sin.sum()
    eval_and_update(a_sin)

    # 6) use a previously found construction if available
    try:
        old = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a_old = np.interp(new_x, old_x, old)
        a_old = np.maximum(a_old, 0.0)
        a_old /= a_old.sum()
        eval_and_update(a_old)
    except Exception:
        pass

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 2 – Smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray, start_time: float, deadline: float,
              best_a: np.ndarray, best_val: float,
              target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation of a smooth approximation of max(conv(a,a))."""
    n = a0.shape[0]
    w = np.log(a0 + 1e-12)                     # logits, Σa = 1

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.10, 0.002
    eps_start, eps_end = 0.1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 30000

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max to obtain a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # exact objective for bookkeeping
        cur_val = _objective_exact(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedules
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)                     # length 2·n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()                        # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        # ∂/∂a 2·n·∑ p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)                     # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]                # n entries that correspond to ∂c/∂a
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the soft‑max (log‑soft‑max)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # small random jitter to escape flat regions
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.02, size=n)

        # restart after long stagnation
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage 3 – Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(a_start: np.ndarray,
               start_time: float,
               deadline: float,
               best_a: np.ndarray,
               best_val: float,
               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Local search that moves tiny mass from a high‑contributing index to a random one."""
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)                # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    no_improve = 0
    IMPROVE_LIMIT = 200_000
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress‑dependent maximal transfer fraction (starts generous, shrinks)
        prog = (now - start_time) / (deadline - start_time)
        max_frac = 0.5 * (1.0 - prog) + 0.001 * prog   # linearly from 0.5 → 0.001

        # index where the convolution is maximal
        k_max = int(np.argmax(b))

        # contributions of each i to that entry
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx = np.arange(i_low, i_high + 1, dtype=np.int64)
        contrib = a[idx] * a[k_max - idx]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(idx, p=probs)
        else:
            i = np.random.choice(idx)

        if a[i] <= 1e-16:
            continue

        # target index (uniformly chosen, different from i)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        # add/subtract the linear terms
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic self‑terms and cross‑term
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept strictly improving moves
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – small random shake around the best seen
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a list of heights (Σ≈1) that minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a safety margin for the final evaluate_sequence call
    deadline_total = start + budget_s - 0.5

    # problem size (large enough to give a good bound but still tractable)
    n = 8000

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric sweep
    # ------------------------------------------------------------------
    # allocate roughly 5 % of the total time to the sweep
    deadline_stage1 = start + 0.05 * (deadline_total - start)
    a_best, best_val = _parametric_sweep(n, deadline_stage1)
    print(f"[Stage 1] best after sweep = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation on a smooth surrogate
    # ------------------------------------------------------------------
    # give about 70 % of the remaining budget to Adam
    deadline_stage2 = start + 0.75 * (deadline_total - start)
    a_best, best_val = _adam_opt(a_best, start, deadline_stage2,
                                 best_a=a_best, best_val=best_val,
                                 target_val=1.503)
    print(f"[Stage 2] best after Adam = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – focused hill‑climbing on the exact objective
    # ------------------------------------------------------------------
    # remaining time goes to hill‑climbing
    best_a, best_val = _hillclimb(a_best,
                                  start_time=deadline_stage2,
                                  deadline=deadline_total,
                                  best_a=a_best,
                                  best_val=best_val,
                                  target_val=1.503)
    print(f"[Stage 3] best after hillclimb = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final renormalisation and secure evaluation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        # fall back to uniform if something went wrong
        best_a = np.full(n, 1.0 / n)
    else:
        best_a /= best_a.sum()

    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
