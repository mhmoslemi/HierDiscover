"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.

The algorithm consists of two stages:

1. **Alternating linear programme** –  Starting from a good
   (or random) normalised vector `a` we repeatedly:
        * compute `M = max(a*a)`,
        * solve a linear programme **maximising** the ℓ₁‑norm of a
          vector `g` under the linear constraints ``a * g ≤ M`` (full
          convolution), and
        * replace `a ← (1‑t)·a + t·ĝ` where `ĝ = g / sum(g)` and `t`
          is a small step size.
   This is the classic “alternating LP” described in the statement and
   provably never increases the objective.

2. **Mass‑transfer hill‑climb** –  After the LP stage we run a fast
   greedy hill‑climber (the same routine used in the previous solution)
   that moves a fraction of the mass away from the lag that attains the
   current convolution maximum.  The climb works on the exact objective
   and therefore finishes with a locally optimal sequence.

All work respects a 1000 s budget; intermediate best values are printed.
The final list of heights is returned.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself."""
    return np.convolve(x, x)


# ----------------------------------------------------------------------
# Build the convolution matrix A such that (a @ g) = conv(a,g)
# ----------------------------------------------------------------------
def _conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Return a sparse matrix A (shape (2*n‑1, n)) satisfying

        (A @ g) == np.convolve(a, g, mode='full')

    for any vector g of length n.
    """
    n = a.shape[0]
    i = np.arange(n)[:, None]          # column index of a (i)
    j = np.arange(n)[None, :]          # column index of g (j)
    rows = (i + j).ravel()             # lag = i + j
    cols = np.repeat(np.arange(n), n)  # column = j
    data = np.tile(a, n)               # coefficient = a_i
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# Hill‑climbing refinement (identical to the one used before)
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray, best_val: float,
                              deadline: float,
                              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Aggressive hill climb that moves a small fraction of mass away from
    the lag where the convolution attains its maximum."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.2

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue

        top_n = min(3, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]
        donors = [(int(i_range[idx]), int(j_range[idx])) for idx in top_idx]

        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue

        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue
        new_a /= s

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.02, 0.5)
        else:
            step_frac = max(step_frac * 0.95, 1e-5)

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise the step‑function heights.

    1. Initialise from ``height_sequence_1`` if present, otherwise draw a
       random Dirichlet vector (length n≈800).
    2. Run a few iterations of the alternating LP scheme that provably
       never increase the exact objective.
    3. Finish with the aggressive hill‑climbing refinement from the
       previous solution.
    Returns the best list of floats found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate size keeps the LP cheap but still allows
    # a good reduction of the objective.
    # ------------------------------------------------------------------
    n = 800

    # ------------------------------------------------------------------
    # Initialise (reuse previous construction if provided)
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    a /= a.sum()                      # ℓ₁‑normalise to 1

    best_a = a.copy()
    best_val = 2.0 * n * np.max(_full_convolution(best_a))
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Alternating linear‑programming stage
    # ------------------------------------------------------------------
    max_lp_iters = 25
    step_t = 0.2                      # initial convex‑combination weight
    lp_iter = 0

    while time.time() < deadline - 5.0 and lp_iter < max_lp_iters:
        conv_a = _full_convolution(a)
        M = conv_a.max()

        # Build the (2n‑1) × n constraint matrix for conv(a,g) ≤ M
        A = _conv_matrix(a)

        # Linear programme: maximise sum(g)  ⇔ minimise –sum(g)
        c = -np.ones(n)
        b = M * np.ones(2 * n - 1)

        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            print(f"[LP] iteration {lp_iter} failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print("[LP] trivial solution, stopping.")
            break

        # Normalise g to ℓ₁‑norm 1 before mixing
        g_norm = g / sum_g

        # Convex combination (step size adapts slowly)
        a = (1.0 - step_t) * a + step_t * g_norm
        a = np.clip(a, 0.0, None)
        a /= a.sum()

        cur_val = 2.0 * n * np.max(_full_convolution(a))

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[LP iter {lp_iter}] new best = {best_val:.6f}")

        # Simple adaptive step‑size rule
        if cur_val >= best_val:
            step_t = max(step_t * 0.9, 0.05)
        else:
            step_t = min(step_t * 1.05, 0.5)

        lp_iter += 1

    # ------------------------------------------------------------------
    # Final aggressive hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _mass_transfer_hill_climb(best_a, best_val,
                                                    deadline, rng)

    # ------------------------------------------------------------------
    # Report the exact score obtained from the provided evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
