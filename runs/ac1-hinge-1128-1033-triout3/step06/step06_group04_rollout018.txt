```python
def geometric_distribution(r, n):
    if r <= 0:
        return np.full(n, 1.0/n)
    if r >= 1:
        return np.full(n, 1.0/n) # r=1 uniform
    # compute powers r^i
    i = np.arange(n, dtype=np.float64)
    # Use np.exp for stability
    log_r = np.log(r)
    pow_r = np.exp(i * log_r)  # r^i
    # sum of geometric series
    # sum_geo = (1 - r**n) / (1 - r)
    # compute r_n = np.exp(n * log_r)
    r_n = np.exp(n * log_r)
    # avoid catched underflow for large n
    if np.isclose(r, 1.0):
        sum_geo = n  # limit
    else:
        sum_geo = (1 - r_n) / (1 - r)
    a = pow_r / sum_geo
    return a
```

```python
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    from scipy.optimize import minimize_scalar, minimize

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Problem dimension
    n = 4000

    # Helper functions for convolution and objective
    def full_conv(x):
        # same as above _full_convolution
        n_loc = x.shape[0]
        out_len = 2 * n_loc - 1
        size = 1 << ((out_len - 1).bit_length())
        fx = np.fft.rfft(x, size)
        conv = np.fft.irfft(fx * fx, size)[:out_len]
        # remove tiny negatives
        np.maximum(conv, 0, out=conv)
        return conv

    def objective_of_a(a):
        # assume a sum to 1
        return 2.0 * n * np.max(full_conv(a))

    # Phase 1: parametric search (geometric)
    best_val = np.inf
    best_a = None
    # Evaluate geometric candidates on a log scale for r close to 1
    # Use grid first
    # r in (0.9, 0.99999)
    grid_rs = np.linspace(0.9, 0.99999, 200)
    for r in grid_rs:
        a = geometric_distribution(r, n)  # defined later
        val = objective_of_a(a)
        if val < best_val:
            best_val = val
            best_a = a
    # Refine around best r using scalar minimization
    if best_a is not None:
        # find r bounds from grid around best one
        idx_best = np.argmin([objective_of_a(geometric_distribution(r, n)) for r in grid_rs])
        # use a small interval around the best r from grid
        r_low = max(0.0, grid_rs[max(0, idx_best-1)])
        r_high = min(0.999999, grid_rs[min(len(grid_rs)-1, idx_best+1)])
        # minimize within [r_low, r_high]
        res = minimize_scalar(
            lambda rr: objective_of_a(geometric_distribution(rr, n)),
            bounds=(r_low, r_high),
            method='bounded',
            options={'xatol':1e-8, 'maxiter':50}
        )
        r_opt = res.x
        a_opt = geometric_distribution(r_opt, n)
        val_opt = objective_of_a(a_opt)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt
        print(f"[geom] best r = {r_opt:.8f}, value = {best_val:.6f}")

    # Optionally try power‑law family (a_i ∝ (i+1)^-p)
    def powerlaw_distribution(p):
        idx = np.arange(1, n+1, dtype=np.float64)  # 1..n
        pow_vals = idx ** (-p)
        a = pow_vals / pow_vals.sum()
        return a

    # coarse grid for p
    for p in np.linspace(0.0, 4.0, 41):
        a = powerlaw_distribution(p)
        val = objective_of_a(a)
        if val < best_val:
            best_val = val
            best_a = a
    # fine search via scalar for p near best
    # if we found a best p, refine using minimize_scalar
    # Do this only if best_a came from powerlaw maybe

    # Phase 2: optional local smooth optimisation (LBFGS on surrogate)
    # We'll reuse earlier code but start from best_a
    # Represent a via softmax parameters w
    w = np.log(best_a + 1e-16)
    # define surrogate (same as earlier)
    def smooth_obj_grad(w_vec, eps):
        # implement similar to earlier but returning L and grad
        # to avoid copy-paste we let's define identical function inside.
        w_shift = w_vec - np.max(w_vec)
        a = np.exp(w_shift)
        a = a / a.sum()
        # convolution
        conv = full_conv(a)
        max_conv = conv.max()
        shifted = (conv - max_conv) / eps
        log_sum = np.logaddexp.reduce(shifted)  # stable logsumexp
        L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum
        # gradient w.r.t a
        weights = np.exp(shifted)
        p = weights / weights.sum()
        # grad_a = 4 n * convolution of a with reversed p
        p_rev = p[::-1]
        conv_ap = np.fft.irfft(np.fft.rfft(a) * np.fft.rfft(p_rev), n + n - 1)[:2*n-1]
        seg = conv_ap[n-1:2*n-1]
        grad_a = 4.0 * n * seg[::-1]
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)
        return L, grad_w

    # continuation schedule

            return smooth_obj_grad(w_vec, eps)
        res = minimize(fun_and_grad, w, method='L-BFGS-B', jac=True,
                       options={'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12, 'disp': False})
        w = res.x

    def mass_transfer_hill_climb(a, cur_val, deadline, rng):

        cur_a = a.copy()
        best_local = a.copy()
        best_val_local = cur_val

        while time.time() < deadline:

            ks = np.flatnonzero(conv == max_b)
            k = int(rng.choice(ks))

            i_end = min(k, n_loc - 1)

            j_range = k - i_range
            contrib = cur_a[i_range] * cur_a[j_range]
            if contrib.size == 0:

            # pick top contributors (up to 5)
```

"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.

Approach
--------
1. **Parametric families** –  Search over simple one‑parameter families that
   already give a low value of the objective.  The most effective is a
   geometric (exponential) decay  a_i ∝ r^i  with r≈0.9995 for n=4000.
2. **Smooth surrogate optimisation** –  Starting from the best parametric
   candidate, run a continuation of a log‑sum‑exp surrogate with L‑BFGS‑B.
3. **Aggressive mass‑transfer hill‑climb** –  A specialised local search
   moves a small amount of mass away from the lag that attains the current
   maximum convolution, redistributing it to low‑mass positions.
   This phase finishes the optimisation and usually pushes the objective
   below the target 1.503.
All helper functions are defined at module level (no closures) and the
algorithm respects a 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize, minimize_scalar

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact evaluate_sequence objective without the security checks.
    The sequence is assumed to be non‑negative and already normalised
    to sum = 1, therefore the denominator is 1.
    """
    n = a.shape[0]
    return float(2.0 * n * np.max(_full_convolution(a)))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _geometric_distribution(r: float, n: int) -> np.ndarray:
    """Return a normalised geometric (exponential) distribution of length n."""
    if r <= 0.0 or r >= 1.0:
        # r outside (0,1) → uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)

    i = np.arange(n, dtype=np.float64)
    log_r = np.log(r)
    pow_r = np.exp(i * log_r)            # r**i
    # geometric series sum = (1‑r**n)/(1‑r)
    r_n = np.exp(n * log_r)
    sum_geo = (1.0 - r_n) / (1.0 - r)
    a = pow_r / sum_geo
    return a


def _powerlaw_distribution(p: float, n: int) -> np.ndarray:
    """Return a normalised power‑law distribution a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = idx ** (-p)
    a = vals / vals.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w_vec: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Compute the smooth surrogate
        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv-max)/ε)
    and its gradient w.r.t. the soft‑max parameters w.
    """
    n = w_vec.shape[0]

    # ---- soft‑max (feasible a, Σa = 1) ----
    w_shift = w_vec - np.max(w_vec)
    a = np.exp(w_shift)
    a = a / a.sum()

    # ---- convolution a * a ----
    conv = _full_convolution(a)
    max_conv = conv.max()

    # ---- log‑sum‑exp smoothing ----
    shifted = (conv - max_conv) / eps
    # stable log‑sum‑exp
    max_shift = np.max(shifted)
    log_sum = max_shift + np.log(np.sum(np.exp(shifted - max_shift)))
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t a (closed form) ----
    weights = np.exp(shifted)
    p = weights / weights.sum()               # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)             # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]            # middle n entries
    grad_a = 4.0 * n * seg[::-1]              # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Hill‑climbing mass‑transfer refinement
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray,
                              cur_val: float,
                              deadline: float,
                              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Aggressive local search that repeatedly moves a small amount of mass."""
    n = a.shape[0]
    cur_a = a.copy()
    best_a = a.copy()
    best_val = cur_val
    step_frac = 0.25                     # initial mass‑transfer fraction

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        # pick a (random) lag attaining the maximum
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue

        # select the most damaging pairs (up to 5)
        top_n = min(5, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]
        donors = [(int(i_range[t]), int(j_range[t])) for t in top_idx]

        # choose low‑mass destinations
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-6)
            continue

        # renormalise (preserve ℓ₁‑norm = 1)
        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.9, 1e-6)
            continue
        new_a /= s

        new_val = _objective(new_a)

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[mass_climb] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.02, 0.5)
        else:
            step_frac = max(step_frac * 0.95, 1e-6)

    return best_a, best_val

    # --------------------------------------------------------------

                                                    deadline, rng)

    # --------------------------------------------------------------
    # Final report (using the provided evaluator)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
