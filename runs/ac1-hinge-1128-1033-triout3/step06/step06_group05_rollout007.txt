def _correlate_w_a(w, a):
    n = len(a)
    a_rev = a[::-1]
    out_len = len(w) + n - 1   # 3n-2
    size = 1 << (out_len - 1).bit_length()
    fw = np.fft.rfft(w, size)
    fa_rev = np.fft.rfft(a_rev, size)
    corr_full = np.fft.irfft(fw * fa_rev, size)[:out_len]
    # central part corresponding to indices i in [0,n-1]
    start = n - 1
    grad = corr_full[start:start+n]
    # add diagonal correction
    diag_idx = np.arange(n)
    k2 = 2*diag_idx
    valid = k2 < len(w)
    grad[valid] += w[k2[valid]] * a[valid]

def _smooth_descent(a, n, deadline, best_a, best_val):
    start = time.time()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0
    beta1=0.9
    beta2=0.999
    eps=1e-8
    # hyperparameters
    step_start = 0.5
    step_end = 0.01
    alpha_start = 10.0
    alpha_end = 5000.0
    while time.time() < deadline:
        # compute convolution
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        # update best

            best_val = cur_val
            best_a = a.copy()

        # compute w = softmax(alpha * b)

def _correlate_with_weights(w, a):
    n = a.shape[0]

Propose a step‑function with minimal evaluation score.

The algorithm mixes three complementary optimisation phases
within the 1000 s wall‑time limit:

1. **Smoothed log‑sum‑exp gradient descent** – the maximum of the
   autocorrelation is approximated by a soft‑max (temperature α).

   linear schedule and we keep the vector on the probability
   simplex.

3. **Greedy hill‑climbing** – a deterministic local search that
   repeatedly moves a small amount of mass from the index that
   contributes most to the current peak of the convolution to the
   index that contributes least.  The full convolution is updated
   incrementally in O(n) time, better values are kept and occasional
   Gaussian perturbations escape stagnation.

All vectors are kept non‑negative and normalised (Σa = 1),
which leaves the evaluation function invariant under scaling.
The best sequence found is finally evaluated with the provided
``evaluate_sequence`` wrapper.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution a∗a (real vectors) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:  # numerical safety
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


def _correlate_with_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a∗a)_k w.r.t. a.

    w – non‑negative weights (typically a softmax over the convolution).
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    out_len = len(w) + n - 1               # = 3n‑2
    size = 1 << (out_len - 1).bit_length()
    fw = np.fft.rfft(w, size)
    fa_rev = np.fft.rfft(a_rev, size)
    corr_full = np.fft.irfft(fw * fa_rev, size)[:out_len]

    start = n - 1
    grad = corr_full[start : start + n].copy()

    # diagonal correction for the terms a_i·a_i
    idx = np.arange(n)
    grad += w[2 * idx] * a
    return grad


# ----------------------------------------------------------------------
# Phase 1 – Smoothed log‑sum‑exp (soft‑max) gradient descent
# ----------------------------------------------------------------------
def _smooth_descent_phase(a: np.ndarray,
                          n: int,
                          deadline: float,
                          best_a: np.ndarray,
                          best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimisation of a smooth surrogate of max(b)."""
    start = time.time()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # schedules
    step_start, step_end = 0.5, 0.02
    alpha_start, alpha_end = 10.0, 5000.0
    it = 0

    while time.time() < deadline:
        # convolution and current objective
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth {it:7d}] new best = {best_val:.6f}")

        # progress‑dependent hyper‑parameters
        progress = (time.time() - start) / (deadline - start)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        step = step_start * (1.0 - progress) + step_end * progress

        # soft‑max weights over the convolution entries
        tb = alpha * b
        shift = tb.max()
        w = np.exp(tb - shift)
        w /= w.sum()

        # gradient of the surrogate
        grad = _correlate_with_weights(w, a)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Exact sub‑gradient descent on max(b)
# ----------------------------------------------------------------------
def _subgradient_phase(a: np.ndarray,
                       n: int,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    """Linear‑schedule sub‑gradient descent on the true L∞ objective."""
    start = time.time()
    max_step = 0.10
    min_step = 1e-4
    it = 0

    while time.time() < deadline:
        b = _full_convolution(a)
        k = int(np.argmax(b))
        cur_max = b[k]
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Subgrad {it:7d}] new best = {best_val:.6f}")

        # sub‑gradient of the peak entry
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                grad[mid] = 2.0 * a[mid]

        # linear step‑size schedule
        progress = (time.time() - start) / (deadline - start)
        step = max_step * (1.0 - progress) + min_step * progress

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – Deterministic greedy hill‑climbing
# ----------------------------------------------------------------------
def _greedy_hillclimb(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> np.ndarray:
    """
    Move a tiny amount of mass from the index that most contributes to
    the current maximum convolution to the index that contributes least.
    Full convolution is updated incrementally (O(n)) after each trial.
    """
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    no_improve = 0
    it = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        k = max_idx

        # contribution of each a_i to the current worst entry b_k
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        if source == target:
            # fall back to a random pair
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-14:
            # pick a non‑empty source
            possible = np.where(a > 1e-14)[0]
            if possible.size == 0:
                break
            source = int(np.random.choice(possible))

        # move a random fraction (up to 30 %) of the source mass
        delta = a[source] * 0.30 * np.random.rand()
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        # cross‑terms: 2·δ·a shifted by target / source
        b_new[target : target + n] += 2.0 * delta * a
        b_new[source : source + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            # accept the move
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # occasional perturbation after long stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimise
    `evaluate_sequence`.  The routine respects the 1000 s wall‑time
    limit and combines three optimisation phases (smooth descent,
    sub‑gradient descent and greedy hill‑climbing).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – 8000 gives a good trade‑off between flexibility and speed
    n = 8000

    # initialise a – try to reuse the previously supplied construction
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # pylint: disable=undefined-variable
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)   # enforce Σa = 1
    best_a = a.copy()
    best_val = float('inf')   # will be updated immediately

    # split the remaining budget
    total = deadline - start_time
    t_smooth_end = start_time + 0.35 * total
    t_subgrad_end = start_time + 0.65 * total
    t_hill_end = deadline

    # ------------------------------------------------------------------
    # Phase 1 – smoothed log‑sum‑exp gradient descent
    # ------------------------------------------------------------------
    best_a, best_val = _smooth_descent_phase(a, n, t_smooth_end, best_a, best_val)
    a = best_a.copy()

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_phase(a, n, t_subgrad_end, best_a, best_val)
    a = best_a.copy()

    # ------------------------------------------------------------------
    # Phase 3 – greedy hill‑climbing
    # ------------------------------------------------------------------
    best_a = _greedy_hillclimb(a, n, t_hill_end, best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (the wrapper may clamp values, but scale invariance
    # guarantees the same objective).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
