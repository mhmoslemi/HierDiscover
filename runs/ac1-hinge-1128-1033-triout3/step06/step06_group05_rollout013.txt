"""
A three‑phase optimizer for the step‑function evaluation problem.

Phase 0 – smoothed‑max descent.
    Uses a log‑sum‑exp surrogate for max(b) and an Adam optimiser.
    The gradient of the surrogate can be written as a convolution,
    which we compute efficiently with an FFT.

Phase 1 – L²‑flattening.
    Minimises the variance of the autocorrelation while keeping the
    sequence non‑negative and normalised (simplex projection).

Phase 2 – deterministic hill‑climbing.
    Moves a little mass from the index that contributes most to the
    current peak of the convolution to the index that contributes least.
    The full convolution is updated incrementally in O(n) time.

The routine respects a 1000 s wall‑time limit, starts from the previously
found construction when available, and finally returns the best
sequence (as a python list) that it discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ( Σa = 1, a_i ≥ 0 )
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:                                   # numerical safety
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Full convolution a * a (real, zero‑padded) using FFT
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Phase 0 – Smoothed‑max descent (log‑sum‑exp surrogate) with Adam
# ----------------------------------------------------------------------
def _smooth_max_phase(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> tuple[np.ndarray, float]:
    """Minimise max(b) via a smooth surrogate and Adam updates."""
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0
    start = time.time()
    while time.time() < deadline:
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth {t:7d}] new best = {best_val:.6f}")

        # schedule for step size and smoothing parameter α
        prog = (time.time() - start) / (deadline - start)
        lr = 0.20 * (1.0 - prog) + 0.001 * prog          # learning rate
        alpha = 0.5 + (30.0 - 0.5) * prog                # smoothing strength

        # log‑sum‑exp weights (stable)
        b_max = b.max()
        exps = np.exp(alpha * (b - b_max))
        sumexp = exps.sum()
        w = exps / sumexp                                 # |w| = 1, non‑negative

        # gradient = 2 * Σ_k w_k * shift(a, k)
        # compute via convolution of w_rev and a
        w_rev = w[::-1]
        conv_len = len(w_rev) + n - 1                    # = 3n‑2
        size = 1 << ((conv_len - 1).bit_length())
        conv = np.fft.irfft(
            np.fft.rfft(w_rev, size) * np.fft.rfft(a, size), size
        )[:conv_len]
        offset = len(w_rev) - 1                         # = 2n‑2
        grad = 2.0 * conv[offset:offset + n]

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (t + 1))
        v_hat = v / (1.0 - beta2 ** (t + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a = _project_simplex(a_candidate)

        t += 1
    return best_a, best_val

def _greedy_hillclimb(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> np.ndarray:
    """Mass‑transfer hill‑climbing with O(n) incremental convolution updates."""
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 150_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # contribution of each a_i to the current worst convolution entry
        k = max_idx
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-14:
            nz = np.where(a > 1e-14)[0]
            if nz.size == 0:
                break
            source = int(np.random.choice(nz))

        # move a small random fraction (up to 30 %) of the source mass
        delta = a[source] * 0.30 * np.random.rand()

            continue

        # incremental update of the convolution (linear‑time)
        b_new = b.copy()
        # cross‑terms
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        # pure‑δ² terms
        idx_tt = 2 * target
        idx_ss = 2 * source
        if idx_tt < b_new.shape[0]:
            b_new[idx_tt] += delta * delta
        if idx_ss < b_new.shape[0]:
            b_new[idx_ss] += delta * delta
        idx_st = source + target
        if idx_st < b_new.shape[0]:
            b_new[idx_st] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            # accept move
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()

            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.001, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")
        it += 1

    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # pylint: disable=undefined-variable
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)
    best_a = a.copy()
    best_val = float('inf')

    # --------------------------------------------------------------
    # Split the remaining wall‑time into three phases.
    # --------------------------------------------------------------
    total_time = deadline - start_time
    t_phase0 = start_time + 0.50 * total_time   # 50 % for smoothed‑max descent
    t_phase1 = start_time + 0.80 * total_time   # next 30 % for L² flattening
    # remaining time for hill climbing

    # ----------------------------
    # Phase 0 – Smoothed‑max descent (Adam)
    # ----------------------------
    best_a, best_val = _smooth_max_phase(
        a, n, t_phase0, best_a, best_val
    )
    a = best_a.copy()

    # ----------------------------
    # Phase 1 – L² flattening
    # ----------------------------
    best_a, best_val = _l2_flatten_phase(
        a, n, t_phase1, best_a, best_val
    )
    a = best_a.copy()

    # ----------------------------
    # Phase 2 – Greedy hill climbing
    # ----------------------------
    best_a = _greedy_hillclimb(
        a, n, deadline, best_a, best_val
    )

    # ------------------------------------------------------------------
    # Final evaluation (the secure wrapper may clamp values)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
