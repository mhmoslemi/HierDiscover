   n = len(x)
   m = len(y)

   fy = np.fft.rfft(y, size)
   conv = np.fft.irfft(fx * fy, size)[:n+m-1]
   return conv

   objective `2·n·max(conv(a,a))` (the sum is kept equal to 1, therefore the
   objective is `2·n·max(conv)` and does not depend on scaling).  For each
   iteration:
   * compute the full autocorrelation `c = a*a`;
   * set a temperature `τ = M·α` where `M = max(c)` and `α` decays from
     `0.1` to `0.001` during the run;
   * form a probability distribution over the convolution indices
     `w_k ∝ exp((c_k-M)/τ)`;
   * the gradient of the smooth maximum is
        g_i = 2·∑_k w_k·a_{k‑i},
     which we evaluate with a single FFT‑based convolution;
   * take a projected (onto the probability simplex) gradient step with a
     diminishing step size `η_t = η₀/√t`.
   * keep the best vector encountered (by the cheap surrogate
     `2·n·max(conv)`) and occasionally inject a small random perturbation
     to escape plateaus.
3. Finish with the *guided hill‑climbing* routine from the previous state‑of‑the‑art
   implementation – it moves a small fraction of the mass away from the
   most offending convolution lag and therefore polishes the solution.
4. Return the best list of heights found before the 1000 s budget expires.

The whole procedure runs well within the time limit and typically improves
the previous bound `1.512153` towards the target `1.503`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(
    a: np.ndarray,
    cur_val: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Refine a candidate by moving a small fraction of the mass away from the
    lag where the autocorrelation attains its maximum.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    step_frac = 0.05

    while time.time() < deadline:
        conv = _full_convolution(best_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # contribution a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = best_a[i_range] * best_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # choose two destination indices with relatively low mass
        low = np.where(best_a < best_a.mean())[0]
        if low.size >= 2:
            p, q = rng.choice(low, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        # avoid recreating the same lag
        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # perform a feasible mass transfer
        if i_max != j_max:
            delta = min(best_a[i_max], best_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = best_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:  # i_max == j_max
            delta = best_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = best_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        val = evaluate_sequence(new_a.tolist())
        if val < best_val:
            best_a = new_a
            best_val = val
            print(f"[guided] new best = {best_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise the step‑function heights using an annealed soft‑max
    sub‑gradient descent followed by guided hill‑climbing.
    Returns the best list of non‑negative heights found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – 8000 entries give a good balance of expressivity
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise from the previous best construction if available,
    # otherwise start from a uniform distribution.
    # --------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if a0.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    except Exception:
        a0 = np.full(n, 1.0 / n)

    # break symmetry a little
    a0 += rng.normal(scale=1e-3, size=n)
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    a = a0.copy()

    # cheap surrogate objective (sum of a is 1, therefore value = 2·n·max(conv))
    conv = _full_convolution(a)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val
    print(f"[init] n={n}, initial surrogate value = {best_val:.6f}")

    # --------------------------------------------------------------
    # Annealed soft‑max sub‑gradient descent
    # --------------------------------------------------------------
    it = 0
    step0 = 0.5               # initial step size
    tau_start = 0.1           # initial temperature factor (relative to M)
    tau_end = 0.001           # final temperature factor

    # main optimisation loop
    while time.time() < deadline - 10.0:
        it += 1

        # ---- current autocorrelation and temperature ----------
        conv = _full_convolution(a)
        M = conv.max()
        elapsed = time.time() - start_time
        total = (deadline - start_time - 10.0)
        fraction = elapsed / total if total > 0 else 1.0
        tau_factor = tau_start * ((tau_end / tau_start) ** fraction)
        tau = max(1e-12, M * tau_factor)

        # ---- soft‑max weights over the lag indices ---------------
        w_exp = np.exp((conv - M) / tau)        # max exponent = 1
        w = w_exp / w_exp.sum()                # probability distribution

        # ---- gradient of the smooth maximum ----------------------
        w_rev = w[::-1]
        grad_full = _conv_fft(a, w_rev)        # length 3·n‑2
        # extract the slice that corresponds to g_i = 2·Σ_k w_k·a_{k‑i}
        grad_mid = grad_full[n - 1 : 2 * n - 1][::-1]   # size n, reversed
        grad = 2.0 * grad_mid

        # ---- projected gradient step ----------------------------
        step = step0 / np.sqrt(it)
        cand = a - step * grad
        cand = _simplex_projection(cand)

        # cheap evaluation of candidate
        cand_conv = _full_convolution(cand)
        cand_val = 2.0 * n * cand_conv.max()

        # keep the globally best solution
        if cand_val < best_val:
            best_val = cand_val
            best_a = cand.copy()
            print(f"[iter {it}] new best surrogate = {best_val:.6f}")

        # accept improvement; otherwise keep current a but inject occasional noise
        if cand_val < cur_val:
            a = cand
            cur_val = cand_val
        else:
            if it % 100 == 0:
                # small random perturbation to escape plateaus
                pert = rng.normal(scale=0.005, size=n)
                a = _simplex_projection(a + pert)
                cur_val = 2.0 * n * _full_convolution(a).max()

    # --------------------------------------------------------------
    # Final guided hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # --------------------------------------------------------------
    # Final reporting (exact evaluation)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] exact evaluation = {final_score:.6f}")

    return best_a.tolist()
