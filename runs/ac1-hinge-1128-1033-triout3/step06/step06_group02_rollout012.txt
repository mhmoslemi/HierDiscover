The LP solver function:

```
def solve_lp(a, M):
    n = len(a)
    # Build A sparse matrix
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, max_iters=2000, abstol=1e-9, reltol=1e-9)
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    return np.array(b_var.value).flatten()
```

def _smooth_grad_and_obj(w, eps, n):
    # compute a via softmax
    w_shift = w - np.max(w)
    exp_w = np.exp(w_shift)
    a = exp_w / np.sum(exp_w)
    # convolution a with itself
    conv = _full_convolution(a)  # length 2n-1
    max_conv = conv.max()
    # smoothed term
    shifted = (conv - max_conv) / eps
    # compute logsumexp
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # compute weights for gradient
    # p = exp(shifted) / sum(exp(shifted))
    # For numerical stability, we can compute weights as exp(shifted) directly using log_sumexp trick:
    # weights = np.exp(shifted - log_sum) # but then sum(weights) = 1.
    # Actually we need probabilities p = exp(shifted) / sum(exp(shifted)).
    # As we already have log_sum, exp(shifted - log_sum) is the probability.
    weights = np.exp(shifted - log_sum)  # shape (2n-1)
    # p is same as weights
    p = weights
    # conv a with reversed p
    p_rev = p[::-1]
    # convolution length needed = n + (2n-1) -1 = 3n-2
    size = _next_pow2(3 * n - 2)

    # extract middle n entries

    grad_a = 4.0 * n * seg[::-1]
    # chain rule
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w, a, max_conv

"""
Propose a step‑function (list of non‑negative heights) that minimizes
``evaluate_sequence``.

The algorithm is a refined mirror‑descent optimiser working in the
log‑soft‑max (mirror) space.  A smooth surrogate of the original
non‑smooth objective is used:

    L_ε(a) = 2·n·max(conv(a)) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

Its gradient w.r.t. the log‑parameters w (a_i = exp(w_i)/Σexp) is
computed analytically (two FFT‑based convolutions).  An Adam optimiser
updates w, while ε is geometrically decreased from 5·10⁻³ down to
10⁻¹⁰.  The step size is also decayed slowly.  Every few hundred
iterations the true objective (``evaluate_sequence``) is evaluated;
if an improvement is found the best vector is stored.  Occasionally an
anti‑symmetric re‑ordering heuristic is applied (pair large values with
small ones) and after the main loop a short hill‑climbing refinement
(mass transfers / swaps) finishes the search.

All operations respect the 1000 s budget; the function returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np
from scipy.special import logsumexp
import scipy.sparse as sp

# ----------------------------------------------------------------------
# FFT convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself
    (length 2·len(x)‑1).  Uses an FFT of size a power of two.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)      # clip tiny negatives
    return conv


def _conv_with_rev(a: np.ndarray, p_rev: np.ndarray) -> np.ndarray:
    """
    Convolution of a (length n) with p_rev (length 2n‑1) using FFT.
    Returns the full result of length n + (2n‑1)‑1 = 3n‑2.
    """
    n = a.shape[0]
    size = _next_pow2(3 * n - 2)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv = np.fft.irfft(fa * fp, size)[: 3 * n - 2]
    return conv


# ----------------------------------------------------------------------
# Smooth surrogate objective and its gradient (mirror space)
# ----------------------------------------------------------------------


def _smooth_obj_grad(w: np.ndarray, eps: float, n: int):
    """
    Given log‑soft‑max parameters w, smoothing ε and length n,
    return three items:
        L      – value of the smooth surrogate,
        grad_w – gradient w.r.t. w,
        a      – the current probability vector (soft‑max of w).
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    exp_w = np.exp(w_shift)
    a = exp_w / np.sum(exp_w)

    # self‑convolution of a
    conv = _full_convolution(a)                # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    # log‑sum‑exp (stable)
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # probability distribution over lags (soft max)
    # p_k = exp((conv_k‑max)/ε) / Σ exp(...)
    p = np.exp(shifted - log_sum)              # shape (2·n‑1,)

    # convolution a * rev(p)  → needed for gradient w.r.t. a
    p_rev = p[::-1]
    conv_ap = _conv_with_rev(a, p_rev)         # length 3·n‑2

    # extract the middle n entries (the derivative of each a_i)
    seg = conv_ap[n - 1 : 2 * n - 1]           # length n
    grad_a = 4.0 * n * seg[::-1]               # shape (n,)

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)                # shape (n,)
    return L, grad_w, a, max_conv


# ----------------------------------------------------------------------
# Simple heuristics
# ----------------------------------------------------------------------


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair large values with small values when the sequence is reversed.
    This heuristic often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Random hill‑climbing: swaps (20 %) and mass transfers (80 %).
    Only strict improvements are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # projection onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimizes ``evaluate_sequence``.  The method combines a smooth‑surrogate
    Adam optimiser (mirror descent) with occasional anti‑symmetric
    re‑ordering and a final hill‑climbing phase.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0    # safety margin

    # Problem size – 4000 coefficients give a good trade‑off
    # --------------------------------------------------------------
    n = 4000

    # it exists, otherwise start from the uniform distribution.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the required length

        else:

    else:

    # --------------------------------------------------------------
    best_a = a0.copy()

    # log‑soft‑max parameters
    w = np.log(best_a + 1e-12)

    # Adam state
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    # learning‑rate schedule
    lr0 = 0.22

    # smoothing schedule (geometric decay)
    eps = 5e-3
    eps_min = 1e-10
    eps_decay = 0.9992   # per 100 iterations

    it = 0
    while time.time() < deadline:
        # ---- smooth surrogate evaluation & gradient ----
        L, grad_w, a, max_conv = _smooth_obj_grad(w, eps, n)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        lr = lr0 * (0.995 ** (it // 300))
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- periodic checks on the true objective ----
        if it % 30 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[adam iter {it}] new best = {best_val:.6f}")

        # ---- occasional anti‑symmetric reorder ----
        if it % 400 == 0 and it > 0:
            a_ord = _anti_symmetric_reorder(best_a)
            val_ord = evaluate_sequence(a_ord.tolist())
            if val_ord < best_val:
                best_val = val_ord
                best_a = a_ord.copy()
                # also bring current w close to the reordered solution
                w = np.log(best_a + 1e-12)
                m[:] = 0.0
                v[:] = 0.0
                beta1_pow = 1.0
                beta2_pow = 1.0
                print(f"[reorder at {it}] improved to {best_val:.6f}")

        # ---- decay smoothing parameter ----
        if it % 100 == 0 and eps > eps_min:
            eps = max(eps_min, eps * eps_decay)

        it += 1

    # --------------------------------------------------------------
    # Final local improvement – hill‑climbing (mass transfers / swaps)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # --------------------------------------------------------------
    # Final sanity check & return
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
