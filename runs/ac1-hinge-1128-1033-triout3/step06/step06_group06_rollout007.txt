"""
Propose a step‑function that minimizes the provided `evaluate_sequence`.

Algorithm
---------

1.  **Initialisation** – Choose a modest length `n` (8000) and a simple
    exponentially decaying starting shape (also interpolates a previous best
    if available).

2.  **Exponentiated‑gradient descent on the exact maximum** –  
    Work on the simplex by keeping logits `w` and using `a = softmax(w)`.  
    At each iteration the convolution `c = a * a` is computed, the index
    `k_max` of the current maximum is located and the gradient of the
    *exact* objective `2·n·c[k_max]` w.r.t. `w` is evaluated analytically.
    A decreasing learning‑rate schedule is used, occasional random
    perturbations and restarts avoid stagnation.

3.  **Targeted hill‑climbing** –  Starting from the best vector found
    above, repeatedly move a tiny amount of mass from indices that
    contribute most to the current maximal convolution entry to indices that
    contribute little.  The move is accepted only when it strictly improves
    the exact objective; after long periods without improvement a small
    random shake is performed.

4.  **Short simulated‑annealing polishing** –  Allows occasional worsening
    moves to escape shallow local minima.

The whole routine respects a 1000 s time budget, printing progress and
returning the best sequence found (as a plain list of non‑negative floats,
already normalised to sum = 1).  The final sequence is evaluated with the
provided `evaluate_sequence` wrapper.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# --------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# --------------------------------------------------------------
# Exact objective (sum(a) assumed = 1)
# --------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# --------------------------------------------------------------
# Exponentiated‑gradient descent on the exact maximum
# --------------------------------------------------------------
def _expgrad_phase(a_start: np.ndarray,
                   n: int,
                   start: float,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """
    Gradient descent on `2·n·max(a*a)` using exponentiated updates.
    The gradient is taken w.r.t. the logits `w` (soft‑max parametrisation)
    of the simplex.
    """
    # initialise logits; tiny offset prevents log(0)
    w = np.log(a_start + 1e-12)

    # parameters
    lr_start, lr_end = 0.12, 0.001
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max to obtain a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a_sum = a.sum()
        if a_sum == 0.0:         # safety – should never happen
            a = np.full(n, 1.0 / n)
            a_sum = 1.0
        a /= a_sum

        # full convolution and current max
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k_max]

        # keep best
        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[ExpGrad {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # gradient of the exact max (w.r.t. a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad_shift = np.zeros_like(a)
        grad_shift[mask] = a[idx[mask]]
        # d/da of 2·n·c[k_max] == 4·n·a_{k_max-i}
        grad_a = 4.0 * n * grad_shift

        # chain‑rule to logits
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # learning‑rate schedule (linear in log‑space)
        progress = (now - start) / (deadline - start)
        lr = lr_start * (lr_end / lr_start) ** progress

        # update
        w -= lr * grad_w

        # occasional random shake to escape plateaus
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.02, size=n)

        # restart if we have been idle for a long time
        if no_improve > IMPROVE_LIMIT:
            print(f"[ExpGrad {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            no_improve = 0

        it += 1

    return best_a, best_val


# --------------------------------------------------------------
# Targeted hill‑climbing on the exact objective
# --------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> np.ndarray:
    """
    Fast local search that moves a small amount of mass from indices that
    heavily contribute to the current maximal convolution entry to indices
    that contribute little.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.20, 0.001   # start aggressive
    IMPROVE_LIMIT = 180_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the current maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # pick source index biased by contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # choose target index – favour low contribution with 30 % prob
        if np.random.rand() < 0.30:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = np.random.choice(n, p=probs_j)
            else:
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept strictly improving moves
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling – perturb and restart from the best seen
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")
        it += 1

    return best_a

# Short simulated‑annealing refinement

def _annealing_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float) -> tuple[np.ndarray, float]:
    """Metropolis‑style search that occasionally accepts worsening moves."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # temperature schedule (linear decay)
    T_start, T_end = 5e-4, 5e-8
    IMPROVE_LIMIT = 100_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        k_max = int(np.argmax(b))

        # source index – same bias as hill‑climber
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # uniform target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # tiny transfer (≤ 0.5 % of source mass)
        delta = a[i] * np.random.rand() * 0.005
        if delta <= 0.0:
            continue

        # update convolution incrementally
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # small random shake
            noise = np.random.normal(scale=0.003, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val

    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Ensure exact sum‑to‑one (tiny floating errors)
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a /= best_a.sum()

    return best_a.tolist()
