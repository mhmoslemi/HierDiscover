"""
Propose a step‑function (non‑negative sequence) that (approximately) minimises
``evaluate_sequence``.

Algorithm
---------

1. **Smooth optimisation** – minimise a log‑sum‑exp surrogate of the
   ``max(conv)`` term with L‑BFGS‑B.  The surrogate is gradually
   “sharpened” by decreasing the smoothing parameter ε (continuation).

2. **Local stochastic search** – starting from the best point found above,
   run a fast simulated‑annealing / hill‑climbing phase that updates the
   sequence by moving a small amount of mass between two indices.
   The convolution is updated *incrementally* in O(n) time, allowing
   many thousands of proposals within the remaining time budget.

3. **Final report** – the best sequence discovered is returned and its
   exact score is printed via ``evaluate_sequence``.

The implementation respects the 1000 s time limit, prints progress
information, and uses only NumPy / SciPy (no file or network I/O).
"""

import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full (linear) autocorrelation of a real vector.
    Returns a length ``2·len(x)‑1`` array.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip tiny negatives from FFT noise
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (linear, non‑circular)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the smooth surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv‑max)/ε)

    together with its gradient w.r.t. the soft‑max parameters ``w``.
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # full autocorrelation (length 2·n‑1)
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()                          # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                        # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]                      # middle n entries
    grad_a = 4.0 * n * seg[::-1]                         # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Incremental convolution update for a single mass‑transfer move
# ----------------------------------------------------------------------
def _apply_move(conv: np.ndarray, a: np.ndarray, i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a copy of ``conv`` after moving ``delta`` mass from index ``i`` to ``j``.
    The update is performed exactly (no FFT) in O(n) time.
    """
    n = a.shape[0]
    new_conv = conv.copy()

    # the linear terms 2·δ·(a_{k‑j} ‑ a_{k‑i})
    # they correspond to slice updates of length n
    new_conv[i:i + n] -= 2.0 * delta * a
    new_conv[j:j + n] += 2.0 * delta * a

    # the quadratic term –2·δ² at lag k = i + j
    idx = i + j
    new_conv[idx] -= 2.0 * delta * delta

    # numerical noise can produce tiny negatives – clip them
    np.maximum(new_conv, 0, out=new_conv)
    return new_conv


# ----------------------------------------------------------------------
# Stochastic local search (simulated annealing / hill‑climbing)
# ----------------------------------------------------------------------
def _stochastic_search(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a fast stochastic local search that repeatedly moves a small
    amount of mass from a “dangerous’’ index (one that contributes a lot
    to the current maximal convolution entry) to a low‑mass index.
    The convolution is updated incrementally, allowing thousands of
    iterations per second.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_conv = _full_convolution(cur_a)
    cur_val = best_val
    best_a = cur_a.copy()
    best_score = best_val

    # temperature schedule – linear decay from an initial value
    T0 = cur_val * 0.02   # start with a modest temperature
    while time.time() < deadline:
        # progress fraction for temperature decay
        progress = (time.time() - (deadline - 1000)) / 1000.0
        T = max(T0 * (1.0 - progress), 1e-12)

        # locate the lag where convolution attains its maximum
        k_max = int(np.argmax(cur_conv))

        # contributions a_i * a_{k_max‑i} (zero where out of range)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=np.intp)
        idx_j = k_max - idx_i
        contrib = cur_a[idx_i] * cur_a[idx_j]

        if contrib.size == 0:
            # shouldn't happen, but safeguard
            donor = rng.integers(n)
        else:
            # pick the index i that contributes the most to the peak
            donor = int(idx_i[np.argmax(contrib)])

        # pick a destination with relatively low mass
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size == 0:
            low_mass = np.arange(n)
        # avoid moving to the donor itself
        cand = np.setdiff1d(low_mass, np.array([donor]), assume_unique=True)
        if cand.size == 0:
            dest = donor
        else:
            dest = int(rng.choice(cand))

        # propose a move: shift a fraction of the donor's mass to the destination
        max_move = cur_a[donor]
        if max_move <= 0:
            continue
        # step size adapts automatically via temperature (larger T → larger steps)
        frac = min(0.5, max(0.1, T / cur_val))        # between 0.1 and 0.5
        delta = max_move * rng.random() * frac

        # compute the new convolution using the incremental formula
        new_conv = _apply_move(cur_conv, cur_a, donor, dest, delta)

        new_val = 2.0 * n * new_conv.max()

        # Metropolis acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = math.exp((cur_val - new_val) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            # update the state
            cur_a[donor] -= delta
            cur_a[dest] += delta
            cur_conv = new_conv
            cur_val = new_val

            if cur_val < best_score:
                best_score = cur_val
                best_a = cur_a.copy()
                print(f"[SA] new best = {best_score:.6f}")

    return best_a, best_score
