"""
Propose a step‑function (non‑negative list) that minimises the
`evaluate_sequence` objective.

Algorithm
---------
1. **Initialisation**
   * If a previous solution `height_sequence_1` exists, interpolate it to the
     working length `n`.
   * Sweep a two‑parameter Beta‑shape family
       a_i ∝ (i+1)^{p-1} (n-i)^{q-1}
     over a grid of (p,q) and keep the best seed.
   * Add a few random seeds for diversity.

2. **Adam optimisation (smooth max)**
   * Optimise on the simplex using the soft‑max approximation of the
     supremum norm of the self‑convolution.
   * Learning‑rate and temperature (ε) are annealed logarithmically.
   * After a prescribed number of idle iterations the optimiser is
     re‑initialised with a fresh random point.

3. **Simulated‑annealing hill‑climbing**
   * Repeatedly move a small amount of mass from an index that contributes
     heavily to the current maximal convolution entry to an index that
     contributes little.
   * Moves are accepted if they improve the exact objective, otherwise
     with Metropolis probability `exp(-(Δ)/T)`, where `T` is a temperature
     that decays linearly to zero.
   * Stagnation triggers a short random perturbation of the whole vector.

4. **Return the best vector found** (sum is kept equal to 1, which does not
   affect the final objective because it is scale‑invariant).

All heavy work (FFT‑based convolutions, gradients, incremental updates)
is done with NumPy.  The routine respects the 1000 s time limit, keeping a
0.5 s safety margin for the final call to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) and gradient w.r.t. a
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float) -> tuple[float, np.ndarray]:
    """
    Returns a smooth approximation of max(conv(a,a)) and its gradient.
    The gradient corresponds to the exact objective
    2·n·max(conv) approximated by a soft‑max with temperature eps.
    """
    n = a.shape[0]
    b = _full_convolution(a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights
    shifted = (b - max_b) / eps
    weights = np.exp(shifted)
    p = weights / weights.sum()                  # shape (2n‑1,)

    # gradient of Σ p_k * b_k w.r.t a
    # Using convolution property: ∂b_k/∂a_i = a_{k-i} + a_i (mirrored)
    # The vector of “soft” contributions is the convolution of a with
    # reversed p, restricted to the n indices that influence a_i.
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)                # length 2n‑1
    segment = conv_ab[n - 1:2 * n - 1]            # n entries
    grad_a = 4.0 * n * segment[::-1]             # factor 4n from 2·n·max
    # Gradient of smooth max term (the factor 2·n is already inside grad_a)
    # Return smooth max value (not multiplied by 2·n) and gradient.
    smooth_val = max_b + eps * np.log(weights.sum())
    return smooth_val, grad_a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   start_time: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises a smooth approximation of the objective.
    Returns the best (a, value) seen before *deadline*.
    """
    # logits for soft‑max (guarantees Σa = 1)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.001
    eps_start, eps_end = 1.0, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    idle = 0
    MAX_IDLE = 25_000                     # restart after too many idle steps

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            idle = 0
            print(f"[Adam {iteration:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            idle += 1

        # schedule for temperature (eps) and learning rate (lr)
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max and gradient
        smooth_val, grad_a = _smooth_max_and_grad(a, eps)

        # chain rule through the soft‑max (w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation to avoid lock‑in
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.015, size=n)

        # restart on stagnation
        if idle > MAX_IDLE:
            print(f"[Adam {iteration:7d}] restarting after {idle} idle steps")
            w = np.random.randn(n) * 0.15
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            idle = 0

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climbing
# ----------------------------------------------------------------------
def _hillclimb_anneal(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Local search that transfers a small amount of mass from
    high‑contribution indices to low‑contribution ones.
    Accepts uphill moves with Metropolis probability.
    """
    a = a0.copy()
    b = _full_convolution(a)               # length 2n‑1
    cur_val = 2.0 * n * b.max()
    eps = 1e-12

    # schedule for maximal transfer fraction and temperature
    max_frac_start, max_frac_end = 0.45, 0.001
    temp_start, temp_end = 0.012, 1e-7

    no_improve = 0
    IMPROVE_LIMIT = 120_000

    iteration = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = temp_start * (temp_end / temp_start) ** progress

        # locate current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ---- source index (probability ∝ contribution) ----
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ---- target index (biased towards *low* contribution) ----
        tgt_weights = np.zeros_like(a)
        tgt_weights[mask] = 1.0 / (contrib[mask] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fallback uniform among all except i
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_val < cur_val - 1e-15:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.025, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Parametric sweep of a Beta‑shaped family
# ----------------------------------------------------------------------
def _beta_family_best(n: int) -> tuple[np.ndarray, float]:
    """Search a two‑parameter Beta family a_i ∝ i^{p-1} (n‑i)^{q-1}."""
    best_a = None
    best_val = np.inf
    # grid (log‑space) for exponents
    p_vals = np.concatenate([np.linspace(0.1, 1.0, 9),
                              np.geomspace(1.0, 6.0, 7)[1:]])
    q_vals = p_vals.copy()
    i = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for q in q_vals:
            a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
            s = a.sum()
            if s == 0.0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    return best_a, best_val
