"""
Optimiser for the step‑function bound.

The algorithm consists of three stages:

1. **Parametric sweep** – a cheap exhaustive search over a few
   low‑dimensional families of “U‑shaped” sequences.
   Families considered:
        * product‑law   a_i ∝ ((i+α)*(n‑1‑i+α))^{-γ}
        * sum‑law      a_i ∝ (i+α)^{-γ} + (n‑1‑i+α)^{-γ}
        * optional mixture with a uniform baseline.
   The best candidate of this sweep is taken as a seed.

2. **Smooth‑max Adam optimisation** – the seed is refined by
   gradient descent on the smoothed objective
        2·n·∑_k p_k·(a⋆a)[k] ,
   where p is a soft‑max over the convolution entries.
   Adam with an exponentially decreasing learning‑rate and
   temperature is run for the bulk of the time budget.

3. **Targeted hill‑climbing** – a fast greedy local search that
   repeatedly transfers a small amount of mass from a “source”
   index (chosen proportionally to its contribution to the current
   maximal convolution entry) towards a “target” index (chosen
   preferentially among indices that hardly contribute to that entry).
   Exact updates of the convolution are performed in O(n) and a move
   is accepted only if it strictly improves the true objective.
   After long stagnation a small random perturbation is applied.

The whole procedure respects a 1000 s wall‑clock limit; if time expires
early the best solution found so far is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (Σa = 1) – 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero -> return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _shape_product(n: int, gamma: float, alpha: float) -> np.ndarray:
    """a_i ∝ ((i+α)*(n‑1‑i+α))^{-γ}."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + alpha) * (n - 1 - i + alpha)
    a = np.power(denom + 1e-12, -gamma)   # add eps to avoid zero
    return a


def _shape_sum(n: int, gamma: float, alpha: float) -> np.ndarray:
    """a_i ∝ (i+α)^{-γ} + (n‑1‑i+α)^{-γ}."""
    i = np.arange(n, dtype=np.float64)
    term1 = np.power(i + alpha + 1e-12, -gamma)
    term2 = np.power(n - 1 - i + alpha + 1e-12, -gamma)
    a = term1 + term2
    return a


def _mix_with_uniform(a: np.ndarray, lam: float) -> np.ndarray:
    """Return (1‑lam)·a + lam·uniform, then re‑normalise."""
    n = a.shape[0]
    uniform = np.full_like(a, 1.0 / n)
    a_mix = (1.0 - lam) * a + lam * uniform
    s = a_mix.sum()
    if s == 0:
        return uniform
    return a_mix / s


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp smoothed version of the max‑convolution."""
    n = a0.shape[0]
    # initialise logits for the soft‑max (guarantees Σa = 1)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters (tuned for this problem)
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.18, 0.001
    eps_start, eps_end = 5e-2, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 30_000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule for temperature (ε) and learning rate
        progress = (now - start_time) / (deadline - start_time)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ---- gradient of smoothed objective ----
        b_full = _full_convolution(a)            # length 2·n‑1
        max_b = b_full.max()
        # soft‑max weights (log‑sum‑exp trick)
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)                # ≤ 1
        p = weights / weights.sum()              # p_k ≈ soft‑max over k

        # gradient w.r.t. a (see derivation in analysis)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)            # length 2·n‑1
        segment = conv_ab[n - 1: 2 * n - 1]      # n entries that correspond to ∂/∂a
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the soft‑max parametrisation
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.02, size=n)

        if no_improve > IMPROVE_LIMIT:
            print("[Adam]  restart after long stagnation")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted hill‑climbing
# ----------------------------------------------------------------------
def _hillclimb(a_start: np.ndarray,
               deadline: float,
               best_a: np.ndarray,
               best_val: float,
               target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill‑climb.  Moves mass from an index that
    contributes heavily to the current maximal convolution entry towards
    an index that contributes little.  Moves are accepted only if the
    exact objective improves.
    """
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # schedule for maximal transfer fraction
    max_frac_start, max_frac_end = 0.30, 0.03
    IMPROVE_LIMIT = 180_000
    no_improve = 0
    iteration = 0
    eps = 1e-12

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ----- source index (biased by contribution) -----
        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ----- target index (biased towards *small* contribution) -----
        target_weights = np.zeros_like(a)
        # give weight only to indices that could affect the same entry
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0                     # avoid i==j
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # uniform fallback (exclude i)
            cand = np.arange(n - 1)
            cand = np.where(cand >= i, cand + 1, cand)
            j = np.random.choice(cand)

        # amount of mass to move (bounded by a[i] and max_frac)
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms (three special positions)
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # accept only if strictly improving
        if new_val + 1e-14 < cur_val:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – global perturbation
        if no_improve > IMPROVE_LIMIT:
            print("[Hill] stagnation – random perturb")
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0

        iteration += 1

    return best_a, best_val
