# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full self‑convolution of a real vector using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w

        else:
            no_improve += 1

        # schedule for γ and step size
        progress = (time.time() - start) / (deadline - start)
        gamma = gamma_start + (gamma_end - gamma_start) * progress
        step = max_step * (1.0 - progress) + min_step * progress

        # soft‑max weights (stable)
        maxb = cur_max
        w = np.exp(gamma * (b - maxb))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of φ w.r.t a
        # conv(a_rev, w) gives the sum_{j} a_j w_{i+j}
        conv_part = _conv_fft(a[::-1], w)[n - 1:2 * n - 1]
        diag_part = a * w[2 * np.arange(n)]           # contribution when k = 2i
        grad = conv_part + diag_part

        # normalise step by the largest absolute gradient entry
        max_abs = np.max(np.abs(grad))
        if max_abs > 0:
            lr = step / max_abs
        else:
            lr = 0.0

        a_candidate = a - lr * grad
        a = _project_simplex(a_candidate)

        # occasional random perturbation if stuck for a long time
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.002, size=n)
            a = _project_simplex(a)
            no_improve = 0
            print(f"[Smooth {it}] stagnation – perturbing")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Sub‑gradient descent on the exact max (worst‑index)
# ----------------------------------------------------------------------
def _subgradient_phase(a: np.ndarray,
                       n: int,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    """Reduce max(b) by classic sub‑gradient steps (one worst index)."""
    start = time.time()
    max_step = 0.10
    min_step = 1e-4
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        k = int(np.argmax(b))
        cur_max = b[k]
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Subgrad {it:7d}] new best = {best_val:.6f}")

        # gradient for the worst index
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                grad[mid] = 2.0 * a[mid]

        # linear step‑size schedule
        progress = (time.time() - start) / (deadline - start)
        step = max_step * (1.0 - progress) + min_step * progress

        max_abs = np.max(np.abs(grad))
        lr = step / max_abs if max_abs > 0 else 0.0
        a_candidate = a - lr * grad
        a = _project_simplex(a_candidate)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Greedy hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _greedy_hillclimb(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> np.ndarray:
    """
    Move a small amount of mass from a source index that contributes most
    to the current peak to a target index that contributes least.
    Full convolution is updated incrementally in O(n).
    """
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        k = max_idx
        contrib = np.zeros_like(a)

        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-14:
            # pick a non‑zero source
            nonzero = np.where(a > 1e-14)[0]
            if nonzero.size == 0:
                break
            source = int(np.random.choice(nonzero))

        # move up to 30 % of the source mass (randomised)
        delta = a[source] * 0.30 * np.random.rand()
        if delta <= 0.0:
            continue

        # incremental update of the full convolution
        b_new = b.copy()
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            # accept move
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb to escape a plateau
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")
        it += 1
    return best_a

# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimise
    `evaluate_sequence`.  The routine respects the 1000 s wall‑time limit
    and combines three optimisation phases: smooth‑max descent,
    sub‑gradient descent on the exact max, and greedy hill‑climbing.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – 8000 gives a good trade‑off between flexibility and speed
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # noqa: F821
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)      # enforce Σa = 1
    best_a = a.copy()
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Split the time budget
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    t_smooth = start_time + 0.30 * total_time   # phase A
    t_subgrad = start_time + 0.60 * total_time   # phase B
    t_hill = deadline                           # phase C

    # ----------------------------
    # Phase A – smooth max reduction
    # ----------------------------
    best_a, best_val = _smooth_max_phase(
        a, n, t_smooth, best_a, best_val
    )
    a = best_a.copy()

    # ----------------------------
    # Phase B – sub‑gradient descent on exact max
    # ----------------------------
    best_a, best_val = _subgradient_phase(
        a, n, t_subgrad, best_a, best_val
    )
    a = best_a.copy()

    # ----------------------------
    # Phase C – greedy hill‑climbing
    # ----------------------------
    best_a = _greedy_hillclimb(
        a, n, t_hill, best_a, best_val
    )

    # ------------------------------------------------------------------
    # Final evaluation (the secure wrapper may clamp values)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
