"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

The search consists of three stages:

1. **Adam optimisation on a smooth log‑sum‑exp surrogate** –  
   We optimise the soft‑max parameters `w` (so that `a = softmax(w)`)
   using an Adam‑style mirror descent.  The surrogate
   `L_ε(a)=2·n·max(conv)+2·n·ε·log Σ exp((conv‑max)/ε)` approximates the
   true objective and provides a usable gradient.  A schedule of decreasing
   ε (from 5·10⁻³ down to 1·10⁻⁸) and several random restarts explores
   the landscape.

2. **Permutation‑annealing** –  
   Keeping the best probability vector found, we repeatedly propose swaps
   (or short block reversals) of the entries.  The move is accepted if it
   improves the true objective or with a probability depending on a
   slowly cooling temperature.  This spreads large values apart and
   reduces the peak of the self‑convolution.

3. **Hill‑climbing refinement** –  
   A light‑weight random‑swap / mass‑transfer local search (the same as
   used in the reference implementation) finishes the optimisation.

All steps respect the non‑negativity and sum‑to‑one constraints,
and the total runtime is limited to the supplied budget (default 1000 s).
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Length of the result is 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _softmax(w: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector that sums to 1."""
    shift = w - np.max(w)
    e = np.exp(shift)
    return e / e.sum()


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)  (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)

    Returns (L, ∇_w L).
    """
    # ------- a from w -------------------------------------------------
    w_shift = w - np.max(w)            # numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ------- convolution of a with itself -----------------------------
    conv = _full_convolution(a)        # length 2·n‑1
    max_conv = conv.max()

    # ------- smooth max (log‑sum‑exp) ---------------------------------
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ------- gradient w.r.t. a ----------------------------------------
    weights = np.exp(shifted)
    p = weights / weights.sum()        # distribution over lag indices
    # convolution of a with reversed p
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # extract the middle n entries that correspond to ∂L/∂a_i
    seg = conv_ap[n - 1: 2 * n - 1]     # length n
    grad_a = 4.0 * n * seg[::-1]       # chain of the two convolutions

    # ------- chain rule through soft‑max -------------------------------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that interleaves large and small values.
    Often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _objective(a: np.ndarray) -> float:
    """
    Compute the exact value of ``evaluate_sequence`` for a vector that
    already satisfies the pre‑checks (non‑negative, sum>0, values < 1000).
    For sum(a)=1 this reduces to 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Light‑weight hill‑climbing: random swaps (20 %) or mass transfers
    (80 %).  Only strict improvements are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # project onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = _objective(new_a)
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val
