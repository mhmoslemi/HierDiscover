    return a

    """
    Construct an arcsine‑law step function (or a slightly larger variant) and
    return the sequence that yields the smallest evaluation score within the
    allotted time budget.

    The algorithm:
        * try a couple of problem sizes (default: 4000, 8000, 12000);
        * for each size generate the arcsine‑law vector;
        * evaluate it with ``evaluate_sequence``;
        * keep the best result.
    No random search is required – the analytically optimal shape is used.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.2      # a tiny safety margin

    # candidate lengths – they trade off discretisation error vs. evaluation time.
    # 4000 is the size used previously; larger sizes give a flatter convolution.
    candidate_ns = [4000, 8000, 12000]

    best_seq = None
    best_val = float("inf")

    for n in candidate_ns:
        if time.time() >= deadline:
            break

        a = _arcsine_sequence(n)
        # convert to a plain Python list (evaluate_sequence expects a list)
        seq = a.tolist()

        try:
            val = evaluate_sequence(seq)
        except Exception as e:
            print(f"[error] evaluation failed for n={n}: {e}")
            continue

        print(f"[candidate] n={n}, evaluation = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = seq

    # If for any reason none of the candidates succeeded (should not happen),
    # fall back to a uniform vector of length 4000.
    if best_seq is None:
        n = 4000
        best_seq = (np.full(n, 1.0 / n)).tolist()
        best_val = evaluate_sequence(best_seq)
        print(f"[fallback] n={n}, evaluation = {best_val:.6f}")

    total_time = time.time() - start_time
    print(f"[final] best evaluation = {best_val:.6f}  (computed in {total_time:.2f}s)")

    # The returned list is the best sequence discovered.
    return best_seq
