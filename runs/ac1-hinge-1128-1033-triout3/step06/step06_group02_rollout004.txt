"""
Propose a step‑function minimizing ``evaluate_sequence``.

Algorithm overview
------------------
1. **Initialisation** – start from the provided construction ``height_sequence_1``,
   otherwise use a uniform distribution.  The vector is normalised to sum = 1.

2. **Smooth surrogate optimisation** – minimise a smooth approximation of the
   max‑convolution (log‑sum‑exp) using L‑BFGS on the soft‑max parameters.
   Several restarts and a geometric schedule of the smoothing parameter
   (ε) are performed.  Each successful run updates the best candidate.

3. **Simulated annealing** – a long run of cheap Monte‑Carlo moves
   (swaps, mass transfers, small perturbations) with an exponential cooling
   schedule.  The move is always accepted if it improves the objective,
   otherwise it is accepted with probability exp(-(Δ)/T).  Periodically a
   deterministic “anti‑symmetric” re‑ordering is tried; a simple
   high‑low alternating placement is also tested.

4. **Final hill‑climb** – a short random‑swap / mass‑transfer refinement
   (strictly improving moves only).

All steps respect the 1000 s wall‑time limit.  The returned sequence is a
list of non‑negative floats whose evaluation is printed for reference.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global containers for the L‑BFGS surrogate (required by the wrapper)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter used in the current L‑BFGS run
_GLOBAL_N = None     # length of the sequence (used by the surrogate)

# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Returns an array of length 2·len(x)‑1.
    Uses FFT for speed and clips tiny negatives caused by rounding.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair large values with small ones when the sequence is reversed.
    This simple deterministic heuristic often reduces the peak of the
    self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)          # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large → left side
        new_a[n - 1 - i] = sorted_vals[i]      # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _alternating_high_low(a: np.ndarray) -> np.ndarray:
    """
    Place the sorted values alternately at the leftmost and rightmost
    still‑free positions (largest → leftmost, second largest → rightmost,
    third largest → next left, …).  This spreads large entries across the
    sequence and often flattens the autocorrelation.
    """
    n = a.shape[0]
    sorted_desc = np.sort(a)[::-1]
    new = np.empty_like(a)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            new[left] = val
            left += 1
        else:
            new[right] = val
            right -= 1
    return new


def _fast_eval(a: np.ndarray, n: int) -> float:
    """
    Fast evaluation of the objective for a vector that already satisfies
    sum(a) == 1 (the scaling invariance of ``evaluate_sequence``).  Returns
    the value 2·n·max(conv) which equals the output of ``evaluate_sequence``.
    """
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate of the original objective and its gradient with
    respect to the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε(w) = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k – max)/ε)

    Returns (L, ∇_w L).
    """
    # --- soft‑max to obtain a (numerically stable) ---
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # convolution of a with itself
    conv = _full_convolution(a)                 # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    # log‑sum‑exp is numerically stable; also needed for gradient
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # --- gradient w.r.t. a -------------------------------------------------
    # weights = exp(shifted) / Σ exp(shifted)   → probability distribution over k
    weights = np.exp(shifted - np.max(shifted))   # prevent overflow
    weights /= weights.sum()                     # p_k

    # The derivative of Σ_k p_k * conv_k w.r.t. a is the convolution of a
    # with the reversed vector p (see the derivation in the original code).
    p_rev = weights[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # Extract the slice that corresponds to ∂/∂a_i (size n)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # --- chain rule: ∂L/∂w = ∂L/∂a · ∂a/∂w (soft‑max Jacobian) ---------------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Simple hill‑climb that uses random swaps (20 %) and mass transfers
    (80 %).  Only strict improvements are kept.  The loop stops when the
    time budget expires.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.2:                         # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                          # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_eval(new_a, n)
        if new_val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = new_val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val

    if s < 1e-12:
        a = np.full(n, 1.0 / n)
        s = a.sum()
    a /= s

    # ------------------------------------------------------------------
    # Evaluate the initial candidate (fast evaluation) and keep it as
    # the current best.
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_val = _fast_eval(best_a, n)
    print(f"[init] n={n}, fast evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS minimisation of the smooth surrogate.
    # ------------------------------------------------------------------
    # A schedule of decreasing smoothing parameters (ε).  The smallest
    # ε we allow is 1e‑12 to avoid numerical overflow.
    eps_schedule = np.logspace(np.log10(5e-3), -12, num=10)

    # Soft‑max parameters (log‑space).  Adding a tiny constant avoids log(0).
    w = np.log(best_a + 1e-12)

    restart = 0
    # we stop the L‑BFGS phase when we are within ~30 s of the deadline
    while time.time() < deadline - 30.0 and restart < 12:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        # Randomise the start occasionally to explore new basins.
        if restart % 4 == 0:
            w = np.random.randn(n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            # expose ε and n to the objective wrapper
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n

            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 300,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # Recover the probability vector from the soft‑max parameters
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()

            cur_val = _fast_eval(a_candidate, n)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # occasional deterministic re‑ordering heuristics
        for reorder_func, name in [( _anti_symmetric_reorder, "anti‑sym"),
                                   ( _alternating_high_low, "alt‑hl )")]:
            a_ord = reorder_func(best_a)
            val_ord = _fast_eval(a_ord, n)
            if val_ord < best_val - 1e-12:
                best_val = val_ord
                best_a = a_ord.copy()
                print(f"[{name} reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Simulated annealing with cheap Monte‑Carlo moves.
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_val = best_val
    # initialise a geometric cooling schedule
    T = 0.005 * cur_val             # start temperature (≈0.5 % of the current value)
    cooling = 0.9995                # per‑iteration multiplicative factor

    while time.time() < deadline - 10.0:
        # ----- generate a random move -----------------------------------
        move_type = rng.integers(0, 3)   # 0: swap, 1: mass transfer, 2: perturbation
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        new_a = cur_a.copy()

        if move_type == 0:                     # pure swap
            new_a[i], new_a[j] = new_a[j], new_a[i]

        elif move_type == 1:                   # mass transfer i → j
            if cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a[i] -= delta
            new_a[j] += delta

        else:                                   # small perturbation at i
            factor = 1.0 + (rng.random() - 0.5) * 0.1   # ±5 %
            new_a[i] *= factor

        # project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        new_val = _fast_eval(new_a, n)

        # ----- acceptance test ------------------------------------------
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance with current temperature
            accept = rng.random() < np.exp(-(new_val - cur_val) / T)

        if accept:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[SA] new global best = {best_val:.6f}")

        # temperature decay
        T *= cooling
        # occasional deterministic reorder attempts (keep the annealing fast)
        if rng.random() < 0.005:   # ~0.5 % of the iterations
            a_ord = _anti_symmetric_reorder(cur_a)
            val_ord = _fast_eval(a_ord, n)
            if val_ord < cur_val:
                cur_a = a_ord
                cur_val = val_ord
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = cur_a.copy()
                    print(f"[SA reorder] improved to {best_val:.6f}")

        # optional: try the alternating high/low placement every few thousand steps
        if (int(T * 1e6) % 2000) == 0:
            a_alt = _alternating_high_low(cur_a)
            val_alt = _fast_eval(a_alt, n)
            if val_alt < cur_val:
                cur_a = a_alt
                cur_val = val_alt
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = cur_a.copy()
                    print(f"[SA alt‑hl] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing refinement (strict improvements only)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final verification using the provided ``evaluate_sequence``.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = float('inf')

    # Return the list form expected by the grader.
    return best_a.tolist()
