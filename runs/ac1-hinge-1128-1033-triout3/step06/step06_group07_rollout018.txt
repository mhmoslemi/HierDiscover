"""
Sparse‑support hill‑climbing optimizer.

The evaluation function is scale‑invariant: after fixing Σ a_i = 1 the score is
    2·n·max_k (a*a)[k] .
If the support S = {i : a_i > 0} has size m and we use a uniform weight
    a_i = 1/m  (i ∈ S) , the convolution entries are
    (a*a)[k] = c_k / m²
where c_k is the *ordered* pair count
    c_k = #{(i,j)∈S×S : i+j = k}.
Thus the objective reduces to
    score = 2·n·max(c_k) / m² .
The algorithm searches for a set S (|S| = m) that makes the maximal count
as small as possible.  Starting from random sets we repeatedly
swap a source index (chosen from the current maximal‐sum) with a random
target index, updating the count array incrementally.
If the swap lowers the maximal count we keep it; otherwise we revert.
Multiple restarts and several values of m are tried within the 1000 s
budget.  The best uniform‑weight vector found is returned.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Helpers for the count‑based objective (uniform weights)
# ----------------------------------------------------------------------
def _initial_counts(S, n):
    """
    Build the ordered‑pair count array for a support set S (list of indices).
    Returns an int32 array of length 2·n‑1 where entry k holds
    the number of ordered pairs (i,j)∈S×S with i+j = k.
    """
    counts = np.zeros(2 * n - 1, dtype=np.int32)
    m = len(S)
    # self‑pairs contribute 1
    for i in S:
        counts[2 * i] += 1
    # i<j pairs contribute 2 (i,j) and (j,i)
    for p in range(m):
        i = S[p]
        for q in range(p + 1, m):
            j = S[q]
            counts[i + j] += 2
    return counts


def _apply_removal(counts, S, i):
    """Subtract the contribution of index i from the count array."""
    changed = []
    for s in S:
        idx = i + s
        delta = -1 if s == i else -2
        counts[idx] += delta
        changed.append((idx, delta))
    return changed


def _apply_addition(counts, S_without_i, j):
    """Add the contribution of a new index j (S_without_i is the support after removal)."""
    changed = []
    for s in S_without_i:
        idx = j + s
        counts[idx] += 2
        changed.append((idx, 2))
    # self‑pair of j
    idx = 2 * j
    counts[idx] += 1
    changed.append((idx, 1))
    return changed


def _revert_changes(counts, changed):
    """Undo a list of (index, delta) updates."""
    for idx, delta in changed:
        counts[idx] -= delta


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Searches for a sparse non‑negative step function that minimises
    ``evaluate_sequence``.  The method works entirely with integer
    pair‑counts and only builds the final floating‑point vector once,
    allowing millions of cheap moves inside the 1000 s budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    n = 8000                     # problem size (same as the reference run)
    target_score = 1.503         # we aim to get below this value
    best_score = float('inf')
    best_S = None                # best support set found
    best_m = None                # its size

    # a few sensible support sizes – larger sets give more leeway,
    # smaller sets make the max‑count bound easier to meet.
    candidate_ms = [180, 220, 260, 300, 340, 380, 420, 460]

    # ------------------------------------------------------------------
    # Repeated random restarts + greedy hill‑climbing
    # ------------------------------------------------------------------
    while time.time() < deadline:
        m = random.choice(candidate_ms)
        # initialise a random support of size m
        S = random.sample(range(n), m)
        S_set = set(S)

        counts = _initial_counts(S, n)
        maxc = int(counts.max())
        cur_score = 2.0 * n * maxc / (m * m)

        # keep the globally best solution seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_S = S.copy()
            best_m = m
            print(f"[init] m={m:3d}  maxc={maxc:3d}  score={cur_score:.6f}")

        # target maximal count for this m (derived from the desired score)
        target_maxc = int(math.floor(target_score * m * m / (2.0 * n)))

        # local hill‑climbing – stop after a streak of non‑improving moves
        no_improve = 0
        MAX_STAGNANT = 20000

        while time.time() < deadline and no_improve < MAX_STAGNANT:
            maxc = int(counts.max())
            if maxc <= target_maxc:
                # we already satisfy the target for this support size
                cur_score = 2.0 * n * maxc / (m * m)
                if cur_score < best_score:
                    best_score = cur_score
                    best_S = S.copy()
                    best_m = m
                    print(f"[good] m={m:3d}  maxc={maxc:3d}  score={cur_score:.6f}")
                break

            # index of a currently maximal convolution entry
            k_max = int(counts.argmax())

            # source candidates: indices that actually participate in the maximal sum
            src_candidates = [idx for idx, val in enumerate(S) if (k_max - val) in S_set]
            if src_candidates:
                src_idx = random.choice(src_candidates)
            else:
                src_idx = random.randint(0, m - 1)
            i = S[src_idx]

            improved = False
            # try a handful of random target positions
            for _ in range(5):
                j = random.randint(0, n - 1)
                if j in S_set:
                    continue

                # --- apply the swap (i → j) ---
                # 1) remove i
                changed = _apply_removal(counts, S, i)

                # 2) add j (S without i)
                S_without_i = S[:src_idx] + S[src_idx + 1:]
                changed += _apply_addition(counts, S_without_i, j)

                new_maxc = int(counts.max())

                if new_maxc < maxc:
                    # accept the move
                    S[src_idx] = j
                    S_set.remove(i)
                    S_set.add(j)
                    maxc = new_maxc
                    improved = True
                    no_improve = 0
                    cur_score = 2.0 * n * maxc / (m * m)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_S = S.copy()
                        best_m = m
                        print(f"[step] m={m:3d}  maxc={maxc:3d}  score={cur_score:.6f}")
                    break                # exit the inner trial loop

                # revert the tentative changes
                _revert_changes(counts, changed)

            if not improved:
                no_improve += 1

        # early exit if we already beat the target
        if best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # Build the final floating‑point vector (uniform weight on the best support)
    # ------------------------------------------------------------------
    if best_S is None:
        # fallback – a simple uniform vector
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        a = np.zeros(n, dtype=float)
        a[best_S] = 1.0 / best_m

    # final verification (uses the supplied secure wrapper)
    try:
        final_val = evaluate_sequence(a.tolist())
        print(f"[final] reported score = {final_val:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return a.tolist()
