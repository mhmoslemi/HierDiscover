def project_simplex(v):
    """Projects vector v onto the probability simplex (sum 1, nonnegative)."""
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # find rho
    rho = np.where(u > cssv / np.arange(1, n+1))[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """

    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clamp tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast exact evaluation of the objective defined in ``evaluate_sequence``.
    ``a`` must be non‑negative and already obey the clipping limits (0 ≤ a_i ≤ 1000).
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values on the left and the smallest on the right,
    i.e. produce a deterministic anti‑symmetric ordering.
    """
    n = a.shape[0]
    s = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = s[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = s[i]          # small → right side
    if n % 2:
        new_a[half] = s[half]
    return new_a


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v <= 0):
        # all non‑positive → return uniform distribution
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]        # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.size) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimizes ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # small safety margin
    print(f"[init] total budget = {budget_s}s, deadline @ {deadline:.2f}")

    # ------------------------------------------------------------------
    # Problem size (4000 offers a good trade‑off between flexibility and speed)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load an existing construction if available; otherwise fall back to uniform
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
            print("[init] interpolated previous construction")
        else:
            a = old.copy()
            print("[init] loaded previous construction")
    except Exception:
        a = np.full(n, 1.0 / n)
        print("[init] using uniform start")

    # enforce feasible domain & normalise
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    # anti‑symmetric reordering (a cheap but strong heuristic)
    a = _anti_symmetric_reorder(a)

    best_a = a.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent on the simplex
    # ------------------------------------------------------------------
    phase1_deadline = start + 0.20 * budget_s
    step_size = 0.3               # initial step magnitude
    decay = 0.9997                # exponential decay of the step size
    grad_iter = 0

    while time.time() < phase1_deadline:
        grad_iter += 1
        conv = _fft_convolve_self(best_a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a:
        g = np.zeros_like(best_a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)                 # inclusive
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * best_a[k_max - idx]        # g_i = 2·a_{k_max‑i}

        # make the direction sum‑zero (project onto the tangent plane)
        g = g - np.mean(g)

        # gradient step & projection onto the simplex
        a_new = best_a - step_size * g
        a_new = _project_onto_simplex(a_new)

        new_val = _eval_fast(a_new)
        if new_val < best_val:
            best_val = new_val
            best_a = a_new
            print(f"[grad] iter {grad_iter:6d}  new best = {best_val:.6f}")

        # decay step size; occasional reset to avoid stagnation
        step_size *= decay
        if step_size < 1e-7:
            step_size = 0.3
            # small random jitter to escape possible flat region
            jitter = rng.normal(scale=0.001, size=n)
            a_tmp = best_a + jitter
            a_tmp = np.maximum(a_tmp, 0.0)
            best_a = _project_onto_simplex(a_tmp)

    # ------------------------------------------------------------------
    # Phase 2 – targeted simulated annealing on the permutation + mass moves
    # ------------------------------------------------------------------
    phase2_deadline = start + 0.80 * budget_s
    sa_iter = 0
    while time.time() < phase2_deadline:
        sa_iter += 1

        conv = _fft_convolve_self(best_a)
        max_conv = conv.max()
        k_max = int(conv.argmax())

        # indices that actually contribute to the current worst shift
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        valid_i = np.arange(i_start, i_end + 1)
        contrib = best_a[valid_i] * best_a[k_max - valid_i]

        # pick a "bad" pair (large contribution) – fallback to random if needed
        if valid_i.size > 0:
            # choose among the top‑10 contributors (or all if fewer)
            top_k = min(10, valid_i.size)
            top_idxs = np.argpartition(-contrib, top_k - 1)[:top_k]
            i = valid_i[top_idxs[rng.integers(top_k)]]
            j = k_max - i
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)

        # pick a small‑value index to exchange with
        low_idxs = np.argpartition(best_a, 10)[:10]
        l = low_idxs[rng.integers(low_idxs.size)]

        # ------------------------------------------------------------------
        # 1) swap move (break a large×large pair)
        # ------------------------------------------------------------------
        a_swapped = best_a.copy()
        a_swapped[j], a_swapped[l] = a_swapped[l], a_swapped[j]
        val_swapped = _eval_fast(a_swapped)

        # accept if improvement; otherwise a tiny Metropolis probability
        if val_swapped < best_val or rng.random() < 1e-3:
            best_a = a_swapped
            best_val = val_swapped
            print(f"[SA swap] iter {sa_iter:6d}  new best = {best_val:.6f}")

        # ------------------------------------------------------------------
        # 2) occasional mass‑transfer move (move a fraction from a large to a small entry)
        # ------------------------------------------------------------------
        if rng.random() < 0.25:
            if best_a[i] > 1e-12:
                delta = best_a[i] * 0.05 * rng.random()
                a_transfer = best_a.copy()
                a_transfer[i] -= delta
                a_transfer[l] += delta
                # re‑normalise (project onto simplex) to preserve ℓ₁‑norm
                a_transfer = _project_onto_simplex(a_transfer)
                val_transfer = _eval_fast(a_transfer)
                if val_transfer < best_val:
                    best_a = a_transfer
                    best_val = val_transfer
                    print(f"[SA mass] iter {sa_iter:6d}  new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final cheap hill‑climbing (pure swaps + small transfers)
    # ------------------------------------------------------------------
    refine_deadline = deadline
    refine_iter = 0
    while time.time() < refine_deadline:
        refine_iter += 1
        if rng.random() < 0.20:          # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = best_a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:                           # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or best_a[i] == 0.0:
                continue
            delta = best_a[i] * 0.05 * rng.random()
            cand = best_a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_onto_simplex(cand)

        val = _eval_fast(cand)
        if val < best_val - 1e-12:            # strict improvement
            best_a = cand
            best_val = val
            print(f"[refine] iter {refine_iter:6d}  new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Report final value using the official evaluator (for completeness)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] official evaluation = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = None

    return best_a.tolist()
