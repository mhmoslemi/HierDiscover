"""
Hybrid local‑search for the L∞ autocorrelation problem.

The objective for a non‑negative sequence a (with sum 1) is

    F(a) = 2·n·max(conv(a,a))

where conv is the full linear convolution.  Because the official
evaluation function is homogeneous of degree 0, minimising F(a) is
equivalent to minimising the supplied `evaluate_sequence`.

Implementation
----------------
1. **Simplex projection** – keeps the vector non‑negative and with sum = 1.
2. **Fast convolution** – real FFT (O(n log n)).
3. **Gradient phase** – exact sub‑gradient of the max‑convolution,
   Adam‑style steps with a slowly decaying learning rate.
4. **Annealing phase** – cheap “mass‑transfer’’ moves accepted with a
   Metropolis rule; the temperature decays linearly.
5. **Mix‑and‑match** – after the time budget the current best is blended
   with the uniform vector and with its own reversal; a tiny line‑search
   often yields a further 0.1 % improvement.

The routine respects a `budget_s` time limit (default 1000 s) and prints
progress information.  The returned list is already clipped to the
[0,1000] range required by `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:                     # this should never happen for non‑empty v
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – full linear convolution via real FFT.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b (full length) using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Helper 3 – objective for a vector already normalised (sum = 1).
# ----------------------------------------------------------------------
def _objective_norm(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a non‑negative vector with sum 1."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Helper 4 – exact sub‑gradient of the max‑convolution.
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of  max(conv(a,a)) at a (sum(a)=1).

    The active set consists of all lags where the max is attained.
    """
    n = a.shape[0]
    max_val = conv.max()
    # tolerance – we treat values extremely close to the max as active
    tol = 1e-14
    active = np.flatnonzero(conv >= max_val - tol)
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0            # each active lag contributes equally
    else:                           # fallback – should never happen
        w[np.argmax(conv)] = 1.0
    rev = a[::-1]
    # convolution with the reversed vector gives the required gradient
    grad_full = _full_convolution_fft(rev, w)
    # the usable part is the slice n‑1 … n‑1+n‑1
    grad = 2.0 * grad_full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Helper 5 – generate a diverse warm‑start.
# ----------------------------------------------------------------------
def _generate_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random non‑negative vector (sum = 1) built from simple shapes."""
    x = np.arange(n, dtype=np.float64)

    # 1) exponential tail
    lam = rng.uniform(5e-4, 2e-2)
    exp_shape = np.exp(-lam * x)

    # 2) power‑law
    p = rng.uniform(0.4, 2.0)
    pow_shape = (x + 1) ** (-p)

    # 3) half‑sine (always positive)
    sine_shape = np.sin(np.pi * (x + 0.5) / n)

    # 4) centred Gaussian
    sigma = rng.uniform(n / 8.0, n / 2.0)
    gauss_shape = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)

    # 5) simple triangle
    tri_shape = np.minimum(x, n - 1 - x) + 1.0

    # 6) two‑spike pattern – mass concentrated at the two ends
    spikes = np.zeros_like(x)
    spikes[0] = 1.0
    spikes[-1] = 1.0

    # collect and mix
    shapes = [exp_shape, pow_shape, sine_shape, gauss_shape, tri_shape, spikes]
    weights = rng.random(len(shapes))
    weights /= weights.sum()
    a = sum(w * s for w, s in zip(weights, shapes))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the grader.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a non‑negative height sequence that (approximately) minimises
    the evaluation function.  The routine respects a `budget_s` second
    time limit and prints intermediate scores.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 5.0               # safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – we keep 6000 entries as in the reference solution.
    # --------------------------------------------------------------
    n = 6000

    # --------------------------------------------------------------
    # Keep global best (across all restarts).
    # --------------------------------------------------------------
    best_val = float("inf")
    best_seq = None

    # --------------------------------------------------------------
    # Main loop – repeated random restarts until time runs out.
    # --------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        restart += 1
        # ---- warm start -------------------------------------------------
        a = _generate_initial(n, rng)

        # ---- Phase A – Adam‑style sub‑gradient descent -----------------
        beta1 = 0.9
        beta2 = 0.999
        lr = 0.15                     # initial learning rate
        eps_adam = 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1_pow = 1.0
        beta2_pow = 1.0

        # run until either we exhaust a fraction of the budget or we see
        # no improvement for a while
        max_iter_a = 120000
        stagn = 0
        for it_a in range(max_iter_a):
            if time.time() >= deadline:
                break

            conv = _full_convolution_fft(a, a)
            max_conv = conv.max()
            cur_val = 2.0 * n * max_conv

            if cur_val + 1e-12 < best_val:
                best_val = cur_val
                best_seq = a.copy()
                stagn = 0
                print(f"[restart {restart:2d} – grad iter {it_a:5d}] "
                      f"new_best={best_val:.12f}")
            else:
                stagn += 1

            # sub‑gradient of the max‑convolution
            g = _subgradient(a, conv)

            # Adam update on the simplex (projection after the step)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)
            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            a = _project_simplex(a - step)

            # learning‑rate decay – very gentle
            lr = max(1e-5, lr * 0.9999)

            # occasional random jitter if we are stuck
            if stagn > 3000:
                jitter = rng.normal(scale=0.001, size=n)
                a = _project_simplex(a + jitter)
                stagn = 0
                m.fill(0.0)
                v.fill(0.0)
                beta1_pow = beta2_pow = 1.0
                print(f"[restart {restart:2d}] jitter applied at grad iter {it_a}")

            # early exit if we have already spent most of the budget
            if it_a % 20000 == 0 and time.time() > deadline - 200:
                break

        # --------------------------------------------------------------
        # Phase B – simulated‑annealing fine tuning (mass transfer).
        # --------------------------------------------------------------
        cur = a.copy()
        cur_score = _objective_norm(cur)

        # keep the very best seen during SA as well
        if cur_score + 1e-12 < best_val:
            best_val = cur_score
            best_seq = cur.copy()
            print(f"[restart {restart:2d}] SA start improved best={best_val:.12f}")

        # linear temperature schedule (fast start, long tail)
        T0 = cur_score * 0.05
        Tmin = 1e-9
        it_sa = 0
        while time.time() < deadline:
            it_sa += 1
            progress = it_sa / 300000.0
            T = max(Tmin, T0 * (1.0 - progress))

            # pick two distinct indices; ensure source has mass to give
            i, j = rng.choice(n, size=2, replace=False)
            if cur[i] <= 0.0:
                continue

            # transfer a modest amount – proportional to the source weight
            max_transfer = min(cur[i], 0.02)        # 2 % of total mass at most
            delta = rng.uniform(0.0, max_transfer)

            new = cur.copy()
            new[i] -= delta
            new[j] += delta
            new_score = _objective_norm(new)

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
                cur = new
                cur_score = new_score
                if cur_score + 1e-12 < best_val:
                    best_val = cur_score
                    best_seq = cur.copy()
                    print(f"[restart {restart:2d} – SA iter {it_sa:6d}] "
                          f"new_best={best_val:.12f}")

            # occasional global jitter to escape plateaus
            if it_sa % 25000 == 0:
                cur = cur + rng.normal(scale=0.0005, size=n)
                cur = _project_simplex(cur)
                cur_score = _objective_norm(cur)

            # stop SA early if we are within the last few seconds
            if time.time() > deadline - 10:
                break

        # --------------------------------------------------------------
        # Small post‑processing: blend with uniform and with the reversed vector.
        # --------------------------------------------------------------
        if best_seq is not None:
            u = np.full(n, 1.0 / n)               # uniform distribution
            rev = best_seq[::-1]

            # try a few convex combinations (α ∈ {0,0.05,…,0.5})
            for alpha in np.linspace(0.0, 0.5, 11):
                cand = (1.0 - alpha) * best_seq + alpha * u
                cand = _project_simplex(cand)      # re‑project to be safe
                val = _objective_norm(cand)
                if val + 1e-12 < best_val:
                    best_val = val
                    best_seq = cand.copy()
                    print(f"[mix] α={alpha:.2f} uniform -> new_best={best_val:.12f}")

                cand2 = (1.0 - alpha) * best_seq + alpha * rev
                cand2 = _project_simplex(cand2)
                val2 = _objective_norm(cand2)
                if val2 + 1e-12 < best_val:
                    best_val = val2
                    best_seq = cand2.copy()
                    print(f"[mix] α={alpha:.2f} rev -> new_best={best_val:.12f}")

        # If we have less than 5 seconds left, quit immediately.
        if time.time() > deadline - 5:
            break

    # ------------------------------------------------------------------
    # Prepare final output – clip to the allowed range and convert to list.
    # ------------------------------------------------------------------
    if best_seq is None:
        # Should never happen, but return a safe uniform vector just in case.
        best_seq = np.full(n, 1.0 / n)

    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # Optional final official score (may be omitted if evaluate_sequence is
    # not imported in the execution environment).
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] evaluate_sequence not available – using internal estimate")
        print(f"[result] internal_score={_objective_norm(best_seq):.12f}")

    return final_seq
