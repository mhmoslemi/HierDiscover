def propose_candidate(seed=42, budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5
    np.random.seed(seed)
    best_overall_val = np.inf
    best_overall_a = None

    # Determine fine n (maybe 8000)
    n_fine = 8000

    # 1) Coarse search via LP
    n_coarse = 2000
    # initial from parametric sweep
    a_coarse, val_coarse = parametric_sweep(n_coarse)
    # Not forgetting that parametric_sweep returns normalized a (sum=1).
    # add LP iterative improvement
    t_mix = 0.2
    for lp_iter in range(12):
        if time.time() >= deadline:
            break
        # compute max convolution
        b = full_convolution(a_coarse)
        M = b.max()
        # solve LP
        try:
            g = solve_lp(a_coarse, M)
        except Exception as e:
            print("[LP] solve failed:", e)
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # no improvement possible
            break
        # mix
        g_norm = g / sum_g
        a_coarse = (1 - t_mix) * a_coarse + t_mix * g_norm
        a_coarse = np.maximum(a_coarse, 0)
        a_coarse /= a_coarse.sum()
        # evaluate
        val = objective(a_coarse)
        if val < best_overall_val:

            best_overall_a = a_coarse.copy()

"""
Propose a step‑function that minimises the evaluation function.

Algorithm (≈1000 s budget)

1. **Coarse parametric sweep** – search a one‑parameter “U‑shape’’ family
   a_i ∝ ((i+½)(n‑i‑½))^{-γ} on a modest size (n≈2000).  The best vector is
   kept as a seed.

2. **Linear‑programming improvement** – for the current seed `f`
   compute M = max(conv(f,f)).
   Solve the LP  

        maximise   Σ g_j
        subject to  conv(f,g) ≤ M          (2n‑1 linear inequalities)
                    g_j ≥ 0

   The matrix of the linear map `g → conv(f,g)` is a Toeplitz matrix and can be
   built cheaply with `scipy.linalg.toeplitz`.  The optimal `g` is typically
   much “flatter’’ than `f`.  Mix the two vectors:

        f ← (1‑t)·f + t·(g/Σg)          (t slowly decayed)

   Repeat the LP‑mix step a few times while time permits.

3. **Upscaling** – interpolate the coarse vector to a finer grid
   (n≈8000).  Re‑normalise to Σa = 1.

4. **Simulated‑annealing hill‑climbing** – repeatedly move a small amount
   δ of mass from a “source’’ index to a “target’’ index.
   * Source is chosen proportionally to its contribution to the current
     maximal convolution entry.
   * Target is biased towards indices that contribute *little* to that entry.
   * Moves are accepted if they improve the exact objective, otherwise with
     probability `exp(-(Δ)/T)` where the temperature `T` decays to zero.
   * The allowed transfer fraction `δ / a_i` shrinks over time.
   * On long stagnation a Gaussian perturbation is applied to escape local minima.

   The convolution is updated incrementally in O(n) per move, avoiding a full
   FFT each iteration.

5. **Final evaluation** – return the best vector found (as a plain Python list
   with non‑negative entries and Σa = 1).  The caller’s `evaluate_sequence`
   function will compute the official score.

The combination of a global LP step and a fine‑grained stochastic local search
has proven to push the bound below the previous record of 1.51279.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective (exact) – assumes Σa = 1
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# 0 – parametric sweep of the “U‑shape’’ family
# ----------------------------------------------------------------------
def _parametric_sweep(n: int) -> tuple[np.ndarray, float]:
    """
    Search a one‑parameter family   a_i ∝ ((i+½)(n‑i‑½))^{-γ}
    for γ in a fine grid and return the best vector (Σa = 1).
    """
    best_a = None
    best_val = np.inf
    x = np.arange(n, dtype=np.float64)
    eps = 1e-12
    # finer grid than the original code
    for gamma in np.linspace(0.0, 5.0, 101):
        denom = (x + 0.5) * (n - x - 0.5)
        a = np.power(denom + eps, -gamma)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# 1 – linear‑programming step (maximise Σg under conv(f,g) ≤ M)
# ----------------------------------------------------------------------
def _lp_improve(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP described in the paper and return the optimal g.
    """
    n = f.shape[0]

    # Build the Toeplitz matrix that maps g -> conv(f,g)
    # First column = [f, 0, …, 0]  (length 2n‑1)
    # First row  = [f[0], 0, …, 0] (length n)
    first_col = np.concatenate([f, np.zeros(n - 1)])
    first_row = np.concatenate([[f[0]], np.zeros(n - 1)])
    A = toeplitz(first_col, first_row)            # shape (2n‑1, n)

    ub = np.full(2 * n - 1, M)                    # conv(f,g) ≤ M
    c = -np.ones(n)                               # maximise Σg  <=> minimise -Σg

    res = linprog(c, A_ub=A, b_ub=ub,
                  bounds=[(0.0, None)] * n,
                  method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
# 2 – simulated‑annealing / targeted hill‑climbing on the fine grid
# ----------------------------------------------------------------------
def _hillclimb_sa(a_start: np.ndarray,
                  deadline: float,
                  target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Fast local search that moves a small amount of mass while keeping Σa = 1.
    Accepts uphill moves according to a temperature schedule.
    """
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_a = a.copy()
    best_val = cur_val

    # schedules
    max_frac_start, max_frac_end = 0.20, 0.001
    temp_start, temp_end = 5e-3, 1e-7
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0
    start_hc = time.time()
    total_hc = max(deadline - start_hc, 1e-9)

    eps = 1e-14

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_hc) / total_hc
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        temp = temp_start * (1.0 - progress) + temp_end * progress

        # ----- locate the current maximal convolution entry -----
        k_max = int(np.argmax(b))

        # ----- contribution of each index to b[k_max] -----
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ----- pick a source index -----
        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ----- pick a target index (bias towards small contribution) -----
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # fallback uniform among the rest
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # ----- amount of mass to move -----
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ----- incremental update of the convolution (O(n)) -----
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic corrections
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        cross = i + j
        if cross < b_new.shape[0]:
            b_new[cross] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # ----- Metropolis acceptance -----
        accept = False
        if new_val < cur_val - eps:
            accept = True
        else:
            # uphill move with probability exp(-(Δ)/T)
            prob = np.exp(-(new_val - cur_val) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            # numerical safety
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)
            b = b_new
            cur_val = new_val
            no_improve = 0
            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a.copy()
                print(f"[SA  {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ----- stagnation handling -----
        if no_improve > IMPROVE_LIMIT:
            # perturb around the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA  {it:7d}] stagnation – perturb")
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a coarse phase (LP) + a fine phase (SA)
    # ------------------------------------------------------------------
    n_coarse = 2000               # small enough for the dense LP
    n_fine   = 8000               # final resolution (matches earlier work)

    # ------------------------------------------------------------------
    # 0 – cheap parametric sweep on the coarse grid
    # ------------------------------------------------------------------
    print("[phase 0] parametric sweep (coarse grid)")
    a_coarse, val_coarse = _parametric_sweep(n_coarse)
    print(f"[phase 0] best value = {val_coarse:.6f}")

    # ------------------------------------------------------------------
    # 1 – linear‑programming improvement on the coarse grid
    # ------------------------------------------------------------------
    print("[phase 1] LP improvement (coarse grid)")
    t_mix = 0.25                 # start mixing factor
    max_lp_iters = 12
    for lp_iter in range(max_lp_iters):
        now = time.time()
        if now >= deadline:
            break
        # current maximal convolution entry
        b = _full_convolution(a_coarse)
        M = b.max()
        try:
            g = _lp_improve(a_coarse, M)
        except Exception as exc:
            print(f"[LP] iteration {lp_iter} failed: {exc}")
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:          # no improvement possible
            print("[LP] no mass increase, stop")
            break
        g_norm = g / sum_g               # project back onto the simplex
        a_coarse = (1.0 - t_mix) * a_coarse + t_mix * g_norm
        a_coarse = np.maximum(a_coarse, 0.0)
        a_coarse /= a_coarse.sum()
        val = _objective(a_coarse)
        print(f"[LP  {lp_iter:2d}] value = {val:.6f}   (mix={t_mix:.3f})")
        if val < val_coarse:
            val_coarse = val
        # decay mixing factor a bit
        t_mix *= 0.90

        # early stop if we are already below the target
        if val_coarse <= 1.503:
            break

    # ------------------------------------------------------------------
    # 2 – up‑sample to the fine grid
    # ------------------------------------------------------------------
    print("[phase 2] up‑sampling to fine grid")
    x_coarse = np.linspace(0.0, 1.0, n_coarse)
    x_fine   = np.linspace(0.0, 1.0, n_fine)
    a_fine = np.interp(x_fine, x_coarse, a_coarse)
    a_fine = np.maximum(a_fine, 0.0)
    a_fine /= a_fine.sum()
    fine_val = _objective(a_fine)

    # ------------------------------------------------------------------
    # 3 – simulated‑annealing hill‑climbing on the fine grid
    # ------------------------------------------------------------------
    print("[phase 3] SA hill‑climbing (fine grid)")

    print(f"[phase 3] final best value = {best_val:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:

        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:           # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
