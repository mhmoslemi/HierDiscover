def hill_climb_sa(a, start, deadline):
    n = a.shape[0]
    b = np.convolve(a,a)
    cur_val = 2.0*n* b.max()
    best_a = a.copy()
    best_val = cur_val
    no_improve = 0
    iteration = 0
    # parameters
    max_frac_start = 0.3
    max_frac_end = 0.001
    T0 = 0.01
    T_end = 1e-8

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1 - progress) + max_frac_end * progress
        T = T0 * (1 - progress) + T_end * progress

        # compute k_max
        k_max = int(np.argmax(b))

        # compute contributions
        idx = k_max - np.arange(n)
        mask = (idx >=0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0 and np.random.rand() < 0.9:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = np.random.choice(n, p=a / a.sum())  # or uniform

        # if a[i] very small skip
        if a[i] < 1e-15:
            continue

        # target selection bias to low contribution
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + 1e-12)
        target_weights[i] = 0.0
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # uniform other
            j = np.random.choice([x for x in range(n) if x != i])

        # delta
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0:
            continue

        # incremental update compute new b
        # copy b to temp
        b_new = b.copy()
        # Add contributions as described: shift a
        # shift a at position j adds 2*delta*a to b[j:j+n]
        b_new[j:j+n] += 2.0 * delta * a
        b_new[i:i+n] -= 2.0 * delta * a
        # add pure quadratic terms
        b_new[2*j] += delta * delta
        b_new[2*i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # acceptance
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / T):
            # accept
            a[i] -= delta
            a[j] += delta
            # enforce nonnegativity and normalization (should stay sum 1)
            # because we sub and add same delta, sum stays constant.
            # Clip small negative due to floating errors
            a[i] = max(a[i], 0.0)
            b = b_new
            cur_val = new_val

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA {iteration:7d}] new best = {best_val:.6f}")
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # stagnation handling: if too many failing moves, perturb
        if no_improve >= 200000:
            # random perturbation around best
            a = best_a.copy()
            # add Gaussian noise and reproject
            noise = np.random.normal(scale=0.01, size=n)
            a += noise
            a = np.maximum(a, 0.0)
            total = a.sum()
            if total < 1e-12:
                a = np.full(n, 1.0/n)
                total = 1.0
            a /= total
            b = np.convolve(a, a)
            cur_val = 2.0*n * b.max()
            no_improve = 0
            print(f"[SA {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

"""
Propose a step‑function with a low value of
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Deterministic U–shape initialization** –  Scan a one‑parameter family
   a_i ∝ ((i+½)(n‑i‑½) + eps)^{-γ} (γ∈[0,6]) and keep the best shape.

2. **Simulated‑annealing hill‑climbing** –  Repeatedly move a small amount of
   mass from an index that contributes heavily to the current maximal
   convolution entry towards an index that contributes little.  Moves are
   accepted either if they improve the exact objective or with a probability
   exp(-(Δ)/T) where the temperature T slowly cools.  Stagnation triggers a
   random perturbation of the best known solution.

3. **Final evaluation** –  The best sequence seen before the deadline is
   returned (as a Python list).  All entries are non‑negative and sum to 1,
   so the security wrapper `evaluate_sequence` never rejects the answer.

The implementation works for any length; a modest size (n≈2000) is used
to keep each iteration cheap, allowing many thousands of iterations within
the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)-1)."""
    return np.convolve(a, a)


def objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    b = full_convolution(a)
    return 2.0 * n * b.max()


def initialise_u_shape(n: int):
    """
    Scan a one‑parameter U‑shape family a_i ∝ ((i+½)(n‑i‑½))^{-γ}
    for γ in [0,6] and return the best vector (sum normalised to 1).
    """
    best_a = None
    best_val = np.inf
    eps = 1e-12
    x = np.arange(n, dtype=np.float64)
    # fine grid: first coarse, then a refined pass around the best γ
    gammas = np.linspace(0.0, 6.0, 121)          # step 0.05
    for gamma in gammas:
        denom = (x + 0.5) * (n - x - 0.5)
        a = np.power(denom + eps, -gamma)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
    # one extra local refinement around the best γ
    if best_a is not None:
        # locate the best gamma roughly
        best_gamma = gammas[np.argmin([objective_value(np.power(((x + 0.5) *
                                   (n - x - 0.5) + eps), -g) /
                                   np.power(((x + 0.5)*(n - x - 0.5) + eps), -g).sum())
                                   for g in gammas])]
        for gamma in np.linspace(max(0.0, best_gamma - 0.2),
                                 best_gamma + 0.2, 41):
            denom = (x + 0.5) * (n - x - 0.5)
            a = np.power(denom + eps, -gamma)
            s = a.sum()
            a /= s
            val = objective_value(a)
            if val < best_val:
                best_val, best_a = val, a.copy()
    return best_a, best_val


def simulated_annealing(a_start: np.ndarray,
                        start_time: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing with a temperature schedule.
    Moves transfer a small amount of mass from a source index (chosen
    proportionally to its contribution to the present maximal convolution
    entry) to a target index (biased towards a low contribution).
    """
    a = a_start.copy()
    n = a.shape[0]
    b = full_convolution(a)                     # length 2·n‑1
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val = cur_val

    # annealing / move schedules
    max_frac_start, max_frac_end = 0.60, 0.02   # biggest allowed mass transfer
    T_start, T_end = best_val * 0.05, 1e-8       # temperature (≈5 % of current value)

    # stagnation handling
    stagnation_limit = 200_000
    no_improve = 0
    iteration = 0
    eps = 1e-12

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear schedules
        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        T = T_start * (1.0 - progress) + T_end * progress

        # index of the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to that entry
        idx = k_max - np.arange(n)                # idx[i] = k_max - i
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ----- choose source index (biased by contribution) -----
        if total_contrib > 0 and np.random.rand() < 0.9:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            # fallback uniform selection weighted by a (larger a easier to move)
            prob = a / a.sum()
            i = np.random.choice(n, p=prob)

        if a[i] <= eps:
            # nothing to take from this index
            continue

        # ----- choose target index (biased towards low contribution) -----
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0                 # avoid self‑transfer
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # uniform amongst the rest
            cand = list(range(n))
            cand.remove(i)
            j = np.random.choice(cand)

        # amount of mass to move (proportional to what is available at i)
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ----- incremental update of the convolution (O(n)) -----
        b_new = b.copy()
        # shift of a by j (adds 2·δ·a) and by i (removes 2·δ·a)
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # ----- acceptance test (SA) -----
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # uphill move accepted with probability exp(-Δ/T)
            prob = math.exp(-(new_val - cur_val) / max(T, 1e-16))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # apply the move permanently
            a[i] -= delta
            a[j] += delta
            # guard against tiny negative numbers due to rounding
            a[i] = max(a[i], 0.0)

            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA  {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= 1.503:
                    # target reached – we can stop early
                    break
        else:
            no_improve += 1

        # ----- stagnation handling -----
        if no_improve >= stagnation_limit:
            # re‑seed around the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a += noise
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum < eps:
                a = np.full(n, 1.0 / n)
                a_sum = 1.0
            a /= a_sum
            b = full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA  {iteration:7d}] stagnation – random perturbation")
        iteration += 1

    return best_a, best_val

            # interpolate to the new size
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_len)
            init_a = np.interp(new_x, old_x, old)
        else:
            init_a = old.copy()
        init_a = np.maximum(init_a, 0.0)
        init_a /= init_a.sum()
        print("[init] loaded previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – deterministic sweep of the U‑shape family.
    # ------------------------------------------------------------------
    best_a, best_val = None, np.inf
    try:
        shape_a, shape_val = initialise_u_shape(n)
        best_a, best_val = shape_a, shape_val
        print(f"[init] shape sweep: value={best_val:.6f}")
    except Exception as exc:
        print("[init] shape sweep failed:", exc)

    # If an older solution was available, compare it with the shape sweep.
    if init_a is not None:
        val = objective_value(init_a)
        if val < best_val:
            best_a, best_val = init_a, val
            print(f"[init] interpolated previous solution better: {best_val:.6f}")

    # Fallback if nothing succeeded (should never happen)
    if best_a is None:
        a = np.random.rand(n)
        a /= a.sum()
        best_a, best_val = a, objective_value(a)
        print("[init] random start, value=", best_val)

    # ------------------------------------------------------------------
    # Phase A – simulated‑annealing hill‑climbing.
    # ------------------------------------------------------------------
    best_a, best_val = simulated_annealing(best_a,
                                           start_time=start_time,
                                           deadline=deadline)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
