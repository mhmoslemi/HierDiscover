def beta_shape(alpha, n):
    x = (np.arange(n) + 0.5) / n
    # compute x^{alpha-1} (if alpha-1 <0, x^neg = 1/x^{abs}
    # use np.power with where to avoid zero division.
    a = np.exp((alpha-1) * np.log(x + 1e-12) + (alpha-1) * np.log(1 - x + 1e-12))
    # Actually for negative exponent we can also compute using np.power.
    a = (x**(alpha-1)) * ((1 - x)**(alpha-1))
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0/n)
    return a / a_sum

```python
"""
Optimisation of a non‑negative step function to minimise
`evaluate_sequence`.  The algorithm combines:
  1) a quick parametric sweep (uniform, linear, quadratic, sinusoidal,
     exponential, Beta‑law and power‑law families) to obtain a good
     initial point;
  2) a projected Adam optimiser on a smooth approximation of the
     max‑convolution; the iterate is forced to be symmetric which
     is known to be optimal for the autocorrelation problem;
  3) a targeted hill‑climber and a short simulated‑annealing phase
     (identical to the reference implementation) to polish the
     solution.
All phases respect the global time budget and the final vector is
symmetrised and normalised before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple parametric initialisers (uniform, linear, …)
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starting vector and its objective value."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    # linear decreasing
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # quadratic decreasing
    a = (np.arange(n, 0, -1, dtype=float) ** 2)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # sine half‑wave
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # exponential tail
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    return best_a, best_val


# ----------------------------------------------------------------------
# Additional families for the parametric sweep
# ----------------------------------------------------------------------
def _beta_shape(alpha: float, n: int) -> np.ndarray:
    """Symmetric Beta(α,α) density on a discrete grid, Σ = 1."""
    # avoid log(0) by clipping to a tiny epsilon
    eps = 1e-12
    x = (np.arange(n, dtype=float) + 0.5) / n
    x = np.clip(x, eps, 1.0 - eps)
    log_a = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    # protect against overflow – cap the exponent
    log_a = np.clip(log_a, -700, 700)          # np.exp(709) ≈ 1e308
    a = np.exp(log_a)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _power_shape(p: float, n: int) -> np.ndarray:
    """Symmetric law a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=float)
    a = (i + 1.0) ** (-p) + (n - i) ** (-p)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Phase A – Adam optimisation on a smooth max‑convolution,
#            the iterate is symmetrised at every step.
# ----------------------------------------------------------------------
def _adam_sym(a0: np.ndarray,
              n: int,
              start: float,
              deadline: float,
              best_a: np.ndarray,
              best_val: float,
              target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected Adam on a log‑soft‑max parametrisation with a symmetry constraint."""
    w = np.log(a0 + 1e-12)                 # logits, no explicit simplex constraint

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.002
    eps_start, eps_end = 0.5, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 20000

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---------- obtain a on the simplex ----------
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        # enforce symmetry – this also preserves the simplex property
        a = (a + a[::-1]) * 0.5

        # ---------- exact objective ----------
        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ---------- schedules ----------
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ---------- smooth‑max weights ----------
        b = _full_convolution(a)
        max_b = b.max()
        shifted = (b - max_b) / eps
        # improve numerical stability
        shifted = shifted - np.max(shifted)
        wgt = np.exp(shifted)
        p = wgt / wgt.sum()                     # “soft‑max” over convolution positions

        # ---------- gradient w.r.t. a ----------
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)           # length 2·n‑1
        segment = conv_ab[n - 1: 2 * n - 1]      # n entries that correspond to ∂c/∂a
        grad_a = 4.0 * n * segment[::-1]

        # ---------- chain rule through the soft‑max ----------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---------- Adam update ----------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # restart if we have been idle for a while
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climber (unchanged, works on the exact objective)
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Local search that transfers a tiny amount of mass from indices that
    contribute most to the current maximal convolution entry to indices that
    contribute little."""
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 180_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the current maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – bias towards large contributions
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – 30 % of the time choose a location that has a
        # *small* contribution to the maximal entry, otherwise uniform.
        if np.random.rand() < 0.30:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            # inverse weighting (favor small values)
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = np.random.choice(n, p=probs_j)
            else:

                if j >= i:
                    j += 1
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac

        # The update formula is taken from the reference implementation.
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a

        b_new[2 * i] += delta * delta          # self‑term at i (removed mass)
        b_new[i + j] -= 2.0 * delta * delta    # cross term

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-14:

            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:

                print(f"[Hill {it:7d}] new best = {best_val:.6f}")

        else:

        if no_improve > IMPROVE_LIMIT:

            cur_val = 2.0 * n * b.max()

        it += 1

"""
Optimisation of a non‑negative step function to minimise
`evaluate_sequence`.

The algorithm proceeds in four stages:

1️⃣  A quick parametric sweep (uniform, linear, quadratic, sinusoidal,
    exponential, *beta* and *power‑law* families) to obtain a solid
    initial point.

2️⃣  A symmetric Adam optimiser on a smooth approximation of
    `max(conv(a,a))`.  The iterate is forced to be symmetric
    (a_i = a_{n-1-i}) after every soft‑max projection – symmetry is
    known to be optimal for the autocorrelation problem.

3️⃣  A targeted hill‑climber that repeatedly moves a tiny amount of mass
    from indices that contribute most to the current maximal
    convolution entry to indices that contribute little.

4️⃣  A short simulated‑annealing phase to occasionally accept worsening
    moves and escape plateaus.

All phases respect the global time budget (default 1000 s).  The final
vector is symmetrised, normalised to Σ = 1 and returned as a list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple parametric initialisers (uniform, linear, …)
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a decent starting vector and its objective value."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    # linear decreasing
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # quadratic decreasing
    a = (np.arange(n, 0, -1, dtype=float) ** 2)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # sine half‑wave
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # exponential tail
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    return best_a, best_val


# ----------------------------------------------------------------------
# Additional families for the parametric sweep
# ----------------------------------------------------------------------
def _beta_shape(alpha: float, n: int) -> np.ndarray:
    """Symmetric Beta(α,α) discretisation, Σ = 1."""
    eps = 1e-12
    x = (np.arange(n, dtype=float) + 0.5) / n
    x = np.clip(x, eps, 1.0 - eps)
    # log‑density to avoid overflow / underflow
    log_a = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    log_a = np.clip(log_a, -700, 700)               # np.exp(709) ≈ 1e308
    a = np.exp(log_a)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _power_shape(p: float, n: int) -> np.ndarray:
    """Symmetric law a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=float)
    a = (i + 1.0) ** (-p) + (n - i) ** (-p)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Phase A – Adam optimisation on a smooth max‑convolution,
#            the iterate is symmetrised at every step.
# ----------------------------------------------------------------------
def _adam_sym(a0: np.ndarray,
              n: int,
              start: float,
              deadline: float,
              best_a: np.ndarray,
              best_val: float,
              target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Projected Adam on a log‑soft‑max parametrisation with a symmetry
    constraint."""
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.002
    eps_start, eps_end = 0.5, 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 20000

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- soft‑max to obtain a on the simplex -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        a = (a + a[::-1]) * 0.5               # enforce symmetry

        # ----- exact objective for tracking -----
        cur_val = _objective_value(a)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- schedules -----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth‑max weights (log‑sum‑exp) -----
        b = _full_convolution(a)
        max_b = b.max()
        shifted = (b - max_b) / eps
        shifted -= np.max(shifted)            # numerical stabilisation
        wgt = np.exp(shifted)
        p = wgt / wgt.sum()

        # ----- gradient w.r.t. a (smooth surrogate) -----
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 2·n‑1
        segment = conv_ab[n - 1: 2 * n - 1]    # n entries
        grad_a = 4.0 * n * segment[::-1]

        # ----- chain rule through the soft‑max -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # restart if stagnating
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence` within the given time budget.

    The routine consists of a parametric sweep, a symmetric Adam optimiser,
    a targeted hill‑climber and a short simulated‑annealing phase.
    The returned vector is normalised to Σ = 1 and symmetrised.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 offers a good trade‑off between resolution and
    # evaluation cost.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (including Beta‑ and power‑law families).
    # ------------------------------------------------------------------
    best_a, best_val = _initial_candidate(n)

    # Beta(α,α) family
    for alpha in np.linspace(0.3, 4.0, 30):
        a = _beta_shape(alpha, n)
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Sweep] Beta α={alpha:.3f} new best = {best_val:.6f}")

    # Power‑law family
    for p in np.linspace(0.5, 3.0, 30):
        a = _power_shape(p, n)
        val = _objective_value(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Sweep] Power p={p:.3f} new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam phase with symmetry.
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + (deadline - start_time) * 0.45
    best_a, best_val = _adam_sym(best_a, n,
                                 start=start_time,
                                 deadline=phase_a_deadline,
                                 best_a=best_a,
                                 best_val=best_val,
                                 target_val=1.503)

    # ------------------------------------------------------------------
    # 3️⃣  Hill‑climber to polish the solution.
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + (deadline - start_time) * 0.80
    best_a = _hillclimb_targeted(best_a, n,
                                 start=phase_a_deadline,
                                 deadline=phase_b_deadline,
                                 best_a=best_a,
                                 best_val=best_val,
                                 target_val=1.503)

    # ------------------------------------------------------------------
    # 4️⃣  Short simulated annealing to escape possible plateaus.
    # ------------------------------------------------------------------
    best_a, best_val = _annealing_phase(best_a, n,
                                        start=phase_b_deadline,
                                        deadline=deadline,
                                        best_a=best_a,
                                        best_val=best_val,
                                        target_val=1.503)

    # ------------------------------------------------------------------
    # Final symmetrisation (counters numerical drift) and normalisation.
    # ------------------------------------------------------------------
    best_a = (best_a + best_a[::-1]) * 0.5
    best_a = best_a / best_a.sum()

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper).
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
