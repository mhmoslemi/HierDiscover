"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.

Algorithm
========
1. **Initialisation** – start from the previous best sequence if available,
   otherwise take a uniform distribution.  The vector is projected onto the
   probability simplex (non‑negative entries with unit sum).

2. **Two‑phase smooth optimisation**  
   • *Phase 1* (≈70 % of the time budget) – minimise the L²‑deviation of the
     autocorrelation from its mean.  The gradient is
       g = 4 · corr(b‑μ, a)
     where *b* is the full autocorrelation of *a* and μ = Σa²/(2n‑1).  
   • *Phase 2* – gradually switch to a smooth approximation of the max‑norm:
       wₖ ∝ exp(γ·(bₖ–max b)),   γ slowly increased up to γₘₐₓ.
     The gradient becomes
       g = 2 · corr(w, a).
   In both phases the update is a projected gradient step with a diminishing
   step size η₀/√t and simplex projection.

3. **Escaping plateaus** – after a long streak without improvement a small
   Gaussian perturbation is added and the current iterate is re‑projected.

4. **Guided hill‑climbing** – once the time for the smooth phase is over a
   short stochastic “mass‑transfer” hill‑climber (the same routine used in
   the previous state‑of‑the‑art code) attempts to push the maximum
   convolution value down further.

5. **Return** – the best sequence found (as a plain list) is returned.
   The final score is printed via ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative round‑off errors
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Correlation (full) via FFT – returns the first n entries reversed,
# i.e.  g_i = Σ_k w_k a_{k-i}.
# ----------------------------------------------------------------------
def _correlation_first_n(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the correlation of w (length 2n‑1) with a (length n) and
    return the first n values needed for the gradient:
        g_i = Σ_k w_k a_{k-i},  i=0…n‑1 .
    """
    n = a.shape[0]
    size = _next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    corr_full = np.fft.irfft(fw * np.conj(fa), size)[: len(w) + n - 1]
    # First n entries (indices 0 … n‑1) correspond to shifts  –(n‑1) … 0 ;
    # we need them in reverse order.
    return corr_full[:n][::-1]


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing that moves a little mass away from the lag(s)
    that attain the current maximal convolution value.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-14)
        k = int(rng.choice(ks))

        # locate the pair (i,j) that contributes most to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # pick two destination indices that have below‑average mass
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # feasible mass transfer
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds,
    first performing a smooth projected‑gradient descent (with a variance‑
    reduction phase followed by a smoothed max phase) and finally a short
    guided hill‑climbing refinement.  The best sequence found is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a lower theoretical bound.
    # 8000 offers a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from a previously known construction if available.
    # ------------------------------------------------------------------
    a0 = None
    try:
        # ``height_sequence_1`` is a global that may be defined by the runner.
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
    except Exception:
        pass

    if a0 is None or a0.ndim != 1:
        a = np.full(n, 1.0 / n, dtype=float)
    else:
        if a0.shape[0] != n:
            # Linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        a = np.maximum(a0, 0.0)
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)

    # keep everything on the simplex (sum = 1)
    a = _simplex_projection(a)

    # ------------------------------------------------------------------
    # Helpers for the optimisation loop
    # ------------------------------------------------------------------
    it = 0
    no_improve = 0
    best_a = a.copy()
    # true objective (since sum = 1) is 2·n·max(conv)
    conv = _full_convolution(a)
    best_val = 2.0 * n * conv.max()
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # Phase‑switch: first 70 % of the budget is variance reduction,
    # remaining 30 % for smoothed‑max optimisation.
    phase1_end = start_time + 0.70 * budget_s

    # learning‑rate schedule
    eta0 = 0.6          # initial step‑size amplitude

    # smoothing schedule (for phase 2)
    gamma = 0.0
    gamma_max = 15.0
    gamma_inc = 0.0007   # per iteration in phase 2

    while time.time() < deadline - 5.0:
        it += 1
        # ----- compute full autocorrelation -----
        conv = _full_convolution(a)
        max_b = conv.max()
        cur_val = 2.0 * n * max_b      # exact objective (sum(a)=1)

        # ----- keep the best ever seen -----
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[iter {it}] new best = {best_val:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # ----- decide which gradient to use -----
        # Phase‑1: minimise L²‑deviation from the mean.
        if time.time() < phase1_end:
            # mean of the convolution (equals sum(a)²/(2n‑1))
            sum_a = a.sum()
            avg = (sum_a * sum_a) / (2 * n - 1)
            diff = conv - avg
            grad = 4.0 * _correlation_first_n(diff, a)
        else:
            # Phase‑2: smooth approximation of the max.
            if gamma < gamma_max:
                gamma = min(gamma_max, gamma + gamma_inc)
            # softmax weights (stable version)
            shift = conv.max()
            exp_w = np.exp(gamma * (conv - shift))
            w = exp_w / exp_w.sum()
            grad = 2.0 * _correlation_first_n(w, a)

        # ----- projected gradient step -----
        step = eta0 / np.sqrt(it)
        cand = a - step * grad
        cand = _simplex_projection(cand)

        # ----- optional escape from plateaus -----
        if no_improve >= 4000:
            # Apply a gentle Gaussian jitter around the current best.
            pert = rng.normal(scale=0.015, size=n)
            cand = _simplex_projection(best_a + pert)
            no_improve = 0

        a = cand

    # ------------------------------------------------------------------
    # Final guided hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # ------------------------------------------------------------------
    # Final reporting and return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    return best_a.tolist()
