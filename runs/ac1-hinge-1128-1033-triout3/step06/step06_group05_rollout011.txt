def _smooth_phase(a, n, deadline, best_a, best_val):
    start = time.time()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    it = 0
    while time.time() < deadline:
        b = _full_convolution_self(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth {it:8d}] new best = {best_val:.6f}")

        # compute softmax smoothing parameter alpha depending on progress
        progress = (time.time() - start) / (deadline - start)
        # alpha start low then increase
        # start from 1 to 500
        alpha = 1.0 + 500.0 * progress

        # Compute p softmax
        # To avoid overflow, shift by max
        scaled = alpha * b
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        p = exp_scaled / np.sum(exp_scaled)

        # compute gradient
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)  # full conv length 3n-2
        grad = 2.0 * conv_rev[n-1:2*n-1]

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        # step size schedule: lr decays from 0.05 to 0.005
        lr = 0.05 * (1 - progress) + 0.005 * progress

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        # enforce non-negativity and symmetry
        a_new = np.maximum(a_new, 0.0)
        a_new = (a_new + a_new[::-1]) / 2.0

        # project onto simplex
        a_new = _project_simplex(a_new)

        a = a_new
        it += 1
    return best_a, best_val

"""
Propose a step‑function that minimises ``evaluate_sequence``.

The optimiser works in two stages.

Stage A – *smooth max* Adam descent
------------------------------------
*   The true objective is 2·n·max(a∗a) with Σa=1.
*   We replace the hard max by a smooth approximation
    f(a)= (1/α)·log Σ_k exp(α·b_k)  with  b=a∗a .
    As α grows the smooth function approaches the hard max,
    but its gradient is everywhere defined.
*   Using the identity
        ∂b_k/∂a_i = a_{k‑i}+a_{k‑i}  (2·a_{k‑i} at the centre)
    the gradient of f can be written as a single convolution:
        g = 2·conv( a_rev , p )[n‑1:2·n‑1],
    where a_rev is a reversed copy of a and
          p_k = exp(α·b_k) / Σ_j exp(α·b_j)  (soft‑max weights).
*   The update follows the Adam optimiser, a step size that decays
    linearly and a projection onto the probability simplex.
*   The sequence is kept exactly symmetric (a_i = a_{n‑1‑i}) after
    every update – symmetry is known to be favourable for flattening
    the autocorrelation.

Stage B – deterministic hill‑climbing
------------------------------------
When the smooth phase stagnates we switch to a greedy local search.
At the current worst convolution index ``k`` we move a small amount
δ of mass from the index that contributes most to ``b_k`` to the
index that contributes least.  The full convolution is updated in
O(n) time, the move is accepted only if it reduces the true
objective, and periodic Gaussian perturbations escape possible
plateaus.

The whole routine respects the 1000 s wall‑time limit, prints
intermediate bounds and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, exact for real vectors)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _full_convolution_self(a: np.ndarray) -> np.ndarray:
    """Full autocorrelation a∗a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:                           # numerical safety
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase A – Smooth‑max Adam descent
# ----------------------------------------------------------------------
def _smooth_phase(a: np.ndarray,
                  n: int,
                  deadline: float,
                  best_a: np.ndarray,
                  best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimisation of the soft‑max approximation of max(b)."""
    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow, beta2_pow = 1.0, 1.0

    start = time.time()
    it = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1.  current objective (hard max) – for bookkeeping only
        # --------------------------------------------------------------
        b = _full_convolution_self(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth {it:7d}] new best = {best_val:.6f}")

        # --------------------------------------------------------------
        # 2.  smooth max – soft‑max weights (progress‑dependent α)
        # --------------------------------------------------------------
        prog = (time.time() - start) / (deadline - start)   # 0 → 1
        alpha = 1.0 + 500.0 * prog                          # grow from 1 to 501

        scaled = alpha * b
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)          # shift for stability
        p = exp_scaled / np.sum(exp_scaled)               # soft‑max (size 2n‑1)

        # --------------------------------------------------------------
        # 3.  gradient of the smooth objective
        # --------------------------------------------------------------
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, p)                    # length 3·n‑2
        grad = 2.0 * conv_rev[n - 1 : 2 * n - 1]           # slice → length n

        # --------------------------------------------------------------
        # 4.  Adam update
        # --------------------------------------------------------------
        beta1_pow *= beta1
        beta2_pow *= beta2

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # learning‑rate schedule: 0.05 → 0.005
        lr = 0.05 * (1.0 - prog) + 0.005 * prog
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        # --------------------------------------------------------------
        # 5.  enforce feasibility (non‑negative, symmetric, Σ=1)
        # --------------------------------------------------------------
        a_new = np.maximum(a_new, 0.0)                     # non‑negative
        a_new = (a_new + a_new[::-1]) / 2.0                # enforce symmetry
        a_new = _project_simplex(a_new)                    # Σ=1, stays non‑negative

        a = a_new
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Greedy hill‑climbing on the true objective
# ----------------------------------------------------------------------
def _greedy_hillclimb(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> np.ndarray:
    """
    Local search moving a small amount of mass from the source index that
    contributes most to the current maximum convolution value to a target
    index that contributes the least.  Full convolution is updated
    incrementally (O(n)) after each trial move.
    """
    b = _full_convolution_self(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 150_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        k = max_idx

        # ----- contribution of each a_i to b_k (the worst entry) -----
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]   # diagonal term

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        # safeguard: avoid degenerate source==target
        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-14:
            # pick a random non‑zero source
            nz = np.where(a > 1e-14)[0]
            if nz.size == 0:
                break
            source = int(np.random.choice(nz))

        # move at most 30 % of the source mass
        delta = a[source] * 0.30 * np.random.rand()
        if delta <= 0.0:
            continue

        # ----- incremental update of the full convolution -----
        b_new = b.copy()
        # the cross terms 2·δ·a
        b_new[target : target + n] += 2.0 * delta * a
        b_new[source : source + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            # accept the move
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # occasional perturbation after long stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.004, size=n)   # tiny noise
            a = np.maximum(a, 0.0)
            a = (a + a[::-1]) / 2.0
            a = _project_simplex(a)
            b = _full_convolution_self(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")
        it += 1
    return best_a
