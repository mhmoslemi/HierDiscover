"""
search_step_function.py

A three‑stage optimizer for the “step‑function” objective
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))² .

Stage 0 – a cheap random sweep over a two‑parameter Beta‑like family
          a_i ∝ (i+½+δ)^α·(n−i−½+δ)^β .
          The best shape found is kept as the seed.

Stage 1 – exponentiated‑gradient descent on the exact max–convolution.
          For the current maximal convolution index kmax,
          the sub‑gradient of max(conv) is
                g_i = 2·a_{kmax−i}  (if 0≤kmax−i<n, else 0).
          The update a ← a·exp(−η·g) followed by a simplex
          renormalisation drives mass away from indices that
          contribute most to the current peak.

Stage 2 – fast targeted hill‑climbing.
          In each move a small amount δ is transferred from the
          index that contributes the most to the current maximal
          convolution entry to a low‑mass index.  The convolution
          is updated incrementally (O(n)) and a move is kept only
          if it strictly lowers the exact objective.  Stagnation is
          broken by a brief random perturbation.

All stages respect the 1000 s overall budget, print progress and
finally return the best sequence found (its sum is 1, any scaling
would give the same score).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum is assumed to be 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Parametric family – Beta‑type shape
# ----------------------------------------------------------------------
def _beta_shape(
    n: int,
    alpha: float,
    beta: float,
    offset: float = 0.0,
    eps: float = 1e-12,
) -> np.ndarray:
    """
    a_i ∝ (i+½+offset)^alpha·(n−i−½+offset)^beta .
    The vector is normalised to have sum = 1 and no entry < eps.
    """
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5 + offset
    right = (n - i - 0.5) + offset
    a = np.power(left, alpha) * np.power(right, beta)
    a = np.maximum(a, eps)          # avoid exact zeros
    a_sum = a.sum()
    if a_sum == 0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (exact, not smoothed)
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at the current maximizing index k_max.
    For 0 ≤ i < n:
        g_i = 2·a_{k_max−i}   if 0 ≤ k_max−i < n, else 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    idx = np.arange(lo, hi + 1)
    partner = k_max - idx
    g[idx] = 2.0 * a[partner]
    return g


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient descent on the exact max‑conv.
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    target_val: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Keeps the best vector seen before *deadline*.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    t = 0
    eta0 = 0.5

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current maximal convolution entry
        b = _full_convolution(a)
        k_max = int(np.argmax(b))
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            if best_val <= target_val:
                break

        # sub‑gradient at the current peak
        g = _subgradient(a, k_max)

        t += 1
        eta = eta0 / np.sqrt(t)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:            # safeguard (should not happen)
            a = best_a.copy()
            s = a.sum()
        a /= s                 # project onto Σa = 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – targeted hill‑climbing
# ----------------------------------------------------------------------
def _hill_climb(
    a_start: np.ndarray,
    start: float,
    deadline: float,
    target_val: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Fast local improvement.  Each step moves a small amount of mass
    from the index that contributes most to the current maximal
    convolution entry to a low‑mass index.  The convolution is updated
    incrementally (O(n)).  The best vector seen is returned.
    """
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a, best_val = a.copy(), cur_val

    max_frac_start, max_frac_end = 0.60, 0.02
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    iteration = 0
    eps = 1e-12

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear schedule for the maximal allowed transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate current maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each (i, k_max‑i) pair to that entry
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        idx = np.arange(lo, hi + 1)
        partner = k_max - idx
        contrib = a[idx] * a[partner]

        # ----- source index (largest contribution) -----
        if contrib.size > 0:
            source = idx[np.argmax(contrib)]
        else:
            source = int(np.argmax(a))

        if a[source] <= eps:
            continue

        # ----- target index (bias to low mass) -----
        target_candidates = np.arange(n)
        target_candidates = target_candidates[target_candidates != source]
        # probability ∝ 1/(a+eps) → prefers tiny entries
        probs = 1.0 / (a[target_candidates] + eps)
        probs /= probs.sum()
        target = np.random.choice(target_candidates, p=probs)

        # amount of mass to move
        delta = a[source] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        # the cross‑terms
        b_new[target : target + n] += 2.0 * delta * a
        b_new[source : source + n] -= 2.0 * delta * a
        # the pure quadratic terms
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # accept only strictly improving moves
        if new_val < cur_val - 1e-14:
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[HillClimb {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – random perturbation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillClimb {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s
    best_a, best_val = _exp_gradient_descent(
        best_a,
        deadline=phase1_deadline,

    )
    print(f"[Phase1] best value = {best_val:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    best_a, best_val = _hill_climb(

    # --------------------------------------------------------------

    final_score = evaluate_sequence(best_a.tolist())
