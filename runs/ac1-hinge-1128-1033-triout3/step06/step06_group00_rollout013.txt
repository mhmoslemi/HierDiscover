    size = a.size + b.size - 1

"""
Propose a step‑function (non‑negative sequence) that (approximately) minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation follows three stages within the 1000 s budget:

1. **Warm‑start** – generate a pool of classic shapes (exponential,
   power‑law, Gaussian, half‑sine, random convex combinations), keep the
   best one (sum normalised to 1).

2. **Linear‑program mixing** – repeatedly solve a tiny LP
   (variables = the sequence, ‑∞‑norm of the convolution as a linear bound)
   to obtain a direction that increases the total mass while keeping the
   convolution ≤ the current worst value.  The new direction is mixed
   into the current sequence with a decreasing step size.  This is the
   “alternating LP” idea from the statement and converges much faster
   than pure gradient descent.

3. **Fine tuning** – exponentiated‑gradient descent on the simplex
   (a mirror‑descent step that uses the exact sub‑gradient of the max‑convolution)
   followed by a simple simulated‑annealing mass‑transfer to escape possible
   plateaus.

The code uses NumPy and SciPy (the latter only for the LP; if SciPy is
missing the algorithm falls back to the gradient+SA part).  All intermediate
sequences are kept on the probability simplex, so the scaling invariance of
the objective is respected automatically.  The best sequence found is
clipped to `[0, 1000]` before being returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (LP phase).  If unavailable we just skip that phase.
# ----------------------------------------------------------------------
try:
    import scipy.optimize
    import scipy.sparse
    _HAVE_SCIPY = True
except Exception:  # pragma: no cover
    _HAVE_SCIPY = False

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al. 2008).
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:                                     # all zero, fallback
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast and accurate).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(Fa * Fb, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Objective for a vector already normalised (sum == 1).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2 * n * max(conv(a, a))  (assumes sum(a)==1)."""
    n = a.size
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the grader.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a non‑negative height sequence that (approximately) minimises
    the evaluation function.  The routine respects a `budget_s` second time
    limit and prints intermediate scores.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate length that still gives many degrees of
    # freedom while keeping the LP (if used) tractable.
    # ------------------------------------------------------------------
    n = 3000

    # ------------------------------------------------------------------
    # Warm start – try to reuse a previously discovered sequence, otherwise
    # build a pool of classical shapes and keep the best.
    # ------------------------------------------------------------------
    a = None
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if base.shape[0] != n:
                if base.shape[0] > n:
                    base = base[:n]
                else:
                    base = np.pad(base, (0, n - base.shape[0]), mode='constant')
            a = np.maximum(0.0, base)
            s = a.sum()
            if s > 0:
                a = a / s
        except Exception:          # pragma: no cover
            a = None

    if a is None:
        # -----------------------------------------------------------------
        # Generate a diverse set of candidate shapes.
        # -----------------------------------------------------------------
        candidates = []
        x = np.arange(n, dtype=np.float64)

        # Exponential decay (lam ~ decay rate)
        for lam in np.geomspace(5e-4, 2e-2, num=8):
            shape = np.exp(-lam * x)
            shape /= shape.sum()
            candidates.append(shape)

        # Power‑law decay
        for p in np.linspace(0.3, 2.0, num=8):
            shape = (x + 1) ** (-p)
            shape /= shape.sum()
            candidates.append(shape)

        # Gaussian centred in the middle
        for sigma in np.linspace(n / 8.0, n / 3.0, num=4):
            shape = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
            shape /= shape.sum()
            candidates.append(shape)

        # Half‑sine (always positive)
        shape = np.sin(np.pi * (x + 0.5) / n)
        shape /= shape.sum()
        candidates.append(shape)

        # Random convex combinations of the above basic shapes
        base_shapes = candidates.copy()
        for _ in range(12):
            w = rng.random(len(base_shapes))
            w /= w.sum()
            combo = sum(wi * s for wi, s in zip(w, base_shapes))
            combo = np.maximum(combo, 0.0)
            combo /= combo.sum()
            candidates.append(combo)

        # -----------------------------------------------------------------
        # Pick the best candidate according to the objective.
        # -----------------------------------------------------------------
        best_val = np.inf
        best_seq = None
        for seq in candidates:
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
        a = best_seq.copy()
        best_val = _objective(a)
        best_seq = a.copy()
        print(f"[warm] n={n} init_score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Keep track of the best solution so far.
    # ------------------------------------------------------------------
    best_val = _objective(a)
    best_seq = a.copy()
    print(f"[start] n={n} score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Phase 1 – Linear‑program mixing (only if SciPy is available).
    # ------------------------------------------------------------------
    if _HAVE_SCIPY:
        # Pre‑compute index arrays for the sparse convolution matrix.
        rows = np.concatenate([np.arange(j, j + n) for j in range(n)], dtype=np.int64)
        cols = np.repeat(np.arange(n, dtype=np.int64), n)
        max_lp_iter = 30
        lp_end = start_time + 0.35 * budget_s
        it_lp = 0
        while time.time() < lp_end and it_lp < max_lp_iter:
            it_lp += 1
            # current worst convolution value
            conv_full = _full_convolution_fft(a, a)
            M = conv_full.max()

            # Build the sparse linear‑program matrix A_ub @ b ≤ M.
            # A_ub is the Toeplitz‑ish matrix encoding the convolution a * b.
            data = np.tile(a, n)                     # length n²
            A_ub = scipy.sparse.csr_matrix((data, (rows, cols)),
                                           shape=(2 * n - 1, n))

            # Linear program: maximise sum(b)  ⇔ minimise -sum(b)
            c = -np.ones(n, dtype=np.float64)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)

            res = scipy.optimize.linprog(c,
                                         A_ub=A_ub,
                                         b_ub=b_ub,
                                         bounds=[(0, None)] * n,
                                         method='highs',
                                         options={'presolve': True})
            if not res.success:
                print(f"[LP] iteration {it_lp} failed: {res.message}")
                break

            b_opt = res.x
            sum_b = b_opt.sum()
            if sum_b <= 0:
                print("[LP] zero‑mass solution, aborting LP phase")
                break

            # Normalise the LP solution to sum‑1 and mix it into a.
            g = b_opt / sum_b
            t = 0.2 / (it_lp)                     # decreasing mixing weight
            a = (1.0 - t) * a + t * g
            a = a / a.sum()                       # protect against round‑off

            cur_val = _objective(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[LP iter {it_lp:02d}] new_best={best_val:.12f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated (mirror) sub‑gradient descent.
    # ------------------------------------------------------------------
    gd_end = start_time + 0.70 * budget_s
    lr = 0.6                      # initial learning rate
    lr_decay = 0.9999
    it_gd = 0
    while time.time() < gd_end:
        it_gd += 1
        conv_full = _full_convolution_fft(a, a)
        M = conv_full.max()
        # Active set (indices where the max is attained)
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]

        w = np.zeros_like(conv_full)
        if active.size:
            w[active] = 1.0 / active.size
        else:                       # fallback – should never happen
            w[np.argmax(conv_full)] = 1.0

        # Sub‑gradient of max(conv) w.r.t. a
        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:2 * n - 1]

        # Mirror‑descent (exponentiated gradient) update on the simplex
        a = a * np.exp(-lr * g)
        if a.sum() == 0.0:          # numerical collapse – re‑initialise
            a = np.full(n, 1.0 / n)
        else:
            a = a / a.sum()

        lr = max(1e-4, lr * lr_decay)

        cur_val = _objective(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[GD iter {it_gd:05d}] new_best={best_val:.12f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing fine tuning (mass‑transfer moves).
    # ------------------------------------------------------------------
    cur = a.copy()
    cur_score = _objective(cur)
    if cur_score < best_val:
        best_val = cur_score
        best_seq = cur.copy()
        print(f"[SA init] score={best_val:.12f}")

    T0 = 1e-3
    Tmin = 1e-7
    it_sa = 0
    while time.time() < deadline:
        it_sa += 1
        T = max(Tmin, T0 * (1.0 - it_sa / 200_000))

        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] <= 0.0:
            continue
        max_transfer = min(cur[i], 0.01)
        delta = rng.uniform(0.0, max_transfer)

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        new_score = _objective(new)
        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            cur = new
            cur_score = new_score
            if cur_score < best_val - 1e-12:
                best_val = cur_score
                best_seq = cur.copy()
                print(f"[SA iter {it_sa:06d}] new_best={best_val:.12f}")

        # occasional global jitter to avoid stagnation
        if it_sa % 20000 == 0:
            cur = cur + rng.normal(scale=0.001, size=n)
            cur = cur / cur.sum()
            cur_score = _objective(cur)

    # ------------------------------------------------------------------
    # Return the best sequence in the format required by `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # Optional official score logging (fails silently if evaluate_sequence
    # is not in the global namespace).
    try:
        final_score = evaluate_sequence(final_seq)      # type: ignore
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:                                   # pragma: no cover
        print("[result] evaluate_sequence not available")

    return final_seq
