import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Build convolution matrix: (conv(f, g))_k = Σ_i f_i * g_{k-i}
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return a sparse matrix A such that A @ g = conv(f, g)."""
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    # k runs from 0 to 2n-2
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # j = k - i
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(k - i)
            data.append(f[i])
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

# Simple hill‑climber used after up‑sampling
# ----------------------------------------------------------------------
def _light_hillclimb(a: np.ndarray,
                     deadline: float,
                     max_frac_start: float = 0.02) -> np.ndarray:
    """Move tiny amounts of mass from a large entry to a random one."""
    n = a.shape[0]
    best_a = a.copy()
    cur_val = 2.0 * n * _full_convolution(a).max()
    best_val = cur_val
    no_improve = 0

    while time.time() < deadline:
        # fraction of source mass that we are allowed to move (decays)
        progress = (time.time() - (deadline - 0.5)) / 0.5
        max_frac = max_frac_start * (1.0 - progress) + 0.001 * progress

        i = int(np.argmax(a))                # source (largest entry)
        if a[i] <= 1e-15:
            break
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        # recompute the objective only occasionally to keep overhead low
        if np.random.rand() < 0.01:
            cur = 2.0 * n * _full_convolution(a).max()
            if cur < best_val:
                best_val = cur
                best_a = a.copy()
                no_improve = 0
                print(f"[Hillclimb] new best = {best_val:.6f}")
            else:
                no_improve += 1
                if no_improve > 5000:
                    break
    return best_a

    x_small = np.linspace(0.0, 1.0, n_small)
    x_big = np.linspace(0.0, 1.0, n_big)
    a_big = np.interp(x_big, x_small, f)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()                     # renormalise to sum ≈ 1

    # ------------------------------------------------------------------
    # Light hill‑climber to polish the up‑sampled vector
    # ------------------------------------------------------------------
    a_big = _light_hillclimb(a_big, deadline)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_big.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return a_big.tolist()
