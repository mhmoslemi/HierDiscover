"""
Propose a step‑function (non‑negative heights) that (approximately) minimises
``evaluate_sequence``.

Algorithm
========
*   Work on a fixed length `n` (here 6000) and keep the sum of the
    coefficients equal to 1 – the objective is scale‑invariant.
*   Optimise a *smooth* approximation of the max‑convolution:
    `L_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)`.  When `β→∞` the
    approximation tends to the true maximum.
*   The gradient of `L_β` w.r.t. `a` is
        `∇L_β = 2·corr(a, p)` ,
    where `p` are the soft‑max weights of the convolution and `corr`
    denotes the (linear) cross‑correlation.  Both the convolution and the
    correlation are computed with a single FFT each, i.e. in `O(n log n)`.
*   An Adam optimiser (adaptive moments) drives the updates.
    After each step we project back onto the probability simplex
    (non‑negative entries with unit sum) using the O(n log n) method of
    Duchi et al.
*   The smoothing parameter `β` is slowly increased (annealed) so that the
    optimisation first sees a smooth landscape and later focuses on the
    true maximum.
*   If progress stalls we inject a small random perturbation around the best
    solution found so far.
*   When the time budget is almost exhausted a short guided hill‑climbing
    phase (mass‑transfer moves, identical to the one used in the previous
    state‑of‑the‑art code) is run to squeeze a little more improvement.

The whole routine respects the 1000 s budget, prints intermediate scores
and finally returns the best list of heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (full convolution, next power‑of‑two)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full (linear) autocorrelation of a real vector `x`.
    Length of the result is ``2*len(x)-1``.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative round‑off errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008) – O(n log n)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local refinement that tries to move a small fraction of mass away from the
    currently most offending convolution lag.  The routine stops when the time
    budget expires.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # contributions a[i] * a[k-i] that belong to the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # pick two destination indices that (hopefully) do not recreate the same lag
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # perform a feasible mass transfer
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:                                   # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        # exact evaluation (fast shortcut is also possible)
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best step‑function heights found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.4   # safety margin

    # --------------------------------------------------------------
    # Problem size – larger n gives a smaller theoretical lower bound.
    # 6000 is a good compromise between quality and speed.
    # --------------------------------------------------------------
    n = 6000

    # --------------------------------------------------------------
    # Initialise the sequence.
    # If a previous construction is available (height_sequence_1) we
    # interpolate it to the current size, otherwise we start from a random
    # Dirichlet vector.
    # --------------------------------------------------------------
    try:
        # ``height_sequence_1`` is provided by the evaluation environment.
        a0 = np.asarray(height_sequence_1, dtype=float)          # type: ignore[name-defined]
        if a0.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    except Exception:
        a0 = rng.random(n)
        a0 = a0 / a0.sum()     # uniform simplex point

    a = a0.copy()
    # direct objective (sum(a)=1 → denominator = 1)
    conv = _full_convolution(a)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # Adam optimiser hyper‑parameters
    # --------------------------------------------------------------
    alpha = 0.03            # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment

    # smoothing (soft‑max) parameter
    beta = 5.0
    beta_max = 80.0

    it = 0
    no_improve = 0

    while time.time() < deadline - 5.0:
        it += 1

        # ----------------------------------------------------------
        # Gradient of the smoothed objective (log‑sum‑exp of conv)
        # ----------------------------------------------------------
        conv = _full_convolution(a)               # length 2n‑1
        max_c = conv.max()
        # soft‑max weights, numerical‑stable
        exp_beta = np.exp(beta * (conv - max_c))
        p = exp_beta / exp_beta.sum()             # length 2n‑1, sums to 1

        # correlation a ⊛ p  (linear, not circular)
        L = _next_pow2(len(a) + len(p) - 1)       # ≥ 3n‑2
        A = np.fft.fft(a, L)
        P = np.fft.fft(p, L)
        corr = np.fft.ifft(A * np.conj(P)).real    # linear correlation
        grad = 2.0 * corr[:n]                     # ∇L_β

        # ----------------------------------------------------------
        # Adam update + simplex projection
        # ----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _simplex_projection(a_candidate)

        # ----------------------------------------------------------
        # Exact objective for the candidate (fast: use conv again)
        # ----------------------------------------------------------
        conv_new = _full_convolution(a_candidate)
        cand_val = 2.0 * n * conv_new.max()

        # keep the globally best point
        if cand_val < best_val - 1e-12:
            best_val = cand_val
            best_a = a_candidate.copy()
            print(f"[iter {it}] new best = {best_val:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # accept the candidate if it improved the *current* objective,
        # otherwise stay where we are (helps stability)
        if cand_val < cur_val - 1e-14:
            a = a_candidate
            cur_val = cand_val
        else:
            # keep current a (do not accept the uphill move)
            pass

        # ----------------------------------------------------------
        # Adaptive schedules
        # ----------------------------------------------------------
        # increase smoothing gradually – we start smooth and finish
        # close to the true max‑convolution
        beta = min(beta * 1.015, beta_max)

        # occasional perturbation when stagnating
        if no_improve >= 300:
            # perturb around the best solution found so far
            noise = rng.normal(scale=0.02, size=n)
            a = _simplex_projection(best_a + noise)
            cur_val = 2.0 * n * _full_convolution(a).max()
            no_improve = 0
            # reset Adam moments to avoid bias from the old trajectory
            m.fill(0.0)
            v.fill(0.0)
            print(f"[perturb] re‑initialised after {it} iterations")

        # modest learning‑rate decay every few thousand iterations
        if it % 2000 == 0:
            alpha *= 0.9

    # --------------------------------------------------------------
    # Final short guided hill‑climbing refinement (takes the remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # --------------------------------------------------------------
    # Final reporting (use the official evaluator to obtain the exact score)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")

    return best_a.tolist()
