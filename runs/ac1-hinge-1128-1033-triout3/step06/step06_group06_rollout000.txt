"""
Optimise a non‑negative step function to minimise
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum a)²

The strategy has three stages:

1. **Rich initialisation** – try several parametric families
   (uniform, linear, quadratic, sinusoidal, exponential,
   power‑law a_i ∝ (i·(n+1‑i))⁻ᵅ).  The best candidate is kept.

2. **Smooth‑max Adam** – optimise a soft approximation of the
   max‑convolution using Adam on the simplex (log‑softmax
   parametrisation).  The smoothing parameter ε is annealed from
   5·10⁻² down to 10⁻⁶.

3. **Targeted hill‑climbing + simulated annealing** – move tiny
   amounts of mass from indices that contribute most to the current
   maximal convolution entry to indices that contribute little.
   Hill‑climbing accepts only improving moves, while a short annealing
   phase allows occasional uphill steps to escape plateaus.

The whole search respects a 1000 s budget (default).  A larger
sequence (n≈20 000) is used because the asymptotic optimum of the
problem is 1.5; increasing n reduces discretisation error and makes it
easier to approach the bound.

The implementation is fully self‑contained, uses only NumPy and the
provided `evaluate_sequence`, and prints progress information.
"""

import time
import numpy as np

# ---------------------------------------------------------------
# Convolution utilities (FFT based, O(N log N))
# ---------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, size)
    conv_full = np.fft.irfft(fa * fa, size)[:out_len]
    return conv_full


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ---------------------------------------------------------------
# Exact objective (works for sum(a)=1)
# ---------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ---------------------------------------------------------------
# Parameteric initialisation – a richer sweep
# ---------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising initial vector and its objective value."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    # linear decreasing
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # quadratic decreasing
    a = (np.arange(n, 0, -1, dtype=float) ** 2)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # half‑wave sine
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # exponential families
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    # power‑law (i·(n+1‑i))⁻ᵅ
    indices = np.arange(1, n + 1, dtype=float)          # 1 … n
    rev = n + 1 - indices
    prod = indices * rev
    for alpha in np.linspace(0.05, 1.0, 20):
        a = prod ** (-alpha)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    return best_a, best_val

# Phase A – Adam on a smooth approximation of the max‑convolution
# ---------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth max (log‑sum‑exp) of conv(a,a)."""
    # log‑softmax parametrisation (ensures Σa = 1, a ≥ 0)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.12, 0.003
    eps_start, eps_end = 5e-2, 1e-6

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 40000  # restart if stagnating

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- soft‑max to obtain vector on simplex -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # ----- exact objective for bookkeeping -----
        cur_val = _objective_value(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- schedule for learning‑rate & smoothing -----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth max via log‑sum‑exp (numerically stable) -----
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        # Clip to avoid overflow in exp
        shifted = np.clip(shifted, -700, 0)
        weights = np.exp(shifted)
        p = weights / weights.sum()          # soft‑max distribution over positions

        # ----- gradient of smooth objective w.r.t. a -----
        # (∂/∂a) 2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)        # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]   # n entries that correspond to ∂/∂a
        grad_a = 4.0 * n * segment[::-1]

        # ----- chain rule through the soft‑max -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to avoid being stuck in a flat region
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        # restart on stagnation
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restarting after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val

# Phase B – Deterministic hill‑climbing on the exact objective
# ---------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Greedy local search that shifts mass from the biggest contributors
    to the least contributors of the current maximal convolution entry."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.6, 0.002
    IMPROVE_LIMIT = 250_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate the index where convolution attains its maximum
        k_max = int(np.argmax(b))

        # contributions of each a[i] to this maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index = argmax contribution (deterministic)
        if total_contrib > 0:
            i = int(np.argmax(contrib))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            # nothing to move – pick another source at random
            i = np.random.randint(n)

        # target index – favour positions with tiny contribution
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        # invert to get preference for small values
        w = 1.0 / (rev_contrib + 1e-12)
        w[i] = 0.0
        if w.sum() > 0:
            probs = w / w.sum()
            j = np.random.choice(n, p=probs)
        else:
            # fallback to uniform distinct index
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move (fraction of a[i])
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta          # self‑term at j
        b_new[2 * i] += delta * delta          # self‑term at i (removed mass)
        b_new[i + j] -= 2.0 * delta * delta    # cross term

        new_val = 2.0 * n * b_new.max()

        # accept only strict improvements
        if new_val < cur_val - 1e-15:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – re‑inject a small random shake around best
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")
        it += 1

    return best_a

# Phase C – brief simulated‑annealing refinement
# ---------------------------------------------------------------
def _annealing_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Metropolis‑style search that occasionally accepts worsening moves."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    T_start, T_end = 6e-4, 4e-8
    IMPROVE_LIMIT = 120_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        k_max = int(np.argmax(b))

        # source index – most harmful contribution
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        if contrib.sum() > 0:
            i = int(np.argmax(contrib))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            i = np.random.randint(n)

        # uniform target distinct from i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # tiny transfer (≤ 0.4 % of source mass)
        delta = a[i] * np.random.rand() * 0.004
        if delta <= 0.0:
            continue

        # incremental convolution update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # gentle shake to jump out of a flat region
            noise = np.random.normal(scale=0.003, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val

# Main entry point required by the evaluation framework
# ---------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of heights) that minimises
    `evaluate_sequence`.  Returns a list whose sum is 1 (scale‑invariant)
    and whose entries are non‑negative.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a short safety margin for the final evaluate_sequence call
    deadline = start_time + budget_s - 0.5

    # -----------------------------------------------------------------
    # Problem size – a larger n yields a better asymptotic bound.
    # -----------------------------------------------------------------
    n = 20000

    # -----------------------------------------------------------------
    # Initialise – reuse a known construction if present, otherwise
    # use the richer parametric sweep.
    # -----------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    TARGET = 1.503

    # -----------------------------------------------------------------
    # Phase A – Adam optimisation on a smooth max‑convolution
    # -----------------------------------------------------------------
    phase_a_deadline = start_time + (deadline - start_time) * 0.30
    best_a, best_val = _adam_phase(a0, n,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=TARGET)

    # -----------------------------------------------------------------
    # Phase B – Deterministic hill‑climbing on the exact objective
    # -----------------------------------------------------------------
    phase_b_deadline = start_time + (deadline - start_time) * 0.80
    best_a = _hillclimb_targeted(best_a, n,
                                 start=phase_a_deadline,
                                 deadline=phase_b_deadline,
                                 best_a=best_a,
                                 best_val=best_val,
                                 target_val=TARGET)

    # -----------------------------------------------------------------
    # Phase C – short simulated annealing
    # -----------------------------------------------------------------
    best_a, best_val = _annealing_phase(best_a, n,
                                        start=phase_b_deadline,
                                        deadline=deadline,
                                        best_a=best_a,
                                        best_val=best_val,
                                        target_val=TARGET)

    # -----------------------------------------------------------------
    # Final evaluation (secure wrapper)
    # -----------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return as plain Python list (the evaluator expects this)
    return best_a.tolist()
