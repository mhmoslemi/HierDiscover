"""
Simulated‑annealing hill‑climber for the step‑function objective.

The objective (with Σ a_i = 1) is  2·n·max_k (a∗a)_k .
We start from a small parametric family (U‑shaped power laws) and keep the
best seed.  A long annealing phase moves small chunks of mass from indices
that contribute a lot to the current maximal convolution entry to indices
that contribute little.  Moves are accepted with the usual Metropolis
criterion, which lets the search jump out of local minima.  When the time
budget is almost exhausted we finish with a few projected‐gradient steps
to polish the solution.

All helpers are top‑level, the entry point is ``propose_candidate`` which
conforms to the required signature.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of ``x`` with itself (length = 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (Σ a_i = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a∗a) for a simplex‑normalised vector."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Quick parametric sweep – one‑parameter U‑shape family
# ----------------------------------------------------------------------
def _initial_candidates(n: int):
    """Generate a few deterministic shapes and keep the best."""
    best_a = None
    best_val = np.inf

    # power‑law family a_i ∝ ((i+½)(n‑i‑½))^{-γ}
    x = np.arange(n, dtype=np.float64)
    eps = 1e-12
    for gamma in np.linspace(0.0, 6.0, 61):            # finer grid than before
        denom = (x + 0.5) * (n - x - 0.5)
        a = np.power(denom + eps, -gamma)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # mixture of the above with a tiny uniform component (helps escaping)
    uniform = np.ones(n) / n
    for mix in (0.0, 0.05, 0.10):
        a_mix = (1.0 - mix) * best_a + mix * uniform
        a_mix /= a_mix.sum()
        val = _objective(a_mix)
        if val < best_val:
            best_val = val
            best_a = a_mix.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill climbing
# ----------------------------------------------------------------------
def _anneal_hillclimb(
    a_start: np.ndarray,
    n: int,
    start_time: float,
    deadline: float,
    best_a: np.ndarray,
    best_val: float,
    target_val: float = 1.503,
) -> tuple[np.ndarray, float]:
    """
    Move a little probability mass from “heavy” to “light” contributors,
    accepting uphill moves with a Metropolis probability.
    """
    a = a_start.copy()
    b = _full_convolution(a)                     # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # annealing schedule
    T_start, T_end = 1e-2, 1e-7                # temperature (relative to objective)
    max_frac_start, max_frac_end = 0.6, 0.01   # biggest allowed fractional move
    IMPROVE_LIMIT = 200_000                    # stagnation trigger
    no_improve = 0
    iteration = 0
    eps = 1e-14

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start_time) / (deadline - start_time)
        T = T_start * (T_end / T_start) ** progress
        max_frac = max_frac_start * (max_frac_end / max_frac_start) ** progress

        # ---- locate the current maximal convolution entry and its contributions
        k_max = int(np.argmax(b))
        idx = k_max - np.arange(n)                     # index of partner for each i
        mask = (idx >= 0) & (idx < n)

        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            # source ~ contribution to the max entry
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ---- target – bias towards indices that *do not* help the max entry
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # uniform fallback
            j = random.randrange(n - 1)
            if j >= i:
                j += 1

        # ---- amount of mass to move (randomised within the allowed fraction)
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution (O(n))
        b_new = b.copy()
        # linear part (cross terms)
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic part
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max
        delta_obj = new_val - cur_val

        # ---- Metropolis acceptance
        if delta_obj < -eps or np.random.rand() < math.exp(-delta_obj / T):
            # accept the move
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - eps:
                best_val = new_val
                best_a = a.copy()
                print(f"[Anneal {iteration:8d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ---- stagnation handling – occasional random jitter
        if no_improve > IMPROVE_LIMIT:
            # re‑inject a small random perturbation around the current best
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal {iteration:8d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple projected sub‑gradient descent (fine‑tuning)
# ----------------------------------------------------------------------
def _subgradient_descent(
    a_start: np.ndarray,
    n: int,
    start_time: float,
    deadline: float,
    best_a: np.ndarray,
    best_val: float,
    target_val: float = 1.503,
) -> tuple[np.ndarray, float]:
    """Project‑gradient steps on the exact max‑convolution objective."""
    a = a_start.copy()
    cur_val = _objective(a)

    eta0 = 0.5               # initial step size (on the simplex)
    eta_min = 1e-4
    iteration = 0
    eps = 1e-14
    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress‑based step‑size schedule (slow decay)
        progress = (now - start_time) / (deadline - start_time)
        eta = eta0 * (eta_min / eta0) ** progress

        # compute the index achieving the current maximum
        b = _full_convolution(a)
        k_max = int(np.argmax(b))

        # sub‑gradient of the active entry: (∂/∂a_i) b_{k_max} = 2·a_{k_max-i}
        grad = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad[mask] = 2.0 * a[idx[mask]]

        # gradient step followed by projection onto the simplex
        a_tent = a - eta * grad
        a_proj = _project_to_simplex(a_tent)

        new_val = _objective(a_proj)

        if new_val < cur_val - eps:
            a = a_proj
            cur_val = new_val
            if new_val < best_val - eps:
                best_val = new_val
                best_a = a.copy()
                print(f"[Subgrad {iteration:8d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        iteration += 1

    return best_a, best_val
