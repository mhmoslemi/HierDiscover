```
def multiplicative_descent(a, grad, eta):
    # Center gradient to have zero sum (so product factors roughly keep sum)
    g = grad - np.mean(grad)
    # Limit magnitude to avoid overflow
    g = np.clip(g, -50, 50)
    factor = np.exp(-eta * g)
    a_new = a * factor
    # Avoid all zero
    if a_new.sum() == 0:
        a_new = a + 1e-12
    a_new /= a_new.sum()
    return a_new
```

"""
Improved step‑function search for minimise evaluate_sequence.

The algorithm consists of three stages:

1. **Smooth surrogate optimisation** – minimise a log‑sum‑exp
   approximation of max(conv) with L‑BFGS‑B while an
   ε‑continuation drives the surrogate towards the true max.
   The variables are kept as a soft‑max (logits) guaranteeing
   non‑negativity and Σa = 1.

2. **Multiplicative sub‑gradient descent** – after the smooth phase
   we tighten the bound by repeatedly moving mass **away** from the
   lag where the convolution attains its maximum.
   The sub‑gradient of max(conv) is  g_i = 2·a_{k_max‑i} (if the index is
   inside the range).  A multiplicative update
        a ← a·exp{-η·(g‑mean(g))}
   (followed by renormalisation) reduces the objective while staying
   on the simplex.

3. **Aggressive mass‑transfer hill‑climbing** – finally we perform a
   more aggressive “donor → recipient” transfer.  The top‑contributing
   indices for the maximal lag are identified as donors, the lowest‑mass
   positions as recipients, and a small fraction of the donor mass is
   moved.  The step size is adapted and the best solution is kept.

The whole routine respects a 1000 s time budget; intermediate results
and the current best evaluation are printed.  The returned list of
heights can be fed to ``evaluate_sequence`` directly.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may give tiny negative values – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full (linear) convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the log‑sum‑exp surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv-max)/ε)

    together with its gradient with respect to the soft‑max logits w.
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # full autocorrelation
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed‑form)
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                  # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]                # middle n entries
    grad_a = 4.0 * n * seg[::-1]                   # shape (n,)

    # chain‑rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution objective
# ----------------------------------------------------------------------
def _max_conv_subgrad(a: np.ndarray):
    """
    Return (grad, max_val, k_max) where grad_i = 2·a_{k_max‑i}
    (zero if the index is out of range).  k_max is a randomly chosen
    lag attaining the maximum.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    k = int(np.random.choice(ks))

    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = k - i_range
    # ∂max/∂a_i = 2·a_{k-i}
    grad[i_range] = 2.0 * a[j_range]
    return grad, max_val, k


# ----------------------------------------------------------------------
# Multiplicative (exponentiated) sub‑gradient descent step
# ----------------------------------------------------------------------
def _multiply_update(a: np.ndarray, grad: np.ndarray, eta: float):
    """
    Perform one multiplicative update a ← a·exp(-η·(grad‑mean(grad))).
    The result is renormalised to sum‑to‑one and clipped to stay non‑negative.
    """
    # Zero‑mean the gradient so that the update roughly preserves the ℓ₁‑norm.
    g = grad - grad.mean()
    # Clamp to avoid overflow/underflow in exp.
    g = np.clip(g, -50, 50)
    factor = np.exp(-eta * g)
    a_new = a * factor
    # Numerical safety
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:
        # If everything vanished (unlikely), fallback to uniform.
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new /= s
    return a_new

# Aggressive mass‑transfer hill‑climbing (improved version)
# ----------------------------------------------------------------------
def _mass_transfer_improved(a: np.ndarray, best_val: float,
                            deadline: float, rng: np.random.Generator):
    """
    Repeatedly transfer a fraction of the mass from the positions that
    contribute most to the current maximal convolution lag to the
    currently lightest positions.  The step fraction is adapted
    dynamically; the routine stops when the time budget expires.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.2   # initial proportion of donor mass to move
    best_a = cur_a.copy()

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # --- contributions to the maximal lag --------------------------------
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]

        # aggregate contribution per index (both i and j receive the same
        # amount because a_i*a_j = a_j*a_i)
        pos_contrib = {}
        for idx, i in enumerate(i_range):
            j = int(j_range[idx])
            c = contrib[idx]
            pos_contrib[i] = pos_contrib.get(i, 0.0) + c
            pos_contrib[j] = pos_contrib.get(j, 0.0) + c

        # donors = indices with largest contributions (up to 10)
        top_n = min(10, len(pos_contrib))
        donors = [p for p, _ in sorted(pos_contrib.items(),
                                       key=lambda kv: kv[1],
                                       reverse=True)[:top_n]]

        # recipients = indices with smallest current mass, distinct from donors
        low_candidates = np.argsort(cur_a)
        recipients = []
        for idx in low_candidates:
            if idx not in donors:
                recipients.append(idx)
            if len(recipients) >= top_n:
                break
        if len(recipients) < top_n:
            # fallback – any index not a donor
            recipients = [p for p in range(n) if p not in donors][:top_n]

        # --- perform the transfer -------------------------------------------
        new_a = cur_a.copy()
        ok = True
        for d, r in zip(donors, recipients):
            delta = new_a[d] * step_frac
            if delta <= 0:
                ok = False
                break
            new_a[d] -= delta
            new_a[r] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue

        # renormalise (exact ℓ₁‑norm = 1)
        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue
        new_a /= s

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[mass_transfer] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.02, 0.5)   # be a bit more aggressive
        else:
            step_frac = max(step_frac * 0.95, 1e-5)

    return best_a, best_val
