import numpy as np
import time

def _full_convolution(a):
    n = a.shape[0]
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:out_len]
    return conv

def _objective(a):
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())

def _exp_gradient_optimize(a0, n, start_time, deadline, best_a, best_val):
    a = a0.copy()
    iteration = 0
    no_improve = 0
    IMPROVE_LIMIT = 20000
    eta0 = 0.5
    while True:
        now = time.time()
        if now >= deadline:
            break
        # compute convolution
        b = _full_convolution(a)
        k_max = int(np.argmax(b))
        # contributions
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        # gradient of b_max w.r.t a_i is 2*a_{k-i} ; we ignore factor 2
        g[mask] = a[idx[mask]]
        # learning rate schedule
        progress = (now - start_time) / (deadline - start_time)
        eta = eta0 * (1.0 - progress)  # linear decay to 0
        # exponentiated update
        a = a * np.exp(-eta * g)
        # renormalize, avoid all zeros
        s = a.sum()
        if s == 0.0:
            # reinitialize uniform
            a = np.full(n, 1.0 / n)
        else:
            a /= s
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[ExpGrad {iteration}] new best = {best_val:.6f}")
        else:
            no_improve += 1
        if no_improve > IMPROVE_LIMIT:
            # perturb
            a = best_a + np.random.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0:
                a = np.full(n, 1.0/n)
            else:
                a /= s
            no_improve = 0
            print(f"[ExpGrad {iteration}] perturb after stagnation")
        iteration += 1
    return best_a, best_val

def _hillclimb_optimize(a_start, n, start_time, deadline, best_a, best_val):
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    iteration = 0
    no_improve = 0
    IMPROVE_LIMIT = 200000
    eps = 1e-12
    max_frac_start, max_frac_end = 0.5, 0.01
    while True:
        now = time.time()
        if now >= deadline:
            break
        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1 - progress) + max_frac_end * progress
        k_max = int(np.argmax(b))
        # get top K indices
        K = 3
        # Determine K highest distinct entries (including the current max)
        # Use np.argpartition for speed:
        flat_indices = np.argpartition(-b, K-1)[:K]  # get K largest
        top_ks = flat_indices
        # compute contribution per i
        contrib = np.zeros_like(a)
        for k in top_ks:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            contrib[mask] += a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        # source selection
        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))
        if a[i] <= eps:
            continue
        # compute target weights (inverse of contrib)
        target_weights = np.zeros_like(a)
        # for each k in top_ks, compute indices that do NOT contribute strongly
        # simpler: use contributions as bias: minimize contribution
        target_weights = 1.0 / (contrib + eps)
        target_weights[i] = 0.0
        wsum = target_weights.sum()
        if wsum > 0:
            j = np.random.choice(n, p=target_weights/ wsum)
        else:
            j = (i+1) % n  # simple fallback
        # delta
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue
        # incremental convolution update O(n)
        b_new = b.copy()
        # The contribution of moving delta from i to j modifies convolution as:
        # subtract 2*delta*a from positions i + [0..n-1] (since each term a_i * a_k)
        # This effect: for each index t, b[t] changes by -2*delta*a[t - i] for t in i..i+n-1 (where t-i in range)
        # Equivalent to vector addition bigger.
        # Simpler: recompute convolution entirely (cost O(n log n)) for accuracy.
        a[i] -= delta
        a[j] += delta
        # recompute b
        b = _full_convolution(a)
        new_val = 2.0 * n * b.max()
        if new_val < cur_val - 1e-14:
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[HillImp {iteration}] new best = {best_val:.6f}")
        else:
            # revert move
            a[i] += delta
            a[j] -= delta
            no_improve += 1
        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            # perturb around best
            a = best_a + np.random.normal(scale=0.02, size=n)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full(n, 1.0/n)
            else:
                a /= a_sum
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillImp {iteration}] stagnation – perturb")
        iteration += 1
    return best_a, best_val

"""
Propose a low‑value step‑function via three stages:

1️⃣  Quick parametric sweep (power‑law, β‑distribution, random Dirichlet)
    to obtain a decent seed.
2️⃣  Exponentiated‑gradient descent on the exact sub‑gradient of
    the max‑convolution (mirror‑descent on the simplex).  The update
    a_i ← a_i·exp(−η·∂max/∂a_i) automatically keeps the vector
    non‑negative and on the simplex.
3️⃣  Targeted hill‑climbing that moves mass from indices that contribute
    heavily to the *K* largest convolution entries toward indices that
    contribute little.  The convolution is updated incrementally (O(n))
    so many moves can be tried inside the remaining budget.

The routine respects the 1000 s limit and prints intermediate progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution x*x (length 2·len(x)‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Value of 2·n·max(conv(a,a)) for a vector whose sum is 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Simplex projection (used only by fallback code)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(v.size) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – generate a collection of sensible seeds
# ----------------------------------------------------------------------
def _seed_candidates(n: int):
    """Return (best_a, best_val) after a modest sweep over several families."""
    best_a = None
    best_val = np.inf
    eps = 1e-12
    x = np.arange(n, dtype=np.float64) + 0.5                # centre of each bin
    n_float = float(n)

    # 1. Power‑law U‑shape  a_i ∝ ((i+½)(n‑i‑½))^{−γ}
    for gamma in np.linspace(0.0, 5.0, 51):
        denom = x * (n_float - x)
        a = np.power(denom + eps, -gamma)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # 2. Symmetric sum of two power laws
    for p in np.linspace(0.1, 3.0, 31):
        a = (x ** -p) + ((n_float - x) ** -p)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    # 3. Beta‑distribution shapes  a_i ∝ x^{α‑1}(n‑x)^{β‑1}
    alphas = np.linspace(0.2, 2.0, 9)
    betas = np.linspace(0.2, 2.0, 9)
    for alpha in alphas:
        for beta in betas:
            a = (x ** (alpha - 1.0)) * ((n_float - x) ** (beta - 1.0))
            s = a.sum()
            if s == 0:
                continue
            a /= s
            val = _objective(a)
            if val < best_val:
                best_val, best_a = val, a.copy()

    # 4. Random Dirichlet draws with small concentration (sparse seeds)
    for _ in range(30):
        # α_i ∈ (0,0.2) produces varied sparse vectors
        alpha_vec = np.random.rand(n) * 0.15 + 0.01
        a = np.random.gamma(alpha_vec, 1.0)
        if a.sum() == 0:
            continue
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient (mirror descent) on the exact sub‑gradient
# ----------------------------------------------------------------------
def _exp_gradient_optimize(a0: np.ndarray,
                           n: int,
                           start: float,
                           deadline: float,
                           best_a: np.ndarray,
                           best_val: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the max‑convolution using a_i ← a_i·exp(−η·∂max)."""
    a = a0.copy()
    iteration = 0
    stagnation = 0
    IMPROVE_LIMIT = 25000
    eta_start = 0.7                              # initial step‑size
    eps = 1e-12

    while True:
        now = time.time()
        if now >= deadline:
            break

        # convolution & location of the current maximum
        b = _full_convolution(a)
        k_max = int(np.argmax(b))

        # sub‑gradient of max(b) w.r.t. a (ignoring constant 2 factor)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = a[idx[mask]]                     # ∂max/∂a_i ≈ a_{k_max‑i}

        # linear schedule for η: from η_start down to 0
        progress = (now - start) / (deadline - start)
        eta = eta_start * (1.0 - progress)

        # mirror‑descent update (keeps a ≥ 0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        cur_val = _objective(a)
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, a.copy()
            stagnation = 0
            print(f"[ExpGrad {iteration:7d}] new best = {best_val:.6f}")
        else:
            stagnation += 1

        # occasional random perturbation if stuck for a long time
        if stagnation > IMPROVE_LIMIT:
            a = best_a + np.random.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum
            stagnation = 0
            print(f"[ExpGrad {iteration:7d}] perturb after stagnation")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – targeted hill‑climbing (moves mass from heavy contributors)
# ----------------------------------------------------------------------
def _hillclimb_optimize(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> tuple[np.ndarray, float]:
    """
    Move a small amount of mass from indices that heavily contribute to the
    K largest convolution entries toward indices that contribute little.
    The convolution is updated incrementally (O(n)) for speed.
    """
    a = a_start.copy()
    b = _full_convolution(a)                      # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # schedule for the maximal transferable fraction
    max_frac_start, max_frac_end = 0.60, 0.02
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    iteration = 0
    eps = 1e-12
    K = 3                                            # how many top entries to watch

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # indices of the K largest convolution entries
        top_ks = np.argpartition(-b, K - 1)[:K]

        # total contribution of each index to those K entries
        contrib = np.zeros_like(a)
        for kk in top_ks:
            idx = kk - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            contrib[mask] += a[mask] * a[idx[mask]]

        total_contrib = contrib.sum()
        if total_contrib > 0.0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target bias – prefer indices that *don't* contribute much
        target_weights = 1.0 / (contrib + eps)
        target_weights[i] = 0.0                     # never pick the source itself
        wsum = target_weights.sum()
        if wsum > 0.0:
            j = np.random.choice(n, p=target_weights / wsum)
        else:
            # fallback uniform (excluding i)
            j = (i + np.random.randint(1, n)) % n

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # -------- incremental update of the convolution (O(n)) --------
        # copy of current convolution to modify
        b_new = b.copy()
        # both ordered pairs (i,·) and (·,i) are altered
        b_new[i:i + n] -= 2.0 * delta * a                # remove contributions of a[i]
        b_new[j:j + n] += 2.0 * delta * a                # add contributions of a[j]
        # quadratic self‑terms
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        if new_val < cur_val - 1e-14:                    # strictly improving
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[HillImp {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            a = best_a + np.random.normal(scale=0.02, size=n)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillImp {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final wrapper call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a few thousand points give a good trade‑off.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise a – try to reuse a previously found construction.
    # --------------------------------------------------------------
    best_a = None
    best_val = np.inf
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        best_a, best_val = a0.copy(), _objective(a0)
        print(f"[init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 0 – cheap deterministic sweep for a decent seed.
    # --------------------------------------------------------------
    try:
        shape_a, shape_val = _seed_candidates(n)
        if shape_val < best_val:
            best_a, best_val = shape_a, shape_val
            print(f"[init] parametric sweep improved value={best_val:.6f}")
    except Exception as exc:
        print("[init] seed sweep failed:", exc)

    # fallback random start if nothing else succeeded
    if best_a is None:
        a = np.random.rand(n)
        a /= a.sum()
        best_a, best_val = a, _objective(a)
        print("[init] random start, value=", best_val)

    # --------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient optimisation (sub‑gradient of max).
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.45 * budget_s
    best_a, best_val = _exp_gradient_optimize(best_a, n,
                                              start=start_time,
                                              deadline=phase1_deadline,
                                              best_a=best_a,
                                              best_val=best_val)

    # --------------------------------------------------------------
    # Phase 2 – targeted hill‑climbing to fine‑tune the convolution peak.
    # --------------------------------------------------------------
    best_a, best_val = _hillclimb_optimize(best_a, n,
                                           start=phase1_deadline,
                                           deadline=deadline,
                                           best_a=best_a,
                                           best_val=best_val)

    # --------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # ensure we output a plain python list of floats
    return best_a.tolist()
