```python
def _full_convolution(x):
    # Use np.fft for speed
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv
```

```python
def _hill_climb(a, n, start_time, deadline, best_a, best_val):
    # full convolve
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()
    # sanity if cur_val != best_val maybe due to rounding
    best_val = min(best_val, cur_val)
    best_a = a.copy()
    max_frac_start = 0.15  # allow up to 15% of mass per move initially
    max_frac_end = 0.001
    temp_start = 0.001
    temp_end = 1e-7
    no_improve = 0
    IMPROVE_LIMIT = 200_000
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break
        # schedule
        progress = (now - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1 - progress) + max_frac_end * progress
        temp = temp_start * (1 - progress) + temp_end * progress

        # find index of max conv
        k_max = int(np.argmax(b))

        # contributions
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))  # fallback

        # target bias opposite contributions
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + 1e-12)
        target_weights[i] = 0.0
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # uniform random else
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # delta amount
        max_delta = a[i] * max_frac
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand()
        if delta <= 0:
            continue

        # incremental update
        b_new = b.copy()
        # update by adding/subtracting 2*delta*a range
        b_new[j:j+n] += 2.0 * delta * a
        b_new[i:i+n] -= 2.0 * delta * a
        # quadratic terms
        b_new[2*j] += delta * delta
        b_new[2*i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            # uphill acceptance (simulated annealing)
            prob = np.exp(-(new_val - cur_val) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                no_improve = 0
                print(f"[SA  {iteration:7d}] new best = {best_val:.6f}")
            else:
                no_improve = 0 if new_val < cur_val else no_improve + 1
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            # random perturbation around best
            a = best_a.copy()
            noise = np.random.normal(scale=0.05, size=n)  # moderate noise
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA  {iteration:7d}] restart after stagnation")
        iteration += 1

    return best_a, best_val
```

"""
Optimising a non‑negative step function for the evaluation metric.

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a)**2)

which is invariant under scaling, so we fix Σa = 1.
The algorithm consists of three stages:

1. **Parametric sweep** – We try monotone power‑law shapes
   a_i ∝ (i+1)^{-β} for β∈[0,2] on a moderate size (n≈2000).
   The shape that gives the smallest exact objective becomes the seed.
2. **Simple simulated‑annealing hill‑climbing** – From the seed we
   repeatedly move a tiny amount of mass from indices that contribute
   heavily to the current maximal convolution entry towards indices that
   contribute little.  Moves that improve the exact objective are always
   accepted; worsening moves are accepted with a temperature‑dependent
   probability to escape local minima.
3. **Final report** – The best vector found is returned as a plain list.
   ``evaluate_sequence`` (already provided in the environment) is called
   once at the end so the contestant can see the reported score.

The implementation is self‑contained, uses only NumPy and the standard
library, and respects the 1000 s time budget (leaving a small safety
margin for the final evaluation call).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full linear convolution (length 2·n‑1)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of x with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negatives – clamp to zero
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value of 2·n·max(conv(a,a)) under the normalization Σa = 1."""
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Stage 0 – cheap power‑law sweep (a_i ∝ (i+1)^‑β)
# ----------------------------------------------------------------------
def _powerlaw_sweep(n: int):
    """
    Scan β∈[0,2] and return the best shape a (Σa = 1) together with its value.
    The sweep also tests a few convex combinations with the uniform vector
    because mixing can sometimes be beneficial.
    """
    # uniform vector – useful both as a baseline and for mixtures
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    # pre‑compute its convolution (the triangular count pattern)
    counts = np.concatenate([np.arange(1, n + 1), np.arange(n - 1, 0, -1)])
    conv_uniform = counts.astype(np.float64) / (n * n)

    best_a = uniform.copy()
    best_val = _objective(uniform, n)

    idx = np.arange(n, dtype=np.float64)

    # coarse grid for β, then a finer grid around the best one
    beta_grid = np.linspace(0.0, 2.0, 41)   # 0.0 ... 2.0 in steps of 0.05
    for beta in beta_grid:
        # power‑law shape (monotone decreasing)
        shape = np.power(idx + 1.0, -beta)
        shape_sum = shape.sum()
        if shape_sum == 0.0:
            continue
        shape /= shape_sum                     # Σshape = 1

        # exact objective for the pure shape
        val = _objective(shape, n)
        if val < best_val:
            best_val, best_a = val, shape.copy()
            # print progress for debugging / visibility
            print(f"[sweep] beta={beta:5.3f} pure shape -> val={best_val:.6f}")

        # also evaluate a few mixtures with the uniform vector
        conv_shape = _full_convolution(shape)
        conv_shape_u = _conv_fft(shape, uniform)

        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            # a = w·shape + (1‑w)·uniform  (still Σa = 1)
            a_mix = w * shape + (1.0 - w) * uniform
            # linear combination of the three pre‑computed convolutions
            conv_mix = (w * w) * conv_shape \
                     + (2.0 * w * (1.0 - w)) * conv_shape_u \
                     + ((1.0 - w) * (1.0 - w)) * conv_uniform
            val_mix = 2.0 * n * conv_mix.max()
            if val_mix < best_val:
                best_val, best_a = val_mix, a_mix.copy()
                print(f"[sweep] beta={beta:5.3f} w={w:5.2f} -> val={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Stage A – simulated‑annealing hill climbing
# ----------------------------------------------------------------------
def _simulated_annealing(a_init: np.ndarray,
                         n: int,
                         start: float,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Move small amounts of mass between indices.
    Moves that improve the exact objective are always accepted;
    worsening moves are accepted with probability exp(-Δ/temperature).
    The temperature and the permitted transfer fraction decay linearly
    with elapsed time.
    """
    a = a_init.copy()
    b = _full_convolution(a)                # current convolution
    cur_val = 2.0 * n * b.max()

    # schedule parameters
    max_frac_start, max_frac_end = 0.15, 0.001   # how much of a[i] can be moved
    temp_start, temp_end = 5e-3, 1e-8           # simulated‑annealing temperature

    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    eps = 1e-12

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear schedules
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        temperature = temp_start * (1.0 - progress) + temp_end * progress

        # locate the currently maximal convolution entry
        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        ok = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[ok] = a[ok] * a[idx[ok]]
        tot = contrib.sum()

        # ---- pick source index (biased by its contribution) ----
        if tot > 0.0:
            src_probs = contrib / tot
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ---- pick target index (biased towards small contribution) ----
        tgt_weights = np.zeros_like(a)
        tgt_weights[ok] = 1.0 / (contrib[ok] + eps)
        tgt_weights[i] = 0.0
        wsum = tgt_weights.sum()
        if wsum > 0.0:
            j = np.random.choice(n, p=tgt_weights / wsum)
        else:
            # fall back to uniform random choice (different from i)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # ---- decide how much mass to move ----
        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution (O(n)) ----
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # quadratic correction terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # ---- Metropolis acceptance test ----
        accept = False
        if new_val < cur_val - 1e-14:
            accept = True
        else:
            # uphill move – accept with a temperature‑dependent probability
            prob = np.exp(-(new_val - cur_val) / max(temperature, eps))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[SA {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ---- stagnation handling – random perturbation & restart ----
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[SA {iteration:7d}] stagnation – re‑perturb")

        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the given time budget.
    Returns a list (Σ = 1) that (tries to) minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a moderate problem size – large enough to be expressive,
    # small enough to allow many fast iterations.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Stage 0 – rapid parametric sweep for a strong deterministic seed.
    # ------------------------------------------------------------------
    best_a, best_val = _powerlaw_sweep(n)
    print(f"[init] best after sweep: value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage A – simulated‑annealing hill‑climbing.
    # ------------------------------------------------------------------
    best_a, best_val = _simulated_annealing(best_a, n,
                                            start=start_time,
                                            deadline=deadline,
                                            best_a=best_a,
                                            best_val=best_val)

    # ------------------------------------------------------------------
    # Final score using the provided secured wrapper.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
