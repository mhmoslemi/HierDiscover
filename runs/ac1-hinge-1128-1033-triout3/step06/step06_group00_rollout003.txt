def upscale(vec, new_n):
    old_n = len(vec)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_vec = np.interp(x_new, x_old, vec)
    # Ensure nonnegative
    new_vec = np.maximum(new_vec, 0)
    new_vec /= new_vec.sum()
    return new_vec
```

"""
Propose a non‑negative step‑function (list of floats) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three stages under a 1000 s budget:

1. **Coarse random search (small size)** – generate many parametric shapes
   (exponential, power‑law, sinusoidal, mixtures) on a modest length
   (≈1500).  Keep the best candidate.

2. **Fast gradient descent on the small problem** – use a smooth soft‑max
   approximation of the max‑convolution and momentum.  The iterate is kept
   on the probability simplex (Duchi et al. 2008).  The result is a refined
   distribution with a low max‑convolution value.

3. **Upscaling + fine Adam optimisation** – linearly interpolate the
   small solution to the target length (6000), renormalise, then run an
   exact sub‑gradient (max‑convolution) descent with Adam.  A short
   simulated‑annealing style mass‑transfer phase finishes the search.

All intermediate scores are printed.  The final sequence is clipped to
[0, 1000] as required by the grader.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (real).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(Fa * Fb, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Objective for a vector that already satisfies sum=1.
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2 * n * max(conv(a, a))   (assumes sum(a)==1)."""
    n = a.size
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Upscaling / downscaling helpers (linear interpolation).
# ----------------------------------------------------------------------
def _resize(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Resize `vec` to length `new_n` by linear interpolation and renormalise."""
    old_n = vec.shape[0]
    if old_n == new_n:
        out = vec.copy()
    else:
        x_old = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
        x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
        out = np.interp(x_new, x_old, vec)
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / new_n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main entry point required by the grader.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a non‑negative height sequence that (approximately) minimizes
    the evaluation function.  The routine respects a `budget_s` second time
    limit and prints intermediate scores.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem sizes
    # ------------------------------------------------------------------
    small_n = 1500            # cheap to iterate on
    large_n = 6000            # final resolution (used by the grader)

    # ------------------------------------------------------------------
    # Warm start: try to reuse a previously discovered sequence.
    # ------------------------------------------------------------------
    a_small = None
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            a_small = _resize(base, small_n)
        except Exception:
            a_small = None

    # ------------------------------------------------------------------
    # If no warm start, create a random pool of classic shapes.
    # ------------------------------------------------------------------
    if a_small is None:
        # generate a few candidates, keep the best after a quick evaluation
        best_val = float("inf")
        best_vec = None
        for _ in range(2000):
            typ = rng.integers(0, 5)
            x = np.arange(small_n, dtype=np.float64)

            if typ == 0:                       # exponential decay
                lam = rng.uniform(5e-4, 2e-2)
                vec = np.exp(-lam * x)
            elif typ == 1:                     # power‑law
                p = rng.uniform(0.2, 3.0)
                vec = 1.0 / (x + 1) ** p
            elif typ == 2:                     # half‑sine with optional decay
                vec = np.sin(np.pi * (x + 0.5) / small_n)
                lam = rng.uniform(0.0, 1e-2)
                vec *= np.exp(-lam * x)
            elif typ == 3:                     # mixture of two exponentials
                lam1 = rng.uniform(5e-4, 2e-2)
                lam2 = rng.uniform(5e-4, 2e-2)
                w = rng.random()
                vec = w * np.exp(-lam1 * x) + (1.0 - w) * np.exp(-lam2 * x)
            else:                              # simple triangle (convex)
                vec = np.minimum(x, small_n - 1 - x) + 1.0

            vec = np.maximum(vec, 0.0)
            s = vec.sum()
            if s == 0.0:
                continue
            vec /= s
            val = _objective(vec)
            if val < best_val:
                best_val = val
                best_vec = vec.copy()
        a_small = best_vec
        best_val = _objective(a_small)
        print(f"[init] random pool best_val={best_val:.12f}")

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max gradient descent on the small problem.
    # ------------------------------------------------------------------
    a = a_small.copy()
    n = small_n
    best_val = _objective(a)
    best_small = a.copy()

    phase1_end = start_time + 0.45 * budget_s
    eta = 0.5
    eta_min = 1e-6
    eta_decay = 0.9999

    conv = _full_convolution_fft(a, a)
    eps = max(1e-3, 0.5 * conv.max())
    eps_min = 1e-6
    eps_decay = 0.9997

    momentum = 0.9
    vel = np.zeros_like(a)
    it1 = 0
    while time.time() < phase1_end:
        it1 += 1
        conv = _full_convolution_fft(a, a)
        mconv = conv.max()
        cur_val = 2.0 * n * mconv
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_small = a.copy()
            print(f"[phase1 iter {it1:6d}] new_best={best_val:.12f}")

        # smooth soft‑max weighting
        w = np.exp((conv - mconv) / eps)
        w_sum = w.sum()
        if w_sum > 0 and np.isfinite(w_sum):
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        rev = a[::-1]
        g = 2.0 * _full_convolution_fft(rev, w)[n - 1:n - 1 + n]

        vel = momentum * vel + (1.0 - momentum) * g
        a_new = a - eta * vel
        a = _project_simplex(a_new)

        eta = max(eta_min, eta * eta_decay)
        eps = max(eps_min, eps * eps_decay)

        # occasional jitter to escape plateaus
        if it1 % 5000 == 0:
            a += rng.normal(scale=0.001, size=n)
            a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Phase 2 – up‑scale to the target size and refine with Adam.
    # ------------------------------------------------------------------
    a = _resize(best_small, large_n)
    n = large_n
    best_seq = a.copy()
    best_val = _objective(a)
    print(f"[phase2 init] upscaled best_val={best_val:.12f}")

    # Adam hyper‑parameters
    lr = 0.20
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # allocate most of the remaining budget to this phase
    phase2_end = start_time + 0.87 * budget_s
    it2 = 0
    while time.time() < phase2_end:
        it2 += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        conv = _full_convolution_fft(a, a)
        mconv = conv.max()
        cur_val = 2.0 * n * mconv
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[phase2 iter {it2:6d}] new_best={best_val:.12f}")

        # exact max‑sub‑gradient (average over all active positions)
        tol = 1e-14
        active = np.flatnonzero(conv >= mconv - tol)
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0
        else:
            w[np.argmax(conv)] = 1.0

        rev = a[::-1]
        g = 2.0 * _full_convolution_fft(rev, w)[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_candidate = a - step
        a = _project_simplex(a_candidate)

        # very occasional random jitter
        if it2 % 5000 == 0:
            a += rng.normal(scale=0.0005, size=n)
            a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Phase 3 – cheap simulated‑annealing / mass‑transfer fine tuning.
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_val = _objective(cur)
    T0 = 1e-3
    Tmin = 1e-7
    sa_iters = 0
    while time.time() < deadline:
        sa_iters += 1
        T = max(Tmin, T0 * (1.0 - sa_iters / 150_000))

        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] <= 0.0:
            continue
        delta = rng.uniform(0.0, min(cur[i], 0.02))  # allow a modest transfer
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_val = _objective(new)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / T):
            cur = new
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_seq = cur.copy()
                print(f"[SA iter {sa_iters:6d}] new_best={best_val:.12f}")

        if sa_iters % 20000 == 0:
            # occasional global jitter
            cur += rng.normal(scale=0.001, size=n)
            cur = _project_simplex(cur)
            cur_val = _objective(cur)

    # ------------------------------------------------------------------
    # Return the best sequence in the format expected by `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # Optional: report the official score (may raise if evaluate_sequence missing)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] evaluate_sequence not available")

    return final_seq
