def project_simplex(v):
    # v can contain any real numbers.
    # Following algorithm: sort v descending, compute cumsum, find rho.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        # all negative, return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative heights) that minimizes
`evaluate_sequence`.  The algorithm consists of four stages:

1.  Extended parametric sweep (uniform, linear, power‑law, exponential,
    sinusoidal) → pick the best shape.
2.  Projected gradient descent on a smooth approximation of the max of the
    auto‑convolution (log‑sum‑exp smoothing).  The iterates are kept on the
    probability simplex by Euclidean projection.
3.  Deterministic hill‑climbing that always moves mass from the index with the
    largest contribution to the maximal convolution entry to the index with the
    smallest contribution.
4.  A short simulated‑annealing refinement (re‑using the previous implementation).

All stages respect the total time budget (default 1000 s).  The best sequence
found is returned as a plain list of floats (its sum is 1, which makes the
objective scale‑invariant).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT – fast for n≈8000)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (Σ=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Extended parametric initialisation
# ----------------------------------------------------------------------
def _initial_candidate_extended(n: int):
    """Search a modest family of analytic shapes and keep the best."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    # linear decreasing
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # quadratic, cubic, quartic, quintic decreasing
    for p in (2, 3, 4, 5):
        a = (np.arange(n, 0, -1, dtype=float) ** p)
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    # power‑law (i+1)^{‑p}
    for p in (0.5, 1.0, 1.5, 2.0):
        idx = np.arange(1, n + 1, dtype=float)
        a = 1.0 / (idx ** p)
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    # exponential decays
    for r in (0.99999, 0.99995, 0.9999, 0.9995, 0.999, 0.998,
              0.997, 0.996, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    # half‑wave sinusoid
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – projected gradient descent on a smooth max
# ----------------------------------------------------------------------
def _smooth_gd_phase(a0: np.ndarray,
                     best_a: np.ndarray,
                     best_val: float,
                     n: int,
                     start: float,
                     deadline: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smoothed objective
        f_ε(a) = 2·n·Σ_k p_k (a*a)_k
    where p is the soft‑max over the convolution entries.
    After each step the iterate is projected onto the simplex.
    """
    a = a0.copy()
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 25000

    # schedule parameters (geometric decay)
    eps_start, eps_end = 5e-2, 1e-12
    lr_start, lr_end = 0.10, 0.001

    while True:
        now = time.time()
        if now >= deadline:
            break
        progress = (now - start) / (deadline - start)

        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # full auto‑convolution and max entry
        b = _full_convolution(a)
        max_c = b.max()

        # smooth weights (log‑sum‑exp)
        # the exponent for the maximal entry is 0 → its contribution is 1
        exps = np.exp((b - max_c) / eps)
        p = exps / exps.sum()                     # shape (2·n‑1,)

        # gradient of the smoothed objective w.r.t a
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)            # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]      # n entries
        grad = 4.0 * n * segment[::-1]           # ∂/∂a

        # descent step + projection
        a_new = a - lr * grad
        a_new = _project_simplex(a_new)

        # exact objective for bookkeeping
        cur_val = _objective_value(a_new)

        if cur_val < best_val - 1e-14:
            best_val, best_a = cur_val, a_new.copy()
            no_improve = 0
            print(f"[GD {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # occasional random jitter to avoid dead‑ends
        if np.random.rand() < 0.003:
            a_new += np.random.normal(scale=0.005, size=n)
            a_new = _project_simplex(a_new)

        # restart if stagnated too long
        if no_improve > IMPROVE_LIMIT:
            print(f"[GD {it:7d}] restart after stagnation")
            a = best_a.copy()
            a += np.random.normal(scale=0.01, size=n)
            a = _project_simplex(a)
            no_improve = 0
        else:
            a = a_new

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – deterministic hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _deterministic_hillclimb(a_start: np.ndarray,
                             best_a: np.ndarray,
                             best_val: float,
                             n: int,
                             start: float,
                             deadline: float,
                             target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    In each iteration move a fraction of the mass from the index that most
    contributes to the current maximal convolution entry to the index that
    contributes the least.  Accept only strictly improving moves.
    """
    a = a_start.copy()
    cur_val = _objective_value(a)

    IMPROVE_LIMIT = 20000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # recompute convolution
        b = _full_convolution(a)
        k_max = int(np.argmax(b))

        # contribution gradient g_i = 2 * a_{k_max-i} (0 if out of range)
        idx = k_max - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[idx[valid]]

        donor = int(np.argmax(g))          # most harmful index
        recipient = int(np.argmin(g))      # least harmful index
        if donor == recipient:
            # pick a random different recipient
            cand = np.where(np.arange(n) != donor)[0]
            recipient = int(np.random.choice(cand))

        if a[donor] <= 1e-15:
            # nothing to move – jitter and continue
            a += np.random.normal(scale=0.005, size=n)
            a = _project_simplex(a)
            cur_val = _objective_value(a)
            continue

        # schedule a decreasing maximal transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = 0.5 * (1.0 - progress) + 0.001 * progress
        delta = a[donor] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[donor] -= delta
        a_new[recipient] += delta
        # projection is not required – the move respects non‑negativity and sum
        # but we guard against tiny negative rounding errors
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        new_val = _objective_value(a_new)

        if new_val < cur_val - 1e-14:
            a = a_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[HillDet {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # shake and restart from the best seen so far
            a = best_a.copy()
            a += np.random.normal(scale=0.003, size=n)
            a = _project_simplex(a)
            cur_val = _objective_value(a)
            no_improve = 0
            print(f"[HillDet {it:7d}] restart after stagnation")

        it += 1

    return best_a, best_val

# Phase 4 – short simulated‑annealing refinement (re‑used from previous work)
# ----------------------------------------------------------------------
def _annealing_phase(a_start: np.ndarray,
                     best_a: np.ndarray,
                     best_val: float,
                     n: int,
                     start: float,
                     deadline: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Metropolis‑style search that occasionally accepts worsening moves."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    T_start, T_end = 5e-4, 5e-8
    IMPROVE_LIMIT = 100_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break
        progress = (now - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        # current maximal index
        k_max = int(np.argmax(b))

        # donor selection (biased by contribution to the current max)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            a += np.random.normal(scale=0.003, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            continue

        # simple uniform target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # tiny transfer (≤ 0.5 % of source mass)
        delta = a[i] * np.random.rand() * 0.005
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.003, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val
