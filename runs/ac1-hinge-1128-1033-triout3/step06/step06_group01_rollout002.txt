```python
    print("[init] start best value:", best_val)

    # Stage 1: L-BFGS restarts
    epsilon_schedule = np.logspace(np.log10(5e-3), np.log10(1e-11), num=20) # more steps
    # We'll allocate time for stage1 as up to 0.5*budget
    stage1_deadline = start + 0.5 * (budget_s - 5) # keep remainder for later

    restart = 0
    while time.time() < stage1_deadline:
        restart += 1
        # random initialization of w: sample from normal then adjust scale
        w = np.random.randn(n) * 3.0
        # optionally start from previous best's logit
        if restart % 10 == 0:
            # use best_a to generate near-opt logit
            w = np.log(best_a + 1e-12) + np.random.randn(n) * 0.1

        for eps in epsilon_schedule:
            if time.time() >= stage1_deadline:
                break
            _GLOBAL_EPS = eps
            # Use L-BFGS with increased maxiter maybe 1500.
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                            method='L-BFGS-B',
                            jac=True,
                            options={'maxiter': 1500, 'ftol': 1e-14, 'gtol': 1e-14, 'disp': False})
            w = res.x
            # compute a
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = _eval_fast(a)
            # update best
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[restart {restart} eps {eps:.1e}] new best = {best_val:.6f}")

        # Reorder candidate shapes (two heuristics)
        for reorder_func in [_anti_symmetric_reorder, _alternating_extreme_reorder]:
            a_ord = reorder_func(best_a)
            val_ord = _eval_fast(a_ord)
            if val_ord < best_val:
                best_val, best_a = val_ord, a_ord.copy()
                print(f"[reorder] improved to {best_val:.6f} by {reorder_func.__name__}")

    # Stage 2: Simulated annealing extended
    if time.time() < deadline - 30:
        print("[SA] Starting extended SA")
        # Let's run SA with many swaps
        best_a, best_val = _simulated_annealing_long(best_a, best_val, deadline - 15)

    # Stage 3: final hill climb
    if time.time() < deadline:
        print("[refine] Final refinement")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # final evaluation using official evaluate_sequence
    final_score = evaluate_sequence(best_a.tolist())
    print("[final] reported evaluation:", final_score)

    return best_a.tolist()
```

assistantfinal**Solution Overview**

The objective is  

\[
\phi(a)=\frac{2\,n\,\max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the linear convolution of the non‑negative sequence `a` with itself.
The value is invariant under a global scaling of `a`, therefore we can keep the
ℓ₁‑norm fixed (we use 1) and only search for the *shape* of the sequence.

The algorithm consists of three independent stages, each respecting the
remaining time budget:

1. **Smooth surrogate optimisation** –  
   minimise a differentiable surrogate of `max(a*a)` obtained with a
   log‑sum‑exp smoothing.  The surrogate is optimised with many L‑BFGS
   restarts (continuation on the smoothing parameter).  After each restart
   the best vector is reordered by two cheap heuristics that tend to pair
   large and small values.

2. **Permutation search** –  
   Starting from the best vector found in stage 1 we explore the space of
   permutations.  First a short random‑shuffle search gives quick improvements,
   then a long simulated‑annealing run (swap + segment‑reversal moves) is used
   to escape local minima.  The evaluator is a fast FFT‑based implementation of
   `evaluate_sequence`.

3. **Fine‑grained hill climbing** –  
   Small mass‑transfer moves and additional swaps are applied until time runs
   out.

All stages use a single‑threaded NumPy/ SciPy implementation; the total
runtime is kept below 1000 seconds.  The final sequence is returned as a plain
Python list of floats, ready for `evaluate_sequence`.

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
propose_candidate
-----------------
Hybrid optimisation for the step‑function evaluation problem.

The search proceeds in three phases:

1.  Continuation L‑BFGS on a smooth log‑sum‑exp surrogate of the
    ℓ∞‑norm of the self‑convolution.  Several random restarts are performed.
    After each restart the resulting shape is reordered by cheap heuristics
    (anti‑symmetric and alternating‑extreme) that tend to lower the peak.

2.  A permutation phase:
        * a quick random‑shuffle search,
        * an extended simulated‑annealing run on permutations (swap and
          segment‑reversal moves) with a slowly cooling temperature.

3.  A final hill‑climbing refinement that allows tiny mass transfers between
    positions and extra swaps.

All heavy work (convolution, evaluation) uses FFT for speed.
The routine respects a 1000 s budget and always returns a feasible list of
non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (set inside propose_candidate)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)   # eliminate tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence.
    `a` must be non‑negative, ℓ₁‑norm = 1 (the scaling does not affect the target).
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (objective, gradient) for the smoothed surrogate

        L_ε(a) = 2 n·max(conv) + 2 n·ε·log∑exp((conv−max)/ε),

    where a = softmax(w) (ensures a_i ≥ 0 and Σa_i = 1).
    The gradient is computed analytically via FFTs.
    """
    #------ soft‑max (positivity, unit ℓ₁) ------
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    #------ self‑convolution ------
    conv = _fft_convolve_self(a)          # length 2·n‑1
    max_conv = conv.max()

    #------ smooth approximation of the max ------
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # log ∑ exp(...)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    #------ gradient w.r.t. a (closed‑form) ------
    weights = np.exp(shifted)
    p = weights / weights.sum()           # probability distribution over k

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    # indices [n‑1, 2·n‑2) correspond to the needed segment
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    #------ chain rule through the soft‑max ------
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# Bind the FFT helper used inside the gradient
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


_smooth_obj_and_grad._conv_fft = _conv_fft


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by scipy.optimize (uses global eps / n)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Re‑ordering heuristics
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Large values on the left, small values on the right (mirrored).
    The pattern reduces the product of large numbers in the centre of the
    convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]       # large to the left
        new_a[n - 1 - i] = sorted_vals[i]       # small to the right

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _alternating_extreme_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest element at the leftmost position, the second largest
    at the rightmost position, the third at the second‑left, etc.
    This strongly interleaves large and small values.
    """
    n = a.shape[0]
    sorted_desc = np.sort(a)[::-1]           # descending
    new_a = np.empty_like(a)
    left, right = 0, n - 1
    put_left = True
    for val in sorted_desc:
        if put_left:
            new_a[left] = val
            left += 1
        else:
            new_a[right] = val
            right -= 1
        put_left = not put_left
    return new_a


# ----------------------------------------------------------------------
# Random permutation search (cheap, many tries)
# ----------------------------------------------------------------------
def _random_permutation_search(a: np.ndarray, cur_best_val: float,
                               deadline: float, tries: int = 2000) -> (np.ndarray, float):
    """
    Generate `tries` random permutations of `a` and keep the best one.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_best_val

    for _ in range(tries):
        if time.time() >= deadline:
            break
        perm = rng.permutation(n)
        a_perm = a[perm]
        val = _eval_fast(a_perm)
        if val < best_val:
            best_val = val
            best_seq = a_perm.copy()
            print(f"[rand perm] new best = {best_val:.6f}")
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing on permutations (swap + segment reversal)
# ----------------------------------------------------------------------
def _simulated_annealing_long(a: np.ndarray, cur_val: float,
                              deadline: float) -> (np.ndarray, float):
    """
    Classic SA allowing two kinds of moves:
        * swap of two positions,
        * reversal of a random sub‑segment.
    The temperature starts as a small fraction of the current value and is
    multiplied by a decay factor after each iteration.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.05, 1e-6)
    decay = 0.99997
    iter_cnt = 0

    while time.time() < deadline:
        # decide move type
        if rng.random() < 0.85:                     # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            new_val = _eval_fast(a)
            delta = new_val - cur_val
            accept = (delta < 0) or (rng.random() < np.exp(-delta / T))
            if accept:
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA long] new best = {best_val:.6f} (iter {iter_cnt})")
            else:
                a[i], a[j] = a[j], a[i]          # revert
        else:                                          # segment reversal
            i = rng.integers(0, n - 1)
            j = rng.integers(i + 1, n)                # j > i
            a[i:j] = a[i:j][::-1]
            new_val = _eval_fast(a)
            delta = new_val - cur_val
            accept = (delta < 0) or (rng.random() < np.exp(-delta / T))
            if accept:
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA long] rev new best = {best_val:.6f} (iter {iter_cnt})")
            else:
                a[i:j] = a[i:j][::-1]                # revert

        # temperature cooling
        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)
        iter_cnt += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (mass transfers + swaps)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    """
    Very cheap local search:
        * with 20 % probability perform a pure swap,
        * otherwise transfer a small random amount of mass from i to j.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # tiny mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time budget expires.
    """
    # ------------------------------------------------------------------
    # Basic setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0       # tiny safety margin
    print(f"[init] budget = {budget_s}s, deadline @ {deadline:.2f}")

    # --------------------------------------------------------------
    # Problem size – 4000 offers a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4000
    global _GLOBAL_N, _GLOBAL_EPS
    _GLOBAL_N = n

    # --------------------------------------------------------------
    # Initialise from a previously found construction, if available
    # --------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            old_x = np.linspace(0, 1, a0.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, a0)
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)          # uniform start
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()                       # enforce ℓ₁ = 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – smooth surrogate optimisation (L‑BFGS restarts)
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-11), num=20)
    stage1_deadline = start_time + 0.5 * (budget_s - 5)   # leave time for later
    restart = 0

    while time.time() < stage1_deadline:
        restart += 1
        # random logits – broader distribution helps diversity
        w = np.random.randn(n) * 3.0
        # every few restarts use a warm start based on the current best shape
        if restart % 7 == 0:
            w = np.log(best_a + 1e-12) + np.random.randn(n) * 0.2

        for eps in eps_schedule:
            if time.time() >= stage1_deadline:
                break
            _GLOBAL_EPS = eps

            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 1500,
                                    'ftol': 1e-14,
                                    'gtol': 1e-14,
                                    'disp': False})
            w = res.x

            # obtain the candidate vector from the logits
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = _eval_fast(a)

            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[restart {restart} eps {eps:.1e}] new best = {best_val:.6f}")

        # --------------------------------------------------------------
        # Apply cheap re‑orderings to the current best shape
        # --------------------------------------------------------------
        for reorder_fn in (_anti_symmetric_reorder, _alternating_extreme_reorder):
            a_ord = reorder_fn(best_a)
            val_ord = _eval_fast(a_ord)
            if val_ord < best_val:
                best_val, best_a = val_ord, a_ord.copy()
                print(f"[reorder {reorder_fn.__name__}] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – permutation exploration
    # --------------------------------------------------------------
    if time.time() < deadline - 30:
        # 2‑a : quick random shuffle search
        best_a, best_val = _random_permutation_search(
            best_a, best_val, deadline - 20, tries=2500)

        # 2‑b : extended simulated annealing on permutations
        print("[SA] starting extended simulated annealing")
        best_a, best_val = _simulated_annealing_long(
            best_a, best_val, deadline - 10)

    # --------------------------------------------------------------
    # Stage 3 – final hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # Report the official score (uses the provided evaluate_sequence)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:                           # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # The competition framework expects a plain Python list
    return best_a.tolist()
```
