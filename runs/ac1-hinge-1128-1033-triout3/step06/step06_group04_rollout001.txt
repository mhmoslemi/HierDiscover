def hill_climb(a, cur_val, deadline, rng):
    n = len(a)
    best_a = a.copy()
    best_val = cur_val
    step_factor = 0.1
    T0 = cur_val - 1   # approximate barrier
    if T0 <= 0:
        T0 = 0.1
    # schedule: T = T0 * (remaining_time / total_time)
    start = time.time()
    while time.time() < deadline:
        # choose donor i from high-mass set
        # donors: indices where a[i] > a.mean()
        high = np.where(a > a.mean())[0]
        if high.size == 0:
            break
        i = int(rng.choice(high))
        # choose recipient j from low-mass set
        low = np.where(a < a.mean())[0]
        if low.size == 0:
            j = int(rng.integers(0, n))
        else:
            j = int(rng.choice(low))
        # step fraction random between 0 and step_factor
        fraction = rng.random() * step_factor
        delta = a[i] * fraction
        if delta <= 0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # compute objective
        new_val = objective_value(new_a, n)
        # acceptance
        # temperature decreasing linearly with time
        t = max(1e-6, (deadline - time.time()) / (deadline - start) * T0)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / t):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[hillclimb] new best = {best_val:.6f}")
                # optionally reduce step_factor
                step_factor = max(step_factor * 0.9, 0.01)
        else:
            # maybe shrink step_factor gradually
            step_factor = min(step_factor * 1.01, 0.2)
    return best_a, best_val

"""
Step‑function optimisation for the ``evaluate_sequence`` attack.

The algorithm consists of three stages

1. **Smoothed L‑BFGS** – minimise a log‑sum‑exp surrogate of
   ``max(conv)`` while keeping the vector on the probability simplex.
   The smoothing parameter ε is decreased on a fine log‑scale down to
   1e‑15, and each ε‑stage is solved with a generous L‑BFGS budget.
   Several random restarts are performed.

2. **Annealed mass‑transfer** – starting from the best vector of stage 1,
   repeatedly move a small amount of mass from the pairs that
   contribute most to the current maximal convolution lag to low‑mass
   positions.  Moves are accepted either if they improve the exact
   objective or with a temperature‑controlled probability (simulated
   annealing).  This digs out of local minima that the smooth phase
   cannot escape.

3. **Final reporting** – the candidate is handed to ``evaluate_sequence``
   (which performs its own clipping and normalisation) and the list of
   heights is returned.

All computations use FFT‑based convolution (``O(n log n)``) so that a
size of a few thousand entries comfortably fits into the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full (auto)‑convolution of a real vector.
    Returns an array of length 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise can produce tiny negative values – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (FFT implementation)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the smooth surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv‑max)/ε)

    and its gradient w.r.t. the unconstrained log‑softmax parameters ``w``.
    The vector ``w`` is transformed to a probability vector ``a`` via a
    soft‑max; this guarantees a_i ≥ 0 and Σ a_i = 1.
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift, dtype=np.float64)
    a /= a.sum()

    # full auto‑convolution (length 2·n‑1)
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing
    shifted = (conv - max_conv) / eps
    # When eps is tiny, shifted contains large negatives; exp handles that.
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}   (derived in the analysis)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                  # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]                # middle n entries
    grad_a = 4.0 * n * seg[::-1]                   # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Exact objective (used for evaluation & hill‑climbing)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact value of evaluate_sequence for a vector that already sums to 1.
    (The original function returns 2·n·max(conv) / sum_a², and we enforce
    sum_a = 1.)
    """
    max_b = _full_convolution(a).max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Annealed mass‑transfer hill climbing
# ----------------------------------------------------------------------
def _annealed_mass_transfer(a: np.ndarray,
                            cur_val: float,
                            deadline: float,
                            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Move small chunks of mass from the most damaging pairs (those that
    realise the current maximal lag) to low‑mass entries.
    The move is accepted if it improves the exact objective or with a
    probability exp(-(Δ)/T) where the temperature T decays linearly.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val

    # initialise temperature – a fraction of the current excess over the
    # absolute lower bound 2·n/(2·n‑1) ≈ 1 (the bound is not tight but safe).
    lower_bound = 2.0 * n / (2 * n - 1)
    T0 = max(0.1, (cur_val - lower_bound) * 0.5)

    start = time.time()
    step_frac = 0.15                     # base fraction of donor mass to move

    while time.time() < deadline:
        # -----------------------------------------------------------------
        # Identify the lag that is currently maximal and the contributing pairs
        # -----------------------------------------------------------------
        conv = _full_convolution(a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))                     # one of the tying lags

        # indices i that take part in this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        if contrib.size == 0:
            # extremely unlikely, safeguard
            continue

        # Take the few biggest contributors – these are the “donor” pairs
        top_n = min(4, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]

        donors = [(int(i_range[idx]), int(j_range[idx])) for idx in top_idx]

        # -----------------------------------------------------------------
        # Choose destination slots – low‑mass positions that are not donors
        # -----------------------------------------------------------------
        mean_a = a.mean()
        low = np.where(a < mean_a)[0]
        if low.size < 2 * len(donors):
            low = np.arange(n)                     # fall back to whole set
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        # -----------------------------------------------------------------
        # Propose a new vector
        # -----------------------------------------------------------------
        new_a = a.copy()
        for d, (i, j) in enumerate(donors):
            if i == j:
                # single‑index pair (contributes a_i²)
                delta = new_a[i] * step_frac
                new_a[i] = max(new_a[i] - delta, 0.0)
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                # two‑different indices – take a fraction of the smaller
                delta = min(new_a[i], new_a[j]) * step_frac
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        # renormalise – keeps the ℓ₁‑norm exactly 1 (required for the
        # simplified objective formula).
        s = new_a.sum()
        if s <= 0:
            # pathological – shrink step size and continue
            step_frac = max(step_frac * 0.7, 1e-5)
            continue
        new_a /= s

        new_val = _objective(new_a, n)

        # -----------------------------------------------------------------
        # Acceptance test (annealed)
        # -----------------------------------------------------------------
        t = max(1e-8, (deadline - time.time()) / (deadline - start) * T0)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis‑type acceptance
            if rng.random() < np.exp(-(new_val - cur_val) / t):
                accept = True

        if accept:
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[annealed] new best = {best_val:.6f}")
            # after success we allow a slightly larger step
            step_frac = min(step_frac * 1.02, 0.4)
        else:
            # unsuccessful – reduce step size gently
            step_frac = max(step_frac * 0.95, 1e-5)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a non‑negative step‑function to minimise ``evaluate_sequence``.
    The routine respects a wall‑clock budget (default 1000 s) and returns a
    list of floats.  The best sequence found so far is stored in the global
    variable ``height_sequence_1`` (if present); otherwise a random Dirichlet
    start is used.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin for final evaluation

    # -------------------------------------------------------------
    # Problem size – a moderate size keeps the search affordable.
    # -------------------------------------------------------------
    n = 4000

    # -------------------------------------------------------------
    # Initialise from a previous construction if it exists.
    # -------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # resample by linear interpolation to the target length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        # fall back to a Dirichlet draw (here a simple random vector)
        a0 = rng.random(n)

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()                     # normalise to unit ℓ₁‑norm
    w = np.log(a0 + 1e-12)             # soft‑max parameters for the smooth phase

    best_a = a0.copy()
    best_val = _objective(best_a, n)
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # -------------------------------------------------------------
    # Smoothed continuation (L‑BFGS‑B) – an extended ε‑schedule.
    # -------------------------------------------------------------
    eps_schedule = np.logspace(-2, -15, num=40)   # finer schedule down to 1e‑15

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def fun_and_grad(w_vec):
            return _smooth_obj_and_grad(w_vec, eps, n)

        res = minimize(fun_and_grad,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 1200,
                                'ftol': 1e-13,
                                'gtol': 1e-13,
                                'disp': False})
        w = res.x

        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

    # -------------------------------------------------------------
    # Random restarts – perturb the best vector and re‑optimise.
    # -------------------------------------------------------------
    restart = 0
    max_restarts = 12
    while time.time() < deadline - 5.0 and restart < max_restarts:
        restart += 1
        # perturb the current best in the log‑domain
        w = np.log(best_a + 1e-12) + rng.normal(scale=0.6, size=n)

        # Only the finer tail of the schedule is needed for a quick refinement
        for eps in eps_schedule[-10:]:
            if time.time() >= deadline:
                break
            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)
            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 600,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x

        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = _objective(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[restart {restart}] new best = {best_val:.6f}")

    # -------------------------------------------------------------
    # Final annealed mass‑transfer hill climbing
    # -------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _annealed_mass_transfer(best_a, best_val, deadline, rng)

    # -------------------------------------------------------------
    # Final reporting using the provided evaluator (for the external score)
    # -------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return as a plain list – the framework expects this.
    return best_a.tolist()
