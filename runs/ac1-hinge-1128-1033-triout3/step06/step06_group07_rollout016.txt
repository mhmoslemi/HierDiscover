    v = np.zeros_like(w)
    beta1=0.9; beta2=0.999

    best_a = a0.copy()
    best_val = _objective(a0)

    t = 0

        # compute a

        a = np.exp(w - maxw)

def _full_convolution(x):
    n = x.shape[0]
    out_len = 2*n - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _conv_fft(x, y):
    n = x.shape[0]; m = y.shape[0]
    out_len = n + m - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

def _objective(a):
    return float(2.0 * a.shape[0] * _full_convolution(a).max())

def _adam_opt(a0, n, start, deadline, best_a, best_val, target_val=1.503):
    w = np.log(a0 + 1e-12)
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    iteration = 0
    # schedule parameters
    lr_start, lr_end = 0.2, 0.001
    eps_start, eps_end = 5e-2, 1e-12
    while True:
        now = time.time()
        if now >= deadline:
            break

        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam {iteration:07d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break

        # compute schedules
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end/eps_start) ** progress
        lr = lr_start * (lr_end/lr_start) ** progress

        # compute gradient
        b_full = _full_convolution(a)
        max_b = b_full.max()
        # avoid division by zero
        shifted = (b_full - max_b) / eps
        # ensure stability: if eps is extremely small, exp may overflow; use clip
        # but we can use np.exp(shifted) with clip to large by np.clip(shifted, -700, 700)
        shifted = np.clip(shifted, -700, 700)
        weights = np.exp(shifted)
        p = weights / weights.sum()
        # compute gradient wrt a:
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        # conv_ab length n+n-1; we need segment positions n-1 to 2n-2 inclusive, which length n
        segment = conv_ab[n-1: 2*n-1]
        grad_a = 4.0 * n * segment[::-1]  # shape n

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad_w
        v = beta2 * v + (1 - beta2) * (grad_w ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random perturbation to escape flat regions
        if np.random.rand() < 0.003:
            w += np.random.normal(scale=0.01, size=n)

    return best_a, best_val

def _hillclimb_opt(a_start, n, start, deadline, best_a, best_val, target_val=1.503):

    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.6, 0.02
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    iteration = 0
    eps = 1e-12

    while True:
        now = time.time()
        if now >= deadline:
            break
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1 - progress) + max_frac_end * progress

        # compute contributions to max entry
        k_max = int(np.argmax(b))

        # compute contributions
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        # choose source
        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = np.argmax(a)
        if a[i] <= eps:
            continue

        # target bias: pick index with small contribution
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)  # small contributions get larger weight
        target_weights[i] = 0
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # uniform
            j = np.random.randint(n-1)
            if j >= i:
                j += 1

        # delta
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0:
            continue

        # fast update
        b_new = b.copy()
        # indices for shift addition
        # add to slice j:j+n: 2*delta*a
        start_j = j
        end_j = j + n
        b_new[start_j:end_j] += 2.0 * delta * a
        start_i = i
        end_i = i + n
        b_new[start_i:end_i] -= 2.0 * delta * a
        # quadratic terms:
        idx_2j = 2 * j
        if idx_2j < b_new.shape[0]:
            b_new[idx_2j] += delta * delta
        idx_2i = 2 * i
        if idx_2i < b_new.shape[0]:
            b_new[idx_2i] += delta * delta
        idx_ij = i + j
        if idx_ij < b_new.shape[0]:
            b_new[idx_ij] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # accept? 
        # Simulated annealing temperature
        T_start = 0.01
        T_end = 1e-6
        T = T_start * (T_end/T_start) ** progress

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # probability
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:07d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            # random perturb and reset
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum
            else:
                a = np.ones(n) / n
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:07d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

def _initial_shape(n):
    best_a=None; best_val=np.inf
    x=np.arange(n, dtype=np.float64)
    eps=1e-12
    for gamma in np.linspace(0.0, 4.0, 401):
        denom = (x + 0.5) * (n - x - 0.5)
        a = np.power(denom + eps, -gamma)

"""
Multi‑scale optimizer for the step‑function bound.

The algorithm works in two resolution levels:
* a modest problem (≈2000 entries) is explored quickly with a parametric
  sweep, Adam optimisation on a smooth max‑approximation and a
  targeted hill‑climber;
* the obtained shape is interpolated to the final resolution (≈8000
  entries) and refined again with the same two‑phase scheme.

Both phases use only NumPy and the provided ``evaluate_sequence`` wrapper.
The code respects the 1000 s budget and returns the best sequence found.
"""

import time
import math
import numpy as np

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a sequence with Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric linear “tent” shape a_i ∝ min(i+1, n‑i)."""
    a = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(a, n + 1 - a)          # 1,2,…,⌈n/2⌉,…,2,1
    a /= a.sum()
    return a


def _initial_shape(n: int):
    """Quick parametric sweep over a U‑shaped power law and a few classics."""
    best_a = None
    best_val = np.inf
    x = np.arange(n, dtype=np.float64)
    eps = 1e-12

    # U‑shaped family a_i ∝ ((i+½)(n‑i‑½))^{‑γ}
    for gamma in np.linspace(0.0, 4.0, 401):
        denom = (x + 0.5) * (n - x - 0.5)
        a = np.power(denom + eps, -gamma)
        s = a.sum()
        if s == 0:
            continue
        a = a / s
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # triangular (tent) shape
    a_tri = _triangular_shape(n)
    val_tri = _objective(a_tri)
    if val_tri < best_val:
        best_val = val_tri
        best_a = a_tri.copy()

    # Gaussian‑like bump
    sigma = n / 8.0
    xv = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a_gauss = np.exp(-0.5 * (xv / sigma) ** 2)
    a_gauss /= a_gauss.sum()
    val_gauss = _objective(a_gauss)
    if val_gauss < best_val:
        best_val = val_gauss
        best_a = a_gauss.copy()

    return best_a, best_val


def _adam_opt(a0: np.ndarray,
              n: int,
              start: float,
              deadline: float,
              best_a: np.ndarray,
              best_val: float,
              target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth log‑sum‑exp approximation of the maximum."""
    # Soft‑max parametrisation (a = softmax(w))
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.2, 0.001
    eps_start, eps_end = 5e-2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    iteration = 0
    no_improve = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # Soft‑max → a (lies on the simplex)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {iteration:07d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # Schedules (progress in the current phase)
        prog = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # Smooth max via log‑sum‑exp
        b = _full_convolution(a)
        max_b = b.max()
        shifted = (b - max_b) / eps
        shifted = np.clip(shifted, -700, 700)          # avoid overflow
        weights = np.exp(shifted)
        p = weights / weights.sum()                    # soft‑max weights

        # Gradient of the smoothed objective w.r.t. a
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)                 # length 2·n‑1
        segment = conv_ab[n - 1: 2 * n - 1]            # n entries
        grad_a = 4.0 * n * segment[::-1]

        # Chain rule through the soft‑max (w → a)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # Tiny random jitter occasionally helps to escape exact plateaus
        if np.random.rand() < 0.003:
            w += np.random.normal(scale=0.01, size=n)

    return best_a, best_val


def _hillclimb_opt(a_start: np.ndarray,
                   n: int,
                   start: float,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Targeted hill‑climbing / simulated‑annealing.
    Mass is transferred from an index that contributes heavily to the current
    maximal convolution entry towards an index that contributes little.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.6, 0.02
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    iteration = 0
    eps = 1e-12

    # annealing schedule (temperature)
    T_start, T_end = 0.01, 1e-7

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # location of the current peak of the convolution
        k_max = int(np.argmax(b))

        # contributions of each index to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ----- pick source (high contribution) -----
        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ----- pick target (low contribution) -----
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0                     # avoid self‑transfer
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # fallback: uniform among all other indices
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # Incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        idx_2j = 2 * j
        if idx_2j < b_new.shape[0]:
            b_new[idx_2j] += delta * delta
        idx_2i = 2 * i
        if idx_2i < b_new.shape[0]:
            b_new[idx_2i] += delta * delta
        idx_ij = i + j
        if idx_ij < b_new.shape[0]:
            b_new[idx_ij] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # simulated‑annealing acceptance
        T = T_start * (T_end / T_start) ** progress
        if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
            # accept
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:07d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – gentle random perturbation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:07d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val

"""
Multi‑scale optimizer for the step‑function bound.

The algorithm works in two resolution levels:
* a modest problem (≈2000 entries) is explored quickly with a parametric
  sweep, Adam optimisation on a smooth max‑approximation and a
  targeted hill‑climber;
* the obtained shape is interpolated to the final resolution (≈8000
  entries) and refined again with the same two‑phase scheme.

Both phases use only NumPy and the provided ``evaluate_sequence`` wrapper.
The code respects the 1000 s budget and returns the best sequence found.
"""

import time
import math
import numpy as np

def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``x`` with itself using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a sequence with Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


def _triangular_shape(n: int) -> np.ndarray:
    """Symmetric linear “tent” shape a_i ∝ min(i+1, n‑i)."""
    a = np.arange(1, n + 1, dtype=np.float64)
    a = np.minimum(a, n + 1 - a)
    a /= a.sum()
    return a


def _initial_shape(n: int):
    """Quick parametric sweep over a U‑shaped power law and a few classics."""
    best_a = None
    best_val = np.inf
    x = np.arange(n, dtype=np.float64)
    eps = 1e-12

    # U‑shaped family a_i ∝ ((i+½)(n‑i‑½))^{‑γ}
    for gamma in np.linspace(0.0, 4.0, 401):
        denom = (x + 0.5) * (n - x - 0.5)
        a = np.power(denom + eps, -gamma)
        s = a.sum()
        if s == 0:
            continue
        a = a / s
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    # triangular (tent) shape
    a_tri = _triangular_shape(n)
    val_tri = _objective(a_tri)
    if val_tri < best_val:
        best_val = val_tri
        best_a = a_tri.copy()

    # Gaussian‑like bump (optional)
    sigma = n / 8.0
    xv = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a_gauss = np.exp(-0.5 * (xv / sigma) ** 2)
    a_gauss /= a_gauss.sum()
    val_gauss = _objective(a_gauss)
    if val_gauss < best_val:
        best_val = val_gauss
        best_a = a_gauss.copy()

    return best_a, best_val


def _adam_opt(a0: np.ndarray,
              n: int,
              start: float,
              deadline: float,
              best_a: np.ndarray,
              best_val: float,
              target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth log‑sum‑exp approximation of the maximum."""
    w = np.log(a0 + 1e-12)                     # soft‑max parametrisation
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.2, 0.001
    eps_start, eps_end = 5e-2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    iteration = 0
    no_improve = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # a = softmax(w)
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {iteration:07d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedules for the current phase
        prog = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b = _full_convolution(a)
        max_b = b.max()
        shifted = (b - max_b) / eps
        shifted = np.clip(shifted, -700, 700)          # protect against overflow
        weights = np.exp(shifted)
        p = weights / weights.sum()

        # gradient w.r.t. a of the smoothed objective
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)                 # length 2·n‑1
        segment = conv_ab[n - 1: 2 * n - 1]            # n entries
        grad_a = 4.0 * n * segment[::-1]

        # back‑propagation through the soft‑max (w → a)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional tiny jitter helps to leave plateaus
        if np.random.rand() < 0.003:
            w += np.random.normal(scale=0.01, size=n)

    return best_a, best_val


def _hillclimb_opt(a_start: np.ndarray,
                   n: int,
                   start: float,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Targeted hill‑climbing with a simple simulated‑annealing acceptance.
    Mass is moved from an index that contributes heavily to the current
    maximal convolution entry towards an index that contributes little.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.6, 0.02
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    iteration = 0
    eps = 1e-12

    # temperature schedule for occasional uphill moves
    T_start, T_end = 0.01, 1e-7

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # index of the current convolution peak
        k_max = int(np.argmax(b))

        # contribution of each index to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ----- source (high contribution) -----
        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ----- target (low contribution) -----
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to transfer
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        idx_2j = 2 * j
        if idx_2j < b_new.shape[0]:
            b_new[idx_2j] += delta * delta
        idx_2i = 2 * i
        if idx_2i < b_new.shape[0]:
            b_new[idx_2i] += delta * delta
        idx_ij = i + j
        if idx_ij < b_new.shape[0]:
            b_new[idx_ij] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # simulated‑annealing acceptance
        T = T_start * (T_end / T_start) ** progress
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:07d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – gentle random perturbation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:07d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val

        prog = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog

        # smooth max via log‑sum‑exp
        b = _full_convolution(a)
        max_b = b.max()
        shifted = (b - max_b) / eps
        shifted = np.clip(shifted, -700, 700)          # protect against overflow
        weights = np.exp(shifted)
        p = weights / weights.sum()

        # gradient of the smoothed objective wrt a
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)                 # length 2·n‑1
        segment = conv_ab[n - 1: 2 * n - 1]            # n entries
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the soft‑max (w → a)
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional tiny jitter to escape plateaus
        if np.random.rand() < 0.003:
            w += np.random.normal(scale=0.01, size=n)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing with simulated‑annealing acceptance
# ----------------------------------------------------------------------
def _hillclimb_opt(a_start: np.ndarray,
                   n: int,
                   start: float,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float,
                   target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Fast local search that transfers a small amount of mass while keeping Σa = 1.
    Source indices are chosen proportionally to their contribution to the
    current maximal convolution entry; targets are biased towards indices that
    contribute little to that entry.  Occasionally uphill moves are accepted
    according to a simple annealing schedule.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.6, 0.02
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    iteration = 0
    eps = 1e-12

    # annealing schedule
    T_start, T_end = 0.01, 1e-7

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # location of the current peak of the convolution
        k_max = int(np.argmax(b))

        # contributions of each index to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ----- source (high contribution) -----
        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ----- target (low contribution) -----
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        idx_2j = 2 * j
        if idx_2j < b_new.shape[0]:
            b_new[idx_2j] += delta * delta
        idx_2i = 2 * i
        if idx_2i < b_new.shape[0]:
            b_new[idx_2i] += delta * delta
        idx_ij = i + j
        if idx_ij < b_new.shape[0]:
            b_new[idx_ij] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # simulated‑annealing acceptance
        T = T_start * (T_end / T_start) ** progress
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if np.random.rand() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill {iteration:07d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – gentle random perturbation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:07d}] stagnation – perturb")

        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Multi‑scale optimiser for the step‑function problem.
    Returns a non‑negative list whose sum is (approximately) 1 and
    minimises ``evaluate_sequence`` under the given 1000 s budget.
    """
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin for final call

    # --------------------------------------------------------------
    # Optional: reuse a previously stored solution, if available.
    # --------------------------------------------------------------
    prior = None
    try:
        prior = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        prior = np.maximum(prior, 0.0)
        prior = prior / prior.sum()
        print("[init] loaded prior solution")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 0 – small problem (≈2000 entries) to obtain a good shape.
    # --------------------------------------------------------------
    n_small = 2000
    if prior is not None and prior.shape[0] == n_small:
        a_small = prior.copy()
        best_val_small = _objective(a_small)
        print(f"[init] using prior (n={n_small}), value={best_val_small:.6f}")
    else:
        a_small, best_val_small = _initial_shape(n_small)
        print(f"[init] parametric sweep (n={n_small}) best={best_val_small:.6f}")

    # --------------------------------------------------------------
    # Small‑scale Adam optimisation (≈15 % of the total budget)
    # --------------------------------------------------------------
    phase_start = time.time()
    small_adam_deadline = start_time + 0.15 * budget_s
    a_small, best_val_small = _adam_opt(a_small, n_small,
                                        start=phase_start,
                                        deadline=small_adam_deadline,
                                        best_a=a_small,
                                        best_val=best_val_small,
                                        target_val=1.503)

    # --------------------------------------------------------------
    # Small‑scale hill‑climbing (≈15 % more of the total budget)
    # --------------------------------------------------------------
    phase_start = time.time()
    small_hill_deadline = start_time + 0.30 * budget_s
    a_small, best_val_small = _hillclimb_opt(a_small, n_small,
                                             start=phase_start,
                                             deadline=small_hill_deadline,
                                             best_a=a_small,
                                             best_val=best_val_small,
                                             target_val=1.503)

    # --------------------------------------------------------------
    # Upsample to the final resolution (≈8000 entries).
    # --------------------------------------------------------------
    n_large = 8000
    old_x = np.linspace(0.0, 1.0, n_small)
    new_x = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(new_x, old_x, a_small)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()
    best_val_large = _objective(a_large)
    print(f"[init] up‑sampled to n={n_large}, value={best_val_large:.6f}")

    # --------------------------------------------------------------
    # Large‑scale Adam optimisation (≈25 % of the total budget)
    # --------------------------------------------------------------
    phase_start = time.time()
    large_adam_deadline = start_time + 0.55 * budget_s
    a_large, best_val_large = _adam_opt(a_large, n_large,
                                        start=phase_start,
                                        deadline=large_adam_deadline,
                                        best_a=a_large,
                                        best_val=best_val_large,
                                        target_val=1.503)

    # --------------------------------------------------------------
    # Large‑scale hill‑climbing (uses the remaining time)
    # --------------------------------------------------------------
    phase_start = time.time()
    a_large, best_val_large = _hillclimb_opt(a_large, n_large,
                                             start=phase_start,
                                             deadline=deadline,
                                             best_a=a_large,
                                             best_val=best_val_large,
                                             target_val=1.503)

    # --------------------------------------------------------------
    # Final evaluation via the provided secure wrapper.
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_large.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return a_large.tolist()
