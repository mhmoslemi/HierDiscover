"""
propose_candidate
-----------------

Search for a non‑negative step function that minimizes

    2 * n * max(conv(a, a)) / sum(a)^2

The algorithm works in three stages and is tuned to make good
progress within the 1000 s time limit.

1. **Warm‑start generation** – a large pool of parametric families
   (exponential, power‑law, Gaussian, half‑sine, triangular) is
   sampled; the best candidate (lowest objective) is kept.

2. **Smoothed gradient descent** – replace the hard max by a
   soft‑max (log‑sum‑exp) with a temperature `eps`.  The resulting
   smooth objective is differentiable; a momentum update followed by
   projection onto the probability simplex keeps the iterate feasible.
   The schedule slowly drives `eps` and the step size to zero,
   allowing the iterate to settle near a (local) optimum of the true
   objective.

3. **Targeted simulated‑annealing** – the current maximum‑convolution
   index `k*` is identified.  The pair `(i, k*‑i)` that contributes the
   most to that peak is weakened by moving a small amount of mass to a
   location that contributes little.  Moves are accepted according to a
   standard Metropolis rule with a linearly decreasing temperature.
   This fine‑tunes the solution beyond what the gradient phase can
   reach.

All intermediate scores are printed; the best sequence seen over the
whole run is returned.  The routine respects the `budget_s` deadline,
clamps the final numbers to the `[0, 1000]` range and therefore works
directly with `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution via FFT (real).  Used many times, therefore a fast
# implementation is important.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()
    FA = np.fft.rfft(a, n=nfft)
    FB = np.fft.rfft(b, n=nfft)
    FC = FA * FB
    conv = np.fft.irfft(FC, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}.
# Duchi et al. (2008) algorithm, O(n log n).
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:                     # should not happen for non‑negative v
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective for a vector that is already normalised (sum == 1).
# Returns 2 * n * max(conv(a, a)).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.size
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Some parametric families used for the warm start.
# ----------------------------------------------------------------------
def _exp_shape(lam: float, n: int) -> np.ndarray:
    """Exponential decay, normalised to sum 1."""
    x = np.arange(n, dtype=np.float64)
    unnorm = np.exp(-lam * x)
    return unnorm / unnorm.sum()


def _pow_shape(p: float, n: int) -> np.ndarray:
    """Power‑law (x+1)^{-p}, normalised."""
    x = np.arange(n, dtype=np.float64)
    unnorm = (x + 1.0) ** (-p)
    return unnorm / unnorm.sum()


def _gauss_shape(sigma: float, n: int) -> np.ndarray:
    """Gaussian centred at the middle, normalised."""
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    unnorm = np.exp(-0.5 * (x / sigma) ** 2)
    return unnorm / unnorm.sum()


def _half_sine_shape(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    unnorm = np.sin(np.pi * (x + 0.5) / n)          # always positive
    return unnorm / unnorm.sum()


def _triangular_shape(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    unnorm = np.minimum(x, n - 1 - x) + 1.0
    return unnorm / unnorm.sum()


def _catalan_shape(n: int) -> np.ndarray:
    catalan = np.empty(n, dtype=np.float64)
    cur = 1.0
    catalan[0] = cur
    for k in range(1, n):
        cur = cur * (2.0 * k - 1) / (2.0 * k)
        catalan[k] = cur
    return catalan / catalan.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a sequence (list of floats) that (approximately) minimizes the
    evaluation function.  The routine respects the given time budget.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 5.0            # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 6000 is a comfortable compromise between
    # optimisation power and per‑iteration cost.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Warm start: sample many classic shapes and keep the best.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # allow reuse of a previously found sequence
    if 'height_sequence_1' in globals():
        try:
            tmp = np.asarray(height_sequence_1, dtype=np.float64)
            if tmp.shape[0] != n:
                if tmp.shape[0] > n:
                    tmp = tmp[:n]
                else:
                    tmp = np.pad(tmp, (0, n - tmp.shape[0]))
            tmp = np.maximum(0.0, tmp)
            s = tmp.sum()
            if s > 0:
                tmp = tmp / s
                val = _objective(tmp)
                best_seq, best_val = tmp, val
                print("[warm] reused previous sequence, score={:.12f}".format(val))
        except Exception:
            pass

    # pool of parametric families
    shapes = []

    # exponential – sample λ on a log‑uniform grid that covers the
    # range where the theory predicts good values (t=λ·n≈0.5…2)
    for _ in range(30):
        lam = rng.uniform(5e-5, 5e-3)               # λ ≈ 2e‑4 gives t≈1.2
        shapes.append(_exp_shape(lam, n))

    # power‑law
    for _ in range(20):
        p = rng.uniform(0.2, 2.5)
        shapes.append(_pow_shape(p, n))

    # Gaussian
    for _ in range(10):
        sigma = rng.uniform(n / 10.0, n / 2.0)
        shapes.append(_gauss_shape(sigma, n))

    # deterministic shapes
    shapes.append(_half_sine_shape(n))
    shapes.append(_triangular_shape(n))
    shapes.append(_catalan_shape(n))

    # random convex combination of the above (20 tries)
    for _ in range(20):
        w = rng.random(len(shapes))
        w /= w.sum()
        combo = sum(wi * sh for wi, sh in zip(w, shapes))
        combo = np.maximum(combo, 0.0)
        combo /= combo.sum()
        shapes.append(combo)

    # evaluate all candidates
    for idx, cand in enumerate(shapes):
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print("[warm] new best #{:02d} score={:.12f}".format(idx, val))

    # ------------------------------------------------------------------
    # (1) Smoothed gradient descent (soft‑max approximation of max).
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s
    a = best_seq.copy()
    eta = 0.3                     # initial learning rate
    eta_min = 1e-6
    eta_decay = 0.9995
    eps = max(1e-4, 0.5 * _full_convolution_fft(a, a).max())
    eps_min = 1e-7
    eps_decay = 0.9997
    momentum = 0.95
    vel = np.zeros_like(a)

    it = 0
    stagn = 0
    while time.time() < phase1_end:
        it += 1
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print("[gd] iter {:7d}  score={:.12f}  eps={:.2e}  eta={:.2e}".format(
                it, best_val, eps, eta))
        else:
            stagn += 1

        # smooth max‑approximation
        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if not np.isfinite(w_sum) or w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        rev_a = a[::-1]
        # gradient of Σ w_k·(a*a)_k  = 2 * (rev_a * w)[shifted]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        vel = momentum * vel + (1.0 - momentum) * g
        a_new = a - eta * vel
        a = _project_simplex(a_new)

        # schedules
        eta = max(eta_min, eta * eta_decay)
        eps = max(eps_min, eps * eps_decay)

        # occasional stochastic kick when stuck
        if stagn > 1500:
            noise = rng.normal(scale=0.003, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            eta = max(eta_min, eta * 1.2)
            print("[gd] perturb at iter", it)

    # ------------------------------------------------------------------
    # (2) Exact max‑sub‑gradient descent with Adam.
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.80 * budget_s
    a = best_seq.copy()
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.12
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    it = 0
    stagn = 0
    while time.time() < phase2_end:
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv

        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print("[adam] iter {:7d}  score={:.12f}".format(it, best_val))
        else:
            stagn += 1

        # exact sub‑gradient of the max (average over all active indices)
        tol = 1e-14
        active = np.flatnonzero(conv >= max_conv - tol)
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        else:
            w[np.argmax(conv)] = 1.0

        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_candidate = a - step
        a = _project_simplex(a_candidate)

        if stagn > 2500:
            # random jitter to escape plateaus
            noise = rng.normal(scale=0.0015, size=n)
            a = a + noise
            a = _project_simplex(a)
            stagn = 0
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta2_pow = 1.0
            print("[adam] perturb at iter", it)

    # ------------------------------------------------------------------
    # (3) Targeted simulated‑annealing fine‑tuning.
    # ------------------------------------------------------------------
    cur = a.copy()
    cur_score = _objective(cur)
    if cur_score < best_val:
        best_val, best_seq = cur_score, cur.copy()
        print("[sa] initial improvement score={:.12f}".format(best_val))

    T0 = 2e-3
    Tmin = 1e-7
    it_sa = 0
    while time.time() < deadline:
        it_sa += 1
        T = max(Tmin, T0 * (1.0 - it_sa / 250_000))

        conv = _full_convolution_fft(cur, cur)
        k_max = int(conv.argmax())                     # index of the peak
        # find the pair (i,j) producing the largest contribution to that peak
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        i_range = np.arange(i_low, i_high + 1)
        j_range = k_max - i_range
        contrib = cur[i_range] * cur[j_range]

        # index that contributes most
        idx_max = np.argmax(contrib)
        i_star = i_range[idx_max]
        j_star = j_range[idx_max]          # partner of i_star (j_star may equal i_star)

        # pick a destination index that currently contributes little to the peak
        # (simple heuristic: choose an index with smallest value)
        low_candidates = np.where(cur < cur.mean())[0]
        if low_candidates.size == 0:
            low_candidates = np.arange(n)
        j_dest = rng.choice(low_candidates)

        # amount to move – a small fraction of the larger of the two partners
        max_transfer = min(cur[i_star], cur[j_star]) * 0.5
        delta = rng.uniform(0.0, max_transfer)

        new = cur.copy()
        # remove from both members of the dominant pair proportionally
        new[i_star] -= delta
        new[j_star] -= delta
        # add the total transferred mass to the destination (keeping sum fixed)
        new[j_dest] += 2 * delta

        # safety: re‑project onto the simplex
        new = _project_simplex(new)

        new_score = _objective(new)

        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            cur = new
            cur_score = new_score
            if cur_score < best_val - 1e-12:
                best_val, best_seq = cur_score, cur.copy()
                print("[sa] iter {:7d}  new_best={:.12f}".format(it_sa, best_val))

        # occasional global jitter to avoid long stagnation
        if it_sa % 30000 == 0:
            cur = cur + rng.normal(scale=0.001, size=n)
            cur = _project_simplex(cur)
            cur_score = _objective(cur)

    # ------------------------------------------------------------------
    # Return the best sequence in the format required by `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    try:
        final_score = evaluate_sequence(final_seq)
        print("[result] final_score={:.12f}  length={}".format(final_score, len(final_seq)))
    except Exception:
        print("[result] evaluate_sequence not available")

    return final_seq
