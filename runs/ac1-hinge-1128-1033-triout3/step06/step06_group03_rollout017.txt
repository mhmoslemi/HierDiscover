"""
Propose a step‑function (non‑negative heights) that approximately minimises
``evaluate_sequence``.  The method combines three ideas:

1. **Exponentiated gradient descent on a smooth approximation of
   the max‑convolution**.  The objective
   `f(a) = 2·n·max(conv(a,a))` (the sum of the sequence is kept equal
   to 1, which makes the objective scale‑invariant).  The non‑smooth
   ``max`` is replaced by a soft‑max with temperature ``α``; the gradient
   of this surrogate is computed efficiently by a single FFT‑based
   convolution.  An exponentiated‑gradient update keeps the iterates on
   the probability simplex without the need for an explicit projection.

2. **Adaptive schedule** – the temperature ``α`` is slowly increased so
   the optimisation first follows the smooth objective and later focuses
   on the true max.  The learning rate ``η`` is adapted depending on
   whether the last step was successful.

3. **Guided hill‑climbing refinement** – after the main optimisation a
   lightweight hill‑climbing routine moves a small amount of mass away
   from the most offending lag (the lag where the convolution attains its
   maximum).  This routine is the same as in the previous state‑of‑the‑art
   code but is applied to the best solution found by the gradient phase.

The whole routine respects the 1000 s budget, continually prints progress
and finally returns the best list of heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full autocorrelation of a real vector ``x`` (length 2·len‑1).
    Uses a real FFT for speed.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative round‑off errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008) – kept for fallback
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Reduce the exact objective by moving mass away from the most offending
    convolution lag.  The routine stops when the time budget expires.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # pick two destination indices that are not recreating the same lag
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # perform a feasible mass transfer
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        # exact evaluation (fast shortcut is also possible)
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val
