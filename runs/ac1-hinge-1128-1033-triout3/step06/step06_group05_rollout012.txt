    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Determine n

        new_x = np.linspace(0,1,n)
        a = np.interp(new_x, old_x, old)
    except Exception:
        a = np.random.rand(n)

    # Normalize to sum 1 and make non-negative
    a = np.maximum(a,0)
    a /= a.sum()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5  # base learning rate, will be decayed

    best_a = a.copy()
    best_val = float('inf')
    iter_num = 0

    # Beta schedule (softmax temperature)
    beta_start = 0.5
    beta_end = 2000.0

    while time.time() < deadline:
        # compute convolution
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Iter {iter_num:7d}] new best val={best_val:.6f}")

        # determine schedules
        progress = (time.time() - start) / (deadline - start)
        # Learning rate decay linear
        lr_t = lr * (1.0 - progress) + 1e-4 * progress
        # Beta increase
        beta = beta_start + (beta_end - beta_start) * progress

        # compute softmax weights
        # shift by max for numerical stability
        b_shifted = b - max_b
        w = np.exp(beta * b_shifted)
        w_sum = w.sum()
        if w_sum == 0:
            # fallback uniform weights
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w = w / w_sum

        # gradient via convolution
        a_rev = a[::-1]
        conv_res = _conv_fft(w, a_rev)   # length 3n-2
        # slice middle part (indices n-1 to 2n-2 inclusive)

        iter_num += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)

        total = a_candidate.sum()

"""
Propose a step‑function with a (near)‑minimal value of
`evaluate_sequence`.  The implementation mixes three ideas:

1️⃣  **Smooth max (soft‑max) descent on the exact objective** –  
    we minimise the smooth approximation  
    `Lβ(a)= (1/β)·log Σ_k exp(β·(a*a)_k)` with an increasing temperature β.  
    The gradient of Lβ is a weighted sum of the convolution gradients and can be
    computed with two FFT‑based convolutions.  An Adam optimiser (with a
    cheap simplex projection) drives the iterates.

2️⃣  **Progressive β‑schedule** – it starts with a low β (behaviour close to the
    L²‑flattening used previously) and ends with a high β so that the
    optimisation finally focuses on the true maximum of the autoconvolution.

3️⃣  **Deterministic greedy hill‑climbing** – after the smooth phase a local
    search moves a small amount of mass from the index that contributes most
    to the current worst convolution value to the index that contributes the
    least.  The full convolution is updated in O(n) time, and several
    candidate step sizes are tried (exact line‑search on a small discrete set).

All phases keep the vector non‑negative and normalised (Σa=1), which does not
change the evaluation because `evaluate_sequence` is scale‑invariant.
The routine respects the 1000 s wall‑time limit and prints intermediate
bounds for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (reuse the fast routines from the previous
# submission – they are fast and numerically stable).
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Simple (cheap) projection onto the probability simplex:
#   1. clip negatives,
#   2. renormalise to sum = 1.
# This is sufficient for a gradient‑based method because the objective is
# scale‑invariant; an exact Euclidean projection is not required.
# ----------------------------------------------------------------------
def _project_simplex_quick(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        return v / s
    # fallback – uniform distribution
    n = v.shape[0]
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Phase 1 – Adam optimiser on the smooth max (soft‑max) surrogate.
# ----------------------------------------------------------------------
def _adam_softmax_phase(a: np.ndarray,
                        n: int,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> tuple[np.ndarray, float]:
    """Run Adam on Lβ(a) = (1/β)·log Σ exp(β·b) where b = a*a."""
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_initial = 0.7                # fairly aggressive – will be decayed
    lr_final = 5e-4

    # temperature schedule for the soft‑max
    beta_start = 0.5
    beta_end = 2000.0

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        # full convolution and current objective (max(b) is the true objective)
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")

        # ---------- schedules ----------
        progress = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - time.time()))
        # linear decay of learning‑rate
        lr = lr_initial * (1.0 - progress) + lr_final * progress
        # linear increase of temperature β
        beta = beta_start + (beta_end - beta_start) * progress

        # ---------- soft‑max weights ----------
        # shift by max for numerical stability
        shifted = b - cur_max
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:               # very unlikely, but protect against overflow
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ---------- gradient ----------
        # grad(Lβ) = 2·conv(w, a_rev)[n‑1:2·n‑1]
        a_rev = a[::-1]
        conv_res = _conv_fft(w, a_rev)          # length 3·n‑2
        grad = 2.0 * conv_res[n - 1:2 * n - 1]

        # ---------- Adam update ----------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- projection onto the simplex ----------
        a = _project_simplex_quick(a_candidate)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Deterministic greedy hill‑climbing (exact max reduction).
# ----------------------------------------------------------------------
def _greedy_hillclimb(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> np.ndarray:
    """
    Local search that moves a fraction of the mass from the index that
    contributes most to the current worst convolution value to the index
    that contributes the least.  Several candidate step sizes are tried and
    the best improvement is taken.  The full convolution is updated in O(n)
    time using the bilinear identity.
    """
    # pre‑compute the full convolution once
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    # stagnation handling
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    it = 0
    while time.time() < deadline:
        k = max_idx

        # contribution of each a_j to b_k (i.e. ∂b_k/∂a_j)
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        # avoid degenerate source/target
        if source == target or a[source] <= 1e-14:
            # pick a random non‑zero source
            nz = np.where(a > 1e-14)[0]
            if nz.size == 0:
                break
            source = int(np.random.choice(nz))
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        # try several fractions of the source mass
        best_delta = 0.0
        best_new_val = cur_val
        best_b = None
        best_max_idx = max_idx

        for frac in (0.05, 0.12, 0.25, 0.40, 0.55, 0.70):
            delta = a[source] * frac
            if delta <= 0.0:
                continue

            # incremental update of the full convolution (O(n))
            b_new = b.copy()

            # cross‑terms 2·δ·a
            b_new[target:target + n] += 2.0 * delta * a
            b_new[source:source + n] -= 2.0 * delta * a

            # pure‑δ² terms
            b_new[2 * target] += delta * delta
            b_new[2 * source] += delta * delta
            b_new[source + target] -= 2.0 * delta * delta

            new_max = b_new.max()
            new_val = 2.0 * n * new_max

            if new_val < best_new_val - 1e-13:
                best_delta = delta
                best_new_val = new_val
                best_b = b_new
                best_max_idx = int(np.argmax(b_new))

        if best_delta > 0.0:
            # accept the best move found
            a[source] -= best_delta
            a[target] += best_delta
            b = best_b
            cur_val = best_new_val
            cur_max = b.max()
            max_idx = best_max_idx
            no_improve = 0

            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # occasional perturbation if we are stuck for a while
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.003, size=n)
            a = _project_simplex_quick(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")
        it += 1

    return best_a
