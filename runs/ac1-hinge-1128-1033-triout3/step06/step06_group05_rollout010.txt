import numpy as np
import time

def _full_convolution(x):
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv

def _conv_fft(x, y):
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

def _grad_softmax_max(a, beta):
    # a: 1D numpy array, sum=1, nonnegative.
    b = _full_convolution(a)  # length 2n-1
    b_max = b.max()
    # compute softmax weights
    # to avoid overflow: shift by max
    exp_term = np.exp(beta * (b - b_max))

    # compute via convolution of a with reversed w
    w_rev = w[::-1]
    conv = _conv_fft(a, w_rev)  # length 3n-2
    L = len(w)
    start = L - 1
    part = conv[start - np.arange(len(a))]
    # gradient of smoothed max w.r.t a
    grad = a + part

def _adam_phase(a_init, n, deadline):
    # a_init is array sum to 1.
    # We'll param via logits z.
    # ensures positivity
    eps = 1e-12
    z = np.log(a_init + eps)  # initial logits
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    alpha = 0.05  # base learning rate
    best_a = a_init.copy()
    best_val = 2.0 * n * _full_convolution(a_init).max()
    start = time.time()
    it = 0
    no_improve = 0
    # schedule beta for softmax
    beta_start = 0.5
    beta_end = 30.0
    while time.time() < deadline:
        # compute softmax a from logits
        max_z = z.max()
        expz = np.exp(z - max_z)
        a = expz / expz.sum()
        # schedule beta linearly based on elapsed fraction
        elapsed = time.time() - start
        total = deadline - start
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)
        # compute gradient wrt a
        grad_a, b = _grad_softmax_max(a, beta)
        cur_val = 2.0 * n * b.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {it:8d}] best approx = {best_val:.6f} (beta={beta:.2f})")
        else:
            no_improve += 1

        # compute gradient w.r.t logits using chain rule
        dot = np.dot(a, grad_a)
        gz = a * (grad_a - dot)  # shape (n,)

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * gz
        v = beta2 * v + (1 - beta2) * (gz * gz)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        # gradient descent: move against gradient
        z -= step

        if no_improve > 2000:
            # jitter to escape plateau
            z += np.random.normal(scale=0.01, size=n)
            no_improve = 0

    return best_a, best_val

def _greedy_hillclimb(a, n, deadline, best_a, best_val):
    # compute initial convolution and max index
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 200000
    no_improve = 0
    it = 0
    while time.time() < deadline:
        # contributions for current worst index
        k = max_idx
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-14:
            # pick any positive source
            candidates = np.where(a > 1e-14)[0]
            if candidates.size == 0:
                break
            source = int(np.random.choice(candidates))

        delta = a[source] * 0.25 * np.random.rand()
        if delta <= 0.0:
            continue

        # incremental update of convolution
        b_new = b.copy()
        # adjust cross terms
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        # pure δ² terms
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:8d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb best solution a bit and restart scroll
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            # ensure nonnegative and renormalize
            a = np.maximum(a, 0.0)
            total = a.sum()
            if total > 0:
                a /= total
            else:
                a = np.full(n, 1.0 / n)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it}] stagnation – perturbing")
        it += 1

    return best_a

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step‑function minimizing `evaluate_sequence`.
    The algorithm runs two phases:
      1. A soft‑max Adam optimisation that minimises a smoothed version of
         the max autoconvolution using a gradually increasing temperature.
      2. A deterministic greedy hill‑climb that moves small amounts of mass
         from the most responsible index to the least.
    The best vector found is finally evaluated and returned as a list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a safety margin
    deadline = start_time + budget_s - 0.5

    # choose problem size (larger n gives more freedom but costs more)
    n = 6000

    # initialise a (try to reuse previous construction)
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # pylint: disable=undefined-variable
        old_len = old.shape[0]
        new_x = np.linspace(0.0, 1.0, n)
        old_x = np.linspace(0.0, 1.0, old_len)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
    except Exception:
        a0 = np.random.rand(n)

    a0 = a0 / a0.sum()  # enforce simplex

    t_mid = start_time + 0.45 * (deadline - start_time)   # 45% for Adam
    # Phase 1 – Adam with smooth max
    best_a, best_val = _adam_phase(a0, n, t_mid)

    # Phase 2 – Greedy hill‑climbing with remaining budget
    best_a = _greedy_hillclimb(best_a, n, deadline, best_a, best_val)

    # final evaluation via provided wrapper

"""
Search for a non‑negative step‑function minimising `evaluate_sequence`.

Algorithm
---------

1. **Parameterisation** – the vector `a` lives on the simplex.  We store
   logits `z` and obtain `a = softmax(z)`.  This guarantees non‑negativity
   and Σa = 1 without an explicit projection.

2. **Phase A – Adam optimisation with a smoothed max**  
   The exact objective is `2·n·max(a*a)`.  We replace `max` by a
   temperature‑controlled soft‑max:
        Lβ(a) = (1/β)·log Σ exp(β·(a*a)ₖ)
   whose gradient is a smooth surrogate of the sub‑gradient of `max`.
   The temperature β is increased linearly from 0.5 to 30 so that the
   optimisation starts with a gentle landscape and ends with a tight
   approximation of the real objective.  Adam (β₁=0.9, β₂=0.999) is used
   on the logits; a tiny random jitter is injected after long
   stagnation.

3. **Phase B – Deterministic greedy hill‑climbing**  
   Starting from the best point of Phase A we repeatedly move a small
   amount of mass from the index that contributes most to the current
   maximal convolution value to the index that contributes the least.
   The convolution is updated in O(n) time using the bilinear identity,
   allowing millions of cheap moves.  After a long period without
   improvement we perturb the current best solution with a tiny Gaussian
   noise and continue.

4. **Final evaluation** – the best vector found is fed to the provided
   `evaluate_sequence` wrapper and returned as a plain Python list.

The routine respects the 1000 s wall‑time limit, prints intermediate
bounds, and works for any previously discovered construction exposed as
`height_sequence_1`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·n‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Smoothed max gradient (soft‑max temperature beta)
# ----------------------------------------------------------------------
def _grad_softmax_max(a: np.ndarray, beta: float):
    """
    Returns the gradient of the smoothed max functional
        Lβ(a) = (1/β)·log Σexp(β·(a*a)ₖ)
    with respect to a, together with the raw convolution b = a*a.
    """
    b = _full_convolution(a)               # length 2·n‑1
    b_max = b.max()

    # soft‑max weights wₖ = exp(β·(bₖ−b_max)) / Σ.
    # subtracting b_max keeps the exponent range safe.
    exp_term = np.exp(beta * (b - b_max))
    w = exp_term / exp_term.sum()          # still length 2·n‑1

    # part_j = Σ_i a_i * w_{i+j}  (correlation of a with w)
    # can be obtained via a convolution with the reversed weight vector.
    w_rev = w[::-1]
    conv = _conv_fft(a, w_rev)             # length 3·n‑2
    L = len(w)                             # = 2·n‑1
    start = L - 1
    part = conv[start - np.arange(len(a))]   # vector of length n

    grad = a + part                         # ∂L/∂a
    return grad, b


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (logits → softmax)
# ----------------------------------------------------------------------
def _adam_phase(a_init: np.ndarray, n: int, deadline: float):
    """
    Adam optimisation of the smoothed max objective.
    Returns (best_a, best_val) where best_val = 2·n·max(a*a).
    """
    eps = 1e-12
    z = np.log(a_init + eps)               # logits
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    alpha = 0.05                           # base learning rate

    best_a = a_init.copy()
    best_val = 2.0 * n * _full_convolution(a_init).max()

    start = time.time()
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 1500

    beta_start = 0.5
    beta_end = 30.0

    while time.time() < deadline:
        # ----- soft‑max to obtain a on the simplex -----
        max_z = z.max()
        expz = np.exp(z - max_z)
        a = expz / expz.sum()

        # ----- schedule the temperature β -----
        elapsed = time.time() - start
        total = deadline - start
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ----- gradient of the smoothed max -----
        grad_a, b = _grad_softmax_max(a, beta)
        cur_val = 2.0 * n * b.max()

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] best ≈ {best_val:.6f} (β={beta:.2f})")
        else:
            no_improve += 1

        # ----- chain‑rule to obtain gradient w.r.t logits -----
        dot = np.dot(a, grad_a)
        gz = a * (grad_a - dot)                # ∂L/∂z

        # ----- Adam update (descent) -----
        it += 1
        m = beta1 * m + (1 - beta1) * gz
        v = beta2 * v + (1 - beta2) * (gz * gz)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        z -= step

        # ----- occasional jitter to escape plateaus -----
        if no_improve > IMPROVE_LIMIT:
            z += np.random.normal(scale=0.01, size=n)
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _greedy_hillclimb(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> np.ndarray:
    """
    Local search that moves a small amount of mass from the index
    contributing most to the current maximum to the index contributing
    the least.  The full convolution is updated incrementally (O(n)).
    """
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        k = max_idx

        # contribution of each a[i] to the worst convolution entry k
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-14:
            positives = np.where(a > 1e-14)[0]
            if positives.size == 0:
                break
            source = int(np.random.choice(positives))

        # move up to 25 % of the source mass
        delta = a[source] * 0.25 * np.random.rand()
        if delta <= 0.0:
            continue

        # ----- incremental update of the full convolution -----
        b_new = b.copy()
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            # accept the move
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb the current best and restart the local search
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a /= s
            else:
                a = np.full(n, 1.0 / n)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it}] stagnation – perturbing")
        it += 1

    return best_a


# ----------------------------------------------------------------------
