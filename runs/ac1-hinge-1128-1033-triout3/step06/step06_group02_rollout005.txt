def max_conv(a):
    n = len(a)
    size = _next_pow2(2*n-1)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2*n-1]
    # due to numerical noise may have tiny negative; clip.
    return conv.max()
```

     reduces the peak (exponential mirror update ``a_i ← a_i·exp(−α·g_i)``).
   * re‑project onto the probability simplex and occasionally apply the
     anti‑symmetric re‑ordering heuristic.

3. **Simulated‑annealing hill‑climbing** – with the remaining time perform
   small random mass‑transfer moves, accepting improvements always and
   occasional worsening moves with a temperature that decays to zero.
   Random re‑orderings are tried as well.

All convolutions are performed with the FFT (``np.fft.rfft``) which is
``O(n log n)`` and much faster than the naïve ``np.convolve`` used inside
``evaluate_sequence``.  The true objective is only queried at the end
(and a few occasional checkpoints) to avoid the quadratic cost of the
built‑in evaluator.

The function returns a plain Python list of floats that respects all
security checks of ``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_size: int) -> np.ndarray:
    """
    Full (linear) convolution of a real vector with itself.
    Uses FFT of length ``fft_size`` (must be ≥ 2·len(a)‑1).
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[:out_len]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def max_convolution(a: np.ndarray, fft_size: int) -> float:
    """Return the maximum entry of ``a*a``."""
    conv = _full_convolution(a, fft_size)
    return float(conv.max())


def max_convolution_and_index(a: np.ndarray, fft_size: int):
    """Return (max value, arg‑max index) of the self‑convolution."""
    conv = _full_convolution(a, fft_size)
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair the largest values with the smallest ones (mirror reordering).
    Often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending order
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _objective_from_max(max_val: float, n: int) -> float:
    """Exact value of the evaluation function when sum(a)=1."""
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‐function (list of non‑negative floats) that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s time budget and
    returns the best sequence found.
    """
    # ------------------------------------------------------------------
    # General set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem size – a moderate length gives a good trade‑off.
    # Larger n gives more freedom but also a larger search space.
    # --------------------------------------------------------------
    n = 8000                           # you may tune this (≈ twice the previous)

    # --------------------------------------------------------------
    # Initialise from a known construction if available
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        # Interpolate to the desired length
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
    except Exception:
        # Random Dirichlet start (sum will be normalised anyway)
        a = rng.random(n)

    # enforce feasibility (non‑negative, normalised)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    # Pre‑compute FFT size (constant for the whole run)
    fft_size = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initial objective
    # ------------------------------------------------------------------
    cur_max = max_convolution(a, fft_size)
    cur_val = _objective_from_max(cur_max, n)
    best_a = a.copy()
    best_val = cur_val
    print(f"[init] n={n}, O={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – mirror‑descent with line‑search
    # ------------------------------------------------------------------
    it = 0
    alpha0 = 0.70                     # initial step‑size scale
    min_alpha = 1e-4                  # do not let it vanish completely

    while time.time() < deadline - 30.0:
        # Full convolution to locate the current peak
        conv = _full_convolution(a, fft_size)
        max_val = float(conv.max())
        max_idxs = np.where(conv >= max_val - 1e-12)[0]   # ties → sub‑gradient set

        # Build sub‑gradient (average of gradients from all maximal lags)
        g = np.zeros_like(a)
        for k in max_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            g[idx] += a[k - idx]                     # g_i = a_{k-i}
        if max_idxs.size > 1:
            g /= float(max_idxs.size)                # take the average

        # --------------------------------------------------------------
        # Adaptive step‑size (geometric decay) + line‑search
        # --------------------------------------------------------------
        # Base step size slowly decays with the iteration count
        base_alpha = max(alpha0 * (0.985 ** (it // 500)), min_alpha)

        # Candidate alphas – try a few decreasing values
        cand_alphas = [base_alpha * (0.5 ** i) for i in range(6)]

        improved = False
        for alpha in cand_alphas:
            # Mirror‑descent exponential update
            a_new = a * np.exp(-alpha * g)

            # Project back onto the simplex (non‑negative & sum=1)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-12:
                continue
            a_new /= s_new

            # Evaluate the new peak (fast FFT version)
            new_max = max_convolution(a_new, fft_size)

            if new_max < max_val - 1e-14:          # strict improvement
                a = a_new
                cur_max = new_max
                improved = True
                break

        # If none of the candidate alphas helped, we simply keep the old vector
        # (the loop will continue with a smaller base_alpha).

        # Periodic bookkeeping
        if it % 200 == 0:
            cur_val = _objective_from_max(cur_max, n)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[grad it {it}] new best O={best_val:.6f} (max={cur_max:.8e})")

        # Every few thousand iterations try the anti‑symmetric re‑ordering
        if it % 2500 == 0 and it > 0:
            a_ord = _anti_symmetric_reorder(a)
            ord_max = max_convolution(a_ord, fft_size)
            if ord_max < cur_max - 1e-14:
                a = a_ord
                cur_max = ord_max
                cur_val = _objective_from_max(cur_max, n)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[reorder it {it}] improved to O={best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing style random perturbations
    # ------------------------------------------------------------------
    # Temperature schedule (starts small – we are already close to a local optimum)
    T_start = 0.02
    while time.time() < deadline - 5.0:
        # Random mass‑transfer move
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] < 1e-15:
            continue
        delta = a[i] * 0.02 * rng.random()          # move up to 2 % of the source mass
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate /= a_candidate.sum()

        cand_max = max_convolution(a_candidate, fft_size)
        cur_val = _objective_from_max(cur_max, n)
        cand_val = _objective_from_max(cand_max, n)

        # Linear temperature decay
        elapsed = time.time() - start_time
        t_frac = min(elapsed / (budget_s - 5), 1.0)
        T = T_start * (1.0 - t_frac)

        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            # occasional uphill moves
            if T > 0 and rng.random() < np.exp(-(cand_val - cur_val) / T):
                accept = True

        if accept:
            a = a_candidate
            cur_max = cand_max
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_a = a.copy()
                print(f"[anneal] time {elapsed:.1f}s new best O={best_val:.6f}")

        # Random occasional re‑ordering (helps escape plateaus)
        if rng.random() < 0.01:
            a_ord = _anti_symmetric_reorder(a)
            ord_max = max_convolution(a_ord, fft_size)
            if ord_max < cur_max:
                a = a_ord
                cur_max = ord_max
                cur_val = _objective_from_max(cur_max, n)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[anneal reorder] improved O={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = float('inf')

    return best_a.tolist()
