"""
Propose a low‑value step‑function for the secret evaluator.

The algorithm consists of three stages, all respecting the 1000 s wall‑time
limit.

1. **Smooth Adam descent** – we minimise a smooth approximation of the
   ``max`` in the objective:
   `f(a) = (1/λ)·log(∑_i exp(λ·b_i))` with `b = a∗a`.  
   The gradient of `f` can be written as a convolution, which is
   evaluated efficiently with FFTs.  An Adam optimiser (momentum‑based
   adaptive steps) performs the updates, and after each step the vector
   is projected onto the probability simplex (`Σa = 1, a ≥ 0`).  The
   smoothing parameter λ is annealed from 1 to 50.

2. **L² flattening** – a smooth surrogate `∑(b_i‑µ)²` (µ = mean of b)
   is minimised using the exact gradient derived in the original
   script.  This phase reduces the variance of the autocorrelation and
   makes the later greedy phase more effective.

3. **Greedy hill‑climbing** – from the best point found so far we move
   a small amount of mass from the index that contributes most to the
   current peak of the convolution to the index that contributes least.
   Full updates of the convolution are performed in `O(n)` time.
   After long stagnation a tiny Gaussian perturbation is added and the
   search continues.

All three phases keep the sequence non‑negative and normalised, and the
best sequence discovered is finally evaluated with the provided
`evaluate_sequence`.  The code uses only NumPy and runs on a single CPU.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·n‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (length = len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:                                                   # numerical safety
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase A – Smooth Adam descent on a soft‑max approximation of max(b)
# ----------------------------------------------------------------------
def _smooth_adam_phase(a: np.ndarray,
                       n: int,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on f(a)=log‑sum‑exp(λ·(a∗a))."""
    start = time.time()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    lam_start = 1.0
    lam_end = 50.0

    lr_start = 0.1
    lr_end = 0.001

    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        # ----- convolution and exact objective (for bookkeeping) -----
        b = _full_convolution(a)               # length 2·n‑1
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Smooth {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ----- annealing schedule for λ -----
        progress = (time.time() - start) / (deadline - start)
        lam = lam_start + (lam_end - lam_start) * progress

        # ----- soft‑max weights -----
        shift = b - b.max()                     # shift for numerical stability
        w = np.exp(lam * shift)                 # ≥0, max entry = 1
        w_sum = w.sum()
        if w_sum == 0.0:                        # should not happen, but guard
            p = np.full_like(w, 1.0 / w.shape[0])
        else:
            p = w / w_sum                       # probabilities, Σp = 1

        # ----- gradient of the smooth objective -----
        a_rev = a[::-1]
        conv_full = _conv_fft(p, a_rev)        # length 3·n‑2
        grad = 2.0 * conv_full[n - 1:2 * n - 1]  # slice corresponds to ∂/∂a

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr_start * (1.0 - progress) + lr_end * progress
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a = _project_simplex(a_candidate)

        # ----- occasional random restart when stuck -----
        if no_improve > IMPROVE_LIMIT:
            a = np.random.rand(n)
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[Smooth {it}] restart after stagnation")

    return best_a, best_val

# Phase C – Greedy hill‑climbing on the exact objective

def _greedy_hillclimb(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> np.ndarray:
    """Local search that shifts mass from the “worst’’ contributors to the “best’’."""
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        k = max_idx
        # compute contribution of each a[i] to b[k]
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))
        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-14:
            # pick any non‑zero source
            nz = np.where(a > 1e-14)[0]
            if nz.size == 0:
                break
            source = int(np.random.choice(nz))

        # random fraction of the source mass (up to 25 %)
        delta = a[source] * 0.25 * np.random.rand()
        if delta <= 0.0:
            continue

        # ----- incremental update of the full convolution -----
        b_new = b.copy()
        # cross‑terms 2·δ·a
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            # accept move
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)   # tiny perturbation
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")

        it += 1

    return best_a

# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Produce a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence``.  The routine respects a 1000 s wall‑time
    budget, runs three optimisation phases (smooth Adam, L² flattening,
    greedy hill‑climbing) and finally returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 gives a good trade‑off between flexibility and speed.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a known construction if available.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # pylint: disable=undefined-variable
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        # Dirichlet with low concentration → sparse start
        a = np.random.dirichlet(alpha=np.full(n, 0.1))

    a = _project_simplex(a)              # enforce Σa = 1
    best_a = a.copy()
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Split the time budget among the three phases.
    # ------------------------------------------------------------------
    total = deadline - start_time
    t_smooth = start_time + 0.30 * total   # Phase A
    t_l2     = start_time + 0.60 * total   # Phase B
    t_hill   = deadline                    # Phase C

    # ----------------------------
    # Phase A – Smooth Adam descent
    # ----------------------------
    best_a, best_val = _smooth_adam_phase(
        a, n, t_smooth, best_a, best_val
    )
    a = best_a.copy()

    # ----------------------------
    # Phase B – L² flattening
    # ----------------------------
    best_a, best_val = _l2_flatten_phase(
        a, n, t_l2, best_a, best_val
    )
    a = best_a.copy()

    # ----------------------------
    # Phase C – Greedy hill climbing
    # ----------------------------
    best_a = _greedy_hillclimb(
        a, n, t_hill, best_a, best_val
    )

    # ------------------------------------------------------------------
    # Final evaluation (the secure wrapper may clamp values)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
