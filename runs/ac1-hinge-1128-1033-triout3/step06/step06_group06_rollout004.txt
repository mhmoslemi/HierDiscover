"""
Propose a step‑function (non‑negative list with Σa = 1) that minimizes
`evaluate_sequence`.  The algorithm works in three stages:

1. **Fast parametric sweep + Adam on a smooth max** – provides a good
   starting point on the simplex.
2. **Deterministic greedy hill‑climber** – repeatedly reduces the
   current maximum of the self‑convolution by moving a carefully chosen
   amount of mass from the index that contributes most to this maximum
   to an index that contributes the least.  A simple line‑search guarantees
   a strict improvement at each accepted step.
3. **Short simulated‑annealing polishing** – occasional worsening moves
   to escape tiny plateaus before the final evaluation.

All phases respect the global time budget (default 1000 s) and keep the
vector non‑negative with unit sum, which makes the objective
scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) assumed = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Parametric initialisation (uniform / linear / quadratic / sine / expo)
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising initial vector and its objective value."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    # linear decreasing
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # quadratic decreasing
    a = (np.arange(n, 0, -1, dtype=float) ** 2)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # half‑wave sine
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # exponential families
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam optimisation on a smooth approximation of max(conv)
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth (log‑sum‑exp) approximation of max(conv)."""
    w = np.log(a0 + 1e-12)                     # soft‑max logits (Σa = 1)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.08, 0.002
    eps_start, eps_end = 5e-2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 25000

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max to recover a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)          # length 2·n‑1
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()            # soft‑max over positions

        # gradient of smooth objective w.r.t. a:
        # ∂/∂a 2·n·∑_k p_k (a*a)_k = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]     # n entries that correspond to ∂c/∂a
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.01, size=n)

        # restart if stagnated
        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Deterministic greedy hill‑climber on the exact objective
# ----------------------------------------------------------------------
def _deterministic_hillclimb(a_start: np.ndarray,
                            n: int,
                            start: float,
                            deadline: float,
                            best_a: np.ndarray,
                            best_val: float,
                            target_val: float = 1.503) -> np.ndarray:
    """
    Greedy local search that at each step moves mass from the index that
    contributes most to the current maximal convolution entry to an index
    that contributes the least.  A short back‑tracking line‑search guarantees
    a strict decrease of the objective.
    """
    a = a_start.copy()
    b = _full_convolution(a)                     # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    IMPROVE_LIMIT = 120_000
    no_improve = 0
    it = 0
    EPS = 1e-14

    while True:
        now = time.time()
        if now >= deadline:
            break

        k_max = int(np.argmax(b))
        b_max = b[k_max]

        # second largest value (handles ties)
        if b.size > 1:
            # partition is O(n) and cheap
            second_max = np.partition(b, -2)[-2]
        else:
            second_max = 0.0
        diff = b_max - second_max

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]

        # source index – biggest contribution
        i_src = int(np.argmax(contrib))
        if a[i_src] <= EPS:
            # nothing to move – give up
            break

        # target index – smallest contribution (different from source)
        contrib_masked = contrib.copy()
        contrib_masked[i_src] = np.inf
        j_tgt = int(np.argmin(contrib_masked))

        # make sure we really have a reduction direction
        src_val = a[idx[i_src]] if (0 <= idx[i_src] < n) else 0.0
        tgt_val = a[idx[j_tgt]] if (0 <= idx[j_tgt] < n) else 0.0
        delta_coeff = src_val - tgt_val
        if delta_coeff <= 0:
            # try a different target (second smallest, etc.)
            sorted_idxs = np.argsort(contrib)
            found = False
            for cand in sorted_idxs:
                if cand == i_src:
                    continue
                cand_val = a[idx[cand]] if (0 <= idx[cand] < n) else 0.0
                if src_val - cand_val > 0:
                    j_tgt = cand
                    tgt_val = cand_val
                    delta_coeff = src_val - tgt_val
                    found = True
                    break
            if not found:
                # cannot find improving direction
                break

        # desired reduction of the maximal entry
        if diff <= 0:
            # tiny artificial reduction to keep moving
            diff = b_max * 1e-12

        # theoretical delta to bring the max down to second_max (linear approx)
        delta_needed = diff / (2.0 * delta_coeff)
        # keep a safe margin
        delta = min(delta_needed * 0.9, a[i_src] * 0.9)
        if delta < EPS:
            delta = a[i_src] * 0.05   # fallback small move

        # line‑search / back‑tracking to guarantee improvement
        a_before = a.copy()
        b_before = b.copy()
        accepted = False
        for _ in range(12):
            # incremental update of the convolution (exact)
            b_new = b_before.copy()
            # linear part
            b_new[j_tgt:j_tgt + n] += 2.0 * delta * a_before
            b_new[i_src:i_src + n] -= 2.0 * delta * a_before
            # quadratic corrections
            b_new[2 * j_tgt] += delta * delta
            b_new[2 * i_src] += delta * delta
            b_new[i_src + j_tgt] -= 2.0 * delta * delta

            new_max = b_new.max()
            if new_max < b_max - EPS:
                # accept
                a = a_before.copy()
                a[i_src] -= delta
                a[j_tgt] += delta
                # re‑normalise (tiny numerical drift may appear)
                a = np.maximum(a, 0.0)
                a /= a.sum()

                b = b_new
                cur_val = 2.0 * n * new_max
                accepted = True
                if cur_val < best_val - 1e-12:
                    best_val, best_a = cur_val, a.copy()
                    no_improve = 0
                    print(f"[DetHill {it:7d}] new best = {best_val:.6f}")
                    if cur_val <= target_val:
                        return best_a
                else:
                    no_improve += 1
                break
            else:
                # too aggressive – shrink delta
                delta *= 0.5

        if not accepted:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            # shake the current best and restart
            noise = np.random.normal(scale=0.015, size=n)
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[DetHill {it:7d}] perturbing after stagnation")
        it += 1

    return best_a

# Phase C – short simulated‑annealing refinement (optional)
# ----------------------------------------------------------------------
def _annealing_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Metropolis‑style short annealing to escape tiny plateaus."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    T_start, T_end = 5e-4, 5e-8
    IMPROVE_LIMIT = 80_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        k_max = int(np.argmax(b))

        # source – bias towards large contribution to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = int(np.random.choice(n, p=probs))
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # uniform target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # tiny transfer (≤ 0.5 % of source mass)
        delta = a[i] * np.random.rand() * 0.005
        if delta <= 0.0:
            continue

        # incremental convolution update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    return best_a, best_val
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # small random shake
            noise = np.random.normal(scale=0.003, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val
