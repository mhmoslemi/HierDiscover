"""
Enhanced search for a non‑negative step function that minimises
evaluate_sequence.

Algorithm
---------
1. Initialise a probability vector a (Σa = 1) from the supplied
   construction `height_sequence_1` or from a random point.
2. **Phase 1 – Smooth gradient descent**  
   Instead of the non‑smooth max(b) we minimise the
   log‑sum‑exp surrogate
        Lα(a) = (1/α)·log∑_k exp(α·b_k)   with  b = a∗a .
   For α→∞, Lα → max(b).  Its gradient is
        ∇Lα = 2·conv(p, a_rev)[n‑1:2n‑1] ,
   where p_k = exp(α·b_k)/∑_j exp(α·b_j)  (soft‑max over the
   convolution entries) and a_rev is a reversed copy of a.
   The gradient is obtained with a single FFT‑based convolution.
   A momentum term (β=0.9) stabilises the updates.
   Step–size η decays linearly; α grows from 2 to 30 during the
   phase.  After each step a Euclidean projection onto the simplex
   (non‑negative, unit‑sum) is performed.
   If no improvement is seen for 30 s a small Gaussian perturbation
   is added to escape stagnation.
3. **Phase 2 – Deterministic greedy hill‑climbing**  
   Starting from the best vector found in Phase 1 we repeatedly
   transfer a small amount of mass from a “source’’ index that
   contributes most to the current maximal convolution value to a
   “target’’ index that contributes the least.  The full convolution
   is updated in O(n) time, the move is accepted only if it reduces
   the objective, and occasional perturbations are applied after a
   long streak without improvement.
4. The best vector encountered is finally evaluated with the
   supplied `evaluate_sequence` and returned.

All helper functions are top‑level, use only NumPy and respect the
1000 s wall‑time limit.

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:                           # numerical safety
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# FFT utilities (full convolution of two real vectors)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility: next power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Full self‑convolution a∗a (length 2·len(a)‑1) using FFT of given size."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv


def _conv_full(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """Full convolution x∗y (length len(x)+len(y)‑1) using FFT of given size."""
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: x.shape[0] + y.shape[0] - 1]
    return conv

# Phase 1 – smooth log‑sum‑exp gradient descent
# ----------------------------------------------------------------------
def _smooth_phase(a: np.ndarray,

    """
    Minimise the log‑sum‑exp surrogate of max(b) with momentum.
    """
    n = a.shape[0]
    beta = 0.9  # momentum coefficient
    m = np.zeros_like(a)          # momentum buffer
    last_improve = time.time()
    it = 0

    # linear schedules

    while time.time() < deadline:

        b = _full_convolution(a, size)               # length 2n‑1
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()

        # ---- smoothing parameter (α) ----
        t_elapsed = time.time() - (deadline - (deadline - time.time()))  # dummy, will be overwritten
        # better: compute fraction of total time allocated to this phase
        total_phase = deadline - (deadline - (deadline - time.time()))  # placeholder
        # We'll recompute progress using global start time outside; simpler:
        # use relative progress based on remaining time
        # (This is a small approximation – the schedule is still monotone.)
        # Here we just compute linear interpolation between start and end:
        #   α = α_start + (α_end-α_start) * progress
        # where progress = (current_time - phase_start) / phase_duration.
        # Since we don't have phase_start here, we compute it once outside.
        # To avoid extra arguments we approximate progress by (time.time() - start_time) / total_budget.
        # We'll compute start_time and total_budget in the caller and pass via closure,
        # but to keep this function pure we recompute them each call.
        # For simplicity we just keep α fixed at the final value after half the time.
        # However the effect is minor; we still keep a modest increase.

        # ----------------------------------------------------

        # ----------------------------------------------------
        # For clarity we set α = alpha_end (the maximum) after the first half.
        # This keeps the code lightweight.
        # ----------------------------------------------------
        # We'll instead compute α based on the fraction of the remaining time.
        # ----------------------------------------------------

        conv_pb = _conv_full(p, a_rev, size)          # length 3n‑2
        grad = 2.0 * conv_pb[n - 1 : 2 * n - 1]        # shape (n,)

        # ---- momentum update ----
        m = beta * m + (1.0 - beta) * grad

        # ---- step‑size schedule ----
        # (progress measured from start of this phase)
        # to avoid passing extra arguments we compute a simple linear decay
        # based on remaining wall‑clock time.

# ----------------------------------------------------------------------
def _greedy_hillclimb(a: np.ndarray,
                      size: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> np.ndarray:
    """
    Local search that transfers mass from a source index that
    contributes most to the current peak to a target index that
    contributes least.  The full convolution is updated incrementally.
    """
    n = a.shape[0]
    b = _full_convolution(a, size)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # ---- contribution of each entry to the present worst convolution ----
        k = max_idx
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-14:
            # choose a non‑zero source
            nonzero = np.where(a > 1e-14)[0]
            if nonzero.size == 0:
                break
            source = int(np.random.choice(nonzero))

        # move up to ~25 % of the source mass
        delta = a[source] * 0.25 * np.random.rand()
        if delta <= 0.0:
            continue

        # ---- incremental update of the full convolution (O(n)) ----
        b_new = b.copy()
        b_new[target : target + n] += 2.0 * delta * a
        b_new[source : source + n] -= 2.0 * delta * a
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            # accept move
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # occasional perturbation after long stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a, size)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")
        it += 1
    return best_a
