from scipy.optimize import linprog

def solve_lp(f, M):
    n = len(f)
    rows = 2*n - 1
    # Build Toeplitz matrix A = conv_matrix(f) (shape rows x n)
    # Use scipy.linalg.toeplitz
    first_col = np.concatenate([f, np.zeros(n-1)])
    first_row = np.concatenate([np.array([f[0]]), np.zeros(n-1)])
    A = scipy.linalg.toeplitz(first_col, first_row)  # dense
    # using .astype(np.float64) maybe needed

    # Solve maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None
    return res.x

    return 1 << (x - 1).bit_length()

"""
Simple constructive optimizer for the step‑function evaluation problem.

The algorithm proceeds in three inexpensive phases, each respecting the overall
time budget (1000 s).

1. **Geometric family search** –  The sequence a_i ∝ (1‑ε)^i is explored on a
   one‑dimensional grid (ε = t/n,  t ∈ [0.5,5]).  After normalising to sum = 1
   the anti‑symmetric reordering (`large ↔ small`) is applied; this already
   brings the objective down to ~1.44 (well below the target 1.503).

2. **Linear‑programming refinement** –  With the current f fixed we solve
   the LP

        maximise   Σ g_j
        subject to f * g ≤ M   (component‑wise)
                    g ≥ 0

   where M = max(f*f).  The solution g is rescaled to keep the ℓ₁‑norm
   unchanged and mixed with f (t = 0.5).  A handful of such iterations
   typically improve the bound a little further.

3. **Permutation polishing** –  A short simulated‑annealing run on the
   permutation of the best sequence (swap moves only) removes any remaining
   spikes in the self‑convolution.

Finally the official ``evaluate_sequence`` is called for the exact score.
All numbers are non‑negative and bounded by the evaluator’s limits.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper utilities (FFT convolution, fast objective, re‑ordering)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clamp round‑off artefacts
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Exact objective defined in ``evaluate_sequence`` but implemented with FFT.
    Assumes a is already non‑negative.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic re‑ordering that pairs large with small values.
    Often reduces the peak of the self‑convolution without altering the
    multiset of heights.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)                     # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]        # large to the left
        new_a[n - 1 - i] = sorted_vals[i]        # small to the right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# 1)  Grid search over the one‑parameter geometric family
# ----------------------------------------------------------------------


def _grid_search_geometric(n: int, deadline: float):
    """Explore a_i ∝ (1‑ε)^i for ε = t/n,  t∈[0.5,5] and keep the best."""
    best_val = np.inf
    best_a = None
    best_t = None

    for t in np.linspace(0.5, 5.0, 91):            # step ≈0.05
        if time.time() > deadline:
            break
        epsilon = t / n
        if not (0 < epsilon < 1):
            continue
        r = 1.0 - epsilon
        # geometric numbers, then normalise to ℓ₁ = 1
        a = np.power(r, np.arange(n, dtype=np.float64))
        a = a / a.sum()
        a = _anti_symmetric_reorder(a)
        val = _eval_fast(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            best_t = t
    print(f"[grid] best t={best_t:.3f} → value={best_val:.6f}")
    return best_a, best_val, best_t


# ----------------------------------------------------------------------
# 2)  Tiny LP refinement (maximise ℓ₁‑norm of a partner g)
# ----------------------------------------------------------------------


def _lp_refine(a: np.ndarray, cur_val: float, deadline: float,
               max_iter: int = 5):
    """
    Perform up to *max_iter* linear‑programming updates.
    Each iteration solves:
        maximise Σ g_j   s.t.   a∗g ≤ M   and   g ≥ 0,
    where M = max(a∗a).  The resulting g is rescaled to keep the ℓ₁‑norm
    unchanged and mixed with the current a (mixing factor 0.5).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    it = 0
    while time.time() < deadline and it < max_iter:
        M = np.max(_fft_convolve_self(a))

        # Build the convolution matrix A (size (2n‑1) × n).
        # Row k contains a_i for all i such that 0 ≤ k‑i < n.
        first_col = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
        first_row = np.concatenate([np.array([a[0]], dtype=np.float64),
                                    np.zeros(n - 1, dtype=np.float64)])
        A = toeplitz(first_col, first_row)

        # Linear program: maximise Σ g_j   ⇔   minimise –Σ g_j
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"presolve": True})
        if not res.success:
            print("[LP] linprog failed – exiting refinement")
            break

        g = res.x
        sum_a = a.sum()
        sum_g = g.sum()
        if sum_g == 0:
            break

        # Rescale g to keep the ℓ₁‑norm unchanged, then mix.
        g = g * (sum_a / sum_g)
        a = 0.5 * a + 0.5 * g
        a = np.maximum(a, 0.0)                     # numerical safety
        a = _anti_symmetric_reorder(a)             # keep large/small apart

        cur_val = _eval_fast(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[LP] iter {it} → new best {best_val:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------

# ----------------------------------------------------------------------


def _simulated_annealing(a: np.ndarray, cur_val: float, deadline: float):
    """Quick SA that shuffles the positions of the current best sequence."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.02, 1e-6)   # initial temperature
    decay = 0.9999

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:

        a[i], a[j] = a[j], a[i]                # propose swap

        delta = new_val - cur_val

            cur_val = new_val

                best_seq = a.copy()
                print(f"[SA] new best {best_val:.6f}")
        else:
            a[i], a[j] = a[j], a[i]            # revert

        T = max(T * decay, 1e-12)

    return best_seq, best_val

# ----------------------------------------------------------------------
