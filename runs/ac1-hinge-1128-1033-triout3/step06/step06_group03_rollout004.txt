"""
Propose a non‑negative step‑function (list of heights) that (approximately)
minimises ``evaluate_sequence``.
Algorithm
=========
1. **Problem size** – we work with n≈8000.  The objective
   2·n·max(conv(a,a)) is scale‑invariant; we keep the vector on the
   probability simplex (sum = 1).

2. **Parametric warm‑start** – a small grid of simple families
   (exponential, symmetric exponential, sine‑power, symmetric polynomial)
   is scanned.  For each family a single shape parameter is varied, the
   vector is normalised and the objective (computed via FFT) is recorded.
   The best candidate is kept as ``a``.

3. **Exponential‑mirror descent** – starting from the best parametric
   vector we perform a few hundred iterations of an exponentiated‑gradient /
   mirror‑descent update on the sub‑gradient of the max‑convolution.
   The sub‑gradient for the active lag(s) `k` is
       g_i = Σ_{k∈S} 2·a_{k‑i}
   (zero when the index would be out of range).  The update
       a ← a·exp(−η·g)   (element‑wise)
   is followed by normalisation.  The step size η is chosen as
       η = min(5.0, 0.5 / (max(g)+ε))
   which guarantees a moderate relative change.  The best vector seen
   during this phase is stored.

4. **Guided hill‑climbing** – a short, greedy “mass‑transfer’’ refinement
   (the same idea as the previous state‑of‑the‑art code) is applied until
   the time budget expires.  In each move the pair (i,j) that contributes
   most to the current maximal convolution value is identified and a small
   fraction of its mass is moved to two low‑mass positions, thereby
   decreasing the peak.  Every improvement is printed.

5. **Final evaluation** – the best vector is returned as a plain Python
   list; ``evaluate_sequence`` is called once for the final score.

The whole routine respects the 1000 s budget, prints progress, and never
uses file or network IO.  All helper functions are defined at top level.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full autocorrelation of a real vector (length 2·len-1).
    The result is guaranteed non‑negative (tiny negative values are clamped).
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate floating‑point noise below zero
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill climbing: move a small fraction of the mass from the
    (i,j) pair that contributes most to the current maximal convolution
    towards two low‑mass indices.  Stops when the time budget expires.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    step_frac = 0.05  # starting transfer fraction

    while time.time() < deadline:
        conv = _full_convolution(best_a)
        max_b = conv.max()
        # all lags that attain the maximum (within tolerance)
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        if ks.size == 0:
            break
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = best_a[i_range] * best_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # pick two destination indices that currently have below‑average mass
        low_mass = np.where(best_a < best_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        # avoid recreating the same lag (which would undo the move)
        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # perform a feasible mass transfer
        if i_max != j_max:
            delta = min(best_a[i_max], best_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = best_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max (peak on the diagonal)
            delta = best_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = best_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        # exact evaluation (fast shortcut also possible)
        val = evaluate_sequence(new_a.tolist())
        if val < best_val:
            best_a = new_a
            best_val = val
            print(f"[guided] new best = {best_val:.6f}")
            # shrink the step slowly as we improve
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            # slowly reduce the step size on failures
            step_frac = max(step_frac * 0.995, 1e-4)

    return best_a, best_val


# ----------------------------------------------------------------------
# Objective helper (fast, assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a already normalised (sum=1)."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The method combines a tiny parametric sweep, an exponentiated‑gradient
    (mirror‑descent) phase and a final guided hill‑climb.
    Returns the best list of heights found before the time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # 1. Problem size (larger n gives a tighter asymptotic constant)
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # 2. Simple parametric families – grid search for a good warm‑start
    # --------------------------------------------------------------
    best_a = None
    best_val = float('inf')

    # helper to normalise and evaluate quickly
    def _norm_and_eval(v):
        s = v.sum()
        if s <= 0:
            return float('inf'), None
        a = v / s
        val = _objective(a)
        return val, a

    # (a) pure exponential decay
    for beta in np.linspace(0.0, 2.0, 25):
        a = np.exp(-beta * np.arange(n, dtype=float))
        val, a_norm = _norm_and_eval(a)
        if val < best_val:
            best_val, best_a = val, a_norm

    # (b) symmetric exponential about the centre
    centre = (n - 1) / 2.0
    for beta in np.linspace(0.0, 2.0, 25):
        a = np.exp(-beta * np.abs(np.arange(n, dtype=float) - centre))
        val, a_norm = _norm_and_eval(a)
        if val < best_val:
            best_val, best_a = val, a_norm

    # (c) sine‑power family (peaks in the centre)
    theta = np.pi * (np.arange(n, dtype=float) + 1) / (n + 1)
    sine_vals = np.sin(theta)
    for alpha in np.linspace(0.0, 6.0, 25):
        a = sine_vals ** alpha
        val, a_norm = _norm_and_eval(a)
        if val < best_val:
            best_val, best_a = val, a_norm

    # (d) symmetric polynomial: ((i+1)*(n-i))**alpha
    i_vals = np.arange(n, dtype=float) + 1.0
    poly_base = i_vals * (n - np.arange(n, dtype=float))
    for alpha in np.linspace(0.0, 4.0, 25):
        a = poly_base ** alpha
        val, a_norm = _norm_and_eval(a)
        if val < best_val:
            best_val, best_a = val, a_norm

    # (e) uniform baseline (just in case)
    a = np.full(n, 1.0 / n)
    val = _objective(a)
    if val < best_val:
        best_val, best_a = val, a.copy()

    print(f"[grid] best after param sweep: {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Exponentiated‑gradient (mirror descent) refinement
    # --------------------------------------------------------------
    a = best_a.copy()
    cur_val = _objective(a)
    it = 0
    # we keep the best vector seen during this phase as well
    best_val_grad = cur_val
    best_a_grad = a.copy()

    while time.time() < deadline - 5.0:
        it += 1
        conv = _full_convolution(a)
        M = conv.max()
        # active lags (within a tiny tolerance)
        ks = np.where(conv >= M - 1e-12)[0]

        # sub‑gradient of max‑convolution
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx_range = np.arange(i_start, i_end + 1, dtype=np.intp)
            g[idx_range] += 2.0 * a[k - idx_range]

        # adaptive step size – scale inversely with max(g)
        step = 0.5 / (g.max() + 1e-12)
        step = min(step, 5.0)          # avoid too‑large steps
        # exponentiated update (mirror descent on the simplex)
        exp_factor = np.exp(-step * g)
        # clip to avoid underflow / overflow
        exp_factor = np.clip(exp_factor, 1e-20, 1e20)
        a_new = a * exp_factor
        a_new_sum = a_new.sum()
        if a_new_sum == 0:
            # fall back to a tiny random perturbation
            a_new = _simplex_projection(rng.normal(scale=0.01, size=n) + a)
        else:
            a_new = a_new / a_new_sum

        new_val = _objective(a_new)

        # keep the global best seen so far
        if new_val < best_val_grad - 1e-12:
            best_val_grad = new_val
            best_a_grad = a_new.copy()
            print(f"[expgrad {it}] new best = {best_val_grad:.6f}")

        # simple acceptance rule: keep the move if it improves
        if new_val < cur_val - 1e-14:
            a = a_new
            cur_val = new_val
        else:
            # occasionally jitter to escape plateaus
            if rng.random() < 0.02:
                a = _simplex_projection(a + rng.normal(scale=0.005, size=n))
                cur_val = _objective(a)

    # Record the best from the gradient phase
    if best_val_grad < best_val:
        best_val, best_a = best_val_grad, best_a_grad.copy()
        print(f"[phase] best after exp‑grad = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Guided hill‑climbing refinement (mass‑transfer moves)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # --------------------------------------------------------------
    # 5. Final evaluation via the provided function (for safety)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")

    return best_a.tolist()
