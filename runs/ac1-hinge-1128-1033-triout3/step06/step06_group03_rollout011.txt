    import time
    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 4000
    try: 
       a0 = np.asarray(height_sequence_1, dtype=float)
       if a0.shape[0] != n:
           # interpolate to n using linear interpolation
           old_x = np.linspace(0,1,a0.shape[0])
           new_x = np.linspace(0,1,n)
           a0 = np.interp(new_x, old_x, a0)
    except Exception:
       a0 = np.full(n, 1.0/n)

    a = np.maximum(a0, 0.0)

    best_a = a.copy()

    print("[init] objective", best_val)

    # define fast objective
    def fast_obj(seq):
        s = seq.sum()
        if s < 1e-12:
            return np.inf
        conv = _full_convolution(seq)
        return float(2 * n * conv.max() / (s * s))

    iter_count = 0
    no_improve_iters = 0

    while time.time() < deadline:
        iter_count += 1
        conv = _full_convolution(a)
        M = conv.max()
        cur_val = 2 * n * M / (a.sum()**2)

        if cur_val < best_val - 1e-12:

"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.

Algorithm
=========
* Initialise a non‑negative vector on the probability simplex (uniform or the
  provided ``height_sequence_1``).
* Repeatedly improve it by solving a tiny linear program that tries to
  decrease the *active* convolution constraints:
      minimise   t
      subject to   g_k·d ≤ t   for every lag k attaining the current maximum,
                   Σ d_i = 0,
                   d_i ≥ –a_i            (keep entries non‑negative).
  Here g_k = ∇_a conv_k = 2·a shifted by k.
  The variables are a direction d (length n) and the scalar t = worst
  linearised increase of the maximal convolution value.
* If the LP yields a negative t we move `a ← a + d` (or a scaled version of
  it) – this guarantees a strict decrease of the true objective.
* When the LP cannot improve the solution we perform a small random jitter
  and project back onto the simplex.
* After the main loop a short guided hill‑climbing phase (mass‑transfer moves)
  is applied to polish the result.
* Throughout the run the best sequence seen so far is reported and finally
  returned.

The LP has only a handful of inequality rows (one per active lag) but
4000+ variables, making it very cheap to solve with ``scipy.optimize.linprog``.
The whole routine respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1) using FFT."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)                 # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Local refinement that moves a little mass away from the worst lag."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise the step‑function heights using a tiny linear‑program based
    active‑set descent, occasional random jitter and a final guided hill‑climb.
    Returns the best list of non‑negative heights found before the time budget
    expires.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – 4000 is a good compromise (same as the reference code)
    n = 4000

    # ------------------------------------------------------------------
    # Initial sequence (reuse previous construction if possible)
    # ------------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    except Exception:
        a0 = np.full(n, 1.0 / n)   # uniform start

    a = a0.copy()
    # internal fast objective (scale‑invariant)
    def fast_obj(v: np.ndarray) -> float:
        s = v.sum()
        if s < 1e-12:
            return np.inf
        conv = _full_convolution(v)
        return float(2.0 * n * conv.max() / (s * s))

    cur_val = fast_obj(a)
    best_a = a.copy()
    best_val = cur_val
    print(f"[init] n={n}, objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Helper to build gradient of a convolution lag
    # ------------------------------------------------------------------
    def _grad_for_k(arr: np.ndarray, k: int) -> np.ndarray:
        """Return ∇_a conv_k (vector of length n) for given lag k."""
        g = np.zeros_like(arr)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        g[i_range] = 2.0 * arr[k - i_range]
        return g

    # ------------------------------------------------------------------
    # Main optimisation loop (LP‑based active‑set descent + jitter)
    # ------------------------------------------------------------------
    it = 0
    no_imp = 0
    try:
        from scipy.optimize import linprog
        SCIPY_AVAILABLE = True
    except Exception:
        SCIPY_AVAILABLE = False
        print("[info] scipy not available – falling back to jitter only.")

    while time.time() < deadline - 5.0:
        it += 1
        conv = _full_convolution(a)
        M = conv.max()
        cur_val = fast_obj(a)

        # keep global best
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[iter {it}] new best (curr) = {best_val:.6f}")

        improved_this_round = False

        if SCIPY_AVAILABLE:
            # active set – top k values (few of them)
            # use a small absolute tolerance to collect the true maximum lag(s)
            eps = 1e-12
            active_ks = np.flatnonzero(conv >= M - eps)
            # limit to at most 10 constraints for speed
            if active_ks.size > 10:
                # take the 10 largest lags
                active_ks = np.argpartition(-conv, 10)[:10]

            # Build LP matrices
            n_vars = n + 1                     # d_0..d_{n-1}, t
            rows = []
            rhs = []
            for k in active_ks:
                g = _grad_for_k(a, int(k))
                row = np.empty(n_vars)
                row[:n] = g
                row[n] = -1.0                 # -t
                rows.append(row)
                rhs.append(0.0)

            A_ub = np.vstack(rows) if rows else None
            b_ub = np.array(rhs) if rows else None

            # Equality Σ d_i = 0  (t is free)
            A_eq = np.zeros((1, n_vars))
            A_eq[0, :n] = 1.0
            b_eq = np.array([0.0])

            # Bounds: d_i ≥ –a_i   (keeps a+d non‑negative),   t free
            bounds = [(-a[i], None) for i in range(n)] + [(None, None)]

            # Linear objective: minimise t
            c = np.concatenate([np.zeros(n), [1.0]])

            try:
                res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                              A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds, method="highs",
                              options={"presolve": True, "time_limit": 10})
            except Exception as e:
                print(f"[LP] exception: {e}")
                res = None

            if res is not None and res.success:
                t_val = res.x[-1]
                if t_val < -1e-12:                     # genuine improvement direction
                    d = res.x[:n]
                    # try full step first
                    a_new = a + d
                    new_val = fast_obj(a_new)
                    if new_val < best_val - 1e-12:
                        a = a_new
                        best_a = a.copy()
                        best_val = new_val
                        improved_this_round = True
                        print(f"[LP full] it {it} → {best_val:.6f}")
                    else:
                        # backtrack with shrinking factors
                        for beta in (0.5, 0.25, 0.125, 0.0625):
                            a_tmp = a + beta * d
                            val_tmp = fast_obj(a_tmp)
                            if val_tmp < best_val - 1e-12:
                                a = a_tmp
                                best_a = a.copy()
                                best_val = val_tmp
                                improved_this_round = True
                                print(f"[LP scaled {beta}] it {it} → {best_val:.6f}")
                                break

        # ------------------------------------------------------------------
        # If LP did not give any progress, try a small random perturbation
        # ------------------------------------------------------------------
        if not improved_this_round:
            no_imp += 1
            if no_imp >= 30:
                # inject a jitter, re‑project onto the simplex
                pert = rng.normal(scale=0.02, size=n)
                cand = a + pert
                cand = np.maximum(cand, 0.0)
                cand = _simplex_projection(cand)
                cand_val = fast_obj(cand)
                if cand_val < best_val - 1e-12:
                    a = cand
                    best_a = cand.copy()
                    best_val = cand_val
                    print(f"[jitter] it {it} → {best_val:.6f}")
                    no_imp = 0
        else:
            no_imp = 0

    # ------------------------------------------------------------------
    # Final guided hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # ------------------------------------------------------------------
    # Final reporting (use the official evaluator)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")

    return best_a.tolist()
