"""
propose_candidate
-----------------
A multi‑stage optimizer for the step‑function evaluation problem.

Stages
------
1. **Smooth surrogate optimisation** – many random restarts of L‑BFGS
   on a log‑sum‑exp smoothed version of the max‑convolution.
   The parameters are the log‑softmax variables; a simple
   ℓ₁‑normalisation enforces a fixed total mass.

2. **Anti‑symmetric re‑ordering** – after each restart the best
   shape is reordered by pairing the largest heights with the smallest
   ones; this typically lowers the self‑convolution peak without
   changing the multiset of values.

3. **Simulated annealing on permutations** – a fast FFT‑based evaluator
   repeatedly swaps positions.  The temperature schedule is deliberately
   aggressive so that the search explores a large part of the permutation
   space.

4. **Gradient‑guided mass transfer** – using the index where the
   self‑convolution attains its maximum we compute the exact sub‑gradient
   (a shifted copy of the sequence) and move a small fraction of mass
   from a “high‑gradient’’ index to a “low‑gradient’’ index.  This
   deterministic refinement can shave the last few per‑cent off the
   objective.

5. **Final hill‑climbing** – cheap local moves (tiny mass transfers
   and random swaps) polish the solution.

All stages respect a global 1000 s budget and always return a feasible
list of non‑negative floats.  The code relies only on NumPy, SciPy and the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate wrapper (avoid closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # tiny negatives can appear due to round‑off – clamp them
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast exact evaluation of the official ``evaluate_sequence``.
    ``a`` is assumed to be non‑negative and already satisfies the clipping
    limits (0 ≤ a_i ≤ 1000).  Returns the same value as
    ``evaluate_sequence(a.tolist())`` but is much faster.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:                     # protection enforced by the official function
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smoothed surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε (log‑sum‑exp smoothing of the max) and its
    gradient w.r.t. the soft‑max parameters ``w``.
    """
    # ----- soft‑max (guarantees positivity and Σa = 1) -----
    w_shift = w - np.max(w)                # improve numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # ----- auto‑convolution a * a (full) -----
    conv = _fft_convolve_self(a)           # length 2·n‑1
    max_conv = conv.max()

    # ----- smooth approximation of max(conv) -----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # log∑ exp((conv−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a (closed‑form) -----
    weights = np.exp(shifted)
    p = weights / weights.sum()           # probability distribution over k

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)   # helper defined below
    # middle n entries correspond to index range [n-1, 2n-2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# Helper: convolution used inside the gradient (FFT based, same as above)
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# Bind the helper to the function (so it can be called from inside)
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Re‑ordering heuristic
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Produce an ordering where large values are paired with small ones
    when the sequence is reversed.  This deterministic heuristic often
    reduces the peak of the self‑convolution without changing the multiset
    of heights.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large to the left
        new_a[n - 1 - i] = sorted_vals[i]      # small to the right

    if n % 2:
        new_a[half] = sorted_vals[half]   # keep the middle element
    return new_a


# ----------------------------------------------------------------------
# Simulated annealing on permutations (swap moves only)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """
    Classic SA on the permutation of ``a``.  Swaps preserve the ℓ₁‑norm.
    ``cur_val`` is the evaluation of ``a`` (computed with ``_eval_fast``).
    Returns the best sequence and its value found before ``deadline``.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # Initialise temperature – a modest fraction of the current value.
    T = max(cur_val * 0.05, 5e-4)          # a slightly hotter start than before
    decay = 0.99996                       # slower cooling (more exploration)

    iter_cnt = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose swap
        a[i], a[j] = a[j], a[i]
        new_val = _eval_fast(a)
        delta = new_val - cur_val

        # Metropolis acceptance
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA   ] new best = {best_val:.6f} (iter {iter_cnt})")
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        # cooling
        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        iter_cnt += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient‑guided mass‑transfer refinement
# ----------------------------------------------------------------------
def _gradient_mass_transfer(a: np.ndarray, cur_val: float,
                           deadline: float) -> (np.ndarray, float):
    """
    Local refinement that looks at the index where the self‑convolution
    attains its maximum.  The sub‑gradient at that point is simply a
    shifted copy of ``a``.  We repeatedly move a small fraction of mass
    from a high‑gradient position to a low‑gradient position and keep
    the move if the objective improves.
    """
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        conv = _fft_convolve_self(cur_a)
        k = int(np.argmax(conv))                # index of the current peak

        # gradient g_i = a_{k-i} (0 if out of range)
        g = np.zeros(n, dtype=cur_a.dtype)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start > i_end:                     # should never happen
            continue
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] = cur_a[j_idx]                 # always non‑negative

        # source = index with large gradient and non‑zero mass
        src_candidates = np.where(cur_a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src_weights = g[src_candidates] + 1e-12   # avoid zero probability
        src_probs = src_weights / src_weights.sum()
        i_src = rng.choice(src_candidates, p=src_probs)

        # destination = index with a small gradient value
        low_thresh = np.percentile(g, 15)        # bottom 15 %
        dst_pool = np.where(g <= low_thresh)[0]
        if dst_pool.size == 0:
            dst_pool = np.arange(n)
        i_dst = rng.choice(dst_pool)

        # propose a modest transfer
        delta = cur_a[i_src] * rng.random() * 0.07   # up to 7 % of source mass
        if delta <= 0:
            continue
        new_a = cur_a.copy()
        new_a[i_src] -= delta
        new_a[i_dst] += delta

        new_val = _eval_fast(new_a)
        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            print(f"[grad ] improved to {cur_val:.6f}")
    return cur_a, cur_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (mass transfers + swaps)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = cur_a.shape[0]

    while time.time() < deadline:
        # 20 % chance of a pure swap, else a tiny mass transfer
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.03 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

    n = 4000
    global _GLOBAL_N, _GLOBAL_EPS
    _GLOBAL_N = n

    # --------------------------------------------------------------
    # Initialise from a known construction if available
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)   # uniform start
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)          # enforce non‑negativity
    a0 = a0 / a0.sum()                # normalise to ℓ₁ = 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # L‑BFGS schedule – a dense log‑space of smoothing parameters
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), -12, num=20)

    # ------------------------------------------------------------------
    # Multi‑restart L‑BFGS phase
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 30.0:   # reserve ~30 s for later phases
        restart += 1
        print(f"[restart {restart}]")
        # Random soft‑max parameters – wide variance → diverse shapes
        w = np.random.randn(n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 20.0:
                break
            _GLOBAL_EPS = eps

            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # Exact evaluation of the current iterate
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[L‑BFGS eps {eps:.1e}] new best = {best_val:.6f}")

        # ----------------------------------------------------------
        # Anti‑symmetric re‑ordering of the best sequence from this restart
        # ----------------------------------------------------------
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing on the permutation of the best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        print("[SA] starting simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val,
                                                deadline - 10.0)

    # ------------------------------------------------------------------
    # Gradient‑guided mass‑transfer refinement (new phase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[grad] starting gradient‑guided mass transfer")
        best_a, best_val = _gradient_mass_transfer(best_a, best_val,
                                                   deadline - 5.0)

    # ------------------------------------------------------------------
    # Final cheap hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing phase")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report final value (using the official evaluator)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required
    return best_a.tolist()
