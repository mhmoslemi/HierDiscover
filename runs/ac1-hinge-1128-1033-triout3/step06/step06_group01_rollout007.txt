Pseudo:

def mass_transfer_step(a):
    conv = _fft_convolve_self(a)
    maxconv = conv.max()
    mask = (conv >= maxconv - 1e-12).astype(float)
    # compute gradient:
    mask_rev = mask[::-1]
    grad = 2 * np.convolve(a, mask_rev)[len(mask)-1 - np.arange(n)]   # vector shape n.
    # Actually N=2n-1, conv_rev length = n+N-1 = 3n-2. index slice: N-1 - i
    # We'll compute using FFT for speed.

    # identify donors: indices where a>0 and grad high
    donors = np.argsort(-grad)  # descending
    recipients = np.argsort(grad)  # ascending

    # pick top donor and recipient pair
    i = donors[0]
    j = recipients[0]
    if i == j: return a (no move)
    # compute delta as small fraction of a_i: e.g., delta = a[i] * 0.1
    delta = a[i] * 0.1
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # project maybe need to clip? but sum preserved.
    return a_new

"""
Propose a step‑function sequence with a low value of
    2 * n * max(conv(a,a)) / (sum(a)^2)

The optimisation proceeds in three phases:

1. **Smoothed surrogate L‑BFGS** – many restarts optimise a
   log‑sum‑exp smoothing of the convolution peak.  The variables are
   unconstrained soft‑max parameters, guaranteeing non‑negativity and
   ℓ₁‑norm = 1.  A schedule of decreasing smoothing parameters (ε) is
   used to approach the true non‑smooth objective.

2. **Permutation SA + guided mass‑transfer** – starting from the best
   sequence of phase 1 we run a simulated‑annealing loop that mixes
   ordinary swaps (reordering) with explicit mass‑transfer moves that
   relocate probability mass away from the indices that realise the
   current convolution maximum.  The temperature is slowly cooled and
   the best sequence seen is retained.

3. **Final hill‑climbing** – a cheap deterministic refinement that
   repeatedly tries swaps and larger mass‑transfer steps until the
   time budget expires.

All heavy work is performed with FFT‑based convolutions
(`_fft_convolve_self`) and the fast evaluator `_eval_fast`,
which yields exactly the same value as the official `evaluate_sequence`
but is orders of magnitude faster.  The routine respects a 1000 s
budget and always returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth‑surrogate wrapper (avoid closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a via FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)      # clamp tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective defined in `evaluate_sequence`
    (except for the final security checks).  `a` must be non‑negative
    and already respect the clipping limits.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smoothed surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns the smoothed surrogate L_ε and its gradient w.r.t. the
    soft‑max parameters w.
    """
    # ---- soft‑max (guarantees a_i ≥ 0 and Σ a_i = 1) ----
    w_shift = w - np.max(w)          # improve numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a /= a_sum

    # ---- convolution a * a (full) ----
    conv = _fft_convolve_self(a)     # length 2*n-1
    max_conv = conv.max()

    # ---- smooth approximation of max(conv) ----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)     # log Σ exp((conv – max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a (closed‑form) ----
    weights = np.exp(shifted)
    p = weights / weights.sum()      # probability distribution over k

    # convolution of a with the reversed probability mask
    # (the gradient of Σ p_k * (a*a)_k w.r.t. a)
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)   # helper defined below
    # middle n entries correspond to indices [n‑1, 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # ---- chain rule through the soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# Helper: convolution used inside the gradient (FFT based, same as above)
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# Bind the helper to the gradient routine
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Heuristic re‑ordering
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic anti‑symmetric ordering:
    large values on the left half, small values on the right half.
    Keeps the multiset of heights unchanged.
    """
    n = a.shape[0]
    s = np.sort(a)               # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[n - 1 - i]          # large → left
        out[n - 1 - i] = s[i]          # small → right
    if n % 2:
        out[half] = s[half]
    return out


# ----------------------------------------------------------------------
# Simulated‑annealing on permutations + guided mass transfers
# ----------------------------------------------------------------------
def _simulated_annealing_augmented(a: np.ndarray,
                                   cur_val: float,
                                   deadline: float) -> (np.ndarray, float):
    """
    Classic SA that mixes two move types:
      * a swap of two positions,
      * a mass‑transfer from one index to another.
    The current temperature is slowly decreased.
    Returns the best sequence seen before ``deadline``.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # initialise temperature – a small fraction of the current value
    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99995
    it = 0

    while time.time() < deadline:
        if rng.random() < 0.6:               # ---- swap move ----
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            new_val = _eval_fast(a)
            delta = new_val - cur_val
            accept = (delta < 0) or (rng.random() < np.exp(-delta / T))
            if accept:
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA] new best (swap) = {best_val:.6f} (iter {it})")
            else:
                a[i], a[j] = a[j], a[i]      # revert

        else:                                 # ---- mass‑transfer move ----
            # Identify the index where the convolution attains its maximum
            conv = _fft_convolve_self(a)
            k_star = int(np.argmax(conv))

            # donors are positions i that actually participate in that peak
            idx = np.arange(n)
            mask = (k_star - idx >= 0) & (k_star - idx < n) & (a > 1e-12)
            donors = idx[mask]
            if donors.size == 0:
                continue
            i = rng.choice(donors)
            # pick a random recipient different from i
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            # transfer a modest fraction of the donor mass
            delta = a[i] * 0.15 * rng.random()
            a[i] -= delta
            a[j] += delta
            new_val = _eval_fast(a)
            delta_obj = new_val - cur_val
            accept = (delta_obj < 0) or (rng.random() < np.exp(-delta_obj / T))
            if accept:
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA] new best (mass) = {best_val:.6f} (iter {it})")
            else:
                # revert the transfer
                a[i] += delta
                a[j] -= delta

        # temperature schedule
        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (mass transfers + swaps)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float) -> (np.ndarray, float):
    """
    Deterministic refinement that alternates swaps (20 % chance) with
    larger mass‑transfer moves (80 %).  The loop stops when the time
    budget expires.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # ---- pure swap ----
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cur_a[i], cur_a[j] = cur_a[j], cur_a[i]
        else:                                      # ---- mass transfer ----
            i = rng.integers(0, n)
            if cur_a[i] == 0.0:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            # transfer a fairly large fraction (up to 20 %)
            delta = cur_a[i] * (0.2 * rng.random())
            cur_a[i] -= delta
            cur_a[j] += delta

        val = _eval_fast(cur_a)
        if val < cur_val - 1e-12:          # strict improvement
            cur_val = val
            if cur_val < best_val:
                best_val = cur_val
                print(f"[refine] new best = {best_val:.6f}")

    return cur_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time
    budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin
    print(f"[init] time budget = {budget_s}s, deadline @ {deadline:.2f}")

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4000
    _globals = globals()
    _globals['_GLOBAL_N'] = n

    # --------------------------------------------------------------
    # Initialise from a known construction if available
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to requested size
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)   # uniform start
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)          # enforce non‑negativity
    a0 /= a0.sum()                    # ℓ₁‑normalise (sum = 1)
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # L‑BFGS schedule (log‑space smoothing)
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=20)

    # ------------------------------------------------------------------
    # Multi‑restart L‑BFGS phase
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 250.0:   # reserve ~250 s for later phases
        restart += 1
        print(f"[restart {restart}]")
        # random soft‑max parameters (wide variance → diverse shapes)
        w = np.random.randn(n) * 2.5

        for eps in eps_schedule:
            if time.time() >= deadline - 200.0:
                break
            _globals['_GLOBAL_EPS'] = eps

            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 600,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # evaluate the current iterate (exact objective)
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # ----------------------------------------------------------
        # Anti‑symmetric re‑ordering of the best sequence from this restart
        # ----------------------------------------------------------
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing on permutations + guided mass transfers
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print("[SA] starting augmented simulated annealing")
        best_a, best_val = _simulated_annealing_augmented(
            best_a, best_val, deadline - 100.0)

    # ------------------------------------------------------------------
    # Final cheap hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing phase")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report final value (using the official evaluator)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required
    return best_a.tolist()
