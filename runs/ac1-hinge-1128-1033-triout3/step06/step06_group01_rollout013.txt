def _lp_improvement(a, M, t, sum_target=None):
    n = len(a)
    if sum_target is None: sum_target = np.sum(a)
    # Build A_ub
    rows = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    cols = np.concatenate([np.full(n, j, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64) # maximize sum b

    # Solve using HiGHS
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                                 method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        print("[LP] Linprog failed:", res.message)
        return a.copy()
    b_opt = res.x
    # Clamp near-zero negatives
    b_opt = np.maximum(b_opt, 0.0)

    sum_b = np.sum(b_opt)
    if sum_b == 0:
        print("[LP] Zero sum solution, skip")
        return a.copy()
    scale = sum_target / sum_b
    b_scaled = b_opt * scale
    # Combine
    new_a = (1 - t) * a + t * b_scaled

    # Clip to bounds
    new_a = np.clip(new_a, 0.0, 1000.0)
    # re-normalize sum to sum_target? Already approx due to clipping but we can renormalize
    new_sum = np.sum(new_a)
    if new_sum == 0:
        return a.copy()
    new_a = new_a * (sum_target / new_sum)
    return new_a

"""
Propose candidate step function via a hybrid of smooth surrogate
optimization, linear-programming refinement and stochastic local search.

The algorithm proceeds as follows:

1. Initialise a non‑negative length‑n sequence (default n=4000) from a
   previously found construction or from a uniform start. The sequence
   is normalised so that its ℓ₁‑norm equals 1 (the objective is scale‑invariant).

2. Repeatedly optimise a smooth surrogate of the original objective
   (log‑sum‑exp approximation of the max‑convolution) with L‑BFGS‑B.
   The surrogate parameters are the soft‑max logits; the gradient is
   obtained analytically using FFT‑based convolutions.

3. After each L‑BFGS restart a *linear‑programming improvement* step is
   performed: keeping the current maximal self‑convolution M fixed, we
   solve a linear program that maximises the ℓ₁‑norm of a new vector
   under the constraints (f * g)≤M.  The solution is rescaled and mixed
   with the current sequence; this tends to flatten the convolution
   profile.

4. A deterministic anti‑symmetric re‑ordering heuristics further reduces
   the peak of the self‑convolution without changing the multiset of
   heights.

5. A simulated‑annealing phase searches over permutations and also
   performs small mass‑transfer moves, allowing the algorithm to escape
   local minima.

6. Finally a cheap hill‑climbing refinement (swaps + mass transfers)
   polishes the best sequence found.

All stages respect a global time‑budget (default 1000 s).  The function
``propose_candidate`` returns the best list of floats before the budget
expires.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import scipy.sparse

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate wrapper (avoid closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()

def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)   # clamp tiny negatives
    return conv

def _eval_fast(a: np.ndarray) -> float:
    """
    Fast evaluation of the exact objective defined in ``evaluate_sequence``.
    The array ``a`` is assumed to be non‑negative and already satisfies the
    clipping limits (0 ≤ a_i ≤ 1000).  Returns the same value as the official
    evaluator but is much faster.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

# Helper convolution used inside the surrogate gradient
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

# ----------------------------------------------------------------------
# Smooth surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate of the original objective and its gradient w.r.t
    the soft‑max logits ``w``.
    """
    # ----- soft‑max (ensures a_i ≥ 0 and Σ a_i = 1) -----
    w_shift = w - np.max(w)                # for numerical stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # ----- self‑convolution -----
    conv = _fft_convolve_self(a)           # length 2·n‑1
    max_conv = conv.max()

    # ----- smooth approximation of the max -----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # log‑sum‑exp
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a (closed‑form) -----
    weights = np.exp(shifted)
    p = weights / weights.sum()           # probability distribution over k

    # convolution of a with reversed p
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)   # helper defined below
    seg = conv_ap[n - 1: 2 * n - 1]        # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w

def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Produce a deterministic ordering where large values are placed at the
    left part and the smallest ones at the symmetric right part.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large to the left
        new_a[n - 1 - i] = sorted_vals[i]      # small to the right

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a

# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------
def _lp_improvement(a: np.ndarray, M: float, t: float, sum_target: float,
                    deadline: float) -> np.ndarray:
    """
    Given a current sequence ``a`` whose self‑convolution never exceeds ``M``,
    solve a linear program that maximises the ℓ₁‑norm while respecting the
    same convolution bound.  The optimum is then mixed with ``a`` using a
    convex combination factor ``t`` (0 < t ≤ 1).  The result is returned
    (clipped to [0, 1000] and rescaled to have ℓ₁‑norm ``sum_target``).
    The function aborts and returns ``a`` unchanged if the remaining time
    budget is insufficient.
    """
    if time.time() > deadline - 5.0:   # leave a few seconds for later phases
        return a.copy()

    n = a.shape[0]

    # Build the constraint matrix A such that (a * b) = A @ b
    # Each column j contributes the pattern a[0], a[1], …, a[n‑1] shifted by j.
    # We construct it in a vectorised way using integer broadcasting.
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)

    # row indices: i + j for every pair (i, j)
    row_idx = (i[:, None] + j[None, :]).ravel()
    # column indices: repeat each j n times
    col_idx = np.tile(j, n)
    # data: a[i] repeated for each column
    data = np.tile(a, n)

    A = scipy.sparse.csr_matrix((data, (row_idx, col_idx)),
                                shape=(2 * n - 1, n))

    # Upper‑bound vector (M for each of the 2n‑1 convolution entries)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)

    # Solve the LP using HiGHS (fast interior‑point method)
    # We set a modest time limit just in case.
    lp_options = {'presolve': True, 'time_limit': max(1.0, deadline - time.time() - 2.0)}
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options=lp_options)

    if not res.success:
        # If the LP fails for any reason we simply skip the refinement.
        print("[LP] linprog failed:", res.message)
        return a.copy()

    b_opt = np.maximum(res.x, 0.0)          # eliminate tiny negatives
    sum_b = b_opt.sum()
    if sum_b == 0.0:
        return a.copy()

    # Rescale to preserve the original ℓ₁‑norm
    scale = sum_target / sum_b
    b_scaled = b_opt * scale

    # Convex combination with the old sequence
    new_a = (1.0 - t) * a + t * b_scaled

    # Clip to the evaluator’s admissible interval and re‑normalise (the
    # clipping can slightly change the ℓ₁‑norm).
    new_a = np.clip(new_a, 0.0, 1000.0)
    new_sum = new_a.sum()
    if new_sum > 0:
        new_a = new_a * (sum_target / new_sum)
    return new_a

# Simulated annealing on permutations (swap + mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """
    Simulated annealing that works on the permutation of ``a`` and also
    supports tiny mass‑transfer moves.  Returns the best sequence and its
    evaluation found before ``deadline``.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.02, 1e-6)   # initial temperature
    decay = 0.99996

    iter_cnt = 0
    while time.time() < deadline:
        if rng.random() < 0.7:   # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            new_val = _eval_fast(a)
            delta = new_val - cur_val
            accept = (delta < 0) or (rng.random() < np.exp(-delta / T))
            if accept:
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA] new best = {best_val:.6f} (iter {iter_cnt})")
            else:
                # revert the swap
                a[i], a[j] = a[j], a[i]
        else:   # mass‑transfer move (similar to hill‑climb but always tried)
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            # transfer a random fraction (up to 5 %) of a[i] to a[j]
            frac = 0.05 * rng.random()
            delta = a[i] * frac
            a[i] -= delta
            a[j] += delta
            new_val = _eval_fast(a)
            delta_obj = new_val - cur_val
            accept = (delta_obj < 0) or (rng.random() < np.exp(-delta_obj / T))
            if accept:
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA] new best (mass) = {best_val:.6f} (iter {iter_cnt})")
            else:
                # revert transfer
                a[i] += delta
                a[j] -= delta

        # cooling schedule
        T = max(T * decay, 1e-12)
        iter_cnt += 1
    return best_seq, best_val

# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    """
    Simple hill‑climbing that repeatedly attempts swaps or small
    mass‑transfer moves and accepts only strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:
            # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

        new_a[n - 1 - i] = sorted_vals[i]      # small to the right

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a

# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------
def _lp_improvement(a: np.ndarray, M: float, t: float, sum_target: float,
                    deadline: float) -> np.ndarray:
    """
    Linear‑programming step: maximise ℓ₁‑norm of a new vector `b` under the
    constraint `conv(a, b) ≤ M`.  The optimal `b` is rescaled to match the
    original ℓ₁‑norm and mixed with `a` using factor `t`.  Returns the
    refined vector (clipped to [0,1000] and renormalised).
    """
    if time.time() > deadline - 5.0:   # give the later phases a chance
        return a.copy()

    n = a.shape[0]

    # Build sparse constraint matrix A such that (a * b) = A @ b.
    # For column j we have a shifted copy of `a` placed starting at row j.
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)

    row_idx = (i[:, None] + j[None, :]).ravel()   # i + j for each pair
    col_idx = np.tile(j, n)                       # repeat each column index n times
    data = np.tile(a, n)                          # a[i] repeated for each column

    A = scipy.sparse.csr_matrix((data, (row_idx, col_idx)),
                                shape=(2 * n - 1, n))

    # Upper bounds: M for each convolution entry
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)

    # Solve with HiGHS (fast interior‑point solver)
    lp_opts = {'presolve': True,
               'time_limit': max(1.0, deadline - time.time() - 2.0)}
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs', options=lp_opts)

    if not res.success:
        print("[LP] linprog failed:", res.message)
        return a.copy()

    b_opt = np.maximum(res.x, 0.0)
    sum_b = b_opt.sum()
    if sum_b == 0.0:
        return a.copy()

    # Rescale so that ℓ₁‑norm matches the original sequence
    scale = sum_target / sum_b
    b_scaled = b_opt * scale

    # Convex combination with the old sequence
    new_a = (1.0 - t) * a + t * b_scaled

    # Clip to admissible range and re‑normalise (clipping may change the sum)
    new_a = np.clip(new_a, 0.0, 1000.0)
    new_sum = new_a.sum()
    if new_sum > 0:
        new_a = new_a * (sum_target / new_sum)
    return new_a

# ----------------------------------------------------------------------
# Simulated annealing on permutations (swap + mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """
    Simulated annealing that manipulates the permutation of ``a`` and also
    performs occasional tiny mass‑transfer moves.  Returns the best sequence
    and its evaluation observed before ``deadline``.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.02, 1e-6)   # initial temperature
    decay = 0.99996
    it = 0

    while time.time() < deadline:
        if rng.random() < 0.7:       # swap move (70 % of the time)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            new_val = _eval_fast(a)
            delta = new_val - cur_val
            accept = (delta < 0) or (rng.random() < np.exp(-delta / T))
            if accept:
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA] new best = {best_val:.6f} (iter {it})")
            else:
                a[i], a[j] = a[j], a[i]     # revert
        else:                         # tiny mass‑transfer move (30 %)
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            frac = 0.05 * rng.random()
            delta = a[i] * frac
            a[i] -= delta
            a[j] += delta
            new_val = _eval_fast(a)
            delta_obj = new_val - cur_val
            accept = (delta_obj < 0) or (rng.random() < np.exp(-delta_obj / T))
            if accept:
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA] new best (mass) = {best_val:.6f} (iter {it})")
            else:
                a[i] += delta
                a[j] -= delta

        T = max(T * decay, 1e-12)
        it += 1
    return best_seq, best_val

# ----------------------------------------------------------------------
# Final cheap hill‑climbing (mass transfers + swaps)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    """
    Simple hill‑climbing that repeatedly attempts swaps or small
    mass‑transfer moves, accepting only strict improvements.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:          # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                           # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function minimising ``evaluate_sequence``.
    Returns the best sequence (list of floats) before the time budget
    expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin
    print(f"[init] time budget = {budget_s}s, deadline @ {deadline:.2f}")

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4000
    globals()['_GLOBAL_N'] = n

    # --------------------------------------------------------------
    # Initialise from a historic construction if possible
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)   # uniform start
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()           # ℓ₁‑norm = 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # L‑BFGS schedule (log‑space smoothing)
    # --------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-11), num=18)

    restart = 0
    # Keep ~40 s for the later stochastic phases
    while time.time() < deadline - 40.0:
        restart += 1
        print(f"[restart {restart}]")
        w = np.random.randn(n) * 2.0   # random logits

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            globals()['_GLOBAL_EPS'] = eps

            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 600,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # evaluate current iterate
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # ----------------------------------------------------------
        # Linear‑programming refinement
        # ----------------------------------------------------------
        if time.time() < deadline - 30.0:
            M = np.max(_fft_convolve_self(best_a))
            t_lp = 0.25                     # mixing factor
            refined = _lp_improvement(best_a, M, t_lp, best_a.sum(), deadline)
            refined_val = _eval_fast(refined)
            if refined_val < best_val:
                best_val = refined_val
                best_a = refined.copy()
                print(f"[LP] improved to {best_val:.6f}")

        # ----------------------------------------------------------
        # Anti‑symmetric re‑ordering
        # ----------------------------------------------------------
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Simulated annealing (swap + mass‑transfer)
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:
        print("[SA] starting simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val,
                                                deadline - 10.0)

    # --------------------------------------------------------------
    # Final cheap hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing phase")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # Report final value using the official evaluator
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
