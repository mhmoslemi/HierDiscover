def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    n = 4000  # or 3000 to be faster?
    
    # compute number of restarts based on time
    best_val = np.inf
    best_a = None

    while time.time() < deadline - 5:
        # generate initial sequence
        # Choose a random spike count k randomly between min_k and max_k.
        # Generate k random distinct positions.
        # Sample random positive weights (e.g., exponential) and normalize sum to 1.
        # Also optionally blend with best to keep some diversity.
        # random geometry.

        if best_a is not None and rng.random() < 0.3:
            # start from best with small perturbation
            a = best_a.copy()
            # add Gaussian noise and project onto simplex
            noise = rng.normal(scale=0.02, size=n)
            a = a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 0:
                a = a / s
            else:
                a = np.full(n, 1.0/n)
        else:
            # random spikes
            k = rng.integers(10, min(n, int(np.sqrt(2*n)*2)))  # at most ~200
            positions = rng.choice(n, size=k, replace=False)
            weights = rng.exponential(scale=1.0, size=k)
            a = np.zeros(n)
            a[positions] = weights
            a_sum = a.sum()
            if a_sum > 0:
                a = a / a_sum
            else:
                a = np.full(n, 1.0/n)

        # perform local simulated annealing for this restart:
        cur_a = a
        cur_val = evaluate_fast(cur_a)  # compute fast objective
        # keep track best local
        local_best_a = cur_a.copy()
        local_best_val = cur_val

        # annealing schedule: temperature decay
        # number of steps ~ some limit based on time
        inner_start = time.time()
        # compute approximate steps allowed: e.g., 2000 steps per restart or time-limited.
        steps = 2000
        for step in range(steps):
            if time.time() >= deadline:
                break
            # propose move
            # use source idx with a_i > small
            src_candidates = np.where(cur_a > 1e-12)[0]
            if src_candidates.size == 0:
                break
            i = rng.choice(src_candidates)
            j = rng.integers(0, n)
            if i == j:
                continue
            # delta proportion
            max_delta = cur_a[i] * 0.5  # can't remove all
            delta = rng.random() * max_delta
            # apply move
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta

            # evaluate
            cand_val = evaluate_fast(cand)
            # accept if better or with probability exp(-(cand_val - cur_val)/T)
            # temperature schedule T = T0 * (1 - step/steps)
            T = max(1e-4, 0.01 * (1 - step/steps))
            accept = False
            if cand_val < cur_val:
                accept = True
            else:
                if rng.random() < np.exp(-(cand_val - cur_val)/T):
                    accept = True
            if accept:
                cur_a = cand
                cur_val = cand_val
                if cand_val < local_best_val:
                    local_best_val = cand_val
                    local_best_a = cand.copy()
        # after inner loop, compare with global best:
        if local_best_val < best_val:
            best_val = local_best_val
            best_a = local_best_a.copy()
            print(f"[restart] new global best = {best_val:.6f}")

    # after time, final evaluation using evaluate_sequence
    if best_a is None:
        # fallback: uniform
        best_a = np.full(n, 1.0/n)
        best_val = evaluate_fast(best_a)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation = {final_score:.6f}")
    return best_a.tolist()

def evaluate_fast(a):
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

"""
Propose a step‑function (non‑negative list of heights) that minimizes
``evaluate_sequence``.

Algorithm
========
The objective  f(a)=2·n·max(conv(a,a))/‖a‖₁²  is convex but non‑smooth.
We minimise it by a two‑phase heuristic:

1. **Smoothed projected gradient descent**
   * ``β`` controls a smooth approximation of the max:
         s(a)= (1/β)·log Σ exp(β·conv(a,a)).
     Its gradient is a weighted sum of the usual sub‑gradient
     (weights are the soft‑max over the convolution values).
   * At each iteration we take a step opposite the gradient and
     project back onto the probability simplex (non‑negative entries,
     unit ℓ₁‑norm).  The step size follows ηₜ = η₀/√t and η₀ is
     adaptively increased when improvements are seen.
   * ``β`` is increased from a small value (≈0.5) to a larger one
     (≈12) so the optimisation first moves in a smooth landscape and
     then tightens to the true max.

2. **Guided hill‑climbing (mass‑transfer)**
   * Starting from the best vector obtained in phase 1 we repeatedly
     move a small fraction of mass away from the most offending lag
     (the lag where the convolution attains its maximum).  The move
     transfers the mass to two positions with below‑average height.
   * The procedure stops when the time budget expires.

Multiple random restarts are performed: each restart starts either from
the current global best (with a tiny Gaussian perturbation) or from a
random “spike” vector (few random positions with exponential weights).
All work is limited by the supplied 1000 s budget.

The code uses only NumPy and the already‑imported ``evaluate_sequence``
function.  No file‑ or network‑IO is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negatives caused by numerical noise
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (same formula as evaluate_sequence, but without the
# security clamps – we guarantee non‑negative inputs ourselves)
# ----------------------------------------------------------------------
def _fast_obj(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a))/‖a‖₁²."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Gradient of the smoothed max (log‑sum‑exp) of the convolution
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of  (1/β)·log Σ exp(β·conv(a,a))
    and also return the exact max value of conv(a,a).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_conv = conv.max()

    # soft‑max weights (numerically stable)
    shift = conv - max_conv               # max_conv is subtracted to avoid overflow
    w = np.exp(beta * shift)
    w /= w.sum()                          # w has length 2·n‑1

    # gradient: g_i = 2· Σ_k w_k·a_{k‑i}
    # This is a correlation of w with a, which we compute via FFT.
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fw = np.fft.rfft(w_rev, size)
    fa = np.fft.rfft(a, size)
    corr_full = np.fft.irfft(fw * fa, size)[:len(w_rev) + n - 1]

    m = len(w_rev)                # = 2·n‑1
    idx = m - 1 - np.arange(n)    # indices that give Σ_j w_{i+j} a_j
    grad = 2.0 * corr_full[idx]
    return grad, max_conv


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Reduce the exact objective by moving mass away from the most offending
    convolution lag.  Stops when the time budget expires.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    step_frac = 0.05

    while time.time() < deadline:
        conv = _full_convolution(best_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # pairs that contribute to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_rng = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_rng = k - i_rng
        contrib = best_a[i_rng] * best_a[j_rng]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_rng[idx])
        j_max = int(j_rng[idx])

        # choose two destination indices with low mass
        low = np.where(best_a < best_a.mean())[0]
        if low.size >= 2:
            p, q = rng.choice(low, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        # avoid recreating the same lag
        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # perform a feasible mass transfer
        if i_max != j_max:
            delta = min(best_a[i_max], best_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = best_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:                     # i_max == j_max (square term)
            delta = best_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = best_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        val = evaluate_sequence(new_a.tolist())
        if val < best_val:
            best_a = new_a
            best_val = val
            print(f"[hill] new best = {best_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return best_a, best_val
