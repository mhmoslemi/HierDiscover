def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of a real vector with itself.
    Length = 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:

    Full convolution of two real vectors using FFT.
    Length = len(x)+len(y)‑1.

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:                           # numerical safety
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w
