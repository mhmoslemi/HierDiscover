"""
Propose a step‑function (non‑negative sequence) that (approximately) minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages inside the 1000 s budget:

1. **Parametric warm‑start** – a dense grid of simple analytic families
   (exponential, power‑law and symmetric β‑like shapes) is scanned.
   Each candidate is normalised to a probability simplex, its objective is
   evaluated with a fast FFT based convolution and the best one is remembered.

2. **Local refinement** – from the best candidate we run a short
   projected gradient descent on the exact sub‑gradient of the max‑convolution
   (the same gradient used in the reference code) with an Adam optimiser.
   This quickly reduces the peak while keeping the vector on the simplex.

3. **Fine‑grained simulated annealing** – finally a cheap mass‑transfer
   Metropolis scheme (move a tiny amount of probability mass from one index
   to another) is performed for the remaining time.  The temperature is
   linearly decreased; every accepted move is kept if it improves the true
   evaluation score.

All vectors are kept non‑negative and are re‑projected onto the simplex after
each modification, guaranteeing that the final value can be fed directly to
`evaluate_sequence`.  The routine prints progress information and always returns
the best sequence seen before the time limit expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helpers ---------------------------------------------------------------
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x | x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (fast and accurate)."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(Fa * Fb, n=nfft)[:size]
    return conv


def _objective(a: np.ndarray) -> float:
    """Objective for a sequence that already sums to 1."""
    n = a.size
    conv = _full_convolution_fft(a, a)
    return float(2.0 * n * conv.max())


def _evaluate(seq: list[float]) -> float:
    """Call the provided evaluate_sequence safely (used only for final logging)."""
    try:
        return evaluate_sequence(seq)
    except Exception:
        return float("inf")

# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a non‑negative height sequence (list of floats) that strives to
    minimise the evaluation function within the given time budget."""

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 6000 works well for the reference runs and gives a
    # nice trade‑off between optimisation power and RAM usage.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # 1️⃣  Parametric warm‑start – exhaustive sweep of a few one‑parameter
    # families.  Every candidate is normalised to sum = 1 (the objective
    # is scale‑invariant) and evaluated with a fast FFT.
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start
    best_val = _objective(best_seq)
    print(f"[init] uniform objective = {best_val:.12f}")

    # ----- exponential families  a_i ∝ exp(-λ i) --------------------
    for lam in np.geomspace(5e-5, 5e-2, num=40):   # λ from very soft to steep
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a = _project_simplex(a)
        val = _objective(a)

            print(f"[exp]  λ={lam:.3e} → {val:.12f}")

    # ----- power‑law families  a_i ∝ (i+1)^{-p} --------------------
    for p in np.linspace(-1.5, 2.5, 40):

        a = _project_simplex(a)

        if val < best_val:

            print(f"[pow]  p={p:.2f} → {val:.12f}")

    # ----- symmetric beta‑like shapes a_i ∝ ((i+1)*(n-i))^{α-1} -----
    #  α = 1 gives uniform, α < 1 produces a U‑shape, α > 1 a hump.
    for alpha in np.linspace(0.05, 5.0, 45):

        shape = (i * (n - i + 1.0)) ** (alpha - 1.0)   # (i·(n‑i+1))^{α‑1}
        a = _project_simplex(shape)

    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        print(f"[result] official_score = {official:.12f}  length = {len(final_seq)}")
