"""
propose_candidate:  A time‑bounded stochastic mirror‑descent search.

The routine builds many random “warm‑start’’ step‑functions (mixtures of
exponential, power‑law, sine, Gaussian, triangular and Dirichlet shapes)
and repeatedly refines them with an exponentiated‑gradient (mirror‑descent)
iteration that directly minimizes the infinity‑norm of the self‑convolution
(and therefore the original evaluation function).  The gradient of the
max‑convolution is obtained analytically; the update is performed in the
log‑domain to guarantee non‑negativity and to stay on the probability simplex.
Occasional random jitter escapes flat regions.  After the main phase the best
candidate is fine‑tuned by a cheap simulated‑annealing “mass‑transfer’’ move.
The whole search respects the supplied time budget (default 1000 s) and prints
intermediate scores.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for the sizes we use).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution a*b using a real FFT."""
    size = a.size + b.size - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Objective for a vector that is already normalised (sum == 1):
#   2 * n * max(conv(a,a))
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.size
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Random warm‑start generator – convex combination of classic shapes and
# a Dirichlet sample.
# ----------------------------------------------------------------------
def _random_warm_start(rng: np.random.Generator, n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)

    shapes = []

    # exponential decays
    for _ in range(2):
        lam = rng.uniform(5e-4, 2e-2)
        shapes.append(np.exp(-lam * x))

    # power‑law tails
    for _ in range(2):
        p = rng.uniform(0.3, 2.0)
        shapes.append((x + 1) ** (-p))

    # half‑sine
    shapes.append(np.sin(np.pi * (x + 0.5) / n))

    # centred Gaussian
    sigma = rng.uniform(n / 8.0, n / 2.0)
    shapes.append(np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2))

    # simple triangle
    shapes.append(np.minimum(x, n - 1 - x) + 1.0)

    # Catalan‑type decreasing sequence
    catalan = np.empty(n, dtype=np.float64)
    cur = 1.0
    catalan[0] = cur
    for k in range(1, n):
        cur = cur * (2.0 * k - 1) / (2.0 * k)
        catalan[k] = cur
    shapes.append(catalan)

    # a Dirichlet‑type random shape to inject additional diversity
    alpha = rng.uniform(0.2, 5.0)
    shapes.append(rng.gamma(alpha, 1.0, size=n))

    # random convex combination of the above
    w = rng.random(len(shapes))
    w /= w.sum()
    a = sum(wi * sh for wi, sh in zip(w, shapes))
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Main entry point required by the grader.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative height sequence that (approximately) minimizes
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds,
    using a mixture of random warm‑starts, exponentiated‑gradient mirror‑descent
    and a short simulated‑annealing polishing phase.
    """
    start_time = time.time()
    # keep a safety margin (a few seconds) for the final conversion & printing
    deadline = start_time + budget_s - 4.0
    rng = np.random.default_rng(seed)

    # problem size – 6000 works well in memory and gives a fine resolution
    n = 6000

    best_val = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Main stochastic search: many short restarts, each running mirror‑descent.
    # ------------------------------------------------------------------
    # Reserve a few seconds for the final simulated‑annealing polish.
    final_sa_time = min(50.0, 0.05 * budget_s)   # at most 5 % of the budget
    search_deadline = deadline - final_sa_time

    restart_counter = 0
    while time.time() < search_deadline:
        restart_counter += 1

        # 1) initialise a random warm‑start
        a = _random_warm_start(rng, n)

        # local best for this restart
        local_best_a = a.copy()
        local_best_val = _objective(a)

        # 2) mirror‑descent (exponentiated gradient) loop
        it = 0
        stagn = 0
        # we allow up to ~8000 iterations per restart – it stays cheap
        max_iters = 8000
        while it < max_iters and time.time() < search_deadline:
            it += 1

            # full self‑convolution and current objective
            conv = _full_convolution_fft(a, a)
            max_conv = conv.max()
            val = 2.0 * n * max_conv

            # keep the best seen so far (global)
            if val < best_val - 1e-13:
                best_val = val
                best_seq = a.copy()
                print(f"[global] restart {restart_counter:03d} iter {it:04d} "
                      f"new_best={best_val:.12f}")

            # keep the best for this restart (used later for SA)
            if val < local_best_val - 1e-13:
                local_best_val = val
                local_best_a = a.copy()
                stagn = 0
            else:
                stagn += 1

            # ----- sub‑gradient of the max‑convolution -----
            # active indices where the max is attained (within tolerance)
            tol = 1e-14
            active = np.flatnonzero(conv >= max_conv - tol)
            w = np.zeros_like(conv)
            if active.size:
                w[active] = 1.0
            else:
                w[np.argmax(conv)] = 1.0

            # g = 2 * (rev_a * w) , take the central n entries
            rev_a = a[::-1]
            g_full = _full_convolution_fft(rev_a, w)
            g = 2.0 * g_full[n - 1:n - 1 + n]

            # ----- exponentiated‑gradient (mirror descent) update -----
            # step size slowly decays (1/sqrt(t))
            eta = 5.0 / math.sqrt(it)

            # log‑domain update to stay on the simplex automatically
            log_a = np.log(a + 1e-30) - eta * g
            a = np.exp(log_a)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= a_sum

            # occasional jitter to avoid getting stuck for a long time
            if stagn > 3000:
                jitter = rng.normal(scale=0.001, size=n)
                a = a + jitter
                a = np.clip(a, 0, None)
                a_sum = a.sum()
                if a_sum == 0:
                    a = np.full_like(a, 1.0 / n)
                else:
                    a /= a_sum
                stagn = 0
                print(f"[restart {restart_counter}] jitter at iter {it}")

        # ------------------------------------------------------------------
        # One short simulated‑annealing polish on the local best of this restart
        # (only if we still have time left after the main phase).
        # ------------------------------------------------------------------
        # We run a very lightweight SA (max 20000 moves) – it is cheap and may
        # improve the local best a little more.
        if time.time() < deadline - 5.0:          # keep a tiny safety margin
            cur = local_best_a.copy()
            cur_score = _objective(cur)

            T0 = 5e-4
            Tmin = 1e-7
            it_sa = 0
            max_sa_moves = 20000
            while it_sa < max_sa_moves and time.time() < deadline:
                it_sa += 1
                T = max(Tmin, T0 * (1.0 - it_sa / max_sa_moves))

                i, j = rng.choice(n, size=2, replace=False)
                if cur[i] <= 0.0:
                    continue
                # transfer a modest fraction of mass
                max_transfer = min(cur[i], 0.01)
                delta = rng.uniform(0.0, max_transfer)
                new = cur.copy()
                new[i] -= delta
                new[j] += delta

                new_score = _objective(new)
                if new_score < cur_score:
                    accept = True
                else:
                    accept = rng.random() < math.exp(-(new_score - cur_score) / T)

                if accept:
                    cur = new
                    cur_score = new_score
                    if cur_score < best_val - 1e-13:
                        best_val = cur_score
                        best_seq = cur.copy()
                        print(f"[SA global] restart {restart_counter} move {it_sa} "
                              f"new_best={best_val:.12f}")

                # occasional global jitter (helps escape plateaus)
                if it_sa % 5000 == 0:
                    cur = cur + rng.normal(scale=0.0005, size=n)
                    cur = np.clip(cur, 0, None)
                    cur_sum = cur.sum()
                    if cur_sum == 0:
                        cur = np.full_like(cur, 1.0 / n)
                    else:
                        cur /= cur_sum
                    cur_score = _objective(cur)

    # ------------------------------------------------------------------
    # If, for any reason, we never found a candidate, fall back to uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Return the best sequence in the format expected by `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # Optional: print the official score (may be unavailable in some envs)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] evaluate_sequence not available")

    return final_seq
