"""
Search for a non‑negative step‑function (list of heights) that minimizes
`evaluate_sequence`.

Algorithm
---------
1. **Coarse multi‑start search on a small problem (n≈1500).**
   * Randomly generate a few thousand initial vectors from simple
     parametric families (uniform, linear, exponential, Dirichlet) and
     shuffle their entries.
   * For each start run a fast targeted hill‑climber (exact convolution)
     for at most a few seconds.  The best small‑scale candidate (lowest
     objective) is retained.

2. **Upsample to the full size (n≈8000).**
   * Linear interpolation preserves the “shape’’ while providing the
     required length.  The vector is re‑normalised to unit sum.

3. **Refine on the full size.**
   * **Phase A – Adam optimisation** on a smooth approximation of
     `max(conv(a,a))` (log‑sum‑exp).  The soft‑max logits are updated on
     the simplex.  Learning‑rate and smoothing temperature decay with
     time.
   * **Phase B – Targeted hill‑climbing** identical to the one used in
     the baseline, but started from the better initial point.
   * **Phase C – Simulated‑annealing** to escape possible plateaus.

All phases respect the total time budget (default 1000 s).  The final
vector is passed to the provided `evaluate_sequence` wrapper and
returned as a plain list of floats (non‑negative, sum unrestricted –
the objective is scale‑invariant).

The combination of a cheap global search on a smaller problem and a
more powerful local optimiser on the full problem usually discovers
a lower bound than the baseline (target ≈ 1.503).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself, length 2·len(a)‑1."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) assumed = 1 => 2·n·max(b))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simple parametric families (used for the coarse small‑scale search)
# ----------------------------------------------------------------------
def _make_candidate(n: int, kind: str) -> np.ndarray:
    """Create a raw (non‑normalised) vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0)
    elif kind == "linear":
        a = np.arange(n, 0, -1, dtype=float)
    elif kind == "quadratic":
        a = (np.arange(n, 0, -1, dtype=float) ** 2)
    elif kind == "exp":
        r = np.random.uniform(0.990, 0.9999)
        a = r ** np.arange(n, dtype=float)
    elif kind == "dirichlet":
        alpha = np.random.uniform(0.3, 3.0)
        a = np.random.gamma(alpha, 1.0, size=n)
    else:
        a = np.full(n, 1.0)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    np.random.shuffle(a)           # break any symmetry
    return a


# ----------------------------------------------------------------------
# Small‑scale hill‑climber (exact objective, O(n) updates)
# ----------------------------------------------------------------------
def _hillclimb_targeted_small(a_start: np.ndarray,
                              n: int,
                              start: float,
                              deadline: float,
                              best_a: np.ndarray,
                              best_val: float,
                              target_val: float = 1.503) -> np.ndarray:
    """
    Same logic as the large‑scale hill‑climber but aborts when the
    deadline is reached.  Returns the best vector seen during the run.
    """
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 120_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contributions to the current maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – biased towards large contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – mostly uniform, sometimes favouring little contribution
        if np.random.rand() < 0.30:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = np.random.choice(n, p=probs_j)
            else:
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # Incremental convolution update (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-14:           # strict improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[small hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # random reshuffle of the current best – a strong perturbation
            a = best_a.copy()
            np.random.shuffle(a)
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[small hill {it:7d}] reshuffling after stagnation")
        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth approximation of max(conv(a,a))."""
    w = np.log(a0 + 1e-12)                 # soft‑max logits, Σa = 1

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.09, 0.001
    eps_start, eps_end = 1e-1, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 30_000

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # ----- soft‑max to obtain a on the simplex -----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()                       # Σa = 1, a ≥ 0

        cur_val = _objective_value(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ----- schedules -----
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # ----- smooth‑max via log‑sum‑exp -----
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()            # soft max over positions

        # ----- gradient of smooth objective w.r.t. a -----
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)          # length 3·n‑2
        segment = conv_ab[n - 1: 2 * n - 1]    # n entries that correspond to ∂c/∂a
        grad_a = 4.0 * n * segment[::-1]

        # ----- chain rule through the soft‑max -----
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional jitter to avoid dead‑ends
        if np.random.rand() < 0.006:
            w += np.random.normal(scale=0.015, size=n)

        if no_improve > IMPROVE_LIMIT:
            print(f"[Adam {it:7d}] restart after {no_improve} idle steps")
            w = np.random.randn(n) * 0.12
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing (large scale)
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> np.ndarray:
    """Fast local search that transfers a tiny amount of mass between indices."""
    return _hillclimb_targeted_small(a_start, n, start, deadline,
                                     best_a, best_val, target_val)


# ----------------------------------------------------------------------
# Phase C – short simulated‑annealing refinement
# ----------------------------------------------------------------------
def _annealing_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Metropolis‑style search that occasionally accepts worsening moves."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    T_start, T_end = 6e-4, 4e-8
    IMPROVE_LIMIT = 100_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        k_max = int(np.argmax(b))

        # source selection – bias to indices contributing to the current max
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        tot = contrib.sum()
        if tot > 0:
            probs = contrib / tot
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # simple uniform target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.005
        if delta <= 0.0:
            continue

        # incremental convolution update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            noise = np.random.normal(scale=0.004, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal {it:7d}] perturbing after stagnation")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Small‑scale exploratory phase
# ----------------------------------------------------------------------
def _small_exploratory_phase(start_time: float,
                             overall_deadline: float,
                             n_small: int = 1500) -> tuple[np.ndarray, float]:
    """
    Run a quick multi‑start hill‑climber on a reduced problem size.
    Returns the best vector found (unit sum) and its objective value.
    """
    deadline = start_time + (overall_deadline - start_time) * 0.30
    best_a = None
    best_val = np.inf
    trial = 0

    while True:
        now = time.time()
        if now >= deadline:
            break
        # allocate a small time slice (≈2 s) for each trial
        remaining = deadline - now
        trial_time = min(2.0, remaining)
        trial_deadline = now + trial_time
        # pick a random family
        kind = np.random.choice(["uniform", "linear", "exp", "dirichlet"])
        a0 = _make_candidate(n_small, kind)

        cur_val = _objective_value(a0)
        # run a focused hill‑climber on this start
        a_opt = _hillclimb_targeted_small(a0, n_small,
                                          start=now,
                                          deadline=trial_deadline,
                                          best_a=a0,
                                          best_val=cur_val,
                                          target_val=1.503)
        val_opt = _objective_value(a_opt)

        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Small trial {trial:4d}] new global best = {best_val:.6f}")

        trial += 1

    # If we never succeeded (highly unlikely) fall back to a uniform vector
    if best_a is None:
        best_a = np.full(n_small, 1.0 / n_small)
        best_val = _objective_value(best_a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Upsampling helper
# ----------------------------------------------------------------------
def _upsample(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation of a small vector to a larger length."""
    old_n = a_small.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, n_large)
    a_large = np.interp(new_x, old_x, a_small)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()
    return a_large


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a step‑function within the given time budget.
    Returns a list of non‑negative floats that minimises `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # 1️⃣  Small‑scale global exploration (≈30 % of the budget)
    # --------------------------------------------------------------
    n_small = 1500
    best_a_small, best_val_small = _small_exploratory_phase(start_time,
                                                             deadline,
                                                             n_small=n_small)
    print(f"[Summary] best small‑scale value = {best_val_small:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Upsample to the full size required for the final evaluation
    # --------------------------------------------------------------
    n_large = 8000
    a0_large = _upsample(best_a_small, n_large)

    # Try to reuse a previously found construction if it exists
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n_large)
        a_old = np.interp(new_x, old_x, old)
        a_old = np.maximum(a_old, 0.0)
        a_old /= a_old.sum()
        # pick the better of the upsampled candidate and the old one
        if _objective_value(a_old) < _objective_value(a0_large):
            a0_large = a_old.copy()
    except Exception:
        pass

    best_a = a0_large.copy()
    best_val = _objective_value(best_a)
    TARGET = 1.503

    # --------------------------------------------------------------
    # 3️⃣  Phase A – Adam optimisation on a smooth max
    # --------------------------------------------------------------
    phase_a_deadline = start_time + (deadline - start_time) * 0.30
    best_a, best_val = _adam_phase(a0_large, n_large,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=TARGET)

    # --------------------------------------------------------------
    # 4️⃣  Phase B – targeted hill‑climbing on the exact objective
    # --------------------------------------------------------------
    phase_b_deadline = start_time + (deadline - start_time) * 0.80
    best_a = _hillclimb_targeted(best_a, n_large,
                                 start=phase_a_deadline,
                                 deadline=phase_b_deadline,
                                 best_a=best_a,
                                 best_val=best_val,
                                 target_val=TARGET)

    # --------------------------------------------------------------
    # 5️⃣  Phase C – short simulated annealing refinement
    # --------------------------------------------------------------
    best_a, best_val = _annealing_phase(best_a, n_large,
                                        start=phase_b_deadline,
                                        deadline=deadline,
                                        best_a=best_a,
                                        best_val=best_val,
                                        target_val=TARGET)

    # --------------------------------------------------------------
    # Final evaluation using the provided secure wrapper
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
