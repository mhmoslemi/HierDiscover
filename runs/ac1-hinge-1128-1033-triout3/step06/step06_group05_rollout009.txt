def _adam_phase(a, n, deadline, best_a, best_val):
    start = time.time()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 1
    max_step = 0.05
    while time.time() < deadline:
        # compute b
        b = _full_convolution(a) # length 2n-1
        # compute exact evaluation for tracking
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam {it}] new best = {best_val:.6f}")

        # compute gamma schedule: small -> large
        # let fraction = (time.time() - start) / (deadline - start)
        # gamma = 0.1 + 10 * fraction**2 maybe.
        progress = (time.time() - start) / (deadline - start)
        gamma = 0.1 + 10 * progress  # gamma from 0.1 to 10.1

        b_max = b.max()
        # compute w normalized
        w = np.exp(gamma * (b - b_max))
        w_sum = w.sum()
        w_norm = w / w_sum

        # compute gradient: 2 * conv(w_norm, a_rev)[n-1:2n-1]
        a_rev = a[::-1]
        grad_full = _conv_fft(w_norm, a_rev)
        grad = 2.0 * grad_full[n-1:2*n-1]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # step size schedule
        step = max_step * (1 - progress) + 1e-4 * progress
        a_candidate = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a = _project_simplex(a_candidate)

        it += 1
    return best_a, best_val

def _optimal_hillclimb(a, n, deadline, best_a, best_val):
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    # but a[max_idx - i] defined if (max_idx - i) in [0,n-1] else 0.
    # We'll compute index range.
    # On each iteration, we will recompute contributions after updates.

    # We'll limit candidate sources/targets to top K=20 and bottom K=20 contributions.
    K = 20
    IMPROV_STREAK = 50000
    no_improve = 0
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)  # recompute full conv maybe expensive but okay; we can reuse b from previous step if we update incrementally.
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        max_idx = int(np.argmax(b))
        # second max
        # we can compute second max quickly
        # Set the max index to -inf temporarily
        tmp = b.copy()
        tmp[max_idx] = -np.inf
        second_max = tmp.max()

        # compute contributions to max
        contrib = np.zeros_like(a)
        i_low = max(0, max_idx - (n-1))
        i_high = min(max_idx, n-1)
        idx_range = np.arange(i_low, i_high+1)
        contrib[idx_range] = a[max_idx - idx_range] * 2.0
        if max_idx % 2 == 0:
            mid = max_idx // 2
            # for i = mid, a[mid] appears twice? but for derivative we used 2*a[mid].
            # Already captured above because when i=mid, a[max_idx - i]=a[mid] same. So still correct.

        # select sources (largest contrib) and targets (smallest)
        # but we also filter out zero a entries for sources.
        # Use argsort
        src_order = np.argsort(-contrib)  # descending
        tgt_order = np.argsort(contrib)   # ascending
        sources = src_order[:K]
        targets = tgt_order[:K]

        best_delta = 0.0
        best_pair = None
        best_new_val = cur_val
        # Evaluate each pair
        for s in sources:
            if a[s] < 1e-12: continue
            for t in targets:
                if s == t: continue
                # Linear coefficient for max index
                # d_max = -2*a[max_idx - s] + 2*a[max_idx - t]
                d_s = -2.0 * (a[max_idx - s] if 0 <= max_idx - s < n else 0.0)
                d_t =  2.0 * (a[max_idx - t] if 0 <= max_idx - t < n else 0.0)
                d = d_s + d_t
                if d >= 0:
                    continue   # moving mass does not reduce max
                # compute linear estimate delta to bring max down to second_max
                delta_est = (second_max - cur_max) / d  # negative/negative -> positive
                # clip to feasible range
                delta = min(a[s], max(0.0, delta_est))
                # also limit delta to fraction of source maybe up to 0.5
                if delta <= 0:
                    continue
                # For safety, bound delta to 0.5*a[s]
                delta = min(delta, 0.5 * a[s])

                # compute new max via incremental update
                # We'll implement a function to compute new max quickly
                # Using vectorized update like earlier
                b_new = b.copy()
                # cross terms
                # add 2*delta*a to target region
                start_t = t
                end_t = t + n
                b_new[start_t:end_t] += 2.0 * delta * a
                # subtract from source region
                start_s = s
                end_s = s + n
                b_new[start_s:end_s] -= 2.0 * delta * a
                # pure delta^2 terms
                b_new[2*t] += delta*delta
                b_new[2*s] += delta*delta
                b_new[s+t] -= 2.0 * delta*delta

                new_max = b_new.max()
                new_val = 2.0 * n * new_max
                if new_val < best_new_val - 1e-14:
                    best_new_val = new_val
                    best_pair = (s, t, delta, b_new)
        if best_pair is not None:
            s, t, delta, b = best_pair
            a[s] -= delta
            a[t] += delta
            cur_val = best_new_val
            max_idx = int(np.argmax(b))
            if best_new_val < best_val:
                best_val = best_new_val
                best_a = a.copy()
                print(f"[Hill_opt {it}] new best = {best_val:.6f}")
            no_improve = 0
        else:
            # no improving pair found: random perturbation
            no_improve += 1
            if no_improve > IMPROV_STREAK:
                # random shuffle small noise
                a = best_a.copy()
                a += np.random.normal(scale=0.001, size=n)

                max_idx = int(np.argmax(b))
                no_improve = 0
                print(f"[Hill_opt {it}] perturb after stagnation")
        it += 1
    return best_a

"""
Propose a non‑negative step‑function (a probability simplex vector) that
minimises  evaluate_sequence().
The optimiser works in two stages:

1.  **Adam on a smoothed max** – we minimise the log‑sum‑exp of the
    self‑convolution b = a⊛a (exactly 2·n·max(b) when the simplex
    constraint Σa=1 is active).  The smoothing parameter γ is increased
    from 0.1 to ~20 (continuation), and the Adam optimiser with a
    time‑decaying step size is applied.  After each update the vector is
    projected onto the probability simplex.

2.  **Greedy hill‑climbing with optimal line‑search** – starting from the
    best point found by Adam we repeatedly move mass from a source index
    that contributes most to the current worst convolution value to a
    target index that contributes the least.  For every source/target
    pair we compute the analytically optimal amount of mass to transfer
    (subject to a≥0) and accept the best improving move.  When no
    improvement is found for a while we perturb the current best with a
    tiny Gaussian noise and continue.

Both phases respect the 1000 s time budget, log intermediate scores
and finally return the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (len = 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())        # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << ((out_len - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # numerically degenerate – fall back to uniform vector
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase 1 – Adam on smooth max (log‑sum‑exp) of the convolution
# ----------------------------------------------------------------------
def _adam_phase(a: np.ndarray,
                n: int,
                deadline: float,
                best_a: np.ndarray,
                best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smoothed max objective."""
    start = time.time()
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    max_step0, min_step = 0.05, 5e-4   # step size schedule
    it = 1
    while time.time() < deadline:
        # ---------- convolution ----------
        b = _full_convolution(a)                 # length 2n‑1
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max              # exact evaluation (Σa = 1)

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam {it:7d}] new best = {best_val:.6f}")

        # ---------- continuation parameter γ ----------
        progress = (time.time() - start) / (deadline - start)
        gamma = 0.1 + 20.0 * progress               # from 0.1 up to ~20

        # ---------- log‑sum‑exp weights ----------
        b_max = b.max()
        w = np.exp(gamma * (b - b_max))            # shift for numerical safety
        w_sum = w.sum()
        w_norm = w / w_sum                         # probabilities, Σw_norm = 1

        # ---------- gradient of smoothed max ----------
        # grad_i = 2 * Σ_k w_norm_k * a_{k-i}
        a_rev = a[::-1]
        grad_full = _conv_fft(w_norm, a_rev)       # length 3n‑2
        grad = 2.0 * grad_full[n - 1:2 * n - 1]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = max_step0 * (1.0 - progress) + min_step * progress
        a_candidate = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- projection ----------
        a = _project_simplex(a_candidate)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Greedy hill‑climbing with optimal line‑search
# ----------------------------------------------------------------------
def _optimal_hillclimb(a: np.ndarray,
                       n: int,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> np.ndarray:
    """
    Local search that moves a small amount of mass from a source
    (large contribution to the current worst convolution entry) to a
    target (small contribution).  For every source/target pair the
    optimal amount of mass is computed analytically and the best
    improving move is applied.
    """
    # keep current convolution and statistics
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    # parameters
    K = 12                     # number of source/target candidates examined
    stagnation_limit = 150_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # recompute second‑largest value
        tmp = b.copy()
        tmp[max_idx] = -np.inf
        second_max = tmp.max()

        # ---- contributions of each coordinate to the worst entry ----
        contrib = np.zeros_like(a)
        low = max(0, max_idx - (n - 1))
        high = min(max_idx, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        contrib[idx] = 2.0 * a[max_idx - idx]   # derivative of b_max w.r.t a_i

        # choose promising sources / targets
        src_order = np.argsort(-contrib)       # descending
        tgt_order = np.argsort(contrib)        # ascending
        sources = src_order[:K]
        targets = tgt_order[:K]

        best_pair = None
        best_new_val = cur_val

        # examine all source/target pairs
        for s in sources:
            if a[s] <= 1e-14:
                continue
            for t in targets:
                if s == t:
                    continue

                # linear coefficient of b_max as a function of δ
                d = 0.0
                if 0 <= max_idx - s < n:
                    d -= 2.0 * a[max_idx - s]
                if 0 <= max_idx - t < n:
                    d += 2.0 * a[max_idx - t]

                if d >= 0.0:               # moving mass would not lower b_max
                    continue

                # ideal δ that would bring b_max down to second_max (linear approx)
                delta_target = (second_max - cur_max) / d   # positive number
                delta = min(a[s], max(0.0, delta_target))
                # limit step size a little to stay in the region where linear approx holds
                delta = min(delta, 0.5 * a[s])
                if delta <= 0.0:
                    continue

                # ----- incremental update of the whole convolution -----
                b_new = b.copy()

                # cross‑terms 2·δ·a
                b_new[t:t + n] += 2.0 * delta * a
                b_new[s:s + n] -= 2.0 * delta * a

                # pure δ² terms
                b_new[2 * t] += delta * delta
                b_new[2 * s] += delta * delta
                b_new[s + t] -= 2.0 * delta * delta

                new_max = b_new.max()
                new_val = 2.0 * n * new_max

                if new_val < best_new_val - 1e-13:
                    best_new_val = new_val
                    best_pair = (s, t, delta, b_new)

        if best_pair is not None:
            s, t, delta, b = best_pair
            a[s] -= delta
            a[t] += delta
            cur_max = b.max()
            cur_val = best_new_val
            max_idx = int(np.argmax(b))
            no_improve = 0
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            # ----- stagnation handling -----
            no_improve += 1
            if no_improve > stagnation_limit:
                # perturb around the current best solution
                a = best_a.copy()
                a += np.random.normal(scale=0.003, size=n)
                a = _project_simplex(a)
                b = _full_convolution(a)
                cur_max = b.max()
                cur_val = 2.0 * n * cur_max
                max_idx = int(np.argmax(b))
                no_improve = 0
                print(f"[Hill {it:7d}] stagnation – perturbing")
        it += 1
    return best_a
