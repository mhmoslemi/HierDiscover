"""
Simple yet powerful optimisation for the “step‑function” challenge.

The code follows three stages:

1. **Initialisation** –  a vector of length *n* (default 8000) is created.
   If a previous construction `height_sequence_1` exists it is interpolated;
   otherwise a random Dirichlet vector is used.  The vector is normalised to
   unit ℓ₁‑norm and encoded as soft‑max parameters ``w`` (guaranteeing
   non‑negativity and the sum constraint).

2. **Continuation L‑BFGS‑B on a log‑sum‑exp surrogate** –  The non‑smooth
   objective ``max(conv)`` is approximated by a smooth surrogate
   (max + ε·log‑sum‑exp).  The surrogate is minimised with L‑BFGS‑B while the
   smoothing parameter ε is decreased from 1e‑1 to 1e‑12 (log‑space).  A
   modest number of random restarts (with Gaussian perturbations of the
   soft‑max parameters) are performed to escape local minima.

3. **Aggressive mass‑transfer hill‑climbing** –  Starting from the best
   vector found, the exact objective is improved by repeatedly moving a
   fraction of mass away from the lag where the convolution attains its
   maximum and depositing it at low‑mass positions.  The step size is
   adaptively increased/decreased and the loop runs until the time budget
   expires.

The routine respects the provided time budget (default 1000 s) and prints
progress information.  The final list of heights is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full autocorrelation of a real vector (length 2·len‑1).
    Uses FFT for speed and clips tiny negative artefacts.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the smooth surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σₖ exp((convₖ - max)/ε)

    together with its gradient w.r.t. the soft‑max parameters ``w``.
    """
    # ---- soft‑max (feasible a, Σa = 1) ----
    w_shift = w - np.max(w)          # improve numerical stability
    a = np.exp(w_shift)
    a /= a.sum()

    # ---- full autocorrelation (length 2·n‑1) ----
    conv = _full_convolution(a)
    max_conv = conv.max()

    # ---- log‑sum‑exp smoothing term ----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)      # scalar
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a (closed form) ----
    # pₖ ∝ exp((convₖ - max)/ε)  .  Σₖ pₖ = 1
    weights = np.exp(shifted)
    p = weights / weights.sum()

    # g_i = 4·n·∑ₖ pₖ·a_{k‑i}
    # The convolution a * p_rev yields exactly the needed sums.
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)     # length 3·n‑2
    # Extract the middle n entries (indices n‑1 … 2n‑2)
    seg = conv_ap[n - 1 : 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # ---- chain rule through the soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Hill‑climbing mass‑transfer refinement
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray, best_val: float,
                              deadline: float,
                              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Reduce the exact objective by repeatedly moving a small amount of mass
    away from the lag where the convolution attains its maximum.
    The algorithm stops when the time budget expires.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.3                     # initial fraction of mass to relocate

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        # Choose one of the maximal lags (there may be several)
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # Identify all ordered pairs (i, j) with i + j = k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue

        top_n = min(5, contrib.size)          # a bit more than before
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]

        donors = [(int(i_range[idx]), int(j_range[idx])) for idx in top_idx]

        # Choose distinct destination indices from low‑mass region
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            # Amount of mass to move – a fraction of the smaller of the two
            if i == j:
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.85, 1e-5)
            continue

        # Project back onto the simplex (ℓ₁‑norm = 1)
        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.85, 1e-5)
            continue
        new_a /= s

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.05, 0.5)
        else:
            step_frac = max(step_frac * 0.93, 1e-5)

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise the step‑function heights.

    1. Initialise a (possibly interpolated) sequence of length *n*.
    2. Run a continuation L‑BFGS‑B on the log‑sum‑exp surrogate with a fine
       ε‑schedule and several random restarts.
    3. Finish with an aggressive hill‑climbing phase that directly reduces
       the true ``evaluate_sequence`` score.
    Returns the best list of floats found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – larger n gives a lower theoretical bound (≈1).
    # --------------------------------------------------------------
    n = 8000   # power‑of‑two size is convenient for the FFT

    # --------------------------------------------------------------
    # Initialise from the previous construction if available.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        a0 = rng.random(n)

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    w = np.log(a0 + 1e-12)                 # soft‑max parameters

    best_a = a0.copy()
    best_val = 2.0 * n * np.max(_full_convolution(best_a))
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # Continuation schedule for ε (coarse → fine)
    # --------------------------------------------------------------
    eps_schedule = np.logspace(-1, -12, num=45)   # 45 stages, more granularity

    # --------------------------------------------------------------
    # Primary optimisation pass (L‑BFGS‑B on the smooth surrogate)
    # --------------------------------------------------------------
    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def fun_and_grad(w_vec):
            return _smooth_obj_and_grad(w_vec, eps, n)

        res = minimize(fun_and_grad,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 1200,
                                'ftol': 1e-13,
                                'gtol': 1e-13,
                                'disp': False})
        w = res.x

        # evaluate exact objective
        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = 2.0 * n * np.max(_full_convolution(a))
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Random restarts – explore new basins
    # --------------------------------------------------------------
    restart = 0
    max_restarts = 20
    while time.time() < deadline - 5.0 and restart < max_restarts:
        restart += 1
        # perturb the current best soft‑max parameters
        w = np.log(best_a + 1e-12) + rng.normal(scale=0.8, size=n)

        # use only the finer tail of the ε‑schedule for speed
        for eps in eps_schedule[-18:]:
            if time.time() >= deadline:
                break

            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)

            res = minimize(fun_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 600,
                                    'ftol': 1e-13,
                                    'gtol': 1e-13,
                                    'disp': False})
            w = res.x

        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = 2.0 * n * np.max(_full_convolution(a))
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[restart {restart}] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final aggressive hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _mass_transfer_hill_climb(best_a, best_val,
                                                    deadline, rng)

    # --------------------------------------------------------------
    # Final report using the provided evaluator (protects against caps)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
