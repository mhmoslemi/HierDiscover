import time, numpy as np

def full_convolution(a):
   n = a.shape[0]
   out_len = 2*n-1
   size = 1 << (out_len-1).bit_length()
   fa = np.fft.rfft(a, size)
   conv = np.fft.irfft(fa * fa, size)[:out_len]
   return conv

def objective_of_a(a):
   n = a.shape[0]
   b = full_convolution(a)
   return float(2.0 * n * b.max())

def parametric_search(n, deadline):
   # search over gamma and delta
   x = (np.arange(n) + 0.5) / n
   best_a = None
   best_val = np.inf
   start = time.time()
   # grid search
   gamma_vals = np.linspace(0.0, 3.0, 31)
   delta_vals = np.linspace(0.0, 0.3, 7)
   for gamma in gamma_vals:
       for delta in delta_vals:
           now = time.time()
           if now >= deadline:
               return best_a, best_val
           denom = x * (1 - x) + delta + 1e-14
           a_raw = np.power(denom, -gamma)
           a = a_raw / a_raw.sum()
           val = objective_of_a(a)
           if val < best_val:
               best_val = val
               best_a = a.copy()
   # random sampling
   while True:
       now = time.time()
       if now >= deadline:
           break
       gamma = np.random.rand() * 3.0
       delta = np.random.rand() * 0.5
       denom = x * (1 - x) + delta + 1e-14
       a_raw = np.power(denom, -gamma)
       a = a_raw / a_raw.sum()
       val = objective_of_a(a)
       if val < best_val:
           best_val = val
           best_a = a.copy()
   return best_a, best_val

def hillclimb(a_start, deadline, target_val=1.503):
   a = a_start.copy()
   n = a.shape[0]
   b = full_convolution(a)
   cur_val = 2.0 * n * b.max()
   best_a = a.copy()
   best_val = cur_val
   # schedule variables
   start = time.time()
   max_frac_start, max_frac_end = 0.5, 0.01
   no_improve = 0
   IMPROVE_LIMIT = 200_000
   eps = 1e-12
   iteration = 0
   while True:
       now = time.time()
       if now >= deadline:
           break
       progress = (now - start) / (deadline - start)
       max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
       # locate current max index
       k_max = int(np.argmax(b))
       idx = k_max - np.arange(n)
       mask = (idx >= 0) & (idx < n)
       contrib = np.zeros_like(a)
       contrib[mask] = a[mask] * a[idx[mask]]
       total_contrib = contrib.sum()
       # source
       if total_contrib > 0:
           source_probs = contrib / total_contrib
           i = np.random.choice(n, p=source_probs)
       else:
           i = np.argmax(a)
       if a[i] <= eps:
           continue
       # target bias
       target_weights = np.zeros_like(a)
       target_weights[mask] = 1.0 / (contrib[mask] + eps)
       target_weights[i] = 0.0
       w_sum = target_weights.sum()
       if w_sum > 0:
           j = np.random.choice(n, p=target_weights / w_sum)
       else:
           # uniform random different index
           j = np.random.randint(n-1)
           if j >= i: j += 1
       delta = a[i] * np.random.rand() * max_frac
       if delta <= 0:
           continue
       if i == j:
           continue
       # incremental reflection
       b_new = b.copy()
       # update slices
       b_new[j:j+n] += 2.0 * delta * a
       b_new[i:i+n] -= 2.0 * delta * a
       # quadratic updates
       b_new[2*j] += delta * delta
       b_new[2*i] += delta * delta
       b_new[i+j] -= 2.0 * delta * delta
       new_max = b_new.max()
       new_val = 2.0 * n * new_max
       if new_val < cur_val - 1e-14:
           a[i] -= delta
           a[j] += delta
           b = b_new
           cur_val = new_val
           no_improve = 0
           if new_val < best_val - 1e-12:
               best_val = new_val
               best_a = a.copy()
               print(f"[HillClimb {iteration:7d}] new best = {best_val:.6f}")
               if best_val <= target_val:
                   break
       else:
           no_improve += 1
       if no_improve > IMPROVE_LIMIT:
           # random perturbation
           a = best_a.copy()
           noise = np.random.normal(scale=0.02, size=n)
           a = np.maximum(a + noise, 0.0)
           a /= a.sum()
           b = full_convolution(a)
           cur_val = 2.0 * n * b.max()
           no_improve = 0
           print(f"[HillClimb {iteration:7d}] stagnation – perturb")
       iteration += 1
   return best_a, best_val

"""
Two‑stage optimizer for the step‑function bound.

* Phase 0 – exhaustive grid + random sampling over a simple symmetric
  U‑shaped family a_i ∝ (x·(1‑x)+δ)^{‑γ}.  The two parameters (γ,δ) are
  tuned by direct evaluation of the exact objective (full convolution).

* Phase 1 – fast hill‑climbing that repeatedly moves a tiny amount of mass
  from an index that contributes strongly to the current maximal
  convolution entry to an index that contributes weakly.  The move is
  accepted only if the exact bound improves.  When progress stalls a
  small random perturbation of the best‑so‑far vector is applied.

The routine respects the 1000 s budget, prints progress, and finally
returns a list of non‑negative heights whose sum is 1.  The returned
sequence is passed to the provided ``evaluate_sequence`` for the final
score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using FFT.
    Length of the result is 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def objective(a: np.ndarray) -> float:
    """
    Returns the exact value of 2·n·max(conv(a,a)) for a vector a that
    already satisfies Σ a = 1.  This is the quantity that
    ``evaluate_sequence`` computes after normalisation.
    """
    n = a.shape[0]
    b = full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Phase 0 – parametric sweep over a simple U‑shaped family
# ----------------------------------------------------------------------
def parametric_search(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Search the two‑parameter family

        a_i ∝ (x_i·(1‑x_i) + δ)^{‑γ},    x_i = (i+½)/n

    for the best bound.  A dense grid is examined first, then a
    moderate number of random points until the time limit is reached.
    The returned pair (best_a, best_val) always satisfies Σ a = 1.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    best_a = None
    best_val = np.inf

    # ---- dense grid ----------------------------------------------------
    gamma_grid = np.linspace(0.0, 3.0, 31)   # 0 … 3 inclusive
    delta_grid = np.linspace(0.0, 0.30, 7)   # 0 … 0.30
    eps = 1e-14

    for gamma in gamma_grid:
        for delta in delta_grid:
            now = time.time()
            if now >= deadline:
                return best_a, best_val
            denom = x * (1.0 - x) + delta + eps
            # power -gamma works also for gamma == 0 (gives 1)
            a_raw = np.power(denom, -gamma)
            a = a_raw / a_raw.sum()
            val = objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()

    # ---- random sampling ------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline:
            break
        gamma = np.random.rand() * 3.0          # uniform in [0,3)
        delta = np.random.rand() * 0.5          # uniform in [0,0.5)
        denom = x * (1.0 - x) + delta + eps
        a_raw = np.power(denom, -gamma)
        a = a_raw / a_raw.sum()
        val = objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – targeted hill‑climbing
# ----------------------------------------------------------------------
def hillclimb(a_start: np.ndarray,
              deadline: float,
              target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Fast local search based on moving a small amount of mass from a
    “source” index (large contribution to the current maximal convolution
    entry) to a “target” index (small contribution).  Moves are accepted
    only when they strictly improve the exact bound.
    """
    a = a_start.copy()
    n = a.shape[0]
    b = full_convolution(a)
    cur_val = 2.0 * n * b.max()

    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    max_frac_start, max_frac_end = 0.50, 0.01   # schedule for transfer size
    IMPROVE_LIMIT = 180_000                    # stagnation threshold
    no_improve = 0
    eps = 1e-14
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear schedule for the maximal transfer fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate the current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each index to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()          # this equals b[k_max]

        # ----- choose source index (biased by contribution) -----
        if total_contrib > 0.0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ----- choose target index (prefer indices with *small* contribution) -----
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0  # never pick the source itself
        w_sum = target_weights.sum()
        if w_sum > 0.0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # fallback to a uniform random different index
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # ----- amount of mass to move -----
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue
        if i == j:
            continue

        # ----- incremental update of the full convolution (O(n)) -----
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ----- accept only improving moves -----
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val = new_val
                best_a = a.copy()
                print(f"[HillClimb {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ----- stagnation handling -----
        if no_improve > IMPROVE_LIMIT:
            # perturb the current best solution and restart from there
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillClimb {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

def propose_candidate(seed: int = 42,

    """
    Optimise a non‑negative step‑function under the 1000 s budget.

    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 8000 entries give a good balance between fidelity
    # and speed for the available time budget.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Try to reuse the previously discovered construction if it exists.
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # Resample to the required length by linear interpolation
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        best_a = a0
        best_val = objective(a0)

        a /= a.sum()

        print("[final] evaluate_sequence reports", final_score)
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
