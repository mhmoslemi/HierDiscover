def _softmax_phase(a, n, deadline, best_a, best_val):
    start = time.time()
    init_b = _full_convolution(a)
    init_max = init_b.max()
    init_val = 2 * n * init_max
    best_val = min(best_val, init_val)
    best_a = a.copy()
    # initial epsilon set as init_max * 0.5 maybe.
    epsilon_start = init_max * 0.5
    epsilon_end = init_max * 0.001   # final very small
    # Step schedule:
    step_start = 0.5
    step_end = 0.001
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Softmax {it:7d}] new best = {best_val:.6f}")

        # epsilon schedule linear
        progress = (time.time() - start) / (deadline - start)
        epsilon = epsilon_start * (1.0 - progress) + epsilon_end * progress
        # if epsilon very small, maybe use subgradient instead but okay.

        # compute weight softmax
        max_b = b.max()
        # use scaled exponents
        scaled = (b - max_b) / epsilon   # stable
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()

        # compute gradient: grad = 2 * conv_rev[n-1:n-1+n]
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, w)   # shape length 3n-2
        grad = 2.0 * conv_rev[n-1 : n-1 + n]

        # step schedule
        step = step_start * (1.0 - progress) + step_end * progress

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        it += 1
    return best_a, best_val

"""
Optimised search for a step‑function minimising `evaluate_sequence`.

Algorithm
---------
* Phase A – Sub‑gradient descent on the true objective `max(a*a)`.
  A decreasing step size and Euclidean projection onto the probability
  simplex keep the vector feasible.

* Phase B – L²‑flattening (as in the reference code) to reduce the
  variance of the autoconvolution.  This provides a good starting point
  for the next phase.

* Phase C – Soft‑max smoothing of the `max` functional.
  We minimise the smooth approximation
        ε·log(∑ exp(b/ε)),   b = a∗a,
  whose gradient is obtained analytically via a single FFT‑convolution.
  The temperature ε is reduced linearly, turning the smooth problem into
  the original one.

* Phase D – Enhanced greedy hill‑climbing.
  From the current worst convolution index we pick a few *source*
  indices (largest contribution) and *target* indices (smallest
  contribution).  For each source–target pair several trial mass
  transfers are evaluated (O(n) each) and the best improving move is
  performed.  After long stagnation a tiny Gaussian perturbation is
  added to escape local minima.

All updates are followed by projection onto the simplex {x≥0, Σx=1}.
The routine respects the 1000 s wall‑time limit, prints periodic
progress messages and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0.0] = 0.0
    w_sum = w.sum()
    if w_sum > 0.0:
        w /= w_sum
    else:                     # numerical safety
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase A – Sub‑gradient descent on the exact objective (max b)
# ----------------------------------------------------------------------
def _subgradient_phase(a: np.ndarray,
                       n: int,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    """Reduce `max(b)` with a sub‑gradient method."""
    start = time.time()
    max_step = 0.10
    min_step = 1e-4
    it = 0

    while time.time() < deadline:
        b = _full_convolution(a)
        k = int(np.argmax(b))
        cur_max = b[k]
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Subgrad {it:7d}] new best = {best_val:.6f}")

        # sub‑gradient of `max(b)` w.r.t. a
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                grad[mid] = 2.0 * a[mid]

        progress = (time.time() - start) / (deadline - start)
        step = max_step * (1.0 - progress) + min_step * progress

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – L²‑flattening (variance of b)
# ----------------------------------------------------------------------
def _l2_flatten_phase(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> tuple[np.ndarray, float]:
    """Minimise Σ(b−\bar b)², where b = a∗a."""
    start = time.time()
    base_step = 0.5
    min_step = 0.001
    avg_target = 1.0 / (2 * n - 1)                 # mean of b for Σa = 1
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[L2 {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        r = b - avg_target
        a_rev = a[::-1]
        conv1 = _conv_fft(r, a)          # r * a
        conv2 = _conv_fft(r, a_rev)      # r * a_rev
        start_idx = n - 1
        end_idx = 2 * n - 1
        grad = 2.0 * (conv1[start_idx:end_idx] + conv2[start_idx:end_idx])

        progress = (time.time() - start) / (deadline - start)
        step = base_step * (1.0 - progress) + min_step * progress

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        if no_improve > IMPROVE_LIMIT:
            a = np.random.rand(n)
            a = _project_simplex(a)
            no_improve = 0
            print(f"[L2 {it}] restart after stagnation")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Soft‑max smoothing of the L∞ functional
# ----------------------------------------------------------------------
def _softmax_phase(a: np.ndarray,
                   n: int,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Gradient descent on a smooth approximation of max(b)."""
    start = time.time()
    # step‑size schedule (larger than in phase A because the gradient is
    # usually smaller after the soft‑max weighting)
    step_start = 0.5
    step_end = 0.001
    it = 0

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Softmax {it:7d}] new best = {best_val:.6f}")

        # linear annealing of the temperature ε
        progress = (time.time() - start) / (deadline - start)
        eps = max(cur_max * 0.5 * (1.0 - progress) + cur_max * 0.001 * progress,
                  1e-12)               # avoid division by zero

        # soft‑max weights (stable formulation)
        scaled = (b - cur_max) / eps
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()

        # gradient of ε·log Σexp(b/ε)  w.r.t. a  =  2·conv_rev[n-1:n-1+n]
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, w)                # length 3n‑2
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        step = step_start * (1.0 - progress) + step_end * progress
        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – Enhanced greedy hill‑climbing
# ----------------------------------------------------------------------
def _enhanced_hillclimb(a: np.ndarray,
                        n: int,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float) -> np.ndarray:
    """
    Local search that evaluates several source/target/delta triples
    per iteration and keeps the move giving the biggest reduction of
    the exact objective 2·n·max(b).  Stagnation is broken by a tiny
    Gaussian perturbation.
    """
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    # hyper‑parameters
    src_cnt = 3          # number of high‑contribution sources examined
    tgt_cnt = 3          # number of low‑contribution targets examined
    delta_fracs = (0.2, 0.5, 0.8)   # trial fractions of the source mass

    while time.time() < deadline:
        # contribution of each index to the current worst convolution entry
        k = max_idx
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        # candidate source / target indices
        src_indices = np.argpartition(contrib, -src_cnt)[-src_cnt:]
        src_indices = src_indices[a[src_indices] > 1e-12]   # discard empty sources
        if src_indices.size == 0:
            src_indices = np.where(a > 1e-12)[0]

        tgt_indices = np.argpartition(contrib, tgt_cnt)[:tgt_cnt]

        best_local_val = cur_val
        best_move = None          # (src, tgt, delta, new_max, new_b)

        for src in src_indices:
            for tgt in tgt_indices:
                if src == tgt:
                    continue
                max_delta = a[src]
                for f in delta_fracs:
                    delta = max_delta * f
                    if delta <= 0.0:
                        continue
                    # incremental update of the full convolution
                    b_new = b.copy()
                    b_new[tgt:tgt + n] += 2.0 * delta * a
                    b_new[src:src + n] -= 2.0 * delta * a
                    b_new[2 * tgt] += delta * delta
                    b_new[2 * src] += delta * delta
                    b_new[src + tgt] -= 2.0 * delta * delta

                    new_max = b_new.max()
                    new_val = 2.0 * n * new_max
                    if new_val < best_local_val - 1e-12:
                        best_local_val = new_val
                        best_move = (src, tgt, delta, new_max, b_new)

        if best_move is not None:
            src, tgt, delta, new_max, b = best_move
            a[src] -= delta
            a[tgt] += delta
            cur_max = new_max
            cur_val = best_local_val
            max_idx = int(np.argmax(b))
            no_improve = 0
            best_a = a.copy()
            best_val = cur_val
            print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # escape a plateau
            a = best_a.copy()
            a += np.random.normal(scale=0.003, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")
        it += 1
    return best_a


# ----------------------------------------------------------------------
# Main entry point expected by the outer evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Return a list of non‑negative step heights that (hopefully) minimise
    `evaluate_sequence`.  The routine respects the 1000 s wall‑time limit,
    runs four optimisation phases, and finally prints the obtained score.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 8000 provides a good trade‑off between expressiveness
    # and speed.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a – try to reuse a previously discovered construction.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(globals().get('height_sequence_1'), dtype=float)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)                 # enforce Σa = 1
    best_a = a.copy()
    cur_b = _full_convolution(a)
    best_val = 2.0 * n * cur_b.max()

    # ------------------------------------------------------------------
    # Split the remaining budget into four phases.
    # ------------------------------------------------------------------
    total = deadline - start_time
    t_subgrad = start_time + 0.20 * total
    t_l2      = start_time + 0.50 * total
    t_softmax = start_time + 0.80 * total
    t_hill    = deadline

    # ----------------------------
    # Phase A – Sub‑gradient descent
    # ----------------------------
    best_a, best_val = _subgradient_phase(
        a, n, t_subgrad, best_a, best_val
    )
    a = best_a.copy()

    # ----------------------------
    # Phase B – L²‑flattening
    # ----------------------------
    best_a, best_val = _l2_flatten_phase(
        a, n, t_l2, best_a, best_val
    )
    a = best_a.copy()

    # ----------------------------
    # Phase C – Soft‑max smoothing
    # ----------------------------
    best_a, best_val = _softmax_phase(
        a, n, t_softmax, best_a, best_val
    )
    a = best_a.copy()

    # ----------------------------
    # Phase D – Enhanced greedy hill‑climbing
    # ----------------------------
    best_a = _enhanced_hillclimb(
        a, n, t_hill, best_a, best_val
    )

    # ------------------------------------------------------------------
    # Final evaluation (the secure wrapper may clamp values)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:                              # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
