def project_onto_simplex(v):
    """Projection of vector v onto probability simplex (sum=1, v>=0)."""
    # Using algorithm from Duchi et al. 2008.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / np.arange(1, n+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Works.

    """Projection onto the probability simplex."""
    # using algorithm: sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        # all zero perhaps
        return np.zeros_like(v)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time, numpy as np

def _next_pow2(x):
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _full_convolution(x):
    """Linear autocorrelation of a real non‑negative vector x (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative values caused by floating‑point noise
    np.maximum(conv, 0, out=conv)
    return conv

def _gradient_of_max_conv(a, k):
    """Sub‑gradient of max_{lag} (a*a) at a given lag k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.intp)
    j = k - i
    # add a_j to g[i] and a_i to g[j] (double count when i==j)
    g[i] += a[j]
    g[j] += a[i]
    return g

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Projected (exponentiated) gradient descent on the *max‑convolution*
    objective.  Starting from the previously found sequence (if available)
    the algorithm repeatedly chooses the lag with the largest convolution
    value, moves probability mass away from the indices participating in
    that lag and re‑normalises onto the simplex.  A few random restarts and
    an adaptive step size are used.  The routine respects the 1000 s time
    budget and returns the best non‑negative sequence it found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # problem size – 4000 gives a good trade‑off between quality and speed
    n = 4000

    # ------------------------------------------------------------------
    # initialise from a previous construction if possible
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            xs_old = np.linspace(0.0, 1.0, old.shape[0])
            xs_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(xs_new, xs_old, old)
        else:
            a0 = old.copy()
    except Exception:
        a0 = rng.random(n)

    a0 = np.maximum(a0, 0.0)
    # normalise onto the probability simplex (sum = 1)
    a0_sum = a0.sum()
    if a0_sum == 0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0_sum

    best_a = a0.copy()
    best_val = 2.0 * n * np.max(_full_convolution(best_a))
    print(f"[init] n={n}, objective={best_val:.6f}")

    # --------------------------------------------------------------
    # hyper‑parameters
    # --------------------------------------------------------------
    max_restarts = 8
    max_iter_per_restart = 6000   # will be limited by the time budget
    eta_initial = 0.12            # step size for exponentiated update
    eta_min = 1e-8
    eta_max = 1.0

    # --------------------------------------------------------------
    # main optimisation loop (exponentiated gradient descent)
    # --------------------------------------------------------------
    restart = 0
    while time.time() < deadline and restart < max_restarts:
        restart += 1

        if restart == 1:
            a = best_a.copy()          # continue from the best known point
        else:
            # a fresh random Dirichlet start (perturbed copy of best helps)
            a = rng.random(n)
            a = a / a.sum()

        cur_val = 2.0 * n * np.max(_full_convolution(a))
        cur_best_val = cur_val
        cur_best_a = a.copy()
        eta = eta_initial

        it = 0

            it += 1

            conv = _full_convolution(a)
            max_b = np.max(conv)
            # list of all lags attaining the maximum – pick one uniformly
            ks = np.flatnonzero(conv == max_b)
            k = int(rng.choice(ks))

            # gradient of the active convolution term
            grad = _gradient_of_max_conv(a, k)

            # exponentiated‑gradient update (always stays on the simplex)
            exp_factor = np.exp(-eta * grad)
            a_candidate = a * exp_factor
            s = a_candidate.sum()
            if s == 0:
                # numerical trouble – fall back to a tiny uniform distribution
                a_candidate = np.full_like(a, 1.0 / n)
            else:
                a_candidate = a_candidate / s

            new_val = 2.0 * n * np.max(_full_convolution(a_candidate))

            if new_val < cur_val:

                a = a_candidate
                cur_val = new_val
                eta = min(eta * 1.08, eta_max)

                if cur_val < cur_best_val:

                eta = max(eta * 0.55, eta_min)

"""
Project (exponentiated) gradient descent on the maximum‑convolution objective.

The algorithm works directly on the probability simplex (∑a_i = 1, a_i ≥ 0).
At each iteration it finds the lag k where (a*a)[k] is maximal, builds the
sub‑gradient of this max‑term, and performs an exponentiated‑gradient update

    a_i ← a_i · exp(‑η·g_i)   and then renormalises to sum‑one.

The step size η is increased after a successful move and decreased after a
rejection; a handful of random restarts (first from the previously best
construction, the others from a fresh Dirichlet draw) are performed until the
time budget expires.  The best sequence found is finally evaluated with the
provided ``evaluate_sequence`` function and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data) – linear autocorrelation a*a
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear autocorrelation of a real vector ``x``.
    Returns a vector of length 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Clip tiny negative rounding artefacts
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of the active convolution term
# ----------------------------------------------------------------------
def _gradient_of_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the term (a*a)[k] w.r.t. ``a``.
    For an index pair (i,j) with i+j=k the contribution is
          ∂/∂a_i (a_i a_j) = a_j  (and symmetrically for a_j).
    The returned vector has shape (len(a),) and contains the sum of
    these contributions for every i.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.intp)
    j = k - i

    # each pair (i,j) adds a[j] to g[i] and a[i] to g[j];
    # the case i==j adds 2·a[i] automatically.
    g[i] += a[j]
    g[j] += a[i]

    return g
