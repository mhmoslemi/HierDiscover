"""
propose_candidate
-----------------

Search for a non‑negative step‑function that minimizes the given
``evaluate_sequence`` objective.  The algorithm works in three phases:

1. **Smooth log‑sum‑exp optimisation** –  The objective
   ``2·n·max(conv)`` is approximated by a smooth function
   ``2·n·ε·log ∑exp(conv/ε)``.  We minimise it with a large number of
   L‑BFGS runs while slowly decreasing ε (from 1e‑1 down to 1e‑12).  The
   optimisation is performed in log‑space (soft‑max parametrisation) so
   the simplex constraints are automatically satisfied.

2. **Direct max‑lag mirror descent** –  Using the exact maximal lag of
   the self‑convolution we perform an exponential mirror‑descent step
   ``a ← a·exp(−α·g)`` where ``g_i = a_{k*−i}``.  The step‑size α decays
   very slowly, giving a fine‑grained reduction of the peak.

3. **Hill‑climbing refinement** –  Random swaps and small mass‑transfer
   moves are tried until the time budget expires; only strict
   improvements are kept.

Occasional anti‑symmetric re‑ordering (pairing large values with small
ones when the sequence is reversed) often flattens the convolution
further and is applied after each major phase.

The whole routine respects the 1000 s budget (a 5 s safety margin) and
returns the best sequence found as a list of Python floats.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)  # clip possible round‑off negatives
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic that pairs large values with small ones when the sequence
    is reversed – often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Random hill‑climb: swaps (20 %) and tiny mass transfers (80 %).
    Only strictly better solutions are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate  L = 2·n·ε·log ∑exp(conv/ε)  and its gradient.
    ``a`` is recovered via a soft‑max parametrisation
    a_i = exp(w_i) / Σ_j exp(w_j).
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)           # length 2·n‑1
    # smooth max via log‑sum‑exp
    L = 2.0 * n * eps * logsumexp(conv / eps)

    # gradient of L w.r.t. a
    # p_k = exp(conv_k/eps) / Σ exp(conv/eps)
    weights = np.exp(conv / eps)
    p = weights / weights.sum()

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # extract the central segment that corresponds to the derivative
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]          # factor 2 from dconv/d a and *2·n from L

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) attains a
    lower bound than the previously reported 1.512114 before the time
    budget runs out.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate length gives a good trade‑off
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Initialise from a previously discovered construction if available
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)   # uniform start

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS on the smooth surrogate (log‑sum‑exp)
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(-1, -12, num=15)   # 0.1 → 1e‑12
    w = np.log(best_a + 1e-12)                 # soft‑max parameters

    restart = 0
    global _GLOBAL_EPS, _GLOBAL_N
    while time.time() < deadline - 30.0 and restart < 12:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")

        # occasional random re‑initialisation to escape possible traps
        if restart % 4 == 0:
            w = np.log(best_a + 1e-12) + 0.1 * np.random.randn(n)

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 1000,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # convert back to probabilities on the simplex
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()

            cur_val = evaluate_sequence(a_candidate.tolist())
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # occasional anti‑symmetric reordering
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – direct max‑lag mirror‑descent (subgradient) refinement
    # ------------------------------------------------------------------
    a = best_a.copy()
    alpha0 = 0.15          # starting step size
    it = 0
    while time.time() < deadline - 10.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())

        # compute subgradient for the current maximal lag
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g = np.zeros_like(a)
        g[idx] = a[jdx]               # g_i = a_{k*−i}

        # step‑size schedule (very slow decay)
        alpha = alpha0 * (0.995 ** (it // 1000))
        a = a * np.exp(-alpha * g)

        # project back onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
            s = 1.0
        a /= s

        if it % 100 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[grad iter {it}] new best = {best_val:.6f}")
            # occasional re‑ordering
            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[grad reorder] improved to {best_val:.6f}")
        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – final stochastic hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final check and return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")

    return best_a.tolist()
