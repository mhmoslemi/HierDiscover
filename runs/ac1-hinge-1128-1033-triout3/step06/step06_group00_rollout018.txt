        # a must be numpy array nonneg, size n

        # However we need to compute maxconv after scaling? Since scaling cancels, we can keep sum=1 to reduce numeric issues.
        s = a.sum()
        if s == 0: return np.inf
        a_norm = a / s
        conv = _full_convolution_fft(a_norm, a_norm)
        return 2.0 * n * conv.max()
    
    # start with warm list of base shapes
    base_shapes = [ ... create few deterministic ones ...]
    for shape in base_shapes:
        a = shape.copy()
        a = np.maximum(0, a)
        a = a / a.sum()
        val = compute_obj(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

        iter += 1
        # randomly choose type
        typ = rng.integers(0, 5)
        if typ ==0:  # power law
            p = rng.uniform(0.0, 0.9)  # exponent
            delta = rng.uniform(0.0, 10.0) + 1e-6
            a = (np.arange(n, dtype=np.float64) + delta) ** (-p)
        elif typ ==1:  # exponential

            a = np.exp(-lam * np.arange(n, dtype=np.float64))
        elif typ ==2:  # mixture exponentials

            lam2 = rng.uniform(0.0, 5e-3)
            w = rng.random()
            a = w * np.exp(-lam1 * np.arange(n)) + (1-w) * np.exp(-lam2 * np.arange(n))
        elif typ ==3:  # piecewise constant with K segments
            K = rng.integers(3, 12)
            # generate random segment lengths that sum to n
            cuts = np.sort(rng.integers(1, n, size=K-1))
            seg_lengths = np.diff(np.concatenate(([0], cuts, [n])))
            seg_values = rng.random(K)
            seg_values = np.maximum(seg_values, 1e-8)
            a = np.repeat(seg_values, seg_lengths)
        elif typ ==4:  # random Dirichlet shape, optionally sorted decreasing
            raw = rng.random(n)
            a = raw
            if rng.random()<0.5:
                a = np.sort(a)[::-1]  # enforce decreasing
        else: # random convex mix of base shapes
            weights = rng.random(len(base_shapes))
            weights /= weights.sum()
            a = np.zeros(n)
            for w, s in zip(weights, base_shapes):
                a += w * s
        # ensure nonnegative
        a = np.maximum(0, a)
        # normalize to sum=1 to avoid scaling issues
        total = a.sum()
        if total < 1e-12:

        a = a / total

        # subgradient refinement steps (optional)

            max_idx = int(conv.argmax())

            start_i = max(0, max_idx - (n - 1))
            end_i = min(n - 1, max_idx)
            # vectorized fill

            j_vals = max_idx - i_vals
            g[i_vals] = 2.0 * a[j_vals]
            # step size
            lr = 0.2 / (1.0 + 0.01*_ ) # maybe degrade
            a_new = a - lr * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            a = a_new

        val = compute_obj(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[iteration {iter}] new best {best_val:.12f}")

        # occasional global perturbation
        if iter % 2000 == 0:
            # try perturbation of best_seq to explore
            cur = best_seq.copy()
            # random mass transfer
            i, j = rng.choice(n, size=2, replace=False)
            delta_transfer = min(cur[i], 0.01)
            cur[i] -= delta_transfer
            cur[j] += delta_transfer
            cur = cur / cur.sum()
            # evaluate
            val2 = compute_obj(cur)
            if val2 < best_val:
                best_val = val2
                best_seq = cur.copy()
                print(f"[perturb] better {best_val}")

    # return final sequence (scale to [0,1000] just clip)
    final = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]
    try:
        final_score = evaluate_sequence(final)
        print(f"[result] final_score={final_score:.12f}")
    except Exception:
        print("[result] evaluate_sequence not available")
    return final

"""
Propose a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages under the given time budget.

1. **Warm‑start** – reuse a previously discovered sequence if it exists,
   otherwise create a simple decreasing power‑law shape and normalise it.

2. **Projected sub‑gradient descent** – the objective is
   f(a)=max_k (a*a)[k]; f is convex.  At each iteration we compute the
   convolution, pick the active shift(s) that attain the maximum and form
   the sub‑gradient g_i = 2·a_{k-i}.  A decreasing step size and a simple
   momentum term are used.  After the gradient step we project back onto the
   probability simplex (non‑negative, sum = 1).  This stage consumes most of
   the budget and usually improves the score well below the uniform
   benchmark (score = 2.0).

3. **Fine‑grained simulated annealing** – when the sub‑gradient phase is
   exhausted we perform a cheap “mass‑transfer” Metropolis search.
   Randomly move a tiny amount of mass from one index to another; the
   move is accepted according to the current temperature.  The temperature
   decays linearly and occasional global jitter helps escape plateaus.

The best sequence seen throughout the run is returned (clipped to the
[0, 1000] range required by `evaluate_sequence`).  The whole procedure
fits easily into the 1000 s time limit on a single CPU core.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for n ≈ 6000).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Evaluate the (scale‑invariant) objective for a normalised vector.
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a, a)) assuming sum(a)==1."""
    n = a.size
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution objective.
# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray):
    """Return sub‑gradient g and the current max‑convolution value."""
    n = a.size
    conv = _full_convolution_fft(a, a)
    max_val = conv.max()
    # indices that are (almost) maximal – using a tiny tolerance
    active = np.flatnonzero(conv >= max_val - 1e-14)
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size               # average over active set
    else:                                            # should not happen
        w[np.argmax(conv)] = 1.0
    # rev_a is a reversed copy; convolution with the sparse selector w
    rev_a = a[::-1]
    g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]
    return g, max_val


# ----------------------------------------------------------------------
# Main entry point required by the grader.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a non‑negative height sequence that (approximately) minimizes
    the evaluation function.  The routine respects a `budget_s` second time
    limit and prints intermediate scores.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 5.0       # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem dimension – 6000 works well for the time budget.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Warm start: reuse a previously discovered sequence if possible,
    # otherwise create a simple decreasing power‑law shape.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if base.shape[0] != n:
                if base.shape[0] > n:
                    base = base[:n]
                else:
                    base = np.pad(base, (0, n - base.shape[0]))
            a = np.maximum(0.0, base)
            if a.sum() > 0:
                a = a / a.sum()
            else:
                a = None
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        # a decreasing power‑law with a random exponent and shift
        p = rng.uniform(0.0, 0.45)          # exponent (keeps us in the safe zone)
        shift = rng.uniform(0.5, 3.0)       # avoid a huge spike at index 0
        idx = np.arange(n, dtype=np.float64)
        a = (idx + shift) ** (-p)
        a = np.maximum(a, 0.0)
        a = a / a.sum()

    # ------------------------------------------------------------------
    # Track the best solution seen so far.
    # ------------------------------------------------------------------
    best_val = _objective(a)
    best_seq = a.copy()
    print(f"[init] n={n} score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent (dominant part of budget).
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.70 * budget_s          # ~70 % of the time
    eta = 0.8                                          # initial step size
    eta_min = 1e-5
    eta_decay = 0.9999
    momentum = 0.85
    vel = np.zeros_like(a)

    it = 0
    stagn = 0
    while time.time() < phase1_end:
        it += 1

        g, cur_max = _subgradient(a)

        # momentum update
        vel = momentum * vel + (1.0 - momentum) * g
        a_new = a - eta * vel

        # protect against negativity and re‑normalise
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() == 0:
            a_new = np.ones_like(a_new) / n
        else:
            a_new = a_new / a_new.sum()

        a = a_new

        # simple projection onto the simplex (already normalised, but
        # guarantees numerical stability)
        a = _project_simplex(a)

        # optional monotone‑decreasing projection – often helpful
        if it % 5000 == 0:
            a = np.sort(a)[::-1]          # enforce non‑increasing order
            a = a / a.sum()

        # evaluate current objective
        score = _objective(a)

        if score < best_val - 1e-12:
            best_val = score
            best_seq = a.copy()
            stagn = 0
            print(f"[sgd iter {it:6d}] new_best={best_val:.12f}  eta={eta:.2e}")
        else:
            stagn += 1

        # schedules
        eta = max(eta_min, eta * eta_decay)

        # shake out of plateaus
        if stagn > 3000:
            noise = rng.normal(scale=0.004, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            eta = max(eta_min, eta * 1.3)
            stagn = 0
            print(f"[sgd perturb] iter {it}")

    # ------------------------------------------------------------------
    # Phase 2 – Simulated‑annealing fine‑tuning.
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = _objective(cur)
    if cur_score < best_val:
        best_val = cur_score
        best_seq = cur.copy()

    # linear temperature schedule
    T0 = 5e-4
    Tmin = 1e-7
    it_sa = 0
    while time.time() < deadline:
        it_sa += 1
        # linear cooling
        T = max(Tmin, T0 * (1.0 - it_sa / 250_000))

        # mass‑transfer move
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] <= 0.0:
            continue
        max_transfer = min(cur[i], 0.008)                # keep moves tiny
        delta = rng.uniform(0.0, max_transfer)
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_score = _objective(new)

        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            cur = new
            cur_score = new_score
            if cur_score < best_val - 1e-12:
                best_val = cur_score
                best_seq = cur.copy()
                print(f"[SA iter {it_sa:6d}] new_best={best_val:.12f}")

        # occasional global jitter
        if it_sa % 20000 == 0:
            cur = cur + rng.normal(scale=0.0008, size=n)
            cur = np.maximum(cur, 0.0)
            cur = cur / cur.sum()
            cur_score = _objective(cur)

    # ------------------------------------------------------------------
    # Return the best sequence in the format required by evaluate_sequence.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # Optional: report the official score (if evaluate_sequence is available).
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] evaluate_sequence not available")

    return final_seq
