    g0 = 0.5

        # compute conv
        b = _full_convolution(a)    # length 2n-1
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val:

def _subgradient_phase_adam(a, n, deadline, best_a, best_val):
    t = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    start = time.time()
    total = deadline - start
    while time.time() < deadline:
        t+=1
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2 * n * max_b
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[subgradAdam {t}] new best {best_val:.6f}")
        # find worst index
        k = np.argmax(b)
        # subgradient
        grad = np.zeros_like(a)
        i_low = max(0, k - (n-1))
        i_high = min(k, n-1)
        idx = np.arange(i_low, i_high+1)
        grad[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                grad[mid] = 2.0 * a[mid]
        # Adam update
        prog = (time.time() - start) / total
        lr = 0.05 * (1 - prog) + 0.001 * prog
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a_candidate, 0)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.random.rand(n)
            a = a / a.sum()
        a = (a + a[::-1]) * 0.5
    return best_a, best_val

"""
Propose a step‑function with a low evaluation score.

Algorithm
---------
1. **Initialisation** – interpolate the supplied `height_sequence_1` (if
   available) to a fixed length `n = 8000` and project it onto the
   probability simplex (Σa = 1).  The vector is kept symmetric
   (a[i] = a[n‑1‑i]) because the objective is invariant under reversal.

2. **Phase A – Smoothed max descent**  
   Minimise a smooth approximation of the ∞‑norm:
   f(a) = (2·n / γ)·log ∑ₖ exp(γ·(a∗a)[k]).
   The gradient of the weighted sum `∑ wₖ·(a∗a)[k]` is a single
   correlation (`corr(w,a)`).  An Adam optimizer with a linearly
   increasing smoothing parameter `γ` (0.5 → 50) drives the iterate
   towards a flat autoconvolution.

3. **Phase B – Exact sub‑gradient descent**  
   Using the index of the current worst convolution value we
   compute the true sub‑gradient of `max(a∗a)`.  Adam with a decaying
   learning rate refines the solution while staying on the simplex.

4. **Phase C – Greedy hill‑climbing**  
   A deterministic local search moves a small amount of mass from the
   index that contributes most to the current peak to the one that
   contributes the least.  The full convolution is updated in O(n)
   time.  After long stagnation a tiny Gaussian perturbation is added
   to escape plateaux.

All phases keep the vector non‑negative and re‑normalise it after each
update.  The best vector seen during the run is finally evaluated with
the provided `evaluate_sequence` and returned as a plain Python list.
The routine respects the 1000 s wall‑time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·n‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    s = w.sum()
    if s > 0:
        w /= s
    else:                       # numerical safety
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase A – Smoothed max descent (log‑sum‑exp surrogate) with Adam
# ----------------------------------------------------------------------
def _smooth_phase_adam(a: np.ndarray,
                       n: int,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    """Minimise a smooth approximation of max(a∗a) using Adam."""
    start = time.time()
    total = deadline - start
    # Adam parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        # convolution a∗a
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth {it:7d}] new best = {best_val:.6f}")

        # smoothing parameter γ (linear schedule)
        progress = (time.time() - start) / total
        gamma = 0.5 + 49.5 * progress          # 0.5 → 50

        # soft‑max weights for the log‑sum‑exp
        w = np.exp(gamma * (b - cur_max))
        w_sum = w.sum()
        if w_sum == 0:                         # safety – should not happen
            w = np.ones_like(w) / w.shape[0]
            w_sum = 1.0
        w /= w_sum

        # gradient = 2·corr(w, a)  (correlation = conv(w, a_rev))
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev)        # length 3·n‑2
        grad = 2.0 * grad_full[n - 1:2 * n - 1]

        # Adam learning‑rate schedule (0.1 → 0.001)
        lr = 0.10 * (1.0 - progress) + 0.001 * progress

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # simplex projection and symmetrisation
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s > 0:
            a_candidate /= s
        else:
            a_candidate = np.random.rand(n)
            a_candidate /= a_candidate.sum()
        a = (a_candidate + a_candidate[::-1]) * 0.5

    return best_a, best_val

# ----------------------------------------------------------------------
def _subgrad_phase_adam(a: np.ndarray,
                       n: int,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:

    start = time.time()
    total = deadline - start
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Subgrad {it:7d}] new best = {best_val:.6f}")

        # index of the current peak
        k = int(np.argmax(b))

        # sub‑gradient of max(b) at index k
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                grad[mid] = 2.0 * a[mid]

        # Adam parameters (learning‑rate decays 0.05 → 0.0005)
        progress = (time.time() - start) / total
        lr = 0.05 * (1.0 - progress) + 0.0005 * progress

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s > 0:
            a_candidate /= s
        else:
            a_candidate = np.random.rand(n)
            a_candidate /= a_candidate.sum()
        a = (a_candidate + a_candidate[::-1]) * 0.5

    return best_a, best_val
