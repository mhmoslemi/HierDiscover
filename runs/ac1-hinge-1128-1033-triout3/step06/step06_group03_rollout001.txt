"""
Propose a step‑function (non‑negative heights) that approximately minimizes
``evaluate_sequence``.

Algorithm
========
1. Initialise a probability vector `a` (sum = 1).  If a previous best
   ``height_sequence_1`` is available it is interpolated to the working size,
   otherwise a uniform start is used.
2. Perform many iterations (until the time budget is exhausted) of a
   smoothed‑max gradient descent:
   *   Compute the full autocorrelation `c = a * a` (FFT based).
   *   Let `β` control the smoothness of the max‑operator.
       We form weights `w_k ∝ exp(β·(c_k‑max(c)))`.  When `β` is small the
       weights are almost uniform, giving a global direction; as `β` grows
       they concentrate on the true maximal lag.
   *   The gradient of the smoothed objective is
       `g_i = 2 Σ_k w_k a_{k‑i}`.
       This convolution is evaluated efficiently by a single FFT:
       `g = 2 * (rev(a)  ⊛  w)[n‑1 : n‑1+n]`.
   *   Take a projected gradient step `a ← Π_S (a – η·g)` where `Π_S`
       projects onto the probability simplex (non‑negative entries that sum
       to one).  The step size `η` decays like `η₀/√t`.
   *   Keep the best vector seen so far.  After a long streak without
       improvement we inject a small random perturbation to escape plateaus.
   *   `β` is increased slowly (every few hundred iterations) so the method
       transitions from a smooth landscape to the true `max` objective.
3. After the gradient phase a short guided hill‑climbing phase moves a
   small amount of mass from the pair of indices that contribute the most
   to the current maximal convolution value to two low‑mass indices.
   This is the same “mass‑transfer” refinement used in the previous
   state‑of‑the‑art code, but it now starts from a potentially better
   solution.
4. The final evaluation is obtained by calling the provided
   ``evaluate_sequence`` function and the resulting list of floats is
   returned.

The whole routine respects the 1000 s budget (a small safety margin is
kept).  By using the smoothed max gradient we obtain a smoother landscape,
allowing the optimiser to escape local plateaux that limited the earlier
sub‑gradient method.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector: (x * x)  length = 2·len‑1."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local refinement: repeatedly move a small fraction of mass away from the
    pair of indices that contributes most to the current maximal convolution.
    Stops when the time budget expires.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        # all lags achieving (or almost achieving) the maximum
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        if ks.size == 0:
            break
        # pick the lag that yields the largest individual product a_i*a_j
        best_pair_val = -1.0
        i_max = j_max = -1
        k_chosen = -1
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            if i_start > i_end:
                continue
            i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
            j_range = k - i_range
            prod = cur_a[i_range] * cur_a[j_range]
            idx = int(np.argmax(prod))
            if prod[idx] > best_pair_val:
                best_pair_val = float(prod[idx])
                i_max = int(i_range[idx])
                j_max = int(j_range[idx])
                k_chosen = int(k)

        if i_max == -1:
            # nothing to improve
            break

        # choose two destination indices that are relatively light
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        # avoid recreating the same lag
        attempts = 0
        while (p + q == k_chosen) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # mass transfer amount
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        # evaluate the exact objective
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val
