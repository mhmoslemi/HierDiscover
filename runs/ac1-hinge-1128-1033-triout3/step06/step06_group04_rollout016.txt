Let's understand their algorithm in detail:

- Initially pick f (nonnegative, sum sqrt(2n)). Compute M = max_{k} (f * f)(k).

- Solve LP:

    maximize sum b_j subject to b_j >= 0, and (f * g)(k) <= M for all k, where g is variable. Since f is fixed, the convolution constraints are linear with respect to g (since convolution is sum_i f_i g_{k-i}, linear in g). So we are maximizing total weight of g while not exceeding M.

- Let g0 be the solution. Then rescale g0 to have sum sqrt(2n). Then update f = (1-t) f + t g (convex combination) for small t.

The function of this is to move mass from positions where f * f attains high values to where it doesn't, while staying within M bound. This reduces M over iterations.

def proj_simplex(v):
    # v is array of floats. returns projection onto probability simplex (sum=1, nonneg).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def random_annealing(a, best_val, deadline, rng):
    n = len(a)
    cur_a = a.copy()
    cur_val = best_val
    temperature = 0.01
    while time.time() < deadline - 0.1:
        # propose random move
        i, j = rng.choice(n, size=2, replace=False)
        delta = min(cur_a[i], 0.1) * rng.random()  # random fraction of mass at i
        if delta <= 0: continue
        cur_a[i] -= delta
        cur_a[j] += delta
        # compute new maxconv quickly
        conv = _full_convolution(cur_a)
        new_val = 2 * n * conv.max()
        if new_val < cur_val or rng.random() < np.exp((cur_val - new_val)/temperature):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
                print(f"[anneal] new best {best_val:.6f}")
            # keep change
        else:
            # revert
            cur_a[i] += delta
            cur_a[j] -= delta
        # cool
        temperature *= 0.99995
    return best_a, best_val
```

"""
A mixed optimisation for the step‑function problem.

The algorithm works in three phases:

1. **Projected sub‑gradient descent** on the exact objective
   `2·n·max(conv)`.  The sub‑gradient of the active convolution lag
   is `g_i = 2·a_{k-i}` (zero outside the support).  After each step the
   vector is projected onto the probability simplex (non‑negative,
   ℓ₁‑norm = 1).

2. **Linear‑programming refinement**.  Given a current distribution `f`,
   we solve the LP

        maximise   Σ g_j
        subject to  (f * g)_k ≤ M   for all k,
                    g_j ≥ 0,

   where `M = max(f * f)`.  The LP tries to add as much mass as possible
   without increasing the worst convolution value.  The solution `g`
   (renormalised to unit ℓ₁‑norm) is mixed with `f` (convex combination)
   and the process is repeated.

3. **Aggressive hill‑climbing** – the same mass‑transfer routine that
   was used previously, now seeded with the best vector found so far.

All phases respect the global time budget (default 1000 s) and keep the
best sequence encountered.  The final list of heights is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full autocorrelation of a real vector (length 2·len‑1).
    Uses FFT for speed and clips tiny negative artefacts.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (ℓ₁‑norm = 1, non‑negative)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform zero vector
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Phase‑1: sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator):
    """
    Projected sub‑gradient descent on 2·n·max(conv) with simplex projection.
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(best_a).max()
    it = 0

    while time.time() < deadline - 0.1 and it < 5000:
        it += 1
        conv = _full_convolution(a)
        M = conv.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[sgd] iter {it:4d} new best = {best_val:.6f}")

        # choose one of the maximal lags (random tie‑break)
        ks = np.flatnonzero(conv == M)
        k = int(rng.choice(ks))

        # sub‑gradient of the active lag
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # diminishing step size
        step = 0.5 / np.sqrt(it)
        a_new = a - step * g

        # projection onto simplex (keeps sum = 1, non‑negative)
        a = _proj_simplex(a_new)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑2: linear‑programming refinement
# ----------------------------------------------------------------------
def _lp_refinement(a0: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator):
    """
    Repeatedly solves the LP

        maximise Σ g_j
        s.t.   (a * g)_k ≤ M   ∀k,
               g ≥ 0,

    where M = max(a * a).  The feasible g is renormalised to unit sum
    and mixed with a (weight 0.5).  The best vector encountered is returned.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(best_a).max()
    iteration = 0

    while time.time() < deadline - 0.5 and iteration < 30:
        iteration += 1
        conv_a = _full_convolution(a)
        M = conv_a.max()

        # Build the constraint matrix A where
        #   A[k, j] = a_{k-j}   (zero outside the support)
        diag_vals = [np.full(n, a[i]) for i in range(n)]   # each diagonal length = n
        A = diags(diag_vals,
                  offsets=np.arange(n),
                  shape=(2 * n - 1, n),
                  format='csr')

        # Linear program: maximise sum(g)  ==> minimise -sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b = np.full(2 * n - 1, M)

        res = linprog(c,
                      A_ub=A,
                      b_ub=b,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': int(deadline - time.time())})

        if not res.success:
            # LP failed – fall back to a tiny random perturbation and continue
            print(f"[lp] iteration {iteration} failed ({res.message}) – perturbing")
            a += 0.001 * rng.random(n)
            a = _proj_simplex(a)
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # Normalise g to unit ℓ₁‑norm and mix with the current vector
        g_norm = g / sum_g
        t = 0.5                     # convex‑combination weight
        a = (1 - t) * a + t * g_norm

        cur_val = 2.0 * n * _full_convolution(a).max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[lp] iter {iteration:2d} new best = {best_val:.6f}")

        # occasional random jitter to escape possible stagnation
        if iteration % 5 == 0:
            noise = rng.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0)
            a = _proj_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑3: aggressive hill‑climbing (mass‑transfer) – same as before
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray,
                              best_val: float,
                              deadline: float,
                              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly relocate a fraction of the mass participating in the
    currently maximal convolution lag to low‑mass positions.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.2                     # initial fraction of mass to move

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue

        top_n = min(5, contrib.size)          # move a few of the largest contributors
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]

        donors = []
        for idx in top_idx:
            i = int(i_range[idx])
            j = int(j_range[idx])
            donors.append((i, j))

        # choose low‑mass destinations
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue

        # renormalise to keep ℓ₁‑norm = 1 (scale‑invariant)
        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue
        new_a /= s

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.02, 0.5)
        else:
            step_frac = max(step_frac * 0.95, 1e-5)

    return cur_a, cur_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimises the step‑function heights.
    Returns the best list of floats found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate size balances expressiveness and speed.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Initialise – reuse a previously found construction if available.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        a0 = rng.random(n)

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    best_a = a0.copy()
    best_val = 2.0 * n * np.max(_full_convolution(best_a))
    print(f"[init] n={n}, initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1: projected sub‑gradient descent
    # ------------------------------------------------------------------
    sub_deadline = min(deadline, start_time + 0.30 * budget_s)
    a_sub, val_sub = _subgradient_descent(best_a, sub_deadline, rng)
    if val_sub < best_val:
        best_val = val_sub
        best_a = a_sub.copy()
        print(f"[phase1] best after sub‑gradient = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2: LP refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        lp_deadline = deadline - 5.0
        a_lp, val_lp = _lp_refinement(best_a, lp_deadline, rng)
        if val_lp < best_val:
            best_val = val_lp
            best_a = a_lp.copy()
            print(f"[phase2] best after LP = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3: aggressive hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        best_a, best_val = _mass_transfer_hill_climb(best_a, best_val,
                                                    deadline, rng)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided evaluate_sequence for safety)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = float('inf')

    return best_a.tolist()
