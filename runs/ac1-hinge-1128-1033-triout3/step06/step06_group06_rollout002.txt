"""
Greedy hill‑climber for the max‑convolution objective.

The problem is:
    minimise  2·n·max(conv(a,a)) / (sum(a)^2)
    subject to a_i ≥ 0  (the scaling of a does not matter, we keep Σa = 1).

The algorithm used here:

1.   A cheap parametric sweep (uniform, linear, quadratic, several exponentials,
     and a few beta‑shaped profiles) to obtain a decent starting vector.
2.   A deterministic greedy local search:
     * At each iteration locate the convolution entry k* where the current
       maximum occurs.
     * The contribution of the i‑th component to this entry is
           c_i = a_i * a_{k*‑i}   (zero when the index is out of range).
     * Move a (random) fraction of the mass from the index with the largest
       contribution to an index whose “partner” a_{k*‑j} is the smallest.
       The fraction starts relatively large (≈30 %) and shrinks
       exponentially to ≈0.1 % as the time budget is spent.
     * The convolution is updated incrementally (O(n)) and the move is kept
       only if the objective strictly improves.
3.   Whenever the search stalls for many iterations a small random
   perturbation of the best solution seen so far is applied in order to
   escape plateaus.
4.   The whole search respects the overall time budget (default 1000 s).
   The final vector is returned as a plain list; `evaluate_sequence`
   will clip values to [0,1000] and check the sum, but the algorithm keeps
   the sum exactly 1, so the objective is scale‑invariant.

The method is deliberately different from the previously supplied
Adam‑phase + random hill‑climber, aiming for a more directed descent
towards lower maximal self‑convolution values.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT‑based full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()       # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (sum(a) is assumed to be 1, otherwise the scaling is
# taken into account explicitly)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    b = _full_convolution(a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:               # same guard used inside evaluate_sequence
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean projection)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = 1}."""
    # Implementation of the algorithm from
    # "Efficient Projections onto the L1-Ball for Learning in High Dimensions"
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric initialisation – a small search over simple families
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising starting vector and its objective value."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective(best_a)

    # linear decreasing
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    val = _objective(a)
    if val < best_val:
        best_a, best_val = a, val

    # quadratic decreasing
    a = (np.arange(n, 0, -1, dtype=float) ** 2)
    a /= a.sum()
    val = _objective(a)
    if val < best_val:
        best_a, best_val = a, val

    # exponential families
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.995, 0.99, 0.98, 0.95, 0.9):
        a = r ** np.arange(n, dtype=float)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a, val

    # beta‑like shapes a_i ∝ (i+1)^p (n-i)^q
    exponents = [(1, 1), (2, 1), (1, 2), (3, 1), (1, 3), (2, 2)]
    for p, q in exponents:
        i = np.arange(n, dtype=float) + 1.0
        a = (i ** p) * ((n - i + 1) ** q)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a, val

    # truncated normal centred near the left edge
    x = np.arange(n, dtype=float)
    sigma = n / 5.0
    a = np.exp(-0.5 * ((x - 0) / sigma) ** 2)
    a /= a.sum()
    val = _objective(a)
    if val < best_val:
        best_a, best_val = a, val

    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy deterministic hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _greedy_hillclimb(a_start: np.ndarray,
                      start: float,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float,
                      target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of mass from a component that most
    contributes to the current maximal convolution entry to a component that
    contributes little.  The move is accepted only if the objective improves.
    """
    a = a_start.copy()
    n = a.shape[0]

    # current convolution and objective
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    # schedule for the maximal fraction of mass that may be moved
    max_frac_start, max_frac_end = 0.30, 0.001

    IMPROVE_LIMIT = 50000       # after this many idle steps we shake the solution
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ------------------------------------------------------------------
        # locate the dominant convolution entry and compute contributions
        # ------------------------------------------------------------------
        k_max = int(np.argmax(b))

        # partner[j] = a_{k_max - j} when the index is valid, else 0
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        partner = np.zeros_like(a)
        partner[mask] = a[idx[mask]]

        # contribution of each a_j to the maximal entry
        contrib = a * partner
        i = int(np.argmax(contrib))          # source index (largest contribution)

        if a[i] <= 1e-15:                     # source already exhausted – random fallback
            i = int(np.argmax(a))

        # ------------------------------------------------------------------
        # pick a target index that has a tiny partner value
        # ------------------------------------------------------------------
        min_partner = partner.min()
        cand = np.where(partner == min_partner)[0]
        # avoid choosing the same index as the source
        cand = cand[cand != i]
        if cand.size == 0:
            # fallback: cheapest a entry (excluding i)
            others = np.arange(n)
            others = others[others != i]
            j = int(others[np.argmin(a[others])])
        else:
            # among candidates choose the one where a is smallest
            j = int(cand[np.argmin(a[cand])])

        # ------------------------------------------------------------------
        # decide how much mass to move (fraction shrinks with time)
        # ------------------------------------------------------------------
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # ------------------------------------------------------------------
        # incremental update of the convolution (exact, O(n))
        # ------------------------------------------------------------------
        b_new = b.copy()
        # the linear part of the update uses the *old* a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # self‑terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        # cross term i + j (removed from the pairwise contributions)
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        # ------------------------------------------------------------------
        # accept only strictly improving moves
        # ------------------------------------------------------------------
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # stagnation handling: shake the current best solution
        # ------------------------------------------------------------------
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")

        # periodic recomputation of the full convolution to avoid drift
        if it % 5000 == 0 and it > 0:
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects the given time budget,
    performs a cheap parametric sweep for initialisation and then runs a
    directed greedy hill‑climber.
    """
    np.random.seed(seed)
    start_time = time.time()
    # give a tiny safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a few thousand entries gives a good trade‑off.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise a – cheap parametric sweep (uniform / linear / …)
    # --------------------------------------------------------------
    a0, _ = _initial_candidate(n)

    # a tiny random jitter helps break symmetry
    a0 += np.random.normal(scale=1e-5, size=n)
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    best_a = a0.copy()
    best_val = _objective(best_a)
    print(f"[Init] initial objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase – deterministic greedy hill‑climbing on the exact objective
    # --------------------------------------------------------------
    best_a, best_val = _greedy_hillclimb(best_a,
                                         start=start_time,
                                         deadline=deadline,
                                         best_a=best_a,
                                         best_val=best_val,
                                         target_val=1.503)

    # --------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:                                   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
