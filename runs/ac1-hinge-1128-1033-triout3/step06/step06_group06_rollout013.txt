"""
Propose a step‑function (non‑negative list of heights) that minimises
`evaluate_sequence`.  The algorithm consists of three stages:

1. **Initialisation** – generate a few simple parametric candidates
   (uniform, decreasing exponential, sine, etc.) and keep the best.

2. **Adam optimisation with variance regularisation** – we minimise a
   smooth surrogate of the true objective:
       L(a) = 2·n·∑ₖ pₖ·(a*a)ₖ  +  λ·Var((a*a)ₖ)
   where *p* is a soft‑max over the convolution entries (temperature ε)
   and λ is a time‑dependent weight that forces the convolution to become
   flat.  The optimisation is performed on the simplex using log‑softmax
   logits and the Adam optimiser.

3. **Targeted hill‑climbing + simulated annealing** – a fast local search
   that moves a tiny amount of mass from indices that contribute most to
   the current maximal convolution entry to indices that contribute little.
   A short annealing phase accepts occasional worsening moves to escape
   plateaus.

The whole search respects the 1000 s time budget; the best vector found
before the deadline is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolutions
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Objective on the simplex (Σa = 1) – 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Exact objective used for bookkeeping (Σa is assumed to be 1)."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used after gradient steps
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector `v` onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric initialisation (uniform / linear / quadratic / sine / expo)
# ----------------------------------------------------------------------
def _initial_candidate(n: int):
    """Return a promising initial vector together with its objective."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    # linear decreasing
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # quadratic decreasing
    a = (np.arange(n, 0, -1, dtype=float) ** 2)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # sine (half‑wave)
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # exponential families
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    # a random Dirichlet – cheap stochastic start
    a = np.random.dirichlet(np.ones(n))
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – Adam on smooth‑max + variance regularisation
# ----------------------------------------------------------------------
def _adam_phase_var(a0: np.ndarray,
                    n: int,
                    start: float,
                    deadline: float,
                    best_a: np.ndarray,
                    best_val: float,
                    target_val: float = 1.45) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth approximation of max(conv(a,a))
    plus a variance penalty on the whole convolution.  The variables are
    kept on the simplex via a log‑softmax parametrisation.
    """
    # logits for the simplex (soft‑max)
    w = np.log(a0 + 1e-12)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.08, 0.002
    eps_adam = 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 40000

    L = 2 * n - 1                     # length of full convolution

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ---- obtain a on the simplex -------------------------------------------------
        w_shift = w - np.max(w)               # numeric stability
        a = np.exp(w_shift)
        a /= a.sum()                           # Σa = 1, a ≥ 0

        # ---- exact objective (for bookkeeping) ---------------------------------------
        cur_val = _objective_value(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # ---- schedule for learning rate / temperature / λ ----------------------------
        progress = (now - start) / (deadline - start)
        lr = lr_start * (lr_end / lr_start) ** progress
        eps_smooth = 0.05 * (1e-12 / 0.05) ** progress   # temperature ε
        lam = 0.8 * (1.0 - progress)                     # λ goes from 0.8 → 0

        # ---- convolution of a with itself ------------------------------------------------
        c = _full_convolution(a)            # length 2·n‑1
        max_c = c.max()
        mu = c.mean()

        # ---- smooth‑max part -------------------------------------------------------------
        shifted = (c - max_c) / eps_smooth
        weights = np.exp(shifted)
        p = weights / weights.sum()          # soft‑max over positions

        # gradient of smooth‑max w.r.t. a (see comments in the original code)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)        # length 3·n‑2
        seg = conv_ab[n - 1: 2 * n - 1]      # slice corresponding to ∂/∂a
        grad_smooth = 4.0 * n * seg[::-1]    # shape (n,)

        # ---- variance part ---------------------------------------------------------------
        w_vec = c - mu                       # (c_i - μ)
        w_rev = w_vec[::-1]
        conv_var = _conv_fft(a, w_rev)
        seg_var = conv_var[n - 1: 2 * n - 1]
        grad_var = (4.0 / L) * seg_var[::-1]

        # ---- total gradient w.r.t. a ----------------------------------------------------
        grad_a = grad_smooth + lam * grad_var

        # ---- chain rule through the soft‑max (log‑softmax logits) ------------------------
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---- Adam update ----------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional small random shake to avoid stagnation
        if np.random.rand() < 0.003:
            w += np.random.normal(scale=0.01, size=n)

        # restart if we have been idle for too long
        if no_improve > IMPROVE_LIMIT:
            print("[Adam] restart after stagnation")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climber (exact objective)
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.45) -> np.ndarray:
    """
    Fast local search that moves a tiny amount of mass from indices that
    contribute most to the current maximal convolution entry to indices that
    contribute little.  Strictly improving moves are kept; after long
    stagnation the algorithm perturbs the current best and restarts.
    """
    a = a_start.copy()
    b = _full_convolution(a)                     # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contribution of each index to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – bias towards large contributions
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – sometimes force a low‑contribution location
        if np.random.rand() < 0.35:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = np.random.choice(n, p=probs_j)
            else:
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-14:          # strictly improving
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb and restart from the best observed sequence
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill] perturb after stagnation (it={it})")
        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase C – short simulated‑annealing refinement
# ----------------------------------------------------------------------
def _annealing_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.45) -> tuple[np.ndarray, float]:
    """
    Metropolis‑style annealing that occasionally accepts worsening moves.
    """
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    T_start, T_end = 5e-4, 5e-8
    IMPROVE_LIMIT = 120_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        k_max = int(np.argmax(b))

        # source – same bias as hill‑climber
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # uniform target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # tiny transfer
        delta = a[i] * np.random.rand() * 0.005
        if delta <= 0.0:
            continue

        # incremental convolution update
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anneal] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # small shake to leave a plateau
            noise = np.random.normal(scale=0.003, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anneal] perturb after stagnation (it={it})")
        it += 1

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function within the allotted time budget.
    Returns a list whose entries sum to 1 (scale‑invariant for the
    `evaluate_sequence` wrapper).
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for the final `evaluate_sequence` call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a few thousand entries work well.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Initialise – try to reuse a known good construction if present.
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    except Exception:
        a0, _ = _initial_candidate(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    TARGET = 1.45          # a modest target lower than the current best

    # --------------------------------------------------------------
    # Phase A – Adam optimisation with variance regularisation
    # --------------------------------------------------------------
    phase_a_deadline = start_time + (deadline - start_time) * 0.50
    best_a, best_val = _adam_phase_var(a0, n,
                                       start=start_time,
                                       deadline=phase_a_deadline,
                                       best_a=best_a,
                                       best_val=best_val,
                                       target_val=TARGET)

    # --------------------------------------------------------------
    # Phase B – targeted hill‑climbing on the exact objective
    # --------------------------------------------------------------
    phase_b_deadline = start_time + (deadline - start_time) * 0.80
    best_a = _hillclimb_targeted(best_a, n,
                                 start=phase_a_deadline,
                                 deadline=phase_b_deadline,
                                 best_a=best_a,
                                 best_val=best_val,
                                 target_val=TARGET)

    # --------------------------------------------------------------
    # Phase C – short simulated‑annealing refinement
    # --------------------------------------------------------------
    best_a, best_val = _annealing_phase(best_a, n,
                                        start=phase_b_deadline,
                                        deadline=deadline,
                                        best_a=best_a,
                                        best_val=best_val,
                                        target_val=TARGET)

    # --------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
