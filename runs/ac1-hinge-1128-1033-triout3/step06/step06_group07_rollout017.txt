import time, numpy as np

def full_convolution(a):
    n = a.shape[0]
    out_len = 2*n-1
    size = 1 << (out_len-1).bit_length()
    ft = np.fft.rfft(a, size)
    conv = np.fft.irfft(ft*ft, size)[:out_len]
    return conv

def objective(a):
    n = a.shape[0]
    b = full_convolution(a)
    return float(2.0 * n * b.max() / (a.sum()**2))

def proj_simplex(v):
    # project onto probability simplex (sum=1, v>=0)
    # Implementation of Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def hillclimb(a, deadline):

    b = full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val = cur_val

    max_frac_end = 0.001
    no_improve = 0
    IMPROVE_LIMIT = 200000
    while time.time() < deadline:
        # compute k_max and contributions as before
        k_max = int(b.argmax())
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0:
            source = np.random.choice(n, p=contrib/total_contrib)
        else:
            source = int(np.argmax(a))
        # target weights low contribution
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0/(contrib[mask]+1e-12)
        target_weights[source] = 0.0
        w_sum = target_weights.sum()
        if w_sum > 0:
            target = np.random.choice(n, p=target_weights/w_sum)
        else:
            target = (source + np.random.randint(1, n)) % n  # ensure diff
        progress = (time.time() - start_time) / (deadline - start_time)
        max_frac = max_frac_start * (1 - progress) + max_frac_end * progress
        delta = a[source] * max_frac * np.random.rand()
        if delta <= 0:
            continue
        # incremental update
        b_new = b.copy()
        # remove source contributions
        b_new[source:source+n] -= 2.0 * delta * a
        # add target contributions
        b_new[target:target+n] += 2.0 * delta * a
        # quadratic corrections
        b_new[2*source] += delta*delta
        b_new[2*target] += delta*delta
        b_new[source+target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-15:
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Hill2 {int(time.time()-start_time)}s] new best {best_val:.6f}")
            # maybe break early if target reached
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # stochastic restart
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(best_a + noise, 0)
            a /= a.sum()
            b = full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print("[Hill2] restart after stagnation")
    return best_a, best_val

"""
Step‑function optimizer.

The algorithm works in three stages within the 1000‑second budget:

1. **Candidate generation** – a small set of analytic shapes
   (uniform, power‑law “U”, exponential, linear, sparse random) is built.
   The best of these (according to an exact convolution objective) seeds
   the subsequent optimisation.

2. **Projected sub‑gradient Adam** – the vector is represented by
   soft‑max logits `w`.  At each iteration the exact maximal convolution
   entry is located; its sub‑gradient w.r.t. `a` is known analytically
   (g_i = 2·a_{k_max‑i} when the index is in range).  The gradient is
   propagated through the soft‑max and fed to Adam.  This quickly drives
   the max‑convolution down while automatically keeping the sum equal to
   one.

3. **Targeted hill‑climbing** – after the smooth phase we perform many
   fast local moves that transfer a small amount of mass from an index
   heavily contributing to the current maximal entry to an index that
   contributes little.  Updates of the full convolution are performed
   incrementally (O(n) per move).  Stagnation triggers a short random
   perturbation and the search restarts from the current best.

All operations respect the non‑negativity and sum‑one constraints, and
the final sequence is returned as a plain Python list for the provided
`evaluate_sequence` wrapper.
"""

import time

# ------------------------------------------------------------
# Convolution helpers (FFT, full linear convolution)
# ------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ------------------------------------------------------------
# Exact objective (under any scaling, sum may be != 1)
# ------------------------------------------------------------
def objective(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) / (sum(a)^2)."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    b = full_convolution(a)
    return float(2.0 * n * b.max() / (s * s))

# Analytic initial families

def generate_initial_candidates(n: int):

    candidates = []

    # uniform
    candidates.append(np.ones(n) / n)

    x = np.arange(n, dtype=np.float64)
    eps = 1e-12

    for gamma in np.linspace(0.0, 6.0, 61):
        a = np.power(denom + eps, -gamma)
        s = a.sum()
        if s > 0:
            candidates.append(a / s)

    # exponential decreasing a_i ∝ exp(-θ·i)
    for theta in np.linspace(0.0, 5.0, 21):
        a = np.exp(-theta * x)
        s = a.sum()
        if s > 0:
            candidates.append(a / s)

    # linear decreasing (clamped at zero)
    for slope in np.linspace(0.0, 2.0, 11):
        a = np.maximum(1.0 - slope * (x / (n - 1)), 0.0)
        s = a.sum()
        if s > 0:
            candidates.append(a / s)

    # sparse random vectors (few non‑zeros)
    for frac in (0.02, 0.05, 0.10):

        a = np.zeros(n, dtype=np.float64)
        a[idx] = np.random.rand(m)
        s = a.sum()
        if s > 0:
            candidates.append(a / s)

# Adam optimisation using the exact max‑convolution sub‑gradient
# ------------------------------------------------------------
def adam_subgradient(start_a: np.ndarray,
                     global_start: float,
                     phase_deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam on the exact max‑convolution sub‑gradient.
    The parameters `global_start` and `phase_deadline` are wall‑clock times.
    Returns the best vector seen in this phase.
    """
    n = start_a.shape[0]
    # logits guaranteeing a simplex point
    w = np.log(start_a + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.20, 0.005
    eps = 1e-8
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0

    best_a = start_a.copy()
    best_val = objective(start_a)

    while True:
        now = time.time()
        if now >= phase_deadline:
            break

        # ---- soft‑max -> a on the simplex ----
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        # ---- exact max convolution and its sub‑gradient ----
        b = full_convolution(a)
        k_max = int(np.argmax(b))                # index of the max entry

        # sub‑gradient of max(b) w.r.t. a:
        # g_i = 2·a_{k_max‑i}   (if the index exists, else 0)
        grad_a = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad_a[mask] = 2.0 * a[idx[mask]]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # ---- Adam update with a linear schedule for the learning rate ----
        progress = (now - global_start) / (phase_deadline - global_start)
        lr = lr_start * (lr_end / lr_start) ** progress

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional perturbation to avoid being stuck
        if np.random.rand() < 0.003:
            w += np.random.normal(scale=0.01, size=n)

        cur_val = float(2.0 * n * b.max())
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam] new best = {best_val:.6f}")

    return best_a, best_val


# ------------------------------------------------------------
# Targeted hill‑climbing (exact O(n) update of the convolution)
# ------------------------------------------------------------
def hillclimb(start_a: np.ndarray,
              global_start: float,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Fast local search that moves a small amount of mass from a source index
    (chosen proportionally to its contribution to the current maximal
    convolution entry) to a target index (biased toward low contribution).
    Restarts with a small random perturbation after prolonged stagnation.
    """
    a = start_a.copy()
    n = a.shape[0]
    b = full_convolution(a)
    cur_val = 2.0 * n * b.max()
    best_a = a.copy()
    best_val = cur_val

    # schedule for how much mass may be moved
    max_frac_start, max_frac_end = 0.50, 0.001
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    iteration = 0
    eps = 1e-12

    while True:
        now = time.time()
        if now >= deadline:
            break

        # ------------------------------------------------------------------
        # Choose source (high contribution) and target (low contribution)
        # ------------------------------------------------------------------
        k_max = int(b.argmax())
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)

        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target weighting – favour indices that *don’t* help the max entry
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # fallback uniform among all other indices
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # ------------------------------------------------------------------
        # Determine amount of mass to move
        # ------------------------------------------------------------------
        progress = (now - global_start) / (deadline - global_start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress
        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # ------------------------------------------------------------------
        # Incremental update of the convolution (ordered pairs)
        # ------------------------------------------------------------------
        b_new = b.copy()
        # remove contributions of the source entry
        b_new[i:i + n] -= 2.0 * delta * a
        # add contributions of the target entry
        b_new[j:j + n] += 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ------------------------------------------------------------------
        # Accept only improving moves
        # ------------------------------------------------------------------
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill] new best = {best_val:.6f}")

        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # Stagnation handling
        # ------------------------------------------------------------------
        if no_improve > IMPROVE_LIMIT:
            # revert to the best known point and add a small Gaussian jitter
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill] stagnation – perturb (iteration {iteration})")
        iteration += 1

    return best_a, best_val


# ------------------------------------------------------------
# Main entry point required by the evaluation framework
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and (attempts to) minimise the
    ``evaluate_sequence`` objective.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------
    # Initialise – try to reuse a previous solution if available.
    # ------------------------------------------------------------
    a0 = None
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        print("[init] interpolated previous solution")
    except Exception:
        pass

    # ------------------------------------------------------------
    # Phase 0 – generate a few analytical seeds and keep the best.
    # ------------------------------------------------------------
    candidates = generate_initial_candidates(n)
    if a0 is not None:
        candidates.append(a0.copy())

    best_a = None
    best_val = np.inf
    for cand in candidates:
        val = objective(cand)
        if val < best_val:
            best_val, best_a = val, cand.copy()
    print(f"[init] best seed = {best_val:.6f}")

    # ------------------------------------------------------------
    # Phase A – Adam optimisation on the exact sub‑gradient.
    # ------------------------------------------------------------
    phase_a_deadline = start_time + 0.35 * budget_s
    if phase_a_deadline > deadline:
        phase_a_deadline = deadline
    best_a, best_val = adam_subgradient(best_a, start_time, phase_a_deadline)
    print(f"[Phase A] best = {best_val:.6f}")

    # ------------------------------------------------------------
    # Phase B – targeted hill‑climbing with restarts.
    # ------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:                     # not enough time for another run
            break
        # allocate at most 200 s per hill‑climb segment (or the whole remainder)
        segment_deadline = time.time() + min(200.0, remaining * 0.9)
        new_a, new_val = hillclimb(best_a.copy(), start_time, segment_deadline)
        if new_val < best_val:
            best_val, best_a = new_val, new_a.copy()
            print(f"[restart] improved best to {best_val:.6f}")

        # occasional random perturbation to diversify the search
        if np.random.rand() < 0.15:
            noise = np.random.normal(scale=0.02, size=n)
            best_a = np.maximum(best_a + noise, 0.0)
            best_a /= best_a.sum()

    # ------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
