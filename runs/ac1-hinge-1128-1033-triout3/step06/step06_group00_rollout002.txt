def propose_candidate(seed=0, budget_s=1000, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 2.0
    rng = np.random.default_rng(seed)

    n = 8000  # adjust maybe 8000
    # compute FFT size for convolution
    size = 2*n - 1
    nfft = 1 << (size - 1).bit_length()

    # Helper functions for conv and objective, using precomputed nfft.

    def conv(a):
        Fa = np.fft.rfft(a, n=nfft)
        conv_full = np.fft.irfft(Fa*Fa, n=nfft)[:size]
        return conv_full

    def objective(a):
        c = conv(a)
        return 2.0 * n * c.max()

    # warm start: if height_sequence_1 exists, use it (trim or pad)

        best_val = objective(a0)

    # Pre-generate base shapes list for mixing

"""
propose_candidate
-----------------
A simple multi‑phase stochastic optimizer for the step‑function problem.

* Phase 0 – warm‑start from a previous solution (if available) or from a
  uniform distribution.
* Phase 1 – random‑shape generation: repeatedly draw a mixture of classic
  shapes (exponential, power‑law, Gaussian, triangular, double‑sided,
  sine) with random parameters, normalise to the simplex and keep the best.
  Every few hundred iterations a short projected sub‑gradient descent
  (exact max‑subgradient) is applied to the current best sequence.
* Phase 2 – final simulated‑annealing fine‑tuning that moves a small amount
  of mass between two indices and accepts moves with a Metropolis rule.

All sequences are kept on the probability simplex (non‑negative,
sum = 1) by Duchi et al.’s projection.  The objective evaluated during the
search is

    f(a) = 2 * n * max(conv(a, a))

which is exactly the quantity that `evaluate_sequence` computes when the
sum of the coefficients is 1 (the scaling cancels out).  The routine respects
a 1000 s time budget and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution via real FFT (full linear convolution).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray, nfft: int) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT of length nfft."""
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)
    # the true length is len(a) + len(b) - 1
    return conv[: a.shape[0] + b.shape[0] - 1]


# ----------------------------------------------------------------------
# Objective for a simplex‑normalised vector (sum = 1).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, nfft_conv: int) -> float:
    """2 * n * max(conv(a, a)).  `a` is assumed to sum to 1."""
    conv = _full_convolution_fft(a, a, nfft_conv)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution term.
# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray,
                     w: np.ndarray,
                     n: int,
                     nfft_rev: int) -> np.ndarray:
    """
    Given a weight vector `w` (same length as conv(a,a)),
    compute the sub‑gradient g = 2 * (rev_a * w) restricted to the
    central `n` entries.
    """
    rev_a = a[::-1]
    conv = _full_convolution_fft(rev_a, w, nfft_rev)   # length = 3n-2
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the grader.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a non‑negative height sequence that (approximately) minimizes
    `evaluate_sequence`.  The routine respects `budget_s` seconds and prints
    progress information.
    """
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a size that still fits comfortably in memory.
    # ------------------------------------------------------------------
    n = 10000                               # length of the step function
    size_conv = 2 * n - 1
    nfft_conv = 1 << ((size_conv - 1).bit_length())
    size_rev = 3 * n - 2
    nfft_rev = 1 << ((size_rev - 1).bit_length())

    # ------------------------------------------------------------------
    # Warm start: reuse a previously discovered sequence if present,
    # otherwise start from the uniform distribution.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if 'height_sequence_1' in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if base.shape[0] != n:
                if base.shape[0] > n:
                    base = base[:n]
                else:
                    base = np.pad(base, (0, n - base.shape[0]))
            base = np.maximum(0.0, base)
            s = base.sum()
            if s > 0:
                a0 = base / s
                val0 = _objective(a0, n, nfft_conv)
                best_seq = a0.copy()
                best_val = val0
                print(f"[init] warm start from global, score={best_val:.12f}")
        except Exception:
            pass

    if best_seq is None:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
        best_seq = a0.copy()
        best_val = _objective(a0, n, nfft_conv)
        print(f"[init] uniform start, score={best_val:.12f}")

    # ------------------------------------------------------------------
    # Helper: generate a random shape (non‑negative, not yet normalised).
    # ------------------------------------------------------------------
    def _make_shape():
        """Return a randomly generated shape of length n."""
        shape_type = rng.integers(0, 7)
        x = np.arange(n, dtype=np.float64)
        if shape_type == 0:                     # exponential
            lam = rng.uniform(5e-5, 5e-2)
            return np.exp(-lam * x)
        if shape_type == 1:                     # power‑law
            p = rng.uniform(0.2, 3.5)
            return (x + 1) ** (-p)
        if shape_type == 2:                     # Gaussian
            sigma = rng.uniform(n / 12.0, n / 2.0)
            mu = rng.uniform(0, n - 1)
            return np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        if shape_type == 3:                     # symmetric triangle
            return np.minimum(x + 1, n - x)      # peaks in the centre
        if shape_type == 4:                     # double‑sided harmonic
            return 1.0 / (x + 1) + 1.0 / (n - x)
        if shape_type == 5:                     # half‑sine
            return np.sin(np.pi * (x + 0.5) / n)
        # Dirichlet‑like random gamma draw
        alpha = rng.uniform(0.1, 5.0)
        return rng.gamma(alpha, 1.0, size=n)

    # ------------------------------------------------------------------
    # Random‑shape generation loop (Phase 1).
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----- build a mixture of a few random shapes -----
        k = rng.integers(1, 5)                 # how many shapes to mix
        a = np.zeros(n, dtype=np.float64)
        for _ in range(k):
            weight = rng.random()
            a += weight * _make_shape()
        # normalise to the simplex
        s = a.sum()
        if s <= 0.0:
            continue
        a /= s

        # ----- evaluate -----
        val = _objective(a, n, nfft_conv)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            print(f"[iter {iteration:6d}] new best={best_val:.12f}")

        # ----- every few hundred iterations run a few exact sub‑gradient steps -----
        if iteration % 400 == 0:
            a_local = best_seq.copy()
            # a tiny fixed learning rate for the short refinement
            lr = 0.08
            for _ in range(6):
                conv_a = _full_convolution_fft(a_local, a_local, nfft_conv)
                M = conv_a.max()
                # active set (indices where conv is (almost) maximal)
                eps_active = 1e-14
                active = np.where(conv_a >= M - eps_active)[0]
                w = np.zeros_like(conv_a)
                if active.size:
                    w[active] = 1.0 / active.size
                else:
                    w[np.argmax(conv_a)] = 1.0
                # sub‑gradient
                g = _max_subgradient(a_local, w, n, nfft_rev)
                # projected gradient step
                a_local = _project_simplex(a_local - lr * g)
            # evaluate the refined candidate
            val_local = _objective(a_local, n, nfft_conv)
            if val_local < best_val - 1e-12:
                best_val = val_local
                best_seq = a_local.copy()
                print(f"[refine after {iteration}] new best={best_val:.12f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing fine‑tuning (Phase 2) – uses the remaining time.
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_val
    # linear temperature schedule
    T0 = 1e-3
    Tmin = 1e-8
    total_sa_iters = 0
    while time.time() < deadline:
        total_sa_iters += 1
        # temperature decays linearly to Tmin
        fraction = total_sa_iters / 200_000.0
        T = max(Tmin, T0 * (1.0 - fraction))

        # propose a small mass‑transfer move
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] <= 0.0:
            continue
        max_transfer = min(cur[i], 0.02)          # allow slightly larger jumps
        delta = rng.uniform(0.0, max_transfer)
        new = cur.copy()
        new[i] -= delta
        new[j] += delta                           # sum stays 1, non‑negativity kept

        new_score = _objective(new, n, nfft_conv)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / T):
            cur = new
            cur_score = new_score
            if cur_score < best_val - 1e-12:
                best_val = cur_score
                best_seq = cur.copy()
                print(f"[SA iter {total_sa_iters:6d}] new best={best_val:.12f}")

        # occasional global jitter to escape plateaus
        if total_sa_iters % 25000 == 0:
            cur = cur + rng.normal(scale=0.001, size=n)
            cur = _project_simplex(cur)
            cur_score = _objective(cur, n, nfft_conv)

    # ------------------------------------------------------------------
    # Return the best sequence in the format expected by `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # optional logging of the official score
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception as exc:
        print("[result] evaluate_sequence not available:", exc)

    return final_seq
