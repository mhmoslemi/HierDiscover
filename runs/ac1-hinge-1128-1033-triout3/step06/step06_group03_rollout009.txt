def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Optimises a non‑negative step‑function using an exponentiated‑gradient
    descent on a smooth log‑sum‑exp approximation of the max‑convolution,
    followed by a guided hill‑climbing refinement.
    """

    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 4000   # can be parameter, but keep moderate

    # initialise a
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)
        # Ensure length n via interpolation/resampling
        if a0.shape[0] != n:
            old_x = np.linspace(0., 1., a0.shape[0])
            new_x = np.linspace(0., 1., n)
            a0 = np.interp(new_x, old_x, a0)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s < 1e-12:
            a = np.full(n, 1.0/n)
        else:
            a = a0 / s
    except Exception:
        a = np.full(n, 1.0/n)

    best_a = a.copy()
    conv = _full_convolution(a)
    max_conv = conv.max()
    best_val = 2.0 * n * max_conv  # objective (sum =1)

    # hyperparameters
    step0 = 0.5
    eps = 0.01
    min_eps = 1e-5
    eps_decay = 0.9995   # each iteration
    no_improve = 0
    max_no_improve = 5000
    iter = 0

    while time.time() < deadline - 2.0:
        # compute convolution and objective for current a
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        # update best
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[iter {iter}] new best = {best_val:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # schedule epsilon and step
        eps = max(eps * eps_decay, min_eps)
        step = step0 / np.sqrt(iter + 1)

        # compute weight distribution w
        # shift for numeric stability
        w = np.exp((conv - max_conv) / eps)
        # sum may be huge? but conv <= max so exponent <= 1
        w_sum = w.sum()
        w = w / w_sum

        # compute gradient via correlation
        # Use FFT for efficiency
        size = _next_pow2(len(a) + len(w) - 1)
        fa = np.fft.rfft(a, size)
        fw = np.fft.rfft(w, size)
        corr = np.fft.irfft(np.conj(fa) * fw, size)
        g = 2.0 * corr[:n]
        g = np.maximum(g, 0.0)

        # multiplicative update (mirror descent on simplex)
        g_clip = np.clip(g, 0.0, 30.0 / max(step, 1e-12))   # prevent extreme exp
        factor = np.exp(-step * g_clip)
        a_new = a * factor
        s_new = a_new.sum()
        if s_new == 0.0:
            # fallback to previous a
            a_new = a.copy()
            s_new = a_new.sum()
        a_new = a_new / s_new

        # compute objective of a_new
        conv_new = _full_convolution(a_new)
        max_conv_new = conv_new.max()
        new_val = 2.0 * n * max_conv_new

            # cur_val set later

            if rng.random() < 0.001:

    if time.time() < deadline:

"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.

Algorithm
========
* Work on the simplex (non‑negative entries with unit sum) – the objective is
  homogeneous, so fixing the sum to 1 does not change its value.
* A smooth surrogate for the non‑smooth max‑convolution is used:
        F_ε(a) = ε·log( Σ_k exp( (a∗a)[k] / ε ) )
  Its gradient is a weighted average of the sub‑gradient of the individual
  convolution lags.  The weights are a softmax over the convolution values
  and become increasingly concentrated on the true maximisers as ε→0.
* We perform mirror‑descent on the simplex via an exponentiated‑gradient
  (multiplicative) update:
        a_i ← a_i·exp(−η·g_i)  ;  a ← a / Σ a .
  This automatically respects non‑negativity.
* The step‑size η decays as 1/√t, the smoothing ε decays geometrically.
* If progress stalls we inject a small random perturbation around the best
  known point (a simple “restart”).
* After the iterative phase a short guided hill‑climbing refinement (mass
  transfers) is executed – the same routine that proved useful in the
  previous state‑of‑the‑art code.

The whole routine respects a user‑supplied time budget (default 1000 s) and
returns the best list of heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full auto‑convolution of a real vector (length 2·len(x)‑1) using FFT.
    Tiny negative rounding artefacts are clipped to zero.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)               # eliminate round‑off negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (FFT based)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local refinement: repeatedly move a small amount of mass away from the
    lag(s) that realise the current maximal convolution value.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # contributions a[i] * a[j] with i+j = k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # pick two destination indices that are not the same lag
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # feasible mass transfer
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise a non‑negative step‑function using a smoothed log‑sum‑exp
    surrogate of the max‑convolution, an exponentiated‑gradient (mirror)
    descent on the simplex, and a final guided hill‑climbing phase.
    Returns the best list of heights found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise a – re‑use a previous construction if available.
    # ------------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a = a0 / s if s > 0 else np.full(n, 1.0 / n)
    except Exception:
        a = np.full(n, 1.0 / n)   # uniform start

    # ensure exact unit sum (helps with the homogeneous objective)
    a = a / a.sum()

    # ------------------------------------------------------------------
    # Track the best solution seen so far.
    # ------------------------------------------------------------------
    conv = _full_convolution(a)
    max_conv = conv.max()
    best_a = a.copy()
    best_val = 2.0 * n * max_conv          # objective = 2·n·max(conv)  (sum(a)=1)
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters for the exponentiated‑gradient phase.
    # ------------------------------------------------------------------
    step0 = 2.0               # initial step magnitude (η₀)
    eps = 0.05                # initial smoothing ε
    min_eps = 1e-5
    eps_decay = 0.9995        # geometric decay factor per iteration
    no_improve = 0
    max_no_improve = 5000
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        it += 1

        # Current objective (exact, no need to call evaluate_sequence).
        conv = _full_convolution(a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        # Record the best solution.
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[iter {it}] new best = {best_val:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 1) Smooth surrogate: soft‑max weights over convolution lags.
        # --------------------------------------------------------------
        eps = max(eps * eps_decay, min_eps)
        w = np.exp((conv - max_conv) / eps)   # w_k ∈ (0,1], Σw = 1 after normalisation
        w /= w.sum()

        # --------------------------------------------------------------
        # 2) Gradient of the smooth surrogate.
        #    g_i = 2 * Σ_k w_k * a_{k-i}   (correlation of a and w)
        # --------------------------------------------------------------
        size_corr = _next_pow2(len(a) + len(w) - 1)
        fa = np.fft.rfft(a, size_corr)
        fw = np.fft.rfft(w, size_corr)
        corr = np.fft.irfft(np.conj(fa) * fw, size_corr)
        g = 2.0 * corr[:n]               # first n entries correspond to i = 0 … n‑1
        g = np.maximum(g, 0.0)           # numerical safety

        # --------------------------------------------------------------
        # 3) Exponentiated (multiplicative) update on the simplex.
        # --------------------------------------------------------------
        step = step0 / np.sqrt(it)       # η_t = η₀ / √t
        log_factor = -step * g
        # Clip to avoid underflow/overflow in the exponent.
        log_factor = np.clip(log_factor, -40.0, 0.0)
        factor = np.exp(log_factor)
        a_new = a * factor
        s_new = a_new.sum()
        if s_new == 0.0:                 # safeguard – fall back to the current point
            a_new = a.copy()
            s_new = a_new.sum()
        a_new /= s_new                    # renormalise to unit sum

        # --------------------------------------------------------------
        # 4) Acceptance test (simple greedy – we only keep genuine improvements).
        # --------------------------------------------------------------
        conv_new = _full_convolution(a_new)
        max_conv_new = conv_new.max()
        new_val = 2.0 * n * max_conv_new

        if new_val < cur_val:
            a = a_new                     # accept improvement
        else:
            # occasional stochastic acceptance to escape tiny plateaus
            if rng.random() < 0.001:
                a = a_new

        # --------------------------------------------------------------
        # 5) Simple restart if we see no progress for a while.
        # --------------------------------------------------------------
        if no_improve >= max_no_improve:
            pert = rng.normal(scale=0.02, size=n)
            a = _simplex_projection(best_a + pert)
            no_improve = 0
            eps = 0.05
            step0 = 2.0
            print(f"[restart] iter {it}, best ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final guided hill‑climbing refinement (uses the exact evaluate_sequence).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # compute the exact evaluation of the current best for the hill‑climber
        best_val = evaluate_sequence(best_a.tolist())
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # ------------------------------------------------------------------
    # Final reporting and return.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] evaluation = {final_score:.6f}")

    return best_a.tolist()
