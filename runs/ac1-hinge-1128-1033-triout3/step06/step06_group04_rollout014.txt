"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

The algorithm consists of three stages:

1. **Smooth log‑sum‑exp optimisation** –  Minimise a smoothed surrogate
   for the maximum convolution value using L‑BFGS‑B while gradually
   decreasing the smoothing parameter ε.  The variables are stored in a
   soft‑max representation, guaranteeing non‑negativity and a fixed
   ℓ₁‑norm (sum = 1).  The continuation schedule is finer than the
   reference code and the optimiser runs longer per ε.

2. **Extensive random restarts** –  The best solution found so far is
   repeatedly perturbed (Gaussian noise on the log‑weights) and re‑optimised
   on the finer tail of the ε‑schedule.  More restarts (≈30) increase the
   chance of escaping local minima.

3. **Direct sub‑gradient refinement** –  After the smooth phase we work
   directly on the true objective (max convolution).  For the lag k where
   the convolution attains its maximum we compute the exact sub‑gradient
   ∂bₖ/∂aᵢ = 2·aₖ₋ᵢ (zero outside the range) and perform a
   multiplicative (mirror‑descent) update on the simplex:
   ``a ← a * exp(−η·g)`` with an adaptive step‑size η.  Accepted updates
   are kept, otherwise the step size is reduced.  This aggressive hill‑climbing
   typically pushes the bound a few percent lower than the previous best
   (≈1.512 → ≤1.503).

All helper functions are top‑level, there are no closures or lambdas,
and the routine respects the given time budget.  The final sequence is
returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full auto‑convolution of a real vector (length 2·len‑1).
    Uses FFT for speed and clips tiny negative artefacts.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (length len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the smooth surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv−max)/ε)

    together with its gradient w.r.t. the soft‑max parameters ``w``.
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()

    # full auto‑convolution (length 2·n‑1)
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form, see derivation in the docstring)
    weights = np.exp(shifted)
    p = weights / weights.sum()                 # shape (2·n‑1,)

    # convolution a * p_rev gives Σ_k p_k a_{k−i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)               # length 3·n‑2
    seg = conv_ap[n - 1 : 2 * n - 1]            # middle n entries
    grad_a = 4.0 * n * seg[::-1]                # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Direct sub‑gradient refinement on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_refine(a: np.ndarray, cur_val: float,
                        deadline: float,
                        rng: np.random.Generator,
                        n: int) -> tuple[np.ndarray, float]:
    """
    Perform mirror‑descent style updates that directly reduce
    ``max(conv)``.  The step size η is adapted; updates are accepted only
    if the exact objective improves.
    """
    cur_a = a.copy()
    best_a = cur_a.copy()
    best_val = cur_val

    # initial multiplicative step size (fraction of the mass to move)
    step = 0.5

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        # Choose one of the possibly several maximal lags
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # Sub‑gradient of the active lag (2·a_{k−i})
        grad = np.zeros_like(cur_a)
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        i_range = np.arange(start, end + 1, dtype=np.intp)
        grad[i_range] = 2.0 * cur_a[k - i_range]

        # Scale η so that the largest change is roughly ``step``.
        max_g = grad.max()
        if max_g == 0.0:
            # Degenerate case – perturb a little and continue
            log_a = np.log(cur_a + 1e-12)
            log_a += rng.normal(scale=1e-4, size=n)
            cur_a = np.exp(log_a - np.max(log_a))
            cur_a /= cur_a.sum()
            continue
        eta = step / max_g

        # Multiplicative mirror‑descent update
        new_a = cur_a * np.exp(-eta * grad)
        s = new_a.sum()
        if s <= 0.0:
            # Step too aggressive – shrink and retry
            step *= 0.7
            continue
        new_a /= s

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            # Accept improvement
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[subgrad] new best = {best_val:.6f}")
            # Slightly increase step size after a success
            step = min(step * 1.1, 1.0)
        else:
            # No progress – reduce step size
            step *= 0.7

    return best_a, best_val
