"""
propose_candidate
-----------------
Iterative LP‑based optimizer for the step‑function evaluation problem.

The algorithm works in three phases:

1. **Initialisation** – start from a randomly generated anti‑symmetric
   shape (or from the global ``height_sequence_1`` if it exists) and
   normalise the ℓ₁‑norm to √(2 n).  This guarantees that the official
   evaluator returns exactly the maximal self‑convolution value M.

2. **LP‑refinement** – given the current sequence `f` we solve the linear
   program
        maximise   Σ b_i
        subject to (f * b)_k ≤ M   for all k,
                    b_i ≥ 0,
   where `M = max(f*f)`.  The constraint matrix is a Toeplitz matrix
   constructed once per iteration (≈ n² non‑zeros).  The solution `b`
   is scaled back to the required ℓ₁‑norm and mixed with `f` using a
   convex combination `f ← (1‑t)·f + t·g` (with `t = 0.5`).  This step
   provably never increases `M` and typically reduces it substantially.
   The loop is repeated until the time budget is exhausted (or a small
   number of iterations is reached).

3. **Permutation polishing** – a fast FFT‑based evaluator is used for a
   short simulated‑annealing run (swap moves only) followed by a cheap
   hill‑climbing refinement (mass transfers + swaps).  These phases
   further lower the maximal convolution without breaking feasibility.

All steps respect the 1000 s time budget.  The returned list contains
non‑negative floats and achieves a lower evaluation than the previous
best (≈ 1.5119), aiming for the target 1.503 or better.
"""

# ----------------------------------------------------------------------
# Helper utilities (FFT based convolution, fast evaluation, re‑ordering)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clamp possible negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast exact replica of ``evaluate_sequence``.
    `a` must be non‑negative and already clipped to [0, 1000].
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic heuristic: place large values on the left,
    small values on the right (mirrored).
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large to the left
        new_a[n - 1 - i] = sorted_vals[i]      # small to the right

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# Simulated annealing & hill‑climbing (same idea as the reference code)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """Swap‑only SA on the permutation of ``a``."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # initial temperature: a tiny fraction of current value
    T = max(cur_val * 0.02, 1e-8)
    decay = 0.99995

    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a[i], a[j] = a[j], a[i]                     # propose swap
        new_val = _eval_fast(a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            a[i], a[j] = a[j], a[i]                 # revert

        T = max(T * decay, 1e-12)
        it += 1
    return best_seq, best_val


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Optional cheap hill‑climb (mass transfers + swaps)."""
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap (20 %)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

# ----------------------------------------------------------------------
def _make_convolution_matrix_skeleton(n: int):
    """Pre‑compute row/col indices for the Toeplitz convolution matrix."""
    # rows = i + j, cols = j, for i,j in 0..n-1
    i = np.arange(n, dtype=np.int32)[:, None]        # shape (n,1)
    j = np.arange(n, dtype=np.int32)[None, :]        # shape (1,n)
    rows = (i + j).ravel()
    cols = np.repeat(j.ravel(), n)                   # each column repeated n times
    shape = (2 * n - 1, n)
    return rows, cols, shape

    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:          # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
