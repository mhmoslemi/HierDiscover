"""
Optimise a non‑negative step function (probability vector) for the
``evaluate_sequence`` objective.

Main ideas
==========

* Keep the sum of the coefficients equal to 1 (the objective is
  scale‑invariant).  The feasible set is therefore the probability simplex.
* At each iteration compute the full autocorrelation
  ``c = a * a`` with an FFT (O(n log n), n≈4000).  Its maximum M is the
  current value of the objective (2 n · M).
* The sub‑gradient of the max‑convolution is
        g_i = Σ_{k∈S} 2·a_{k-i}
  where S is the (tiny) set of lags attaining the maximum.  This can be
  evaluated efficiently as a convolution of ``a`` with the reversed indicator
  of S.
* Perform a projected (Adam‑style) descent on the simplex:
        m ← β₁·m + (1‑β₁)·g
        v ← β₂·v + (1‑β₂)·g²
        â ← a – α·m̂/(√v̂+ε)          (bias‑corrected moments)
        a ← simplex_projection(â)
  The step size α is reduced like 1/√t, but a small line‑search over a few
  scaling factors (1, ½, ¼, ⅛) is tried each iteration to guarantee a
  monotone improvement whenever possible.
* After a long sub‑gradient phase a short guided hill‑climbing refinement
  (mass‑transfer moves) is applied – this is the routine that proved useful
  in the previous state‑of‑the‑art code.
* The whole optimisation runs for at most the given time budget (≈ 1000 s)
  and returns the best coefficient vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negatives caused by numerical errors
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (FFT based)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local refinement: move a small fraction of the mass from the pair that
    contributes most to the current maximal lag to two low‑mass positions.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.07          # a bit larger than the previous version

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        # pick a lag uniformly among the active ones
        k = int(rng.choice(ks))

        # indices i that appear in the sum for this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # pick two low‑mass destination indices (avoid recreating the same lag)
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # feasible transfer
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max (the contribution comes from a_i^2)
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 5e-4)
        else:
            step_frac = max(step_frac * 0.995, 5e-4)

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Objective helper (fast, uses FFT)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) assuming Σa = 1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------

    expires.
    """

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
