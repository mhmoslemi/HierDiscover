def _adam_optimize(w, n, eps, max_iters, deadline, best_a, best_val):
    b1=0.9; b2=0.999; lr = 0.1; eps_adam = 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)

    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        L, grad = _smooth_obj_and_grad(w, eps, n)
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*(grad**2)
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        w = w - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if t % 50 == 0:
            # Evaluate actual objective
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_a = a.copy()
    return w, best_a, best_val

"""
Propose a step‑function (non‑negative sequence) that minimizes
``evaluate_sequence``.

The algorithm combines several complementary heuristics:

1. **Random restarts** – many trials start from a random Dirichlet
   distribution (or from the known best construction if available).

2. **Smooth surrogate optimisation** – we minimise a smoothed version of
   the objective
       Lε(a) = 2·n·max(conv(a)) + 2·n·ε·log∑_k exp((conv_k−max)/ε)
   using the Adam optimiser on the soft‑max parameters w (a_i = exp(w_i)
/∑exp(w)).  A continuation schedule gradually reduces ε from 5e‑2 down
   to 1e‑6, allowing the optimiser to first see a smooth landscape and
   then focus on the true max‑norm.

3. **Anti‑symmetric re‑ordering** – after each ε‑stage we apply a cheap
   heuristic that pairs large values with small ones when the sequence
   is reversed; this frequently flattens the autocorrelation peak.

4. **Hill‑climbing refinement** – finally a fast stochastic hill‑climber
   (random swaps / small mass transfers) is run on the best sequence
   found so far.

All steps respect the 1000 s time budget; the routine returns the
best sequence discovered before the deadline.
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative artefacts
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothing surrogate for max‑convolution, gradient w.r.t. soft‑max
# parameters w (a_i = exp(w_i)/∑exp(w))
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the smoothed objective L_ε and its gradient w.r.t. w.

    Returns
    -------
    L : float
        Value of the surrogate objective.
    grad_w : ndarray
        Gradient of L w.r.t. the parameters w.
    """
    # soft‑max (numerically stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()                     # a ≥ 0, ∑a = 1

    # full autocorrelation of a
    conv = _full_convolution(a)      # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    weights = np.exp(shifted)
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    p = weights / weights.sum()                # distribution over lag indices
    p_rev = p[::-1]

    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the segment that corresponds to the derivative of each a_i
    seg = conv_ap[n - 1: 2 * n - 1]            # length n
    grad_a = 4.0 * n * seg[::-1]               # chain rule for the quadratic term

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return float(L), grad_w.astype(np.float64)


# ----------------------------------------------------------------------
# Simple Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------


def _adam_optimize(w, n, eps, max_iters, deadline, best_a, best_val):
    """Run Adam on the smooth surrogate for at most max_iters steps."""
    b1, b2 = 0.9, 0.999
    lr = 0.12                     # base learning rate – tuned empirically
    eps_adam = 1e-8

    m = np.zeros_like(w)
    v = np.zeros_like(w)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        L, grad = _smooth_obj_and_grad(w, eps, n)

        # Adam updates
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad ** 2)

        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Periodically evaluate the true objective
        if t % 70 == 0:
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[adam eps={eps:.1e} iter={t}] new best = {best_val:.6f}")

    return w, best_a, best_val


# ----------------------------------------------------------------------
# Heuristic utilities
# ----------------------------------------------------------------------


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair large values with small ones when the sequence is reversed.
    This often reduces the peak of the self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)            # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large → left side
        new_a[n - 1 - i] = sorted_vals[i]      # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Very cheap hill‑climbing: random swaps (20 %) and small mass
    transfers (80 %).  Only strict improvements are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.04 * rng.random()   # 4 % of mass at i
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val
