def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    data = []
    # for each column i
    for i in range(n):
        # rows where b_i contributes: from i to i+n-1
        start = i
        end = i + n
        rows_i = np.arange(start, end)
        cols_i = np.full(n, i, dtype=int)
        data_i = a
        rows.append(rows_i)
        cols.append(cols_i)
        data.append(data_i)
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    return sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

"""
Propose a candidate step‑function sequence.

The algorithm is a hybrid of several complementary optimisation ideas:

1. **Smooth surrogate L‑BFGS** – optimise a soft‑max parametrisation of the
   sequence using a log‑sum‑exp approximation of the convolution peak.
   Many random restarts and a schedule of decreasing smoothing parameters
   explore the continuous landscape.

2. **Anti‑symmetric re‑ordering** – after the continuous phase the
   multiset of heights is reordered so that large values are paired with
   small ones when the sequence is mirrored.  This cheap deterministic
   transformation often reduces the convolution peak.

3. **Linear‑programming improvement** – given a current shape `a`,
   we solve a linear programme that maximises the total mass of a new
   sequence `b` under the *same* convolution bound `M = max(a*a)`.
   Mixing a fraction of `b` into `a` provably cannot increase the bound
   and empirically yields a lower objective.

4. **Simulated‑annealing on permutations** – a fast FFT based evaluator
   repeatedly proposes swaps of two positions.  Moves are accepted either
   because they improve the objective or with a temperature‑controlled
   probability, allowing escape from local minima.

5. **Final hill‑climbing** – cheap random swaps and small mass‑transfer
   moves polish the best sequence found.

All steps respect a global deadline (default 1000 s).  The function
returns a plain Python list of non‑negative floats that can be fed to the
provided ``evaluate_sequence`` evaluator.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import logsumexp
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Global placeholders for the L‑BFGS surrogate (avoid closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution of ``a`` with itself using FFT (full length)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clamp tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Exact objective (identical to ``evaluate_sequence``) but much faster.
    ``a`` is assumed non‑negative and within the clipping limits.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smooth surrogate used by L‑BFGS (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns surrogate value and gradient w.r.t. the soft‑max parameters ``w``.
    The surrogate approximates 2·n·max(a*a), where ``a`` = softmax(w).
    """
    # soft‑max (guarantees a_i ≥ 0 and Σ a_i = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # self‑convolution of a
    conv = _fft_convolve_self(a)              # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)               # scalar
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()                # probability over k

    # convolution of a with reversed p  (a * p_rev)
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)   # helper defined below
    # slice the central n entries (indices n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]                # derivative of max term

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper for ``scipy.optimize.minimize`` using the global eps/n."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# helper convolution used inside the gradient (FFT based, same as above)
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Re‑ordering heuristic (large ↔ small pairing)
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Deterministic re‑ordering that places large values on the left
    and small values on the right, mirroring the anti‑symmetric idea."""
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large on the left
        new_a[n - 1 - i] = sorted_vals[i]      # small on the right
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# Linear‑programming step
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix `A` such that  A·b = a * b (full convolution).
    The matrix is Toeplitz, constructed efficiently with ``scipy.sparse.diags``.
    """
    n = a.shape[0]
    # Every diagonal offset k (col‑row) = -d carries the constant a[d]
    diags = [np.full(n, a[d]) for d in range(n)]
    offsets = [-d for d in range(n)]
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def _lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ b_i   s.t.   a * b ≤ M  (component‑wise),  b ≥ 0,
    with an upper bound of 1000 on each b_i (identical to evaluate_sequence).
    Returns the optimal b (or the original a if the solve fails).
    """
    n = a.shape[0]
    A = _build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M, dtype=float)

    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)
    bounds = [(0.0, 1000.0) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success and res.x is not None:
        return res.x
    # fallback – return a copy of the current vector
    return a.copy()


# ----------------------------------------------------------------------
# Simulated annealing on permutations (swap moves only)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """Fast SA that swaps positions; uses the FFT evaluator for speed."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # initial temperature: a modest fraction of the current value
    T = max(cur_val * 0.02, 1e-8)
    decay = 0.99996
    iter_cnt = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose swap
        a[i], a[j] = a[j], a[i]
        new_val = _eval_fast(a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {iter_cnt})")
        else:
            # revert
            a[i], a[j] = a[j], a[i]

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        iter_cnt += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (swaps + mass transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        # 15 % chance of a pure swap, else a small mass transfer
        if rng.random() < 0.15:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 0.0:
                continue
            # transfer a random fraction (up to 5 %) of a[i] to a[j]
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

    print(f"[init] initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – smooth surrogate L‑BFGS with many random restarts
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-10), num=12)
    max_restarts = 30
    restart = 0

    while time.time() < deadline - 40.0 and restart < max_restarts:
        restart += 1
        print(f"[restart {restart}]")
        # random soft‑max parameters (wide variance → diverse shapes)
        w = np.random.randn(n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 35.0:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps

            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 250,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # obtain the current a from the soft‑max parameters
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # ------------------------------------------------------------------
        # Re‑order the best shape found in this restart (anti‑symmetric)
        # ------------------------------------------------------------------
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val - 1e-12:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – linear‑programming mixing step (few aggressive iterations)
    # ------------------------------------------------------------------
    lp_iters = 5
    t_mix = 0.2                 # initial mixing fraction
    while time.time() < deadline - 30.0 and lp_iters > 0:
        # compute current convolution bound M
        conv_a = _fft_convolve_self(best_a)
        M = conv_a.max()

        # solve LP to obtain a candidate b
        b = _lp_max_sum(best_a, M)

        # optional re‑ordering of b (helps when b has a very skewed multiset)
        b = _anti_symmetric_reorder(b)

        # mix a fraction of b into the current shape
        new_a = (1.0 - t_mix) * best_a + t_mix * b
        new_a = np.maximum(new_a, 0.0)
        new_a /= new_a.sum()          # renormalise (scale invariant)
        new_val = _eval_fast(new_a)

        if new_val < best_val - 1e-12:
            best_val = new_val
            best_a = new_a.copy()
            print(f"[LP] mix improved to {best_val:.6f} (t={t_mix:.3f})")
            # keep the same mixing fraction for the next iteration
        else:
            # if no improvement, shrink the mixing fraction
            t_mix *= 0.5
            print(f"[LP] no improvement, reducing mixing fraction to {t_mix:.3f}")

        lp_iters -= 1

    # ------------------------------------------------------------------
    # Stage 3 – simulated annealing on the permutation of the best shape
    # ------------------------------------------------------------------
    if time.time() < deadline - 20.0:
        print("[SA] starting simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val,
                                                deadline - 10.0)

    # ------------------------------------------------------------------
    # Stage 4 – final cheap hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing phase")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report the final score using the official evaluator and return the list
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required
    return best_a.tolist()
