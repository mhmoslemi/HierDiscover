        g[i_range] = a[k - i_range]

        # Compute contribution_i = a[i]*g[i] (= a[i]*a[k-i]) = 2*? Actually conv contributions.
        contrib = a * g  # equal to a_i * a_{k-i}
        # Choose donors: top approxs.
        num_donors = min(5, np.count_nonzero(contrib))
        if num_donors == 0:
            break
        donor_idx = np.argpartition(-contrib, num_donors - 1)[:num_donors]
        # recipients: indices with smallest g (i.e., a_{k-i}) and not donors
        # small g ensures moving mass there will increase conv_k the least.
        # Excluding donors to maintain balance.
        sorted_low = np.argsort(g)  # ascending
        # skip donors
        rec_candidates = [idx for idx in sorted_low if idx not in donor_idx]
        # take as many recipients as donors
        rec_idx = np.array(rec_candidates[:num_donors])
        # define amount to move from each donor

        new_a = a.copy()
        new_a[donor_idx] -= delta
        new_a[rec_idx] += delta  # same total mass moved

        s = new_a.sum()
        if s <= 0:
            # reduce step and continue
            step_frac = max(step_frac * 0.5, 1e-6)
            continue

"""
Hybrid optimisation for the self‑convolution objective.

The goal is to minimise

    f(a) = 2·n·max(conv(a,a))           with  a_i ≥ 0,  Σ a_i = 1

where `conv` is the full discrete convolution.  The objective is
scale‑invariant, therefore we work with L¹‑normalised vectors.

The optimiser consists of three light‑weight phases that respect the
1000 s wall‑clock budget:

1. **Exponentiated‑gradient (EG) descent** – a smooth sub‑gradient step
   on the active lag `k* = argmax conv`.  The EG update
   `a_i ← a_i·exp(-η·a_{k*‑i})` is guaranteed to stay in the simplex.
   The step size η is decreased with the iteration counter.

2. **Gradient‑guided mass‑transfer hill‑climb** – repeatedly move a
   fraction of mass from indices that contribute most to the active
   lag (`a_i·a_{k*‑i}` large) to indices that have the smallest
   partner value `a_{k*‑p}`.  This directly reduces the current peak
   `max(conv)` by an amount proportional to the difference of the two
   partner values.

3. **Stochastic fine‑tuning** – occasional random perturbations of the
   log‑weights (the “mirror‑descent’’ space) to escape shallow local
   minima.  Accept only improvements.

All phases use an FFT based convolution (`_full_convolution`) which
runs in O(n log n) and is fast for n≈4000.  The best sequence seen so
far is always retained and finally evaluated with the provided
`evaluate_sequence` function.

The function `propose_candidate` can be called directly; it returns a
list of non‑negative floats that respects the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full autocorrelation of a real vector (length 2·len(x)‑1).
    Uses FFT for speed and clips tiny negative artefacts.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Exact objective (fast, no clipping)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(a,a)) for a simplex‑normalised vector a.
    """
    n = a.shape[0]
    max_conv = np.max(_full_convolution(a))
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient descent on the active lag
# ----------------------------------------------------------------------
def _exponentiated_gradient(a: np.ndarray,
                            max_iters: int,
                            deadline: float,
                            rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a few hundred EG steps.  Returns the best vector found and its
    objective value.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    eta0 = 0.7                     # initial step size (tuned experimentally)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a)
        max_val = conv.max()
        ks = np.flatnonzero(conv == max_val)
        k = int(rng.choice(ks))   # pick one of the maximal lags

        # sub‑gradient of max(conv) w.r.t. a at the chosen lag:
        # g_i = a_{k-i} (zero outside the support)
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1, dtype=int)
        g[i_range] = a[k - i_range]

        # EG step with a diminishing learning rate
        eta = eta0 / (it + 1.0)
        a_new = a * np.exp(-eta * g)
        a_new /= a_new.sum()      # projection onto the simplex

        cur_val = _objective(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            print(f"[EG] it {it:4d}  new best = {best_val:.6f}")

        # Accept improvement unconditionally; otherwise keep the old a.
        if cur_val < _objective(a):
            a = a_new

        # occasional random perturbation to avoid stagnation
        if it % 50 == 0 and rng.random() < 0.08:
            perturb = np.exp(rng.normal(scale=0.2, size=n))
            a = a * perturb
            a /= a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – gradient‑guided mass‑transfer hill‑climb
# ----------------------------------------------------------------------
def _gradient_hill_climb(a: np.ndarray,
                         cur_best_val: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the worst contributors (largest
    a_i·a_{k*-i}) to positions with the smallest partner a_{k*-p}.
    The step size adapts to the success rate.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_best_val
    step_frac = 0.25                     # initial transfer proportion

    while time.time() < deadline:
        conv = _full_convolution(a)
        max_conv = conv.max()
        ks = np.flatnonzero(conv == max_conv)
        k = int(rng.choice(ks))

        # partner values for the active lag
        partner = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner[idx] = a[k - idx]

        # contribution of each index to the current peak
        contrib = a * partner                     # a_i * a_{k-i}
        if not np.any(contrib > 0):
            break

        # donors – top contributors (but we need a_i>0 to move mass)
        n_donors = min(5, np.count_nonzero(contrib))
        donor_idx = np.argpartition(-contrib, n_donors - 1)[:n_donors]

        # recipients – indices with smallest partner values (least impact)
        # ensure they are different from donors
        sorted_low = np.argsort(partner)
        rec_candidates = [i for i in sorted_low if i not in donor_idx]
        if len(rec_candidates) < n_donors:
            # fallback to any index not a donor
            rec_candidates = [i for i in range(n) if i not in donor_idx]
        rec_idx = np.array(rec_candidates[:n_donors])

        # amount to move from each donor (a fraction of its current mass)
        delta = a[donor_idx] * step_frac
        # protect against numerical issues
        delta = np.minimum(delta, a[donor_idx])

        # apply the transfer
        a_new = a.copy()
        a_new[donor_idx] -= delta
        a_new[rec_idx] += delta

        # renormalise (floating‑point drift)
        total = a_new.sum()
        if total <= 0:
            step_frac = max(step_frac * 0.6, 1e-6)
            continue
        a_new /= total

        new_val = _objective(a_new)

        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            print(f"[HG] new best = {best_val:.6f}")

        if new_val < _objective(a):
            a = a_new
            # slightly increase step size after success
            step_frac = min(step_frac * 1.03, 0.6)
        else:
            # shrink step size on failure
            step_frac = max(step_frac * 0.94, 1e-6)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – random log‑weight perturbations (small‑scale simulated annealing)
# ----------------------------------------------------------------------
def _random_perturbations(a: np.ndarray,
                          cur_best_val: float,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Tiny random moves in the log‑space to escape remaining plateaus.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_best_val
    while time.time() < deadline:
        # log‑space Gaussian perturbation
        log_a = np.log(a + 1e-14)
        log_a += rng.normal(scale=0.03, size=n)
        a_new = np.exp(log_a)
        a_new /= a_new.sum()
        new_val = _objective(a_new)
        if new_val < best_val:
            best_val = new_val
            best_a = a_new.copy()
            a = a_new
            print(f"[RND] new best = {best_val:.6f}")
        else:
            # occasionally accept a worsening move (very low probability)
            if rng.random() < 0.001:
                a = a_new
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # small safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate sized vector keeps the search affordable.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise from a previously saved solution if available,
    # otherwise start from a random Dirichlet vector.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to the new length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.ones(n)
    a /= a.sum()                     # L¹‑normalisation

    best_a = a.copy()
    best_val = _objective(best_a)
    print(f"[init] n={n}, objective={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent
    # ------------------------------------------------------------------
    max_eg_iters = 1500
    best_a, best_val = _exponentiated_gradient(best_a,
                                               max_iters=max_eg_iters,
                                               deadline=deadline,
                                               rng=rng)

    # ------------------------------------------------------------------
    # Phase 2 – gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _gradient_hill_climb(best_a,
                                                cur_best_val=best_val,
                                                deadline=deadline,
                                                rng=rng)

    # ------------------------------------------------------------------
    # Phase 3 – small random perturbations for fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _random_perturbations(best_a,
                                                 cur_best_val=best_val,
                                                 deadline=deadline,
                                                 rng=rng)

    # ------------------------------------------------------------------
    # Final evaluation via the provided helper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
