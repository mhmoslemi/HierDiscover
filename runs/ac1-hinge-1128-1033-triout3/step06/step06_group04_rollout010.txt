def simplex_projection(v):
    # ensure nonnegative sum to 1
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors (len(x)+len(y)-1)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used after arbitrary moves
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex {x >= 0, sum x = 1}.
    Returns a new array (the input is not modified).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Should not happen for non‑trivial vectors – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Generate a symmetric Beta(α,α) density (discretised on n points)
# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Returns a non‑negative vector of length n that approximates a Beta(α,α)
    density on [0,1].  The vector is already normalised to sum to 1.
    """
    # Mid‑point discretisation avoids the singularities at 0 and 1
    x = (np.arange(n) + 0.5) / n
    # log‑form to stay stable for α<1 (which yields a U‑shape)
    log_vals = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    vals = np.exp(log_vals)
    # Guard against NaNs/Infs that can appear for extreme α
    vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
    s = vals.sum()
    if s == 0.0:                     # fall back to uniform if everything vanished
        vals = np.ones(n, dtype=float)
        s = n
    return vals / s


# ----------------------------------------------------------------------
# Mass‑transfer hill climbing (same idea as the reference implementation)
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(
    a: np.ndarray,
    best_val: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Aggressive hill climbing that moves a fraction of the mass from the
    contributing pair(s) of the current maximal lag to low‑mass positions.
    The function stops when the time budget is exhausted and returns the
    best vector seen together with its exact objective value.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.2  # initial fraction of the donor mass to move

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue

        # pick a few strongest contributing pairs
        top_n = min(3, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]
        donors = [(int(i_range[t]), int(j_range[t])) for t in top_idx]

        # choose low‑mass destinations
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:                     # a_i^2 contributes
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue

        # re‑normalise (the ℓ₁‑norm is invariant for the objective,
        # but we keep the simplex for numerical stability)
        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue
        new_a /= s

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            # increase step size modestly after success
            step_frac = min(step_frac * 1.02, 0.5)
        else:
            # shrink step size when no improvement
            step_frac = max(step_frac * 0.95, 1e-5)

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Initialise from the previous construction if it exists; otherwise use
    # a uniform vector.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if old.shape[0] != n:
            # Interpolate to the required length
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
    except Exception:
        a0 = np.ones(n, dtype=float)          # start from uniform

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()
    best_a = a0.copy()
    best_val = 2.0 * n * np.max(_full_convolution(best_a))
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1) Grid search over symmetric Beta(α,α) shapes and uniform mixes
    # ------------------------------------------------------------------
    alphas = np.linspace(0.30, 0.80, 26)          # 0.30 … 0.80
    mixes = np.linspace(0.0, 1.0, 11)            # weight of the beta component

    uniform = np.ones(n, dtype=float) / n

    for alpha in alphas:
        if time.time() >= deadline:
            break
        beta = _beta_shape(n, alpha)
        for lam in mixes:
            if time.time() >= deadline:
                break
            cand = lam * beta + (1.0 - lam) * uniform
            cand /= cand.sum()
            val = 2.0 * n * np.max(_full_convolution(cand))
            if val < best_val:
                best_val = val
                best_a = cand.copy()
                print(f"[grid] α={alpha:.3f}, λ={lam:.2f} → best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Optional light random perturbations (helps to escape tiny plateaus)
    # ------------------------------------------------------------------
    perturb_interval = 2000
    it = 0
    while time.time() < deadline:
        it += 1
        # Move a small amount of mass from the most offending pair
        conv = _full_convolution(best_a)
        max_conv = conv.max()
        ks = np.flatnonzero(conv == max_conv)
        k = int(rng.choice(ks))

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = best_a[i_range] * best_a[j_range]
        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # Destination – a low‑mass index
        low = np.where(best_a < best_a.mean())[0]
        if low.size == 0:
            low = np.arange(n)
        forb = {i, j}
        low = np.setdiff1d(low, list(forb), assume_unique=True)
        if low.size == 0:
            low = np.setdiff1d(np.arange(n), list(forb), assume_unique=True)
        p = int(rng.choice(low))

        # Transfer 10 % of the donor mass
        if i == j:
            delta = best_a[i] * 0.10
            if delta <= 0:
                continue
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[p] += delta
        else:
            delta = min(best_a[i], best_a[j]) * 0.10
            if delta <= 0:
                continue
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] -= delta
            # give the removed mass to the destination (all of it)
            new_a[p] += 2.0 * delta

        new_a = np.maximum(new_a, 0.0)
        new_a /= new_a.sum()
        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        # Accept only improving moves – the check is cheap and keeps the
        # algorithm monotone.
        if new_val < best_val:
            best_a = new_a
            best_val = new_val
            print(f"[refine] iter {it} → new best = {best_val:.6f}")

        # Occasionally inject a tiny random Dirichlet blend to shake the
        # simplex and possibly discover a new basin.
        if it % perturb_interval == 0:
            blend = 0.02
            rnd = rng.dirichlet(alpha=np.ones(n))
            best_a = (1.0 - blend) * best_a + blend * rnd
            best_a /= best_a.sum()
            best_val = 2.0 * n * np.max(_full_convolution(best_a))
            print(f"[perturb] iter {it} → best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Aggressive mass‑transfer hill‑climbing (same as reference)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _mass_transfer_hill_climb(best_a, best_val, deadline, rng)

    # ------------------------------------------------------------------
    # Final verification with the provided evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
