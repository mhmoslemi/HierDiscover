def _refine_hill_climb(a, best_val, deadline):
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]
    while time.time() < deadline:
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val

"""
Hybrid optimiser for the step‑function evaluation problem.

The search proceeds in three phases:

1. **Smooth surrogate optimisation** – a differentiable surrogate of the
   original max‑convolution objective is minimised with many independent
   L‑BFGS restarts.  The surrogate is a log‑sum‑exp approximation of the
   maximum of the autocorrelation; a soft‑max parametrisation guarantees
   non‑negativity and a fixed ℓ₁‑norm (∑a=1).

2. **Permutation optimisation** – the current best sequence is refined
   by simulated‑annealing on its indices.  In addition to plain random
   swaps the algorithm uses *targeted* swaps that attack the convolution
   peak directly and occasional small mass‑transfer moves.

3. **Final hill‑climbing** – a cheap greedy pass (swaps + transfers)
   finishes the search.

All helper functions are top‑level, there are no closures or lambdas.
The routine respects a 1000 s budget and always returns a feasible list
of non‑negative floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global parameters that are injected into the smooth‑surrogate wrapper.
# ----------------------------------------------------------------------
_SB_EPS = None   # smoothing parameter for the current L‑BFGS run
_SB_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using FFT.  Result length = 2·len(a)‑1.
    Output is clamped to be non‑negative (tiny negative artefacts from
    round‑off are removed).
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (non‑negative output)."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast version of `evaluate_sequence` (exact, no clipping checks)."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _smooth_obj_and_grad(w: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate of the original objective and its gradient
    w.r.t. the soft‑max parameters w.
    """
    n = _SB_N
    eps = _SB_EPS

    # ----- soft‑max (ensures a_i ≥ 0 and Σa = 1) -----
    w_shift = w - np.max(w)          # numeric stability
    a = np.exp(w_shift)
    a = a / a.sum()                  # shape (n,)

    # ----- autocorrelation a * a (full) -----
    conv = _fft_convolve_self(a)     # length 2n‑1
    max_conv = conv.max()

    # ----- smooth approximation of max(conv) -----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)     # = log Σ exp((conv‑max)/eps)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a (closed form) -----
    weights = np.exp(shifted)
    p = weights / weights.sum()          # length 2n‑1, behaves like a soft‑max over k
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)        # length 3n‑2
    # the segment [n‑1, 2n‑2] (inclusive) corresponds to the needed slice
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]         # reverse to match original orientation

    # ----- chain rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic reordering: largest value left‑most, smallest right‑most,
    second‑largest next‑to‑left, etc.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[n - 1 - i]          # large to the left
        out[n - 1 - i] = s[i]          # small to the right
    if n % 2:
        out[half] = s[half]
    return out


def _interleaved_reorder(a: np.ndarray) -> np.ndarray:
    """
    Alternative deterministic ordering: largest, smallest, second‑largest,
    second‑smallest, …  (interleaving extremes).
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    out = np.empty_like(a)
    lo, hi = 0, n - 1
    idx = 0
    while lo <= hi:
        out[idx] = s[hi]
        idx += 1
        hi -= 1
        if lo <= hi:
            out[idx] = s[lo]
            idx += 1
            lo += 1
    return out


def _simulated_annealing(a: np.ndarray,
                         cur_val: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated annealing on the permutation of ``a``.
    Three move types are used:
      * random swap,
      * targeted swap (uses the index of the current convolution peak),
      * small mass transfer.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val
    cur_seq = a.copy()
    cur_val_cur = cur_val

    # initialise temperature – a modest fraction of the current value
    T = max(cur_val_cur * 0.02, 1e-8)
    decay = 0.99996
    iter_cnt = 0

    while time.time() < deadline:
        # decide move type
        r = rng.random()
        cand = cur_seq.copy()

        if r < 0.55:                                   # ---- random swap ----
            i, j = rng.integers(0, n, size=2)
            if i != j:
                cand[i], cand[j] = cand[j], cand[i]

        elif r < 0.85:                                 # ---- targeted swap ----
            conv = _fft_convolve_self(cur_seq)
            k_max = int(np.argmax(conv))
            # contribution of each position to the peak
            idx = np.arange(n)
            counterpart = k_max - idx
            mask = (counterpart >= 0) & (counterpart < n)
            contrib = np.zeros(n)
            contrib[mask] = cur_seq[idx[mask]] * cur_seq[counterpart[mask]]
            i = int(np.argmax(contrib))
            j = rng.integers(0, n)
            if i != j:
                cand[i], cand[j] = cand[j], cand[i]

        else:                                          # ---- mass transfer ----
            i, j = rng.integers(0, n, size=2)
            if i != j and cand[i] > 0.0:
                delta = cand[i] * 0.05 * rng.random()
                cand[i] -= delta
                cand[j] += delta

        new_val = _eval_fast(cand)
        delta = new_val - cur_val_cur

        # Metropolis acceptance
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur_seq = cand
            cur_val_cur = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cand.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {iter_cnt})")

        # cooling
        T *= decay
        if T < 1e-12:
            T = max(cur_val_cur * 0.02, 1e-12)

        iter_cnt += 1

    return best_seq, best_val


def _refine_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Fast greedy refinement: occasional swaps (20 % chance) and
    small mass‑transfer moves.  Only strictly improving moves are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                       # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                         # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point required by the evaluation framework.

    The routine combines smooth surrogate optimisation (L‑BFGS), deterministic
    reorderings, simulated‑annealing on permutations (including targeted moves)
    and a final hill‑climbing pass.  It respects the time budget and returns a
    plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 5000 offers a good trade‑off between richness and speed.
    # ------------------------------------------------------------------
    n = 5000
    global _SB_N, _SB_EPS
    _SB_N = n

    # ------------------------------------------------------------------
    # Initialise a starting sequence.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required length
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        # fallback: random Dirichlet (guarantees sum = 1)
        a0 = np.random.rand(n)
        a0 = a0 / a0.sum()
        print("[init] random start")

    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth surrogate optimisation (multiple L‑BFGS restarts)
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(1e-2), np.log10(1e-12), num=20)

    restart = 0
    # keep ~120 s for the later phases
    while time.time() < deadline - 120.0:
        restart += 1
        print(f"[restart {restart}]")
        # random initial soft‑max parameters (wide spread)
        w = np.random.randn(n) * 3.0

        for eps in eps_schedule:
            if time.time() >= deadline - 110.0:
                break
            _SB_EPS = eps
            # L‑BFGS on the smooth surrogate
            res = minimize(_smooth_obj_and_grad,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 500,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # recover a from w (soft‑max)
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a = a / a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # ------------------------------------------------------------------
        # Deterministic re‑orderings after each restart
        # ------------------------------------------------------------------
        for fn, name in [( _anti_symmetric_reorder, "anti_sym"),
                         ( _interleaved_reorder,    "interleaved")]:
            a_ord = fn(best_a)
            v_ord = _eval_fast(a_ord)
            if v_ord < best_val:
                best_val = v_ord
                best_a = a_ord.copy()
                print(f"[{name} reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing on permutations (swap + targeted)
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        print("[SA] starting simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val,
                                                deadline - 30.0)

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report the final score through the official evaluator (for logging)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # return a plain Python list as required by the framework
    return best_a.tolist()
