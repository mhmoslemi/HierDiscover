def project_simplex(v):
    # v can be any real numbers
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def hill_climb_step(a, cur_obj, step_frac, rng):
    n = len(a)
    conv = full_convolution(a)
    max_val = conv.max()
    # indices where max occurs:
    ks = np.flatnonzero(conv == max_val)
    k = int(rng.choice(ks))
    # compute donor contributions:
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end+1)
    j_range = k - i_range
    contrib = a[i_range] * a[j_range]
    # pick the pair with maximum contribution
    idx = int(np.argmax(contrib))
    i = int(i_range[idx])
    j = int(j_range[idx])
    # choose destination with low mass:
    low_candidates = np.where(a < a.mean())[0]
    if low_candidates.size == 0:
        low_candidates = np.arange(n)
    # ensure not same as i or j
    forbidden = np.unique([i, j])
    dest_candidates = np.setdiff1d(low_candidates, forbidden, assume_unique=True)
    if dest_candidates.size == 0:
        dest_candidates = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
    dest = int(rng.choice(dest_candidates))
    # determine delta
    if i == j:
        delta = a[i] * step_frac
        if delta <= 0:
            return a, cur_obj
        a_new = a.copy()
        a_new[i] -= delta
        a_new[dest] += delta
    else:
        delta = min(a[i], a[j]) * step_frac
        if delta <= 0:
            return a, cur_obj
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] -= delta
        a_new[dest] += 2 * delta
    # re-normalize (sum unchanged)
    # but due to rounding might be slightly off. Ensure nonnegative.
    a_new = np.maximum(a_new, 0)
    a_new_sum = a_new.sum()
    if a_new_sum == 0:
        return a, cur_obj
    a_new /= a_new_sum
    new_obj = 2.0 * len(a) * np.max(full_convolution(a_new))
    # accept if better
    if new_obj < cur_obj:
        return a_new, new_obj
    else:

"""
Propose a step‑function (non‑negative sequence) that approximately minimises
``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – start from the previously found sequence
   ``height_sequence_1`` if it exists, otherwise use a uniform vector.
2. **Sub‑gradient descent on the exact max‑convolution** – the objective is
   `2·n·max(conv(a,a))` (the denominator is fixed to 1 by normalising the
   vector).  For the current maximal lag `k` the sub‑gradient is
   `g_i = 2·a_{k-i}` (zero outside the support).  A simple projected step
   `a ← a – η·g` is performed, followed by non‑negativity clipping and
   re‑normalisation onto the probability simplex.
3. **Exploratory random mass moves** – occasional stochastic moves,
   transferring a small fraction of mass from a high‑mass index to a
   low‑mass one.  The move is kept if it improves the objective.
4. **Fine‑grained hill‑climbing** – repeatedly locate the lag where the
   convolution attains its maximum, pick the pair `(i,j)` that contributes
   most to that lag and move a tiny amount of mass from the donors to a
   low‑mass destination.  The step‑size is adapted in a multiplicative
   fashion.
5. The best sequence encountered is finally evaluated with the provided
   ``evaluate_sequence`` function and returned.

The whole search respects the supplied time budget (default 1000 s) and
prints intermediate progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data) – fast full self‑convolution
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a * a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # clip rounding artefacts
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Objective (sum of the sequence is kept equal to 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective = 2·n·max(conv(a,a)) under the normalisation Σa=1."""
    n = a.shape[0]
    return 2.0 * n * _full_convolution(a).max()


# ----------------------------------------------------------------------
# Sub‑gradient step for the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray, eta: float) -> tuple[np.ndarray, float]:
    """
    Perform one projected sub‑gradient step.
    Returns the new vector and its objective value.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_val = conv.max()
    # pick one maximal lag (any works)
    k = int(np.flatnonzero(conv == max_val)[0])

    # sub‑gradient g_i = 2·a_{k-i} (zero outside the support)
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.intp)
    j = k - i
    g[i] = 2.0 * a[j]

    # explicit Euler step, clip negatives, renormalise onto the simplex
    a_new = a - eta * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s == 0.0:               # degenerate – fall back to the old point
        return a, max_val * 2.0 * n
    a_new /= s
    new_obj = _objective(a_new)
    return a_new, new_obj


# ----------------------------------------------------------------------
# Simple hill‑climbing move that reduces the contribution of the worst pair
# ----------------------------------------------------------------------
def _hill_climb_step(a: np.ndarray, cur_obj: float, step_frac: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Identify the pair (i,j) with the largest product a_i·a_j among the
    indices that realise the maximal convolution value and move a small
    amount of mass to a low‑mass location.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)
    k = int(rng.choice(ks))

    # all contributing pairs for this lag
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = k - i_range
    contrib = a[i_range] * a[j_range]

    if contrib.size == 0:
        return a, cur_obj

    # the donor pair (i,j) with the biggest contribution
    idx = int(np.argmax(contrib))
    i = int(i_range[idx])
    j = int(j_range[idx])

    # destination – a low‑mass index that is not a donor
    low = np.where(a < a.mean())[0]
    if low.size == 0:
        low = np.arange(n)
    forbidden = np.unique([i, j])
    cand = np.setdiff1d(low, forbidden, assume_unique=True)
    if cand.size == 0:
        cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
    dest = int(rng.choice(cand))

    # amount of mass to relocate
    if i == j:
        delta = a[i] * step_frac
        if delta <= 0.0:
            return a, cur_obj
        a_new = a.copy()
        a_new[i] -= delta
        a_new[dest] += delta
    else:
        delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            return a, cur_obj
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] -= delta
        a_new[dest] += 2.0 * delta

    # keep feasibility
    a_new = np.maximum(a_new, 0.0)
    total = a_new.sum()
    if total == 0.0:
        return a, cur_obj
    a_new /= total

    new_obj = _objective(a_new)
    if new_obj < cur_obj:
        return a_new, new_obj
    else:
        return a, cur_obj
