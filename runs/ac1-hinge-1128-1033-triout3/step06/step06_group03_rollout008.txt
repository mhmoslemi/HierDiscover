"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initial shape** – use the discrete analogue of the *arcsine*
   distribution.  For a continuous arcsine law the self‑convolution is
   (almost) uniform, which is the ideal shape for minimising the
   worst‑case convolution entry.
2. **Smoothed mirror‑descent** – at each iteration we
   * compute the full convolution `c = a * a`,
   * turn it into a probability distribution `w` with a soft‑max
     (temperature `T` slowly annealed),
   * obtain the gradient of the smoothed objective
     `∇ = 2·(w * a_rev)[n‑1:n‑1+n]` via FFT convolution,
   * perform an exponentiated‑gradient update
     `a ← a·exp(‑η·∇)` and renormalise onto the simplex.
   The step size `η` decays like `1/√t`.  This is a mirror‑descent
   method on the probability simplex that smoothly interpolates
   between a full‑gradient step (large `T`) and the classic sub‑gradient
   step (tiny `T`).
3. **Escaping stagnation** – after a long period without improvement
   we add a small Gaussian jitter around the current best point.
4. **Polish** – a short guided hill‑climb (mass‑transfer moves) is run
   until the time budget expires.
5. **Return** – the best list of heights found.

The whole routine respects the 1000 s budget, prints intermediate
progress, and returns a Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    # eliminate tiny negatives that appear due to rounding
    np.maximum(conv, 0, out=conv)
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full autocorrelation a*a (length 2·n‑1)."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _grad_from_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Given a weight vector `w` (length 2·n‑1, sum = 1) compute

        g_i = Σ_k w_k · ∂(a*a)_k / ∂a_i  =  2·Σ_k w_k·a_{k‑i}

    for all i = 0…n‑1.
    This is performed by a single FFT convolution.
    """
    a_rev = a[::-1]                 # reversed a
    # convolution of w (len 2n‑1) with a_rev (len n)
    conv = _conv_fft(w, a_rev)      # length = 3n‑2
    n = a.shape[0]
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g

            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----------------------
        g = _grad_from_weights(w, a)

        # ----- step size (decreasing) ----------------------------------
        eta = step0 / np.sqrt(it)

        # ----- exponentiated‑gradient update ----------------------------
        a_new = a * np.exp(-eta * g)
        eps = 1e-15
        a_new = np.maximum(a_new, eps)          # avoid exact zeros → underflow
        a_new = a_new / a_new.sum()

        # ----- stagnation handling --------------------------------------
        if cur_obj - best_val > -1e-14:
            no_improve += 1
        else:
            no_improve = 0

        if no_improve >= 300:
            # add a small random jitter around the best point to escape a flat region
            jitter = rng.normal(scale=0.005, size=n)
            a = best_a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
            continue

        a = a_new

    # ------------------------------------------------------------------
    # 3️⃣  Guided hill‑climbing refinement (optional but cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # ------------------------------------------------------------------
    # 4️⃣  Final reporting
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")

    return best_a.tolist()
