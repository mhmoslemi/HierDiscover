"""
Propose a step‑function (non‑negative vector) minimizing
    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Fast parametric sweep** – Randomly generate many symmetric
   “U‑shaped’’ and beta‑distributed candidates (few hundred
   milliseconds each) and keep the best one.

2. **Adam optimisation** – Smooth‑max (log‑sum‑exp) approximation of
   the maximum convolution term.  The variables are the logits of a
   soft‑max, guaranteeing Σa = 1.  Adam with a decreasing learning‑rate
   and a temperature schedule refines the best candidate.

3. **Targeted hill‑climbing** – Repeatedly move a small amount of mass
   from indices that contribute the most to the current maximal
   convolution entry toward indices that contribute the least.
   Moves are accepted only when they strictly improve the exact
   objective.  After long stagnation a small random perturbation is
   applied.

The final vector is symmetrised (a ↦ (a+a[::-1])/2) and returned.
All operations respect the 1000 s budget; heavy calls to the given
`evaluate_sequence` are made only once at the end.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective under Σa = 1
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Phase‑0 – random parametric sweep (U‑shape & beta‑shape)
# ----------------------------------------------------------------------
def _parametric_sweep(n: int, deadline: float, best_a, best_val):
    """Search a large number of symmetric parametric families."""
    x = np.arange(n, dtype=np.float64)
    eps = 1e-12
    while time.time() < deadline:
        family = np.random.choice(["u", "beta"])
        if family == "u":
            # a_i ∝ ((i+½)(n‑i‑½)+δ)^(-γ)
            gamma = 10 ** np.random.uniform(-1.0, 0.7)   # ~0.1 … 5
            delta = np.random.uniform(0.0, 2.0)          # small offset
            denom = (x + 0.5) * (n - x - 0.5) + delta
            a = np.power(denom + eps, -gamma)
        else:
            # beta‑shape: a_i ∝ (i+½)^{α‑1} (n‑i‑½)^{β‑1}
            alpha = 10 ** np.random.uniform(-2.0, 0.7)   # ~0.01 … 5
            beta = 10 ** np.random.uniform(-2.0, 0.7)
            a = np.power(x + 0.5, alpha - 1.0) * np.power(n - x - 0.5, beta - 1.0)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        # enforce symmetry (helps the later stages)
        a = (a + a[::-1]) * 0.5
        val = _objective(a)
        if val < best_val:
            best_a, best_val = a.copy(), val
            print(f"[param] new best = {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑A – Adam optimisation on a smooth‑max approximation
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp smooth approximation of max(conv(a,a))."""
    # logits for the soft‑max (ensure Σa = 1)
    w = np.log(a0 + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.08, 0.001
    eps_start, eps_end = 5e-2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 25000

    while True:
        now = time.time()
        if now >= deadline:
            break

        # a from logits
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule for learning‑rate & temperature
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)          # ≤ 1
        p = weights / weights.sum()        # soft‑max over all 2n‑1 positions

        # gradient of smoothed max with respect to a
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)
        segment = conv_ab[n - 1: 2 * n - 1]        # n entries
        grad_a = 4.0 * n * segment[::-1]           # factor 2 (max) * 2 (conv) * n

        # back‑prop through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter to avoid stagnation
        if np.random.rand() < 0.004:
            w += np.random.normal(scale=0.01, size=n)

        if no_improve > IMPROVE_LIMIT:
            # restart from a fresh random point
            print("[Adam] restart after stagnation")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑B – targeted hill‑climbing
# ----------------------------------------------------------------------
def _hillclimb_improved(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """Local search that transfers a small amount of mass between indices."""
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.60, 0.02
    IMPROVE_LIMIT = 180_000
    no_improve = 0
    iteration = 0
    eps = 1e-12

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progressive bound on transferred fraction
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # current maximal convolution entry
        k_max = int(np.argmax(b))

        # contributions of each a[i] to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # --- source index (biased by contribution) ---
        if total_contrib > 0.0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # --- target index (bias towards low contribution) ---
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0                     # avoid self‑move
        w_sum = target_weights.sum()
        if w_sum > 0.0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # uniform fallback (exclude i)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to transfer
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (vectorised, O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic terms
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # accept only strictly improving moves
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[HillImp] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling
        if no_improve > IMPROVE_LIMIT:
            # perturb around the current best and restart the local search
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillImp] stagnation – random perturbation at it={iteration}")

        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    random_state = np.random.RandomState(seed)

    start_time = time.time()
    # keep a short safety margin for the final evaluation
    deadline = start_time + budget_s - 0.5

    # problem size – a few thousand entries give a good trade‑off
    n = 8000

    # ------------------------------------------------------------------
    # initialise a – try to reuse a previously discovered solution
    # ------------------------------------------------------------------
    best_a = None
    best_val = np.inf
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.size != n:
            old_x = np.linspace(0.0, 1.0, old.size)
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        # enforce symmetry (helps later stages)
        a0 = (a0 + a0[::-1]) * 0.5
        best_a, best_val = a0.copy(), _objective(a0)
        print(f"[init] interpolated previous solution, value={best_val:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (randomised families)
    # ------------------------------------------------------------------
    if best_a is None:
        # start from a random uniform vector if nothing else is available
        a0 = np.random.rand(n)
        a0 /= a0.sum()
        best_a, best_val = a0.copy(), _objective(a0)
        print(f"[init] random start, value={best_val:.6f}")

    param_deadline = start_time + 0.15 * budget_s
    best_a, best_val = _parametric_sweep(n, param_deadline, best_a, best_val)

    # ------------------------------------------------------------------
    # Phase A – Adam optimisation on smooth max‑convolution
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + 0.45 * budget_s
    best_a, best_val = _adam_phase(best_a, n,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=1.503)

    # ------------------------------------------------------------------
    # Phase B – targeted hill‑climbing
    # ------------------------------------------------------------------
    best_a, best_val = _hillclimb_improved(best_a, n,
                                           start=phase_a_deadline,
                                           deadline=deadline,
                                           best_a=best_a,
                                           best_val=best_val,
                                           target_val=1.503)

    # final symmetrisation (keeps sum = 1)
    best_a = (best_a + best_a[::-1]) * 0.5

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
