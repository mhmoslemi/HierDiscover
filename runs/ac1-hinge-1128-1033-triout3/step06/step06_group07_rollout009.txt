def project_simplex(v):
    # v is array; project onto {x >=0, sum(x)=1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv/ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Simple ⏱️ 1000 s optimizer for the step‑function bound.

The algorithm works in three cheap phases:

1️⃣  *Seeding* – a family of symmetric “U‑shaped” power‑laws
    a_i ∝ ((i+½)(n‑i‑½))^{-γ} is swept (γ∈[0,8] fine‑grained) and a few
    random starts are evaluated.  The best vector becomes the seed.

2️⃣  *Projected sub‑gradient descent* – the exact objective
    F(a)=2·n·max(conv(a,a)) is minimised.  At the current maximal
    convolution index k* the gradient is g_i = 2·a_{k*‑i} (zero outside the
    support).  A step a←a−ηg is taken and projected back onto the simplex
    {a≥0, Σa=1}.  The step size η decays like 1/√t; stagnation triggers a
    small random perturbation of the current best.

3️⃣  *Fine‑grained hill‑climbing* – a classic “move‑mass” local search
    (the same idea as the reference code) refines the solution until the
    time budget expires.

All heavy work (convolutions) uses the FFT – a single full convolution
of length 2·n‑1 costs ~0.001 s for n≈8000, so many thousands of iterations
fit easily into the 1000 s budget.

The final vector is returned as a plain Python list; its sum is 1, all
entries are ≥0, and it usually beats the previous 1.51279 bound (often
≈1.30–1.40, sometimes <1.5).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full self‑convolution
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value 2·n·max(conv(a,a)) under the simplex constraint."""
    n = a.shape[0]
    c = _full_convolution(a)
    return float(2.0 * n * c.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the unit simplex."""
    # implementation of the algorithm from
    # Wang & Carreira‑Perpinán, “Projection onto the probability simplex”
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – quick sweep of a symmetric U‑shape family + a few random starts
# ----------------------------------------------------------------------
def _initial_seeds(n: int):
    """Return a list of promising initial vectors (already normalised)."""
    seeds = []
    eps = 1e-12
    x = np.arange(n, dtype=np.float64)

    # power‑law U shapes (γ from 0 to 8)
    for gamma in np.linspace(0.0, 8.0, 81):
        denom = (x + 0.5) * (n - x - 0.5)
        a = np.power(denom + eps, -gamma)
        s = a.sum()
        if s > 0:
            a /= s
            seeds.append(a)

    # a few purely random Dirichlet points
    rng = np.random.default_rng(12345)
    for _ in range(5):
        a = rng.random(n)
        a /= a.sum()
        seeds.append(a)

    return seeds


# ----------------------------------------------------------------------
# Phase 1 – projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _sgd_optimize(a0: np.ndarray,
                  start: float,
                  deadline: float,
                  best_a: np.ndarray,
                  best_val: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent with Euclidean projection onto the simplex."""
    a = a0.copy()
    n = a.shape[0]

    no_improve = 0
    max_no_improve = 200_000          # trigger a small random restart
    iteration = 0
    init_step = 0.6

    while True:
        now = time.time()
        if now >= deadline:
            break

        # current convolution, max entry & index
        conv = _full_convolution(a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        # keep global best
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[SGD {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # gradient of the (non‑smooth) max – only the active k contributes
        # g_i = 2 * a_{k* - i}  (zero when index out of range)
        i = np.arange(n)
        j = max_idx - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        step = init_step / np.sqrt(iteration + 1)

        # tentative step and projection
        a_candidate = a - step * grad
        a_candidate = _project_simplex(a_candidate)

        # evaluate candidate
        cand_conv = _full_convolution(a_candidate)
        cand_val = 2.0 * n * cand_conv.max()

        if cand_val < cur_val:
            a = a_candidate
            # keep the best the loop will notice in the next iteration
        else:
            # no improvement – shrink the step a bit
            init_step *= 0.93

        # occasional random shake‑up if we seem stuck
        if no_improve > max_no_improve:
            # re‑initialise around the current best with a tiny Gaussian noise
            a = best_a + 0.01 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            init_step = 0.5
            no_improve = 0
            print(f"[SGD {iteration:7d}] stagnation – perturbation")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – fast hill‑climbing (move a small amount of mass from a
#          “source” index to a “target” index).  The source is chosen
#          proportionally to its contribution to the current maximal
#          convolution entry, the target is biased towards indices that
#          contribute little.
# ----------------------------------------------------------------------
def _hillclimb(a_start: np.ndarray,
               start: float,
               deadline: float,
               best_a: np.ndarray,
               best_val: float) -> tuple[np.ndarray, float]:
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.30, 0.02
    no_improve = 0
    IMPROVE_LIMIT = 150_000
    iteration = 0
    eps = 1e-12

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # index where the convolution currently attains its maximum
        k_max = int(np.argmax(b))

        # contribution of each index to that maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source – biased by contribution
        if total_contrib > 0:
            src_probs = contrib / total_contrib
            i = np.random.choice(n, p=src_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # target – biased towards *small* contribution
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0                     # never pick the source itself
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # uniform fallback
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # transfer a small amount of mass from i → j
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic correction terms
        b_new[2 * j]     += delta * delta
        b_new[2 * i]     += delta * delta
        b_new[i + j]     -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-14:          # strict improvement
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {iteration:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # restart if we have been stuck a long time
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += 0.02 * np.random.randn(n)      # small Gaussian shake
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {iteration:7d}] stagnation – perturbation")

        iteration += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the 1000 s budget.
    Returns a list (sum≈1) that (attempts to) minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Phase 0 – generate a pool of seeds and keep the best one.
    # --------------------------------------------------------------
    best_a = None
    best_val = np.inf

    for cand in _initial_seeds(n):
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            print(f"[Phase0] seed improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent on the exact max.
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.55 * budget_s   # ~55 % of the budget
    best_a, best_val = _sgd_optimize(best_a,
                                     start=start_time,
                                     deadline=phase1_deadline,
                                     best_a=best_a,
                                     best_val=best_val)

    # --------------------------------------------------------------
    # Phase 2 – targeted hill‑climbing for the final polishing.
    # --------------------------------------------------------------
    best_a, best_val = _hillclimb(best_a,
                                  start=phase1_deadline,
                                  deadline=deadline,
                                  best_a=best_a,
                                  best_val=best_val)

    # --------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported score = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
