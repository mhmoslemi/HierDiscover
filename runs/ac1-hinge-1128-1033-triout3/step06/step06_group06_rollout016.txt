```python
"""
Search for a step function that minimizes `evaluate_sequence`.

The algorithm consists of:

1. Random sampling of Dirichlet-distributed vectors (fast FFT evaluation)
   to obtain a good initial point.
2. A fast hill‑climber that moves a small amount of mass from indices
   that contribute most to the current maximal convolution entry to
   indices that contribute little.  The convolution is updated incrementally
   (O(n)) so that millions of moves can be performed.
3. A short simulated‑annealing phase that occasionally accepts
   worsening moves to escape plateaus.

All phases respect the overall time budget (default 1000 s).
The final candidate is returned as a list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT) – length 2·n‑1 linear convolution
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv

# ----------------------------------------------------------------------
# Exact objective (assuming Σ a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())

# ----------------------------------------------------------------------
# Random initialisation – Dirichlet samples with optional sorting
# ----------------------------------------------------------------------
def _random_initial_samples(n: int,
                            max_samples: int,
                            start: float,
                            deadline: float) -> tuple[np.ndarray, float]:
    """Generate up to `max_samples` random probability vectors and keep the best."""
    best_a = None
    best_val = np.inf
    it = 0
    while it < max_samples and time.time() < deadline:
        # Dirichlet(1) == normalized exponentials
        raw = np.random.rand(n)
        a = raw / raw.sum()
        # optional ordering: decreasing (helps to lower max)
        a = np.sort(a)[::-1]          # monotone decreasing (empirically better)
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
        it += 1
    return best_a, best_val

# ----------------------------------------------------------------------
# Targeted hill‑climber (exact objective) – same idea as earlier code
# ----------------------------------------------------------------------
def _hillclimb(a_start: np.ndarray,
               start: float,
               deadline: float,
               best_val: float,
               best_a: np.ndarray) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass from heavy contributors to light ones."""
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = _objective(a)

    # schedule for the fraction of mass moved in a step
    max_frac_start, max_frac_end = 0.5, 0.001
    no_improve = 0
    IMPROVE_LIMIT = 150_000
    it = 0
    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate lag with maximal convolution value
        k_max = int(np.argmax(b))

        # contributions of each index to b[k_max]
        # for each i, term a[i] * a[k_max - i] (if index inside range)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # pick a source index proportional to its contribution
        if total > 0:
            probs = contrib / total
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – try to choose a place that contributes little
        # to the *same* lag (inverse weighting)
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        # Inverse weighting – avoid places already large
        w = np.where(rev_contrib > 0, 1.0 / rev_contrib, 0.0)
        w[i] = 0.0
        w_sum = w.sum()
        if w_sum > 0 and np.random.rand() < 0.7:
            probs_j = w / w_sum
            j = np.random.choice(n, p=probs_j)
        else:
            # uniform fallback
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # move a fraction of the source mass
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution (O(n))
        b_new = b.copy()
        # a[i] participates in all lags i..i+n-1; removing delta reduces those entries
        b_new[i:i + n] -= 2.0 * delta * a
        # a[j] receives delta
        b_new[j:j + n] += 2.0 * delta * a
        # self‑terms at i and j
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        # cross term for lag i + j (appears twice, one positive one negative)
        b_new[i + j] -= 2.0 * delta * delta

        new_val = _objective(a)  # recompute quick? Actually we need using b_new.
        new_val = float(2.0 * a.shape[0] * b_new.max())

        # accept only improvements (strict) – this keeps algorithm stable
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= 1.5:   # early exit if already very good
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb and restart from current best
            a = best_a.copy()
            a += np.random.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = _objective(a)
            no_improve = 0
            print(f"[Hill {it:7d}] perturb after stagnation")
        it += 1

    return best_a, best_val

# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (optional)
# ----------------------------------------------------------------------
def _anneal(a_start: np.ndarray,
            start: float,
            deadline: float,
            best_val: float,
            best_a: np.ndarray) -> tuple[np.ndarray, float]:
    """Metropolis search that sometimes accepts worsening moves."""
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = _objective(a)

    T_start, T_end = 5e-4, 5e-9
    no_improve = 0
    IMPROVE_LIMIT = 100_000
    it = 0
    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        k_max = int(np.argmax(b))

        # pick source biased by contribution
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # pick a random target different from i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # tiny transfer (≤ 0.5 % of source mass)
        delta = a[i] * np.random.rand() * 0.005
        if delta <= 0.0:
            continue

        # incremental update of convolution
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max())

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anne {it:7d}] new best = {best_val:.6f}")
                if best_val <= 1.5:
                    break
        else:
            no_improve += 1

"""
Search for a non‑negative step function that minimizes `evaluate_sequence`.

The algorithm proceeds in three stages that respect the overall
time budget (default 1000 s):

1. **Random sampling** – generate many Dirichlet‑distributed vectors,
   evaluate them with a fast FFT based objective and keep the best.
   This gives a diverse starting point that usually beats the simple
   parametric candidates used before.

2. **Targeted hill‑climbing** – repeatedly move a small amount of mass
   from an index that contributes heavily to the current maximal
   convolution entry to an index that contributes little.  The full
   convolution is updated incrementally (O(n)) so that millions of
   moves can be performed quickly.

3. **Simulated‑annealing refinement** – a short Metropolis style phase
   that occasionally accepts worsening moves, helping to escape
   plateaus that the greedy hill‑climber can get stuck in.

All auxiliary functions are top‑level, no closures are used and the
final sequence is normalised to unit sum (the evaluator rescales
automatically).  The routine returns a list of floats ready for the
external `evaluate_sequence` wrapper.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    # next power of two for an efficient FFT
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector normalised to sum = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Phase 0 – cheap random search
# ----------------------------------------------------------------------
def _random_initial_samples(n: int,
                            max_samples: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """
    Generate up to `max_samples` random probability vectors (Dirichlet(1))
    and keep the one with the smallest objective.
    """
    best_a = None
    best_val = np.inf
    it = 0
    while it < max_samples and time.time() < deadline:
        # Dirichlet(1) = normalised exponentials; np.random.rand is enough.
        raw = np.random.rand(n)
        a = raw / raw.sum()               # Σa = 1
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 1 – targeted hill‑climber (exact objective)
# ----------------------------------------------------------------------
def _hillclimb(a_start: np.ndarray,
               start: float,
               deadline: float,
               best_val: float,
               best_a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Move tiny amounts of mass from indices that heavily contribute to the
    current maximal convolution entry to indices that contribute little.
    The convolution is updated incrementally, so the loop is very fast.
    """
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = _objective(a)

    # Fraction of source mass moved in a single step – linearly decays.
    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # lag where the convolution attains its maximum
        k_max = int(np.argmax(b))

        # contribution of each index to this maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()

        # choose a source index proportional to its contribution
        if total > 0:
            probs = contrib / total
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # choose a target index – preferably one that contributes little
        rev_idx = k_max - np.arange(n)
        rev_mask = (rev_idx >= 0) & (rev_idx < n)
        rev_contrib = np.zeros_like(a)
        rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
        w = np.where(rev_contrib > 0, 1.0 / rev_contrib, 0.0)
        w[i] = 0.0
        w_sum = w.sum()
        if w_sum > 0 and np.random.rand() < 0.7:
            probs_j = w / w_sum
            j = np.random.choice(n, p=probs_j)
        else:
            # uniform fallback
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max())

        # strict improvement acceptance
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                # early exit if we are already very close to the theoretical limit
                if best_val <= 1.2:
                    break
        else:
            no_improve += 1

        # restart from the best seen after prolonged stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.01, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = _objective(a)
            no_improve = 0
            print(f"[Hill {it:7d}] perturb after stagnation")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – short simulated‑annealing refinement
# ----------------------------------------------------------------------
def _anneal(a_start: np.ndarray,
            start: float,
            deadline: float,
            best_val: float,
            best_a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Metropolis style search that occasionally accepts worsening moves.
    The temperature is linearly decayed; the move set is the same as in
    the hill‑climber but with a much smaller transfer fraction.
    """
    a = a_start.copy()
    n = a.shape[0]
    b = _full_convolution(a)
    cur_val = _objective(a)

    T_start, T_end = 5e-4, 5e-9
    IMPROVE_LIMIT = 100_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        k_max = int(np.argmax(b))

        # source index – weighted by contribution to the maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total = contrib.sum()
        if total > 0:
            probs = contrib / total
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target – uniform
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # tiny transfer (≤ 0.5 % of source mass)
        delta = a[i] * np.random.rand() * 0.005
        if delta <= 0.0:
            continue

        # incremental convolution update
        b_new = b.copy()
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[j:j + n] += 2.0 * delta * a
        b_new[2 * i] += delta * delta
        b_new[2 * j] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = float(2.0 * n * b_new.max())

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anne {it:7d}] new best = {best_val:.6f}")
                if best_val <= 1.2:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.003, size=n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = _objective(a)
            no_improve = 0
            print(f"[Anne {it:7d}] perturb after stagnation")
        it += 1

    return best_a, best_val

    a_best, best_val = _random_initial_samples(
        n=n,
        max_samples=30_000,          # upper bound; the loop respects the deadline
        deadline=rnd_deadline,
    )
    if a_best is None:
        # Fallback to the uniform distribution if the time ran out immediately.
        a_best = np.full(n, 1.0 / n)
        best_val = _objective(a_best)

    print(f"[Init] best from random search = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – targeted hill‑climbing on the exact objective
    # ------------------------------------------------------------------
    hill_deadline = start_time + (deadline - start_time) * 0.70
    a_best, best_val = _hillclimb(
        a_start=a_best,
        start=start_time,
        deadline=hill_deadline,
        best_val=best_val,
        best_a=a_best,
    )
    print(f"[Hill] best after hill‑climb = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – brief simulated‑annealing refinement
    # ------------------------------------------------------------------
    a_best, best_val = _anneal(
        a_start=a_best,
        start=hill_deadline,
        deadline=deadline,
        best_val=best_val,
        best_a=a_best,
    )
    print(f"[Final] best evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check using the supplied secure wrapper
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_best.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Normalise just for neatness (the evaluator does its own rescaling)
    a_best = np.maximum(a_best, 0.0)
    if a_best.sum() > 0:
        a_best /= a_best.sum()
    return a_best.tolist()
