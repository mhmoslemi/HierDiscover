def smooth_phase(a, n, deadline, best_a, best_val):
    start_time = time.time()
    iteration = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; epsilon=1e-8
    # base learning rate for gradient magnitude scaling, maybe 0.01
    while time.time() < deadline:
        b = _full_convolution(a)
        max_b = b.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth {iteration}] new best = {best_val:.6f}")
        # alpha schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        alpha = 0.5 + 30.0 * progress  # linear increase
        # compute softmax weights
        w = np.exp(alpha * b)
        w_sum = w.sum()
        softmax = w / w_sum
        # compute gradient: 2 * cross_corr(a, softmax)
        # compute conv of softmax with reversed a, then reverse slice
        conv_res = np.convolve(softmax, a[::-1])
        # cross = conv_res[n-1::-1][:n]   # reversed slice
        cross = conv_res[n-1::-1][:n]
        grad = 2.0 * cross

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        lr = 0.05 * (1 - progress) + 0.001 * progress  # decaying lr
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a = _project_simplex(a_new)
    return best_a, best_val

def _simulated_hillclimb(a, n, deadline, best_a, best_val):
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    start = time.time()
    it = 0
    no_improve = 0
    # Temperature schedule: start T = initial_temp = cur_val * 0.005 maybe (5% of value)
    init_temp = cur_val * 0.01
    T = init_temp

    while time.time() < deadline:
        # compute contributions
        k = max_idx
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = np.zeros_like(a)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-15:
            # no mass, choose another source
            source_candidates = np.where(a > 1e-15)[0]
            if source_candidates.size == 0:
                break

"""
propose_candidate()
===================

A compact, time‑bounded optimiser for the “step‑function” evaluation
problem.  The goal is to minimise

    E(a) = 2·n·max(a∗a) / (sum(a)²)

subject to a_i ≥ 0.  The optimiser works in four stages, each of which
fits comfortably into the 1000 s wall‑time limit:

1. **Sub‑gradient descent** on the exact max‑convolution (as in the
   reference implementation) – fast, coarse reduction.
2. **Smooth‑max descent** – replaces the non‑differentiable max by a
   log‑sum‑exp surrogate.  The gradient is obtained analytically via a
   single FFT‑based convolution; an Adam optimiser accelerates
   convergence.
3. **L²‑flattening** – minimises Σ(b–b̄)² (the variance of the
   convolution) using the same FFT trick as the original code.
4. **Simulated‑annealing hill‑climb** – a greedy move of a small mass
   from the index that contributes most to the current worst
   convolution value to an index that contributes least.  The move is
   accepted either when it improves the exact objective or with a
   probability that decays with a temperature schedule, allowing the
   search to escape shallow plateaux.  Occasional Gaussian perturbations
   re‑seed the search.

All intermediate points are projected onto the probability simplex
(Σa = 1, a_i ≥ 0), which leaves the evaluation invariant.  The best
vector found is returned as a plain Python list; its quality is
reported with `evaluate_sequence` before returning.

The implementation relies only on NumPy and the supplied `evaluate_sequence`
function, respects the 1000 s limit, and prints occasional progress
messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Length = 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.
    Length = len(x)+len(y)‑1.
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # fall back to the uniform vector
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    # re‑normalise to compensate for numerical loss
    ws = w.sum()
    if ws > 0:
        w /= ws
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase A – Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_phase(a: np.ndarray,
                       n: int,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    start = time.time()
    max_step = 0.10   # initial step size
    min_step = 1e-4   # final step size
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        k = int(np.argmax(b))          # index of the current worst shift
        cur_max = b[k]
        cur_val = 2.0 * n * cur_max    # exact objective (Σa = 1)

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Subgrad {it:7d}] new best = {best_val:.6f}")

        # gradient of max(b) w.r.t. a (sub‑gradient at the worst index)
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = a[k - idx]          # derivative a_{k‑i}
        if k % 2 == 0:                  # middle term a_{k/2}² contributes twice
            mid = k // 2
            if 0 <= mid < n:
                grad[mid] = 2.0 * a[mid]

        # linear step‑size schedule
        progress = (time.time() - start) / (deadline - start)
        step = max_step * (1.0 - progress) + min_step * progress

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Smooth‑max descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_max_phase(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> tuple[np.ndarray, float]:
    """
    Minimises the smoothed objective
        f(a) = (1/α)·log Σ exp(α·(a∗a))
    whose gradient tends to the sub‑gradient of max(a∗a) as α → ∞.
    An Adam optimiser is used for fast convergence.
    """
    start = time.time()
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth {it:7d}] new best = {best_val:.6f}")

        # linearly raise the smoothing parameter α
        progress = (time.time() - start) / (deadline - start)
        alpha = 0.5 + 30.0 * progress          # α ∈ [0.5, 30.5]

        # stable soft‑max weights for the surrogate
        shift = b.max()
        exp_weights = np.exp(alpha * (b - shift))
        w = exp_weights / exp_weights.sum()    # length 2n‑1, Σw = 1

        # gradient = 2 * cross‑correlation(a, w)
        # cross‑correlation can be obtained by convolving w with the reversed a
        conv_res = _conv_fft(w, a[::-1])       # length = 3n‑2
        cross = conv_res[n - 1::-1][:n]        # reverse the first n entries
        grad = 2.0 * cross

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (fast at the start, then decays)
        lr = 0.10 * (1.0 - progress) + 0.001 * progress

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a_new)
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – L²‑flattening (variance of the convolution)
# ----------------------------------------------------------------------
def _l2_flatten_phase(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> tuple[np.ndarray, float]:
    """Minimises Σ (b−b̄)² to flatten the convolution."""
    start = time.time()
    base_step = 0.5
    min_step = 0.001
    avg_target = 1.0 / (2 * n - 1)    # mean of b when Σa = 1
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[L2 {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        r = b - avg_target
        conv1 = _conv_fft(r, a)          # length 3n‑2
        conv2 = _conv_fft(r, a[::-1])
        start_idx = n - 1
        end_idx = 2 * n - 1
        grad = 2.0 * (conv1[start_idx:end_idx] + conv2[start_idx:end_idx])

        # schedule
        progress = (time.time() - start) / (deadline - start)
        step = base_step * (1.0 - progress) + min_step * progress

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        if no_improve > IMPROVE_LIMIT:
            a = np.random.rand(n)
            a = _project_simplex(a)
            no_improve = 0
            print(f"[L2 {it}] restart after stagnation")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase D – Simulated‑annealing hill climbing on the exact objective
# ----------------------------------------------------------------------
def _simulated_hillclimb(a: np.ndarray,
                         n: int,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float) -> np.ndarray:
    """
    Greedy local search that moves a small amount of mass from the index
    that contributes most to the current worst convolution value to the
    index that contributes least.  Moves are accepted either if they
    improve the objective or with a probability exp(-(ΔE)/T), where T
    decays linearly.
    """
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0
    start = time.time()
    # initial temperature: a few per‑cent of the current value
    T0 = max(1e-6, cur_val * 0.01)

    while time.time() < deadline:
        # ---- contribution vector for the current worst index ----
        k = max_idx
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-15:
            # pick a random non‑empty source
            src_cand = np.where(a > 1e-15)[0]
            if src_cand.size == 0:
                break
            source = int(np.random.choice(src_cand))

        # move a fraction of the source mass (5 % – 30 %)
        delta = a[source] * (0.05 + 0.25 * np.random.rand())
        if delta <= 0.0:
            continue

        # ---- incremental update of the full convolution (O(n)) ----
        b_new = b.copy()
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # temperature (linear cooling)
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = T0 * (1.0 - frac) + 1e-8

        accept = False
        if new_val < cur_val - 1e-12:
            accept = True
        else:
            # Metropolis‑type acceptance
            if np.random.rand() < np.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_val = new_val
            cur_max = new_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[HillSA {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # occasional perturbation after long stagnation
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[HillSA {it}] stagnation – perturbing")
        it += 1
    return best_a

        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
