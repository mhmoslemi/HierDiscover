def _gradient_descent(a, cur_val, deadline):

    best_val = cur_val
    eta = 1e-6
    t = 0
    while time.time() < deadline:
        grad, max_val = _grad_exact(a)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0:
            break
        a_new /= s
        new_val = _eval_fast(a_new)
        if new_val < cur_val:
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
            # maybe increase step

        else:
            # revert not needed as we used a_new, keep a unchanged
            # reduce step
            eta *= 0.5
            if eta < 1e-12:

        # optional early exit

            # print progress
            print(f"[GD] iter {t}, cur_val {cur_val:.6f}, eta {eta:.2e}")
        if time.time() > deadline:
            break

    best_a = a.copy()
    best_val = cur_val
    eta = 1e-6
    step = 0
    while time.time() < deadline:
        grad, _ = _grad_exact(a)
        # propose step
        new_a = a - eta * grad
        # enforce non-negativity
        new_a = np.maximum(new_a, 0.0)
        # renorm to sum 1
        s = new_a.sum()
        if s <= 0:
            break
        new_a /= s
        # compute new objective
        new_val = _eval_fast(new_a)
        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            # maybe increase step size

        else:
            # reject progression and reduce step size

    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_val = conv.max()
    # choose random index among ties
    ks = np.flatnonzero(conv == max_val)
    k = ks[0]  # deterministic but fine

"""
Propose a candidate sequence for the step‑function optimisation problem.

The algorithm has three main phases:

1. **Initialisation** – loads a previously found good sequence (if
   available) and interpolates it to the chosen size; otherwise a random
   Dirichlet start is used.  The sequence is normalised to ℓ₁‑norm = 1.
2. **Exact sub‑gradient descent** – uses the analytical sub‑gradient of the
   true objective (no smoothing).  Each iteration performs a tiny
   projected gradient step, adapts the step‑size and keeps the best
   result seen so far.  This phase exploits the fact that moving mass
   away from the peak of the self‑convolution strictly reduces the
   objective.
3. **Permutation‑based refinement** – a short simulated‑annealing run
   (swaps only) followed by a cheap hill‑climbing mass‑transfer pass.
   Both stages run only on the remaining time budget.

All heavy work is performed with FFT‑based convolution, which makes the
search feasible even for several‑thousand‑element sequences.  The
function returns a plain list of non‑negative floats and prints the
final evaluation obtained from ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – FFT based convolution, fast evaluation, etc.
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT (length 2·n‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    # tiny negatives may appear because of rounding – clamp them
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Exact objective from ``evaluate_sequence`` but much faster."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simple re‑ordering heuristic – places large values at the ends.
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Large–small alternating order (large on both ends, small in centre)."""
    n = a.shape[0]
    s = np.sort(a)                # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[n - 1 - i]     # largest on the left
        out[n - 1 - i] = s[i]     # smallest on the right
    if n % 2:
        out[half] = s[half]
    return out


# ----------------------------------------------------------------------
# Exact sub‑gradient of the true objective.
# ----------------------------------------------------------------------
def _grad_exact(a: np.ndarray):
    """
    Returns (grad, max_conv) where ``grad`` is the sub‑gradient of
    ``2·n·max(conv)/sum(a)^2`` w.r.t. ``a``.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_val = conv.max()
    # pick the first (deterministic) index attaining the maximum
    k = int(np.flatnonzero(conv == max_val)[0])

    S = a.sum()
    # grad of max(conv) : 2 * a reversed and shifted by k
    i = np.arange(n)
    j = k - i
    valid = (j >= 0) & (j < n)
    grad_M = np.zeros_like(a)
    grad_M[valid] = 2.0 * a[j[valid]]

    # full gradient of the objective
    coeff = 2.0 * n / (S * S)
    const = (4.0 * n * max_val) / (S * S * S)
    grad = coeff * grad_M - const * np.ones_like(a)
    return grad, max_val


# ----------------------------------------------------------------------
# Exact gradient descent (projected onto the simplex).
# ----------------------------------------------------------------------
def _gradient_descent(a: np.ndarray, cur_val: float, deadline: float):
    """
    Sub‑gradient descent with adaptive step size.
    Returns the best sequence and its objective seen before ``deadline``.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = cur_val
    eta = 1e-5                     # initial step size
    it = 0
    while time.time() < deadline:
        grad, _ = _grad_exact(a)

        # projected gradient step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)          # enforce non‑negativity
        s = a_new.sum()
        if s == 0.0:    # shouldn't happen, but guard against division by zero
            break
        a_new /= s                               # renormalise to ℓ₁ = 1

        new_val = _eval_fast(a_new)

        if new_val < cur_val:
            # successful step – keep it and try a slightly larger step size
            a = a_new
            cur_val = new_val
            eta *= 1.02
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[GD] new best = {best_val:.6f} (iter {it})")
        else:
            # step not improving – shrink step size
            eta *= 0.5
            if eta < 1e-12:
                eta = 1e-12

        it += 1
        # occasional progress report
        if it % 20000 == 0:
            print(f"[GD] iter {it}, cur_val {cur_val:.6f}, eta {eta:.2e}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing on permutations (swap moves only).
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float):
    """
    Fast SA on the ordering of ``a`` (swaps preserve the ℓ₁‑norm).
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.02, 1e-6)   # initial temperature
    decay = 0.99997                # slower cooling for larger sequences

    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose swap
        a[i], a[j] = a[j], a[i]
        new_val = _eval_fast(a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            # revert swap
            a[i], a[j] = a[j], a[i]

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (mass transfers + swaps).
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float):
    """
    Simple local search: occasional swaps (20 % chance) else a small
    mass transfer between two random positions.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    cur_a = a.copy()
    while time.time() < deadline:
        if rng.random() < 0.2:               # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                 # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_val = _eval_fast(new_a)
        if new_val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = new_val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Sequence initialisation (load previous or random Dirichlet).
# ----------------------------------------------------------------------
def _initial_sequence(n: int, rng: np.random.Generator):
    """Return a non‑negative ℓ₁‑normalised sequence of length ``n``."""
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required size
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a = np.interp(new_x, old_x, old)
            print("[init] interpolated previous construction")
        else:
            a = old.copy()
            print("[init] loaded previous construction")
    except Exception:
        # fallback: random Dirichlet (Gamma(1) i.i.d.)
        a = rng.gamma(shape=1.0, size=n)
        print("[init] random Dirichlet start")
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Main entry point.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time
    budget expires.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # small safety margin
    print(f"[init] time budget = {budget_s}s, deadline @ {deadline:.2f}")

    # problem size – 6000 gives a good trade‑off between flexibility
    # and per‑iteration cost.
    n = 6000
    # ------------------------------------------------------------------
    # Initialise a candidate (try to reuse a previously good construction).
    # ------------------------------------------------------------------
    a = _initial_sequence(n, rng)

    # Very cheap initial shaping – anti‑symmetric ordering.
    a = _anti_symmetric_reorder(a)

    cur_val = _eval_fast(a)
    best_a = a.copy()
    best_val = cur_val
    print(f"[init] initial evaluation = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – exact sub‑gradient descent.
    # ------------------------------------------------------------------
    gd_deadline = deadline - 120.0   # keep ~2 min for later phases
    if time.time() < gd_deadline:
        print("[gd] starting exact sub‑gradient descent")
        best_a, best_val = _gradient_descent(best_a, best_val, gd_deadline)

    # ------------------------------------------------------------------
    # Phase 2 – permutation simulated annealing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        print("[sa] starting simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val,
                                                deadline - 60.0)

    # ------------------------------------------------------------------
    # Phase 3 – final cheap hill‑climbing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing phase")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report the exact score using the official evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:            # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list.
    return best_a.tolist()
