"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises the provided ``evaluate_sequence`` objective.

The search proceeds in three stages:

1.  **Exploratory sampling** – a cheap parametric sweep (uniform,
    linear, quadratic, sine, exponential) followed by a short random
    “Dirichlet/β‑distribution” sampling loop.  The best vector found so
    far is kept.

2.  **Smooth‑max Adam optimisation** – the current best vector is
    refined on a smooth (log‑sum‑exp) approximation of the
    ``max(conv(a,a))`` objective.  The optimisation works on soft‑max
    logits, stays on the probability simplex (Σa = 1) and uses a
    learning‑rate / temperature schedule.

3.  **Exact local search** – a targeted hill‑climber that moves tiny
    amounts of mass from indices that contribute most to the current
    maximal convolution entry to indices that contribute little, followed
    by a short simulated‑annealing phase to escape plateaus.

All stages respect the global time budget (default 1000 s) and the
candidate is finally evaluated with the secure ``evaluate_sequence``.
The algorithm is deliberately stochastic and performs many cheap random
restarts, which helps to escape the local minima that limited the
previous implementation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective used during the search ( Σa = 1 )
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Parametric families (used for the initial sweep and for random sampling)
# ----------------------------------------------------------------------
def _shape_uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def _shape_linear_decreasing(n: int) -> np.ndarray:
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    return a


def _shape_quadratic_decreasing(n: int) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=float) ** 2)
    a /= a.sum()
    return a


def _shape_sine_halfwave(n: int) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _shape_exponential(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=float)
    a /= a.sum()
    return a


def _shape_power(n: int, p: float) -> np.ndarray:
    # decreasing power shape (p > 0)
    a = (np.arange(n, 0, -1, dtype=float) ** p)
    a /= a.sum()
    return a


def _shape_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    # Beta distribution on the discrete grid
    x = (np.arange(1, n + 1) / (n + 1))
    a = (x ** alpha) * ((1.0 - x) ** beta)
    a /= a.sum()
    return a


def _initial_candidate(n: int):
    """Return a promising initial vector from a small deterministic sweep."""
    candidates = [
        _shape_uniform(n),
        _shape_linear_decreasing(n),
        _shape_quadratic_decreasing(n),
        _shape_sine_halfwave(n),
    ]
    # a few exponential rates
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.995, 0.99, 0.98, 0.95):
        candidates.append(_shape_exponential(n, r))

    best_a = None
    best_val = float('inf')
    for a in candidates:
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase A – smooth‑max Adam optimisation on the simplex
# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                start: float,
                deadline: float,
                best_a: np.ndarray,
                best_val: float,
                target_val: float = 1.5) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth approximation of max(conv(a,a))."""
    w = np.log(a0 + 1e-12)                 # soft‑max logits (Σa = 1)

    beta1, beta2 = 0.90, 0.999
    lr_start, lr_end = 0.09, 0.001
    eps_start, eps_end = 5e-2, 1e-12

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    t = 0
    no_improve = 0
    IMPROVE_LIMIT = 20000

    it = 0
    while True:
        now = time.time()
        if now >= deadline:
            break

        # soft‑max → a on the simplex
        w_shift = w - np.max(w)
        a = np.exp(w_shift)
        a /= a.sum()

        cur_val = _objective_value(a)

        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            no_improve = 0
            print(f"[Adam  {it:7d}] new best = {best_val:.6f}")
            if best_val <= target_val:
                break
        else:
            no_improve += 1

        # schedule
        progress = (now - start) / (deadline - start)
        eps = eps_start * (eps_end / eps_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        # smooth max via log‑sum‑exp
        b_full = _full_convolution(a)
        max_b = b_full.max()
        shifted = (b_full - max_b) / eps
        weights = np.exp(shifted)
        p = weights / weights.sum()          # soft‑max over convolution positions

        # gradient of smooth objective w.r.t. a :
        # d/d a  (2 n Σ_k p_k (a*a)_k) = 4 n conv(a, rev(p))
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)         # length 3n‑2
        segment = conv_ab[n - 1: 2 * n - 1]   # n entries that correspond to ∂/∂a
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        w -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional random jitter to escape flat regions
        if np.random.rand() < 0.005:
            w += np.random.normal(scale=0.01, size=n)

        if no_improve > IMPROVE_LIMIT:
            # restart from fresh random logits
            print(f"[Adam  {it:7d}] restart after stagnation")
            w = np.random.randn(n) * 0.1
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.5) -> np.ndarray:
    """Exact local search that moves a small amount of mass between indices."""
    a = a_start.copy()
    b = _full_convolution(a)                 # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.5, 0.001
    IMPROVE_LIMIT = 150_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        k_max = int(np.argmax(b))

        # contributions of each index to the currently maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # source index – bias towards large contributions
        if total_contrib > 0 and np.random.rand() < 0.9:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # target index – sometimes favour positions with little contribution
        if np.random.rand() < 0.35:
            rev_idx = k_max - np.arange(n)
            rev_mask = (rev_idx >= 0) & (rev_idx < n)
            rev_contrib = np.zeros(n)
            rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
            w = 1.0 / (rev_contrib + 1e-12)
            w[i] = 0.0
            w_sum = w.sum()
            if w_sum > 0:
                probs_j = w / w_sum
                j = np.random.choice(n, p=probs_j)
            else:
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()

        if new_val < cur_val - 1e-14:          # accept only improving
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Hill  {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb and restart from the best seen so far
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill  {it:7d}] perturb after stagnation")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Phase C – short simulated‑annealing refinement
# ----------------------------------------------------------------------
def _annealing_phase(a_start: np.ndarray,
                     n: int,
                     start: float,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float,
                     target_val: float = 1.5) -> tuple[np.ndarray, float]:
    """Metropolis‑style search that occasionally accepts worsening moves."""
    a = a_start.copy()
    b = _full_convolution(a)
    cur_val = 2.0 * n * b.max()

    T_start, T_end = 5e-4, 5e-8
    IMPROVE_LIMIT = 80_000
    no_improve = 0
    it = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        progress = (now - start) / (deadline - start)
        T = T_start * (1.0 - progress) + T_end * progress

        k_max = int(np.argmax(b))

        # source – bias towards contributing indices
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()
        if total_contrib > 0 and np.random.rand() < 0.85:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # uniform target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # tiny transfer
        delta = a[i] * np.random.rand() * 0.003
        if delta <= 0.0:
            continue

        # update convolution
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * b_new.max()
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val, best_a = new_val, a.copy()
                print(f"[Anne {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # shake the solution a little
            noise = np.random.normal(scale=0.005, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Anne {it:7d}] perturb after stagnation")

        it += 1

    return best_a, best_val

# Random‑sampling helpers (cheap generators for the exploratory stage)
# ----------------------------------------------------------------------
def _rand_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex using a Gamma‑based Dirichlet."""
    # concentration parameter sampled logarithmically to favour both sparse and dense draws
    conc = 10 ** np.random.uniform(-2, 1)      # ~0.01 … 10
    g = np.random.gamma(conc, 1.0, size=n)
    a = g / g.sum()
    return a


def _rand_beta_shape(n: int) -> np.ndarray:
    """Random Beta‑shaped distribution on the grid."""
    alpha = np.random.exponential(scale=1.0) + 0.1
    beta  = np.random.exponential(scale=1.0) + 0.1
    return _shape_beta(n, alpha, beta)


def _rand_power_shape(n: int) -> np.ndarray:
    """Random power‑law decreasing shape."""
    p = np.random.uniform(0.1, 5.0)
    return _shape_power(n, p)


def _rand_exponential_shape(n: int) -> np.ndarray:
    """Random exponential decay."""
    r = np.random.uniform(0.94, 0.9999)
    return _shape_exponential(n, r)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function within the given time budget.
    Returns a list of floats (Σ = 1) that minimises ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 0.5

    # problem size – a few thousand entries gives a good trade‑off
    n = 8000

    # ------------------------------------------------------------------
    # 1️⃣  Initialise from cheap parametric families
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a0 = np.interp(new_x, old_x, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        best_a, best_val = a0.copy(), _objective_value(a0)
        print("[Init] started from previous construction")
    except Exception:
        best_a, best_val = _initial_candidate(n)
        print(f"[Init] parametric best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Exploratory random sampling (≈25 % of total budget)
    # ------------------------------------------------------------------
    exploration_deadline = start_time + (deadline - start_time) * 0.25
    sampling_methods = [
        _rand_dirichlet,
        _rand_beta_shape,
        _rand_power_shape,
        _rand_exponential_shape,
    ]

    while time.time() < exploration_deadline:
        # pick a generator at random
        gen = np.random.choice(sampling_methods)
        a = gen(n)

        # occasional tiny Gaussian noise to increase diversity
        if np.random.rand() < 0.3:
            a += np.random.normal(scale=1e-4, size=n)
            a = np.clip(a, 0.0, None)

        a /= a.sum()
        val = _objective_value(a)

        if val < best_val:
            best_a, best_val = a, val
            print(f"[Sample] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max Adam optimisation (≈30 % of budget)
    # ------------------------------------------------------------------
    phase_a_deadline = start_time + (deadline - start_time) * 0.55
    best_a, best_val = _adam_phase(best_a, n,
                                   start=start_time,
                                   deadline=phase_a_deadline,
                                   best_a=best_a,
                                   best_val=best_val,
                                   target_val=1.5)

    # ------------------------------------------------------------------
    # 4️⃣  Targeted hill‑climbing (≈30 % of budget)
    # ------------------------------------------------------------------
    phase_b_deadline = start_time + (deadline - start_time) * 0.85
    best_a = _hillclimb_targeted(best_a, n,
                                 start=phase_a_deadline,
                                 deadline=phase_b_deadline,
                                 best_a=best_a,
                                 best_val=best_val,
                                 target_val=1.5)

    # ------------------------------------------------------------------
    # 5️⃣  Short simulated‑annealing refinement (remaining time)
    # ------------------------------------------------------------------
    best_a, best_val = _annealing_phase(best_a, n,
                                        start=phase_b_deadline,
                                        deadline=deadline,
                                        best_a=best_a,
                                        best_val=best_val,
                                        target_val=1.5)

    # ------------------------------------------------------------------
    # Final secure evaluation
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
