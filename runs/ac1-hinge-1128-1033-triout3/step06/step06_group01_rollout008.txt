"""
Hybrid optimizer for the step‑function evaluation problem.

The pipeline consists of four main stages:

1. **Smooth surrogate optimisation** – many short L‑BFGS runs on a
   log‑sum‑exp (soft‑max) surrogate of the max‑convolution.  The surrogate
   guarantees positivity, a fixed ℓ₁‑norm and a smooth gradient.

2. **Re‑ordering heuristics** – after each L‑BFGS run we try several
   deterministic reorderings (anti‑symmetric, alternating) and a small
   set of random permutations.  The goal is to break the alignment of large
   values that creates high peaks in the self‑convolution.

3. **Enhanced simulated annealing** – a long SA phase that proposes three
   kinds of moves: swaps, mass‑transfer (shifting a tiny amount of weight
   from one position to another) and sub‑array reversals.  The temperature
   schedule is generous at the start and cools slowly, allowing the
   algorithm to escape local minima.

4. **Final hill‑climbing** – cheap deterministic refinements (mass
   transfers + swaps) that run until the time budget expires.

All heavy work (convolutions and objective evaluations) is performed
with FFT‑based routines (`_fft_convolve_self`) which are much faster than
the reference `evaluate_sequence`.  The algorithm respects the 1000 s
budget and always returns a feasible list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth surrogate (avoid closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # guard against rounding artefacts
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast version of `evaluate_sequence`.  The array ``a`` must be non‑negative
    and already satisfy 0 ≤ a_i ≤ 1000.  Returns the same value as the
    official evaluator.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Gradient of the smooth surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Returns (L, grad_w) where L is the smoothed proxy of the true objective
    and grad_w its gradient with respect to the soft‑max parameters ``w``.
    """
    # soft‑max (guarantees positivity and Σ a = 1)
    w_shift = w - np.max(w)          # numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # self‑convolution
    conv = _fft_convolve_self(a)          # length 2·n‑1
    max_conv = conv.max()

    # smooth approximation of max(conv)
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # log ∑ exp((conv−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed‑form)
    weights = np.exp(shifted)
    p = weights / weights.sum()           # probability distribution over k
    # convolution of a with reversed p (used in the derivation)
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    seg = conv_ap[n - 1:2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# Helper convolution used inside the gradient (FFT based, same as above)
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# Bind the helper for use inside the gradient
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Re‑ordering heuristics
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair large values on the left with small values on the right.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large to the left
        new_a[n - 1 - i] = sorted_vals[i]      # small to the right

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """
    Large / small alternating pattern: L, S, L, S, …
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    lo = 0
    hi = n - 1
    for i in range(n):
        if i % 2 == 0:                     # even index → largest remaining
            new_a[i] = sorted_vals[hi]
            hi -= 1
        else:                              # odd index → smallest remaining
            new_a[i] = sorted_vals[lo]
            lo += 1
    return new_a


def _try_random_reorders(a: np.ndarray, cur_val: float,
                         deadline: float, trials: int = 30) -> (np.ndarray, float):
    """
    Try a few random permutations of ``a`` and keep the best.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a
    best_val = cur_val

    for _ in range(trials):
        if time.time() >= deadline:
            break
        perm = rng.permutation(n)
        cand = a[perm]
        val = _eval_fast(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[rand reorder] new best = {best_val:.6f}")
    return best_seq, best_val


# ----------------------------------------------------------------------
# Enhanced simulated annealing (swap / mass‑transfer / sub‑array reverse)
# ----------------------------------------------------------------------
def _enhanced_simulated_annealing(a: np.ndarray, cur_val: float,
                                  deadline: float) -> (np.ndarray, float):
    """
    Simulated annealing that mixes three move types:
    * swap two positions,
    * transfer a small amount of mass from i to j,
    * reverse a random sub‑array.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # start with a relatively high temperature (10 % of current value)
    T = max(cur_val * 0.1, 1e-5)
    decay = 0.99997  # very slow cooling

    it = 0
    while time.time() < deadline:
        move_type = rng.integers(0, 3)   # 0=swap, 1=mass‑transfer, 2=reverse
        if move_type == 0:               # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            new_val = _eval_fast(a)
            delta = new_val - cur_val

        elif move_type == 1:              # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            # transfer a small fraction (up to 20 %) of a[i] to a[j]
            frac = rng.random() * 0.2
            delta_mass = a[i] * frac
            a[i] -= delta_mass
            a[j] += delta_mass
            new_val = _eval_fast(a)
            delta = new_val - cur_val

        else:                             # sub‑array reversal
            l = rng.integers(0, n - 1)
            r = rng.integers(l + 1, n)   # ensure length ≥ 1
            a[l:r] = a[l:r][::-1]
            new_val = _eval_fast(a)
            delta = new_val - cur_val

        # Metropolis acceptance
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            # revert move
            if move_type == 0:
                a[i], a[j] = a[j], a[i]
            elif move_type == 1:
                # undo the mass transfer
                a[i] += delta_mass
                a[j] -= delta_mass
            else:
                # reverse again restores original order
                a[l:r] = a[l:r][::-1]

        # cooling
        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.01, 1e-12)

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (mass transfers + swaps)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:  # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                   # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * (0.02 + 0.08 * rng.random())   # 2‑10 % of a_i
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

    # --------------------------------------------------------------
    n = 5000

    try:
