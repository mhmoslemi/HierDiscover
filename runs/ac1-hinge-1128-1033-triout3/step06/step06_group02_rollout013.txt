"""
Propose a step‐function that (almost) minimizes ``evaluate_sequence``.

The search consists of three phases that each run until the time budget
(≈1000 s) is almost exhausted.

1. **Smooth L‑BFGS refinement** – the objective ``max(conv)`` is smoothed
   with a log‑sum‑exp penalty; a soft‑max parametrisation guarantees the
   simplex constraints.  Several restarts and a decreasing ε‑schedule
   (1e‑2 → 1e‑12) are performed.

2. **Mirror‑descent on the true max‐conv** – the exact sub‑gradient
   ``g_i = a_{k*‑i}`` (where *k* is the current maximal lag) is used.
   An exponential update (a_i ← a_i·exp(−α·g_i)) is followed by a
   simple re‑normalisation onto the simplex.  A back‑tracking line‑search
   is performed at every iteration, guaranteeing a monotone improvement
   of the true objective.

3. **Intensive hill‑climbing / simulated‑annealing** – random swaps and
   small mass‑transfer moves are repeatedly tried; worsening moves are
   occasionally accepted according to a temperature schedule.  The best
   vector seen over the whole run is finally returned.

All helper functions are top‑level (no closures) and the code respects
the 1000 s runtime limit by checking the deadline at every major loop.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the L‑BFGS surrogate (set before each run)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter
_GLOBAL_N = None     # length of the current sequence


# ----------------------------------------------------------------------
# Basic utilities (FFT based convolution, simple re‑ordering, projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (useful for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself (length 2·len‑1).
    Uses an FFT of size a power of two for speed.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise can create tiny negatives; clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic re‑ordering: pair large values with small ones from the
    opposite side of the array.  Often reduces the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """
    Project a non‑negative vector onto the probability simplex by
    scaling: set negatives to zero and renormalise to sum = 1.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-12:                     # protective fallback
        n = v.shape[0]
        v = np.full(n, 1.0 / n)
    else:
        v /= s
    return v


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient for L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective:
        L = 2·n·max(conv) + 2·n·eps·log Σ_k exp((conv_k−max)/eps)

    The variables w are soft‑max parameters:
        a_i = exp(w_i) / Σ_j exp(w_j)

    Returns (L, ∇_w L).
    """
    # soft‑max (numerically safe)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)               # length 2·n−1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)                  # ∝ exp((conv_k−max)/eps)
    p = weights / weights.sum()                # probability distribution over k

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # derivative of L w.r.t. a (see derivation in the docstring)
    seg = conv_ap[n - 1: 2 * n - 1]            # middle n entries
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` – uses the global eps / n."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Exact objective (used for fast internal checks)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Compute 2·n·max(conv) / (sum(a)^2)  – identical to ``evaluate_sequence``
    for non‑negative vectors that respect the clipping limits.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 0.01:                     # matches the security check
        return np.inf
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Gradient of the (non‑smooth) max‑convolution
# ----------------------------------------------------------------------
def _max_conv_gradient(a: np.ndarray, kstar: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at the current active lag k*.
    g_i = a_{k*‑i} for indices where the term is defined, otherwise 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_start = max(0, kstar - (n - 1))
    i_end = min(n - 1, kstar)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    jdx = kstar - idx
    g[idx] = a[jdx]
    return g


# ----------------------------------------------------------------------
# Hill‑climbing / simulated‑annealing refinement
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Intensive stochastic local search.
    Random swaps (20 %) and small mass‑transfer moves (80 %) are tried.
    Worsening moves are accepted with a temperature that decays
    exponentially – a classic simulated‑annealing scheme.
    The routine stops when the deadline is reached.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()

    # annealing schedule
    T0 = 1e-3          # initial temperature (objective units)
    t_start = time.time()
    while time.time() < deadline:
        # linear cooling in terms of elapsed fraction
        frac = (time.time() - t_start) / (deadline - t_start + 1e-12)
        T = T0 * (1.0 - frac) + 1e-12

        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            # transfer a modest fraction (up to 5 %) of the source mass
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        new_a = _simplex_normalise(new_a)
        val = _objective(new_a)

        # accept if improvement or with Metropolis probability
        if val < cur_val - 1e-12:
            cur_a, cur_val = new_a, val
            print(f"[SA] new best = {cur_val:.6f}")
        else:
            Δ = val - cur_val
            if rng.random() < np.exp(-Δ / T):
                cur_a, cur_val = new_a, val

        # keep the globally best vector
        if cur_val < best_val - 1e-12:
            best_val, best_a = cur_val, cur_a.copy()
            print(f"[global] improved to {best_val:.6f}")

    return cur_a, cur_val

    print(f"[init] n={n}, objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth L‑BFGS refinement (several restarts & ε‑schedule)
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(1e-2), np.log10(1e-12), num=10)
    max_restarts = 6
    w = np.log(best_a + 1e-12)           # soft‑max parameters

    for restart in range(1, max_restarts + 1):
        if time.time() >= deadline - 30.0:
            break
        print(f"[L‑BFGS restart {restart}]")
        # occasional randomisation of starting point
        if restart % 3 == 0:
            w = np.random.randn(n) * 2.0

        for eps in eps_schedule:
            if time.time() >= deadline - 30.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 300,
                                    'ftol': 1e-14,
                                    'gtol': 1e-14,
                                    'disp': False})
            w = res.x
            # back‑transform to probabilities
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate /= a_candidate.sum()
            cur_val = _objective(a_candidate)

            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a_candidate.copy()
                print(f"[eps {eps:e}] new best = {best_val:.6f}")

        # occasional anti‑symmetric re‑ordering
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _objective(a_ord)
        if val_ord < best_val - 1e-12:
            best_val, best_a = val_ord, a_ord.copy()
            print(f"[reorder after L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent on the true max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_val = best_val
    it = 0
    alpha0 = 0.25                     # initial step size for the exponential update

    while time.time() < deadline - 30.0:
        conv = _full_convolution(a)
        kstar = int(conv.argmax())
        max_conv = conv[kstar]

        # exact sub‑gradient of the active lag
        g = _max_conv_gradient(a, kstar)

        # back‑tracking line‑search (guarantees improvement)
        alpha = alpha0
        improved = False
        for _ in range(7):           # at most 7 reductions of α
            a_new = a * np.exp(-alpha * g)
            a_new = _simplex_normalise(a_new)
            new_val = _objective(a_new)

            if new_val < cur_val - 1e-12:
                a, cur_val = a_new, new_val
                improved = True
                break
            else:
                alpha *= 0.5

        # if none of the tried step sizes improve, just reduce α0 a little
        if not improved:
            alpha0 *= 0.9

        # bookkeeping
        if it % 100 == 0:
            if cur_val < best_val - 1e-12:
                best_val, best_a = cur_val, a.copy()
                print(f"[grad iter {it}] new best = {best_val:.6f} "
                      f"(max conv {max_conv:.8f})")
            if it % 800 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = _objective(a_ord)
                if val_ord < best_val - 1e-12:
                    best_val, best_a = val_ord, a_ord.copy()
                    a = a_ord.copy()
                    print(f"[grad reorder] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final check & return (use the provided secure evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
