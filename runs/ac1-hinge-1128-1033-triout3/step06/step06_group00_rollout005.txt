"""
Propose a step‑function (non‑negative sequence) that approximately
minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three phases inside the 1000 s budget:

1. **Diverse warm‑start** – generate many random simplex vectors using
   several parametric families (uniform, linear “U‑shape”, exponentials,
   beta‑type power laws, mixtures).  Keep the best candidate.

2. **Projected soft‑max gradient descent** – minimise a smooth approximation
   of the L∞‑norm (log‑sum‑exp) while staying on the probability simplex.
   Momentum (Nesterov) and a slowly decaying step‑size are used.

3. **Simulated‑annealing fine‑tuning** – cheap mass‑transfer moves are
   accepted according to a Metropolis rule with a linear cooling schedule.
   Occasionally a global Gaussian jitter is applied to escape plateaux.

All operations are performed with FFT‑based full convolution, which is
~O(n log n) and fast for the chosen size (≈10‑12 k).  The final vector is
clipped to the `[0,1000]` range required by `evaluate_sequence`.  The best
objective value seen is printed together with the official score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT.
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(Fa * Fb, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Objective for a vector already normalised (sum = 1).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2 * n * max(conv(a, a))  (assumes sum(a)==1)."""
    n = a.size
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Random shape generators (all return a normalised probability vector).
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    a = np.ones(n, dtype=np.float64)
    return a / a.sum()


def _linear_u_shape(n: int) -> np.ndarray:
    """Linear increase from the centre to the borders (U‑shape)."""
    x = np.arange(n, dtype=np.float64)
    a = np.maximum(x, n - 1 - x) + 1.0
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _exp_decay(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.uniform(5e-4, 2e-2)
    a = np.exp(-lam * np.arange(n, dtype=np.float64))
    return a / a.sum()


def _exp_both_ends(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.uniform(5e-4, 2e-2)
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * x) + np.exp(-lam * (n - 1 - x))
    return a / a.sum()


def _beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    # beta density on a discrete grid, parameters drawn at random
    alpha = rng.uniform(0.3, 4.0)
    beta = rng.uniform(0.3, 4.0)
    x = (np.arange(1, n + 1, dtype=np.float64)) / n
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _power_law(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.3, 2.0)
    a = (np.arange(1, n + 1, dtype=np.float64)) ** (-p)
    return a / a.sum()


def _random_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mix a random number (2‑4) of the basic shapes."""
    bases = [_uniform,
             _linear_u_shape,
             _exp_decay,
             _exp_both_ends,
             _beta_shape,
             _power_law]
    k = rng.integers(2, 5)          # how many components to mix
    weights = rng.random(k)
    weights /= weights.sum()
    a = np.zeros(n, dtype=np.float64)
    for w in weights:
        func = rng.choice(bases)
        a += w * func(n, rng) if func.__code__.co_argcount == 2 else w * func(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = _uniform(n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Main entry point required by the grader.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative height sequence that (approximately) minimizes the
    evaluation function.  The routine respects a `budget_s` second time limit
    and prints intermediate scores.
    """
    start_time = time.time()
    # keep a small safety margin for the final print/evaluation
    deadline = start_time + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate size that allows many iterations while
    # giving enough degrees of freedom to improve over the previous best.
    # ------------------------------------------------------------------
    n = 11000

    # ------------------------------------------------------------------
    # Warm‑start: try a few classic shapes, then many random mixtures.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # (a) deterministic seeds (uniform, U‑shape, etc.)
    for init_func in [_uniform, _linear_u_shape]:
        a = init_func(n)
        val = _objective(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[warm‑start deterministic] val={best_val:.12f}")

    # (b) propose a candidate from a previously discovered sequence, if any
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.shape[0] != n:
                if prev.shape[0] > n:
                    prev = prev[:n]
                else:
                    prev = np.pad(prev, (0, n - prev.shape[0]))
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s > 0:
                prev = prev / s
                val = _objective(prev)
                if val < best_val:
                    best_val, best_seq = val, prev.copy()
                    print(f"[warm‑start previous] val={best_val:.12f}")
        except Exception:
            pass

    # (c) many random mixtures – continue until ~20 % of total time elapsed
    phase1_end = start_time + 0.20 * budget_s
    while time.time() < phase1_end:
        a = _random_mixture(n, rng)
        val = _objective(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[warm‑start random] new_best={best_val:.12f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected soft‑max gradient descent with momentum.
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.55 * budget_s
    a = best_seq.copy()
    vel = np.zeros_like(a)

    # hyper‑parameters
    eta = 0.6                 # initial step size
    eta_min = 1e-5
    eta_decay = 0.9997
    momentum = 0.9
    eps = None                # will be initialised from the first conv

    iter_gd = 0
    while time.time() < phase2_end:
        iter_gd += 1

        conv = _full_convolution_fft(a, a)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val - 1e-12:
            best_val, best_seq = val, a.copy()
            print(f"[gd iter {iter_gd:6d}] new_best={best_val:.12f}")

        # initialise / decay smoothing parameter
        if eps is None:
            eps = max_c * 0.3
        else:
            eps = max(1e-8, eps * 0.9995)

        # soft‑max weights (stable version)
        if eps < 1e-12:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w_raw = (conv - max_c) / eps
            # subtract max to avoid overflow
            w_raw = w_raw - np.max(w_raw)
            w = np.exp(w_raw)
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum

        # gradient of the smooth max (convolution with reversed a)
        rev_a = a[::-1]
        grad = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # Nesterov‑style momentum update
        vel = momentum * vel + (1.0 - momentum) * grad
        a_new = a - eta * vel
        a = _project_simplex(a_new)

        # schedules
        eta = max(eta_min, eta * eta_decay)

        # occasional random perturbation if stuck
        if iter_gd % 5000 == 0:
            noise = rng.normal(scale=0.002, size=n)
            a = _project_simplex(a + noise)
            print(f"[gd perturb] iter {iter_gd}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing fine‑tuning (mass‑transfer moves).
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_val = best_val
    it_sa = 0
    T0 = 1e-3
    Tmin = 1e-7
    max_transfer_ratio = 0.02          # max fraction of a[i] that can be moved

    while time.time() < deadline:
        it_sa += 1
        # linear temperature schedule
        T = max(Tmin, T0 * (1.0 - it_sa / 300_000))

        # pick distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] <= 0.0:
            continue

        # transfer a small amount from i to j
        max_transfer = min(cur[i], max_transfer_ratio * cur[i])
        delta = rng.uniform(0.0, max_transfer)

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        new_val = _objective(new)

        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_val - cur_val) / T)

        if accept:
            cur, cur_val = new, new_val
            if cur_val < best_val - 1e-12:
                best_val, best_seq = cur_val, cur.copy()
                print(f"[SA iter {it_sa:6d}] new_best={best_val:.12f}")

        # global jitter every 30 k iterations
        if it_sa % 30000 == 0:
            cur += rng.normal(scale=0.001, size=n)
            cur = _project_simplex(cur)
            cur_val = _objective(cur)

    # ------------------------------------------------------------------
    # Return the best sequence in the format expected by `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        # `evaluate_sequence` may not be available in the local test harness
        print("[result] evaluate_sequence not available")

    return final_seq
