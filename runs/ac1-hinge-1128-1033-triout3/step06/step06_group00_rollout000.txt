"""
Step‑function optimization via multiplicative (soft‑max) Adam.

The search works on the log‑probabilities `z` so that
    a_i = exp(z_i) / sum_j exp(z_j)
is always a non‑negative vector on the probability simplex.
At each iteration we evaluate the true objective

    F(a) = 2·n·max(conv(a,a))

and obtain a sub‑gradient of `max(conv(a,a))` by taking a uniform
convex combination of all active shifts (those attaining the maximum).
The sub‑gradient in the `z`‑space is obtained by the Jacobian of the
soft‑max map.  Adam (Kingma & Ba) is used to adapt the step size.
If progress stalls, a small Gaussian perturbation of `z` is applied to
escape flat regions.  The best vector seen during the run is returned
as the final candidate.

The routine respects the given `budget_s` time limit (≤1000 s) and
prints occasional intermediate scores.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: full linear convolution using a real FFT.
# --------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * b (real output)."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()      # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(Fa * Fb, n=nfft)[:size]
    return conv

    Return a non‑negative height sequence that (approximately) minimizes
    `evaluate_sequence`.  The algorithm runs an Adam‑based multiplicative
    gradient descent on the simplex, keeps the best solution found, and
    respects the given time budget.
    """

    deadline = start_time + budget_s - 2.0

    # ----------------------------------------------------------

    # build a convex combination of a few classic shapes.

                    base = base[:n]
                else:

            a = np.maximum(0.0, base)
            s = a.sum()
            if s > 0:
                a = a / s
        except Exception:
            a = None

    if a is None:
        # generate a pool of simple shapes
        x = np.arange(n, dtype=np.float64)

        shapes = []

        # exponential decays
        for _ in range(2):
            lam = rng.uniform(5e-4, 2e-2)
            shapes.append(np.exp(-lam * x))

        # power‑law tails
        for _ in range(2):
            p = rng.uniform(0.3, 2.0)
            shapes.append((x + 1) ** (-p))

        # half‑sine
        shapes.append(np.sin(np.pi * (x + 0.5) / n))

        # centred Gaussian
        sigma = rng.uniform(n / 8.0, n / 2.0)
        shapes.append(np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2))

        # simple triangle
        shapes.append(np.minimum(x, n - 1 - x) + 1.0)

        # Catalan‑type decreasing sequence
        catalan = np.empty(n, dtype=np.float64)
        cur = 1.0
        catalan[0] = cur
        for k in range(1, n):
            cur = cur * (2.0 * k - 1) / (2.0 * k)

    # ----------------------------------------------------------

    v = np.zeros_like(z)           # second moment estimate
    beta1, beta2 = 0.9, 0.999

        a = np.exp(z_shift)
        a_sum = a.sum()
        a = a / a_sum

        # ---- evaluate objective ----
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        cur_obj = 2.0 * n * max_conv

        if cur_obj < best_val - 1e-12:
            best_val = cur_obj
            best_seq = a.copy()
            stagn = 0
            print(f"[iter {it:07d}] new_best={best_val:.12f}")
        else:
            stagn += 1

        # ---- sub‑gradient of max(conv) ----
        tol = 1e-14
        active = np.flatnonzero(conv >= max_conv - tol)
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size       # uniform convex combination
        else:
            w[np.argmax(conv)] = 1.0

        rev_a = a[::-1]
        g_a = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        # ---- map gradient to z‑space (soft‑max Jacobian) ----
        a_dot_g = np.dot(g_a, a)                # scalar ⟨g,a⟩
        g_z = g_a * a - a_dot_g * a             # ∂F/∂z

        # ---- Adam update ----
        it += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        m = beta1 * m + (1.0 - beta1) * g_z
        v = beta2 * v + (1.0 - beta2) * (g_z * g_z)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)
        z = z - step

        # keep the log‑parameters in a numerically safe range
        z = np.clip(z, -60.0, 60.0)

        # simple learning‑rate decay
        lr *= 0.9999

        # ---- stagnation handling ----
        if stagn > 5000:
            # inject a modest random kick and reset Adam moments
            z += rng.normal(scale=0.04, size=n)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta2_pow = 1.0
            stagn = 0
            print(f"[iter {it:07d}] stagnation reset (random perturbation)")

        # occasional progress report
        if it % 20000 == 0:
            print(f"[iter {it}] best={best_val:.12f}")

    # ----------------------------------------------------------
    # Final conversion to the format expected by `evaluate_sequence`.
    # ----------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq.tolist()]

    # optional official score logging
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] evaluate_sequence not available")

    return final_seq
