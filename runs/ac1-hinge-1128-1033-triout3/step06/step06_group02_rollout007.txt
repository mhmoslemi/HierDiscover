The optimizer searches for a non‑negative step function `a = (a₀,…,aₙ₋₁)` that minimises  

\[
\text{score}(a)=\frac{2\,n\,\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

i.e. the peak of the self‑convolution for a fixed ℓ₁‑norm.  
The algorithm consists of three phases and respects the 1000 s budget:

1. **Smooth surrogate optimisation (L‑BFGS)** –  
   The non‑smooth `max` is replaced by a log‑sum‑exp smoothing
   \[
   L_\varepsilon(a)=2n\,\max(a*a)+2n\varepsilon\,
            \log\!\sum_k\exp\!\bigl((a*a)_k-\max(a*a))/\varepsilon\bigr).
   \]
   The variables are expressed with a soft‑max (`a_i = e^{w_i}/\sum e^{w_j}`) so
   the feasible set (non‑negative, sum = 1) is automatically respected.
   A schedule of decreasing `ε` drives the optimisation towards the true
   max‑convolution.

2. **Multi‑lag mirror‑descent** –  
   Instead of using only the single lag that attains the maximum, the
   `top_k` largest lags are taken, and the sub‑gradient  

   \[
   g_i=\sum_{k\in\text{top}_k} a_{k-i}
   \]

   is applied with an exponential (mirror) update `a_i ← a_i·exp(−α·g_i)`.
   The step size `α` decays slowly, the vector is projected back onto the
   probability simplex and the current best score is logged.  Occasionally the
   anti‑symmetric re‑ordering heuristic is applied to flatten the convolution
   further.

3. **Final stochastic hill‑climbing** –  
   Random swaps and tiny mass‑transfer moves are accepted only if they improve
   the true objective.  This phase explores the remaining local neighbourhood.

All heavy linear‑algebra work (full convolutions) uses FFT‑based
`np.fft.rfft`, giving an *O(n log n)* cost per iteration.  The whole search
fits comfortably into the allotted time and returns the best sequence found.

```python
"""
Propose a step‑function that (approximately) minimises ``evaluate_sequence``.
The algorithm consists of

1. L‑BFGS on a log‑sum‑exp smoothing of the max‑convolution.
2. Multi‑lag exponential mirror‑descent (gradient of several peak lags).
3. A final hill‑climbing refinement with random swaps / small mass transfers.

All updates keep the vector non‑negative and normalised to sum = 1.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global parameters used by the smooth surrogate (L‑BFGS)
_GLOBAL_EPS = None   # smoothing parameter ε
_GLOBAL_N = None     # length of the current vector


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT zero‑padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (FFT implementation)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Heuristic: pair large values with small values when the sequence is reversed.
    This often flattens the self‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (ensures a_i ≥ 0, Σ a_i = 1)

    L = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k−max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)            # length 2·n−1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)              # >0
    p = weights / weights.sum()            # probability over lags

    # convolution of a with reversed p (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # the central n entries correspond to ∂L/∂a
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]           # factor 4 comes from derivative of 2·n·max

    # chain rule through the soft‑max (standard soft‑max gradient)
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """
    Wrapper required by ``scipy.minimize`` – it reads the global ε and n.
    """
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _refine_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       n: int):
    """
    Simple hill‑climbing: random swaps (20 %) and tiny mass transfers (80 %).
    Only strict improvements are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val
```
