def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

"""
Optimisation of a non‑negative step function (vector a) to minimise

        evaluate_sequence(a) = 2·n·max(a∗a) / (sum(a)²)

where ∗ denotes the full discrete convolution.
The algorithm consists of three stages:

1.  Smooth‑max projected gradient descent.
    The non‑smooth objective max(b) (with b = a∗a) is replaced by the
    smooth approximation
            fβ(a) = (1/β)·log Σ_i exp(β·b_i) .
    Its gradient can be written in closed form and evaluated with two FFT‑based
    convolutions per iteration, allowing thousands of cheap steps.  β is
    increased from 1 to a few hundred (annealing) so that the algorithm first
    follows a gentle landscape and finally concentrates on the true maximum.

2.  L²‑flattening (the same variance‑reduction phase from the reference
    implementation) – it removes residual spikes in the convolution.

3.  Deterministic greedy hill‑climbing that moves a small amount of mass from
    the index that contributes most to the current worst convolution value to
    the index that contributes least.  The full convolution is updated
    incrementally in O(n) time.

All updates are followed by Euclidean projection onto the probability simplex
{a_i≥0, Σa_i=1}.  The best vector found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (len = 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow_two(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.
    Length = len(x)+len(y)‑1.
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow_two(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0

    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    # re‑normalise (numerical safety)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Smooth‑max gradient (β–annealed)
# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, beta: float):
    """
    Compute:
        b = a∗a
        max_b = max(b)
        grad = ∇_a ( (1/β)·log Σ_i exp(β·b_i) )
    using two FFT convolutions.
    Returns (b, max_b, grad).
    """
    # full convolution a∗a
    b = _full_convolution(a)
    max_b = b.max()

    # soft‑max weights (stabilised)
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:                     # should not happen, but be safe
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                         # now Σ w_i = 1

    # first term = a * w   (normal convolution)
    conv1 = _conv_fft(a, w)                # length = 3n‑2
    # second term = Σ_j a_j w_{j+k} – use correlation = a * rev(w)
    w_rev = w[::-1]
    conv2 = _conv_fft(a, w_rev)            # length = 3n‑2
    L = w.size - 1                         # = 2n‑2

    # gradient for each component k = 0..n‑1
    grad = conv1[:a.size] + conv2[L - np.arange(a.size)]
    return b, max_b, grad


# ----------------------------------------------------------------------
# L²‑flattening (smooth variance of the convolution)
# ----------------------------------------------------------------------
def _l2_flatten_phase(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> tuple[np.ndarray, float]:
    """Minimise Σ (b_i - mean(b))² while staying on the simplex."""
    start = time.time()
    base_step = 0.5
    min_step = 0.001
    avg_target = 1.0 / (2 * n - 1)           # mean of b when Σa = 1
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 150_000

    while time.time() < deadline:
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[L2   {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        r = b - avg_target
        a_rev = a[::-1]
        conv1 = _conv_fft(r, a)            # length = 3n‑2
        conv2 = _conv_fft(r, a_rev)
        # pick the centre part that corresponds to indices n‑1 … 2n‑2
        grad = 2.0 * (conv1[n - 1:2 * n - 1] + conv2[n - 1:2 * n - 1])

        # linear schedule for the step size
        prog = (time.time() - start) / (deadline - start)
        step = base_step * (1.0 - prog) + min_step * prog

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        if no_improve > IMPROVE_LIMIT:
            a = np.random.rand(n)
            a = _project_simplex(a)
            no_improve = 0
            print(f"[L2   {it}] restart after stagnation")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Greedy hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _greedy_hillclimb(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> np.ndarray:
    """Local search that moves a small amount of mass from a bad index to a good one."""
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        # contribution of each position to the current worst convolution entry
        k = max_idx
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]                     # a_i * a_{k-i}
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]           # derivative of a_mid²

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        # if source and target degenerate, pick random distinct pair
        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-14:                        # nothing to move
            src_candidates = np.where(a > 1e-14)[0]
            if src_candidates.size == 0:
                break
            source = int(np.random.choice(src_candidates))

        # amount to move (up to 30 % of the source mass, randomised)
        delta = a[source] * 0.25 * np.random.rand()
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n) work)
        b_new = b.copy()
        # cross‑terms 2·δ·a
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            # accept
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb and continue
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")
        it += 1

    return best_a
