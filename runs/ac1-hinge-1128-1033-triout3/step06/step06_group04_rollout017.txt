Implementation of projection:

def _project_simplex(v):
    # v is a numpy array. Return projection onto simplex.
    # reference algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    """Project onto the probability simplex (>=0, sum=1)."""
    # v is a 1-D np.ndarray.
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) that approximately minimises
``evaluate_sequence``.  

The method consists of four loosely coupled phases:

1. **Warm‑start generation** – a few parametrised families (exponential,
   power‑law, their product) are sampled and the best candidate is kept.
   The evaluation uses a fast FFT based computation of ``2·n·max(conv)``,
   i.e. the exact objective when the ℓ₁‑norm is forced to 1.

2. **Smooth surrogate optimisation** – the nonsmooth objective
   ``max(conv)`` is approximated by a log‑sum‑exp surrogate.
   The surrogate is minimised with L‑BFGS‑B on a soft‑max (log‑probability)
   parametrisation while the smoothing parameter ε is decreased (continuation).
   Several random restarts improve robustness.

3. **Projected sub‑gradient descent** – the current maximal lag ``k*`` is
   identified, a sub‑gradient of the exact max‑convolution is constructed,
   a step is taken opposite to that gradient and the result is projected back
   onto the probability simplex (non‑negative, unit ℓ₁‑norm).  The step size
   is adapted on‑the‑fly.

4. **Aggressive mass‑transfer hill‑climb** – a fine‑grained local search
   (the same routine used in the reference solution) is executed until the
   time budget expires.

All phases respect the global deadline; the best sequence seen so far is
returned.  The final score is also printed via ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full autocorrelation of a real vector (length 2·len−1).
    Uses FFT for speed and clips tiny negative artefacts.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the smooth surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv‑max)/ε)

    together with its gradient w.r.t. the soft‑max parameters ``w``.
    """
    # soft‑max to obtain a feasible a (Σa = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()

    # full auto‑convolution (length 2·n‑1)
    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()          # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)       # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]      # middle n entries
    grad_a = 4.0 * n * seg[::-1]        # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al.)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return the uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         best_val: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Reduce ``2·n·max(conv)`` by a sub‑gradient step on the exact objective.
    The step direction is the negative of the gradient of the active lag.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_obj = best_val
    best_a = cur_a.copy()
    best_obj = best_val
    step = 0.3                       # initial step size

    while time.time() < deadline - 0.2:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        cur_obj = 2.0 * n * max_b

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = cur_a.copy()
            print(f"[subgrad] new best = {best_obj:.6f}")

        # pick a maximal lag (break ties randomly)
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # sub‑gradient of conv[k] (see analysis)
        grad = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.intp)
        grad[idx] = cur_a[k - idx]           # a_{k-i}

        # tentative update
        new_a = cur_a - step * grad
        new_a = np.maximum(new_a, 0.0)
        new_a = _project_simplex(new_a)

        new_obj = 2.0 * n * np.max(_full_convolution(new_a))

        if new_obj < cur_obj:
            cur_a = new_a
            cur_obj = new_obj
            step = min(step * 1.05, 1.0)      # gently increase step after success
        else:
            step = max(step * 0.7, 1e-6)      # shrink step on failure

    return best_a, best_obj


# ----------------------------------------------------------------------
# Hill‑climbing mass‑transfer refinement (same as reference)
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray,
                              best_val: float,
                              deadline: float,
                              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Aggressive local search that repeatedly moves a small amount of mass away
    from the lag attaining the current maximum of the convolution.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.2                     # fraction of mass moved each trial

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue

        top_n = min(3, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]

        donors = []
        for idx in top_idx:
            i = int(i_range[idx])
            j = int(j_range[idx])
            donors.append((i, j))

        # destination positions – preferably low‑mass indices
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue

        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue
        new_a /= s

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.02, 0.5)
        else:
            step_frac = max(step_frac * 0.95, 1e-5)

    return cur_a, cur_val

    # include the old construction if we have one
    if a0 is not None:
        cur_val = exact_obj(a0)
        best_a = a0.copy()
        best_val = cur_val
        print(f"[init] loaded previous sequence, obj = {best_val:.6f}")

    # generate candidates from simple families
    while time.time() < deadline - 30.0:
        shape_type = rng.integers(3)
        if shape_type == 0:                     # pure exponential
            alpha = rng.uniform(0.0, 0.005)     # decay rate
            a = np.exp(-alpha * np.arange(n, dtype=float))
        elif shape_type == 1:                   # power‑law
            p = rng.uniform(0.0, 5.0)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
        else:                                   # product of both
            alpha = rng.uniform(0.0, 0.005)
            p = rng.uniform(0.0, 5.0)
            a = np.exp(-alpha * np.arange(n, dtype=float)) \
                * (np.arange(1, n + 1, dtype=float) ** (-p))

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a /= s
        val = exact_obj(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[randshape] new best = {best_val:.6f}")

    # safety: if nothing was produced, fall back to a uniform vector
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = exact_obj(best_a)

    print(f"[after warm‑start] best objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Smooth surrogate optimisation (log‑sum‑exp + L‑BFGS‑B continuation).
    # ------------------------------------------------------------------
    w = np.log(best_a + 1e-12)          # soft‑max parameters

    eps_schedule = np.logspace(-2, -12, num=22)   # finer at the end

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def fun_and_grad(w_vec):
            return _smooth_obj_and_grad(w_vec, eps, n)

        res = minimize(fun_and_grad,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 600,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = exact_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[smooth eps {eps:.1e}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. A few random restarts of the smooth phase.
    # ------------------------------------------------------------------
    for restart in range(5):
        if time.time() >= deadline - 10.0:
            break
        w = np.log(best_a + 1e-12) + rng.normal(scale=0.7, size=n)
        for eps in eps_schedule[-10:]:
            if time.time() >= deadline - 5.0:
                break
            def fun_and_grad(w_vec):
                return _smooth_obj_and_grad(w_vec, eps, n)
            res = minimize(fun_and_grad,
                           w,
                           method='L‑BFGS‑B',
                           jac=True,
                           options={'maxiter': 400,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = exact_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[restart {restart}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient descent on the exact max‑convolution.
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        best_a, best_val = _subgradient_descent(best_a, best_val, deadline, rng)
        print(f"[after subgrad] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final aggressive hill‑climbing refinement.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _mass_transfer_hill_climb(best_a, best_val,
                                                    deadline, rng)
        print(f"[final hillclimb] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final report using the provided evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)
        final_score = None

    return best_a.tolist()
