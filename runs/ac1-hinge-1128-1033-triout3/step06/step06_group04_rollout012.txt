```
def project_onto_simplex(v):
    # from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def subgradient_descent(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_val = 2 * n * np.max(full_convolution(a))
    print(f"[subgrad] start val = {best_val:.6f}")
    iteration = 0

"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.

The algorithm consists of three phases:

1. **Exponentiated sub‑gradient descent** –  The exact objective
   ``max(conv)`` is non‑smooth.  Its sub‑gradient at the maximal lag ``k*``
   is ``g_i = 2·a_{k*-i}`` for the overlapping indices.
   We perform a mirror‑descent (exponentiated gradient) update
   ``a_i ← a_i·exp(‑η·g_i)`` and re‑normalise to the simplex.
   The step size η is decreased like 1/√t.  This quickly drives the
   convolution peak down.

2. **Aggressive hill‑climbing (mass transfer)** –  After the continuous
   phase we repeatedly move a small fraction of mass away from the
   pairs that contribute most to the current maximal lag and deposit it
   on low‑mass positions.  The move is accepted only if the exact
   ``evaluate_sequence`` value improves.

3. **Final reporting** –  The best sequence found within the time
   budget is returned (the evaluator is called once for the final score).

The whole routine respects the given budget (default 1000 s) and prints
progress information.  It works for any length; we use n≈8000 to give the
optimizer enough degrees of freedom while keeping the run fast.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two not smaller than x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Full autocorrelation of a real vector a (length 2·len(a)‑1).
    Uses FFT for speed and clips tiny negative artefacts.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate tiny negative values caused by rounding
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Exponentiated sub‑gradient descent on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray, deadline: float, rng: np.random.Generator):
    """
    Perform exponentiated sub‑gradient descent on
        f(a) = max_k (a * a)[k]
    while keeping a on the probability simplex.
    Returns the best vector (sum=1) and its objective value
    (2·n·max_b, i.e. the value that ``evaluate_sequence`` returns for sum=1).
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise best known solution
    conv = _full_convolution(a)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    print(f"[subgrad] start value = {best_val:.6f}")

    iteration = 0
    eta0 = 0.5                     # initial step size (will decay)

    # main loop – stop a few seconds before the overall deadline
    while time.time() < deadline - 3.0:
        conv = _full_convolution(a)
        max_b = conv.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[subgrad] iter {iteration:5d} new best = {best_val:.6f}")

        # lag where the convolution attains its maximum
        k = int(np.argmax(conv))

        # sub‑gradient of the max‑lag w.r.t. a (g_i = 2·a_{k‑i})
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        g = np.zeros(n, dtype=np.float64)
        if i_start <= i_end:
            i_idx = np.arange(i_start, i_end + 1, dtype=np.intp)
            j_idx = k - i_idx
            g[i_idx] = 2.0 * a[j_idx]

        # step size schedule η_t = η0 / sqrt(t+1)
        eta = eta0 / np.sqrt(iteration + 1.0)

        # exponentiated (multiplicative) update and renormalisation
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            # numerical safeguard – revert to the best known point
            a = best_a.copy()
            s = a.sum()
        a = a / s

        iteration += 1

        # occasional tiny random jitter to avoid stagnation
        if iteration % 1000 == 0:
            jitter = rng.uniform(0.0, 1e-5, size=n)
            a = a + jitter
            a = a / a.sum()

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing based on mass transfers
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a0: np.ndarray, cur_val0: float,
                             deadline: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass away from the lag with the
    largest convolution value and deposit it on low‑mass positions.
    The move is accepted only if the exact objective improves.
    Returns the best vector found and its objective value.
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_val = cur_val0
    best_a = a0.copy()
    best_val = cur_val0

    step_frac = 0.25          # initial fraction of mass to move

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        # choose one of the maximal lags uniformly at random
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # contributions a_i * a_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]

        if contrib.size == 0:
            continue

        # pick a few (up to 4) largest contributing pairs
        top_n = min(4, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]

        donors = []
        for idx in top_idx:
            i = int(i_range[idx])
            j = int(j_range[idx])
            donors.append((i, j))

        # destination positions – pick low‑mass indices not used as donors
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)

        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)

        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                # move a fraction of the single entry's mass
                delta = new_a[i] * step_frac
                if delta <= 0.0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                # move the same amount from both members of the pair
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0.0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-6)
            continue

        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.9, 1e-6)
            continue
        new_a /= s

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")

            # increase step size modestly after a success
            step_frac = min(step_frac * 1.02, 0.5)
        else:
            # shrink step size when we cannot improve
            step_frac = max(step_frac * 0.95, 1e-6)

    return best_a, best_val
