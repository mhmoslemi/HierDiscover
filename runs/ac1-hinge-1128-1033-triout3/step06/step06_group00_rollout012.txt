"""
Propose a step‑function (non‑negative sequence) that approximately minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works inside a 1000 s budget:

1. **Warm‑start** – generate a pool of classic shapes (exponential,
   power‑law, half‑sine, Gaussian, triangle, arcsine, …) and form a few
   random convex combinations.  The best among them (according to the
   cheap objective 2 n·max(a∗a) with sum(a)=1) seeds the optimisation.

2. **Soft‑max gradient descent** – minimise a smooth approximation of the
   max‑convolution using momentum.  The step size and the “soft‑max”
   temperature are decayed slowly.

3. **Exact sub‑gradient descent (Adam)** – after the soft‑max phase the
   active set (indices where the convolution attains its maximum) is
   used to compute an exact sub‑gradient.  Adam’s adaptive moments speed
   up convergence.

4. **Simulated‑annealing fine‑tuning** – cheap “mass‑transfer’’ moves
   (move a small fraction of mass from one index to another) are accepted
   according to a Metropolis rule with a linearly decreasing temperature.
   Periodic global jitter helps escape plateaus.

The best sequence seen in any phase is returned (clipped to [0, 1000] as
required by ``evaluate_sequence``).  The whole routine respects the
time‑budget and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1: projection onto the probability simplex {x ≥ 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2: full linear convolution via real FFT (fast and exact)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of a and b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()        # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Helper 3: objective for a vector already on the simplex (sum == 1)
# ----------------------------------------------------------------------
def _objective_on_simplex(a: np.ndarray) -> float:
    """Return 2 * n * max(a*a)  (a already normalised, sum(a)=1)."""
    n = a.size
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Warm‑start shape generators (all produce non‑negative arrays)
# ----------------------------------------------------------------------
def _shape_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.uniform(5e-4, 2e-2)
    x = np.arange(n, dtype=np.float64)
    return np.exp(-lam * x)


def _shape_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.3, 2.0)
    x = np.arange(n, dtype=np.float64)
    return (x + 1) ** (-p)


def _shape_half_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * (x + 0.5) / n)


def _shape_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(n / 8.0, n / 2.0)
    x = np.arange(n, dtype=np.float64)
    centred = (x - (n - 1) / 2.0) / sigma
    return np.exp(-0.5 * centred * centred)


def _shape_triangle(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    return np.minimum(x, n - 1 - x) + 1.0


def _shape_arcsine(n: int) -> np.ndarray:
    # Beta(0.5, 0.5) density on [0,1] (U‑shaped, heavy at the ends)
    x = (np.arange(n) + 0.5) / n
    eps = 1e-12
    return 1.0 / np.sqrt(x * (1.0 - x) + eps)


def _shape_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.uniform(0.2, 5.0)
    b = rng.uniform(0.2, 5.0)
    x = (np.arange(n) + 0.5) / n
    # use scipy-like beta pdf, but avoid external dependencies:
    from math import gamma
    # Gamma values (might be slightly slower but called only a few times)
    const = gamma(a + b) / (gamma(a) * gamma(b))
    return const * (x ** (a - 1)) * ((1 - x) ** (b - 1))


def _initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a single warm‑start candidate by mixing several basic shapes."""
    shapes = [
        _shape_exponential(n, rng),
        _shape_powerlaw(n, rng),
        _shape_half_sine(n),
        _shape_gaussian(n, rng),
        _shape_triangle(n),
        _shape_arcsine(n),
        _shape_beta(n, rng),
    ]
    weights = rng.random(len(shapes))
    weights /= weights.sum()
    a = sum(w * s for w, s in zip(weights, shapes))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a
