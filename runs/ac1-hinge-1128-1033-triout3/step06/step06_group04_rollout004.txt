assistantfinal**Solution Overview**

The evaluation function is  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\qquad a\ge 0 .
\]

Because the score is homogeneous we may normalise \(\sum a_i = 1\); then  

\[
\text{score}(a)=2n\;\max (a*a).
\]

The task is therefore to find a non‑negative vector with unit ℓ₁‑norm that makes the
maximum entry of the full auto‑convolution as small as possible.

The algorithm consists of three stages

1. **Continuation on a smooth surrogate** –  
   the non‑smooth objective \(\max\) is replaced by a log‑sum‑exp
   approximation.  A sequence of decreasing smoothing parameters
   \(\varepsilon\) is used and a standard L‑BFGS‑B optimiser (with a
   soft‑max parametrisation) is run for each \(\varepsilon\).  
   The continuation quickly brings us into a good basin.

2. **Projected sub‑gradient descent** –  
   After the smooth phase we switch to the exact objective.  For the
   lag(s) where the convolution attains its current maximum we compute the
   sub‑gradient  
   \(\partial_i =2\,a_{k-i}\) (zero when the index is out of range) and
   perform a projected sub‑gradient step onto the probability simplex.
   An adaptive step‑size together with occasional random restarts drives the
   method to a tighter local optimum.

3. **Aggressive mass‑transfer hill‑climb** –  
   A final refinement similar to the one used in the reference code moves a
   small fraction of the mass from the most contributing pairs of the
   maximal lag to low‑mass positions.  The hill‑climb accepts a move only if
   the exact score improves.

All three phases respect the given time budget (default 1000 s).  
The best sequence found so far is stored globally as `height_sequence_1` and
is used as a warm‑start.

```python
"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.  The algorithm combines:

1.  A continuation L‑BFGS‑B optimisation on a log‑sum‑exp surrogate
    (smooth approximation of the max‑convolution).

2.  Projected sub‑gradient descent on the exact objective
    (max‑convolution) with an adaptive step size and occasional
    random restarts.

3.  A final aggressive hill‑climb that moves mass away from the most
    active lag of the convolution.

The routine works for any vector length (default 4000) and respects the
global time budget (default 1000 seconds).  The best sequence found is
returned as a plain ``list[float]``.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full auto‑convolution of a real vector (length 2·len‑1).
    Uses FFT for speed.  Small negative artefacts are clipped to zero.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # protect against rounding negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Compute the log‑sum‑exp surrogate

        L_ε(a) = 2·n·max(conv) + 2·n·ε·log Σ exp((conv−max)/ε)

    together with its gradient w.r.t. the soft‑max parameters ``w``.
    """
    # soft‑max (feasible a, Σa = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a = a / a.sum()

    conv = _full_convolution(a)
    max_conv = conv.max()

    # log‑sum‑exp smoothing term
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # shape (2·n‑1,)

    # g_i = 4·n·∑_k p_k·a_{k‑i}
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                  # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]                # middle n entries
    grad_a = 4.0 * n * seg[::-1]                   # shape (n,)

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the rho where u_i - (cssv_i / i) > 0
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------
def _subgradient_optimize(a: np.ndarray,
                          best_a: np.ndarray,
                          best_val: float,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a projected sub‑gradient descent on
    f(a) = 2·n·max(a*a).  Returns the best vector seen before the time
    limit expires.
    """
    n = a.shape[0]
    step = 1.0                # initial step size
    no_improve = 0            # counter for stagnation

    while time.time() < deadline:
        conv = _full_convolution(a)
        max_b = conv.max()
        cur_val = 2.0 * n * max_b

        # keep global best
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[subgrad] new best = {best_val:.6f}")

        # pick a random active lag (where the max is attained)
        ks = np.flatnonzero(conv >= max_b - 1e-12)
        k = int(rng.choice(ks))

        # sub‑gradient of max‑convolution at lag k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        g[i_range] = 2.0 * a[j_range]

        # trial step
        a_trial = a - step * g
        a_trial = _project_onto_simplex(a_trial)

        new_val = 2.0 * n * np.max(_full_convolution(a_trial))

        if new_val < cur_val:
            # successful step – accept and increase step size a bit
            a = a_trial
            step = min(step * 1.2, 10.0)
            no_improve = 0
        else:
            # reject – shrink step size
            step = max(step * 0.5, 1e-7)
            no_improve += 1

        # occasional random restart to escape possible plateaus
        if no_improve >= 500:
            a = best_a + 0.01 * rng.normal(size=n) * best_a
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a)
            step = 1.0
            no_improve = 0

    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive hill‑climbing (mass‑transfer) refinement
# ----------------------------------------------------------------------
def _mass_transfer_hill_climb(a: np.ndarray,
                              best_val: float,
                              deadline: float,
                              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small fraction of mass away from the most active
    lag of the convolution.  The move is accepted only if the exact
    evaluation improves.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.2                     # initial fraction of mass to move

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(conv == max_b)
        k = int(rng.choice(ks))

        # pairs (i, j) that contribute to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]

        if contrib.size == 0:
            continue

        top_n = min(3, contrib.size)
        top_idx = np.argpartition(-contrib, top_n - 1)[:top_n]

        donors = [(int(i_range[idx]), int(j_range[idx])) for idx in top_idx]

        # destinations: choose low‑mass positions
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size < 2 * len(donors):
            low_mass = np.arange(n)
        forbidden = np.unique(np.ravel(donors))
        cand = np.setdiff1d(low_mass, forbidden, assume_unique=True)
        if cand.size < 2 * len(donors):
            cand = np.setdiff1d(np.arange(n), forbidden, assume_unique=True)
        dest = rng.choice(cand, size=2 * len(donors), replace=False)

        new_a = cur_a.copy()
        ok = True
        for d, (i, j) in enumerate(donors):
            if i == j:
                delta = new_a[i] * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                p = int(dest[2 * d])
                new_a[p] += delta
            else:
                delta = min(new_a[i], new_a[j]) * step_frac
                if delta <= 0:
                    ok = False
                    break
                new_a[i] -= delta
                new_a[j] -= delta
                p = int(dest[2 * d])
                q = int(dest[2 * d + 1])
                new_a[p] += delta
                new_a[q] += delta

        if not ok:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue

        # renormalise (the evaluation is scale‑invariant, but we keep sum=1)
        s = new_a.sum()
        if s <= 0:
            step_frac = max(step_frac * 0.9, 1e-5)
            continue
        new_a /= s

        new_val = 2.0 * n * np.max(_full_convolution(new_a))

        if new_val < cur_val:
            cur_a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
            step_frac = min(step_frac * 1.02, 0.5)
        else:
            step_frac = max(step_frac * 0.95, 1e-5)

    return cur_a, cur_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise the step‑function heights.

    1. Initialise from ``height_sequence_1`` if present, otherwise draw a
       random Dirichlet vector.
    2. Run a continuation L‑BFGS‑B on a log‑sum‑exp surrogate (several
       ε‑stages) to obtain a good basin.
    3. Refine with a projected sub‑gradient descent on the exact
       objective for the remaining time budget.
    4. Finish with an aggressive hill‑climbing mass‑transfer phase.
    5. Return the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep it moderate for speed, but reuse previous length
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = old.shape[0]
    except Exception:
        n = 4000   # default if no previous construction is available

    # ------------------------------------------------------------------
    # Initialise sequence (reuse previous construction if possible)
    # ------------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            # resample the old vector onto the new grid
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
    except Exception:
        a0 = rng.random(n)

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()                     # enforce Σa = 1

    # ------------------------------------------------------------------
    # Initial (smooth) optimisation via continuation on the LSE surrogate
    # ------------------------------------------------------------------
    w = np.log(a0 + 1e-12)            # soft‑max parameters
    best_a = a0.copy()
    best_val = 2.0 * n * np.max(_full_convolution(best_a))
    print(f"[init] n={n}, initial exact score = {best_val:.6f}")

    # epsilon schedule – from coarse to very fine
    eps_schedule = np.logspace(-2, -16, num=35)

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        def fun_and_grad(w_vec):
            return _smooth_obj_and_grad(w_vec, eps, n)

        # L‑BFGS‑B – fairly generous iteration limit, but cheap per iteration
        res = minimize(fun_and_grad,
                       w,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 800,
                                'ftol': 1e-12,
                                'gtol': 1e-12,
                                'disp': False})
        w = res.x

        # evaluate the exact objective at the new point
        a = np.exp(w - np.max(w))
        a /= a.sum()
        cur_val = 2.0 * n * np.max(_full_convolution(a))
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        best_a, best_val = _subgradient_optimize(best_a,
                                                 best_a,
                                                 best_val,
                                                 deadline,
                                                 rng)

    # ------------------------------------------------------------------
    # Final aggressive hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _mass_transfer_hill_climb(best_a,
                                                    best_val,
                                                    deadline,
                                                    rng)

    # ------------------------------------------------------------------
    # Final report (use the provided evaluator for the final score)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # expose the best sequence for possible reuse in the next round
    global height_sequence_1   # type: ignore
    height_sequence_1 = best_a.tolist()   # pylint: disable=global-statement

    return best_a.tolist()
```
