from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def solve_lp(f, M):
    n = len(f)
    K = 2*n - 1
    rows = []
    cols = []
    data = []
    for k in range(K):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # equivalent j = k - i
        # For each i, coefficient on g_j is f[i]
        # We'll iterate i
        for i in range(i_start, i_end+1):
            j = k - i  # g index
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A_ub = csr_matrix((data, (rows, cols)), shape=(K, n))
    b_ub = np.full(K, M)
    c = -np.ones(n)  # maximize sum(g)
    # variable bounds: (0, None)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'time_limit': 30})
    # time_limit maybe not needed; we can trust it's fast.
    if not res.success:
        # fallback to uniform or previous f
        return np.copy(f)
    g = res.x
    return g

"""
Propose a step‑function that (hopefully) improves on the previous
1.512114 bound.

The optimiser works in three stages

1️⃣  A smooth surrogate (log‑sum‑exp of the convolution) is minimised
    with L‑BFGS‑B for several decreasing smoothing parameters.  This
    quickly drives the whole convolution curve towards a flat shape.

2️⃣  A direct sub‑gradient descent on the *actual* max‑convolution.
    Instead of reacting to a single maximal lag we react to all lags
    whose value is within a small tolerance of the current maximum.
    The update is a multiplicative mirror‑descent step
    `a_i ← a_i·exp(−α·g_i)` preserving non‑negativity, followed by a
    projection onto the simplex.  Momentum and an exponential decay of
    the step size are used together with the “anti‑symmetric reorder”
    heuristic that often flattens the peak further.

3️⃣  A fast hill‑climbing refinement that performs random swaps and
    mass‑transfer moves; only strict improvements are kept.

The whole routine respects a 1000 s budget (a 5 s safety margin is kept)
and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Low‑level utilities (no closures – required by the task)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of a real vector with itself (length 2·n‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # remove tiny negative artefacts caused by rounding
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair large values with small ones when the sequence is reversed.
    This simple heuristic often reduces the tallest peak of the
    self‑convolution.
    """
    n = a.shape[0]
    s = np.sort(a)                 # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[n - 1 - i]        # large → left side
        b[n - 1 - i] = s[i]        # small → right side
    if n % 2:
        b[half] = s[half]
    return b


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Very light‑weight hill‑climber: random swaps (20 %) and
    mass transfers (80 %).  Only strict improvements are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:          # should never happen
            continue
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


# ----------------------------------------------------------------------
# Smooth surrogate for L‑BFGS (global variables needed by wrapper)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate of 2·n·max(conv) together with its gradient
    w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)  (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # ----- soft‑max (stable) -----
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # ----- smoothed max via log‑sum‑exp -----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a -----
    weights = np.exp(shifted)
    p = weights / weights.sum()           # a probability distribution over k

    # convolution a * reversed(p)  –‑ efficient via FFT
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries correspond to ∂L/∂a
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # ----- chain rule through soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)
