def proj_simplex(v):
    # Projects onto the probability simplex {x >=0, sum=1}
    v_shift = v - np.max(v)  # For numerical stability? Not needed
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector whose ℓ₁‑norm is 1) that
minimises the evaluation function

    2·n·max(conv(a,a)) / (sum(a)²)

where ``conv`` is the linear convolution.  The algorithm is a
combination of three simple ingredients that run within the 1000 s
budget:

1. **Initialisation** – use a smooth sine‑shaped vector (or fall back to
   a random Dirichlet) and normalise it to the probability simplex.

2. **Projected sub‑gradient descent** – at the current point ``a`` the
   lag ``k`` that attains the maximal convolution value is found.
   The gradient of this particular lag with respect to ``a`` is
   computed analytically, a step is taken in the opposite direction and
   the result is projected back onto the simplex (Duchi et al. 2008).
   The step size is adapted (increase on improvement, decrease otherwise).

3. **Simulated‑annealing style hill‑climbing** – small random mass
   transfers between two indices are tried; a move is accepted if it
   improves the exact objective, otherwise it may be accepted with a
   probability that decays with a temperature schedule.  Successful
   moves are kept and the temperature is slowly lowered.

Both phases use the fast FFT‑based full convolution (``_full_convolution``)
to evaluate the objective.  The best vector found is finally handed to
the provided ``evaluate_sequence`` for the official score.

The whole routine respects the 1000 s limit and prints occasional
progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear autocorrelation of a real vector ``x``.
    Length = 2·len(x)‑1.  Uses real FFT for speed.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Tiny negative round‑off artefacts are clipped to zero.
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implementation follows Duchi et al., ICML 2008.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]                       # last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise exactly to 1
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:                                     # all zero – fall back to uniform
        w[:] = 1.0 / len(v)
    return w


# ----------------------------------------------------------------------
# Exact objective (same formula as ``evaluate_sequence`` but faster)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    max_b = np.max(_full_convolution(a))
    return 2.0 * n * max_b

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a low‑value step‑function using
    sub‑gradient descent + simulated‑annealing hill‑climbing.
    Returns the best list of floats found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # a tiny safety margin

    # --------------------------------------------------------------
    # Problem size – moderate (4000) keeps FFTs cheap while offering
    # enough degrees of freedom for a good optimum.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Initialise a – try to reuse a previous construction, otherwise fall
    # back to a smooth sine shape (positive, zero at the ends) and
    # normalise to the probability simplex.
    # --------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if a0.shape[0] != n:
            # linear interpolation to the desired length
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
    except Exception:
        # sine‑shaped start (zero at the boundaries, positive in the middle)
        x = np.arange(n, dtype=float) / (n - 1)
        a0 = np.sin(np.pi * x) + 1e-12

    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    best_a = a0.copy()
    best_val = _objective(best_a)
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent on the exact max‑convolution.
    # --------------------------------------------------------------
    step = 1.0                     # initial step size (in the simplex space)
    it = 0
    # we stop this phase a few seconds before the overall deadline
    subgrad_deadline = deadline - 30.0

    while time.time() < subgrad_deadline:
        it += 1
        conv = _full_convolution(best_a)
        max_val = conv.max()
        # index (lag) where the maximum is attained – pick the first one
        k = int(np.flatnonzero(conv == max_val)[0])

        # analytic gradient of conv_k w.r.t. a (see description above)
        grad = np.zeros_like(best_a)
        if k < n:                         # term a_i * a_{i+k}
            grad[: n - k] += best_a[k:]
        if k > 0:                         # term a_{i-k} * a_i
            grad[k:] += best_a[: n - k]

        # take a step opposite to the gradient and project back onto the simplex
        cand = best_a - step * grad
        cand = _proj_simplex(cand)

        cand_val = _objective(cand)
        if cand_val < best_val:

            best_val = cand_val

        else:
            step = max(step * 0.7, 1e-4)          # shrink step on failure

        if it % 200 == 0:

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing hill‑climbing.
    # Randomly move a small fraction of mass from a “donor’’ index to a
    # “receiver’’ index.  Accept improvements, otherwise accept with
    # probability exp(−Δ/temperature).  Temperature decays slowly.
    # --------------------------------------------------------------
    temperature = 0.02
    sa_iter = 0
    # allocate most of the remaining time to this phase
    while time.time() < deadline:
        sa_iter += 1
        # pick a donor with non‑trivial mass
        donor_idx = int(rng.integers(0, n))
        if best_a[donor_idx] < 1e-12:
            continue
        # pick a receiver that currently holds relatively little mass
        low_mass = np.where(best_a < best_a.mean())[0]
        if low_mass.size == 0:
            receiver_idx = int(rng.integers(0, n))
        else:
            receiver_idx = int(rng.choice(low_mass))

        # fraction of donor mass to move (adaptable random size)
        frac = rng.random() * 0.08 + 0.01          # between 1 % and 9 %
        delta = best_a[donor_idx] * frac
        if delta <= 0:
            continue

        cand = best_a.copy()
        cand[donor_idx] -= delta
        cand[receiver_idx] += delta

        # numerical safety – enforce non‑negativity (tiny negatives may appear)
        if cand[donor_idx] < -1e-15:
            continue
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()                         # re‑normalise to keep Σ = 1

        cand_val = _objective(cand)
        improve = cand_val < best_val

        if improve or rng.random() < np.exp(-(cand_val - best_val) / max(temperature, 1e-12)):
            best_a = cand
            best_val = cand_val
            temperature = max(temperature * 0.9995, 1e-6)   # cool down slowly
            if improve:
                print(f"[SA] iter {sa_iter:6d}, new best = {best_val:.6f}")

        # occasional temperature bump to avoid being frozen too early
        if sa_iter % 5000 == 0:
            temperature = min(temperature * 1.2, 0.1)

    # --------------------------------------------------------------
    # Final evaluation with the provided ``evaluate_sequence`` (ensures
    # the identical security checks are applied).
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] official score = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
