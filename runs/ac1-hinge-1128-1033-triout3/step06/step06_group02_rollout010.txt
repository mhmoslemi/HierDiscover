    iteration = 0
    lambda0 = 0.01, lambda_decay = maybe 0.9 per 200 iterations.
    alpha0 = 0.2 (or 0.1), decays as (0.99^something)
    while time.time() < deadline:

        max_c = conv.max()
        # compute weights with lambda
        lam = max(1e-12, lambda0 * (0.9 ** (iteration // 500)))
        # compute weights
        shifted = (conv - max_c) / lam
        # to avoid overflow, subtract max
        shifted_max = np.max(shifted)
        w = np.exp(shifted - shifted_max)
        w_sum = np.sum(w)

        # update

"""
Optimise a non‑negative step function to minimise the evaluation
function  ``evaluate_sequence``.  The algorithm combines three stages
for each problem size:

1.  A smooth surrogate of the max‑convolution (log‑sum‑exp) is
    minimised with many L‑BFGS restarts.  The surrogate is gradually
    tightened (ε → 0) and the optimiser is allowed far more iterations
    than in the reference solution.

2.  A weighted mirror‑descent loop on the original non‑smooth objective
    is performed.  The weight distribution is a softmax over the
    (conv – max)/λ, i.e. a smoothed sub‑gradient that simultaneously
    reduces all large lobes, not only the global maximum.

3.  A final simulated‑annealing hill‑climber (mass transfers) refines
    the best candidate.

The procedure is executed for several sequence lengths (2000…6000)
under the 1000 s budget; the best sequence overall is returned.
All helper functions are top‑level, there are no closures or file I/O.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables used by the L‑BFGS wrapper (required by scipy)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative artefacts that may appear due to rounding
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair the largest values with the smallest ones at opposite ends.
    This heuristic often flattens the auto‑convolution peak.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k−max)/ε)
    """
    # soft‑max (numerically stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)           # length 2·n−1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted - np.max(shifted))   # normalised in the next line
    weights /= weights.sum()                      # p_k = softmax over (conv - max)/ε

    # convolution of a with reversed p (efficient via FFT)
    p_rev = weights[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries correspond to the derivative
    seg = conv_ap[n - 1: 2 * n - 1]          # length n
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _initialise_a(n: int, rng: np.random.Generator, restart_index: int):
    """
    Produce an initial feasible probability vector of length ``n``.
    The first restart uses the known construction (if available);
    later restarts are random Dirichlet draws.
    """
    # try to use the previously discovered construction
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # interpolate to the new size
            old_x = np.linspace(0.0, 1.0, old.shape[0])

            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()

        s = a.sum()
        if s < 1e-12:

        a /= s

    except Exception:
        # random Dirichlet (α=0.5 gives a moderately sparse start)

def _optimize_for_n(n: int, deadline: float, rng: np.random.Generator):
    """
    Run the three‑phase optimisation for a fixed length ``n``.
    Returns the best sequence (as a NumPy array) and its internal
    objective value 2·n·max(conv).
    """
    best_a = None
    best_val = np.inf
    max_restarts = 8                     # number of independent starts
    eps_schedule = np.logspace(np.log10(5e-2), np.log10(1e-12), num=12)

    for restart in range(max_restarts):
        if time.time() > deadline - 5.0:
            break

        # ----- initialise -------------------------------------------------
        a = _initialise_a(n, rng, restart)
        conv = _full_convolution(a)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val, best_a = cur_val, a.copy()
            print(f"[n={n} restart={restart}] init value {best_val:.6f}")

        # ----- Phase 1: smooth L‑BFGS on the log‑sum‑exp surrogate -----
        w = np.log(a + 1e-12)                     # soft‑max parameters
        for eps in eps_schedule:
            if time.time() > deadline - 5.0:
                break
            global _GLOBAL_EPS, _GLOBAL_N
            _GLOBAL_EPS = eps
            _GLOBAL_N = n
            res = minimize(
                _smooth_obj_and_grad_wrapper,
                w,
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': 400,
                         'ftol': 1e-15,
                         'gtol': 1e-15,
                         'disp': False})
            w = res.x
            # back‑to probability vector

            a = np.exp(w_shift)
            a /= a.sum()
            conv = _full_convolution(a)
            cur_val = 2.0 * n * conv.max()
            if cur_val < best_val:

                if val_ord < best_val:

                    a = a_ord.copy()
                    print(f"[n={n} reorder] improved to {best_val:.6f}")

        it = 0

        while time.time() < deadline - 5.0 and it < max_md_iters:
            conv = _full_convolution(a)
            max_c = conv.max()

            # smoothing parameter for the soft‑max weights (decreases slowly)
            lam = 0.01 * (0.85 ** (it // 500))
            # soft‑max over conv entries
            shifted = (conv - max_c) / lam
            shifted_max = np.max(shifted)          # always 0, kept for stability
            wgt = np.exp(shifted - shifted_max)
            wgt /= wgt.sum()

            # g_i = Σ_k wgt_k·a_{k‑i}  (implemented via a single FFT)
            a_rev = a[::-1]
            size = _next_pow2(len(wgt) + n - 1)
            fw = np.fft.rfft(wgt, size)
            fa_rev = np.fft.rfft(a_rev, size)
            conv_w = np.fft.irfft(fw * fa_rev, size)[: len(wgt) + n - 1]
            g = conv_w[n - 1: n - 1 + n]          # length n

            # exponential mirror‑descent update
            alpha = 0.18 * (0.95 ** (it // 300))
            a = a * np.exp(-alpha * g)

            # project back onto the simplex
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:
                a = np.full(n, 1.0 / n)
                s = a.sum()
            a /= s

            cur_val = 2.0 * n * max_c
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[n={n} MD iter={it}] new best {best_val:.6f}")

            # periodic anti‑symmetric re‑ordering
            if it % 400 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = 2.0 * n * _full_convolution(a_ord).max()
                if val_ord < best_val:
                    best_val, best_a = val_ord, a_ord.copy()
                    a = a_ord.copy()
                    print(f"[n={n} MD reorder] improved to {best_val:.6f}")

            it += 1

        # ----- Phase 3: simulated‑annealing hill climbing -----------------
        cur_a = a.copy()
        cur_val = 2.0 * n * _full_convolution(cur_a).max()
        total_hc_iters = 30000
        temp_start = 1e-3
        temp_end = 1e-6

            T = temp_start * (temp_end / temp_start) ** (hc_it / total_hc_iters)

            if i == j or cur_a[i] <= 1e-15:

            delta = cur_a[i] * 0.07 * rng.random()

            new_a = np.maximum(new_a, 0.0)
            s = new_a.sum()
            if s < 1e-12:

            new_a /= s

            if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / T):
                cur_a, cur_val = new_a, new_val
                if cur_val < best_val:
                    best_val, best_a = cur_val, cur_a.copy()
                    print(f"[n={n} HC iter={hc_it}] new best {best_val:.6f}")

        # end of this restart – keep the best among all restarts
        if best_a is not None and best_val < np.inf:
            # continue to the next restart with the globally best as a warm start
            a = best_a.copy()
        else:
            # if everything failed (should not happen) re‑initialise
            a = _initialise_a(n, rng, restart)

    return best_a, best_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Entry point required by the evaluation harness.
    Searches over several sequence lengths and returns the best
    non‑negative step function found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # 5 s safety margin for the final return
    deadline = start_time + budget_s - 5.0

    best_overall_seq = None
    best_overall_val = np.inf

    # A modest list of lengths – more are tried if time permits
    length_candidates = [2000, 3000, 4000, 5000, 6000]

    for n in length_candidates:
        if time.time() > deadline:
            break
        print(f"[main] Optimising length n={n}")
        a_opt, val_opt = _optimize_for_n(n, deadline, rng)
        if a_opt is None:
            continue
        if val_opt < best_overall_val:
            best_overall_val = val_opt
            best_overall_seq = a_opt.copy()
            print(f"[main] New global best from n={n}: value={best_overall_val:.6f}")

    # Final sanity check with the provided evaluate_sequence function
    if best_overall_seq is None:
        # fall back to a uniform distribution of length 1000
        best_overall_seq = np.full(1000, 1.0 / 1000)
        best_overall_val = None

    try:
        final_score = evaluate_sequence(best_overall_seq.tolist())
        print(f"[final] evaluate_sequence reports {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_overall_seq.tolist()
