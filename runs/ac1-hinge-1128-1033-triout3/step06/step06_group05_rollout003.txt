"""
Propose a step‑function with a low max‑autoconvolution value.

Algorithm
---------
1. **Problem size** – use a longer vector (default n = 12 000).  More
   degrees of freedom give the optimizer a better chance to flatten the
   full convolution while keeping the total running time under the 1000 s
   limit.

2. **Initialisation** – start from the supplied `height_sequence_1`
   (interpolated to the new length) if it exists, otherwise from a
   random point on the probability simplex.

3. **Phase A – Smoothed‑max gradient descent**  
   Replace the non‑smooth `max(b)` by a log‑sum‑exp surrogate  
   `f_α(a)= (1/α)·log∑_k exp(α·b_k)` with `b = a∗a`.  Its gradient can be
   written in closed form using two convolutions.  Perform projected
   gradient steps with a linearly decaying step size while slowly
   increasing `α` (from 5 000 up to 50 000).  This phase gives a fast,
   global improvement.

4. **Phase B – Exact sub‑gradient with safe line‑search**  
   Work directly on `max(b)`.  At the current worst index `k` we move a
   small amount of mass `δ` from a *source* index that contributes most
   to `b_k` to a *target* index that contributes least.  The maximal
   admissible `δ` is obtained analytically from the linear effect on all
   convolution entries, guaranteeing that the peak never increases.
   The move is accepted only if the exact objective improves.

5. **Phase C – Greedy hill‑climbing (fine tuning)**  
   Re‑apply the line‑search move repeatedly, now allowing tiny random
   perturbations when stagnation is detected.  This cleans up the last
   remaining peaks.

All three phases keep the vector non‑negative and project it onto the
simplex `{x≥0, Σx=1}` after every update.  The best vector found so far
is evaluated with the supplied `evaluate_sequence` at the very end and
returned as a plain Python list.

The implementation uses only NumPy and the provided `evaluate_sequence`,
runs on a single CPU, and respects the 1000 s wall‑time limit.

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:                                           # numerical safety
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase A – Smoothed max gradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_max_phase(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent on a smooth approximation of max(b)."""
    start = time.time()
    # step‑size schedule
    step_start = 0.15
    step_end = 0.001
    # α (smoothness) schedule
    alpha_start = 5_000.0
    alpha_end = 50_000.0

    it = 0
    while time.time() < deadline:
        # ----- compute convolution, smooth weights and gradient -----
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max          # exact objective for monitoring

        # keep best
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth {it:7d}] new best = {best_val:.6f}")

        # log‑sum‑exp smoothing (stable)
        diff = b - cur_max                     # ≤ 0
        progress = (time.time() - start) / (deadline - start)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        w = np.exp(alpha * diff)               # w_i ∈ (0,1]
        w_sum = w.sum()
        if w_sum == 0.0:                       # pathological safety check
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        conv1 = _conv_fft(w, a)                # length 3·n‑2
        conv2 = _conv_fft(w, a[::-1])
        grad = 2.0 * (conv1[n - 1:2 * n - 1] + conv2[n - 1:2 * n - 1])

        # ----- step and projection -----
        step = step_start * (1.0 - progress) + step_end * progress
        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Exact sub‑gradient descent with safe line‑search
# ----------------------------------------------------------------------
def _exact_subgrad_phase(a: np.ndarray,
                         n: int,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float) -> tuple[np.ndarray, float]:
    """Move mass from the most responsible source to a weak target,
    limiting the step so that the current worst convolution never grows."""
    start = time.time()
    it = 0
    while time.time() < deadline:
        # current convolution and objective
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max

        # record best
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Exact {it:7d}] new best = {best_val:.6f}")

        # index of the current peak
        k = int(np.argmax(b))

        # ----- contributions to the peak (sub‑gradient) -----
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))
        if source == target:
            target = (target + 1) % n

        # ----- compute safe Δ (line‑search) -----
        # linear effect L = 2·a shifted by target – 2·a shifted by source
        L = np.zeros(2 * n - 1)
        L[target:target + n] += 2.0 * a
        L[source:source + n] -= 2.0 * a

        # allowable Δ so that no entry exceeds the current maximum
        pos = L > 1e-18
        if np.any(pos):
            bound = (cur_max - b[pos]) / L[pos]          # ≥ 0
            safe_delta = np.min(bound)
        else:
            safe_delta = a[source]                        # no increasing direction

        # final Δ is also limited by the mass that exists at `source`
        delta = min(a[source], safe_delta * 0.95)        # a little safety margin
        if delta < 1e-12:                                 # fall back to a tiny random move
            src_candidates = np.where(a > 1e-12)[0]
            if src_candidates.size == 0:
                break
            source = int(np.random.choice(src_candidates))
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)
            delta = a[source] * 0.02 * np.random.rand()
            if delta < 1e-12:
                it += 1
                continue

        # ----- incremental update of convolution (O(n)) -----
        a_old = a.copy()
        b_new = b.copy()
        # linear terms
        b_new[target:target + n] += 2.0 * delta * a_old
        b_new[source:source + n] -= 2.0 * delta * a_old
        # quadratic terms
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # ----- accept if we really improve -----
        if new_val < cur_val - 1e-12:
            a = a_old
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
        else:
            # If the computed safe Δ does not improve (quadratic terms can hurt),
            # simply halve it and retry a few times.
            trial_delta = delta
            improved = False
            for _ in range(5):
                trial_delta *= 0.5
                if trial_delta < 1e-14:
                    break
                b_test = b.copy()
                b_test[target:target + n] += 2.0 * trial_delta * a_old
                b_test[source:source + n] -= 2.0 * trial_delta * a_old
                b_test[2 * target] += trial_delta * trial_delta
                b_test[2 * source] += trial_delta * trial_delta
                b_test[source + target] -= 2.0 * trial_delta * trial_delta
                if b_test.max() < cur_max - 1e-12:
                    a = a_old
                    a[source] -= trial_delta
                    a[target] += trial_delta
                    b = b_test
                    cur_max = b.max()
                    cur_val = 2.0 * n * cur_max
                    improved = True
                    break
            if not improved:
                # random small perturbation to escape possible plateaus
                a = _project_simplex(a + np.random.normal(scale=0.001, size=n))

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Greedy hill‑climbing (fine tuning)
# ----------------------------------------------------------------------
def _hillclimb_phase(a: np.ndarray,
                     n: int,
                     deadline: float,
                     best_a: np.ndarray,
                     best_val: float) -> np.ndarray:
    """Deterministic hill climbing with the same safe line‑search as in
    the exact phase, plus occasional random perturbations when stagnating."""
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 200_000

    while time.time() < deadline:
        # contribute vector for the current worst index
        k = max_idx
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))
        if source == target:
            target = (target + 1) % n

        # safe Δ (same logic as in the exact phase)
        L = np.zeros(2 * n - 1)
        L[target:target + n] += 2.0 * a
        L[source:source + n] -= 2.0 * a

        pos = L > 1e-18
        if np.any(pos):
            bound = (cur_max - b[pos]) / L[pos]
            safe_delta = np.min(bound)
        else:
            safe_delta = a[source]

        delta = min(a[source], safe_delta * 0.90)
        if delta < 1e-12:
            delta = a[source] * 0.02 * np.random.rand()
            if delta < 1e-12:
                it += 1
                continue

        # incremental update
        a_old = a.copy()
        b_new = b.copy()
        b_new[target:target + n] += 2.0 * delta * a_old
        b_new[source:source + n] -= 2.0 * delta * a_old
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            a = a_old
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb and reset
            a = best_a.copy()
            a += np.random.normal(scale=0.003, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")
        it += 1
    return best_a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence`.  The routine respects the 1000 s wall‑time limit
    and combines three optimisation phases.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5            # safety margin

    # ------------------------------------------------------------------
    # Problem size – a little larger than the previous 8000‑item construction
    # ------------------------------------------------------------------
    n = 12_000

    # ------------------------------------------------------------------
    # Initialisation – try to reuse the previously discovered construction
    # ------------------------------------------------------------------
    old = globals().get("height_sequence_1", None)
    if old is not None:
        old = np.asarray(old, dtype=float)
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    else:
        a = np.random.rand(n)

    a = _project_simplex(a)            # enforce Σa = 1
    best_a = a.copy()
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Split the budget among the three phases
    # ------------------------------------------------------------------
    total = deadline - start_time
    t_smooth = start_time + 0.40 * total
    t_exact   = start_time + 0.75 * total
    t_hill    = deadline

    # -----------------
    # Phase A – smooth
    # -----------------
    best_a, best_val = _smooth_max_phase(
        a, n, t_smooth, best_a, best_val
    )
    a = best_a.copy()

    # -----------------
    # Phase B – exact sub‑gradient with line‑search
    # -----------------
    best_a, best_val = _exact_subgrad_phase(
        a, n, t_exact, best_a, best_val
    )
    a = best_a.copy()

    # -----------------
    # Phase C – greedy hill‑climbing
    # -----------------
    best_a = _hillclimb_phase(
        a, n, t_hill, best_a, best_val
    )

    # ------------------------------------------------------------------
    # Final evaluation (the secure wrapper may clamp values)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:                       # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
