"""
Optimiser for `evaluate_sequence`.

The objective is 2·n·max(a∗a) when the vector is normalised
(∑a = 1).  The global lower bound is 1 (a∗a everywhere equal to the
average 1/(2n‑1)).  The programme works in three stages:

1. **Smart initialisation** – a small 2‑parameter family
   a_i ∝ (1‑λ)·1 + λ·((i+½)(n‑½‑i)+ε)^‑p is scanned on a coarse grid.
   The best member (in terms of the true objective) is taken as the
   start point.

2. **Adam‑L2 flattening** – we minimise the variance
   Σ(b‑b̄)² (b = a∗a) with an Adam optimiser while keeping a on the
   probability simplex.  Each iteration the gradient is
   `2·[conv(r,a)+conv(r,a[::-1])][n‑1:2·n‑1]` with `r=b‑b̄`.  The step‑size
   decays linearly, and after every update the vector is symmetrised
   (a←(a+a[::-1])/2) and projected onto the simplex.

3. **Simulated‑annealing hill climbing** – a greedy move transfers a
   small amount of mass from the index that contributes most to the
   current worst convolution value to the index that contributes least.
   Moves that increase the objective are accepted with probability
   `exp(-(Δval)/T)` where the temperature T cools linearly to zero.
   The full convolution is updated incrementally (O(n)) so the phase
   runs fast even for n≈8000.

The best vector found is finally passed to the provided
`evaluate_sequence` function and returned as a plain Python list.
All timing respects the 1000 s wall‑time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Length = 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.
    Length = len(x)+len(y)‑1.
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:                       # numerical safety
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Objective utilities (sum = 1 assumed)
# ----------------------------------------------------------------------
def _objective_max(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(a∗a) with a already normalised (∑a = 1)."""
    b = _full_convolution(a)
    return 2.0 * n * b.max()


def _objective_l2(a: np.ndarray, n: int) -> float:
    """L2 loss Σ(b‑b̄)² (used only for flattening, not final score)."""
    b = _full_convolution(a)
    avg = 1.0 / (2 * n - 1)                     # mean of b when ∑a = 1
    r = b - avg
    return np.dot(r, r)


# ----------------------------------------------------------------------
# Phase 1 – Adam optimiser that minimises L2‑flattening loss
# ----------------------------------------------------------------------
def _adam_l2_phase(a: np.ndarray,
                   n: int,
                   deadline: float,
                   best_a: np.ndarray,
                   best_val: float) -> tuple[np.ndarray, float]:
    """Adam optimisation of Σ(b‑b̄)² with simplex projection & symmetry."""
    start = time.time()
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    base_lr = 0.50
    min_lr = 0.01

    it = 0
    while time.time() < deadline:
        # convolution and L2 loss
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max                    # true objective
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[AdamL2 {it:7d}] new best = {best_val:.6f}")

        # gradient of Σ(b‑b̄)²
        avg = 1.0 / (2 * n - 1)
        r = b - avg
        # two convolutions (same formula as in the original code)
        conv1 = _conv_fft(r, a)               # length 3·n‑2
        conv2 = _conv_fft(r, a[::-1])
        start_idx = n - 1
        end_idx = 2 * n - 1
        grad = 2.0 * (conv1[start_idx:end_idx] + conv2[start_idx:end_idx])

        # Adam update with linear learning‑rate schedule
        progress = (time.time() - start) / (deadline - start)
        lr = max_lr = base_lr * (1.0 - progress) + min_lr * progress

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        t = it + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep the vector symmetric (helps flattening) and project
        a_candidate = (a_candidate + a_candidate[::-1]) * 0.5
        a = _project_simplex(a_candidate)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Sub‑gradient descent on max(b) (exact objective)
# ----------------------------------------------------------------------
def _subgradient_phase(a: np.ndarray,
                       n: int,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    """Standard sub‑gradient descent for min max(b)."""
    start = time.time()
    max_step = 0.10
    min_step = 1e-4
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        k = int(np.argmax(b))                 # index of the current worst conv.
        cur_max = b[k]
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Subgrad {it:7d}] new best = {best_val:.6f}")

        # sub‑gradient of max(b)
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                grad[mid] = 2.0 * a[mid]

        # linear step‑size schedule
        progress = (time.time() - start) / (deadline - start)
        step = max_step * (1.0 - progress) + min_step * progress

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – Simulated‑annealing greedy hill‑climber
# ----------------------------------------------------------------------
def _simulated_hillclimb(a: np.ndarray,
                         n: int,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float) -> np.ndarray:
    """
    Hill climbing that may accept uphill moves according to a temperature.
    Uses O(n) incremental updates of the convolution.
    """
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    it = 0
    # temperature schedule: start = 1e-3 * cur_val, end = 0
    T0 = cur_val * 1e-3
    start = time.time()
    while time.time() < deadline:
        # --- contribution of each coordinate to the current worst index ---
        k = max_idx
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        # fallback to random pair if degenerate
        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-14:
            # choose another source with appreciable mass
            nonzero = np.where(a > 1e-14)[0]
            if nonzero.size == 0:
                break
            source = int(np.random.choice(nonzero))

        # amount to move (up to 25 % of source mass)
        delta = a[source] * 0.25 * np.random.rand()
        if delta <= 0.0:
            continue

        # ---- incremental update of the full convolution (O(n)) ----
        b_new = b.copy()
        # cross‑terms 2·δ·a
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        # pure‑δ² terms
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max
        # temperature decreasing linearly
        progress = (time.time() - start) / (deadline - start)
        T = T0 * (1.0 - progress)

        accept = False
        if new_val < cur_val - 1e-12:
            accept = True
        elif T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
            accept = True

        if accept:
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")

        it += 1
    return best_a

                best_val = val
                best_a = a.copy()
                print(f"[Init] p={p:4.2f} λ={lam:4.2f} → obj={val:.6f}")

    print(f"[Init] best initial objective = {best_val:.6f}")
    return best_a


# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence`.  The routine fits the whole 1000 s wall‑time
    budget and combines a parametrised initial guess, an Adam‑L2
    flattening phase, sub‑gradient descent on the exact max‑objective,
    and a simulated‑annealing hill‑climber.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # problem size – 8000 gives a good trade‑off between flexibility and speed
    n = 8000

    # ------------------------------------------------------------------
    # 1. smart initial guess
    # ------------------------------------------------------------------
    a = _initial_guess(n)

    # store the best solution seen so far
    best_a = a.copy()
    best_val = _objective_max(a, n)

    # ------------------------------------------------------------------
    # split remaining budget into three phases
    # ------------------------------------------------------------------
    total = deadline - start_time
    t_phase1 = start_time + 0.40 * total   # Adam L2 flattening
    t_phase2 = start_time + 0.65 * total   # sub‑gradient descent
    t_phase3 = deadline                    # simulated annealing hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimiser on Σ(b‑b̄)²
    # ------------------------------------------------------------------
    best_a, best_val = _adam_l2_phase(a, n, t_phase1, best_a, best_val)
    a = best_a.copy()

    # ------------------------------------------------------------------
    # Phase 2 – Sub‑gradient descent on max(b)
    # ------------------------------------------------------------------
    best_a, best_val = _subgradient_phase(a, n, t_phase2, best_a, best_val)
    a = best_a.copy()

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    best_a = _simulated_hillclimb(a, n, t_phase3, best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation with the official wrapper (clipping, etc.)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
