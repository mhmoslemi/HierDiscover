# Helper: projection onto the probability simplex {x≥0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex using the algorithm of Duchi et al."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear convolution via FFT (real, full length)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution a*b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv

    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * np.max(conv)

        a = _project_simplex(a_candidate)

        if stagn > 5000:

            a = _project_simplex(a + noise)
            stagn = 0
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta2_pow = 1.0
            print(f"[phase2] perturb at iter {it2}")

    # Phase 3 – simulated‑annealing fine‑tuning.

    cur = best_a.copy()
    cur_score = best_score

    Tmin = 1e-7

"""
Strategic search for a step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the full linear convolution.  Because the objective is
scale‑invariant we keep the vector on the probability simplex
(`sum(a)=1, a_i≥0`).  The algorithm proceeds in four stages:

1. **Warm start** – reuse a previously found solution if it exists,
   otherwise generate a large pool of analytic shapes (beta, exponential,
   power‑law, Gaussian, half‑sine, triangle) with random parameters and
   keep the best after normalising to the simplex.

2. **Smooth‑max gradient descent** – minimise a soft‑max approximation of
   `max(conv)` with momentum, projecting onto the simplex after each step.

3. **Exact sub‑gradient descent** – use the true sub‑gradient (averaged
   over all active maximal entries) together with Adam.

4. **Simulated‑annealing fine‑tuning** – repeatedly transfer a small amount
   of mass from one index to another and accept the move according to a
   Metropolis rule.  Temperature decays linearly; occasional global jitter
   helps escape plateaus.

All phases respect the overall time budget (default 1000 s).  The best
vector seen is returned as a list of floats clipped to `[0, 1000]`.  The
result is also stored in the global variable `height_sequence_1` for future
warm‑starts.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x) = 1}.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex using the algorithm of Duchi et al."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = cssv[-1] / n
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for the sizes we use).
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution a * b using a real FFT."""
    size = a.shape[0] + b.shape[0] - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Objective for a vector that already satisfies sum(a)=1.
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2 * n * max(conv(a, a)) (with n = len(a))."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Main entry point required by the grader.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """Return a non‑negative height sequence that (approximately) minimizes
    the evaluation function.  The routine respects a `budget_s` second time
    limit and prints intermediate scores.
    """
    start_time = time.time()
    # Keep a small safety margin for the final call to evaluate_sequence.
    deadline = start_time + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem dimensionality – a size that balances richness and speed.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Tiny wrapper around the objective (vector is already normalised).
    # ------------------------------------------------------------------
    def _eval(a: np.ndarray) -> float:
        return _objective(a)

    # ------------------------------------------------------------------
    # Initialise: try to reuse a previously discovered sequence, otherwise
    # start from a broad collection of analytic shapes.
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # Reuse previous warm start if present.
    if 'height_sequence_1' in globals():
        try:
            seq = np.asarray(height_sequence_1, dtype=np.float64)
            if seq.shape[0] != n:
                if seq.shape[0] > n:
                    seq = seq[:n]
                else:
                    seq = np.pad(seq, (0, n - seq.shape[0]))
            seq = np.maximum(0.0, seq)
            s = seq.sum()
            if s > 0:
                a0 = seq / s
                sc = _eval(a0)
                if sc < best_score:
                    best_score = sc
                    best_a = a0.copy()
                    print(f"[init] reused height_sequence_1, score={best_score:.12f}")
        except Exception:
            pass

    # Uniform distribution (simple baseline).
    a_uni = np.full(n, 1.0 / n, dtype=np.float64)
    sc_uni = _eval(a_uni)
    if sc_uni < best_score:
        best_score = sc_uni
        best_a = a_uni.copy()
        print(f"[init] uniform baseline, score={best_score:.12f}")

    # ------------------------------------------------------------------
    # 1) Random search over several analytic families.
    # ------------------------------------------------------------------
    max_search_iters = 800
    xs = np.linspace(0.0, 1.0, n, dtype=np.float64)

    for it in range(max_search_iters):
        if time.time() > deadline:
            break

        family = rng.integers(0, 6)
        if family == 0:                     # Beta distribution (both parameters random)
            alpha = 10 ** rng.uniform(-1.0, 1.0)   # 0.1 … 10
            beta_ = 10 ** rng.uniform(-1.0, 1.0)
            eps = 1e-12
            log_pdf = (alpha - 1) * np.log(np.clip(xs, eps, 1 - eps)) + \
                      (beta_ - 1) * np.log(np.clip(1 - xs, eps, 1 - eps))
            pdf = np.exp(log_pdf - np.max(log_pdf))
            a = pdf / pdf.sum()
        elif family == 1:                   # Exponential decay
            lam = rng.uniform(5e-4, 2e-2)
            a = np.exp(-lam * np.arange(n, dtype=np.float64))
            a = a / a.sum()
        elif family == 2:                   # Power‑law decreasing
            p = rng.uniform(0.2, 2.5)
            a = (np.arange(n, dtype=np.float64) + 1) ** (-p)
            a = a / a.sum()
        elif family == 3:                   # Gaussian centred in the middle
            sigma = rng.uniform(n / 8.0, n / 2.0)
            centre = (n - 1) / 2.0
            a = np.exp(-0.5 * ((np.arange(n, dtype=np.float64) - centre) / sigma) ** 2)
            a = a / a.sum()
        elif family == 4:                   # Half‑sine (always positive)
            a = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
            a = a / a.sum()
        else:                               # Simple triangle
            a = np.minimum(np.arange(n, dtype=np.float64),
                           n - 1 - np.arange(n, dtype=np.float64)) + 1.0
            a = a / a.sum()

        # Random perturbation to increase diversity.
        if rng.random() < 0.3:
            noise = rng.normal(scale=0.05, size=n)
            a = np.maximum(0.0, a * (1.0 + noise))
            a = a / a.sum()

        sc = _eval(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[search] iter {it:04d} new best {best_score:.12f}")

    if best_a is None:                     # safety fallback
        best_a = a_uni.copy()
        best_score = _eval(best_a)

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max gradient descent with momentum.
    # ------------------------------------------------------------------
    phase1_end = time.time() + 0.45 * (deadline - time.time())
    a = best_a.copy()
    eta = 0.5
    eta_min = 1e-8
    eta_decay = 0.9995
    eps = 0.2
    eps_min = 1e-8
    eps_decay = 0.9995
    momentum = 0.9
    vel = np.zeros_like(a)
    stagn = 0
    it1 = 0

    while time.time() < phase1_end:
        it1 += 1

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"[phase1] iter {it1:06d} new best {best_score:.12f}")
        else:
            stagn += 1

        w = np.exp((conv - max_conv) / eps)
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        vel = momentum * vel + (1.0 - momentum) * g
        a_new = a - eta * vel
        a = _project_simplex(a_new)

        eta = max(eta_min, eta * eta_decay)
        eps = max(eps_min, eps * eps_decay)

        if stagn > 5000:
            noise = rng.normal(scale=0.001, size=n)
            a = _project_simplex(a + noise)
            stagn = 0
            eta = max(eta_min, eta * 2.0)
            print(f"[phase1] perturb at iter {it1}")

    # ------------------------------------------------------------------
    # Phase 2 – exact max‑sub‑gradient descent with Adam.
    # ------------------------------------------------------------------
    phase2_end = time.time() + 0.30 * (deadline - time.time())
    beta1, beta2 = 0.9, 0.999
    lr = 0.10
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow, beta2_pow = 1.0, 1.0
    it2 = 0
    stagn = 0

    while time.time() < phase2_end:
        it2 += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        score = 2.0 * n * max_conv
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagn = 0
            print(f"[phase2] iter {it2:06d} new best {best_score:.12f}")
        else:
            stagn += 1

        tol = 1e-12
        active = np.flatnonzero(conv >= max_conv - tol)
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        else:
            w[np.argmax(conv)] = 1.0

        rev_a = a[::-1]
        g = 2.0 * _full_convolution_fft(rev_a, w)[n - 1:n - 1 + n]

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_candidate = a - step
        a = _project_simplex(a_candidate)

        if stagn > 5000:
            noise = rng.normal(scale=0.0005, size=n)
            a = _project_simplex(a + noise)
            stagn = 0
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta2_pow = 1.0
            print(f"[phase2] perturb at iter {it2}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing fine‑tuning.
    # ------------------------------------------------------------------
    cur = best_a.copy()
    cur_score = best_score
    T0 = 1e-3
    Tmin = 1e-7
    it_sa = 0
    max_transfer = 0.02

    while time.time() < deadline:
        it_sa += 1
        T = max(Tmin, T0 * (1.0 - it_sa / 250_000))

        i, j = rng.choice(n, size=2, replace=False)
        if cur[i] <= 0.0:
            continue
        delta = rng.uniform(0.0, min(cur[i], max_transfer))

        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        conv_new = _full_convolution_fft(new, new)
        new_score = 2.0 * n * conv_new.max()

        if new_score < cur_score:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_score - cur_score) / T)

        if accept:
            cur = new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = cur.copy()
                print(f"[SA] iter {it_sa:07d} new best {best_score:.12f}")

        if it_sa % 50000 == 0:
            cur = _project_simplex(cur + rng.normal(scale=0.001, size=n))
            cur_score = 2.0 * n * _full_convolution_fft(cur, cur).max()

    # ------------------------------------------------------------------
    # Return the best sequence in the format expected by evaluate_sequence.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_a.tolist()]

    # Store the result for possible future warm‑starts.
    globals()['height_sequence_1'] = final_seq

    # Optional final official score (may be unavailable in the testing harness).
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] evaluate_sequence not available")

    return final_seq
