def _project_simplex(v):
    # ensure nonnegative and sum to 1
    # Algorithm from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _annealing_full(a, cur_val, deadline):
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_val = cur_val
    T = max(cur_val * 0.01, 1e-8)  # initial temperature
    decay = 0.9999

    while time.time() < deadline:
        # decide move type: swap (50%) or transfer (50%)
        if rng.random() < 0.5:
            i, j = rng.integers(0, n, size=2)
            if i == j: continue
            a[i], a[j] = a[j], a[i]
            new_val = _eval_fast(a)
            delta = new_val - cur_val
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
            else:
                a[i], a[j] = a[j], a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0: continue
            # transfer up to a[i]
            delta_mass = a[i] * (0.2 * rng.random())
            a[i] -= delta_mass
            a[j] += delta_mass
            # no need to project since sum unchanged and nonneg may become slightly negative if delta_mass > a[i]; but we limit.
            new_val = _eval_fast(a)
            delta = new_val - cur_val
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
            else:
                # revert
                a[i] += delta_mass
                a[j] -= delta_mass
        T *= decay
        # occasional re-heat: if T too low, set to something
        if T < 1e-12:
            T = max(cur_val * 0.01, 1e-12)
    return best_a, best_val

def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Compute smoothed max of the convolution (log-sum-exp) and its gradient w.r.t. a.
    Returns (L, grad) where L = max_conv + eps*logsumexp((conv - max_conv)/eps) which approximates max_conv.
    a is assumed to be non-negative and sum to 1.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)  # length 2n-1
    max_conv = conv.max()
    # shift for numerical stability
    shifted = (conv - max_conv) / eps
    # log-sum-exp
    log_sum = logsumexp(shifted)
    L = max_conv + eps * log_sum
    
    # compute weights p = exp(shifted) / sum_exp
    p = np.exp(shifted - log_sum)  # shape 2n-1
    
    # compute gradient: a + reverse(p) convolve a (segment)
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)  # length 3n-2
    seg = conv_ap[n - 1: 2 * n - 1]  # size n
    grad = a + seg[::-1]  # shape n
    return L, grad
```

"""
propose_candidate
-----------------
Hybrid optimisation for the step‑function minimisation problem.

1. Initialise from the previously discovered best construction (if
   available) and normalise it to ℓ₁‑norm = 1.
2. A few short L‑BFGS runs on a smooth log‑sum‑exp surrogate of the
   peak convolution find a good shape of the heights.
3. The obtained shape is anti‑symmetrically reordered (large ↔ small)
   – a cheap deterministic improvement.
4. A long simulated‑annealing phase (≈ 800 s) repeatedly performs
   *swaps* of two positions and *mass‑transfer* moves (moving a
   random fraction of the content of one index to another).  Moves are
   accepted if they improve the exact objective or with a Metropolis
   probability depending on a slowly cooling temperature.
5. Finally a few gradient‑descent steps on the same log‑sum‑exp
   surrogate (projected onto the probability simplex) provide a fine
   polish.
6. The best sequence found is returned as a plain Python list.  All
   steps respect the 1000 s wall‑clock budget and the sequence consists
   only of non‑negative numbers.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the smooth‑surrogate wrapper (avoid closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (full length 2·n‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)                # clamp tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast exact evaluation – same as evaluate_sequence(a.tolist())."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smooth surrogate used by the L‑BFGS optimiser (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε = 2·n·max(conv) + 2·n·ε·log∑exp((conv−max)/ε)
    together with its gradient w.r.t. the soft‑max parameters w.
    """
    # soft‑max → a ≥ 0, Σa = 1
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # self‑convolution
    conv = _fft_convolve_self(a)          # length 2·n‑1
    max_conv = conv.max()

    # smooth max (log‑sum‑exp)
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed‑form)
    weights = np.exp(shifted)
    p = weights / weights.sum()            # probability distribution over shifts

    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by scipy.optimize.minimize (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# bind the FFT convolution helper to the function (so it can be called from inside)
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Direct smooth max and its gradient w.r.t. a (used for final polishing)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Returns a smooth approximation of max(a*a) (log‑sum‑exp version)
    and its gradient w.r.t. the vector a.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)           # length 2·n‑1
    max_conv = conv.max()

    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = max_conv + eps * log_sum           # smoothed max

    # weights for the gradient
    p = np.exp(shifted - log_sum)          # sum(p) = 1

    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)          # length 3·n‑2
    seg = conv_ap[n - 1: 2 * n - 1]         # n entries
    grad = a + seg[::-1]                   # ∂L/∂a
    return L, grad


# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex Σx = 1, x ≥ 0
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the (non‑negative) simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – all zeros
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple re‑ordering heuristic (large ↔ small)
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place the largest values on the left and the smallest on the right."""
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large to the left
        new_a[n - 1 - i] = sorted_vals[i]      # small to the right

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# Simulated annealing on permutations + mass‑transfer moves
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float, deadline: float):
    """
    Classic SA that mixes swaps (permutation) and mass‑transfer moves.
    Returns the best sequence found before ``deadline``.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # initial temperature – 2 % of the current value (heuristic)
    T = max(cur_val * 0.02, 1e-8)
    decay = 0.99995
    iteration = 0

    while time.time() < deadline:
        if rng.random() < 0.5:
            # ---- swap move ----
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            new_val = _eval_fast(a)
            delta = new_val - cur_val
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA swap] new best = {best_val:.6f} (iter {iteration})")
            else:
                a[i], a[j] = a[j], a[i]        # revert
        else:
            # ---- mass‑transfer move ----
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            # transfer a random fraction (up to 20 %) of a[i] to a[j]
            delta_mass = a[i] * (0.2 * rng.random())
            a[i] -= delta_mass
            a[j] += delta_mass
            new_val = _eval_fast(a)
            delta = new_val - cur_val
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA transfer] new best = {best_val:.6f} (iter {iteration})")
            else:
                # revert
                a[i] += delta_mass
                a[j] -= delta_mass

        # cooling
        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        iteration += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Final gradient‑descent polishing (smooth max surrogate)
# ----------------------------------------------------------------------
def _gradient_refine(a0: np.ndarray, deadline: float):
    """
    A few gradient‑descent steps on the log‑sum‑exp surrogate, with
    projection onto the simplex after each step.
    """
    a = a0.copy()
    cur_val = _eval_fast(a)
    eps = 1e-3                # start smoothing
    step_factor = 0.5

    while time.time() < deadline:
        # smooth max and gradient
        L, grad = _smooth_max_and_grad(a, eps)

        # simple normalised step
        step = step_factor * cur_val / (np.linalg.norm(grad) + 1e-12)
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        new_val = _eval_fast(a_new)
        if new_val < cur_val:
            a, cur_val = a_new, new_val
            print(f"[refine] improved to {cur_val:.6f}")

        # slowly tighten smoothing and step size
        eps = max(1e-12, eps * 0.9)
        step_factor *= 0.99

    return a, cur_val
