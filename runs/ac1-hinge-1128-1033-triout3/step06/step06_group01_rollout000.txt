The optimizer works in two stages.

1. **Iterative linear‑programming (LP) refinement**  
   For a fixed length `n` we keep a non‑negative vector `f` with
   `sum(f)=1`.  
   Let `M = max(f*f)`.  The LP
   ```
   maximize   1ᵀ g
   subject to g ≥ 0
              f * g ≤ M               (2n‑1 linear constraints)
   ```
   finds a vector `g` whose convolution with the current `f` does not
   exceed `M`.  After scaling `g` to the same ℓ₁‑norm as `f` we
   replace `f ← (1‑t)·f + t·g` (small mixing weight `t`).  Because the
   convolution bound stays valid, the maximal convolution value never
   increases – the objective `2 n·max(f*f)` is monotone decreasing.
   The LP is solved with SciPy’s HiGHS interior‑point solver; the
   constraint matrix is built once (it only depends on `n`) and reused.

2. **Permutation polishing**  
   After the LP stage we have a multiset of heights that is already
   close to optimal.  A fast FFT based evaluator is used together with
   simulated‑annealing (swap moves) and a cheap hill‑climbing pass
   (mass‑transfer + swaps) to find a good ordering of the heights.
   The ordering influences the self‑convolution peak, therefore these
   final steps often shave a few × 10⁻³ off the objective.

The whole procedure respects the 1000 s budget, prints intermediate
information and finally returns a plain Python list that can be fed to
`evaluate_sequence`.  The implementation does **not** contain closures,
uses only top‑level functions and is fully self‑contained.  

```python
"""
Hybrid LP + permutation optimisation for the step‑function problem.

Stage 1 – LP refinement:
    * keep a vector f with sum(f)=1
    * compute M = max(f*f)
    * solve   max Σ g_i   s.t. g_i≥0  and  (f*g)_k ≤ M  (k=0…2n‑2)
    * scale g to sum 1 and mix: f ← (1‑t)f + t·g  (t≈0.2)
    * repeat until no improvement or time runs out

Stage 2 – permutation polishing:
    * fast FFT evaluator of the exact objective
    * simulated‑annealing on swaps
    * final hill‑climbing (mass transfers + swaps)

The algorithm runs under the 1000 s budget and returns the best
sequence found.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# FFT based convolution / fast evaluation
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)               # clamp tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Exact objective from ``evaluate_sequence`` but much faster.
    The vector a must be non‑negative and already respects clipping.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Re‑ordering heuristic: pair large with small values
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic re‑ordering that puts the largest values on the left
    half and the smallest on the right half, alternating them.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = s[n - 1 - i]          # large values left side
        out[n - 1 - i] = s[i]          # small values right side
    if n % 2:
        out[half] = s[half]
    return out


# ----------------------------------------------------------------------
# Simulated annealing on permutations (swap moves only)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """
    Classic SA on the ordering of ``a``.
    Swaps preserve the ℓ₁‑norm, only the arrangement changes.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val
    T = max(cur_val * 0.02, 1e-6)      # initial temperature
    decay = 0.99995

    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose swap
        a[i], a[j] = a[j], a[i]
        new_val = _eval_fast(a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f} (iter {it})")
        else:
            # revert
            a[i], a[j] = a[j], a[i]

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (mass transfers + swaps)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Helper: static template for the convolution constraint matrix
# ----------------------------------------------------------------------
def _conv_matrix_template(n: int):
    """
    Build the row/col index arrays for the matrix that represents
    (f * g) ≤ M.  For a given f the data array is simply
    ``np.repeat(f, n)``.
    """
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n)   # rows where column i participates
        cols[pos:pos + n] = i
        pos += n
    return rows, cols

        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # Report final value using the official evaluator
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    # Return as plain Python list
    return best_a.tolist()
```
