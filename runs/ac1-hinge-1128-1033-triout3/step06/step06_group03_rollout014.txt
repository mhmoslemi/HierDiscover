"""
Propose a step‑function (non‑negative heights) that (approximately) minimizes
``evaluate_sequence``.  

The algorithm consists of two phases:

1. **Projected gradient descent on the L2 distance between the
   autocorrelation ``a*a`` and the uniform target**.  Making the
   convolution flat also reduces its maximum.  The gradient of
   ``∑(conv−c)²`` can be computed with two FFT convolutions, after which a
   step is taken and the vector is projected onto the probability simplex
   (Duchi et al. 2008).  A diminishing step‑size and occasional random
   perturbations avoid stagnation.

2. **Guided hill‑climbing** – identical to the previous state‑of‑the‑art
   routine but run on the best solution found in phase 1.  It repeatedly
   removes a little mass from the pair of indices that contributes most
   to the current worst lag and deposits it on two low‑mass positions,
   accepting only improvements.

Both phases respect the 1000 s budget and keep the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility: next power of two for FFT zero‑padding
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for FFT length)."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Full (linear) convolution – generic version (no clipping)
# ----------------------------------------------------------------------
def _full_convolution_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors (no post‑processing)."""
    n, m = int(x.shape[0]), int(y.shape[0])
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Autocorrelation (a * a) – with tiny‑negative clipping for safety
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    # eliminate round‑off negatives – they should be zero
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008) – projects onto {x≥0, Σx = 1}
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (identical to the previous state‑of‑the‑art)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        val = evaluate_sequence(new_a.tolist())
        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Optimise the step‑function heights using a projected gradient descent on
    the L2 deviation of the autocorrelation from the uniform target, followed
    by a guided hill‑climbing phase.
    Returns the best list of non‑negative heights before the time budget elapses.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = a0.shape[0]
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    except Exception:
        n = 4000
        a0 = np.full(n, 1.0 / n)

    a = a0.copy()
    a = a / a.sum()                         # enforce Σa = 1

    # ------------------------------------------------------------------
    # Pre‑computed constants
    # ------------------------------------------------------------------
    avg_target = 1.0 / (2 * n - 1)           # theoretical average of the convolution
    size1 = _next_pow2(2 * n - 1)            # FFT size for a * a
    size2 = _next_pow2(3 * n)                # FFT size for diff * a_rev

    # initial evaluation
    conv = _full_convolution(a)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – projected gradient descent on L2 error of the convolution
    # ------------------------------------------------------------------
    step0 = 0.2
    it = 0
    no_improve = 0

    while time.time() < deadline * 0.7:
        it += 1

        # --- current autocorrelation ---
        fa = np.fft.rfft(a, size1)
        conv = np.fft.irfft(fa * fa, size1)[:2 * n - 1]
        np.maximum(conv, 0, out=conv)               # tiny‑negative clipping
        max_b = conv.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[grad {it}] new best = {best_val:.6f}")

        # --- gradient of Σ(conv - avg)² ---
        diff = conv - avg_target                     # length 2n‑1
        a_rev = a[::-1]
        fd = np.fft.rfft(diff, size2)
        fra = np.fft.rfft(a_rev, size2)
        grad_full = np.fft.irfft(fd * fra, size2)[:3 * n - 1]
        # slice corresponding to indices n‑1 … 2n‑2
        grad = 4.0 * grad_full[n - 1 : n - 1 + n]

        # --- projected gradient step ---
        step = step0 / np.sqrt(it)
        cand = a - step * grad
        cand = _simplex_projection(cand)

        # evaluate candidate
        fc = np.fft.rfft(cand, size1)
        conv_c = np.fft.irfft(fc * fc, size1)[:2 * n - 1]
        np.maximum(conv_c, 0, out=conv_c)
        cand_val = 2.0 * n * conv_c.max()

        if cand_val < cur_val - 1e-14:
            a = cand
            cur_val = cand_val
            no_improve = 0
            step0 = min(step0 * 1.02, 2.0)          # modestly increase aggressiveness
        else:
            no_improve += 1
            step0 *= 0.9
            if no_improve >= 200:
                # random perturbation to escape flat region
                pert = rng.normal(scale=0.005, size=n)
                a = _simplex_projection(a + pert)
                no_improve = 0
                step0 = 0.2

    # ------------------------------------------------------------------
    # Phase 2 – guided hill‑climbing refinement (uses the best found so far)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")

    return best_a.tolist()
