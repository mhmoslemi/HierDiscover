def smoothmax_objective(w, eps):
    # a = softmax(w)
    w_shift = w - np.max(w)
    exp_w = np.exp(w_shift)
    a = exp_w / np.sum(exp_w)
    b = _full_convolution(a)
    max_b = b.max()
    # smooth max: S = max_b + eps * np.log(np.sum(np.exp((b - max_b)/eps)))
    shifted = (b - max_b) / eps
    exp_shift = np.exp(shifted)
    sum_exp = np.sum(exp_shift)
    S = max_b + eps * np.log(sum_exp)
    obj = 2.0 * n * S
    # gradient w.r.t a
    p = exp_shift / sum_exp   # derivative of S w.r.t b
    # compute grad_a = 2 * n * 2? Actually we need derivative of obj: obj = 2*n*S. So grad_S = 2*n.
    # but S derivative w.r.t b is p, so dObj/da = 2*n * sum_k p_k * ∂b_k/∂a.
    # ∂b/∂a = 2 * conv(a, p_rev) -- Actually we need vector gradient length n.
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)   # length 2n-1
    segment = conv_ab[n-1:2*n-1]    # n entries
    grad_a = 4.0 * n * segment[::-1]   # as before

    dot = np.dot(grad_a, a)

"""
search_step_function.py

Optimises a non‑negative step function (list of heights) that minimises
`evaluate_sequence`.  The optimiser consists of three parts:

1.  A rich parametric sweep (uniform, linear, quadratic, symmetric
    beta‑shapes, sine‑powers, exponentials) to obtain a very good
    starting point.
2.  A sequence of L‑BFGS optimisations on a smooth approximation of
    the max‑convolution.  The variables are the soft‑max logits, so the
    simplex constraints are satisfied automatically.  The smoothing
    parameter ε is decreased geometrically, allowing a coarse‑to‑fine
    optimisation.
3.  An aggressive hill‑climber that repeatedly transfers a small amount
    of mass from indices that contribute heavily to the current maximal
    convolution entry to indices that contribute little.  For each
    source index a few target candidates are examined and the best
    improving move is taken.  If no improvement is seen for a long
    stretch the current solution is perturbed and the search resumes.

All phases respect the global time budget (default 1000 s).  The final
sequence is returned as a plain ``list[float]``; it is evaluated with
the provided ``evaluate_sequence`` wrapper.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of `a` with itself (length 2·len(a)‑1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σ a = 1)
# ----------------------------------------------------------------------
def _objective_value(a: np.ndarray) -> float:
    """Return the exact value of `evaluate_sequence` for a normalised vector."""
    n = a.shape[0]
    return float(2.0 * n * _full_convolution(a).max())


# ----------------------------------------------------------------------
# Parametric initialisation – a large palette of symmetric shapes
# ----------------------------------------------------------------------
def _parametric_initialisation(n: int):
    """Return the best vector among a large family of analytic shapes."""
    best_a = np.full(n, 1.0 / n)
    best_val = _objective_value(best_a)

    # linear decreasing (asymmetric)
    a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # quadratic decreasing (asymmetric)
    a = (np.arange(n, 0, -1, dtype=float) ** 2)
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # symmetric quadratic (i·(n‑i‑1))
    idx = np.arange(n, dtype=float)
    a = idx * (n - 1 - idx)
    a += 1e-12                    # avoid the all‑zero situation
    a /= a.sum()
    val = _objective_value(a)
    if val < best_val:
        best_a, best_val = a, val

    # beta‑like symmetric shapes  a_i ∝ (x(1‑x))^{p‑1}
    xs = (np.arange(n) + 0.5) / n        # avoid exact 0 / 1
    for p in np.linspace(1.1, 5.0, 9):   # p≈1 gives uniform, larger p gives a bell
        a = (xs * (1 - xs)) ** (p - 1)
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    # sin‑power families  a_i ∝ sin(π·x)^q
    for q in (1, 2, 3, 4):
        a = np.sin(np.pi * xs) ** q
        a = np.maximum(a, 0.0)
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    # exponential decays
    for r in (0.9999, 0.9995, 0.999, 0.998, 0.995, 0.99, 0.98, 0.95):
        a = r ** np.arange(n, dtype=float)
        a /= a.sum()
        val = _objective_value(a)
        if val < best_val:
            best_a, best_val = a, val

    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser on the simplex (log‑soft‑max parametrisation)
# ----------------------------------------------------------------------
def _smooth_lbfgs_optimize(a0: np.ndarray,
                           eps: float,
                           maxiter: int,
                           n: int,
                           deadline: float):
    """
    Optimises the smooth approximation of the max‑convolution using L‑BFGS.
    Parameters are the soft‑max logits w (i.e. a = softmax(w)).
    """
    from scipy.optimize import minimize

    w0 = np.log(a0 + 1e-12)                      # initialise logits

    # objective+gradient for minimise()
    def fun(w):
        # soft‑max (ensures Σa = 1, a ≥ 0)
        w_shift = w - np.max(w)
        exp_w = np.exp(w_shift)
        a = exp_w / np.sum(exp_w)

        # convolution and smooth‑max
        b = _full_convolution(a)
        max_b = b.max()
        shifted = (b - max_b) / eps
        exp_shift = np.exp(shifted)
        sum_exp = np.sum(exp_shift)
        smooth = max_b + eps * np.log(sum_exp)               # smooth max
        obj = 2.0 * n * smooth

        # soft‑max probabilities over the k‑indices
        p = exp_shift / sum_exp

        # gradient wrt a : 2·n·∑_k p_k ∂b_k/∂a = 4·n·conv(a, p_rev)
        p_rev = p[::-1]
        conv_ab = _conv_fft(a, p_rev)            # length 2·n‑1
        segment = conv_ab[n - 1: 2 * n - 1]       # n entries that correspond to ∂c/∂a
        grad_a = 4.0 * n * segment[::-1]

        # chain rule through the soft‑max parametrisation
        dot = np.dot(grad_a, a)
        grad_w = a * (grad_a - dot)

        return obj, grad_w

    # run L‑BFGS; we bound the number of iterations so that we never exceed
    # the global deadline.
    opts = {
        "maxiter": maxiter,
        "disp": False,
    }

    res = minimize(fun, w0, method="L-BFGS-B", jac=True, options=opts)
    w_opt = res.x
    w_shift = w_opt - np.max(w_opt)
    exp_w = np.exp(w_shift)
    a_opt = exp_w / np.sum(exp_w)
    # clip tiny negatives that may appear due to numerical noise
    a_opt = np.clip(a_opt, 0.0, None)
    a_opt /= a_opt.sum()
    return a_opt


def _run_smooth_optimisation(a0: np.ndarray,
                             n: int,
                             start: float,
                             deadline: float,
                             best_a: np.ndarray,
                             best_val: float):
    """
    Executes several L‑BFGS passes with decreasing smoothing.
    Returns the best vector found.
    """
    # a decreasing schedule for the smoothing parameter ε
    eps_schedule = [1e-2, 5e-3, 1e-3, 5e-4, 1e-4, 5e-5, 1e-5]
    a_cur = a0.copy()
    for eps in eps_schedule:
        now = time.time()
        if now >= deadline:
            break
        # allocate a modest number of L‑BFGS iterations; the number is
        # scaled by the remaining time budget.
        remaining = deadline - now
        maxiter = max(5, int(min(300, remaining / 0.01)))  # each iteration ≈10 ms
        a_cur = _smooth_lbfgs_optimize(a_cur, eps, maxiter, n, deadline)

        cur_val = _objective_value(a_cur)
        if cur_val < best_val:
            best_val, best_a = cur_val, a_cur.copy()
            print(f"[L‑BFGS eps={eps:.1e}] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimber(a_start: np.ndarray,
                 n: int,
                 start: float,
                 deadline: float,
                 best_a: np.ndarray,
                 best_val: float,
                 target_val: float = 1.503) -> np.ndarray:
    """
    Moves a tiny amount of mass from an index that contributes heavily to
    the current maximal convolution entry to a low‑contribution index.
    Several target candidates are examined per source; the best improving
    move is performed.
    """
    a = a_start.copy()
    b = _full_convolution(a)          # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    # smoothing schedule for the transferred fraction
    max_frac_start, max_frac_end = 0.5, 0.001
    no_improve = 0
    IMPROVE_LIMIT = 180_000
    it = 0

    while time.time() < deadline:
        progress = (time.time() - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # index where the current maximum occurs
        k_max = int(np.argmax(b))

        # contributions of each index to that maximal entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # choose a source index i biased by contribution
        if total_contrib > 0:
            probs = contrib / total_contrib
            i = np.random.choice(n, p=probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= 1e-15:
            continue

        # generate a few candidate target indices and keep the best improvement
        best_new_val = cur_val
        best_j = None
        best_delta = None
        best_b_new = None

        n_trials = 7
        for _ in range(n_trials):
            # biased target selection – 30 % chance to pick a low‑contribution slot
            if np.random.rand() < 0.30:
                rev_idx = k_max - np.arange(n)
                rev_mask = (rev_idx >= 0) & (rev_idx < n)
                rev_contrib = np.zeros(n)
                rev_contrib[rev_mask] = a[rev_idx[rev_mask]]
                w = 1.0 / (rev_contrib + 1e-12)
                w[i] = 0.0
                w_sum = w.sum()
                if w_sum > 0:
                    probs_j = w / w_sum
                    j = np.random.choice(n, p=probs_j)
                else:
                    j = np.random.randint(n - 1)
                    if j >= i:
                        j += 1
            else:
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1

            if j == i:
                continue

            delta = a[i] * np.random.rand() * max_frac
            if delta <= 0.0:
                continue

            # incremental update of the convolution (O(n))
            b_new = b.copy()
            b_new[j:j + n] += 2.0 * delta * a
            b_new[i:i + n] -= 2.0 * delta * a
            b_new[2 * j] += delta * delta
            b_new[2 * i] += delta * delta
            b_new[i + j] -= 2.0 * delta * delta

            new_val = 2.0 * n * b_new.max()
            if new_val < best_new_val - 1e-14:
                best_new_val = new_val
                best_j = j
                best_delta = delta
                best_b_new = b_new

        # perform the best improving move if any
        if best_j is not None:
            a[i] -= best_delta
            a[best_j] += best_delta
            b = best_b_new
            cur_val = best_new_val
            no_improve = 0
            if cur_val < best_val:
                best_val, best_a = cur_val, a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # stagnation handling – perturb loosely around the best seen so far
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.01, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[Hill {it:7d}] perturbing after stagnation")
        it += 1

    return best_a

    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to `evaluate_sequence`
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a – reuse a known construction if available.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)

    except Exception:
        a0, _ = _parametric_initialisation(n)

    best_a = a0.copy()
    best_val = _objective_value(best_a)
    print(f"[Init] best objective = {best_val:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

                                                start_time,
                                                phase1_deadline,
                                                best_a,
                                                best_val)

    print(f"[After L‑BFGS] best objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – targeted hill‑climbing on the exact objective
    # ------------------------------------------------------------------
    phase2_deadline = start_time + (deadline - start_time) * 0.90
    best_a = _hillclimber(best_a, n,
                          start=phase1_deadline,
                          deadline=phase2_deadline,
                          best_a=best_a,
                          best_val=best_val,
                          target_val=1.503)

    best_val = _objective_value(best_a)
    print(f"[After Hillclimb] best objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final fine‑tuning with a very small smoothing parameter
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_deadline = deadline
        best_a, best_val = _run_smooth_optimisation(best_a, n,
                                                    time.time(),
                                                    fine_deadline,

                                                    best_val)

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:               # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)
        final_score = float('inf')

    return best_a.tolist()
