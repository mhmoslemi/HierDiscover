"""
A three‑phase optimizer for the step‑function evaluation problem.

Phase A – Sub‑gradient descent on the exact max‑convolution objective.
Phase B – Smooth “log‑sum‑exp” descent (Adam) that approximates the
          max‑function by a differentiable surrogate; a small L2‑flattening
          term is added to keep the whole autocorrelation flat.
Phase C – Deterministic greedy hill‑climbing with O(n) incremental updates.

All updates keep the vector non‑negative and projected onto the probability
simplex (Σa = 1).  The routine respects the 1000 s wall‑time limit and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – fast for n≈8 k)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full convolution of a real vector with itself.
    Length = 2·len(x)‑1.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.
    Length = len(x) + len(y) – 1.
    """
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of a vector onto the simplex. """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0.0] = 0.0
    s = w.sum()
    if s > 0.0:
        w /= s
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# Phase A – Sub‑gradient descent on the exact objective (max b)
# ----------------------------------------------------------------------
def _subgradient_phase(a: np.ndarray,
                       n: int,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, float]:
    """ Reduce max(b) by sub‑gradient steps. """
    start = time.time()
    max_step = 0.10
    min_step = 1e-4
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        k = int(np.argmax(b))
        cur_max = b[k]
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Subgrad {it:7d}] new best = {best_val:.6f}")

        # build sub‑gradient of max(b)
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                grad[mid] = 2.0 * a[mid]

        prog = (time.time() - start) / (deadline - start)
        step = max_step * (1.0 - prog) + min_step * prog

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Smooth max descent (log‑sum‑exp) + small L2 flattening
# ----------------------------------------------------------------------
def _smooth_descent_phase(a: np.ndarray,
                         n: int,
                         deadline: float,
                         best_a: np.ndarray,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of a smoothed max objective.
    The soft‑max temperature (beta) is increased gradually.
    """
    start = time.time()
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule for soft‑max temperature
    beta_start, beta_end = 10.0, 500.0
    beta = beta_start

    # step‑size schedule (linear decay)
    step_start, step_end = 0.05, 0.001

    # L2‑flattening weight
    lam_l2 = 0.15

    it = 0
    while time.time() < deadline:
        # exact convolution and exact max (for monitoring)
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Smooth {it:7d}] new best = {best_val:.6f}")

        # ---- smooth max gradient (log‑sum‑exp) ----
        maxb = cur_max
        # stability: subtract max before exponentiation
        exp_beta = np.exp(beta * (b - maxb))
        p = exp_beta / exp_beta.sum()          # softmax probabilities, shape (2n‑1)

        # grad of smooth max: 2 * conv(p, a_rev) sliced to length n
        conv_max = _conv_fft(p, a[::-1])
        grad_max = 2.0 * conv_max[n - 1:2 * n - 1]

        # ---- optional L2 flattening gradient (variance of b) ----
        avg_target = 1.0 / (2 * n - 1)          # because Σb = 1 when Σa = 1
        r = b - avg_target
        conv1 = _conv_fft(r, a)
        conv2 = _conv_fft(r, a[::-1])
        grad_l2 = 2.0 * (conv1[n - 1:2 * n - 1] + conv2[n - 1:2 * n - 1])

        # total gradient
        grad = grad_max + lam_l2 * grad_l2

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        prog = (time.time() - start) / (deadline - start)
        step = step_start * (1.0 - prog) + step_end * prog

        a_candidate = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a_candidate)

        # slowly anneal the temperature
        beta = min(beta_end, beta * 1.001)

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Greedy hill‑climbing with incremental convolution updates
# ----------------------------------------------------------------------
def _greedy_hillclimb(a: np.ndarray,
                      n: int,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> np.ndarray:
    """Local greedy search that moves mass from the “worst” contributor to a “best” one."""
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 200_000
    no_improve = 0
    it = 0

    while time.time() < deadline:
        k = max_idx
        # contribution of each coordinate to the current max entry
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-14:
            # pick a non‑zero source
            nz = np.where(a > 1e-14)[0]
            if nz.size == 0:
                break
            source = np.random.choice(nz)

        # move a random fraction (up to 25 %) of the source mass
        delta = a[source] * 0.25 * np.random.rand()
        if delta <= 0.0:
            continue

        # incremental update of the full convolution (O(n))
        b_new = b.copy()
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            # accept move
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0

            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # perturb around current best to escape a plateau
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            a = _project_simplex(a)
            b = _full_convolution(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")
        it += 1
    return best_a
