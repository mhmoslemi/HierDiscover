def smooth_obj_and_grad(w, eps):
   w_shift = w - max(w)
   a = np.exp(w_shift)
   a /= a.sum()
   b = _full_convolution(a)
   max_b = b.max()
   shifted = (b - max_b)/eps
   exp_shift = np.exp(shifted - np.max(shifted)) # to avoid overflow? But typical.
   weights = np.exp(shifted) # may overflow? Use stable.
   sum_weights = weights.sum()
   p = weights / sum_weights
   # grad w.r.t a
   conv = _conv_fft(a, p[::-1])
   segment = conv[n-1: 2*n-1]
   grad_a = 4.0 * n * segment[::-1]   # this is gradient of 2*n*smoothed_max

   # value = 2 * n * (max_b + eps * log sum_weights)

   return val, grad_w

"""
Step‑function optimizer for the “max‑convolution” objective.

The search proceeds in three stages for each problem size *n*:
  1.  A dense one‑parameter “U‑shape’’ sweep (γ‑law) yields a deterministic
      starting vector.
  2.  A continuation of smooth‐max minimisation using L‑BFGS‑B on the
      soft‑max parametrisation a = softmax(w).  The smoothing parameter ε is
      decreased geometrically (0.20 → 0.005) and the optimisation is restarted
      from a jittered copy of the current best vector.
  3.  A fast, gradient‑guided hill‑climber that moves a small fraction of mass
      from a source index (high contribution to the current maximal
      convolution entry) to a target index (low contribution).  Stagnation is
      broken by a short random perturbation.

Several problem sizes (2000, 4000, 8000) are tried within the 1000 s budget,
and the best solution found (lowest exact objective) is returned.
"""

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Convolution helpers (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full linear convolution of `x` with itself (length 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Exact objective (assumes Σa = 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) under Σa = 1."""
    n = a.shape[0]
    b = _full_convolution(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Initial parametric sweep (U‑shape power‑law)
# ----------------------------------------------------------------------
def _initial_shape(n: int):
    """Search the family a_i ∝ ((i+½)(n‑i‑½))^{-γ} for the best γ."""
    best_a = None
    best_val = np.inf
    eps = 1e-12
    x = np.arange(n, dtype=np.float64)
    # fine grid to give the optimiser a good launch point
    for gamma in np.linspace(0.0, 5.0, 101):
        denom = (x + 0.5) * (n - x - 0.5)
        a = np.power(denom + eps, -gamma)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑max + L‑BFGS optimisation (log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _smooth_obj_grad(w: np.ndarray, n: int, eps: float):
    """
    Smooth approximation of the exact objective and its gradient w.r.t. the
    logits w (a = softmax(w)).  Returns (value, grad_w).
    """
    # soft‑max to obtain a on the simplex
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # full self‑convolution
    b = _full_convolution(a)
    max_b = b.max()

    # log‑sum‑exp of (b-max_b)/eps  (stable)
    shifted = (b - max_b) / eps
    max_shift = np.max(shifted)
    exp_shift = np.exp(shifted - max_shift)
    sum_exp = exp_shift.sum()
    log_sum_exp = max_shift + np.log(sum_exp)

    # smooth max = max_b + eps * log_sum_exp
    smoothed_max = max_b + eps * log_sum_exp
    val = float(2.0 * n * smoothed_max)

    # soft‑max weights (gradient of smoothed max w.r.t. each b entry)
    p = exp_shift / sum_exp

    # gradient of 2·n·(∑ p_k b_k) w.r.t. a
    p_rev = p[::-1]
    conv_ab = _conv_fft(a, p_rev)
    segment = conv_ab[n - 1: 2 * n - 1]          # length n
    grad_a = 4.0 * n * segment[::-1]            # chain rule for a*a and factor 2·n

    # chain rule through the soft‑max parametrisation
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)

    return val, grad_w


# ----------------------------------------------------------------------
# Targeted hill‑climbing (local improvement)
# ----------------------------------------------------------------------
def _hillclimb_improved(a_start: np.ndarray,
                        n: int,
                        start: float,
                        deadline: float,
                        best_a: np.ndarray,
                        best_val: float,
                        target_val: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Fast local search that transfers a small amount of mass while keeping Σa = 1.
    Moves are guided by the contribution of each index to the current maximal
    convolution entry.  Stagnation triggers a brief random perturbation.
    """
    a = a_start.copy()
    b = _full_convolution(a)               # length 2·n‑1
    cur_val = 2.0 * n * b.max()

    max_frac_start, max_frac_end = 0.60, 0.02
    IMPROVE_LIMIT = 120_000                # more aggressive than before
    no_improve = 0
    iteration = 0
    eps = 1e-12

    while True:
        now = time.time()
        if now >= deadline:
            break

        # linear schedule for the maximal fraction of mass that may be moved
        progress = (now - start) / (deadline - start)
        max_frac = max_frac_start * (1.0 - progress) + max_frac_end * progress

        # locate the (first) maximal convolution index
        k_max = int(np.argmax(b))

        # contribution of each a[i] to b[k_max]
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[idx[mask]]
        total_contrib = contrib.sum()

        # ----- source (biased by contribution) -----
        if total_contrib > 0:
            source_probs = contrib / total_contrib
            i = np.random.choice(n, p=source_probs)
        else:
            i = int(np.argmax(a))

        if a[i] <= eps:
            continue

        # ----- target (biased towards *small* contribution) -----
        target_weights = np.zeros_like(a)
        target_weights[mask] = 1.0 / (contrib[mask] + eps)
        target_weights[i] = 0.0                     # avoid self‑transfer
        w_sum = target_weights.sum()
        if w_sum > 0:
            j = np.random.choice(n, p=target_weights / w_sum)
        else:
            # fallback uniform among all except i
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount to move (fraction of a[i])
        delta = a[i] * np.random.rand() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (O(n))
        b_new = b.copy()
        b_new[j:j + n] += 2.0 * delta * a
        b_new[i:i + n] -= 2.0 * delta * a
        # pure quadratic corrections
        b_new[2 * j] += delta * delta
        b_new[2 * i] += delta * delta
        b_new[i + j] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # accept only if strictly improving
        if new_val < cur_val - 1e-14:
            a[i] -= delta
            a[j] += delta
            b = b_new
            cur_val = new_val
            no_improve = 0

            if new_val < best_val - 1e-12:
                best_val, best_a = new_val, a.copy()
                print(f"[HillImp {iteration:7d}] new best = {best_val:.6f}")
                if best_val <= target_val:
                    break
        else:
            no_improve += 1

        # ----- stagnation handling -----
        if no_improve > IMPROVE_LIMIT:
            a = best_a.copy()
            noise = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            b = _full_convolution(a)
            cur_val = 2.0 * n * b.max()
            no_improve = 0
            print(f"[HillImp {iteration:7d}] stagnation – perturb")
        iteration += 1

    return best_a, best_val

# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under a 1000 s budget.
    Returns a list whose sum is 1 and that (attempts to) minimise
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem sizes – we try a few to let the optimiser work with a
    # smaller n (more iterations) and then a larger n (finer discretisation).
    # ------------------------------------------------------------------
    n_candidates = [2000, 4000, 8000]

    best_overall_a = None
    best_overall_val = np.inf

    for n in n_candidates:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising for n = {n} ===")

        # --------------------------------------------------------------
        # Initialise a – try to reuse a previous construction if available.
        # --------------------------------------------------------------
        a0 = None
        try:
            old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
            old_len = old.shape[0]
            new_x = np.linspace(0.0, 1.0, n)
            old_x = np.linspace(0.0, 1.0, old_len)
            a0 = np.interp(new_x, old_x, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            print("[init] interpolated previous solution")
        except Exception:
            pass

        # --------------------------------------------------------------
        # Phase 0 – cheap parametric sweep for a deterministic seed.
        # --------------------------------------------------------------
        best_a = None
        best_val = np.inf
        if a0 is not None:
            best_a = a0.copy()
            best_val = _objective(best_a)
            print(f"[init] start from interpolated solution, value={best_val:.6f}")

        try:
            shape_a, shape_val = _initial_shape(n)
            if shape_val < best_val:
                best_a, best_val = shape_a, shape_val
                print(f"[init] shape sweep improved value={best_val:.6f}")
        except Exception as exc:
            print("[init] shape sweep failed:", exc)

        # fallback random start
        if best_a is None:
            a = np.random.rand(n)
            a /= a.sum()
            best_a, best_val = a, _objective(a)
            print(f"[init] random start, value={best_val:.6f}")

        # --------------------------------------------------------------
        # Phase 1 – continuation L‑BFGS on the smooth max.
        # --------------------------------------------------------------
        eps_schedule = [0.20, 0.10, 0.05, 0.02, 0.005]
        for eps in eps_schedule:
            if time.time() >= deadline:
                break
            # Perform a few stochastic restarts at each ε
            for restart in range(2):
                if time.time() >= deadline:
                    break
                # jitter the logits to escape possible plateaus
                w0 = np.log(best_a + 1e-12)
                w0 += np.random.normal(scale=0.02, size=n)

                res = opt.minimize(_smooth_obj_grad,
                                   w0,
                                   args=(n, eps),
                                   method='L-BFGS-B',
                                   jac=True,
                                   options={'maxiter': 150, 'disp': False})
                w_opt = res.x
                a_opt = np.exp(w_opt - np.max(w_opt))
                a_opt /= a_opt.sum()
                val_opt = _objective(a_opt)

                if val_opt < best_val:
                    best_val, best_a = val_opt, a_opt.copy()
                    print(f"[L‑BFGS n={n:4d} eps={eps:5.3f} restart={restart}] "
                          f"new best = {best_val:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – fast targeted hill‑climbing.
        # --------------------------------------------------------------
        if time.time() < deadline:
            best_a, best_val = _hillclimb_improved(best_a,
                                                   n,
                                                   start=time.time(),
                                                   deadline=deadline,
                                                   best_a=best_a,
                                                   best_val=best_val,
                                                   target_val=1.503)

        print(f"[summary n={n}] best value = {best_val:.6f}")

        # keep the overall best across different n
        if best_val < best_overall_val:
            best_overall_val, best_overall_a = best_val, best_a.copy()
            print(f"[overall] new global best = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (uses the provided secure wrapper)
    # ------------------------------------------------------------------
    if best_overall_a is None:
        # should never happen, but provide a safe fallback
        best_overall_a = np.array([1.0])

    final_score = evaluate_sequence(best_overall_a.tolist())
    print(f"[final] reported score = {final_score:.6f}")

    return best_overall_a.tolist()
