"""
Propose a step‑function (list of non‑negative heights) that minimizes
``evaluate_sequence``.

The search combines several complementary phases:

1. **Diverse initialisations** – uniform, sine, triangular, exponential,
   a previous best construction (if available) and a few random Dirichlet
   draws.
2. **Focused mirror‑descent** – multiplicative updates
   a_i ← a_i·exp(−α·g_i) where g_i aggregates the contributions of all
   lags whose convolution value is close to the current maximum.
   The step size α decays slowly; a cheap anti‑symmetric re‑ordering is
   applied periodically to keep the reversed product small.
3. **Local stochastic refinement** – if no improvement is observed for a
   while a random perturbation (reshuffling a subset of the mass) is
   performed to escape plateaus.
4. **Smooth L‑BFGS polishing** – a log‑sum‑exp surrogate of the
   ``max(conv)`` objective is minimised with a few L‑BFGS runs;
   gradients are derived analytically.
5. **Final hill‑climbing** – random swaps and mass transfers finish the
   search.

All phases respect a global 1000 s budget (a 5 s safety margin).  The
algorithm returns the best sequence found.
"""

import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself.
    Length = 2·len(x)‑1, computed via FFT for speed.
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negative values – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Pair large values with small ones when the sequence is reversed.
    This heuristic (the rearrangement inequality) lowers the centre
    convolution peak (a ⋆ rev(a)).
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)           # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]     # large → left side
        new_a[n - 1 - i] = sorted_vals[i]     # small → right side

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, n: int):
    """
    Simple hill‑climb: random swaps (20 %) and mass transfers (80 %).
    Only strict improvements are kept; the routine stops at the deadline.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * rng.random() * 0.5   # up to 50 % of source
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex (non‑negative, sum = 1)
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate of 2·n·max(conv) and its gradient w.r.t. the
    soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)  (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (numerically stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)               # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()               # probability distribution over k

    # convolution of a with reversed p  (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # extract the middle n entries that correspond to the derivative
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function of length ≈4000 that
    minimises ``evaluate_sequence`` within the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – a moderate length gives a good balance
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build a list of diverse starting points
    # ------------------------------------------------------------------
    seeds = []

    # 1) Uniform
    seeds.append(np.full(n, 1.0 / n))

    # 2) Previous best (if available)
    try:
        prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if prev.shape[0] != n:
            # Linear interpolation onto the target grid
            old_x = np.linspace(0.0, 1.0, prev.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            prev = np.interp(new_x, old_x, prev)
        seeds.append(prev.copy())
        print("[init] loaded height_sequence_1")
    except Exception:
        pass

    # 3) Sinusoidal (positive half‑wave)
    i = np.arange(n)
    sine = np.sin(np.pi * (i + 0.5) / n)          # strictly positive
    seeds.append(sine)

    # 4) Triangular (mountain shape)
    tri = np.minimum(i + 1, n - i)               # 1,…,n/2,…,1
    seeds.append(tri.astype(float))

    # 5) Exponential decay
    lam = 5.0 / n
    expdec = np.exp(-lam * i)
    seeds.append(expdec)

    # 6‑8) Random Dirichlet draws (a few)
    for _ in range(3):
        seeds.append(rng.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # Global best trackers
    # ------------------------------------------------------------------
    best_a = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper: evaluate and possibly update the global best
    # ------------------------------------------------------------------
    def try_update(candidate):
        nonlocal best_a, best_val
        try:
            val = evaluate_sequence(candidate.tolist())
        except Exception:
            return
        if val < best_val - 1e-12:
            best_val = val
            best_a = candidate.copy()
            print(f"[global] new overall best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – gradient descent on each seed (time‑sliced)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(seeds):
        remaining = deadline - time.time()
        if remaining < 30.0:          # not enough time for another full run
            break
        # give each seed roughly an equal slice, but never exceed 200 s
        slice_time = min(200.0, remaining / (len(seeds) - idx))
        slice_deadline = time.time() + slice_time
        print(f"[seed {idx+1}/{len(seeds)}] time slice → {slice_time:.1f}s")

        # ---- initialise -------------------------------------------------
        a = np.array(seed_vec, dtype=float)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        try_update(a)

        # ---- optimisation loop -----------------------------------------
        it = 0
        last_improve_it = 0
        alpha0 = 0.30                     # initial step size
        rel_tol = 0.0                     # tolerance for “near‑max” lags

        while time.time() < slice_deadline:
            # compute full convolution (FFT)
            conv = _full_convolution(a)
            max_val = conv.max()

            # Determine which lags should participate in the sub‑gradient.
            # Early in optimisation we focus on the single maximum;
            # later we broaden to a small neighbourhood.
            if it > 3000:
                rel_tol = 0.02            # 2 % window after a while
            mask = conv >= max_val * (1.0 - rel_tol)
            ks = np.where(mask)[0]

            # aggregate gradient over the selected lags
            g = np.zeros_like(a)
            for k in ks:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                idx_range = np.arange(i_start, i_end + 1)
                jdx_range = k - idx_range
                g[idx_range] += a[jdx_range]

            # geometric decay of the step size
            alpha = alpha0 / (1.0 + it / 2000.0)

            # multiplicative mirror‑descent update (preserves non‑negativity)
            a = a * np.exp(-alpha * g)

            # project back onto the simplex Σ a_i = 1
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s <= 1e-12:
                a = np.full(n, 1.0 / n)
            else:
                a /= s

            # occasional anti‑symmetric re‑ordering (helps the centre lag)
            if it % 2500 == 0 and it > 0:
                a = _anti_symmetric_reorder(a)

            # periodic evaluation
            if it % 300 == 0:
                cur_val = evaluate_sequence(a.tolist())
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_a = a.copy()
                    last_improve_it = it
                    print(f"[seed {idx+1}] iter {it:5d} → new best = {best_val:.6f}")

            # stagnation handling – random perturbation after many idle steps
            if it - last_improve_it > 5000:
                # reshape 10 % of the mass randomly
                subset = rng.choice(n, size=int(0.10 * n), replace=False)
                random_mass = rng.random(subset.size)
                a[subset] = random_mass
                a = np.maximum(a, 0.0)
                a /= a.sum()
                last_improve_it = it
                print(f"[seed {idx+1}] iter {it:5d} → random perturbation")

            it += 1

        # after the slice, make a final update
        try_update(a)

    # ------------------------------------------------------------------
    # Phase 2 – optional L‑BFGS polishing of the current best
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline - 30.0:
        print("[L‑BFGS] polishing the current best sequence")
        a = best_a.copy()
        w = np.log(a + 1e-12)                 # soft‑max parameters
        eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-10), num=7)

        while time.time() < deadline - 30.0:
            for eps in eps_schedule:
                if time.time() >= deadline - 30.0:
                    break
                global _GLOBAL_EPS, _GLOBAL_N
                _GLOBAL_EPS = eps
                _GLOBAL_N = n
                res = minimize(_smooth_obj_and_grad_wrapper,
                               w,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': 120,
                                        'ftol': 1e-12,
                                        'gtol': 1e-12,
                                        'disp': False})
                w = res.x
                ws = w - np.max(w)
                a_candidate = np.exp(ws)
                a_candidate /= a_candidate.sum()
                val = evaluate_sequence(a_candidate.tolist())
                if val < best_val - 1e-12:
                    best_val = val
                    best_a = a_candidate.copy()
                    print(f"[L‑BFGS] eps {eps:.1e} → new best = {best_val:.6f}")

        # final anti‑symmetric check after L‑BFGS
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val - 1e-12:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[L‑BFGS] post‑reorder improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final stochastic hill‑climbing refinement
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final sanity check & return
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] reported evaluation = {final_score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
