   where ``w`` are the normalised soft‑max weights and ``a_rev`` is ``a``
   reversed.  This follows from the formula
       g_i = 2 Σk w_k·a_{k‑i}.
4. Optimise Lα with the Adam adaptive‑step method, while projecting each
   iterate back onto the simplex (Duchi et al. 2008).  The smoothing
   parameter α is increased slowly during the run, starting from a small
   value (≈5) and ending around 2000.
5. Whenever the *exact* objective improves (checked with the provided
   ``evaluate_sequence``), store the current best sequence.
6. After the gradient phase, run a short guided hill‑climbing refinement
   that moves small amounts of mass away from the most offending lag
   (the same idea as the previous state‑of‑the‑art code but with a
   finer‑grained step schedule).
7. Return the best list of heights found before the time budget expires.

The whole routine respects the 1000 s budget, prints progress and the
best score obtained, and works with any existing global ``height_sequence_1``.
"""

import time
import numpy as np

# ------------------------------------------------------------
# FFT utilities (real data)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len−1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative values caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0.0, out=conv)
    return conv


# ------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Small local search that moves a tiny fraction of mass away from the
    most offending convolution lag.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # indices (i,j) that realise the product a_i*a_j for this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # pick two destination indices with relatively low mass
        low = np.where(cur_a < cur_a.mean())[0]
        if low.size >= 2:
            p, q = rng.choice(low, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # perform a feasible mass transfer
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:  # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        # exact evaluation (cheap shortcut is also possible)
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] new best = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence``.  Uses a smooth‑max surrogate, Adam optimisation
    and a final hill‑climbing refinement.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------
    # Problem size – 4000 is a good trade‑off
    # --------------------------------------------------------
    n = 4000

    # --------------------------------------------------------
    # Initial sequence (reuse previous result if available)
    # --------------------------------------------------------
    try:
        a0 = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if a0.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, a0.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, a0)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    except Exception:
        a0 = np.full(n, 1.0 / n)  # uniform start

    a = a0.copy()
    # quick objective (ignores scaling which is already fixed)
    conv = _full_convolution(a)
    cur_approx = 2.0 * n * conv.max()
    best_a = a.copy()
    best_approx = cur_approx
    best_val = float('inf')
    print(f"[init] n={n}, initial approx objective = {best_approx:.6f}")

    # --------------------------------------------------------
    # Adam optimiser on the smooth max surrogate
    # --------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.25               # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smoothing schedule
    alpha = 5.0
    alpha_inc = 1.02      # multiplicative increase per iteration
    alpha_max = 2000.0

    iteration = 0
    no_improve = 0

    while time.time() < deadline - 5.0:
        iteration += 1

        # ---------- compute convolution and smooth weights ----------
        conv = _full_convolution(a)               # length 2n‑1
        max_conv = conv.max()
        approx_obj = 2.0 * n * max_conv

        # store a better *exact* solution (occasionally evaluate)
        if approx_obj < best_approx - 1e-12:
            try:
                exact = evaluate_sequence(a.tolist())
                if exact < best_val:
                    best_val = exact
                    best_a = a.copy()
                    best_approx = approx_obj
                    print(f"[iter {iteration}] new best exact = {best_val:.6f}")
                    no_improve = 0
                else:
                    # exact not better – keep the approximation as record
                    best_approx = approx_obj
            except Exception:
                # fallback – treat as no improvement
                best_approx = approx_obj

        # ---------- soft‑max weights ----------
        # shift by max to avoid overflow
        shifted = conv - max_conv
        w_raw = np.exp(alpha * shifted)
        w_sum = w_raw.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # ---------- gradient via convolution ----------
        a_rev = a[::-1]                               # reversed copy
        grad_full = _full_convolution_fft(w, a_rev)   # length 3n‑2
        start = n - 1
        g = 2.0 * grad_full[start:start + n]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _simplex_projection(a_candidate)

        # ---------- acceptance ----------
        new_conv = _full_convolution(a_candidate)
        new_approx = 2.0 * n * new_conv.max()

        if new_approx < cur_approx - 1e-14:
            a = a_candidate
            cur_approx = new_approx
            no_improve = 0
        else:
            no_improve += 1

        # ---------- schedule adjustments ----------
        alpha = min(alpha * alpha_inc, alpha_max)

        # If we have been stuck for a while, inject a tiny random jitter
        if no_improve >= 200:
            jitter = rng.normal(scale=0.005, size=n)
            a = _simplex_projection(a + jitter)
            cur_approx = 2.0 * n * _full_convolution(a).max()
            no_improve = 0
            # reset Adam moments to avoid stale direction
            m.fill(0.0)
            v.fill(0.0)

        # minimal printing for long runs
        if iteration % 200 == 0:
            print(f"[iter {iteration}] approx = {cur_approx:.6f}, best = {best_approx:.6f}")

    # --------------------------------------------------------
    # Final refinement (guided hill‑climbing)
    # --------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # --------------------------------------------------------
    # Final reporting
    # --------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")

    return best_a.tolist()
