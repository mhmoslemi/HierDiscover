"""
Hybrid optimization for minimizing the convolution peak of a non‑negative step function.
Roughly:
1. Generate diverse seed sequences (random, Dirichlet, geometric, power‑law).
2. Run a cheap L‑BFGS optimisation of a smoothed surrogate (log‑sum‑exp of the max).
3. Refine the best candidates with a longer L‑BFGS run.
4. Apply several deterministic reorderings (anti‑symmetric, alternating, descending).
5. Perform an extended simulated‑annealing that mixes swaps, mass‑transfer
   and a targeted transfer that tries to lower the current maximal convolution entry.
6. Finish with a short hill‑climbing pass.
All work stops before the given time budget; the best feasible sequence is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

_GLOBAL_EPS = None
_GLOBAL_N = None

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _fft_convolve_self(a):
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

def _eval_fast(a):
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _conv_fft(x, y):
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

def _smooth_obj_and_grad(w, eps, n):
    # softmax to get a (positive, sum to 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    conv = _fft_convolve_self(a)
    max_conv = conv.max()

    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()

    # compute convolution of a with reversed p
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)
    # middle n entries correspond to indices n-1 .. 2n-2 (full conv length 2n-1)
    seg = conv_ap[n - 1:2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w

def _smooth_obj_and_grad_wrapper(w):
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

def _anti_symmetric_reorder(a):
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large to the left
        new_a[n - 1 - i] = sorted_vals[i]      # small to the right

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a

def _alternating_reorder(a):
    # largest, smallest, second‑largest, second‑smallest, …
    sorted_vals = np.sort(a)
    left = 0
    right = sorted_vals.shape[0] - 1
    out = np.empty_like(a)
    idx = 0
    while left <= right:
        out[idx] = sorted_vals[right]
        idx += 1
        right -= 1
        if left <= right:
            out[idx] = sorted_vals[left]
            idx += 1
            left += 1
    return out

def _descending_reorder(a):
    return np.sort(a)[::-1]

# Simple parametric families

def _geometric_sequence(q, n):
    """Return a_i proportional to q**i, normalised to sum 1."""
    if q <= 0.0 or q >= 1.0:
        # fallback to uniform
        return np.full(n, 1.0 / n)
    # geometric series sum = (1 - q**n) / (1 - q)
    denom = (1.0 - q**n) / (1.0 - q)
    a = q ** np.arange(n) / denom
    return a

def _power_sequence(p, n):
    """Return a_i proportional to (i+1)**p, normalised to sum 1.
    p should be negative to keep sum finite."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** p
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n)
    return a / a_sum

def _simulated_annealing_extended(a, cur_val, deadline):
    """
    SA that mixes:
    * simple swaps of two positions
    * mass transfer from i to j
    * targeted transfer from the pair that contributes most to the current peak
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # start temperature as a fraction of the current objective
    T = max(cur_val * 0.05, 1e-6)
    decay = 0.9999
    it = 0
    while time.time() < deadline:
        move_type = rng.random()
        if move_type < 0.4:          # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        elif move_type < 0.8:        # simple mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            delta = a[i] * 0.05 * rng.random()
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
        else:                         # targeted transfer
            # compute full convolution once
            conv = _fft_convolve_self(a)
            k_max = int(np.argmax(conv))
            # find the pair (i,j) that realises the largest product a_i*a_j with i+j=k_max
            # iterate over possible i
            i_best = None
            max_product = -1.0
            for i in range(max(0, k_max - (n - 1)), min(n - 1, k_max) + 1):
                j = k_max - i
                if j < 0 or j >= n:
                    continue
                prod = a[i] * a[j]
                if prod > max_product:
                    max_product = prod
                    i_best = i
                    j_best = j
            if i_best is None:
                # fallback to regular mass transfer
                i, j = rng.integers(0, n, size=2)
                if i == j or a[i] == 0.0:
                    continue
                delta = a[i] * 0.05 * rng.random()
                new_a = a.copy()
                new_a[i] -= delta
                new_a[j] += delta
            else:
                # decide from which of the two positions to take mass
                source = i_best if a[i_best] >= a[j_best] else j_best
                target = rng.integers(0, n)
                if source == target:
                    continue
                delta = a[source] * 0.05 * rng.random()
                new_a = a.copy()
                new_a[source] -= delta
                new_a[target] += delta

        new_val = _eval_fast(new_a)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA-ext] new best = {best_val:.6f} (iter {it})")
        # cooling
        T = max(T * decay, 1e-12)
        it += 1
    return best_seq, best_val

"""
Hybrid optimiser for the step‑function convolution problem.

The algorithm proceeds in several stages:

1. **Diverse seed generation** – a handful of hand‑crafted families
   (geometric, power‑law) together with random Dirichlet draws are
   evaluated; the best seed is kept.

2. **Coarse L‑BFGS** – a cheap L‑BFGS optimisation of a smooth
   log‑sum‑exp surrogate (soft‑max parametrisation, sum = 1) using a
   relatively large smoothing parameter.  Several random restarts are
   performed; after each restart a few deterministic reorderings
   (anti‑symmetric, alternating, descending) are tested.

3. **Refined L‑BFGS** – the current best vector is further optimised with
   a finer ε‑schedule and more L‑BFGS iterations, again followed by the
   deterministic reorderings.

4. **Extended simulated annealing** – a temperature‑controlled search
   that mixes three move types:
   * swaps of two positions,
   * small mass transfers between two positions,
   * a *targeted* transfer that tries to lower the currently maximal
     convolution entry.
   The annealing runs for the bulk of the remaining time.

5. **Final hill‑climbing** – cheap local improvements (swap / transfer)
   until the deadline is reached.

All heavy work is performed with an FFT‑based evaluator
(`_eval_fast`) that reproduces the behaviour of the supplied
`evaluate_sequence`.  The function returns a plain Python list of
floats and respects the 1000 s budget.
"""

from scipy.special import logsumexp

_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using FFT."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)  # clamp tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """Fast version of `evaluate_sequence` for a non‑negative vector."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε (log‑sum‑exp approximation of the max) and its
    gradient w.r.t. the soft‑max parameters w.
    """
    # ---- soft‑max (positivity + ℓ₁ = 1) ----
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= np.sum(a)

    # ---- autocorrelation a*a ----
    conv = _fft_convolve_self(a)
    max_conv = conv.max()

    # ---- log‑sum‑exp smoothing of the max ----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a (closed‑form) ----
    weights = np.exp(shifted)
    p = weights / weights.sum()                     # probability over convolution indices
    p_rev = p[::-1]
    conv_ap = _conv_fft(a, p_rev)                  # a * p_rev
    seg = conv_ap[n - 1: 2 * n - 1]                 # central part, length n
    grad_a = 4.0 * n * seg[::-1]

    # ---- chain rule through the soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by scipy.optimize.minimize."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Large values left, small values right (pairwise)."""
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    out = np.empty_like(a)
    half = n // 2
    for i in range(half):
        out[i] = sorted_vals[n - 1 - i]      # large to the left
        out[n - 1 - i] = sorted_vals[i]      # small to the right
    if n % 2:
        out[half] = sorted_vals[half]
    return out


def _alternating_reorder(a: np.ndarray) -> np.ndarray:
    """Largest, smallest, second‑largest, second‑smallest, …."""
    sorted_vals = np.sort(a)
    left, right = 0, len(sorted_vals) - 1

    idx = 0
    while left <= right:
        out[idx] = sorted_vals[right]
        idx += 1
        right -= 1
        if left <= right:
            out[idx] = sorted_vals[left]
            idx += 1
            left += 1
    return out


def _descending_reorder(a: np.ndarray) -> np.ndarray:
    """Simply sort descending."""

            new_a[i], new_a[j] = new_a[j], new_a[i]

        elif move < 0.8:                   # random mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            delta = a[i] * 0.05 * rng.random()
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        else:                               # targeted transfer from the pair that forms the current peak
            conv = _fft_convolve_self(a)
            k_max = int(np.argmax(conv))
            # locate the pair (i,j) with i+j=k_max and maximal product a_i*a_j
            i_best, prod_best = None, -1.0
            i_start = max(0, k_max - (n - 1))
            i_end = min(k_max, n - 1)
            for i in range(i_start, i_end + 1):
                j = k_max - i
                prod = a[i] * a[j]
                if prod > prod_best:
                    prod_best = prod
                    i_best = i
                    j_best = j
            if i_best is None:
                # fallback to a regular transfer
                i, j = rng.integers(0, n, size=2)
                if i == j or a[i] == 0.0:
                    continue
                delta = a[i] * 0.05 * rng.random()
                new_a = a.copy()
                new_a[i] -= delta
                new_a[j] += delta
            else:
                source = i_best if a[i_best] >= a[j_best] else j_best
                target = rng.integers(0, n)
                if source == target:
                    continue
                delta = a[source] * 0.05 * rng.random()
                new_a = a.copy()
                new_a[source] -= delta
                new_a[target] += delta

        new_val = _eval_fast(new_a)
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            a = new_a
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA-ext] new best = {best_val:.6f} (iter {it})")

        T = max(T * decay, 1e-12)
        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Final hill‑climbing (swap + small transfers)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:           # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                             # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a step‑function minimising ``evaluate_sequence``."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = 4000
    global _GLOBAL_N
    _GLOBAL_N = n

    # --------------------------------------------------------------
    # initial sequence (reuse previous construction if available)
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)
        print("[init] using uniform start")
    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1. hand‑crafted seed families
    # --------------------------------------------------------------
    seed_candidates = []

    # geometric progression
    for q in np.linspace(0.7, 0.999, 20):
        seed_candidates.append(_geometric_sequence(q, n))

    # power‑law progression
    for p in np.linspace(-3.0, -0.2, 20):
        seed_candidates.append(_power_sequence(p, n))

    # random Dirichlet‑like draws
    for _ in range(30):
        a = np.random.rand(n)
        a /= a.sum()
        seed_candidates.append(a)

    for a in seed_candidates:
        val = _eval_fast(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[seed] new best = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. coarse L‑BFGS (many cheap restarts)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    stage_a_deadline = time.time() + min(0.25 * budget_s, time_left * 0.5)

    eps_coarse = np.logspace(np.log10(1e-2), np.log10(5e-7), num=8)

    while time.time() < stage_a_deadline:
        w = np.random.randn(n) * 2.0
        for eps in eps_coarse:
            if time.time() >= stage_a_deadline:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': 100,
                                    'ftol': 1e-8,
                                    'gtol': 1e-8,
                                    'disp': False})
            w = res.x
            a = np.exp(w - np.max(w))
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[coarse] eps {eps:.1e} new best = {best_val:.6f}")

        # try a few deterministic reorderings after each restart
        for reorder in (_anti_symmetric_reorder,
                        _alternating_reorder,
                        _descending_reorder):
            a_r = reorder(best_a)
            v_r = _eval_fast(a_r)
            if v_r < best_val:
                best_val = v_r
                best_a = a_r.copy()
                print(f"[reorder] {reorder.__name__} improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. refined L‑BFGS on the current best (fine schedule)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    stage_b_deadline = time.time() + min(200.0, max(0.0, time_left - 30.0))

    eps_fine = np.logspace(np.log10(5e-3), np.log10(1e-12), num=12)

    # initialise w from the current best (log‑softmax)
    w = np.log(best_a + 1e-16)

    while time.time() < stage_b_deadline:
        for eps in eps_fine:
            if time.time() >= stage_b_deadline:
                break
            _GLOBAL_EPS = eps
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': 300,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x
            a = np.exp(w - np.max(w))
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[fine] eps {eps:.1e} new best = {best_val:.6f}")

        # deterministic re‑orderings after each full pass
        for reorder in (_anti_symmetric_reorder, _alternating_reorder):
            a_r = reorder(best_a)
            v_r = _eval_fast(a_r)
            if v_r < best_val:
                best_val = v_r
                best_a = a_r.copy()
                print(f"[reorder‑fine] {reorder.__name__} improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. extended simulated annealing (permutations + transfers)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 150.0:
        sa_deadline = time.time() + remaining - 100.0   # keep ~100 s for final polish
    else:
        sa_deadline = deadline - 10.0
    print(f"[SA] extended simulated annealing until {sa_deadline:.2f}")
    best_a, best_val = _simulated_annealing_extended(best_a, best_val, sa_deadline)

    # --------------------------------------------------------------
    # 5. final hill‑climbing refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # --------------------------------------------------------------
    # 6. report exact evaluation using the provided function
    # --------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return best_a.tolist()
