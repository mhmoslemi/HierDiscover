The algorithm works in three stages (all executed inside the 1000 s budget):

   unit sum) and the step size is slowly decayed.

   directly reduces the current maximum convolution value.  The
   sub‑gradient is the average of the active (max‑attaining) rows of the
   Toeplitz matrix and is obtained again by one convolution.  Adam is

3. **Simulated‑annealing fine‑tuning** – cheap “mass‑transfer’’ moves

   a Metropolis rule.  Temperature decays linearly; periodic global

The search is performed on a modest problem size (n≈300).  For such a

import math

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project `v` onto the simplex {x | x ≥ 0, sum(x)=1} (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sort descending
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

    return 2.0 * n * conv.max()

        it2 += 1
        beta1_pow *= beta1
        beta2_pow *= beta2

        conv = np.convolve(a, a)
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[phase2 iter {it2:6d}] new_best={best_val:.12f}")

        # exact max‑sub‑gradient (average over all active entries)
        tol = 1e-14
        active = np.flatnonzero(conv >= max_conv - tol)
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size   # average weight
        else:
            w[np.argmax(conv)] = 1.0

        # gradient of weighted sum = 2·rev(a) * w
        g = 2.0 * np.convolve(a[::-1], w)[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr_adam * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_candidate = a - step
        a = _project_simplex(a_candidate)

        # occasional random perturbation if stuck
        if it2 % 4000 == 0:
            a = a + rng.normal(scale=0.001, size=n)
            a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing fine tuning.
    # ------------------------------------------------------------------
    cur = best_a.copy()
    cur_val = _objective(cur)

    T0 = 5e-3
    Tmin = 1e-8
    iter_sa = 0
    while time.time() < deadline:
        iter_sa += 1
        # linear cooling schedule
        T = max(Tmin, T0 * (1.0 - iter_sa / 200_000))

        # pick two distinct indices
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # transfer a small amount from i to j (preserving non‑negativity)
        max_transfer = min(cur[i], 0.015)       # keep moves modest
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        new_val = _objective(new)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / T):
            cur = new
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = cur.copy()
                print(f"[SA iter {iter_sa:6d}] new_best={best_val:.12f}")

        # global jitter every 25 k steps to break long plateaus
        if iter_sa % 25000 == 0:
            cur = cur + rng.normal(scale=0.001, size=n)
            cur = _project_simplex(cur)
            cur_val = _objective(cur)

    # ------------------------------------------------------------------
    # Return the best sequence in the format expected by `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_seq = [float(max(0.0, min(1000.0, x))) for x in best_a.tolist()]

    # optional logging of the official score
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[result] final_score={final_score:.12f}, length={len(final_seq)}")
    except Exception:
        print("[result] evaluate_sequence not available")

    return final_seq
