"""
propose_candidate
=================

Search for a non‑negative step‑function (a probability vector) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The scaling invariance allows us to keep sum(a)=1, so the objective reduces to
`2·n·max(conv(a,a))`.  The algorithm consists of three stages:

1. **Smoothed‑max gradient descent** (mirror descent on the probability simplex).
   We replace the non‑smooth max by a log‑sum‑exp with a temperature β and
   minimise the resulting smooth objective.  The gradient can be computed by a
   single convolution using FFTs.  The update is performed with an
   exponentiated‑gradient step (mirror descent w.r.t. the KL‑divergence) and the
   vector is re‑normalised to stay on the simplex.

2. **Automatic β‑schedule** – start with a small β (very smooth) and grow it
   geometrically up to a large value.  This gradually shifts the focus from the
   whole convolution to its maximal entries.

3. **Guided hill‑climbing refinement** – after the smooth phase we repeatedly
   move a small amount of mass away from the most offending lag (the lag that
   attains the current maximum) and deposit it at positions that are currently
   light.  This local search is cheap and often yields a noticeable final gain.

The whole routine respects the supplied time budget (default 1000 s).  All
helper functions are top‑level and use only NumPy / math / time.  The final
sequence is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector `x`.  Length = 2·len(x)-1."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # Numerical noise may produce tiny negative values – clip them.
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors `x` and `y` using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector `v` onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray,
                       cur_val: float,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local improvement: repeatedly move a small fraction of the mass that most
    contributes to the current maximal convolution lag to two low‑mass
    positions.  Stops when the time budget expires.
    """
    n = a.shape[0]
    step_frac = 0.05
    best_a = a.copy()
    best_val = cur_val

    while time.time() < deadline:
        conv = _full_convolution(best_a)
        max_b = conv.max()
        # all lags attaining (almost) the maximum
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        if ks.size == 0:
            break
        k = int(rng.choice(ks))

        # find the pair (i,j) with the largest contribution a_i*a_j to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = best_a[i_range] * best_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # pick two destination indices that are currently light
        low_mass = np.where(best_a < best_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        # avoid recreating the same lag (k = p+q)
        attempts = 0
        while p + q == k and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # perform a feasible transfer
        if i_max != j_max:
            delta = min(best_a[i_max], best_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = best_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            delta = best_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = best_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        # exact evaluation (fast shortcut not needed – we call the public function)
        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:       # defensive – should never happen
            continue

        if val < best_val:
            best_a = new_a
            best_val = val
            print(f"[guided] new best = {best_val:.6f}")
            # slowly shrink the transfer amount to fine‑tune
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            # if not improving, back off a little
            step_frac = max(step_frac * 0.995, 1e-4)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine runs for at most ``budget_s`` seconds,
    employing a smoothed‑max mirror‑descent phase followed by a short guided
    hill‑climb refinement.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size: a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 6000

    # initialise with a uniform distribution (sum = 1)
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # initialise bookkeeping
    conv = _full_convolution(a)
    best_val = 2.0 * n * conv.max()   # exact objective when sum(a)=1
    best_a = a.copy()
    print(f"[init] n={n}, initial objective = {best_val:.6f}")

    # optimisation hyper‑parameters
    beta = 1.0                # smoothing temperature
    beta_max = 400.0
    beta_growth = 1.03        # multiply every iteration
    eta0 = 0.30               # base learning rate for exponentiated step
    not_improve_cnt = 0
    stagnation_limit = 500    # trigger a random perturbation after this many idle iters

    iteration = 0
    while time.time() < deadline - 3.0:          # keep a few seconds for final hill‑climb
        iteration += 1

        # ----- compute current convolution and objective -----
        conv = _full_convolution(a)
        max_b = conv.max()
        cur_obj = 2.0 * n * max_b

        # update global best if improved
        if cur_obj < best_val - 1e-12:
            best_val = cur_obj
            best_a = a.copy()
            not_improve_cnt = 0
            print(f"[iter {iteration}] new best = {best_val:.6f}")
        else:
            not_improve_cnt += 1

        # ----- smoothed‑max weights (softmax over convolution) -----
        # we use β·conv for numerical stability
        scaled = beta * conv
        max_scaled = scaled.max()
        exps = np.exp(scaled - max_scaled)      # subtract max to avoid overflow
        p = exps / exps.sum()                  # softmax probabilities, Σp = 1

        # ----- gradient of the smoothed max w.r.t. a -----
        # g_i = 2 * Σ_j a_j * p_{i+j}
        p_rev = p[::-1]
        grad_full = _conv_fft(a, p_rev)        # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)       # mapping to the needed slice
        g = 2.0 * grad_full[idx]               # shape (n,)

        # ----- exponentiated‑gradient (mirror descent) update -----
        eta = eta0 / math.sqrt(iteration)      # diminishing step size
        a_new = a * np.exp(-eta * g)           # element‑wise
        # re‑normalise to stay on the simplex
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full(n, 1.0 / n)
        else:
            a_new /= s
        a = a_new

        # ----- schedule for β (smoothness) -----
        if beta < beta_max:
            beta = min(beta * beta_growth, beta_max)

        # ----- occasional random restart to escape plateaus -----
        if not_improve_cnt >= stagnation_limit:
            perturb = rng.normal(scale=0.02, size=n)
            a = _simplex_projection(a + perturb)
            not_improve_cnt = 0
            print(f"[iter {iteration}] random perturbation to escape stagnation")

    # ------------------------------------------------------------------
    # Final guided hill‑climbing refinement (uses the exact evaluate_sequence)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")

    return best_a.tolist()
