"""
propose_candidate
-----------------
An improved search for the step‑function problem.

The algorithm consists of three stages:

1. **Smooth surrogate optimisation** – many independent L‑BFGS runs on a
   soft‑max parameterisation of the step heights, minimising a
   log‑sum‑exp smoothing of the maximum of the self‑convolution.
   A finer ε‑schedule and a smaller problem size (n≈2500) give many
   cheap restarts within the time budget.

2. **Targeted simulated annealing** – rather than blind random swaps,
   each SA iteration focuses on the shift *k* that currently attains the
   convolution maximum.  Indices are chosen with probability proportional
   to their contribution a[i]·a[k‑i] to that peak, and either a swap or a
   small mass‑transfer move is proposed.  This concentrates the search
   where it matters most and accelerates convergence.

3. **Final hill‑climbing** – a cheap neighbourhood search consisting of
   swaps and tiny mass transfers to polish the best solution.

All phases respect the 1000 s budget, print progress information and
finally return a plain Python list of non‑negative floats ready for
``evaluate_sequence``.  The implementation uses only NumPy, SciPy and the
standard library.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders for the smooth surrogate (avoid closures)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)   # clamp tiny negatives
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate of the max‑convolution and its gradient w.r.t.
    the soft‑max parameters w.
    """
    # --- soft‑max (guarantees non‑negativity and Σa = 1) ----
    w_shift = w - np.max(w)                # numerical stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # --- self‑convolution (full) ----
    conv = _fft_convolve_self(a)           # length 2·n‑1
    max_conv = conv.max()

    # --- smooth approximation of max(conv) ----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # log∑ exp((conv−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # --- gradient w.r.t. a (closed‑form) ----
    weights = np.exp(shifted)
    p = weights / weights.sum()           # probability over shift indices
    p_rev = p[::-1]                       # needed for convolution

    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    # middle n entries correspond to indices n‑1 … 2n‑2
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # --- chain rule through the soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper for scipy.optimize.minimize (uses global eps & n)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# bind the helper convolution to the function (so it can be called inside)
_smooth_obj_and_grad._conv_fft = _conv_fft


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Deterministic ordering that pairs large heights with small ones.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large to the left
        new_a[n - 1 - i] = sorted_vals[i]      # small to the right

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _eval_fast(a: np.ndarray) -> float:
    """Exact evaluation of ``evaluate_sequence`` – much faster via FFT."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _targeted_simulated_annealing(a: np.ndarray, cur_val: float,
                                 deadline: float) -> (np.ndarray, float):
    """
    Simulated annealing that concentrates moves on the shift presently
    responsible for the maximum convolution value.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    T = max(cur_val * 0.02, 1e-9)         # initial temperature
    decay = 0.9999
    eps_mass = 0.07                       # mass‑transfer factor

    it = 0
    while time.time() < deadline:
        # ---- current peak shift and contributions ----
        conv = _fft_convolve_self(a)
        k_max = int(np.argmax(conv))

        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[partner[mask]]
        total = contrib.sum()

        # ---- pick i with bias towards large contributions ----
        if total > 0:
            probs = contrib / total
            i = rng.choice(n, p=probs)
        else:
            i = rng.integers(0, n)

        j = rng.integers(0, n)
        if i == j:
            continue

        # ---- decide move type (swap vs mass transfer) ----
        if rng.random() < 0.3:          # 30 % swaps
            a[i], a[j] = a[j], a[i]
            moved = 'swap'
        else:                           # 70 % tiny mass transfers
            if a[i] <= 0:
                continue
            delta = a[i] * eps_mass * rng.random()
            a[i] -= delta
            a[j] += delta
            moved = 'transfer'

        new_val = _eval_fast(a)
        delta_val = new_val - cur_val

        # ---- Metropolis acceptance ----
        if delta_val < 0 or rng.random() < np.exp(-delta_val / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[TS‑SA] iter {it:6d} – new best = {best_val:.6f}")
        else:
            # revert the move
            if moved == 'swap':
                a[i], a[j] = a[j], a[i]
            else:
                a[i] += delta
                a[j] -= delta

        # ---- cooling schedule ----
        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        it += 1

    return best_seq, best_val


def _refine_hill_climb(a: np.ndarray, cur_val: float,
                       deadline: float) -> (np.ndarray, float):
    """
    Straightforward hill‑climbing (pure swaps + tiny mass transfers).
    """
    cur_a = a.copy()
    cur_val = cur_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:                     # a pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                       # a mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point.  Returns a list of non‑negative heights that
    (hopefully) improves over the previous best bound.
    """
    # ------------------------------------------------------------------
    # Basic timing set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # little safety margin
    print(f"[init] total budget = {budget_s}s → deadline @ {deadline:.2f}")

    # ------------------------------------------------------------------
    # Problem size – a modest 2500 gives many cheap iterations
    # ------------------------------------------------------------------
    n = 2500
    globals()['_GLOBAL_N'] = n

    # ------------------------------------------------------------------
    # Starting point – try to reuse a previous construction if present
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to required size
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            a0 = np.interp(new_x, old_x, old)
        else:
            a0 = old.copy()
        print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)      # uniform start
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()                # enforce Σa = 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Smooth surrogate optimisation – many inexpensive L‑BFGS restarts
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-10), num=18)

    restart = 0
    while time.time() < deadline - 200.0:          # reserve ~200 s for later phases
        restart += 1
        print(f"[restart {restart}]")
        w = np.random.randn(n) * 2.0                # random soft‑max parameters

        for eps in eps_schedule:
            if time.time() >= deadline - 180.0:
                break
            globals()['_GLOBAL_EPS'] = eps

            # L‑BFGS on the smoothed objective
            res = minimize(_smooth_obj_and_grad_wrapper, w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 800,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # Extract the current a
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()

            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # ------------------------------------------------------------------
        # Deterministic anti‑symmetric re‑ordering (large ↔ small)
        # ------------------------------------------------------------------
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

        # ------------------------------------------------------------------
        # Optional extra symmetry check (average with its mirror)
        # ------------------------------------------------------------------
        a_sym = (best_a + best_a[::-1]) * 0.5
        val_sym = _eval_fast(a_sym)
        if val_sym < best_val:
            best_val = val_sym
            best_a = a_sym.copy()
            print(f"[symmetrise] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Targeted simulated annealing on the permutation of the best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        print("[TS‑SA] starting targeted simulated annealing")
        best_a, best_val = _targeted_simulated_annealing(best_a,
                                                         best_val,
                                                         deadline - 60.0)

    # ------------------------------------------------------------------
    # Final cheap hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report the official evaluation (using the provided function)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list – required by the framework
    return best_a.tolist()
