def _simulated_annealing(a, cur_val, deadline, rng, n):
    cur_a = a.copy()
    cur_val = cur_val
    best_local_a = cur_a.copy()
    best_local_val = cur_val

    min_temp = 1e-6

"""
Optimising a non‑negative step‑function to minimise
evaluate_sequence().

The algorithm works in three phases:

1. **Projected sub‑gradient optimisation** – minimise the
   maximal value of the full autocorrelation (the “max‑conv”
   term).  The sub‑gradient is taken over all convolution
   indices that are within a small relative tolerance of the
   current maximum, which gives a more informative direction
   than using only the exact maximisers.  After each gradient
   step the vector is projected onto the probability simplex
   (non‑negative entries that sum to one).

2. **Guided hill‑climbing** – repeatedly pick the offending
   lag (where the convolution is maximal) and move a small
   fraction of the mass from the pair that contributes most
   to that lag to two other positions with low mass.  This
   locally reduces the peak while keeping the sum fixed.

3. **Simple simulated‑annealing mass‑transfer** – perform many
   tiny random transfers of mass between two randomly chosen
   indices and accept worsening moves with a temperature that
   slowly cools.  The objective used internally is the same
   as evaluate_sequence but computed directly (avoiding the
   extra validation overhead).

The whole routine respects a 1000 s time budget; progress and
intermediate scores are printed.  The final list of heights is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full autocorrelation of a real vector (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative artefacts caused by floating point errors
    np.maximum(conv, 0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient optimisation on max‑convolution
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator,
                          rel_tol: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns the best vector found and its objective value.
    """
    n = a0.shape[0]
    a = a0.copy()
    # initialise objective (sum is (very close to) 1 after projection)
    cur_val = 2.0 * n * _full_convolution(a).max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = cur_val

    step0 = 0.5               # initial step amplitude
    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution(a)
        M = conv.max()
        eps = max(1e-12, M * rel_tol)           # active set tolerance
        ks = np.where(conv >= M - eps)[0]       # indices with near‑max value

        # compute sub‑gradient: g_i = Σ_{k∈S} 2·a_{k‑i}
        g = np.zeros_like(a)
        for k in ks:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
            g[i_range] += 2.0 * a[k - i_range]

        # diminishing step size
        step = step0 / np.sqrt(it)

        cand = a - step * g
        cand = _simplex_projection(cand)

        cand_val = 2.0 * n * _full_convolution(cand).max() / (cand.sum() ** 2)

        # store global best
        if cand_val < best_val - 1e-12:
            best_val = cand_val
            best_a = cand.copy()
            print(f"[subgrad iter {it}] new best = {best_val:.6f}")

        # accept improvement, otherwise adapt step size
        if cand_val < cur_val - 1e-14:
            a = cand
            cur_val = cand_val
            no_improve = 0
            step0 = min(step0 * 1.05, 1.0)   # gently increase aggressiveness
        else:
            no_improve += 1
            step0 *= 0.7
            if no_improve >= 200:
                # escape a flat region: small random perturbation around the best solution
                pert = rng.normal(scale=0.01, size=n)
                a = _simplex_projection(best_a + pert)
                cur_val = 2.0 * n * _full_convolution(a).max() / (a.sum() ** 2)
                no_improve = 0
                step0 = 0.5

    return best_a, best_val


# ----------------------------------------------------------------------
# Guided hill‑climbing refinement (mass transfers)
# ----------------------------------------------------------------------
def _guided_hill_climb(a: np.ndarray,
                       cur_val: float,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Local optimisation that repeatedly moves a small fraction of
    mass away from the most offending convolution lag.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = a.copy()
    best_val = cur_val
    step_frac = 0.05       # fraction of mass that will be moved each step

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        # find the pair (i,j) that contributes most to this lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # choose two destination indices that currently have low mass
        low_mass = np.where(cur_a < cur_a.mean())[0]
        if low_mass.size >= 2:
            p, q = rng.choice(low_mass, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        # avoid recreating the same lag when possible
        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # perform the mass transfer
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max, only one source index
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        # fast evaluation (identical to evaluate_sequence for sum≈1)
        s = new_a.sum()
        max_b_new = _full_convolution(new_a).max()
        val = 2.0 * n * max_b_new / (s ** 2)

        if val < cur_val:
            cur_a = new_a
            cur_val = val
            if val < best_val:
                best_a = new_a.copy()
                best_val = val
                print(f"[guided] new best = {best_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing mass‑transfer phase
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         cur_val: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very cheap SA that repeatedly transfers a tiny amount of mass
    between two random positions.  Worsening moves are accepted
    with probability exp(-Δ/temperature).  Temperature and step size
    slowly decay.
    """
    n = a.shape[0]
    cur_a = a.copy()
    best_a = a.copy()
    best_val = cur_val

    temp = 0.01                # initial temperature
    min_temp = 1e-6
    sigma = max(0.01 / n, 1e-8)   # typical magnitude of a single transfer
    min_sigma = 1e-9

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = rng.random() * sigma
        delta = min(delta, cur_a[i])   # keep non‑negative
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        s = new_a.sum()
        max_b = _full_convolution(new_a).max()
        val = 2.0 * n * max_b / (s ** 2)

        accept = False
        if val < cur_val:
            accept = True
        else:
            prob = np.exp(-(val - cur_val) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_a = new_a
            cur_val = val
            if val < best_val:
                best_a = new_a.copy()
                best_val = val
                print(f"[SA] new best = {best_val:.6f}")

        # cool down
        temp = max(temp * 0.9999, min_temp)
        sigma = max(sigma * 0.99995, min_sigma)

    return best_a, best_val
