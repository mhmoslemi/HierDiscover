We minimise the maximum of the autoconvolution `b = a∗a`.  The code
uses three ideas:

1. **Smooth approximation of the max** – a temperature‑controlled
   soft‑max `Φ_β(b) = (1/β)·log Σ exp(β·b)`.  Its gradient is a
   weighted sum of the contributions of every convolution entry,
   which can be computed in `O(n log n)` with a pair of FFTs.

2. **Projected gradient descent on the simplex** – after each
   (additive) step we project back onto the probability simplex
   `{x≥0, Σx = 1}` using the classic O(n log n) algorithm.

3. **Deterministic hill‑climbing polish** – once the soft‑max phase
   stalls we switch to a fast greedy local search that moves a tiny
   amount of mass from the index that contributes most to the current
   peak towards the index that contributes least, updating the full
   convolution in linear time.

The routine respects the 1000 s wall‑time limit, prints occasional
progress reports and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x >= 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    # numerical safety – renormalise exactly to 1
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution a∗a (real vectors)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a∗a (length 2·len(a)-1)."""
    n = a.shape[0]
    out_len = 2 * n - 1
    size = 1 << ((out_len - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Linear correlation of a with w (returns length len(a)+len(w)-1)
# ----------------------------------------------------------------------
def _correlation_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear correlation (a ⋆ w)[k] = Σ_i a[i]·w[i+k].
    Returned length = len(a)+len(w)-1.
    """
    n = a.shape[0]
    m = w.shape[0]
    out_len = n + m - 1
    size = 1 << ((out_len - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w, size)
    corr = np.fft.irfft(fa * np.conj(fw), size)[:out_len]
    return corr


# ----------------------------------------------------------------------
# Greedy hill‑climbing: move a little mass from the “source”
# (largest contribution to the current peak) to the “target”
# (smallest contribution).  Convolution updated in O(n).
# ----------------------------------------------------------------------
def _greedy_hillclimb(a: np.ndarray,
                      deadline: float,
                      best_a: np.ndarray,
                      best_val: float) -> np.ndarray:
    n = a.shape[0]
    b = _full_convolution_fft(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    IMPROVE_LIMIT = 150_000   # stagnation trigger
    no_improve = 0
    it = 0

    while time.time() < deadline:
        k = max_idx

        # contribution vector for the current worst index
        contrib = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))

        if source == target:
            # degenerate – pick a random pair
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] < 1e-15:
            # fallback source if empty
            nz = np.where(a > 1e-15)[0]
            if nz.size == 0:
                break
            source = int(np.random.choice(nz))

        # move a modest fraction of the source mass
        delta = a[source] * 0.25 * np.random.rand()
        if delta <= 0.0:
            continue

        # incremental update of the full convolution
        b_new = b.copy()
        # add 2·δ·a to the target‑shifted block, subtract from source‑shifted block
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        # pure δ² terms (appear at positions 2·target, 2·source and sum index)
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val - 1e-12:
            # accept move
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[Hill {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        if no_improve > IMPROVE_LIMIT:
            # escape stagnation with a tiny Gaussian jitter
            a = best_a.copy()
            a += np.random.normal(scale=0.005, size=n)
            a = _project_simplex(a)
            b = _full_convolution_fft(a)
            cur_max = b.max()
            cur_val = 2.0 * n * cur_max
            max_idx = int(np.argmax(b))
            no_improve = 0
            print(f"[Hill {it:7d}] stagnation – perturbing")
        it += 1
    return best_a


# ----------------------------------------------------------------------
# Main entry point expected by the outer harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Uses a soft‑max gradient descent on the exact max‑convolution objective,
    followed by a greedy hill‑climbing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # 0.5 s safety margin

    # ------------------------------------------------------------------
    # Problem size – a few ten‑thousands of coefficients give extra freedom
    # ------------------------------------------------------------------
    n = 12000 if budget_s >= 800 else 8000   # adapt to very short budgets

    # ------------------------------------------------------------------
    # Initialise a – try to reuse the previously discovered construction.
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # pylint: disable=undefined-variable
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)   # enforce Σa = 1, a ≥ 0
    best_a = a.copy()
    b = _full_convolution_fft(a)
    best_val = 2.0 * n * b.max()

    # ------------------------------------------------------------------
    # Split the wall‑time: ~70 % for smooth gradient descent,
    # ~30 % for final hill‑climbing.
    # ------------------------------------------------------------------
    split = start_time + 0.70 * (deadline - start_time)
    grad_deadline = split
    hill_deadline = deadline

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max gradient descent on max(b)
    # ------------------------------------------------------------------
    it = 0
    no_improve = 0
    IMPROVE_LIMIT = 50_000
    while time.time() < grad_deadline:
        # ---- current convolution & max ----
        b = _full_convolution_fft(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            no_improve = 0
            print(f"[Grad {it:7d}] new best = {best_val:.6f}")
        else:
            no_improve += 1

        # ---- schedules for step size and temperature β ----
        elapsed = time.time() - start_time
        total = deadline - start_time
        progress = elapsed / total
        # step size decays from 0.15 to 0.001
        step = 0.15 * (1.0 - progress) + 0.001 * progress
        # temperature rises from 0.5 to 25 (more aggressive later)
        beta = 0.5 + 24.5 * (progress ** 0.6)

        # ---- soft‑max weights ----
        max_b = cur_max
        shifted = b - max_b                 # ≤ 0, zero at the peak(s)
        w = np.exp(beta * shifted)          # w_i ∈ (0,1], sum ≤ len(b)
        Z = w.sum() + 1e-30                 # avoid div‑by‑zero

        # ---- gradient of the smooth max ----
        # correlation of a with w gives ∂/∂a_i Σ_j w_{i+j} a_j
        corr = _correlation_fft(a, w)       # length = n + len(w) - 1
        grad = 2.0 * corr[:n] / Z

        # optional aggressive clipping (helps stability)
        clip_val = 5.0 / n
        grad = np.clip(grad, -clip_val, clip_val)

        # ---- projected additive update ----
        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        # ---- occasional random restart to escape plateaus ----
        if no_improve > IMPROVE_LIMIT:
            a = np.random.rand(n)
            a = _project_simplex(a)
            no_improve = 0
            print(f"[Grad {it}] restart after stagnation")
        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – deterministic hill‑climbing polish
    # ------------------------------------------------------------------
    best_a = _greedy_hillclimb(best_a, hill_deadline, best_a, best_val)

    # ------------------------------------------------------------------
    # Final evaluation (the wrapper may clamp values)
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported = {final_score:.6f}")
    except Exception as exc:                    # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()
