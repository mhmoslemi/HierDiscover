"""
Candidate generation for the convolution‑maximisation problem.

The objective is

    f(a) = 2·n·max(conv(a,a)) / (sum(a)^2) ,

with a_i ≥ 0.  It is invariant under a global scaling of the vector,
therefore the task is to *flatten* the self‑convolution: minimise its
largest entry while keeping the L1‑norm as large as possible.

The algorithm works in three stages:

1. **Initialisation** – a Catalan (central‑binomial) sequence,
   which satisfies  conv(a,a)[k] = 1 for 0 ≤ k ≤ n‑1 and is a strong
   starting point (objective ≈ 1.57).

2. **Mass‑transfer simulated annealing** – repeatedly move a small
   amount of mass from indices that contribute heavily to the current
   maximal convolution entries to indices that contribute little.
   The “gradient” of the max‑convolution is approximated by looking at
   the top‑K convolution coefficients (K≈50).  Moves are accepted if
   they improve the objective, otherwise they are accepted with a
   temperature‑dependent probability (simulated annealing).

3. **Guided hill‑climbing refinement** – a short deterministic routine
   that builds on the previous implementation, moving mass away from
   the single pair (i,j) that attains the current maximum.

All steps respect the 1000 s budget; the best sequence found is printed
and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (real data)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full (linear) convolution of a real vector with itself."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negative round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Problem‑specific helpers
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    s = a.sum()
    # guard against division by zero (never happens here)
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _approx_gradient(a: np.ndarray, conv: np.ndarray, top_k: int = 50) -> np.ndarray:
    """
    Approximate the sub‑gradient of the max‑convolution.
    Only the `top_k` largest convolution entries are taken as the
    active set – this reduces the O(n²) work to O(top_k·n).
    """
    n = a.shape[0]
    # indices of the largest coefficients (might be many ties)
    if top_k >= conv.size:
        active = np.arange(conv.size)
    else:
        # argpartition gives the indices of the top_k values (unsorted)
        part = np.argpartition(-conv, top_k - 1)[:top_k]
        # keep only those that are truly close to the maximum
        max_val = conv.max()
        tol = max_val * 1e-12
        active = part[conv[part] >= max_val - tol]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        # each pair (i, k-i) contributes 2·a_{k-i} to the gradient
        g[i_range] += 2.0 * a[k - i_range]
    return g


def _simple_transfer(a: np.ndarray, i_src: int, i_dst: int, frac: float) -> np.ndarray:
    """
    Transfer a fraction `frac` of the mass at `i_src` to `i_dst`.
    The operation respects non‑negativity.
    """
    delta = a[i_src] * frac
    if delta <= 0.0:
        return None
    new_a = a.copy()
    new_a[i_src] -= delta
    new_a[i_dst] += delta
    return new_a


def _guided_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Very short deterministic refinement (identical to the previous
    implementation).  Moves a small fraction of the mass away from the
    pair (i,j) that currently realises the maximal convolution entry.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    step_frac = 0.05

    while time.time() < deadline:
        conv = _full_convolution(cur_a)
        max_b = conv.max()
        ks = np.flatnonzero(np.abs(conv - max_b) < 1e-15)
        k = int(rng.choice(ks))

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = cur_a[i_range] * cur_a[j_range]
        if contrib.size == 0:
            continue
        idx = int(np.argmax(contrib))
        i_max = int(i_range[idx])
        j_max = int(j_range[idx])

        # choose a low‑mass destination (different from i_max, j_max)
        low = np.where(cur_a < cur_a.mean())[0]
        if low.size >= 2:
            p, q = rng.choice(low, size=2, replace=False)
        else:
            p, q = rng.integers(0, n, size=2)

        attempts = 0
        while (p + q == k) and attempts < 5:
            p, q = rng.integers(0, n, size=2)
            attempts += 1

        # perform a feasible transfer
        if i_max != j_max:
            delta = min(cur_a[i_max], cur_a[j_max]) * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[j_max] -= delta
            new_a[p] += delta
            new_a[q] += delta
        else:   # i_max == j_max
            delta = cur_a[i_max] * step_frac
            if delta <= 1e-14:
                continue
            new_a = cur_a.copy()
            new_a[i_max] -= delta
            new_a[p] += delta

        if (new_a < 0).any():
            continue

        val = _objective(new_a)
        if val < cur_val:
            cur_a = new_a
            cur_val = val
            print(f"[guided] improved = {cur_val:.6f}")
            step_frac = max(step_frac * 0.99, 1e-4)
        else:
            step_frac = max(step_frac * 0.995, 1e-4)

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    Returns the best list of non‑negative heights found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – 4000 gives a good trade‑off between quality and speed
    n = 4000

    # ------------------------------------------------------------------
    # 1. Initialise with the Catalan (central‑binomial) sequence
    # ------------------------------------------------------------------
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # C_{i} = C_{i-1} * (2*i-1) / (2*i)
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)

    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val
    print(f"[init] n={n}, start objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing style mass‑transfer optimisation
    # ------------------------------------------------------------------
    # annealing schedule
    T0 = best_val * 0.01          # initial temperature (≈ 1 % of the score)
    temperature = T0
    it = 0
    step_frac = 0.25              # maximal fraction of a source entry to move
    no_improve_cnt = 0

    while time.time() < deadline:
        it += 1

        # compute the convolution once per iteration
        conv = _full_convolution(a)
        max_b = conv.max()

        # approximate sub‑gradient based on the largest convolution entries
        g = _approx_gradient(a, conv, top_k=50)

        # source = index with largest gradient (most responsible for the max)
        src = int(np.argmax(g))
        # destination = index with smallest gradient (least responsible)
        # avoid picking the same index
        dst_candidates = np.argsort(g)[:5]  # a few of the smallest entries
        dst = int(dst_candidates[0])
        if dst == src:
            if len(dst_candidates) > 1:
                dst = int(dst_candidates[1])
            else:
                # fallback: pick a random low‑mass location
                low = np.where(a < a.mean())[0]
                if low.size == 0:
                    dst = (src + n // 2) % n
                else:
                    dst = int(rng.choice(low))

        # randomise the actual fraction a little to keep the search stochastic
        frac = step_frac * rng.random()
        cand = _simple_transfer(a, src, dst, frac)
        if cand is None:
            continue
        cand_val = _objective(cand)

        # simulated annealing acceptance
        if cand_val < cur_val:
            a = cand
            cur_val = cand_val
            no_improve_cnt = 0
        else:
            # accept worse move with probability exp(-Δ/T)
            prob = np.exp(-(cand_val - cur_val) / temperature)
            if rng.random() < prob:
                a = cand
                cur_val = cand_val
                no_improve_cnt = 0
            else:
                no_improve_cnt += 1

        # keep the globally best solution
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[iter {it}] new best = {best_val:.6f}")

        # cooling schedule (linear in time)
        elapsed = time.time() - start_time
        temperature = max(T0 * (1.0 - elapsed / budget_s), 1e-12)

        # gradually shrink the step size if we are stuck
        if no_improve_cnt > 2000:
            step_frac = max(step_frac * 0.7, 0.01)
            no_improve_cnt = 0

    # ------------------------------------------------------------------
    # 3. Final deterministic hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _guided_hill_climb(best_a, best_val, deadline, rng)

    # ------------------------------------------------------------------
    # Final reporting (use the provided `evaluate_sequence` for the exact
    # score that will be judged).
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] reported score = {final_score:.6f}")

    return best_a.tolist()
