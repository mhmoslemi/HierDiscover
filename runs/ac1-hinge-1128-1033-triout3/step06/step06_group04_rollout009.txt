"""
Improved search for step-function heights minimizing evaluate_sequence.

Algorithm:
1. Initialise from previous solution if present; otherwise uniform.
2. Perform a projected sub‑gradient descent on the nonsmooth
   objective f(a)=2·n·max(conv(a,a)).  The sub‑gradient is the
   derivative of the convolution entry that currently attains the
   maximum.  After each trial step we project onto the probability
   simplex (non‑negative, sum 1) by clipping and renormalising.
   A simple back‑tracking line‑search adapts the step size.
3. Follow up with an aggressive “mass‑transfer” hill‑climbing phase.
   In each iteration we locate the lag where the convolution is
   maximal, pick the pair of indices that contributes most to that
   lag and move a fraction of their mass to two low‑mass positions.
   The move is kept only if it strictly improves the exact objective.
   The step‑fraction is increased after successes and decreased after
   failures.
4. Occasionally (after a given number of unsuccessful moves) a small
   random perturbation of the current best solution is tried to escape
   plateaus.
5. The routine respects the supplied time budget and returns the best
   list of heights found.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using an FFT (real input)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # Numerical noise may create tiny negatives; clip them.
    np.maximum(conv, 0, out=conv)
    return conv

def _objective(a: np.ndarray) -> float:
    """Exact evaluate_sequence value for a (sum = 1)."""
    n = a.shape[0]
    return 2.0 * n * np.max(_full_convolution(a))

def _gradient_of_lag(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry conv[k] with respect to a.
    For non‑negative a the gradient is 2 * a_{k‑i} on the feasible range.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = k - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x >= 0, sum x = 1}.  Implementation follows the algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    # shift to make non‑negative
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / (np.arange(len(u)) + 1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 10000) -> tuple[np.ndarray, float]:
    """
    Run a projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    # initial step size
    step0 = 0.5
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # sub‑gradient for this lag
        g = _gradient_of_lag(a, k)

        # back‑tracking line search
        lr = step0 / np.sqrt(t)
        improved = False
        for _ in range(10):
            a_trial = a - lr * g
            a_trial = np.clip(a_trial, 0.0, None)
            s = a_trial.sum()
            if s == 0.0:
                # degenerate move – shrink step and try again
                lr *= 0.5
                continue
            a_trial /= s  # project onto simplex (exact normalisation)
            val = _objective(a_trial)
            if val < best_val:
                # accept the move
                a = a_trial
                best_a = a_trial.copy()
                best_val = val
                improved = True
                # slowly increase step size after a success
                step0 = min(step0 * 1.02, 2.0)
                # print progress occasionally
                if t % 200 == 0:
                    print(f"[subgrad] iter {t:5d}  val = {best_val:.6f}")
                break
            else:
                lr *= 0.5   # shrink step and retry
        if not improved:
            # after failing to improve, shrink the maximal step size a bit
            step0 = max(step0 * 0.95, 1e-4)

    return best_a, best_val

def _mass_transfer_hillclimb(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_no_improve: int = 2000) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climbing using mass‑transfer moves.
    The routine stops after the time budget expires or after
    `max_no_improve` consecutive unsuccessful attempts.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    # adaptive fraction of mass to move
    step_frac = 0.2
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices that contribute to the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        # pick the single pair with largest contribution
        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass we are willing to relocate
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            # nothing to move – shrink step and continue
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        # remove delta from the donors
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # select two distinct destination positions with low mass
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        # re‑normalise to keep ℓ₁‑norm equal to 1 (exact scaling invariance)
        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue
        new_a /= s

        new_val = _objective(new_a)

        if new_val < cur_val:
            # successful move
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            # move rejected – reduce step size
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val

    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Determine problem size and initial point.
    # ------------------------------------------------------------------
    default_n = 4000
    try:
        old = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = int(old.shape[0])
        a0 = old.copy()
        # make sure the vector is non‑negative and normalised

    except Exception:

    except Exception as exc:   # pragma: no cover

Improved optimisation for the step‑function attack.

Algorithm
---------
1. Initialise a probability vector `a` (non‑negative, sum = 1).  If the
   global `height_sequence_1` is available it is used, otherwise a random
   Dirichlet start is generated.

2. Projected sub‑gradient descent on the exact objective  
   f(a) = 2·n·max_k (a * a)[k] .  
   The sub‑gradient is the derivative of the convolution entry that
   currently attains the maximum.  After each trial step the vector is
   clipped to non‑negative values and re‑normalised (projection onto the
   probability simplex).  A small back‑tracking line search adapts the
   step size.

3. Aggressive mass‑transfer hill‑climbing.  
   In each iteration the lag `k` where the convolution is maximal is
   located, the pair of indices contributing the most to that lag is
   identified, and a fraction of their mass is moved to two low‑mass
   positions.  The move is kept only if the exact objective improves.
   The transferred‑mass fraction is increased after successes and
   decreased after failures.

4. The routine respects the supplied time budget (default 1000 s) and
   returns the best list of heights found.  The final score is also
   printed via the provided `evaluate_sequence` function.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using an FFT (real input)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    np.maximum(conv, 0, out=conv)          # eliminate tiny negatives
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact value of `evaluate_sequence` when sum(a)=1."""
    n = a.shape[0]
    return 2.0 * n * np.max(_full_convolution(a))


def _gradient_of_lag(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry conv[k] w.r.t. a.
    For non‑negative a the derivative is 2·a_{k‑i} on the feasible range.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
    j_range = k - i_range
    grad[i_range] = 2.0 * a[j_range]
    return grad


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 10000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(conv(a,a)).
    Returns the best vector encountered and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(a)

    step0 = 0.5                     # initial step size multiplier
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        grad = _gradient_of_lag(a, k)

        # simple back‑tracking line search
        lr = step0 / np.sqrt(t)
        improved = False
        for _ in range(10):
            a_trial = a - lr * grad
            a_trial = np.clip(a_trial, 0.0, None)
            s = a_trial.sum()
            if s == 0.0:
                lr *= 0.5
                continue
            a_trial /= s                # projection onto the simplex
            val = _objective(a_trial)
            if val < best_val:
                a = a_trial
                best_a = a_trial.copy()
                best_val = val
                improved = True
                step0 = min(step0 * 1.02, 2.0)   # modestly increase step after success
                if t % 200 == 0:
                    print(f"[subgrad] iter {t:5d}  val = {best_val:.6f}")
                break
            else:
                lr *= 0.5

        if not improved:
            step0 = max(step0 * 0.95, 1e-4)

    return best_a, best_val


def _mass_transfer_hillclimb(a0: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_no_improve: int = 3000) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing phase that repeatedly moves a fraction of the most
    damaging mass to low‑mass positions.
    Stops after the time budget expires or after `max_no_improve`
    consecutive unsuccessful attempts.
    """
    a = a0.copy()
    n = a.shape[0]
    cur_val = _objective(a)
    best_a = a.copy()
    best_val = cur_val

    step_frac = 0.2                # initial fraction of mass to relocate
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        conv = _full_convolution(a)
        maxb = conv.max()
        ks = np.flatnonzero(conv == maxb)
        k = int(rng.choice(ks))

        # indices that make up the maximal lag
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
        j_range = k - i_range
        contrib = a[i_range] * a[j_range]

        idx = int(np.argmax(contrib))
        i = int(i_range[idx])
        j = int(j_range[idx])

        # amount of mass we are willing to shift
        if i == j:
            delta = a[i] * step_frac
        else:
            delta = min(a[i], a[j]) * step_frac
        if delta <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue

        new_a = a.copy()
        if i == j:
            new_a[i] -= delta
        else:
            new_a[i] -= delta
            new_a[j] -= delta

        # pick two distinct low‑mass destinations, avoiding the donors
        low = np.where(new_a < np.mean(new_a))[0]
        forbidden = {i, j}
        cand = np.setdiff1d(low, list(forbidden), assume_unique=True)
        if cand.size < 2:
            cand = np.setdiff1d(np.arange(n), list(forbidden), assume_unique=True)
        dest = rng.choice(cand, size=2, replace=False)
        p, q = int(dest[0]), int(dest[1])

        new_a[p] += delta
        new_a[q] += delta

        s = new_a.sum()
        if s <= 0.0:
            step_frac = max(step_frac * 0.8, 1e-6)
            continue
        new_a /= s

        new_val = _objective(new_a)

        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            no_improve = 0
            step_frac = min(step_frac * 1.02, 0.5)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[massclimb] new best = {best_val:.6f}")
        else:
            step_frac = max(step_frac * 0.9, 1e-6)
            no_improve += 1

    return best_a, best_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Initialise – reuse previous construction if it exists.
    # ------------------------------------------------------------------
    default_n = 4000
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = int(old.shape[0])
        a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0.0:
            a0 = rng.random(n)
        a0 /= a0.sum()
    except Exception:
        n = default_n
        a0 = rng.random(n)
        a0 /= a0.sum()

    print(f"[init] n={n}, initial objective = {_objective(a0):.6f}")

    # ------------------------------------------------------------------
    # 1. Projected sub‑gradient descent.
    # ------------------------------------------------------------------
    a_sub, val_sub = _subgradient_descent(a0, deadline, rng, max_iters=8000)
    print(f"[subgrad] best value after descent = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 2. Aggressive mass‑transfer hill‑climbing.
    # ------------------------------------------------------------------
    a_final, val_final = _mass_transfer_hillclimb(a_sub, deadline, rng)
    print(f"[final] best value before evaluate_sequence = {val_final:.6f}")

    # ------------------------------------------------------------------
    # 3. Report the score using the provided evaluator.
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(a_final.tolist())
        print(f"[final] evaluate_sequence reports = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    return a_final.tolist()
