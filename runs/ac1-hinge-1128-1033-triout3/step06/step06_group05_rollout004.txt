def _lse_gradient_phase(a, n, deadline, best_a, best_val):
    start = time.time()
    initial_alpha = 2.0
    final_alpha = 200.0
    step0 = 0.5
    step_end = 0.01
    it = 0
    while time.time() < deadline:
        # compute self convolution
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        # possibly update best
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[LSE {it:7d}] new best = {best_val:.6f}")

        # compute alpha schedule
        progress = (time.time() - start) / (deadline - start)
        alpha = initial_alpha * (1 - progress) + final_alpha * progress

        # compute softmax weights
        # subtract max to avoid overflow
        b_max = b.max()
        w = np.exp(alpha * (b - b_max))
        sum_w = w.sum()
        if sum_w == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / sum_w

        # compute gradient
        # conv(w, a) and conv(w, a[::-1]) using FFT
        grad_part1 = _conv_fft(w, a)  # length n + (2n-1) -1 = 3n-2
        grad_part2 = _conv_fft(w, a[::-1])
        # slice part [n-1:2*n-1]
        grad = 2.0 * (grad_part1[n-1:2*n-1] + grad_part2[n-1:2*n-1])

        # step schedule
        step = step0 * (1 - progress) + step_end * progress

        # update and project onto simplex
        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        it += 1
    return a, best_a, best_val

"""
Propose a step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

with n≈8000.  The search consists of three phases:

1. Sub‑gradient descent on the exact max (fast cheap descent).
2. Gradient descent on a smooth approximation of the max via a soft‑max
   (log‑sum‑exp) surrogate.  The temperature α grows from 2 to 200,
   focusing the descent increasingly on the worst convolution entries.
3. Simulated‑annealing hill‑climbing that moves a small amount of mass
   from a “source” index (largest contribution to the current worst
   convolution index) to a “target” index (smallest contribution).  Moves
   are accepted if they improve the objective or with a temperature‑
   dependent probability, allowing the search to escape local minima.

All updates keep the vector non‑negative and on the probability simplex
(Σa = 1).  The routine respects the 1000 s wall‑time limit and prints
intermediate best scores.  The final sequence is returned as a plain
Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length = 2·len(x)‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = 1 << (out_len - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx=1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = v - theta
    w[w < 0] = 0.0
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:                           # safety fallback
        w = np.full(n, 1.0 / n, dtype=np.float64)
    return w

    min_step = 1e-4
    it = 0
    while time.time() < deadline:
        b = _full_convolution(a)
        k = int(np.argmax(b))
        cur_max = b[k]
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Subgrad {it:7d}] new best = {best_val:.6f}")

        # gradient of max(b) w.r.t. a (sub‑gradient at index k)
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        grad[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                grad[mid] = 2.0 * a[mid]

        # linear step‑size schedule
        progress = (time.time() - start) / (deadline - start)
        step = max_step * (1.0 - progress) + min_step * progress

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        it += 1
    return a, best_a, best_val


# ----------------------------------------------------------------------
# Phase B – Gradient descent on a smoothed max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _lse_gradient_phase(a: np.ndarray,
                       n: int,
                       deadline: float,
                       best_a: np.ndarray,
                       best_val: float) -> tuple[np.ndarray, np.ndarray, float]:
    """Minimise log‑sum‑exp of the convolution (smooth max)."""
    start = time.time()
    it = 0
    # schedules
    alpha_start, alpha_end = 2.0, 200.0      # temperature for the soft‑max
    step_start, step_end = 0.5, 0.001      # learning‑rate schedule
    while time.time() < deadline:
        b = _full_convolution(a)
        cur_max = b.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[LSE {it:7d}] new best = {best_val:.6f}")

        # schedule for α (soft‑max temperature) and step size
        progress = (time.time() - start) / (deadline - start)
        alpha = alpha_start * (1 - progress) + alpha_end * progress
        step = step_start * (1 - progress) + step_end * progress

        # soft‑max weights (stable formulation)
        b_max = b.max()
        w = np.exp(alpha * (b - b_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the weighted sum of convolutions
        conv1 = _conv_fft(w, a)               # length = n + (2n‑1) - 1 = 3n‑2
        conv2 = _conv_fft(w, a[::-1])
        grad = 2.0 * (conv1[n - 1:2 * n - 1] + conv2[n - 1:2 * n - 1])

        a_candidate = a - step * grad
        a = _project_simplex(a_candidate)

        it += 1
    return a, best_a, best_val


# ----------------------------------------------------------------------
# Phase C – Simulated‑annealing hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_sa(a: np.ndarray,
                  n: int,
                  deadline: float,
                  best_a: np.ndarray,
                  best_val: float) -> np.ndarray:
    """Local search that moves mass between indices with SA acceptance."""
    start = time.time()
    b = _full_convolution(a)
    cur_max = b.max()
    cur_val = 2.0 * n * cur_max
    max_idx = int(np.argmax(b))

    it = 0
    # temperature schedule for SA (tiny values – objective differences are ≈10⁻⁶)
    T_start, T_end = 1e-5, 1e-9

    while time.time() < deadline:
        # ---- contribution vector for the current worst index ----
        k = max_idx
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1, dtype=int)
        contrib = np.zeros_like(a)
        contrib[idx] = a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if 0 <= mid < n:
                contrib[mid] = 2.0 * a[mid]

        # pick source (largest contribution) / target (smallest)
        source = int(np.argmax(contrib))
        target = int(np.argmin(contrib))
        if source == target:
            source = np.random.randint(n)
            target = np.random.randint(n)
            while target == source:
                target = np.random.randint(n)

        if a[source] <= 1e-14:
            # fall back to any positive source
            pos = np.where(a > 1e-14)[0]
            if pos.size == 0:
                break
            source = int(np.random.choice(pos))

        # amount of mass to move (up to 30 % of source)
        delta = a[source] * 0.30 * np.random.rand()
        if delta <= 0.0:
            continue

        # ---- incremental update of the full convolution ----
        b_new = b.copy()
        # linear contributions
        b_new[target:target + n] += 2.0 * delta * a
        b_new[source:source + n] -= 2.0 * delta * a
        # quadratic terms
        b_new[2 * target] += delta * delta
        b_new[2 * source] += delta * delta
        b_new[source + target] -= 2.0 * delta * delta

        new_max = b_new.max()
        new_val = 2.0 * n * new_max

        # SA acceptance
        progress = (time.time() - start) / (deadline - start)
        T = T_start * (1 - progress) + T_end * progress
        accept = False
        if new_val < cur_val - 1e-12:
            accept = True
        else:
            # uphill move probability
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a[source] -= delta
            a[target] += delta
            b = b_new
            cur_max = new_max
            cur_val = new_val
            max_idx = int(np.argmax(b))

            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[SA Hill {it:7d}] new best = {best_val:.6f}")

        it += 1
    return best_a

# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with minimal evaluate_sequence().
    Uses three optimisation phases and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – 8000 gives a good trade‑off between flexibility and speed
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a – try to interpolate the previously discovered construction
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # pylint: disable=undefined-variable
        old_x = np.linspace(0.0, 1.0, old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        a = np.interp(new_x, old_x, old)
        a = np.maximum(a, 0.0)
    except Exception:
        a = np.random.rand(n)

    a = _project_simplex(a)       # enforce Σa = 1
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[Init] initial evaluation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Split the time budget
    # ------------------------------------------------------------------
    total = deadline - start_time
    sub_end = start_time + 0.15 * total   # 15 % for sub‑gradient
    lse_end = start_time + 0.70 * total   # next 55 % for LSE gradient
    # remainder for hill‑climbing with SA

    # ------------------------------------------------------------------
    # Phase A – sub‑gradient descent
    # ------------------------------------------------------------------
    a, best_a, best_val = _subgradient_phase(a, n, sub_end, best_a, best_val)

    # ------------------------------------------------------------------
    # Phase B – smooth max (log‑sum‑exp) gradient descent
    # ------------------------------------------------------------------
    a, best_a, best_val = _lse_gradient_phase(a, n, lse_end, best_a, best_val)

    # ------------------------------------------------------------------
    # Phase C – simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    best_a = _hillclimb_sa(a, n, deadline, best_a, best_val)

    # final evaluation (the wrapper may clamp values, but our vector obeys the constraints)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[Final] best evaluation reported = {final_score:.6f}")

    return best_a.tolist()
