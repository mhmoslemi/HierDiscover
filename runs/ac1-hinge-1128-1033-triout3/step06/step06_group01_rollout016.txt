def _simulated_annealing(a, cur_val, deadline):

    n = len(a)
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val
    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99996  # slower cooling

    while time.time() < deadline:
        move_type = rng.random()
        if move_type < 0.5:
            # swap
            i, j = rng.integers(0, n, size=2)
            if i == j: continue
            a[i], a[j] = a[j], a[i]
        elif move_type < 0.8:
            # reverse segment
            i, j = sorted(rng.integers(0, n, size=2))
            if j - i <= 1: continue
            a[i:j] = a[i:j][::-1]
        else:
            # rotate
            k = rng.integers(1, n)  # shift amount
            a[:] = np.roll(a, k)

        new_val = _eval_fast(a)
        delta = new_val - cur_val

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f}")
        else:
            # revert move
            if move_type < 0.5:
                a[i], a[j] = a[j], a[i]
            elif move_type < 0.8:
                a[i:j] = a[i:j][::-1]  # reversed again to original
            else:
                a[:] = np.roll(a, -k)  # roll opposite

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

    return best_seq, best_val

def _project_simplex(v):
    # Ensure nonnegative and sum 1
    # Using algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Propose candidate for step function minimization.
Uses multi‑shape L‑BFGS with log‑sum‑exp smoothing,
anti‑symmetric reordering, extensive simulated annealing,
and a projected sub‑gradient refinement on the simplex.
The implementation stays within a 1000 s budget.
"""
import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# Global parameters for the smooth surrogate
_GLOBAL_EPS = None
_GLOBAL_N = None

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using FFT.
    Returns length 2*len(a)-1 array.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)      # clamp tiny negatives
    return conv

def _eval_fast(a: np.ndarray) -> float:
    """Exact evaluation of the objective defined in evaluate_sequence."""
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))

# ----- smooth surrogate objective for L‑BFGS -----
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε = 2 n max(conv) + 2 n ε log‑sum‑exp((conv‑max)/ε)
    and its gradient w.r.t. the soft‑max parameters w.
    """
    # soft‑max (ensures a_i ≥ 0 and Σ a_i = 1)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # self‑convolution (full)
    conv = _fft_convolve_self(a)          # length 2 n‑1
    max_conv = conv.max()

    # smoothed approximation of max
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # scalar
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a (closed form)
    weights = np.exp(shifted)
    p = weights / np.sum(weights)         # probability distribution over k

    # convolution a * reversed(p)
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    # middle n entries correspond to indices [n‑1, 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w

def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by scipy.optimize.minimize (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)

# helper convolution used inside the gradient
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv

_smooth_obj_and_grad._conv_fft = _conv_fft

# ----- re‑ordering heuristics -----
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """Place large values on one side and small ones on the opposite side."""
    n = a.shape[0]
    sorted_vals = np.sort(a)                # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]      # large
        new_a[n - 1 - i] = sorted_vals[i]      # small
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a

def _random_reorder(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    b = a.copy()
    rng.shuffle(b)
    return b

# ----- simulated annealing on permutations -----
def _simulated_annealing(a: np.ndarray, cur_val: float, deadline: float) -> (np.ndarray, float):
    """Swap / reverse / rotate moves on the ordering."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val
    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99996

    while time.time() < deadline:
        r = rng.random()
        if r < 0.5:
            # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            revert = lambda: (a.__setitem__(i, a[i]), a.__setitem__(j, a[j]))
            # we will revert manually later if needed
        elif r < 0.8:
            # reverse a random segment
            i, j = sorted(rng.integers(0, n, size=2))
            if j - i <= 1:
                continue
            a[i:j] = a[i:j][::-1]
            revert = lambda: (a.__setitem__(slice(i, j), a[i:j][::-1]))
        else:
            # rotate (circular shift)
            k = rng.integers(1, n)
            a[:] = np.roll(a, k)
            revert = lambda: (a.__setitem__(slice(None), np.roll(a, -k)))

        new_val = _eval_fast(a)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA] new best = {best_val:.6f}")
        else:
            # revert move
            revert()
        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

    return best_seq, best_val

# ----- projected sub‑gradient refinement (simple) -----
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # From Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _subgradient_refine(a: np.ndarray, deadline: float) -> (np.ndarray, float):
    """Projected sub‑gradient descent on max(conv) while staying on the simplex."""
    n = a.shape[0]
    cur_val = _eval_fast(a)
    it = 0
    while time.time() < deadline and it < 20000:
        conv = _fft_convolve_self(a)
        k = np.argmax(conv)               # index of the current peak
        # gradient of conv_k w.r.t a
        grad = np.zeros_like(a)
        idx1 = k - np.arange(n)
        mask1 = (idx1 >= 0) & (idx1 < n)
        grad[mask1] += a[idx1[mask1]]
        idx2 = np.arange(n) - k
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad[mask2] += a[idx2[mask2]]
        # correct the double‑count on the diagonal term (i=j)
        if k % 2 == 0:
            i_mid = k // 2
            if 0 <= i_mid < n:
                grad[i_mid] -= a[i_mid]   # we added it twice, need only once

        # step size schedule (simple diminishing)
        step = 0.3 / np.sqrt(it + 1)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        new_val = _eval_fast(a_new)
        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            # occasional progress report
            if it % 1000 == 0:
                print(f"[subgrad] iter {it}, val = {cur_val:.6f}")
        else:
            # keep current a (no update)
            pass
        it += 1
    return a, cur_val

# ----- cheap hill‑climbing refinement (swap + mass transfer) -----
def _refine_hill_climb(a: np.ndarray, best_val: float, deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val

# ----- initial shape generators -----
def _generate_initial_shapes(n: int, rng: np.random.Generator):
    shapes = []
    # uniform
    shapes.append(np.full(n, 1.0 / n))
    # exponential families
    for alpha in [5e-4, 1e-3, 2e-3, 5e-3]:
        a = np.exp(-alpha * np.arange(n))
        a /= a.sum()
        shapes.append(a)
    # power‑law families
    for p in [0.5, 1.0, 1.5, 2.0]:
        i = np.arange(1, n + 1)
        a = 1.0 / (i ** p)
        a /= a.sum()
        shapes.append(a)
    # half‑sine
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a /= a.sum()
    shapes.append(a)
    # Dirichlet‑like random (concentration < 1)
    a = rng.gamma(shape=0.5, scale=1.0, size=n)
    a /= a.sum()
    shapes.append(a)
    return shapes

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine combines:
        * multi‑shape L‑BFGS on a smoothed surrogate,
        * anti‑symmetric re‑ordering,
        * several rounds of simulated annealing (swap / reverse / rotate),
        * a projected sub‑gradient refinement,
        * a final cheap hill‑climb.
    It respects a strict time budget and always returns a feasible list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin
    print(f"[init] budget = {budget_s}s, deadline @ {deadline:.2f}")

    # problem size – a few thousand gives a good trade‑off
    n = 6000
    global _GLOBAL_N
    _GLOBAL_N = n

    # create a diverse collection of starting shapes
    shapes = _generate_initial_shapes(n, rng)

    # schedule of smoothing parameters for the surrogate problem
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=20)

    best_a = None
    best_val = np.inf
    restart = 0

    # ------------------------------------------------------------------
    # L‑BFGS exploration phase
    # ------------------------------------------------------------------
    for shape in shapes:
        # start from the exact log‑softmax of the shape
        w_base = np.log(shape + 1e-12)

        # two randomised restarts per shape
        for r in range(2):
            if time.time() >= deadline - 30.0:
                break
            restart += 1
            w = w_base + rng.normal(scale=0.1, size=n)   # jitter
            for eps in eps_schedule:
                if time.time() >= deadline - 25.0:
                    break
                _GLOBAL_EPS = eps
                # L‑BFGS
                res = minimize(_smooth_obj_and_grad_wrapper, w,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': 300,
                                        'ftol': 1e-12,
                                        'gtol': 1e-12,
                                        'disp': False})
                w = res.x
                # evaluate current point
                w_shift = w - np.max(w)
                a = np.exp(w_shift)
                a /= a.sum()
                cur_val = _eval_fast(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[restart {restart} eps {eps:.1e}] new best = {best_val:.6f}")

            # after the whole eps trajectory, try the anti‑symmetric ordering
            if best_a is not None:
                a_ord = _anti_symmetric_reorder(best_a)
                val_ord = _eval_fast(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    print(f"[reorder after restart {restart}] improved to {best_val:.6f}")

    if best_a is None:
        # fallback – uniform
        best_a = np.full(n, 1.0 / n)
        best_val = _eval_fast(best_a)
        print("[fallback] using uniform start")

    # ------------------------------------------------------------------
    # Additional random reorder attempts (cheap)
    # ------------------------------------------------------------------
    for attempt in range(5):
        if time.time() >= deadline - 20.0:
            break
        a_rand = _random_reorder(best_a, rng)
        val_rand = _eval_fast(a_rand)
        if val_rand < best_val:
            best_val = val_rand
            best_a = a_rand.copy()
            print(f"[rand reorder {attempt}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing on the permutation (swap / reverse / rotate)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        print("[SA] starting simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val,
                                                deadline - 5.0)

    # ------------------------------------------------------------------
    # Projected sub‑gradient refinement on the simplex
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[subgrad] projected sub‑gradient refinement")
        best_a, best_val = _subgradient_refine(best_a, deadline - 2.0)

    # ------------------------------------------------------------------
    # Final cheap hill‑climbing phase
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report final value using the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised:", exc)

    return best_a.tolist()

"""
Propose a candidate step‑function that minimises the objective defined in
``evaluate_sequence``.

The algorithm combines several ideas:

1. **Multi‑shape L‑BFGS** – a smoothed surrogate of the max‑convolution
   (log‑sum‑exp) is optimised for a variety of initial shapes
   (uniform, exponential, power‑law, half‑sine, U‑shaped, etc.).
   The descent uses a schedule of decreasing smoothing parameters.

2. **Re‑ordering** – the best vector after each L‑BFGS run is re‑ordered by the
   anti‑symmetric heuristic (large values opposite small ones).  A few random
   shuffles are also tested.

3. **Simulated annealing** – swaps, segment reversals and circular rotations
   are applied to the ordering with a temperature schedule, keeping the
   ℓ₁‑norm fixed.

4. **Projected sub‑gradient refinement** – a simple sub‑gradient descent on
   the true max‑convolution is performed while projecting back onto the
   probability simplex (non‑negative, sum = 1).

5. **Final cheap hill‑climb** – local mass transfers and swaps polish the
   solution.

All steps respect a 1000 s time budget; the function always returns a
feasible list of non‑negative floats.
"""

import time

# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # current smoothing parameter
_GLOBAL_N = None     # length of the sequence


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)          # clip tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for a non‑negative vector ``a``.
    The vector is assumed to satisfy 0 ≤ a_i ≤ 1000 and Σa_i > 0.01.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:            # mirrors the security check
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smoothed surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective
        L_ε = 2·n·max(conv) + 2·n·ε·log‑sum‑exp((conv−max)/ε)
    and its gradient w.r.t. the soft‑max parameters ``w``.
    """
    # ----- soft‑max (ensures non‑negativity & Σa = 1) -----
    w_shift = w - np.max(w)            # improve numeric stability
    a = np.exp(w_shift)
    a /= np.sum(a)

    # ----- convolution a * a (full) -----
    conv = _fft_convolve_self(a)       # length 2·n‑1
    max_conv = conv.max()

    # ----- smooth approximation of max(conv) -----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ----- gradient w.r.t. a (closed‑form) -----
    weights = np.exp(shifted)
    p = weights / np.sum(weights)      # probability distribution over lag index

    # convolution a * reversed(p)            (FFT based, helper defined below)
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)
    # middle n entries correspond to indices [n‑1, 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # ----- chain‑rule through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# ----------------------------------------------------------------------
# Helper convolution used inside the gradient (FFT based)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Re‑ordering heuristics
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Place the largest values on one side and the smallest on the opposite side.
    This often reduces the peak of the self‑convolution.
    """
    n = a.shape[0]
    s = np.sort(a)                     # ascending
    b = np.empty_like(a)
    half = n // 2
    for i in range(half):
        b[i] = s[n - 1 - i]            # large to the left
        b[n - 1 - i] = s[i]            # small to the right
    if n % 2:
        b[half] = s[half]
    return b


def _random_reorder(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a random permutation of ``a``."""
    b = a.copy()
    rng.shuffle(b)
    return b


# ----------------------------------------------------------------------
# Simulated annealing on permutations (swap / reverse / rotate)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """Classic SA on the ordering of ``a``."""
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val
    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99996

    while time.time() < deadline:
        r = rng.random()
        if r < 0.5:                     # ----- swap move -----
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            new_val = _eval_fast(a)
            delta = new_val - cur_val
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA] new best = {best_val:.6f}")
            else:
                a[i], a[j] = a[j], a[i]          # revert

        elif r < 0.8:                 # ----- segment reversal -----
            i, j = sorted(rng.integers(0, n, size=2))
            if j - i <= 1:
                continue
            a[i:j] = a[i:j][::-1]
            new_val = _eval_fast(a)
            delta = new_val - cur_val
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA] new best = {best_val:.6f}")
            else:
                a[i:j] = a[i:j][::-1]            # revert

        else:                         # ----- circular rotation -----
            k = rng.integers(1, n)
            a[:] = np.roll(a, k)
            new_val = _eval_fast(a)
            delta = new_val - cur_val
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA] new best = {best_val:.6f}")
            else:
                a[:] = np.roll(a, -k)           # revert

        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient refinement on the simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_refine(a: np.ndarray, deadline: float) -> (np.ndarray, float):
    """
    Simple projected sub‑gradient descent on ``max(conv)`` while staying on
    the ℓ₁‑simplex (non‑negative, sum = 1).
    """
    n = a.shape[0]
    cur_val = _eval_fast(a)
    it = 0
    while time.time() < deadline and it < 20000:
        conv = _fft_convolve_self(a)
        k = np.argmax(conv)                     # index of the current peak

        # gradient of conv_k w.r.t a
        grad = np.zeros_like(a)
        idx1 = k - np.arange(n)
        mask1 = (idx1 >= 0) & (idx1 < n)
        grad[mask1] += a[idx1[mask1]]
        idx2 = np.arange(n) - k
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad[mask2] += a[idx2[mask2]]
        # Diagonal term counted twice – keep only one copy
        if k % 2 == 0:
            i_mid = k // 2
            if 0 <= i_mid < n:
                grad[i_mid] -= a[i_mid]

        step = 0.3 / np.sqrt(it + 1)            # simple diminishing step size
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        new_val = _eval_fast(a_new)
        if new_val < cur_val - 1e-12:
            a = a_new
            cur_val = new_val
            if it % 1000 == 0:
                print(f"[subgrad] iter {it}, val = {cur_val:.6f}")
        it += 1
    return a, cur_val


# ----------------------------------------------------------------------
# Cheap hill‑climbing refinement (swap + mass transfer)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:            # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                             # small mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Initial shape generators
# ----------------------------------------------------------------------
def _generate_initial_shapes(n: int, rng: np.random.Generator):
    """
    Return a list of diverse probability vectors of length ``n``.
    All vectors are normalised to sum = 1.
    """
    shapes = []

    # 1) uniform
    shapes.append(np.full(n, 1.0 / n))

    # 2) exponential decay
    for alpha in [5e-4, 1e-3, 2e-3, 5e-3]:
        a = np.exp(-alpha * np.arange(n))
        a /= a.sum()
        shapes.append(a)

    # 3) power‑law decay
    for p in [0.5, 1.0, 1.5, 2.0]:
        i = np.arange(1, n + 1)
        a = 1.0 / (i ** p)
        a /= a.sum()
        shapes.append(a)

    # 4) half‑sine (positive on the whole interval)
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a /= a.sum()
    shapes.append(a)

    # 5) U‑shaped exponential (large on both ends, tiny in the centre)
    for alpha in [0.001, 0.005, 0.01]:
        dist = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a = np.exp(-alpha * dist)
        a /= a.sum()
        shapes.append(a)

    # 6) Dirichlet‑like random (concentration < 1)
    a = rng.gamma(shape=0.5, scale=1.0, size=n)
    a /= a.sum()
    shapes.append(a)

    return shapes


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine mixes L‑BFGS on a smoothed surrogate, clever re‑ordering,
    simulated annealing, a projected sub‑gradient phase and a final
    hill‑climb.  It respects a strict time budget and always returns a
    feasible list of non‑negative floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a small safety margin
    print(f"[init] time budget = {budget_s}s, deadline @ {deadline:.2f}")

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good balance
    # ------------------------------------------------------------------
    n = 6000
    global _GLOBAL_N
    _GLOBAL_N = n

    # ------------------------------------------------------------------
    # Build a diverse collection of initial shapes
    # ------------------------------------------------------------------
    shapes = _generate_initial_shapes(n, rng)

    # ------------------------------------------------------------------
    # Smoothing schedule for the surrogate objective
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=20)

    best_a = None
    best_val = np.inf
    restart = 0

    # ------------------------------------------------------------------
    # L‑BFGS exploration (multiple shapes, a couple of restarts each)
    # ------------------------------------------------------------------
    for shape in shapes:
        # log‑softmax of the shape – this yields a feasible starting point
        w_base = np.log(shape + 1e-12)

        for r in range(2):                     # two jittered restarts per shape
            if time.time() >= deadline - 30.0:
                break
            restart += 1
            w = w_base + rng.normal(scale=0.1, size=n)   # slight random perturbation

            for eps in eps_schedule:
                if time.time() >= deadline - 25.0:
                    break
                global _GLOBAL_EPS
                _GLOBAL_EPS = eps

                # L‑BFGS on the smooth surrogate
                res = minimize(_smooth_obj_and_grad_wrapper, w,
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': 300,
                                        'ftol': 1e-12,
                                        'gtol': 1e-12,
                                        'disp': False})
                w = res.x

                # evaluate the current iterate (exact objective)
                w_shift = w - np.max(w)
                a = np.exp(w_shift)
                a /= a.sum()
                cur_val = _eval_fast(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    print(f"[restart {restart} eps {eps:.1e}] new best = {best_val:.6f}")

            # after finishing the eps trajectory, try the anti‑symmetric reorder
            if best_a is not None:
                a_ord = _anti_symmetric_reorder(best_a)
                val_ord = _eval_fast(a_ord)
                if val_ord < best_val:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    print(f"[reorder after restart {restart}] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely to be needed)
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _eval_fast(best_a)
        print("[fallback] using uniform start")

    # ------------------------------------------------------------------
    # A few cheap random re‑order attempts
    # ------------------------------------------------------------------
    for attempt in range(5):
        if time.time() >= deadline - 20.0:
            break
        a_rand = _random_reorder(best_a, rng)
        val_rand = _eval_fast(a_rand)
        if val_rand < best_val:
            best_val = val_rand
            best_a = a_rand.copy()
            print(f"[rand reorder {attempt}] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing on the permutation (swap / reverse / rotate)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        print("[SA] starting simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val,
                                                deadline - 5.0)

    # ------------------------------------------------------------------
    # Projected sub‑gradient refinement (still on the simplex)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[subgrad] projected sub‑gradient refinement")
        best_a, best_val = _subgradient_refine(best_a, deadline - 2.0)

    # ------------------------------------------------------------------
    # Final cheap hill‑climbing (swap + small mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[refine] final hill‑climbing")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report the final value using the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required by the framework
    return best_a.tolist()
