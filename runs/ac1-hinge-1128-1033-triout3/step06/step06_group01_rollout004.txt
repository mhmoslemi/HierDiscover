"""
Propose a low‑value step‑function for `evaluate_sequence`.

The algorithm is a multi‑phase optimizer:

1. **Initialisation** – start from the previously found construction
   (`height_sequence_1` if available) and linearly interpolate to a larger
   problem size (default 8000).  The sequence is normalised to unit ℓ₁‑norm.

2. **Smooth surrogate optimisation** – a log‑sum‑exp (soft‑max) surrogate
   of the max‑convolution is minimised with many L‑BFGS restarts.
   The surrogate guarantees non‑negativity and a fixed ℓ₁‑norm.
   An ε‑schedule (`5·10⁻³ → 10⁻¹²`) gradually sharpens the approximation.

3. **Anti‑symmetric reordering** – after each restart the multiset of
   heights is reordered by pairing the largest values with the smallest
   (the `_anti_symmetric_reorder` heuristic).  This usually lowers the
   true peak of the convolution without changing the ℓ₁‑norm.

4. **Simulated annealing on permutations** – a fast FFT‑based evaluator
   (`_eval_fast`) repeatedly proposes swaps *or* small mass‑transfer moves.
   A temperature schedule accepts occasionally worsening moves, allowing
   the search to escape local minima.

5. **Final hill‑climbing refinement** – cheap greedy moves (swaps and
   transfers) are applied until the time budget expires.

All phases respect the 1000 s budget, report intermediate scores and finally
return the best sequence as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global placeholders used by the surrogate optimiser
# ----------------------------------------------------------------------
_GLOBAL_EPS = None   # smoothing parameter for the current L‑BFGS run
_GLOBAL_N = None     # length of the current sequence


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of ``a`` with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:out_len]
    np.maximum(conv, 0, out=conv)       # clamp tiny negatives
    return conv


def _eval_fast(a: np.ndarray) -> float:
    """
    Fast evaluation of the exact objective defined in ``evaluate_sequence``.
    ``a`` must be non‑negative and respect the clipping limits.
    Returns the same value as ``evaluate_sequence(a.tolist())`` but is
    considerably faster.
    """
    n = a.shape[0]
    conv = _fft_convolve_self(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Smooth surrogate objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate L_ε (log‑sum‑exp smoothing of the max) and its
    gradient w.r.t. the log‑softmax parameters ``w``.
    The parameter vector ``w`` is transformed by a soft‑max into a
    probability vector ``a`` that satisfies a_i ≥ 0 and Σ a_i = 1.
    """
    # ---- soft‑max (guarantees positivity and Σ a = 1) ----
    w_shift = w - np.max(w)          # numeric stability
    a = np.exp(w_shift)
    a_sum = np.sum(a)
    a = a / a_sum

    # ---- auto‑convolution a * a (full) ----
    conv = _fft_convolve_self(a)           # length 2·n‑1
    max_conv = conv.max()

    # ---- smooth approximation of max(conv) ----
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)          # log∑ exp((conv−max)/ε)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # ---- gradient w.r.t. a (closed‑form) ----
    weights = np.exp(shifted)
    p = weights / weights.sum()           # probability distribution over k

    # convolution a * p_rev  (full)
    p_rev = p[::-1]
    conv_ap = _smooth_obj_and_grad._conv_fft(a, p_rev)   # helper defined below

    # middle n entries correspond to indices [n‑1 … 2n‑2)
    seg = conv_ap[n - 1: 2 * n - 1]        # length n
    grad_a = 4.0 * n * seg[::-1]          # reverse to align with a_i

    # ---- chain rule through the soft‑max ----
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.optimize.minimize`` (uses globals)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


# Helper: convolution used inside the gradient (FFT based, same as above)
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    out_len = n + m - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:out_len]
    np.maximum(conv, 0, out=conv)
    return conv


# Bind the helper to the gradient function (so it can be called internally)
_smooth_obj_and_grad._conv_fft = _conv_fft


# ----------------------------------------------------------------------
# Re‑ordering heuristic
# ----------------------------------------------------------------------
def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Produce an ordering where large values are paired with small ones
    when the sequence is reversed.  The multiset of values stays the same.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)

    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large on the left
        new_a[n - 1 - i] = sorted_vals[i]          # small on the right

    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


# ----------------------------------------------------------------------
# Simulated annealing on permutations (swap + mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, cur_val: float,
                         deadline: float) -> (np.ndarray, float):
    """
    Classic SA on the permutation of ``a``.  Two kinds of moves are used:
      * swap two positions,
      * transfer a small fraction of mass from one position to another.
    The temperature is slowly cooled; the best solution seen is retained.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = cur_val

    # initial temperature – a small fraction of the current value
    T = max(cur_val * 0.02, 1e-6)
    decay = 0.99995

    iter_cnt = 0
    while time.time() < deadline:
        if rng.random() < 0.25:          # --- swap move ---
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            a[i], a[j] = a[j], a[i]
            new_val = _eval_fast(a)
            delta = new_val - cur_val
            accept = (delta < 0) or (rng.random() < np.exp(-delta / T))
            if accept:
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA] swap new best = {best_val:.6f} (iter {iter_cnt})")
            else:
                a[i], a[j] = a[j], a[i]   # revert
        else:                             # --- mass‑transfer move ---
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            # move a random fraction (up to 5 %) of a[i] to a[j]
            frac = rng.random() * 0.05
            delta_mass = a[i] * frac
            a[i] -= delta_mass
            a[j] += delta_mass
            new_val = _eval_fast(a)
            delta = new_val - cur_val
            accept = (delta < 0) or (rng.random() < np.exp(-delta / T))
            if accept:
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[SA] transfer new best = {best_val:.6f} (iter {iter_cnt})")
            else:
                # revert transfer
                a[i] += delta_mass
                a[j] -= delta_mass

        # cooling
        T *= decay
        if T < 1e-12:
            T = max(cur_val * 0.02, 1e-12)

        iter_cnt += 1
    return best_seq, best_val


# ----------------------------------------------------------------------
# Final cheap hill‑climbing (mass transfers + swaps)
# ----------------------------------------------------------------------
def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> (np.ndarray, float):
    """Greedy local search: occasional swaps and small mass transfers."""
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = a.shape[0]

    while time.time() < deadline:
        if rng.random() < 0.2:          # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                           # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] == 0.0:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        val = _eval_fast(new_a)
        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")

    return cur_a, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin
    print(f"[init] time budget = {budget_s}s, deadline @ {deadline:.2f}")

    # ------------------------------------------------------------------
    # Choose problem size (larger size gives more degrees of freedom)
    # ------------------------------------------------------------------
    # Use 8000 if we have enough total budget; otherwise fall back to 5000.
    n = 8000 if budget_s >= 900 else 5000
    global _GLOBAL_N
    _GLOBAL_N = n
    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Initialise from a known construction if possible
    # ------------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            # linear interpolation to requested size
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, old)
            print("[init] interpolated previous construction")
        else:
            a0 = old.copy()
            print("[init] loaded previous construction")
    except Exception:
        a0 = np.full(n, 1.0 / n)      # uniform start
        print("[init] using uniform start")

    a0 = np.maximum(a0, 0.0)          # enforce non‑negativity
    a0 = a0 / a0.sum()                # ℓ₁‑normalise to 1
    best_a = a0.copy()
    best_val = _eval_fast(best_a)
    print(f"[init] initial evaluation (fast) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # ε‑schedule for the smooth surrogate (log‑space)
    # ------------------------------------------------------------------
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-12), num=20)

    # ------------------------------------------------------------------
    # Multi‑restart L‑BFGS phase
    # ------------------------------------------------------------------
    max_restarts = 12
    restart = 0
    while time.time() < deadline - 120 and restart < max_restarts:
        restart += 1
        print(f"[restart {restart}]")
        # start from the current best, add moderate random noise
        w = np.log(best_a + 1e-12) + np.random.randn(n) * 0.5

        for eps in eps_schedule:
            if time.time() >= deadline - 100:
                break
            global _GLOBAL_EPS
            _GLOBAL_EPS = eps

            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 400,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # evaluate the real objective (fast version)
            w_shift = w - np.max(w)
            a = np.exp(w_shift)
            a /= a.sum()
            cur_val = _eval_fast(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[eps {eps:.1e}] new best = {best_val:.6f}")

        # --------------------------------------------------------------
        # Anti‑symmetric re‑ordering of the best sequence from this restart
        # --------------------------------------------------------------
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = _eval_fast(a_ord)
        if val_ord < best_val:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"[reorder] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing on the permutation of the best sequence
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        print("[SA] starting simulated annealing")
        best_a, best_val = _simulated_annealing(best_a, best_val,
                                                deadline - 20)

    # ------------------------------------------------------------------
    # Final cheap hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        print("[refine] final hill‑climbing phase")
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline)

    # ------------------------------------------------------------------
    # Report final value using the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"[final] best evaluation reported by evaluate_sequence = {final_score:.6f}")
    except Exception as exc:      # pragma: no cover
        print("[final] evaluate_sequence raised an exception:", exc)

    # Return a plain Python list as required by the framework
    return best_a.tolist()
