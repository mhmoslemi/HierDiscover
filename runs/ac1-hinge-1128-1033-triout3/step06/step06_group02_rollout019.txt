def _full_convolution(x):
    n = x.shape[0]
    out_len = 2*n - 1
    size = 1 << (out_len - 1).bit_length()
    fx = np.fft.rfft(x, size)

    return conv

"""
Propose a step function that (approximately) minimizes
``evaluate_sequence``.

The search consists of three stages:

1. **Warm‑start** – use a previously discovered construction if it
   exists; otherwise fall back to a uniform distribution.

2. **Smoothed L‑BFGS refinement** – optimize a smooth surrogate of the
   objective (log‑sum‑exp approximation of the max‑convolution) with a
   soft‑max parametrisation.  A schedule of decreasing smoothing
   parameters together with several random restarts allows the solver
   to explore many basins.  After each run the sequence is projected back
   onto the probability simplex and the true objective is evaluated.
   The best sequence so far is retained.

3. **Mirror‑descent on the true max‑convolution** – perform a
   sub‑gradient (exponential mirror‑descent) update that directly reduces
   the current peak of the self‑convolution.  The step size decays as
   1/√t.  Every few hundred iterations a cheap hill‑climbing phase
   (random swaps / small mass transfers) and an anti‑symmetric re‑order
   are applied to escape shallow local minima.

Finally a short stochastic hill‑climbing refinement is run until the
time budget expires.  The best sequence encountered is returned as a
plain ``list[float]`` compatible with ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import numpy as np
from scipy.special import logsumexp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global variables used by the smooth surrogate
# ----------------------------------------------------------------------
_SMOOTH_EPS = None   # smoothing parameter for the current L‑BFGS run
_SMOOTH_N = None     # length of the current sequence


# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (used for FFT padding)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """Linear convolution of a real vector with itself (length 2·len‑1)."""
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0, out=conv)
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Anti‑symmetric reorder: pair a large value on the left with a small
    value on the right, a second large with a second small, etc.
    This heuristic tends to flatten the self‑convolution.
    """
    n = a.shape[0]
    sorted_vals = np.sort(a)               # ascending
    new_a = np.empty_like(a)
    half = n // 2
    for i in range(half):
        new_a[i] = sorted_vals[n - 1 - i]          # large → left side
        new_a[n - 1 - i] = sorted_vals[i]          # small → right side
    if n % 2:
        new_a[half] = sorted_vals[half]
    return new_a


def _refine_hill_climb(a: np.ndarray,
                       best_val: float,
                       deadline: float,
                       n: int):
    """
    Very cheap stochastic hill‑climbing: random swaps (20 %) and
    mass transfers (80 %).  Only strict improvements are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # re‑project onto the simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        try:
            val = evaluate_sequence(new_a.tolist())
        except Exception:
            continue
        if val < cur_val - 1e-12:          # strict improvement
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.9f}")
    return cur_a, cur_val


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed objective L_ε and its gradient w.r.t. the soft‑max parameters w.

    a_i = exp(w_i) / Σ_j exp(w_j)        (ensures a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    # full self‑convolution
    conv = _full_convolution(a)           # length 2·n‑1
    max_conv = conv.max()

    # smooth max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()           # probability distribution over k

    # convolution a * reversed(p) (efficient via FFT)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # middle n entries correspond to ∂L/∂a (see derivation)
    seg = conv_ap[n - 1: 2 * n - 1]
    grad_a = 4.0 * n * seg[::-1]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses the globals)."""
    return _smooth_obj_and_grad(w, _SMOOTH_EPS, _SMOOTH_N)

    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem size – a moderate power‑of‑two eases FFT work
    # --------------------------------------------------------------
    n = 4096                     # ≈ 4 k entries, fits easily in memory

    # --------------------------------------------------------------
    # Initialise from a previously discovered construction if possible
    # --------------------------------------------------------------
    try:
        old = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if old.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x, old_x, old)
        else:
            a = old.copy()
    except Exception:
        a = np.full(n, 1.0 / n)   # uniform start

    # enforce feasibility (non‑negative & normalise to sum = 1)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    # --------------------------------------------------------------
    # Evaluate the initial candidate
    # --------------------------------------------------------------
    best_a = a.copy()
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] n={n}, evaluation = {best_val:.9f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on the smooth surrogate
    # ------------------------------------------------------------------
    # A logarithmic schedule for the smoothing parameter ε.
    eps_schedule = np.logspace(np.log10(5e-3), np.log10(1e-13), num=13)

    # soft‑max parameters (log‑space) for the current iterate
    w = np.log(best_a + 1e-12)

    # reserve roughly three quarters of the total time for the L‑BFGS phase
    lbf_deadline = deadline - 250.0
    restart = 0
    while time.time() < lbf_deadline and restart < 12:
        restart += 1
        print(f"[L‑BFGS restart {restart}]")
        # every few restarts randomise the starting point to avoid being trapped
        if restart % 4 == 0:
            w = np.random.randn(n) * 0.5

        for eps in eps_schedule:
            if time.time() >= lbf_deadline:
                break
            global _SMOOTH_EPS, _SMOOTH_N
            _SMOOTH_EPS = eps
            _SMOOTH_N = n
            # a fairly generous iteration budget per epsilon
            res = minimize(_smooth_obj_and_grad_wrapper,
                           w,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 500,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            w = res.x

            # convert back to probabilities
            w_shift = w - np.max(w)
            a_candidate = np.exp(w_shift)
            a_candidate_sum = a_candidate.sum()
            if a_candidate_sum < 1e-12:
                a_candidate = np.full(n, 1.0 / n)
            else:
                a_candidate /= a_candidate_sum

            cur_val = evaluate_sequence(a_candidate.tolist())
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a_candidate.copy()
                print(f"  [eps {eps:.1e}] new best = {best_val:.9f}")

        # occasionally try the anti‑symmetric reorder
        a_ord = _anti_symmetric_reorder(best_a)
        val_ord = evaluate_sequence(a_ord.tolist())
        if val_ord < best_val - 1e-12:
            best_val = val_ord
            best_a = a_ord.copy()
            print(f"  [reorder] improved to {best_val:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Mirror‑descent on the true max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    it = 0
    alpha0 = 0.18                     # initial step size (mirror‑descent)

    while time.time() < deadline - 60.0:
        conv = _full_convolution(a)
        max_k = int(conv.argmax())
        max_val = conv[max_k]

        # sub‑gradient for the current maximal lag
        g = np.zeros_like(a)
        i_start = max(0, max_k - (n - 1))
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = max_k - idx
        g[idx] = a[jdx]                     # g_i = a_{k*‑i}

        # step‑size schedule – 1/√t decay
        alpha = alpha0 / np.sqrt(it + 1)

        # exponential mirror‑descent update (preserves non‑negativity)
        a = a * np.exp(-alpha * g)

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:                       # under‑flow protection
            a = np.full(n, 1.0 / n)
            s = a.sum()
        a /= s

        if it % 100 == 0:
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                print(f"[mirror {it}] new best = {best_val:.9f} (max conv {max_val:.9f})")

            # every few hundred iterations run a short hill‑climb
            if it % 500 == 0:
                a_hc, val_hc = _refine_hill_climb(a.copy(),
                                                  cur_val,
                                                  deadline - 30.0,
                                                  n)
                if val_hc < best_val - 1e-12:
                    best_val = val_hc
                    best_a = a_hc.copy()
                    a = a_hc.copy()
                    print(f"[mirror {it}] hill‑climb improved to {best_val:.9f}")

            # occasional anti‑symmetric re‑ordering
            if it % 1000 == 0:
                a_ord = _anti_symmetric_reorder(a)
                val_ord = evaluate_sequence(a_ord.tolist())
                if val_ord < best_val - 1e-12:
                    best_val = val_ord
                    best_a = a_ord.copy()
                    a = a_ord.copy()
                    print(f"[mirror {it}] reorder improved to {best_val:.9f}")

        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – final stochastic hill‑climbing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_val = _refine_hill_climb(best_a, best_val, deadline, n)

    # ------------------------------------------------------------------
    # Final check & return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] evaluation reported by evaluate_sequence = {final_score:.9f}")
    return best_a.tolist()
