"""
Propose a step function (non‑negative heights) that minimizes the
objective defined by ``evaluate_sequence``.

The algorithm works in three phases:

1. **Smooth surrogate optimisation** –  minimise a smoothed version of
   the objective using a soft‑max approximation of the max‑convolution
   and SciPy's L‑BFGS.  Several restarts and a decreasing temperature
   schedule are used to reach a good basin.

2. **Adaptive multiplicative mirror‑descent** –  starting from the best
   point obtained above, we repeatedly apply an exponential update
   a_i ← a_i·exp(−α·g_i) where g_i aggregates the contributions from
   *all* convolution lags that are (almost) maximal (instead of a
   single lag).  The step size α is adapted on‑the‑fly: it is increased
   when a move improves the objective and decreased otherwise.  The
   iterate is re‑projected onto the probability simplex after each
   step.  Periodically we apply the “anti‑symmetric” reordering
   (descending sort) which is known to flatten the self‑convolution.

3. **Final hill‑climbing refinement** –  a cheap stochastic local search
   (random swaps / mass transfers) runs until the time budget expires.

All convolutions are performed with FFT (O(n log n)) and the objective
value is computed directly as `2 * n * max(conv)` because the sum of
the coefficients is kept equal to one.  The routine respects the 1000 s
budget and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables for the smooth surrogate (required by scipy.minimize)
# ----------------------------------------------------------------------
_GLOBAL_EPS = None  # smoothing temperature
_GLOBAL_N = None    # length of the sequence


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a real vector with itself (length 2·len‑1)
    using FFT.  The result is clipped to be non‑negative (tiny rounding
    errors may produce negative values).
    """
    n = x.shape[0]
    out_len = 2 * n - 1
    size = _next_pow2(out_len)
    fx = np.fft.rfft(x, size)
    conv = np.fft.irfft(fx * fx, size)[:out_len]
    np.maximum(conv, 0, out=conv)  # remove possible round‑off negatives
    return conv


def _anti_symmetric_reorder(a: np.ndarray) -> np.ndarray:
    """
    Rearrange the vector into descending order (largest values on the
    left, smallest on the right).  This simple heuristic often reduces
    the peak of the self‑convolution.
    """
    return np.sort(a)[::-1]


def _smooth_obj_and_grad(w: np.ndarray, eps: float, n: int):
    """
    Smoothed surrogate of the objective and its gradient w.r.t.
    the soft‑max parameters ``w``.

    a_i = exp(w_i) / Σ_j exp(w_j)   (guarantees a_i ≥ 0, Σ a_i = 1)

    L_ε = 2·n·max(conv) + 2·n·ε·log Σ_k exp((conv_k‑max)/ε)
    """
    # soft‑max (numerically stable)
    w_shift = w - np.max(w)
    a = np.exp(w_shift)
    a /= a.sum()

    conv = _full_convolution(a)          # length 2·n‑1
    max_conv = conv.max()

    # smoothed max via log‑sum‑exp
    shifted = (conv - max_conv) / eps
    log_sum = logsumexp(shifted)
    L = 2.0 * n * max_conv + 2.0 * n * eps * log_sum

    # gradient w.r.t. a
    weights = np.exp(shifted)
    p = weights / weights.sum()          # soft distribution over k

    # derivative of max(conv) part:
    #   ∂/∂a_i ( Σ_k p_k * conv_k ) = 2· (a * p_rev)  (see derivation)
    p_rev = p[::-1]
    size = _next_pow2(n + p_rev.shape[0] - 1)
    fa = np.fft.rfft(a, size)
    fp = np.fft.rfft(p_rev, size)
    conv_ap = np.fft.irfft(fa * fp, size)[: n + p_rev.shape[0] - 1]

    # Extract the segment that matches the gradient formula
    seg = conv_ap[n - 1: 2 * n - 1]      # length n
    grad_a = 4.0 * n * seg[::-1]        # ∂L/∂a_i

    # Chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_w = a * (grad_a - dot)
    return L, grad_w


def _smooth_obj_and_grad_wrapper(w: np.ndarray):
    """Wrapper required by ``scipy.minimize`` (uses the global eps/n)."""
    return _smooth_obj_and_grad(w, _GLOBAL_EPS, _GLOBAL_N)


def _refine_hill_climb(a: np.ndarray, best_val: float,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb: random swaps (20 %) and small mass
    transfers (80 %).  Only strict improvements are kept.
    """
    cur_a = a.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    n = len(a)

    while time.time() < deadline:
        if rng.random() < 0.2:                     # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            new_a = cur_a.copy()
            new_a[i], new_a[j] = new_a[j], new_a[i]
        else:                                      # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_a[i] <= 1e-15:
                continue
            delta = cur_a[i] * 0.05 * rng.random()
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # project onto simplex
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-12:
            continue
        new_a /= s

        # evaluate (fast, no call to evaluate_sequence)
        conv = _full_convolution(new_a)
        val = 2.0 * len(new_a) * conv.max()

        if val < cur_val - 1e-12:
            cur_a = new_a
            cur_val = val
            print(f"[refine] new best = {cur_val:.6f}")
    return cur_a, cur_val
