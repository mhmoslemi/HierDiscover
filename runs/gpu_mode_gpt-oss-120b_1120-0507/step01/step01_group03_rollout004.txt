        right = right * mask_h

    # 4. Gating
    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # 5. TriMul heavy operation via Triton kernel:
    # Permute to B, D, N, N
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    out_t = torch.empty_like(left_t, dtype=torch.float16)

    B, D, N, _ = left_t.shape  # because shape is (B, D, N, N)

    stride_l_batch, stride_l_d, stride_l_i, stride_l_k = left_t.stride()
    stride_r_batch, stride_r_d, stride_r_j, stride_r_k = right_t.stride()
    stride_o_batch, stride_o_d, stride_o_i, stride_o_j = out_t.stride()

    # Determine tile sizes: use compile-time constants but may adjust if N small
    BLOCK_M = 128 if N >= 256 else 64
    BLOCK_N = 128 if N >= 256 else 64
    BLOCK_K = 64 if N >= 256 else 32

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * D )

    trmul_outgoing_kernel[grid](
        left_t, right_t, out_t,
        B, N, D,
        stride_l_batch, stride_l_d, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_d, stride_r_j, stride_r_k,
        stride_o_batch, stride_o_d, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout
    out = out_t.permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,D]

    # 6. Post‑processing
    # to_out_norm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    # gating
    out = out * out_gate  # out_gate is fp16, broadcast; out is fp32 after layernorm, convert out_gate to fp32
    # final linear
    out = F.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) – fully‑fused forward pass.

The kernel implements the only O(N³) part:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix‑multiply (N×N)·(N×N)ᵀ for each hidden channel *d*.
All tensors are cast to FP16 before the heavy compute and the
accumulator is kept in FP32 so the kernel can use Tensor‑Core WMMA on H100.
The surrounding LayerNorms, linear projections, gates and the final projection
are performed with cheap PyTorch ops.  The implementation works for any
batch‑size (≤2) and sequence length up to 1024, and supports optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trmul_outgoing_kernel(
    left_ptr,   # *Pointer* to left tensor   (B, D, N, N)   – FP16
    right_ptr,  # *Pointer* to right tensor  (B, D, N, N)   – FP16
    out_ptr,    # *Pointer* to output tensor (B, D, N, N)   – FP16
    B, N, D,    # static problem sizes
    stride_l_batch, stride_l_d, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_d, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_d, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,  # tile size in i dimension
    BLOCK_N: tl.constexpr,  # tile size in j dimension
    BLOCK_K: tl.constexpr,  # tile size in k dimension
):
    """
    Batched “outgoing” TriMul:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    The batch&channel dimension is flattened (B*D independent GEMMs).
    The kernel works in FP16 (Tensor‑Core) with FP32 accumulation.
    """
    pid_i = tl.program_id(0)          # block id along i
    pid_j = tl.program_id(1)          # block id along j
    pid_bc = tl.program_id(2)         # flattened batch*channel id

    # tile start positions
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    # decode flattened batch*channel index
    b = pid_bc // D
    d = pid_bc % D

    # base pointers for the current (b,d) matrix pair
    left_base  = left_ptr  + b * stride_l_batch + d * stride_l_d
    right_base = right_ptr + b * stride_r_batch + d * stride_r_d
    out_base   = out_ptr   + b * stride_o_batch + d * stride_o_d

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the shared k axis
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile  (i,k)  – shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i[:, None] * stride_l_i + k[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0,
                            dtype=tl.float16).to(tl.float32)

        # right tile (k,j) – shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + k[:, None] * stride_r_k + j[None, :] * stride_r_j
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0,
                             dtype=tl.float16).to(tl.float32)

        # matrix‑multiply the tile (Tensor‑Core will be used)
        acc += tl.dot(left_tile, right_tile)   # (BLOCK_M, BLOCK_N)

    # store the result back in FP16
    out_ptrs = out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:         torch.Tensor [B, N, N] or None
            - weights:      dict of torch.Tensors (float32)
            - config:       dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Cast to FP16 for all heavy linear work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – FP16
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,hidden_dim]
    right = F.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension) – FP16
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4. Gating – FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy O(N³) update – batched GEMM via Triton kernel
    #    Data layout: (B, D, N, N) – contiguous
    # ------------------------------------------------------------------
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, D, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, D, N, N]
    out_t = torch.empty_like(left_t, dtype=torch.float16)

    B, D, N, _ = left_t.shape

    # strides in element units (torch returns element‑wise strides)
    stride_l_batch, stride_l_d, stride_l_i, stride_l_k = left_t.stride()
    stride_r_batch, stride_r_d, stride_r_j, stride_r_k = right_t.stride()
    stride_o_batch, stride_o_d, stride_o_i, stride_o_j = out_t.stride()

    # -----------------------------------------------------------------------------
    # Tile sizes – tuned for H100 Tensor‑Core (fits into shared memory)
    # Larger tiles reduce launch overhead for big N.
    # -----------------------------------------------------------------------------
    BLOCK_M = 128 if N >= 256 else 64
    BLOCK_N = 128 if N >= 256 else 64
    BLOCK_K = 64  if N >= 256 else 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * D,                          # flattened (batch, channel)
    )

    trmul_outgoing_kernel[grid](
        left_t,
        right_t,
        out_t,
        B,
        N,
        D,
        stride_l_batch,
        stride_l_d,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_d,
        stride_r_j,
        stride_r_k,
        stride_o_batch,
        stride_o_d,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # enough warps for 128×128 tiles
        num_stages=2,
    )

    # reshape back to original layout: [B, N, N, hidden_dim] (still FP16)
    out = out_t.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Post‑processing (float32)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    # Layer‑norm over the hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply the output gate (convert gate to float32 for the multiplication)
    out = out * out_gate.to(out.dtype)

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
