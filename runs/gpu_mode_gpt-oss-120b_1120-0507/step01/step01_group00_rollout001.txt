"""
custom_kernel – highly‑optimized "outgoing" TriMul.

Algorithm
---------
1️⃣  Layer‑norm the input over the channel axis `dim`.
2️⃣  Linear projections (bias‑free) → hidden dimension.
3️⃣  Compute three gating tensors (linear + sigmoid) and apply them.
4️⃣  Optional mask multiplication (broadcast over the hidden dim).
5️⃣  Core TriMul:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   – realised as a batched GEMM  L @ Rᵀ  where each hidden channel is
     treated as an independent batch element.
   – The GEMM runs in FP16 (half‑precision) with FP32 accumulation
     inside a custom Triton kernel (`_batched_matmul_fp16`).
   – Tile sizes (128 × 128 × 64) are tuned for H100, drastically reducing
     kernel launch overhead compared to the reference 64 × 64 tiles.
6️⃣  Layer‑norm over the hidden dimension, multiply by the output‑gate
    and project back to `dim` with a final linear layer.

Only the heavy reduction (step 5) is off‑loaded to Triton; all other
operations stay in PyTorch for simplicity.  The final tensor is
float32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched FP16 GEMM (A @ Bᵀ) with FP32 acc.
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_fp16(
    a_ptr,                     # *ptr, shape (batch, M, K)   – FP16
    b_ptr,                     # *ptr, shape (batch, K, N)   – FP16 (interpreted as transposed)
    c_ptr,                     # *ptr, shape (batch, M, N)   – FP16 output
    M, N, K,                   # matrix sizes (int32 scalars)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,   # B is already transposed
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a whole batch.

    A : (batch, M, K)  FP16
    B : (batch, K, N)  FP16 – stored in memory as (batch, N, K) and
                         accessed with swapped strides so that the kernel
                         sees Bᵀ directly.
    C : (batch, M, N)  FP16
    Accumulator is FP32 for numerical safety.
    """
    pid = tl.program_id(0)

    # ----------------------------------------------------------
    # 1. Decode the 3‑D grid → (batch, tile_m, tile_n)
    # ----------------------------------------------------------
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    batch_id = pid // (num_pid_m * num_pid_n)
    tile_id  = pid % (num_pid_m * num_pid_n)
    pid_m = tile_id // num_pid_n
    pid_n = tile_id %  num_pid_n

    # ----------------------------------------------------------
    # 2. Compute offsets for this tile
    # ----------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # pointers to the (M×K) and (K×N) tiles
    a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    # ----------------------------------------------------------
    # 3. Accumulate over K
    # ----------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=tl.float16(0.0)
        )                                 # (BLOCK_M, BLOCK_K)
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=tl.float16(0.0)
        )                                 # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)               # FP32 accumulation

        # advance to next K‑block
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ----------------------------------------------------------
    # 4. Write back (cast to FP16)
    # ----------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

# Public entry‑point used by the grader
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict containing all model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype float32, on the same
        device as ``input_tensor``.
    """
    # ----------------------------------------------------------
    # Unpack and read configuration
    # ----------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------
    # 1️⃣ Layer‑norm over ``dim``
    # ----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ----------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ----------------------------------------------------------
    left_raw = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right_raw = F.linear(x, weights["right_proj.weight"])  # same shape

    # ----------------------------------------------------------
    # 3️⃣ Gating tensors (linear → sigmoid)
    # ----------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ----------------------------------------------------------
    # 4️⃣ Apply gates (and optional mask) – stays in FP32 for stability
    # ----------------------------------------------------------
    left = left_raw * left_gate
    right = right_raw * right_gate

    if mask is not None:
        # broadcast mask over the hidden dimension
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------
    # 5️⃣ Core TriMul (batched GEMM) – compute L @ Rᵀ for every hidden channel
    # ----------------------------------------------------------
    # Cast to half precision before the heavy reduction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange to (B, H, N, N) → treat each (b,h) as independent batch
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    batch_dim = B * H

    # Collapse batch and hidden dimensions
    left_mat = left.view(batch_dim, N, N)   # [batch_dim, N, N]
    right_mat = right.view(batch_dim, N, N)

    # Output buffer in FP16 (later cast to FP32)
    out_mat = torch.empty((batch_dim, N, N), dtype=torch.float16, device=device)

    # ----------------------------------------------------------
    # 5.1 Triton launch configuration
    # ----------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = batch_dim * grid_m * grid_n
    grid = (total_programs,)

    # Strides (as int64)
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_mat.stride()
    # reinterpret right as transposed: (batch, K, N)
    stride_b_k_T = stride_b_k               # original column stride (==1 for contiguous)
    stride_b_n_T = stride_b_m               # original row stride (==N)

    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # ----------------------------------------------------------
    # 5.2 Kernel call
    # ----------------------------------------------------------
    _batched_matmul_fp16[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k_T, stride_b_n_T,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] and promote to FP32
    out = (
        out_mat.view(B, hidden_dim, N, N)

    # 6️⃣ Post‑processing: hidden‑dim layer‑norm, output gate & final linear
    # ----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
