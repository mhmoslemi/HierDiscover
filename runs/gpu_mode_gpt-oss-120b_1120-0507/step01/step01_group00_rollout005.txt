import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # compute addresses of current tile
    a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_batch + offs_n[:, None] * stride_b_n + offs_k[None, :] * stride_b_k

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b, out_dtype=tl.float32)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write back
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul kernel (forward only).
    Implements the AlphaFold3 TriMul operation:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    where left/right are linear projections of the input with gating.
    The heavy kernel (batched GEMM) is realised in Triton.
    All other ops (LayerNorm, Linear, Sigmoid) are performed with PyTorch.
    Mixed‑precision (FP16) is used for the mass‑multiply part to reduce
    memory traffic while keeping the final output in FP32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm ----------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b, eps=eps)  # FP32

    # Cast to half for cheap ops
    x = x.to(torch.float16)

    # ---------- Linear projections ----------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim] FP16
    right = F.linear(x, right_proj_w)

    # ---------- Optional mask ----------
    if mask_tensor is not None:
        # mask broadcast to hidden dim
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ---------- Gates ----------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, hidden_dim] FP16
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ---------- Core TriMul (batched GEMM) ----------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # reshape to (batch_dim, N, N) where batch_dim = B * hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_dim = B * hidden_dim
    left_flat = left_perm.view(batch_dim, N, N)   # FP16
    right_flat = right_perm.view(batch_dim, N, N)

    out_flat = torch.empty_like(left_flat)  # FP16 output

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_M = (N + BLOCK_M - 1) // BLOCK_M
    grid_N = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch_dim, grid_M, grid_N)

    # strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_flat.stride()
    # For transposed access of B we swap the last two strides
    stride_b_n = stride_b_m          # rows of original B
    stride_b_k = stride_b_k          # cols of original B

    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ---------- Post‑processing ----------
    # Convert to FP32 for layer‑norm stability
    out_fp32 = out.to(torch.float32)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # Apply output gate (cast gate to FP32)
    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    # Final linear projection back to `dim`
    to_out_w = weights["to_out.weight"]
    out_fp32 = F.linear(out_fp32, to_out_w)   # [B, N, N, dim]

    return out_fp32

# -*- coding: utf-8 -*-
"""
Optimised Triton implementation of the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the 4‑D input over the channel axis (`dim`).
2. Cast to FP16 and compute four bias‑free linear projections:
      left  = x @ W_left    (→ hidden_dim)
      right = x @ W_right   (→ hidden_dim)
   as well as three gates (left, right, out) via linear + sigmoid.
3. (Optional) apply the pair‑wise mask (broadcast over hidden_dim).
4. Fuse the heavy reduction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   into a *batched* matrix multiply L @ Rᵀ for every hidden channel.
   The multiply is carried out in a custom Triton kernel that works on
   FP16 tensors and accumulates in FP32.
5. Reshape the result back to (B,N,N,hidden_dim), cast to FP32 and
   apply a second LayerNorm, the output‑gate and a final linear
   projection to the original `dim`.
All arithmetic except the tiny LayerNorm / Linear parts is performed
in half‑precision, which cuts memory traffic dramatically while the
final result is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B : [batch, M, K])
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,                     # *ptr, shape (batch, M, K)
    b_ptr,                     # *ptr, shape (batch, K, N) – we read it as transposed
    c_ptr,                     # *ptr, shape (batch, M, N)
    M, N, K,                   # matrix dimensions (int32)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,   # b is transposed on‑the‑fly
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ  where A∈[B,M,K], B∈[B,K,N]."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch pointer
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # pointer to current tile (M×K and K×N → read B as transposed)
    a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_batch + offs_n[:, None] * stride_b_n + offs_k[None, :] * stride_b_k

    # accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        )                              # (BLOCK_M, BLOCK_K)
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )                              # (BLOCK_K, BLOCK_N)

        # dot produces FP32 because of out_dtype
        acc += tl.dot(a, b, out_dtype=tl.float32)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask)

# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (FP32)
        - mask_tensor  : torch.Tensor or None, shape [B, N, N] (bool/float)
        - weights      : dict of torch.Tensor containing module parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (FP32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over channel axis (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 – the heavy arithmetic runs in half precision
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]  FP16
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Gates: linear + sigmoid
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))    # FP16
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core TriMul : batched GEMM L @ Rᵀ
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # B, seq_len, seq_len, hidden_dim
    # bring hidden_dim to the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_dim = B * hidden_dim
    left_flat = left_perm.view(batch_dim, N, N)   # (batch_dim, N, N)  FP16
    right_flat = right_perm.view(batch_dim, N, N)

    out_flat = torch.empty_like(left_flat)        # FP16 output buffer

    # block sizes – chosen to fit H100 registers and give good occupancy
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_M = (N + BLOCK_M - 1) // BLOCK_M
    grid_N = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch_dim, grid_M, grid_N)           # (batch, tile_M, tile_N)

    # strides in number of elements
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_flat.stride()
    # for transposed read of B we swap the last‑two strides
    stride_b_n = stride_b_m   # row stride of original right
    stride_b_k = stride_b_k   # column stride of original right

    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to (B, N, N, hidden_dim) and cast to FP32 for the remaining ops
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7️⃣ Post‑processing: layer‑norm, output gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    out = out * out_gate.to(out.dtype)   # apply output gate

    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)        # → [B, N, N, dim]  FP32

    return out
