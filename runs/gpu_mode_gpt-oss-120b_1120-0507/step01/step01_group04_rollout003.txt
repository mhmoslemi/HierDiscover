@triton.jit
def _mask_gate_fp16_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # decode flat index
    b = offs // (N * N * H)
    r1 = offs % (N * N * H)
    i = r1 // (N * H)
    r2 = r1 % (N * H)
    j = r2 // H
    h = r2 % H

    # compute memory offsets
    left_off = b * stride_l_b + i * stride_l_i + j * stride_l_j + h * stride_l_h
    right_off = b * stride_r_b + i * stride_r_i + j * stride_r_j + h * stride_r_h
    left_gate_off = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h * stride_lg_h
    right_gate_off = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + h * stride_rg_h
    mask_off = b * stride_m_b + i * stride_m_i + j * stride_m_j

    # load as float16
    l = tl.load(left_ptr + left_off, mask=mask, other=tl.float16(0.0))
    r = tl.load(right_ptr + right_off, mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + left_gate_off, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + right_gate_off, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(1.0))

    # fused multiplication
    out_l = l * lg * m
    out_r = r * rg * m

    tl.store(out_left_ptr + left_off, out_l, mask=mask)
    tl.store(out_right_ptr + right_off, out_r, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul operation.
    All heavy tensor‑wise work (linear projections, mask+gate fusion,
    and the N³ contraction) is performed in float16 to exploit
    H100 Tensor‑Core throughput; only the final tiny layers remain in
    float32.  A tiny Triton kernel fuses mask multiplication with the
    per‑channel gates.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------ #
    # Config & dimensions
    # ------------------------------------------------------------------ #
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------ #
    # Cast everything to float16 for the bulk of the computation
    # ------------------------------------------------------------------ #
    # Input (float16) and LayerNorm
    x = input_tensor.to(torch.float16)
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------ #
    # Linear projections (half‑precision)
    # ------------------------------------------------------------------ #
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x, left_proj_weight)   # (B,N,N,H)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------ #
    # Gating tensors (half‑precision)
    # ------------------------------------------------------------------ #
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------ #
    # Prepare mask (half‑precision, broadcasted on hidden dim)
    # ------------------------------------------------------------------ #
    if mask is None or config.get("nomask", False):
        mask_half = torch.ones((bs, seq_len, seq_len), dtype=torch.float16, device=device)
    else:
        mask_half = mask.to(dtype=torch.float16, device=device)

    # ------------------------------------------------------------------ #
    # Fuse mask & gates with a tiny Triton kernel (produces half‑precision)
    # ------------------------------------------------------------------ #
    left_out = torch.empty_like(left, dtype=torch.float16)
    right_out = torch.empty_like(right, dtype=torch.float16)

    # strides (element‑wise)
    stride_l = left.stride()
    stride_r = right.stride()
    stride_lg = left_gate.stride()
    stride_rg = right_gate.stride()
    stride_m = mask_half.stride()   # (N*N, N, 1)

    total = bs * seq_len * seq_len * hidden_dim
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

        left, right,
        left_gate, right_gate,
        mask_half,
        left_out, right_out,

"""
TriMul (outgoing) – fast Triton‑accelerated forward.

The expensive N³ contraction is performed as a batched GEMM in float16,
leveraging H100 Tensor‑Core throughput.  All linear projections,
LayerNorms and gates are also run in float16, while a tiny Triton kernel
fuses mask multiplication with the two per‑channel gates.  The final
output is cast back to float32 to match the reference implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fp16_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse mask & per‑channel gates; all work in float16."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # Decode flat index → (b,i,j,h)
    b = offs // (N * N * H)
    r1 = offs % (N * N * H)
    i = r1 // (N * H)
    r2 = r1 % (N * H)
    j = r2 // H
    h = r2 % H

    # Offsets for every tensor (element‑wise strides)
    left_off = b * stride_l_b + i * stride_l_i + j * stride_l_j + h * stride_l_h
    right_off = b * stride_r_b + i * stride_r_i + j * stride_r_j + h * stride_r_h
    left_gate_off = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h * stride_lg_h
    right_gate_off = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + h * stride_rg_h
    mask_off = b * stride_m_b + i * stride_m_i + j * stride_m_j

    # Load as float16
    l = tl.load(left_ptr + left_off, mask=mask, other=tl.float16(0.0))
    r = tl.load(right_ptr + right_off, mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + left_gate_off, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + right_gate_off, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(1.0))

    # Fuse mask · gate · value
    out_l = l * lg * m
    out_r = r * rg * m

    tl.store(out_left_ptr + left_off, out_l, mask=mask)
    tl.store(out_right_ptr + right_off, out_r, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward of AlphaFold3 "outgoing" TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------
    # shape / config
    # ------------------------------------------------------------------
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Cast everything to float16 for the heavy work
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float16)

    # LayerNorm on the input (float16 weights)
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(x, (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left = F.linear(x, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Per‑channel gates (half‑precision)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # Prepare mask (half‑precision, broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_half = torch.ones((bs, seq_len, seq_len),
                               dtype=torch.float16,
                               device=device)
    else:
        mask_half = mask.to(dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Fuse mask + gates (tiny Triton kernel)
    # ------------------------------------------------------------------
    left_out = torch.empty_like(left, dtype=torch.float16)
    right_out = torch.empty_like(right, dtype=torch.float16)

    stride_l = left.stride()
    stride_r = right.stride()
    stride_lg = left_gate.stride()
    stride_rg = right_gate.stride()
    stride_m = mask_half.stride()   # (N*N, N, 1)

    total = bs * seq_len * seq_len * hidden_dim
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    _mask_gate_fp16_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_half,
        left_out, right_out,
        bs, seq_len, hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_lg[0], stride_lg[1], stride_lg[2], stride_lg[3],
        stride_rg[0], stride_rg[1], stride_rg[2], stride_rg[3],
        stride_m[0], stride_m[1], stride_m[2],
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # N³ contraction: batched GEMM in half‑precision (Tensor Cores)
    # ------------------------------------------------------------------
    # left: (B,N,N,H) → (B*H, N, N) where axes are (i,k)
    left_mat = left_out.permute(0, 3, 1, 2).reshape(bs * hidden_dim,
                                                   seq_len,
                                                   seq_len)
    # right: (B,N,N,H) → (B*H, N, N) where axes are (k,j)
    right_mat = right_out.permute(0, 3, 2, 1).reshape(bs * hidden_dim,
                                                    seq_len,
                                                    seq_len)

    out_mat = torch.bmm(left_mat, right_mat)    # (B*H, N, N) half

    # reshape back to (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # LayerNorm over hidden dimension (half‑precision)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    # ------------------------------------------------------------------
    # Output gate and final projection (both half‑precision)
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) half

    # ------------------------------------------------------------------
    # Cast back to float32 for downstream use
    # ------------------------------------------------------------------
    return out.to(torch.float32)
