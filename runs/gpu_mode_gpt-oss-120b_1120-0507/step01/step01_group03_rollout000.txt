"""
TriMul (outgoing) – fast Triton implementation.

The heavy O(N³) part of the AlphaFold3 “outgoing” TriMul is
∑ₖ left[b,i,k,d] * right[b,j,k,d]   for every batch b, hidden channel d.
After permuting to (B, D, N, N) this is exactly a batched matrix‑multiply:
    out[b,d] = left[b,d] @ right[b,d].T
We fuse this multiplication in a Triton kernel that:
* works on FP16 for the two inputs (half the memory traffic),
* uses Tensor‑Core friendly tiles (64×64×64),
* accumulates in FP32 for numerical stability,
* handles arbitrary N (masking the tail tiles) and a flattened (B·D) batch axis.
All other operations (LayerNorm, Linear, gating) are cheap PyTorch calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trmul_outgoing_kernel(
    # pointers
    left_ptr,          # [B, D, N, N]  (FP16)
    right_ptr,         # [B, D, N, N]  (FP16)
    out_ptr,           # [B, D, N, N]  (FP32)
    # problem sizes
    B, N, D,
    # strides (in element units)
    stride_l_batch, stride_l_d, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_d, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_d, stride_o_i, stride_o_j,
    # tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr,  # i‑tile
    BLOCK_N: tl.constexpr,  # j‑tile
    BLOCK_K: tl.constexpr,  # k‑tile
):
    """
    out[b,d,i,j] = Σₖ left[b,d,i,k] * right[b,d,j,k]
    (i‑j tile of the matrix product left @ rightᵀ)
    """
    pid_i = tl.program_id(0)          # tile‑row index
    pid_j = tl.program_id(1)          # tile‑col index
    pid_bd = tl.program_id(2)         # flattened (batch, hidden) index

    # ---- decode positions --------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # (BLOCK_M,)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # (BLOCK_N,)

    mask_i = i < N
    mask_j = j < N

    b = pid_bd // D          # original batch index
    d = pid_bd % D           # hidden‑channel index

    # base pointers for this (b,d) slice
    left_base  = left_ptr  + b * stride_l_batch + d * stride_l_d
    right_base = right_ptr + b * stride_r_batch + d * stride_r_d
    out_base   = out_ptr   + b * stride_o_batch + d * stride_o_d

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over k -------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)      # (BLOCK_K,)
        mask_k = k_idx < N

        # left tile: (i, k)
        left_ptrs = left_base + i[:, None] * stride_l_i + k_idx[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=tl.float16(0.0))
        # right tile (transposed): (k, j)
        right_ptrs = right_base + k_idx[:, None] * stride_r_k + j[None, :] * stride_r_j
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=tl.float16(0.0))

        # half‑precision dot → cast to FP32 and accumulate
        acc += tl.dot(left_tile, right_tile).to(tl.float32)

    # ---- store result ------------------------------------------------------
    out_ptrs = out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:        torch.Tensor [B, N, N] or None
            - weights:     dict of torch.Tensors containing all model parameters
            - config:      dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Layer‑norm over the last dimension
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])    # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gated projections
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Heavy O(N³) part – Triton GEMM on half‑precision tensors
    # --------------------------------------------------------------
    # Cast to fp16 before launching the kernel (memory‑bandwidth saving)
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # Rearrange to (B, D, N, N) for easy striding
    left_t = left_h.permute(0, 3, 1, 2).contiguous()   # [B, D, N, N]
    right_t = right_h.permute(0, 3, 1, 2).contiguous() # [B, D, N, N]

    B, D, N, _ = left_t.shape

    # Output tensor stays in FP32 (to match later LayerNorm)
    out_t = torch.empty((B, D, N, N), dtype=torch.float32, device=left_t.device)

    # Strides in element counts (Torch returns element‑wise strides)
    stride_l_batch, stride_l_d, stride_l_i, stride_l_k = left_t.stride()
    stride_r_batch, stride_r_d, stride_r_j, stride_r_k = right_t.stride()
    stride_o_batch, stride_o_d, stride_o_i, stride_o_j = out_t.stride()

    # Tile sizes – tuned for H100 (fit in registers / shared memory)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * D,                          # flattened batch × hidden
    )

    trmul_outgoing_kernel[grid](
        left_t,
        right_t,
        out_t,
        B,
        N,
        D,
        stride_l_batch,
        stride_l_d,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_d,
        stride_r_j,
        stride_r_k,
        stride_o_batch,
        stride_o_d,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,           # Tensor‑Core friendly launch config
    )

    # Back to the original layout: [B, N, N, hidden_dim]
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Post‑processing: LayerNorm → out‑gate → final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```

This implementation:
* Keeps the high‑level logic in PyTorch (LayerNorm, Linear, gating) for readability.
* Moves the cubic‑complexity multiplication to a fused Triton kernel that operates on half‑precision inputs and accumulates in FP32.
* Uses 64×64×64 tiles and 8 warps per CTA, which matches the H100 Tensor‑Core granularity and dramatically reduces the runtime compared with a naïve implementation.
