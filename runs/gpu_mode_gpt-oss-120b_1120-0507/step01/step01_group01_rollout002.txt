"""
TriMul (outgoing) forward pass – highly‑optimized Triton implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (last dimension = dim).
2. Linear projections → left, right (shape [B,N,N,H]) and optional
   element‑wise masking.
3. Gating (sigmoid of three more linear projections) → left_gate,
   right_gate, out_gate.  Apply left_gate/right_gate to left/right.
4. Heavy O(N³) contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication for each hidden channel:
        left  : (N × N)      (row=i, col=k)
        right : (N × N)ᵀ    (row=k, col=j)
   We fuse all (B*H) independent GEMMs in a single Triton kernel.
   The kernel works on fp16 inputs, accumulates in fp32 and writes fp32.
   Double‑buffered loads and 128×128×64 tiles exploit H100 tensor‑cores.
5. Layer‑norm over the hidden dimension, element‑wise out_gate, and final
   linear projection back to `dim`.

Only the N³‑contraction is written in Triton; all other ops use
torch.nn.functional for simplicity and negligible cost.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM: C = A @ Bᵀ   (A: [B, M, K], B: [B, N, K] -> C: [B, M, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                         # matrix sizes
    stride_a_batch, stride_a_m, stride_a_k,   # strides of A
    stride_b_batch, stride_b_n, stride_b_k,   # strides of B (already transposed)
    stride_c_batch, stride_c_m, stride_c_n,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Grid dimensions: (ceil(M/BLOCK_M), ceil(N/BLOCK_N), batch)
    Each program computes a TILE_M × TILE_N output block.
    Double‑buffered loading of A and B tiles hides memory latency.
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (B*H)

    # --------------------------------------------------------------
    # Offsets for this tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds (used for load/store)
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # ------------------------------------------------------------------
    # Pointers for A and B (B already transposed, i.e. shape [B,N,K])
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + pid_b * stride_a_batch \
                   + offs_m[:, None] * stride_a_m \
                   + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_b * stride_b_batch \
                   + offs_n[None, :] * stride_b_n \
                   + offs_k[:, None] * stride_b_k  # B is stored as (N,K) after transpose

    # ------------------------------------------------------------------
    # Accumulator (fp32) – ensure safety for large reductions
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Prefetch the first tiles (double buffering)
    # ------------------------------------------------------------------
    a_tile = tl.load(a_ptrs,
                     mask=mask_m[:, None] & mask_k[None, :],
                     other=0.0).to(tl.float16)
    b_tile = tl.load(b_ptrs,
                     mask=mask_k[:, None] & mask_n[None, :],
                     other=0.0).to(tl.float16)

    a_ptrs += BLOCK_K * stride_a_k
    b_ptrs += BLOCK_K * stride_b_k

    num_k = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------
    # Main K‑loop – compute dot products while pre‑fetching next tiles
    # ------------------------------------------------------------------
    for _ in range(1, num_k):
        # Load the next tiles (masked)
        a_next = tl.load(a_ptrs,
                         mask=mask_m[:, None] & mask_k[None, :],
                         other=0.0).to(tl.float16)
        b_next = tl.load(b_ptrs,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0).to(tl.float16)

        # Accumulate current tile
        acc += tl.dot(a_tile, b_tile)   # (M,K)·(K,N) → (M,N)

        # Advance pointers for next iteration
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

        # Shift tiles
        a_tile = a_next
        b_tile = b_next

    # Final accumulation for the last prefetched tile
    acc += tl.dot(a_tile, b_tile)

    # ------------------------------------------------------------------
    # Write back the result – apply output masks
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + pid_b * stride_c_batch \
                   + offs_m[:, None] * stride_c_m \
                   + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(A: torch.Tensor,
                    B: torch.Tensor,
                    *,
                    BLOCK_M=128,
                    BLOCK_N=128,
                    BLOCK_K=64) -> torch.Tensor:
    """
    A : (batch, M, K)  – fp16 or fp32 (will be cast to fp16)
    B : (batch, N, K)  – fp16 or fp32 (already transposed to N×K)
    Returns C : (batch, M, N)  – fp32
    """
    assert A.shape[0] == B.shape[0]
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K == K2

    # Ensure fp16 for tensor‑core execution
    A = A.to(torch.float16)
    B = B.to(torch.float16)

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    grid = (
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )
    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                 # input / output channel dimension
    hidden_dim = config["hidden_dim"]   # projection dimension (H)
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                     # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left / right
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]    # [H, dim]
    right_proj_weight = weights["right_proj.weight"]  # [H, dim]

    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections of the *normalized* x)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]   # [H, dim]
    right_gate_weight = weights["right_gate.weight"]  # [H, dim]
    out_gate_weight   = weights["out_gate.weight"]    # [H, dim]

    left_gate  = F.linear(x, left_gate_weight).sigmoid()   # [B, N, N, H]
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Prepare tensors for Triton batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    → treat each hidden channel as an independent batch.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B == batch size, N == seq_len

    # Rearrange to (B, H, N, N) where the last two dims are (i,k) for left
    # and (k,j) for right.  Then collapse batch & hidden into a single
    # batch dimension for the Triton kernel.
    left_h = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_h = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k,j)

    # Flatten batch*hidden
    left_h = left_h.view(B * hidden_dim, N, N)   # [B*H, N, N]
    right_h = right_h.view(B * hidden_dim, N, N) # [B*H, N, N]

    # ------------------------------------------------------------------
    # 6️⃣ Batched GEMM via Triton (result in FP32)
    # ------------------------------------------------------------------
    # Choose tile sizes dynamically: larger tiles for larger seq_len.
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    out_h = _batched_matmul(left_h, right_h,
                            BLOCK_M=BLOCK_M,
                            BLOCK_N=BLOCK_N,
                            BLOCK_K=BLOCK_K)           # [B*H, N, N] fp32

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to original layout: [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Hidden‑dim LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)                     # [B, N, N, H]

    # Apply element‑wise out_gate (same shape as out)
    out = out * out_gate

    # Final projection back to `dim`
    to_out_weight = weights["to_out.weight"]          # [dim, H]
    out = F.linear(out, to_out_weight)               # [B, N, N, dim]

    # Ensure output is float32 as required
    out = out.to(torch.float32)

    return out
