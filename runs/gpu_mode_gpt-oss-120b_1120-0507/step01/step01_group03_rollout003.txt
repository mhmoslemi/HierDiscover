import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_batch, stride_l_d, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_d, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_d, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    # decode batch*channel index
    b = pid_batch // D
    d = pid_batch % D

    # base pointers to the start of (b,d) slice
    left_base = left_ptr + b * stride_l_batch + d * stride_l_d
    right_base = right_ptr + b * stride_r_batch + d * stride_r_d
    out_base = out_ptr + b * stride_o_batch + d * stride_o_d

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left[i,k] tile
        left_ptrs = left_base + i[:, None] * stride_l_i + k_off[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0).to(tl.float16)

        # right[k,j] tile
        right_ptrs = right_base + k_off[:, None] * stride_r_k + j[None, :] * stride_r_j
        right_tile = tl.load(right_ptrs,

"""
TriMul (outgoing) – fast Triton implementation.

The forward pass consists of:
1️⃣ Layer‑norm over the last dimension of the input.
2️⃣ Two bias‑free linear projections → left / right.
3️⃣ Optional pair‑wise mask (broadcasted on the hidden dimension).
4️⃣ Gated projections (sigmoid of another linear projection) and
   conversion to fp16 to halve the memory traffic of the cubic kernel.
5️⃣ Heavy O(N³) update:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply (B·D independent N×N GEMMs).
   A custom Triton kernel computes the GEMM in fp16 multiplication with
   fp32 accumulation, using 64×64 tiles and a 64‑wide k‑panel.  The
   kernel is fully fused (no Python loops, no extra temporaries) and
   works for the “outgoing’’ variant where the k‑axis is summed over.
6️⃣ Layer‑norm over the hidden dimension, out‑gate scaling and a final
   linear projection back to the original channel size.

All heavy arithmetic (step 5) runs on the GPU via Triton; the rest of
the operations use regular PyTorch kernels.  Mixed‑precision (fp16
for the cubic part) reduces memory bandwidth while keeping the final
result in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trmul_outgoing_kernel(
    left_ptr,               # *ptr* to left tensor   (B, D, N, N)  fp16
    right_ptr,              # *ptr* to right tensor  (B, D, N, N)  fp16
    out_ptr,                # *ptr* to output tensor (B, D, N, N)  fp32
    B, N, D,                # static problem sizes
    stride_l_batch, stride_l_d, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_d, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_d, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched “outgoing’’ TriMul:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    The batch*hidden dimension (B·D) is flattened onto program_id(2).
    Each program processes a (BLOCK_M×BLOCK_N) tile of the output matrix.
    """
    pid_i = tl.program_id(0)   # tile index along i
    pid_j = tl.program_id(1)   # tile index along j
    pid_bd = tl.program_id(2)  # flattened batch*channel index

    # tile coordinates
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    # decode (batch, hidden) from flattened id
    b = pid_bd // D
    d = pid_bd % D

    # pointers to the (b,d) slice of each operand
    left_base  = left_ptr  + b * stride_l_batch + d * stride_l_d
    right_base = right_ptr + b * stride_r_batch + d * stride_r_d
    out_base   = out_ptr   + b * stride_o_batch + d * stride_o_d

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left tile: (i, k) – contiguous in k
        left_ptrs = left_base + i[:, None] * stride_l_i + k_off[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)                     # fp16

        # right tile (transposed view): (k, j)
        right_ptrs = right_base + k_off[:, None] * stride_r_k + j[None, :] * stride_r_j
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)                    # fp16

        # half‑precision multiplication, accumulate in fp32
        acc += tl.dot(left_tile, right_tile).to(tl.float32)

    # write the block back
    out_ptrs = out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Entry point used by the evaluation harness.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor  [B, N, N, dim] (float32)
            - mask:         torch.Tensor  [B, N, N]   (bool/float) or None
            - weights:      dict of torch.Tensors (float32) containing all model parameters
            - config:       dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.to(x.dtype).unsqueeze(-1)           # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of a linear projection) and fp16 cast
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # multiply by gates and drop to half‑precision (memory‑bandwidth saving)
    left = (left * left_gate).half()   # now fp16
    right = (right * right_gate).half()  # now fp16

    # ------------------------------------------------------------------
    # 5️⃣ Triton kernel: batched GEMM (outgoing TriMul)
    #    Layout: (B, D, N, N) → contiguous => easy striding
    # ------------------------------------------------------------------
    #   a) permute to (B, D, N, N) and make contiguous
    left_t = left.permute(0, 3, 1, 2).contiguous()    # [B, D, N, N] fp16
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, D, N, N] fp16

    #   b) allocate output in fp32 (the kernel accumulates to fp32)
    out_t = torch.empty_like(left_t, dtype=torch.float32)  # [B, D, N, N] fp32

    B, D, N, _ = left_t.shape

    #   c) strides in element units (torch returns element strides)
    stride_l_batch, stride_l_d, stride_l_i, stride_l_k = left_t.stride()
    stride_r_batch, stride_r_d, stride_r_j, stride_r_k = right_t.stride()
    stride_o_batch, stride_o_d, stride_o_i, stride_o_j = out_t.stride()

    #   d) tile sizes – tuned for H100 fp16 tensor‑core execution
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    #   e) grid dimensions: (i‑tiles, j‑tiles, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # launch kernel
    trmul_outgoing_kernel[grid](
        left_t,
        right_t,
        out_t,
        B,
        N,
        hidden_dim,
        stride_l_batch,
        stride_l_d,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_d,
        stride_r_j,
        stride_r_k,
        stride_o_batch,
        stride_o_d,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Restore original layout and remaining post‑processing
    # ------------------------------------------------------------------
    # out_t is (B, D, N, N) → permute back to (B, N, N, D)
    out = out_t.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # layer‑norm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # out‑gate scaling
    out = out * out_gate

    # final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
