import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy O(N³) part of the module is the tensor contraction
    out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d] .
For each batch and hidden channel d this is a regular matrix multiplication
    (N×N) @ (N×N)ᵀ.
We fuse this contraction across all (batch, hidden) pairs with a
batched matmul kernel written in Triton.  All surrounding operations
(LayerNorm, linear projections, gating, final projection) are performed
with PyTorch functional calls.

The kernel works on half‑precision inputs (fp16) and accumulates in
float‑32 for numerical stability, finally writing the result as fp32.
The implementation supports the full range of test shapes (B∈{1,2},
N∈{128,…,1024}, dim∈{128,384}, hidden_dim∈{128}).

Usage
-----
custom_kernel(data) where
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

returns a tensor of shape [bs, seq_len, seq_len, dim] (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched matrix multiplication: C = A @ B  (A: [B, M, K], B: [B, K, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                             # matrix sizes
    stride_a_batch, stride_a_m, stride_a_k,   # strides of A
    stride_b_batch, stride_b_k, stride_b_n,   # strides of B
    stride_c_batch, stride_c_m, stride_c_n,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Grid: (ceil(M/BLOCK_M), ceil(N/BLOCK_N), batch)
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (combined batch*hidden)

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers with the batch offset baked in
    a_ptrs = a_ptr + pid_b * stride_a_batch \
                     + offs_m[:, None] * stride_a_m \
                     + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_b * stride_b_batch \
                     + offs_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of tiles along the K dimension
    num_k = tl.cdiv(K, BLOCK_K)

    for _ in range(num_k):
        # Load a tile of A and B (masked for out‑of‑bounds)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)
        # Accumulate
        acc += tl.dot(a, b)               # (M,K)·(K,N) → (M,N)

        # Move to the next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # --------------------------------------------------------------
    # Write back the result
    # --------------------------------------------------------------
    c_ptrs = c_ptr + pid_b * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


def _batched_matmul(A: torch.Tensor, B: torch.Tensor,
                    BLOCK_M=64, BLOCK_N=64, BLOCK_K=64) -> torch.Tensor:
    """
    A: (batch, M, K)   fp16 or fp32
    B: (batch, K, N)   fp16 or fp32
    Returns C: (batch, M, N)  fp32
    """
    assert A.shape[0] == B.shape[0]
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    grid = (triton.cdiv(M, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            batch)

    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]            # input / output channel dimension
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                     # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]      # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)               # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask (broadcast over the hidden channel)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)           # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections of the *normalized* x)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = F.linear(x, left_gate_weight).sigmoid()   # [B, N, N, hidden]
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the Triton batched matmul:
    #   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #
    # We reshape to (B*hidden, N, N) and apply a transpose on the
    # second N dimension of `right` so that the multiplication becomes
    # a regular batched matmul:  A @ B   with
    #   A = left  (N × N)
    #   B = rightᵀ (N × N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # fp16 for the heavy GEMM
    left_h = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # [B, H, N, N]
    right_h = right.permute(0, 3, 2, 1).contiguous().to(torch.float16)  # [B, H, N, N] (transposed)

    left_h = left_h.view(B * hidden_dim, N, N)      # [B*H, N, N]
    right_h = right_h.view(B * hidden_dim, N, N)    # [B*H, N, N]

    # ------------------------------------------------------------------
    # Triton batched matmul (output is fp32)
    # ------------------------------------------------------------------
    out_h = _batched_matmul(left_h, right_h)       # [B*H, N, N] fp32

    # ------------------------------------------------------------------
    # Reshape back to original layout [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden]
    out = out.to(torch.float32)                                 # ensure float32 for following ops

    # ------------------------------------------------------------------
    # Output LayerNorm (hidden_dim) and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # Element‑wise gating after the hidden‑dim LayerNorm
    out = out * out_gate

    # Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"]           # [dim, hidden_dim]
    out = F.linear(out, to_out_weight)                # [B, N, N, dim]

    return out
