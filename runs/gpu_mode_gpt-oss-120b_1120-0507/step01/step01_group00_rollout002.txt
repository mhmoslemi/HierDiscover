"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. Layer‑norm on the channel axis (dim).
2. Cast the tensor to fp16 and fuse the two linear projections
   (left_proj, right_proj) into a single mat‑mul.  Split the result.
3. Fuse the three gating linear layers (left_gate, right_gate, out_gate)
   into a single mat‑mul and split.  Apply sigmoid.
4. Optional mask is broadcast and applied to left/right before the gates.
5. Core operation:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix multiplication L @ Rᵀ for each hidden channel.
   We reshape (B, N, N, H) → (B·H, N, N) and run a custom Triton
   kernel that computes C = A @ B  where A and B are fp16
   (Tensor‑Core friendly) and the accumulator is fp32.
6. Layer‑norm on the hidden dimension, gate with out_gate and linear
   projection back to the original channel dimension.

All heavy arithmetic (steps 2‑5) runs in fp16; the final output is
float‑32.  The Triton kernel processes many (M×N) tiles per batch and
uses block sizes 128×128×32 which are well‑suited for the H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ B  (A,B are fp16, accumulator fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_fp16(
    a_ptr,               # *fp16,   shape (batch, M, K)
    b_ptr,               # *fp16,   shape (batch, K, N) ← transposed view of right
    c_ptr,               # *fp32,   shape (batch, M, N)
    M, N, K,             # int32 matrix dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C = A @ B   (batch‑wise matrix multiply)"""

    pid = tl.program_id(0)

    # ---- grid decomposition ------------------------------------------------
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    batch_id = pid // (num_pid_m * num_pid_n)
    tile_id  = pid % (num_pid_m * num_pid_n)
    pid_m = tile_id // num_pid_n
    pid_n = tile_id % num_pid_n

    # ---- tile offsets -------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- base pointers for this batch ----------------------------------------
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # ---- pointers to the current tile ----------------------------------------
    a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    # ---- accumulation (fp32) -------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K ---------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # load fp16 tiles, zero‑pad out‑of‑bounds elements
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=tl.float16(0.0),
                    dtype=tl.float16)                       # (BM, BK)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=tl.float16(0.0),
                    dtype=tl.float16)                       # (BK, BN)

        # Tensor‑core multiply‑accumulate (result is fp32)
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ---- write back ----------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor        [B, N, N, dim]
        mask         : torch.Tensor|None  [B, N, N]
        weights      : dict of torch.Tensor (model parameters)
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor  [B, N, N, dim] (float32, same device as inputs)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                         # (B, N, N, dim), dtype float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear / matmul work
    # ------------------------------------------------------------------
    x_h = x.half()          # fp16 tensor

    # ------------------------------------------------------------------
    # 3️⃣ Fuse left/right projections (2 * hidden_dim outputs)
    # ------------------------------------------------------------------
    proj_w = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(x_h.dtype)        # (2*H, dim)
    proj = F.linear(x_h, proj_w)                 # (B, N, N, 2*H)
    left, right = torch.split(proj, hidden_dim, dim=-1)  # each (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse gate linear layers (3 * hidden_dim outputs) and sigmoid
    # ------------------------------------------------------------------
    gate_w = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(x_h.dtype)        # (3*H, dim)
    gates = torch.sigmoid(F.linear(x_h, gate_w))   # (B,N,N,3*H)
    left_gate, right_gate, out_gate = torch.split(gates, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: (B,N,N) → (B,N,N,1) and cast to fp16
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)
        left = left * mask_h
        right = right * mask_h

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core TriMul: batched matrix multiplication L @ Rᵀ
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # bring hidden channel to axis 1
    left = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    # we need right transposed on the last two axes
    right = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N) == Rᵀ

    batch_dim = B * hidden_dim
    left_flat  = left.view(batch_dim, N, N)               # fp16
    right_flat = right.view(batch_dim, N, N)              # fp16 (already transposed)

    # output will be accumulated in fp32
    out_flat = torch.empty((batch_dim, N, N), dtype=torch.float32, device=device)

    # ----- launch parameters ------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = batch_dim * grid_m * grid_n
    grid = (total_programs,)

    # strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_flat.stride()   # (batch, K, N)
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # ----- run Triton GEMM ---------------------------------------------------
    _batched_matmul_fp16[grid](
        left_flat,
        right_flat,
        out_flat,
        N,               # M
        N,               # N
        N,               # K
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # (B, N, N, H), dtype float32

    # ------------------------------------------------------------------
    # 7️⃣ Post‑processing: hidden‑dim layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # out_gate is fp16 – cast to fp32 before the element‑wise product
    out = out * out_gate.to(out.dtype)

    # final projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
