"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
The module first Layer‑Normalises the 4‑D input X ∈ ℝ^{B×N×N×C},
projects it to a hidden dimension H (two linear layers), masks and
gates the projected tensors, and finally performs the N³ reduction

    out_{b,i,j,h} = Σ_k left_{b,i,k,h} · right_{b,j,k,h}

which is a batched matrix‑multiply `left @ rightᵀ` for every hidden
channel.  The heavy contraction is carried out in a fused Triton
kernel that processes many independent GEMMs (`B·H` of them) in a
single launch, using FP16 inputs with FP32 accumulation.  The remaining
operations (LayerNorm, gating, final projection) use highly‑optimised
PyTorch kernels.

The kernel tiles the output with BLOCK_M × BLOCK_N tiles (default
64×64) and a BLOCK_K depth (default 32) that matches Tensor‑Core
requirements on H100.  Each block accumulates in FP32 for numerical
stability and writes back in FP16.  The surrounding Python code reshapes
the tensors so that the hidden dimension becomes part of the batch axis,
allowing a single kernel launch for the whole batch.
"""

import torch
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A:[batch,M,K], B:[batch,N,K])
# ---------------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr,                # *Pointer* to A   shape [batch, M, K]
    b_ptr,                # *Pointer* to B   shape [batch, N, K] (interpreted as K×N)
    c_ptr,                # *Pointer* to C   shape [batch, M, N]
    M, N, K,              # matrix dimensions
    stride_a_batch, stride_am, stride_ak,   # strides for A
    stride_b_batch, stride_bk, stride_bn,   # strides for B (treated as K×N)
    stride_c_batch, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication where the second matrix is transposed."""
    pid_batch = tl.program_id(0)   # batch index (B*H)
    pid_m = tl.program_id(1)       # tile index along M (i‑axis)
    pid_n = tl.program_id(2)       # tile index along N (j‑axis)

    # Compute the offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the tile in A and B
    a_tile_ptr = a_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak

    b_tile_ptr = b_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_tile_ptr,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)
        # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑block
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bk

    # Write the result back as FP16
    c = acc.to(tl.float16)
    c_tile_ptr = c_ptr + pid_batch * stride_c_batch \
                 + offs_m[:, None] * stride_cm \
                 + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, c,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


# ---------------------------------------------------------------------------
# Python entry point
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/int) or None
        - weights      : dict of model parameters
        - config       : dict with keys ``dim`` (C) and ``hidden_dim`` (H);
                         optional ``nomask`` (bool) and ``eps``.

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ---------------------------------------------------------------
    # Unpack inputs
    # ---------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]                # input channel dim C
    hidden_dim = config["hidden_dim"]  # hidden dim H
    eps = config.get("eps", 1e-5)

    # ---------------------------------------------------------------
    # Input LayerNorm (over the last dimension)
    # ---------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, C]

    # ---------------------------------------------------------------
    # Linear projections (no bias) → hidden dimension
    # ---------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (H, C)
    right_proj_weight = weights["right_proj.weight"]  # (H, C)

    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,H]

    # ---------------------------------------------------------------
    # Optional masking (mask shape [B,N,N])
    # ---------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask = mask.to(x.dtype).unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------------------------------------------------------------
    # Gating (sigmoid of a linear projection)
    # ---------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]   # (H, C)
    right_gate_weight = weights["right_gate.weight"] # (H, C)
    out_gate_weight   = weights["out_gate.weight"]   # (H, C)

    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------
    # Prepare for the Triton GEMM:
    #   - move hidden dim to the 2nd axis,
    #   - flatten batch × hidden into one dimension,
    #   - cast to FP16 (Tensor‑Core friendly)
    # ---------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left.shape
    batch_flat = B * H

    left_mat  = left.view(batch_flat, N, N)   # [B*H, N, N]  (A)
    right_mat = right.view(batch_flat, N, N)  # [B*H, N, N]  (B)

    # Use half‑precision for the GEMM
    left_half  = left_mat.to(torch.float16)
    right_half = right_mat.to(torch.float16)
    out_half   = torch.empty_like(left_half)   # will hold [B*H, N, N] in fp16

    # ---------------------------------------------------------------
    # Compute strides (in elements, not bytes) for the kernel
    # ---------------------------------------------------------------
    stride_a_batch = left_half.stride(0)   # N*N
    stride_am      = left_half.stride(1)   # N   (row stride)
    stride_ak      = left_half.stride(2)   # 1   (col stride)

    stride_b_batch = right_half.stride(0)  # N*N
    stride_bk      = right_half.stride(2)  # 1   (step over K)
    stride_bn      = right_half.stride(1)  # N   (step over N)

    stride_c_batch = out_half.stride(0)    # N*N
    stride_cm      = out_half.stride(1)    # N
    stride_cn      = out_half.stride(2)    # 1

    # ---------------------------------------------------------------
    # Launch the Triton kernel
    # ---------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch_flat,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_matmul_kernel[grid](
        left_half,
        right_half,
        out_half,
        N,               # M
        N,               # N
        N,               # K
        stride_a_batch, stride_am, stride_ak,
        stride_b_batch, stride_bk, stride_bn,
        stride_c_batch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # H100 shows best performance with 8 warps
    )

    # ---------------------------------------------------------------
    # Reshape back to the original layout and cast to fp32 for the final ops
    # ---------------------------------------------------------------
    out = out_half.view(B, H, N, N)                 # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()      # [B, N, N, H]
    out = out.to(torch.float32)

    # ---------------------------------------------------------------
    # to_out_norm (LayerNorm over the hidden dimension H)
    # ---------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ---------------------------------------------------------------
    # Apply the output gate and final linear projection back to dim C
    # ---------------------------------------------------------------
    out = out * out_gate                         # gating
    to_out_weight = weights["to_out.weight"]     # (C, H)
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,C]

    return out
