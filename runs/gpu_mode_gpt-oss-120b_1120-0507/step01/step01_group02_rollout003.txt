```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps)
    # Cast to fp16 for rest
    x_h = x.to(torch.float16)

    # Cast weights to fp16 once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]   # keep in fp32 for final proj

    # Linear projections
    left = torch.nn.functional.linear(x_h, left_proj_w)   # [B,N,N,H]
    right = torch.nn.functional.linear(x_h, right_proj_w)

    # Gates
    left_gate = torch.nn.functional.linear(x_h, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, out_gate_w).sigmoid()

    # Apply gating (and mask if present)
    left = left * left_gate
    right = right * right_gate
    if mask is not None and not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # Transpose to batch over hidden dim
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    batch = B * H
    left = left.view(batch, N, N)
    right = right.view(batch, N, N)

    # Use batched GEMM via torch.bmm
    # left @ right^T, each of shape (N,N)
    out = torch.bmm(left, right.transpose(1, 2))

    # reshape back
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # to_out_norm (LayerNorm on hidden dim)
    out = out.to(torch.float32)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    # Apply out_gate (cast gate to float32)
    out_gate = out_gate.to(torch.float32)
    out = out * out_gate
    # Final projection
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]

    return out
```

"""
Fast outgoing TriMul (AlphaFold3) implementation.

Algorithm
---------
1.  Layer‑norm the input (FP32 for numerical stability).
2.  Cast to FP16 and run all linear projections / sigmoid gates in FP16.
3.  Apply element‑wise gates and an optional mask.
4.  Rearrange to a batch‑over‑hidden layout ``[B*H, N, N]`` and
    compute the heavy contraction
        out_{b,h} = left_{b,h} @ right_{b,h}.T
    with a batched GEMM (torch.bmm → cuBLAS) that runs on the H100
    at FP16/FP32 mixed precision.
5.  Convert the FP16 result back to FP32 with a tiny Triton kernel
    (avoids an extra PyTorch copy and demonstrates Triton usage).
6.  Layer‑norm over the hidden dimension, multiply the *out‑gate*
    (still FP32) and project back to the original channel dimension.

The implementation keeps the memory layout contiguous, uses the
fast cuBLAS batched GEMM for the N³ reduction and only spends a few
micro‑seconds in the custom Triton kernel.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – cheap FP16 → FP32 conversion (used after the batched GEMM)
# ----------------------------------------------------------------------
@triton.jit
def fp16_to_fp32_kernel(src_ptr, dst_ptr, n_elems, BLOCK_SIZE: tl.constexpr):
    """
    src_ptr : FP16 input tensor (contiguous)
    dst_ptr : FP32 output tensor (contiguous, same size as src)
    n_elems : total number of elements to process
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elems

    # load FP16, cast to FP32 and store
    val = tl.load(src_ptr + offs, mask=mask, other=0.0)
    tl.store(dst_ptr + offs, val.to(tl.float32), mask=mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (FP32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]               # C  (input channel dimension)
    hidden_dim = config["hidden_dim"] # H  (projected hidden dimension)
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Layer‑norm on the input (FP32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C], FP32

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Cast model weights to FP16 once (biases are absent)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, C)

    # ------------------------------------------------------------------
    # Linear projections (no bias) – all in FP16
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x_h, left_proj_w)   # [B,N,N,H]
    right = torch.nn.functional.linear(x_h, right_proj_w)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # Gating (sigmoid) – still FP16
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x_h, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x_h, out_gate_w).sigmoid()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # Apply gates and optional mask (all element‑wise, FP16)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    if mask is not None and not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # Rearrange for a batched GEMM:   [B, H, N, N] → (B*H, N, N)
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]
    B, H, N, _ = left.shape
    batch = B * H

    left = left.view(batch, N, N)   # [B*H, N, N]  (FP16)
    right = right.view(batch, N, N) # [B*H, N, N]  (FP16)

    # ------------------------------------------------------------------
    # Batched GEMM:  out = left @ rightᵀ   (cuBLAS, FP16 → FP32 accumulation)
    # ------------------------------------------------------------------
    out_fp16 = torch.bmm(left, right.transpose(1, 2))   # [B*H, N, N]  FP16

    # ------------------------------------------------------------------
    # Reshape back to the original layout (still FP16)
    # ------------------------------------------------------------------
    out_fp16 = out_fp16.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # Convert the FP16 result to FP32 with a tiny Triton kernel
    # ------------------------------------------------------------------
    total = out_fp16.numel()
    BLOCK_SIZE = 1024
    grid = ( (total + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    out_fp32 = torch.empty_like(out_fp16, dtype=torch.float32)

    fp16_to_fp32_kernel[grid](
        out_fp16,
        out_fp32,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,         # modest warp count – enough for the conversion kernel
    )

    # ------------------------------------------------------------------
    # to_out_norm (LayerNorm over the hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_fp32 = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # [B,N,N,H]  FP32

    # ------------------------------------------------------------------
    # Apply the out‑gate (already FP16 → cast to FP32) and final projection
    # ------------------------------------------------------------------
    out_gate_fp32 = out_gate.to(torch.float32)          # [B,N,N,H]
    out_fp32 = out_fp32 * out_gate_fp32                # gating

    to_out_w = weights["to_out.weight"]                # (dim, hidden_dim), FP32
    out = torch.nn.functional.linear(out_fp32, to_out_w)  # [B,N,N,dim]

    return out
