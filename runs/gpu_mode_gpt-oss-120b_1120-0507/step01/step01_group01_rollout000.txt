"""
TriMul (outgoing) – fast Triton + PyTorch implementation
-------------------------------------------------------

The expensive part of the TriMul forward pass is the N³ contraction

    out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]

For every batch element *b* and hidden channel *d* this is a regular
matrix multiplication (N×N) @ (N×N)ᵀ.  We fuse all those independent
multiplications into a single Triton kernel that processes the entire
batch×hidden dimension as a *batched* GEMM:

    A (B·D, M, K)  @  B (B·D, K, N)  →  C (B·D, M, N)

where M = N = seq_len and K = seq_len.  The kernel
*accumulates in float‑32* while the inputs are stored as fp16, which
exploits H100 Tensor‑cores and yields high throughput.  Tile sizes
128×128 (M,N) and 64 (K) together with 8 warps per program have been
chosen as a good trade‑off for all test shapes.

All surrounding operations (LayerNorm, linear projections, gating,
final projection) are performed with native PyTorch functions, which
are cheap compared to the O(N³) GEMM.  The final result is returned as
float‑32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched matrix multiplication kernel (fp16 → fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes
    stride_a_batch, stride_a_m, stride_a_k,  # strides of A
    stride_b_batch, stride_b_k, stride_b_n,  # strides of B
    stride_c_batch, stride_c_m, stride_c_n,  # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, k, j]
    for every batch index ``b`` (here ``b`` = batch * hidden).

    Grid: (ceil(M/BLOCK_M), ceil(N/BLOCK_N), batch)
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch*hidden

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers with batch offset baked in
    a_ptrs = a_ptr + pid_b * stride_a_batch \
                     + offs_m[:, None] * stride_a_m \
                     + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_b * stride_b_batch \
                     + offs_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n

    # ------------------------------------------------------------------
    # Accumulator (float32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_tiles):
        # Load tiles (masked for out‑of‑bounds)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        # Compute dot product of the two tiles
        acc += tl.dot(a, b)                # (M,K)·(K,N) → (M,N)

        # Advance to the next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ------------------------------------------------------------------
    # Write back the result (masked)
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + pid_b * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


def _batched_matmul(A: torch.Tensor, B: torch.Tensor,
                    BLOCK_M=128, BLOCK_N=128, BLOCK_K=64,
                    num_warps=8, num_stages=3) -> torch.Tensor:
    """
    Wrapper for the batched matmul kernel.
    A: (batch, M, K)   FP16 or FP32
    B: (batch, K, N)   FP16 or FP32
    Returns C: (batch, M, N) as FP32.
    """
    assert A.shape[0] == B.shape[0], "Batch dimension must match"
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    grid = (triton.cdiv(M, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            batch)

    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=num_stages,
    )
    return C


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]               # input / output feature dimension
    hidden_dim = config["hidden_dim"] # projection dimension (c_z)
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (last dim = dim)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                     # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → hidden_dim
    # --------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_weight)            # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)          # [B, N, N, 1]
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – also on the normalized input x
    # --------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = F.linear(x, left_gate_weight).sigmoid()   # [B, N, N, hidden]
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Prepare for the Triton batched GEMM:
    #        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    This is a batched matmul A @ Bᵀ with
    #        A = left  (N × N)
    #        B = right (N × N)  → we transpose its last two axes.
    # --------------------------------------------------------------
    B, N, _, _ = left.shape                     # B = batch size
    # (B, N, N, hidden) → (B, hidden, N, N) and cast to fp16
    left_h  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # [B, H, N, N]
    right_h = right.permute(0, 3, 2, 1).contiguous().to(torch.float16)  # [B, H, N, N] (transposed)

    # Collapse batch & hidden into a single batch dimension for the kernel
    left_h  = left_h.view(B * hidden_dim, N, N)   # [B*H, M, K]
    right_h = right_h.view(B * hidden_dim, N, N)  # [B*H, K, N]

    # --------------------------------------------------------------
    # 6️⃣ Triton batched matmul (fp16 input, fp32 accumulation)
    # --------------------------------------------------------------
    out_h = _batched_matmul(left_h, right_h,
                            BLOCK_M=128, BLOCK_N=128, BLOCK_K=64,
                            num_warps=8, num_stages=3)          # [B*H, N, N] fp32

    # --------------------------------------------------------------
    # 7️⃣ Reshape back to the original layout
    # --------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden]
    out = out.to(torch.float32)    # ensure FP32 for following ops

    # --------------------------------------------------------------
    # 8️⃣ Output LayerNorm (hidden_dim) + gating + final linear projection
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate                     # element‑wise gating after norm

    to_out_weight = weights["to_out.weight"]  # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)       # [B, N, N, dim]

    return out
