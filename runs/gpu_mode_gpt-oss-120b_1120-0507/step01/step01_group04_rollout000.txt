"""
TriMul (outgoing) forward pass – Triton‑accelerated mask/gate fusion and
half‑precision batched GEMM.

Algorithm
---------
1. Layer‑norm the input (float32).
2. Cast to float16 and apply four linear projections:
   * left_proj  → (B,N,N,H)
   * right_proj → (B,N,N,H)
   * left_gate  → (B,N,N,H)  then sigmoid
   * right_gate → (B,N,N,H)  then sigmoid
   * out_gate   → (B,N,N,H)  then sigmoid
3. Fuse mask, left_gate and right_gate with a tiny Triton kernel:
   out_left  = left  * left_gate  * mask
   out_right = right * right_gate * mask
   (all in float16, mask possibly all‑ones when ``nomask`` is True)
4. Compute the “Triangle‑multiplicative” product as a batched matrix
   multiplication:
   * reshape left  → (B*H, N, N)   (rows = i, cols = k)
   * reshape right → (B*H, N, N)   (rows = k, cols = j)   via a permute
   * out_mat = torch.bmm(left_mat, right_mat)   (float16)
   * reshape back → (B,N,N,H)
5. Cast to float32, apply the output LayerNorm, multiply by out_gate,
   and a final linear projection back to the original channel dimension.
6. Return the result (float32) with shape (B, N, N, dim).

The kernel in step 3 handles the mask and the two per‑channel gates in a
single pass, dramatically reducing memory traffic.  All heavy linear
operations are executed in float16 to exploit Tensor‑cores on an H100,
while the final LayerNorm and linear layers stay in float32 for numerical
stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_f16_kernel(
    left_ptr,               # *float16  [B,N,N,H]
    right_ptr,              # *float16  [B,N,N,H]
    left_gate_ptr,          # *float16  [B,N,N,H]
    right_gate_ptr,         # *float16  [B,N,N,H]
    mask_ptr,               # *float16  [B,N,N] (broadcasted on H)
    out_left_ptr,           # *float16  [B,N,N,H]
    out_right_ptr,          # *float16  [B,N,N,H]
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse mask, left gate and right gate:
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    Operates on flat indices and works for any (B,N,H) that fit in the
    total number of threads.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # decode flat index → (b,i,j,h)
    b = offs // (N * N * H)
    i = (offs // (N * H)) % N
    j = (offs // H) % N
    h = offs % H

    # memory offsets (in elements, not bytes)
    left_off      = b * stride_l_b  + i * stride_l_i  + j * stride_l_j  + h * stride_l_h
    right_off     = b * stride_r_b  + i * stride_r_i  + j * stride_r_j  + h * stride_r_h
    left_gate_off = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h * stride_lg_h
    right_gate_off= b * stride_rg_b + i * stride_rg_i+ j * stride_rg_j+ h * stride_rg_h
    mask_off      = b * stride_m_b  + i * stride_m_i  + j * stride_m_j          # no h dim

    # load with safety mask
    l  = tl.load(left_ptr      + left_off,      mask=mask, other=tl.float16(0.0))
    r  = tl.load(right_ptr     + right_off,     mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + left_gate_off, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr+ right_gate_off,mask=mask, other=tl.float16(0.0))
    m  = tl.load(mask_ptr      + mask_off,      mask=mask, other=tl.float16(1.0))

    # fused multiplication
    out_l = l * lg * m
    out_r = r * rg * m

    # write results
    tl.store(out_left_ptr  + left_off,  out_l, mask=mask)
    tl.store(out_right_ptr + right_off, out_r, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward of AlphaFold3 “outgoing” TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # unpack inputs
    # ------------------------------------------------------------------ #
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    else:  # fallback when mask is omitted
        input_tensor, weights, config = data
        mask = None

    device = input_tensor.device
    dtype  = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # Prepare mask (broadcasted later on hidden dim)
    # ------------------------------------------------------------------ #
    if mask is None or nomask:
        mask = torch.ones((bs, seq_len, seq_len), dtype=dtype, device=device)
    else:
        mask = mask.to(dtype=dtype, device=device)

    # ------------------------------------------------------------------ #
    # Load weights (convert to half where advantageous)
    # ------------------------------------------------------------------ #
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------ #
    # 1) LayerNorm on the input (float32)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------ #
    # 2) Cast to fp16 and perform linear projections (bias‑free)
    # ------------------------------------------------------------------ #
    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_weight.to(torch.float16))
    right = F.linear(x_fp16, right_proj_weight.to(torch.float16))

    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_weight.to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight.to(torch.float16)))
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_weight.to(torch.float16)))

    # ------------------------------------------------------------------ #
    # 3) Fuse mask and per‑channel gates with a Triton kernel (fp16)
    # ------------------------------------------------------------------ #
    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # strides are expressed in element counts (not bytes)
    stride_l  = left.stride()          # (B*N*N*H, N*H, H, 1)
    stride_r  = right.stride()
    stride_lg = left_gate.stride()
    stride_rg = right_gate.stride()
    stride_m  = mask.stride()          # (N*N, N, 1)

    total_elements = bs * seq_len * seq_len * hidden_dim
    BLOCK = 1024 * 4  # 4096 threads per launch – good occupancy on H100
    grid = ((total_elements + BLOCK - 1) // BLOCK,)

    # Cast mask to fp16 once
    mask_fp16 = mask.to(torch.float16)

    _mask_gate_f16_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_fp16,
        left_out, right_out,
        bs, seq_len, hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_lg[0], stride_lg[1], stride_lg[2], stride_lg[3],
        stride_rg[0], stride_rg[1], stride_rg[2], stride_rg[3],
        stride_m[0], stride_m[1], stride_m[2],
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------ #
    # 4) Batched matrix multiplication (fp16) implementing the einsum
    # ------------------------------------------------------------------ #
    # left: (B,N,N,H) → (B*H, N, N)  rows=i, cols=k
    left_mat = left_out.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)
    # right: (B,N,N,H) → (B*H, N, N)  rows=k, cols=j  (need transpose of last two axes)
    right_mat = right_out.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, seq_len, seq_len)

    # batched GEMM – Tensor‑core accelerated
    out_mat = torch.bmm(left_mat, right_mat)      # (B*H, N, N)  fp16

    # reshape back to (B,N,N,H)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    # ------------------------------------------------------------------ #
    # 5) Cast to fp32, LayerNorm over hidden dim, gate, final projection
    # ------------------------------------------------------------------ #
    out_fp32 = out.to(torch.float32)

    out_fp32 = F.layer_norm(out_fp32,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=1e-5)

    # Apply output gate (cast gate to fp32)
    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    # Final linear back to original dim
    out_fp32 = F.linear(out_fp32, to_out_weight)  # (B,N,N,dim)

    return out_fp32
```

This implementation:

* Uses a Triton kernel to fuse mask multiplication with the two per‑channel gates,
  eliminating three separate element‑wise passes.
* Performs all large linear projections, gates and the expensive pairwise
  multiplication in **float16**, leveraging the H100 Tensor‑cores.
* Keeps the more numerically sensitive LayerNorms and the final projection in
  **float32** to preserve model quality.
* Handles both masked and un‑masked scenarios transparently.

The kernel works for any batch size (1 – 2), sequence length up to 1024, and
hidden dimensions up to 128, matching the test configurations.
