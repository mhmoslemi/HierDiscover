"""
TriMul (outgoing) – fast Triton implementation of the O(N³) “pairwise” update.

Algorithm
---------
1. Layer‑norm on the input tensor x ∈ ℝ^{B×N×N×C}.
2. Two linear projections (left/right) → ℝ^{B×N×N×H} (H = hidden_dim).
3. Optional mask (broadcast over H) and gated projection:
        left  = left  * sigmoid(left_gate)
        right = right * sigmoid(right_gate)
   The tensors are cast to float16 to reduce memory traffic.
4. **Heavy O(N³) core**: for each batch b and hidden channel h we compute
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] .
   This is a batched matrix‑multiply (N×N)‑by‑(N×N) performed by the
   Triton kernel `trmul_outgoing_kernel`.  The kernel tiles the i‑j plane
   with BLOCK_M×BLOCK_N blocks and iterates over k in BLOCK_K steps,
   using fp16 loads and tf32‑like accumulation in float32 (via tl.dot).
5. The result (still in float16) is reshaped back to (B,N,N,H), cast to
   float32, layer‑normed, multiplied by the output‑gate, and finally
   projected back to the original channel dimension C.

The kernel is written for H100 (SM‑90) – BLOCK_M=N_BLOCK_N=64 and
BLOCK_K=32 give good occupancy while keeping shared‑memory usage low.
All other operations stay in PyTorch for simplicity and correctness.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trmul_outgoing_kernel(
    left_ptr,      # [B, H, N, N] – left[b, h, i, k]
    right_ptr,     # [B, H, N, N] – right[b, h, k, j]
    out_ptr,       # [B, H, N, N] – out[b, h, i, j]
    B, N, H,       # problem sizes
    stride_l_batch, stride_l_h, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_h, stride_r_k, stride_r_j,
    stride_o_batch, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix‑multiply for the outgoing TriMul:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    """
    pid_i = tl.program_id(0)          # block index along i
    pid_j = tl.program_id(1)          # block index along j
    pid_batch_h = tl.program_id(2)    # flattened (batch, hidden) index

    # ------------------------------------------------------------------
    # Decode batch and hidden‑channel indices
    # ------------------------------------------------------------------
    b = pid_batch_h // H
    h = pid_batch_h % H

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Pointers to the start of the (b, h) slice
    left_base  = left_ptr \
        + b * stride_l_batch + h * stride_l_h
    right_base = right_ptr \
        + b * stride_r_batch + h * stride_r_h
    out_base   = out_ptr \
        + b * stride_o_batch + h * stride_o_h

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left[i, k] tile  (BLOCK_M × BLOCK_K)
        left_ptrs = left_base \
            + i[:, None] * stride_l_i \
            + k_off[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # fp16 → fp32

        # right[k, j] tile  (BLOCK_K × BLOCK_N)
        right_ptrs = right_base \
            + k_off[:, None] * stride_r_k \
            + j[None, :] * stride_r_j
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # GEMM on the tile (uses GPU Tensor Cores under the hood)
        acc += tl.dot(left_tile, right_tile)

    # ------------------------------------------------------------------
    # Store the computed sub‑matrix
    # ------------------------------------------------------------------
    out_ptrs = out_base \
        + i[:, None] * stride_o_i \
        + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask is [B, N, N]; expand to [B, N, N, 1] and cast to same dtype
        mask_exp = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gated projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to fp16 for the O(N³) kernel (halves memory bandwidth)
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange layout to (B, H, N, N) for the kernel.
    #   left: (b, i, k, h) → (b, h, i, k)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    #   right: (b, j, k, h) → (b, h, k, j)
    right_t = right.permute(0, 3, 2, 1).contiguous()

    # Output buffer (fp16)
    out_t = torch.empty_like(left_t, dtype=torch.float16)

    B, H, N, _ = left_t.shape  # N is sequence length

    # ------------------------------------------------------------------
    # 6. Triton kernel launch
    # ------------------------------------------------------------------
    stride_l_batch, stride_l_h, stride_l_i, stride_l_k = left_t.stride()
    stride_r_batch, stride_r_h, stride_r_k, stride_r_j = right_t.stride()
    stride_o_batch, stride_o_h, stride_o_i, stride_o_j = out_t.stride()

    # Tile sizes – tuned for H100 (register‑friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑axis blocks
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑axis blocks
        B * H,                          # flattened batch×hidden dimension
    )

    trmul_outgoing_kernel[grid](
        left_t,
        right_t,
        out_t,
        B,
        N,
        H,
        stride_l_batch,
        stride_l_h,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_h,
        stride_r_k,
        stride_r_j,
        stride_o_batch,
        stride_o_h,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,        # enough warps per block for H100
    )

    # ------------------------------------------------------------------
    # 7. Post‑processing (back to float32)
    # ------------------------------------------------------------------
    # (B, H, N, N) → (B, N, N, H) and cast to float32
    out = out_t.permute(0, 2, 3, 1).to(torch.float32)

    # Layer‑norm over the hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Apply the output gate
    out = out * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
