"""
TriMul (outgoing) – fast half‑precision implementation.

The heavy O(N³) pairwise update
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed as a batched GEMM using a custom Triton kernel that
leverages FP16 tensor‑core matmul (BLOCK_M = BLOCK_N = 64, BLOCK_K = 64)
and stores the result in FP16.  All cheap operations (LayerNorm,
linear projections, sigmoid gates, final LayerNorm + linear) are kept
in FP32 for numerical stability, with a single FP32→FP16 conversion
right before the kernel and an FP16→FP32 conversion afterwards.

The kernel is launched with a three‑dimensional grid:
    (ceil(N / BLOCK_M), ceil(N / BLOCK_N), B * hidden_dim)
where each program instance computes one (i, j) tile of one
(batch, channel) pair.  The implementation works for any batch
size ≤ 2, any sequence length up to 1024, and hidden dimensions
as used in AlphaFold‑3 (hidden_dim = 128).  It supports optional
binary masks and works on the H100 (TRITON 3.3.1).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trmul_outgoing_fp16_kernel(
    left_ptr,          # *pointer* to left tensor   (B, D, N, N)  FP16
    right_ptr,         # *pointer* to right tensor  (B, D, N, N)  FP16
    out_ptr,           # *pointer* to output tensor (B, D, N, N)  FP16
    B, N, D,           # problem sizes (int64)
    stride_l_batch, stride_l_d, stride_l_i, stride_l_k,   # left strides
    stride_r_batch, stride_r_d, stride_r_j, stride_r_k,   # right strides
    stride_o_batch, stride_o_d, stride_o_i, stride_o_j,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched outgoing TriMul:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    Each program instance computes a (BLOCK_M × BLOCK_N) tile of the
    N×N output matrix for a single (batch, channel) pair.
    """
    pid_i = tl.program_id(0)          # tile row index (i)
    pid_j = tl.program_id(1)          # tile col index (j)
    pid_bc = tl.program_id(2)         # flattened batch*channel index

    # ----- compute coordinates -----
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # (BLOCK_M,)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # (BLOCK_N,)
    mask_i = i < N
    mask_j = j < N

    # decode batch and channel from flattened index
    b = pid_bc // D
    d = pid_bc % D

    # base pointers for this (b,d) slice
    left_base  = left_ptr  + b * stride_l_batch + d * stride_l_d
    right_base = right_ptr + b * stride_r_batch + d * stride_r_d
    out_base   = out_ptr   + b * stride_o_batch + d * stride_o_d

    # accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # ----- loop over the reduction dimension k -----
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)   # (BLOCK_K,)
        mask_k = k < N

        # left[i, k]  shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i[:, None] * stride_l_i + k[None, :] * stride_l_k
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0)
        )

        # right[k, j]  shape (BLOCK_K, BLOCK_N) – note the transposed layout
        right_ptrs = right_base + k[:, None] * stride_r_k + j[None, :] * stride_r_j
        right_tile = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=tl.float16(0.0)
        )

        # tensor‑core matmul (FP16 × FP16 → FP16)
        acc += tl.dot(left_tile, right_tile)

    # ----- write back the tile -----
    out_ptrs = out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N] or None (float32)
        - weights      : dict of torch.Tensors containing model parameters
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp32 = torch.float32
    dtype_fp16 = torch.float16

    # ------------------------------------------------------------------
    # cast all weights to the proper dtype once (avoid repeated .to())
    # ------------------------------------------------------------------
    def to_fp16(t):
        return t.to(dtype_fp16)

    norm_w = to_fp16(weights["norm.weight"])
    norm_b = to_fp16(weights["norm.bias"])
    left_proj_w = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w = to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # keep the main computation in FP32 up to the point of the GEMM,
    # then down‑cast to FP16 for the heavy part.
    x_fp32 = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – still FP32
    # ------------------------------------------------------------------
    left = F.linear(x_fp32, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x_fp32, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask is float (0/1) – ensure same type as left/right
        mask_ = mask.to(dtype_fp32).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating – sigmoid(linear(x))
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp32, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_fp32, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_fp32, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the O(N³) GEMM
    # ------------------------------------------------------------------
    left_fp16 = left.to(dtype_fp16).contiguous()
    right_fp16 = right.to(dtype_fp16).contiguous()

    # Bring tensors to layout (B, D, N, N) required by kernel
    #   original: [B, N, N, D]   →   [B, D, N, N]
    left_t = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_t = right_fp16.permute(0, 3, 1, 2).contiguous()

    B, D, N, _ = left_t.shape   # note: last dim = N after permute
    out_t = torch.empty_like(left_t, dtype=dtype_fp16, device=device)

    # ------------------------------------------------------------------
    # 6. Triton kernel launch – compute Σₖ left[...,i,k] * right[...,j,k]
    # ------------------------------------------------------------------
    stride_l_batch, stride_l_d, stride_l_i, stride_l_k = left_t.stride()
    stride_r_batch, stride_r_d, stride_r_j, stride_r_k = right_t.stride()
    stride_o_batch, stride_o_d, stride_o_i, stride_o_j = out_t.stride()

    # Tuned tile sizes for H100 FP16 tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (tiles_i, tiles_j, batch*channel)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # Launch the kernel (num_warps=8 gives good occupancy on H100)
    trmul_outgoing_fp16_kernel[grid](
        left_t,
        right_t,
        out_t,
        B,
        N,
        hidden_dim,
        stride_l_batch,
        stride_l_d,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_d,
        stride_r_j,
        stride_r_k,
        stride_o_batch,
        stride_o_d,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 7. Post‑processing – back to (B, N, N, D) layout, FP32
    # ------------------------------------------------------------------
    out_fp16 = out_t.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim] FP16
    out_fp32 = out_fp16.to(dtype_fp32)                  # FP32 for stable BN/LN

    # 8. LayerNorm + out‑gate
    out_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_fp32 = out_fp32 * out_gate   # out_gate is still FP32

    # 9. Final linear projection back to `dim`
    out = F.linear(out_fp32, weights["to_out.weight"])

    return out
