@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    batch_id = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_id = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = block_id // tl.cdiv(N, BLOCK_N)
    pid_n = block_id % tl.cdiv(N, BLOCK_N)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a = a_ptr + batch_id * stride_a_batch
    b = b_ptr + batch_id * stride_b_batch
    c = c_ptr + batch_id * stride_c_batch

    a_ptrs = a + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=tl.float16(0.0))
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=tl.float16(0.0))
        acc += tl.dot(a, b)  # float32 accumulator, computes dot in fp16->fp32
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    c_ptrs = c + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask)
```

"""
custom_kernel
==============

High‑performance forward implementation of the *outgoing* TriMul operator
used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1. **Layer‑norm** over the channel dimension (float32 – cheap).
2. Cast to FP16 and evaluate all linear projections and gating functions
   (`left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`) in
   half‑precision.  All ELU/SiLU/​sigmoid operations stay in FP16.
3. Apply optional pairwise mask (broadcast over the hidden dimension) and
   fuse the gate multiplicatively.
4. **Core TriMul**:
   The per‑channel operation
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is an independent batched matrix‑multiplication L @ Rᵀ for each
   `(b,d)`.  We fuse the many GEMMs into a single Triton kernel that
   treats the combined batch dimension `B*hidden_dim` as the Triton batch.
   The kernel works on FP16 inputs but accumulates in FP32 for numerical
   stability.
5. **Post‑processing** – layer‑norm, output‑gate and final linear projection
   are performed again in FP16; the final tensor is cast back to float32.

The kernel uses a 128×128 tile (configurable) and a 64‑element K‑tile,
which maps perfectly to the Tensor‑Core size on an NVIDIA H100.
All heavy arithmetic (step 4) runs on the GPU via Triton, while the
remaining lightweight operations use PyTorch.  The implementation supports
`bs∈{1,2}`, `seq_len≤1024`, `dim∈{128,384}`, `hidden_dim≤384` and optional
pair‑wise masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM  C = A @ Bᵀ
#   A : (batch, M, K)   – FP16
#   B : (batch, N, K)   – FP16 (stored transposed, i.e. we pass strides of Bᵀ)
#   C : (batch, M, N)   – FP16 (accumulated in FP32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,                     # *ptr, shape (batch, M, K)
    b_ptr,                     # *ptr, shape (batch, N, K)  (transposed view of original right)
    c_ptr,                     # *ptr, shape (batch, M, N)
    M, N, K,                   # matrix dimensions (int32 scalars)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,   # B is already transposed: (batch, K, N)
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for many independent batches."""
    pid_m = tl.program_id(0)               # tile index in M dimension
    pid_n = tl.program_id(1)               # tile index in N dimension
    batch_id = tl.program_id(2)            # which (batch, hidden) pair

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the current batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # ------------------------------------------------------------------
    # K‑loop
    # ------------------------------------------------------------------
    offs_k = tl.arange(0, BLOCK_K)
    a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=tl.float16(0.0))
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=tl.float16(0.0))
        acc += tl.dot(a, b)                     # fp16·fp16 → fp32 accumulation
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ------------------------------------------------------------------
    # Write back (cast accumulator to FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (FP16‑accelerated).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]  (float32)
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensor (float32)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as inputs)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input layer‑norm (keep in FP32 – cheap and numerically stable)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )                     # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Switch to half precision for the heavy part
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free) – FP16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)     # [B, N, N, hidden_dim]  (fp16)
    right = F.linear(x_fp16, right_proj_w)   # same shape

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1]  (fp16)
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating (linear → sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # stored for later

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul – batched GEMM using Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # B, seq_len, seq_len, hidden_dim
    batch_dim = B * hidden_dim                  # total GEMM batch count

    # Move hidden dimension to the leading axis and make the tensor contiguous.
    # Layout after .contiguous():   (B*hidden_dim, N, N)  – row‑major
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(batch_dim, N, N)   # fp16
    right_flat = right_perm.view(batch_dim, N, N) # fp16 (will be read as transposed)

    # Output buffer (fp16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Kernel launch parameters
    # ------------------------------------------------------------------
    # Tile sizes – chosen to match H100 Tensor‑Core granularity.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_m, grid_n, batch_dim)   # 3‑D grid: (tile_m, tile_n, batch)

    # Strides (int64) – for a contiguous (batch, N, N) tensor:
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_m, stride_b_n = right_flat.stride()
    # Right matrix is accessed as transposed (K,N) -> swap last two strides
    stride_b_k = stride_b_n          # column stride (=1)
    stride_b_n = stride_b_m          # row stride (=N)

    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # Launch
    _batched_gemm_kernel[grid](
        left_flat,                      # a_ptr
        right_flat,                     # b_ptr (as transposed)
        out_flat,                       # c_ptr
        N, N, N,                        # M, N, K
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Reshape back to (B, N, N, hidden_dim)  (fp16)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, hidden_dim] (fp16)

    # ------------------------------------------------------------------
    # 7. Post‑processing – all in FP16 then cast to FP32
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )   # fp16

    out = out * out_gate                       # gating after norm (fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    out = F.linear(out, to_out_w)               # [B, N, N, dim]  (fp16)

    # Cast back to FP32 for the public API
    return out.to(torch.float32)
