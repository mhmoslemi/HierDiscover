def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to start of batch
    a_ptrs = A_ptr + batch_id * stride_Ab
    b_ptrs = B_ptr + batch_id * stride_Bb
    c_ptrs = C_ptr + batch_id * stride_Cb

    # Offsets for A and B for block
    a_offs = a_ptrs + offs_m[:, None] * stride_Am + (tl.arange(0, BLOCK_K)[None,:]) * stride_Ak
    b_offs = b_ptrs + (tl.arange(0, BLOCK_K)[:, None]) * stride_Bk + offs_n[None,:] * stride_Bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A and B tiles
        a = tl.load(a_offs, mask=(mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K)), other=0.0, dtype=tl.float16)
        b = tl.load(b_offs, mask=((k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None,:]), other=0.0, dtype=tl.float16)

        # Compute dot product

    tl.store(c_ptrs + offs_m[:, None] * stride_Cm + offs_n[None,:] * stride_Cn,

"""
TriMul (outgoing) – Triton‐accelerated forward pass.

The implementation follows the reference PyTorch module but replaces the two
most expensive steps with custom Triton kernels:

1. **Mask + gating fusion** – a 1‑D kernel that loads `left`, `right`,
   the two per‑channel gates and the pairwise mask, applies the mask and the
   sigmoid gates and writes the result back as FP16 tensors.
2. **Batched GEMM (A @ Bᵀ)** – a classic blocked matrix‑multiply kernel that
   processes the `hidden_dim` independent heads in a single launch.
   `left` and `right` (already gated) are reshaped to
   (B·hidden_dim, N, N) and multiplied as
   C = A @ Bᵀ  →  out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].

All other operations (LayerNorms, Linear projections, final
normalisation and output projection) are performed with PyTorch (they are
tiny compared to the N³ work).  The kernels are written for Triton 3.3.1
and assume an H100 GPU, but they work for any CUDA device.

The final output tensor has shape `[bs, seq_len, seq_len, dim]` and dtype
`torch.float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# 1️⃣  Mask + gate fusion (produces FP16 tensors)
# --------------------------------------------------------------------------- #
@triton.jit
def _mask_gate_kernel(
    left_ptr,                # *float32  [B,N,N,H]
    right_ptr,               # *float32  [B,N,N,H]
    left_gate_ptr,           # *float32  [B,N,N,H]
    right_gate_ptr,          # *float32  [B,N,N,H]
    mask_ptr,                # *float32  [B,N,N]  (broadcasted on H)
    out_left_ptr,            # *float16  [B,N,N,H]
    out_right_ptr,           # *float16  [B,N,N,H]
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_SIZE: tl.constexpr,
):
    """
    For every element (b,i,j,h) compute
        out_left  = left  * left_gate  * mask
        out_right = right * right_gate * mask
    and store the results as FP16.  The kernel is 1‑D (flat index) to keep the
    launch overhead low.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # ----- decode flat index to (b,i,j,h) ------------------------------
    b = offs // (N * N * H)
    r = offs % (N * N * H)
    i = r // (N * H)
    r = r % (N * H)
    j = r // H
    h = r % H

    # ----- compute memory offsets ------------------------------------
    left_off = b * stride_l_b + i * stride_l_i + j * stride_l_j + h * stride_l_h
    right_off = b * stride_r_b + i * stride_r_i + j * stride_r_j + h * stride_r_h
    lg_off = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h * stride_lg_h
    rg_off = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + h * stride_rg_h
    mask_off = b * stride_m_b + i * stride_m_i + j * stride_m_j        # no h dim

    # ----- load -------------------------------------------------------
    l = tl.load(left_ptr + left_off, mask=mask, other=0.0)          # float32
    r = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + lg_off, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + rg_off, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)          # broadcasted mask

    # ----- fused multiplication ---------------------------------------
    out_l = tl.cast(l * lg * m, tl.float16)
    out_r = tl.cast(r * rg * m, tl.float16)

    # ----- store ------------------------------------------------------
    tl.store(out_left_ptr + left_off, out_l, mask=mask)
    tl.store(out_right_ptr + right_off, out_r, mask=mask)

    """
    Batched matrix multiplication:
        for each batch b:
            C[b] = A[b] @ B[b]ᵀ
    where
        A : (BATCH, M, K)   FP16
        B : (BATCH, K, N)   FP16   (already transposed version of the original tensor)
        C : (BATCH, M, N)   FP32
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ----- tile offsets ------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- base pointers for the current batch -------------------------
    a_batch_ptr = A_ptr + batch_id * stride_Ab
    b_batch_ptr = B_ptr + batch_id * stride_Bb
    c_batch_ptr = C_ptr + batch_id * stride_Cb

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K dimension --------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # k‑tile indices
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # bounds for the k‑tile
        mask_k = k_offset < K

        # load A tile (M × K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_Am + k_offset[None, :] * stride_Ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # load B tile (K × N)
        b_ptrs = b_batch_ptr + k_offset[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # dot product over the K dimension
        acc += tl.dot(a, b)   # result is FP32

    # ----- write back -------------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# 3️⃣  Public entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the AlphaFold3 “outgoing” TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------- #
    # unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # fetch model weights
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------- #
    # 0️⃣  optional mask handling
    # ------------------------------------------------------------------- #
    if mask is None or config.get("nomask", False):
        mask = torch.ones((bs, seq_len, seq_len), dtype=dtype, device=device)
    else:
        mask = mask.to(dtype=dtype)

    # ------------------------------------------------------------------- #
    # 1️⃣  Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------- #
    # 2️⃣  Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left = F.linear(x, left_proj_weight)      # (B,N,N,hidden_dim)  float32
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------- #
    # 3️⃣  Gate pre‑activations (sigmoid)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------- #
    # 4️⃣  Fuse mask & gates (produces FP16 tensors)
    # ------------------------------------------------------------------- #
    left_out_fp16 = torch.empty_like(left, dtype=torch.float16, device=device)
    right_out_fp16 = torch.empty_like(right, dtype=torch.float16, device=device)

    total_elements = bs * seq_len * seq_len * hidden_dim
    BLOCK = 1024
    grid_mask = ((total_elements + BLOCK - 1) // BLOCK, )

    _mask_gate_kernel[grid_mask](
        left, right,
        left_gate, right_gate,
        mask,
        left_out_fp16, right_out_fp16,
        bs, seq_len, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------- #
    # 5️⃣  Batched GEMM: out = left @ rightᵀ   (hidden dim treated as batch)
    # ------------------------------------------------------------------- #
    # reshape to (B*hidden, N, N)   –  left is (i,k), right is (k,j)
    left_mat = left_out_fp16.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right_out_fp16.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    batch = bs * hidden_dim
    M = N = K = seq_len

    left_mat = left_mat.view(batch, M, K)   # (B*H, N, N)  FP16
    right_mat = right_mat.view(batch, K, N) # (B*H, N, N)  FP16 (already transposed)

    out_mat = torch.empty((batch, M, N), dtype=torch.float32, device=device)

    # strides (in elements, not bytes)
    stride_a_b, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_b, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_b, stride_c_m, stride_c_n = out_mat.stride()

    # tile sizes – chosen to hit the sweet spot on an H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid_gemm = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid_gemm](
        left_mat, right_mat, out_mat,
        batch, M, N, K,
        stride_a_b, stride_a_m, stride_a_k,
        stride_b_b, stride_b_k, stride_b_n,
        stride_c_b, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1)  # (B,N,N,hidden)

    # ------------------------------------------------------------------- #
    # 6️⃣  Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------- #
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    # ------------------------------------------------------------------- #
    # 7️⃣  Apply output gate
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8️⃣  Final linear projection back to original dimension
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
