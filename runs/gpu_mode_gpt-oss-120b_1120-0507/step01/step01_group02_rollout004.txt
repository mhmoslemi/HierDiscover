"""
TriMul (outgoing) – forward only.

Algorithm
---------
1. Layer‑norm the input (float32) and cast to FP16.
2. Apply the three linear projections (left_proj, right_proj,
   left_gate/right_gate/out_gate) with FP16 weights → tensors of shape
   [B, N, N, H] (H = hidden_dim).  Gates are passed through a sigmoid.
3. Optional masking (mask broadcast on the hidden dimension) is applied
   in FP16 before gating.
4. Reshape both projected tensors to a batched matrix‑multiply layout
   [B·H, N, N] (row‑major, K‑stride = 1).  This layout is consumed by a
   custom Triton gemm that computes

        out[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]

   i.e. a batched GEMM C = A @ Bᵀ with FP16 arithmetic and FP32 accumulation.
   Block size = 128 × 128 × 32, 8 warps per block for H100.
5. The result is reshaped back to [B, N, N, H] and cast to FP32.
6. Layer‑norm over the hidden dimension, element‑wise multiplication with
   the out‑gate, and a final linear projection back to the original dim
   (float32) produce the module output.

The heavy N³ reduction is performed by the Triton kernel; all other
operations use highly‑optimised PyTorch primitives.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A: [batch, M, K], B: [batch, N, K])
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr,                # *Pointer* to A   shape [batch, M, K]
    b_ptr,                # *Pointer* to B   shape [batch, N, K] (interpreted as K×N)
    c_ptr,                # *Pointer* to C   shape [batch, M, N]
    M, N, K,              # matrix sizes (runtime)
    stride_a_batch, stride_am, stride_ak,   # strides for A
    stride_b_batch, stride_bk, stride_bn,   # strides for B (treated as K×N)
    stride_c_batch, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM where B is implicitly transposed (K,N)."""
    pid_batch = tl.program_id(0)   # combined batch × hidden index
    pid_m = tl.program_id(1)       # tile along M (i‑axis)
    pid_n = tl.program_id(2)       # tile along N (j‑axis)

    # Offsets within the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the beginning of the tile in A and B
    a_tile_ptr = a_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak

    b_tile_ptr = b_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

    # Accumulator in FP32 (Tensor‑Core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_tile_ptr,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)
        # a : (BLOCK_M, BLOCK_K), b : (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bk

    # Write back (convert to fp16)
    c = acc.to(tl.float16)
    c_tile_ptr = c_ptr + pid_batch * stride_c_batch \
                 + offs_m[:, None] * stride_cm \
                 + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, c,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


# -------------------------------------------------------------------------
# Python entry‑point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only implementation of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float)
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    # Cast to fp16 for the heavy path
    x = x_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Project to hidden dimension (FP16 linear layers, no bias)
    # ------------------------------------------------------------------
    # Cast weights once to FP16
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left = torch.nn.functional.linear(x, left_proj_w)          # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_w)        # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)       # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Prepare tensors for the Triton GEMM (batch over hidden dim)
    # ------------------------------------------------------------------
    # Bring hidden dim to the second axis and make memory contiguous.
    # This copy is cheap relative to the N³ work and yields a row‑major layout.
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left.shape
    batch_flat = B * H

    # Reshape to [batch_flat, N, N] – the layout expected by the kernel
    left_mat = left.view(batch_flat, N, N)   # A
    right_mat = right.view(batch_flat, N, N) # B

    # Cast to fp16 (they are already fp16, but ensure dtype)
    left_fp16 = left_mat
    right_fp16 = right_mat
    out_fp16 = torch.empty_like(left_fp16)   # placeholder for C

    # ------------------------------------------------------------------
    # 6️⃣ Compute strides (in elements, not bytes) for the kernel launch
    # ------------------------------------------------------------------
    stride_a_batch = left_fp16.stride(0)   # N * N
    stride_am = left_fp16.stride(1)        # N   (row stride)
    stride_ak = left_fp16.stride(2)        # 1   (col stride)

    stride_b_batch = right_fp16.stride(0)  # N * N
    # B is interpreted as K×N → stride across K (original col) = 1,
    # stride across N (original row) = N
    stride_bk = right_fp16.stride(2)       # 1
    stride_bn = right_fp16.stride(1)       # N

    stride_c_batch = out_fp16.stride(0)    # N * N
    stride_cm = out_fp16.stride(1)         # N
    stride_cn = out_fp16.stride(2)         # 1

    # ------------------------------------------------------------------
    # 7️⃣ Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_flat,                                 # pid 0 : batch*hidden
        (N + BLOCK_M - 1) // BLOCK_M,               # pid 1 : tiles over M (i)
        (N + BLOCK_N - 1) // BLOCK_N,               # pid 2 : tiles over N (j)
    )

    tri_mul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_a_batch, stride_am, stride_ak,
        stride_b_batch, stride_bk, stride_bn,
        stride_c_batch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps works well on H100
        num_stages=3,         # double‑buffering
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to the original layout and cast to FP32
    # ------------------------------------------------------------------
    out = out_fp16.view(B, H, N, N)          # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 9️⃣ to_out_norm (LayerNorm over hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 10️⃣ Apply out_gate and final projection back to `dim`
    # ------------------------------------------------------------------
    # out_gate was computed in FP16 → cast to FP32 for the element‑wise product
    out_gate_fp32 = out_gate.to(torch.float32)
    out = out * out_gate_fp32

    to_out_w = weights["to_out.weight"]      # (dim, hidden_dim) – float32
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim]

    return out
