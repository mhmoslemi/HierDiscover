TriMul (outgoing) – Triton‑accelerated mask + gate fusion, fp16 batched GEMM.

Algorithm
---------
1. Layer‑norm the input (float32 – same eps as nn.LayerNorm).
2. Cast to float16 and run five bias‑free Linear projections:
   left, right, left_gate, right_gate, out_gate.
3. Fuse the mask, sigmoid‑gates and element‑wise mask multiplication
   in a tiny Triton kernel (`_mask_gate_kernel`).  The kernel writes
   `left_out` and `right_out` (both fp16) directly.
4. Compute the expensive pairwise reduction
      out[b,i,j,d] = Σ_k left_out[b,i,k,d] * right_out[b,j,k,d]
   as a batched fp16 GEMM: for each head d we multiply
   left_out[...,d] (N×N) with right_out[...,d]ᵀ.  This uses
   `torch.bmm`, which on an H100 runs on Tensor‑Cores.
5. Convert the result back to float32, apply a second LayerNorm on
   the hidden dimension, multiply by the output‑gate, and perform the
   final Linear projection back to `dim`.
6. Return the result (float32) with shape [bs, seq_len, seq_len, dim].

The kernel is required by the problem statement and significantly
reduces memory traffic by fusing three element‑wise operations.
"""

import triton.language as tl

# projections.  All tensors are stored in the same dtype (fp16 or fp32).
# ----------------------------------------------------------------------

    left_gate_ptr,           # *{dtype}   [B,N,N,H]
    right_gate_ptr,          # *{dtype}   [B,N,N,H]
    mask_ptr,                # *{dtype}   [B,N,N]
    out_left_ptr,            # *{dtype}   [B,N,N,H]
    out_right_ptr,           # *{dtype}   [B,N,N,H]
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_SIZE: tl.constexpr,
):
    """Apply mask and sigmoid‑gates to the projected tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    active = offs < total

    # ---- Decode linear index into (b,i,j,h) ----
    b = offs // (N * N * H)
    r1 = offs % (N * N * H)
    i = r1 // (N * H)
    r2 = r1 % (N * H)
    j = r2 // H
    h = r2 % H

    # ---- Compute memory offsets for each tensor ----
    left_off      = b * stride_l_b  + i * stride_l_i  + j * stride_l_j  + h * stride_l_h
    right_off     = b * stride_r_b  + i * stride_r_i  + j * stride_r_j  + h * stride_r_h
    left_gate_off = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h * stride_lg_h
    right_gate_off= b * stride_rg_b + i * stride_rg_i+ j * stride_rg_j+ h * stride_rg_h
    mask_off      = b * stride_m_b  + i * stride_m_i  + j * stride_m_j          # no h dimension

    # ---- Load values -------------------------------------------------
    l  = tl.load(left_ptr      + left_off,      mask=active, other=0.0)
    r  = tl.load(right_ptr     + right_off,     mask=active, other=0.0)
    lg = tl.load(left_gate_ptr + left_gate_off, mask=active, other=0.0)
    rg = tl.load(right_gate_ptr+ right_gate_off,mask=active, other=0.0)
    m  = tl.load(mask_ptr      + mask_off,      mask=active, other=1.0)   # broadcasted mask

    # ---- Fuse the three element‑wise operations ----------------------
    out_l = l * lg * m
    out_r = r * rg * m

    # ---- Store results ------------------------------------------------
    tl.store(out_left_ptr  + left_off,  out_l, mask=active)
    tl.store(out_right_ptr + right_off, out_r, mask=active)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------- #
    # unpack inputs
    # --------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------- #
    # 1) Input layer‑norm (float32)
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B,N,N,dim]  float32

    # --------------------------------------------------------------- #
    # 2) Cast to fp16 for all heavy linear work
    # --------------------------------------------------------------- #
    x = x.to(torch.float16)

    # Cast all projection / gate weights to fp16 once (tiny tensors)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # --------------------------------------------------------------- #
    # 3) Linear projections (bias‑free) and gate pre‑activations
    # --------------------------------------------------------------- #
    # shape after each: [B,N,N,hidden_dim]  (fp16)
    left_raw   = F.linear(x, left_proj_w)      # left projection
    right_raw  = F.linear(x, right_proj_w)     # right projection
    left_gate_raw  = F.linear(x, left_gate_w) # left gate pre‑sigmoid
    right_gate_raw = F.linear(x, right_gate_w)# right gate pre‑sigmoid
    out_gate_raw   = F.linear(x, out_gate_w)  # out gate pre‑sigmoid

    left_gate  = torch.sigmoid(left_gate_raw)   # fp16
    right_gate = torch.sigmoid(right_gate_raw)  # fp16
    out_gate   = torch.sigmoid(out_gate_raw)    # fp16, will be used later

    # --------------------------------------------------------------- #
    # 4) Prepare mask (broadcasted on hidden dim)
    # --------------------------------------------------------------- #
    if mask is None or config.get("nomask", False):
        mask = torch.ones((bs, seq_len, seq_len), dtype=torch.float16, device=device)
    else:
        mask = mask.to(dtype=torch.float16, device=device)

    # --------------------------------------------------------------- #
    # 5) Fuse mask and per‑head gates with a tiny Triton kernel
    # --------------------------------------------------------------- #
    left_out  = torch.empty_like(left_raw)
    right_out = torch.empty_like(right_raw)

    # stride information (element‑wise, not bytes)
    stride_l  = left_raw.stride()          # (B*N*N*H, N*H, H, 1)
    stride_r  = right_raw.stride()
    stride_lg = left_gate.stride()
    stride_rg = right_gate.stride()
    stride_m  = mask.stride()              # (N*N, N, 1)

    total_elements = bs * seq_len * seq_len * hidden_dim
    BLOCK = 1024  # each thread processes one element
    grid = ((total_elements + BLOCK - 1) // BLOCK,)

    _mask_gate_kernel[grid](
        left_raw, right_raw,
        left_gate, right_gate,
        mask,
        left_out, right_out,
        bs, seq_len, hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_lg[0], stride_lg[1], stride_lg[2], stride_lg[3],
        stride_rg[0], stride_rg[1], stride_rg[2], stride_rg[3],
        stride_m[0], stride_m[1], stride_m[2],
        BLOCK_SIZE=BLOCK,
    )

    # --------------------------------------------------------------- #
    # 6) Pairwise multiplication over the sequence dimension k
    #    (equivalent to the einsum in the reference code)
    # --------------------------------------------------------------- #
    # reshape to (B*H, N, N) – each head becomes a separate batch
    left_mat  = left_out.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)
    # right_mat must be transposed on the last two axes to match the einsum
    right_mat = right_out.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, seq_len, seq_len)

    # Batched half‑precision GEMM (Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat)                     # (B*H, N, N)  fp16

    # reshape back to [B, N, N, H]
    out = (
        out_mat.view(bs, hidden_dim, seq_len, seq_len)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # --------------------------------------------------------------- #
    # 7) Second LayerNorm on the hidden dimension (float32)
    # --------------------------------------------------------------- #
    out = out.to(torch.float32)  # LayerNorm runs in FP32 for stability
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------- #
    # 8) Apply the output gate (broadcasted)
    # --------------------------------------------------------------- #
    out_gate = out_gate.to(torch.float32)  # fp32 now
    out = out * out_gate

    # --------------------------------------------------------------- #
    # 9) Final linear projection back to the original `dim`
    # --------------------------------------------------------------- #
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim) float32

    return out
