import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) implementation.

The operator projects the input tensor X ∈ ℝ^{B×N×N×C} to a hidden dimension H,
applies element‑wise gates, masks the result, and finally computes for each
batch b and hidden 𝘩 the contraction

    out_{b,i,j,𝘩} = Σ_{k=0}^{N−1} left_{b,i,k,𝘩} · right_{b,j,k,𝘩}

which is mathematically equivalent to a batched matrix multiplication
    out_{b,𝘩} = left_{b,𝘩} @ right_{b,𝘩}ᵀ .

The heavy N³ reduction is performed in a fused Triton kernel that computes many
independent GEMMs (one per hidden channel) in a single launch.  The rest of the
pipeline (LayerNorms, linear layers, gating, masking) is carried out with
high‑performance PyTorch operators.  The kernel works on FP16 data with FP32
accumulation and writes the result back in FP16; the surrounding code converts
to FP32 before the final LayerNorm and output projection.
"""

import torch
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ   (A: [B, M, K], B: [B, N, K])
# ---------------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr,                # *Pointer* to A   shape [batch, M, K]
    b_ptr,                # *Pointer* to B   shape [batch, N, K] (interpreted as K×N)
    c_ptr,                # *Pointer* to C   shape [batch, M, N]
    M, N, K,              # matrix sizes
    stride_a_batch, stride_am, stride_ak,   # strides for A
    stride_b_batch, stride_bk, stride_bn,   # strides for B (treated as K×N)
    stride_c_batch, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM where B is conceptually transposed (K,N)."""
    pid_batch = tl.program_id(0)   # batch index (B*H)
    pid_m = tl.program_id(1)       # tile along M (i‑axis)
    pid_n = tl.program_id(2)       # tile along N (j‑axis)

    # Offsets within the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the tile in A and B
    a_tile_ptr = a_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak

    b_tile_ptr = b_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_tile_ptr,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)
        # a : (BLOCK_M, BLOCK_K), b : (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)

        # Advance to next K‑block
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bk

    # Write the result back as FP16
    c = acc.to(tl.float16)
    c_tile_ptr = c_ptr + pid_batch * stride_c_batch \
                 + offs_m[:, None] * stride_cm \
                 + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr, c,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


# ---------------------------------------------------------------------------
# Python entry point
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only TriMul (outgoing) implementation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of model parameters (see code)
        - config       : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, C]
    """
    # -----------------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    eps = config.get("eps", 1e-5)

    # -----------------------------------------------------------------------
    # LayerNorm on the input (norm over the last dimension)
    # -----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # -----------------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (H, C)
    right_proj_weight = weights["right_proj.weight"]  # (H, C)

    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,H]

    # -----------------------------------------------------------------------
    # Optional masking (mask has shape [B,N,N])
    # -----------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # broadcast mask to hidden dim and ensure float type
        mask = mask.to(x.dtype).unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------------
    # Gating: sigmoid(linear) multiplied elementwise
    # -----------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # Prepare tensors for the Triton GEMM (batch over hidden dimension)
    # -----------------------------------------------------------------------
    # bring hidden dimension to 2‑nd axis and make memory contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left.shape
    batch_flat = B * H

    # reshape to [batch_flat, N, N]
    left_mat  = left.view(batch_flat, N, N)   # A: [batch, M, K]  with M=K=N
    right_mat = right.view(batch_flat, N, N)  # B: [batch, N, K]

    # Cast to fp16 for the kernel
    left_half  = left_mat.to(torch.float16)
    right_half = right_mat.to(torch.float16)
    out_half   = torch.empty_like(left_half)   # will hold [batch_flat, N, N] in fp16

    # -----------------------------------------------------------------------
    # Compute strides required by the kernel (in elements, not bytes)
    # -----------------------------------------------------------------------
    stride_a_batch = left_half.stride(0)   # N * N
    stride_am      = left_half.stride(1)   # N   (row stride)
    stride_ak      = left_half.stride(2)   # 1

    stride_b_batch = right_half.stride(0)  # N * N
    # B is interpreted as K×N, thus:
    stride_bk = right_half.stride(2)       # 1   (step over K)
    stride_bn = right_half.stride(1)       # N   (step over N)

    stride_c_batch = out_half.stride(0)    # N * N
    stride_cm      = out_half.stride(1)    # N
    stride_cn      = out_half.stride(2)    # 1

    # -----------------------------------------------------------------------
    # Launch the Triton kernel
    # -----------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch_flat,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_matmul_kernel[grid](
        left_half,
        right_half,
        out_half,
        N,               # M
        N,               # N
        N,               # K
        stride_a_batch, stride_am, stride_ak,
        stride_b_batch, stride_bk, stride_bn,
        stride_c_batch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,        # modest number of warps for H100
    )

    # -----------------------------------------------------------------------
    # Reshape back to original layout and cast to FP32 for the remaining ops
    # -----------------------------------------------------------------------
    out = out_half.view(B, H, N, N)                 # [B,H,N,N]
    out = out.permute(0, 2, 3, 1).contiguous()      # [B,N,N,H]
    out = out.to(torch.float32)

    # -----------------------------------------------------------------------
    # to_out_norm (LayerNorm over hidden dimension)
    # -----------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # -----------------------------------------------------------------------
    # Apply out‑gate and final linear projection back to dim
    # -----------------------------------------------------------------------
    out = out * out_gate                         # gating
    to_out_weight = weights["to_out.weight"]     # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
