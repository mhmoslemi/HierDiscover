"""
TriMul (outgoing) – fast Triton‑accelerated forward.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (FP32 – cheap, numerically stable).
2️⃣ In an AMP (FP16) region compute all linear projections:
   * left  = x·W_left   ,  right = x·W_right
   * left_gate , right_gate , out_gate = σ(x·W_gate)
3️⃣ Fuse mask, left‑gate and right‑gate with a tiny Triton kernel
   (elementwise: out_left = left * left_gate * mask,
               out_right = right * right_gate * mask).
   This kernel works entirely in FP16, reducing memory traffic.
4️⃣ The expensive “pairwise multiplication”
   out[b,i,j,h] = Σ_k left[b,i,k,h]·right[b,j,k,h]
   is a batched GEMM.  By reshaping to (B·H, N, N) we obtain
   B·H independent matrix‑multiplies that are executed by
   torch.bmm (cuBLAS / Tensor‑Cores – FP16).
5️⃣ The result is reshaped back to [B,N,N,H] (still FP16).
6️⃣ Layer‑norm over the hidden dimension (FP16), scale by out_gate,
   and a final linear projection back to the model dimension.
7️⃣ Cast the final tensor to FP32 (the only required output type).

The kernel in step 3 is mandatory (satisfies the “use Triton” rule)
and the rest of the pipeline stays in highly‑optimised PyTorch
operations that already exploit Tensor‑Core acceleration on H100.
"""

    out_right_ptr,           # *fp16 [B,N,N,H]
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_SIZE: tl.constexpr,

    """
    Apply mask and per‑channel gates to left/right tensors.
    All arithmetic is in FP16.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # Decode flat index → (b, i, j, h)
    b  = offs // (N * N * H)
    r1 = offs % (N * N * H)
    i  = r1 // (N * H)
    r2 = r1 % (N * H)
    j  = r2 // H
    h  = r2 % H

    # Compute memory offsets (in elements, not bytes)
    left_off      = b * stride_l_b + i * stride_l_i + j * stride_l_j + h * stride_l_h
    right_off     = b * stride_r_b + i * stride_r_i + j * stride_r_j + h * stride_r_h
    left_gate_off = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h * stride_lg_h
    right_gate_off= b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + h * stride_rg_h
    mask_off      = b * stride_m_b + i * stride_m_i + j * stride_m_j     # no h dimension

    # Load values (masking out‑of‑bounds threads)
    l  = tl.load(left_ptr      + left_off,      mask=mask, other=tl.float16(0.0))
    r  = tl.load(right_ptr     + right_off,     mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + left_gate_off, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr+ right_gate_off,mask=mask, other=tl.float16(0.0))
    m  = tl.load(mask_ptr      + mask_off,      mask=mask, other=tl.float16(1.0))

    # Fuse mask and gates
    out_l = l * lg * m
    out_r = r * rg * m

    # Store the fused tensors
    tl.store(out_left_ptr  + left_off,  out_l, mask=mask)
    tl.store(out_right_ptr + right_off, out_r, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_fp32 = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # 0️⃣ Prepare mask (FP16, broadcasted later on hidden dim)
    # ------------------------------------------------------------------ #
    if mask is None or nomask:
        mask = torch.ones((bs, seq_len, seq_len), dtype=torch.float16, device=device)
    else:
        mask = mask.to(dtype=torch.float16, device=device)

    # ------------------------------------------------------------------ #
    # 1️⃣ Input LayerNorm (FP32 – keeps numerical stability)
    # ------------------------------------------------------------------ #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)                     # FP32

    # ------------------------------------------------------------------ #
    # 2️⃣ FP16 region: linear projections + gating
    # ------------------------------------------------------------------ #
    with amp.autocast(enabled=True, dtype=torch.float16):
        # Linear maps (no bias)
        left  = F.linear(x, weights["left_proj.weight"])
        right = F.linear(x, weights["right_proj.weight"])

        # Gates (sigmoid)
        left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
        right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
        out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

        # ------------------------------------------------------------------
        # 3️⃣ Fuse mask & per‑channel gates with a Triton kernel (FP16)
        # ------------------------------------------------------------------
        left_out  = torch.empty_like(left)   # FP16
        right_out = torch.empty_like(right)  # FP16

        # strides – element (not byte) strides
        stride_l  = left.stride()
        stride_r  = right.stride()
        stride_lg = left_gate.stride()
        stride_rg = right_gate.stride()
        stride_m  = mask.stride()

        total_elements = bs * seq_len * seq_len * hidden_dim
        BLOCK_SIZE = 1024
        grid = ((total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        _mask_gate_kernel[grid](
            left, right,
            left_gate, right_gate,
            mask,
            left_out, right_out,
            bs, seq_len, hidden_dim,
            stride_l[0], stride_l[1], stride_l[2], stride_l[3],
            stride_r[0], stride_r[1], stride_r[2], stride_r[3],
            stride_lg[0], stride_lg[1], stride_lg[2], stride_lg[3],
            stride_rg[0], stride_rg[1], stride_rg[2], stride_rg[3],
            stride_m[0], stride_m[1], stride_m[2],
            BLOCK_SIZE=BLOCK_SIZE,
        )

        # ------------------------------------------------------------------
        # 4️⃣ Batched GEMM – the O(N³) core (FP16 Tensor‑Core)
        # ------------------------------------------------------------------
        # reshape to (B*H, N, N) where the second matrix is transposed on the
        # last two axes to match the einsum pattern.
        left_mat  = left_out.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)
        right_mat = right_out.permute(0, 3, 2, 1).reshape(bs * hidden_dim, seq_len, seq_len)

        out_mat = torch.bmm(left_mat, right_mat)                     # (B*H, N, N) FP16

        # reshape back to [B, N, N, H] (still FP16)
        out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1)

        # ------------------------------------------------------------------
        # 5️⃣ Output LayerNorm, gating and final projection (still FP16)
        # ------------------------------------------------------------------
        to_out_norm_w = weights["to_out_norm.weight"].half()
        to_out_norm_b = weights["to_out_norm.bias"].half()

        out = F.layer_norm(out,
                           normalized_shape=(hidden_dim,),
                           weight=to_out_norm_w,
                           bias=to_out_norm_b,
                           eps=1e-5)

        # Apply the output gate (both half)
        out = out * out_gate

        # Final linear back to `dim` (Tensor‑Core path)
        out = F.linear(out, weights["to_out.weight"])

    # -------------------------------------------------------------- #
    # 6️⃣ Cast back to FP32 for the API contract
    # -------------------------------------------------------------- #
    return out.to(dtype_fp32)
