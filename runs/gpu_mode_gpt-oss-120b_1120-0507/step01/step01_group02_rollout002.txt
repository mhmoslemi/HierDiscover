"""
TriMul (outgoing) – fast Triton implementation

The heavy part of the TriMul layer is the contraction

    out_{b,i,j,h} = Σ_k left_{b,i,k,h} · right_{b,j,k,h}

for a batch of size B, sequence length N and hidden dimension H (the
``hidden_dim``).  This is a batched matrix‑multiply where each hidden
channel h performs an N×N × N×N^T product.  The kernel below computes all
B·H matrix products in a single launch, using FP16 data and FP32
accumulation (Tensor‑Core friendly).  Block sizes are chosen adaptively
(64² or 128²) and the kernel is launched with 4‑8 warps and 4 stages to
hide memory latency.  All surrounding ops (LayerNorm, Linear,
gating, final projection) are kept in PyTorch for simplicity; only the
N³ reduction is Triton‑accelerated.

The implementation follows the reference PyTorch module but skips
gradient computation (forward‑only) and respects optional masking.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# -----------------------------------------------------------------------
# Triton kernel:  C = A @ Bᵀ   (A: [B, M, K]   B: [B, N, K]  -> Bᵀ is K×N)
# -----------------------------------------------------------------------
@triton.jit
def tr_mul_gemm_kernel(
    a_ptr,               # [batch, M, K]     (FP16)
    b_ptr,               # [batch, N, K]     (FP16, interpreted as K×N)
    c_ptr,               # [batch, M, N]     (FP16 output)
    M, N, K,             # matrix dimensions (M = N = seq_len)
    stride_a_batch, stride_am, stride_ak,   # strides for A
    stride_b_batch, stride_bk, stride_bn,   # strides for B (K×N)
    stride_c_batch, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM with B transposed (K×N).  Uses FP16 load/store,
    FP32 accumulation and Tensor‑Core dot."""
    pid_batch = tl.program_id(0)      # combined B×H index
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # Row / column offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the tile in A and B
    a_tile_ptr = a_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak

    b_tile_ptr = b_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

    # Accumulator in FP32 (Tensor‑Core dot returns FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension (K = seq_len)
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_tile_ptr,
            mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
            other=0.0,
        )
        b = tl.load(
            b_tile_ptr,
            mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
            other=0.0,
        )
        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑block
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bk

    # Cast back to FP16 and store
    c = acc.to(tl.float16)
    c_tile_ptr = c_ptr + pid_batch * stride_c_batch \
                 + offs_m[:, None] * stride_cm \
                 + offs_n[None, :] * stride_cn
    tl.store(
        c_tile_ptr,
        c,
        mask=(offs_m < M)[:, None] & (offs_n < N)[None, :],
    )


# -----------------------------------------------------------------------
# Python entry point
# -----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only TriMul (outgoing) implementation.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict containing all model parameters
        - config       : dict with at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, C]  (float32)
    """
    # -------------------------------------------------------------------
    # unpack inputs
    # -------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]                # C  (input channel dimension)
    hidden_dim = config["hidden_dim"]  # H  (projection dimension)
    eps = config.get("eps", 1e-5)

    # -------------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension (C)
    # -------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]  FP32

    # -------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (no bias)
    # -------------------------------------------------------------------
    # Cast the projection matrices once to the dtype of x (FP32) – they are
    # stored as FP32 in the checkpoint.
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_w)   # [B,N,N,H]
    right = F.linear(x, right_proj_w)  # [B,N,N,H]

    # -------------------------------------------------------------------
    # 3) Optional mask (broadcasted over the hidden dimension)
    # -------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------------
    # 4) Gating – sigmoid(linear) and element‑wise multiplication
    # -------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------
    # 5) Prepare tensors for the Triton batched GEMM
    #    Layout: [batch_flat, N, N] where batch_flat = B * H
    # -------------------------------------------------------------------
    B, N, _, _ = left.shape
    # move hidden dim to the second axis and make everything contiguous
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    batch_flat = B * hidden_dim

    left_mat  = left.view(batch_flat, N, N).to(torch.float16)   # A
    right_mat = right.view(batch_flat, N, N).to(torch.float16)  # B
    out_mat   = torch.empty_like(left_mat)                     # C (FP16)

    # ---------------------------------------------------------------
    # 6) Compute strides (in elements) required by the kernel
    # ---------------------------------------------------------------
    stride_a_batch = left_mat.stride(0)   # N * N
    stride_am      = left_mat.stride(1)   # N   (row stride)
    stride_ak      = left_mat.stride(2)   # 1   (col stride)

    stride_b_batch = right_mat.stride(0)  # N * N
    stride_bk      = right_mat.stride(2)   # 1   (step over K)
    stride_bn      = right_mat.stride(1)   # N   (step over N)

    stride_c_batch = out_mat.stride(0)    # N * N
    stride_cm      = out_mat.stride(1)    # N
    stride_cn      = out_mat.stride(2)    # 1

    # ---------------------------------------------------------------
    # 7) Choose tile sizes based on sequence length
    # ---------------------------------------------------------------
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
        num_warps = 4
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
        num_warps = 8

    # ---------------------------------------------------------------
    # 8) Launch the Triton kernel (batched GEMM)
    # ---------------------------------------------------------------
    grid = (
        batch_flat,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    tr_mul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,               # M (rows)
        N,               # N (cols)
        N,               # K (shared dim)
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_c_batch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=4,          # double‑buffering of loads
    )

    # ---------------------------------------------------------------
    # 9) Reshape back to [B, N, N, H] and cast to FP32 for the rest
    # ---------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N)          # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]
    out = out.to(torch.float32)

    # ---------------------------------------------------------------
    # 10) Output LayerNorm over hidden dimension, gating, final projection
    # ---------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,H] FP32

    out = out * out_gate                     # apply out gate (FP32)

    # final linear projection back to original channel size C
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,C]

    return out
