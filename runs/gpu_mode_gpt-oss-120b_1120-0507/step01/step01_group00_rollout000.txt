@triton.jit
def batched_matmul_kernel_fp16(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    batch_id = pid // (num_pid_m * num_pid_n)
    tile_id = pid % (num_pid_m * num_pid_n)
    pid_m = tile_id // num_pid_n
    pid_n = tile_id % num_pid_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # pointers
    a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over K
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float16)
        # dot returns float16, cast to fp32 to accumulate
        acc += tl.dot(a, b).to(tl.float32)
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store result as half
    c = acc.to(tl.float16)

    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=mask)
```

"""
TriMul (outgoing) – fast half‑precision Triton implementation
================================================================
The module computes

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are linear projections of the input tensor,
optionally masked and gated.  The reduction over the third axis `k`
is exactly a batched matrix multiplication:

    L_d   : (N, N)   ←  left  permuted to (batch, d, i, k)
    R_dᵀ : (N, N)   ←  right permuted to (batch, d, k, j)

    out_flat = batch_matmul(L, Rᵀ)      # shape (B*H, N, N)

All heavy arithmetic (the N³ reduction) is performed in FP16 with
accumulation in FP32 on the H100 tensor‑core‑friendly Triton kernel.
Layer‑norm, gating and the final projection use the standard PyTorch
functional API (which also runs on tensor‑cores when the data is FP16).

The kernel tiles the multiplication with
    BLOCK_M = BLOCK_N = 128, BLOCK_K = 64
which gives good occupancy on H100 while staying inside the 64 KB
shared‑memory limit.  The result is returned as FP32 to match the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched FP16 GEMM with FP32 accumulation
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_fp16_kernel(
    a_ptr,                     # *ptr, shape (batch, M, K) – FP16
    b_ptr,                     # *ptr, shape (batch, K, N) – transposed view, FP16
    c_ptr,                     # *ptr, shape (batch, M, N) – FP16 output
    M, N, K,                   # matrix sizes (int32)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B   where
        A : (batch, M, K)   FP16
        B : (batch, K, N)   FP16 (already transposed view of the original right)
        C : (batch, M, N)   FP16
    Accumulation is performed in FP32 for numerical stability.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Grid decomposition
    # --------------------------------------------------------------
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    batch_id = pid // (num_pid_m * num_pid_n)
    tile_id  = pid % (num_pid_m * num_pid_n)
    pid_m = tile_id // num_pid_n
    pid_n = tile_id % num_pid_n

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Pointers for this batch
    # --------------------------------------------------------------
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    # --------------------------------------------------------------
    # Accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension
    # --------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        ).to(tl.float16)                                   # (BM, BK)
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        ).to(tl.float16)                                   # (BK, BN)

        # dot returns FP16; cast to FP32 before accumulating
        acc += tl.dot(a, b).to(tl.float32)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # --------------------------------------------------------------
    # Store result (cast back to FP16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, c, mask=mask)


# ----------------------------------------------------------------------
# Public entry point used by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator implemented with a fused
    half‑precision Triton GEMM.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensor containing model parameters
        - config       : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same device as inputs)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the channel dimension `dim` (use FP16)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(x.dtype)
    norm_b = weights["norm.bias"].to(x.dtype)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b, eps=eps)  # FP16/FP32...

    # Cast to half for the rest of the pipeline (biggest speed win)
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free) – all in FP16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(x.dtype)
    right_proj_w = weights["right_proj.weight"].to(x.dtype)

    left = F.linear(x, left_proj_w)       # [B, N, N, H]
    right = F.linear(x, right_proj_w)    # same shape

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (linear → sigmoid) – still FP16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(x.dtype)
    right_gate_w = weights["right_gate.weight"].to(x.dtype)
    out_gate_w = weights["out_gate.weight"].to(x.dtype)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul : batched GEMM L @ Rᵀ  (FP16 kernel)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                       # batch, seq_len, _, hidden
    # Move hidden dimension to batch axis: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_dim = B * hidden_dim                     # effective batch for GEMM

    # View as (batch_dim, N, N) – each “batch” corresponds to a hidden channel
    left_flat = left_perm.view(batch_dim, N, N)
    right_flat = right_perm.view(batch_dim, N, N)

    # Output buffer (FP16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_progs = batch_dim * grid_m * grid_n
    grid = (total_progs,)

    # Strides in elements (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_flat.stride()
    # For the transposed view of `right`, swap the last‑two strides
    stride_b_k = stride_b_k          # column stride (==1 for contiguous)
    stride_b_n = stride_b_m          # row stride (==N for contiguous)

    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    _batched_matmul_fp16_kernel[grid](
        left_flat,                      # a_ptr
        right_flat,                     # b_ptr (treated as transposed)
        out_flat,                       # c_ptr
        N, N, N,                        # M, N, K
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H] – still FP16

    # ------------------------------------------------------------------
    # 6. Post‑processing: layer‑norm (hidden), output gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(out.dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(out.dtype)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                       bias=to_out_norm_b, eps=eps)

    out = out * out_gate   # out_gate is FP16

    to_out_w = weights["to_out.weight"].to(out.dtype)
    out = F.linear(out, to_out_w)   # [B, N, N, dim] in FP16

    # Cast back to FP32 for the final result (matches reference dtype)
    return out.to(torch.float32)
