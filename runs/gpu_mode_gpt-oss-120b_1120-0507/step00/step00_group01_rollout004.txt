```
@triton.jit
def triamul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    b = pid // H
    h = pid % H

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base ptr for this batch and hidden channel
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # load a tile (M, K)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load b tile (K, N) so that we can dot a (M,K) @ b (K,N)
        b_ptrs = right_base + cur_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write out result
    out_base = out_ptr + b * stride_out_b + h * stride_out_h
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) implementation with a fused Triton kernel.

The kernel performs the O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
as a batch of GEMMs (one per batch and hidden channel) using
Triton.  All other operations (LayerNorms, linear projections,
sigmoids and the final projection) are carried out with PyTorch
for simplicity.  Half‑precision is used for the large intermediate
tensors (left/right) to reduce memory traffic while accumulation is
done in FP32.

Input:
    data = (x, mask, weights, config)
    x                : [B, N, N, C]   (float32)
    mask (optional)  : [B, N, N]      (float32 or bool)
    weights          : dict of torch.nn.Parameter tensors
    config           : {"dim": C, "hidden_dim": H, ...}

Output:
    y : [B, N, N, C] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def triamul_gemm_kernel(
    left_ptr,            # const * half
    right_ptr,           # const * half
    out_ptr,             # * float32
    B, N, H,             # scalar ints
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: for each (batch, hidden) compute
       out = left @ right.T   where left/right are (N,N) matrices per hidden.
    """
    pid = tl.program_id(0)               # combined batch*hidden index
    b = pid // H
    h = pid % H

    pid_m = tl.program_id(1)             # row block index
    pid_n = tl.program_id(2)             # col block index

    # offsets within the N dimension
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in blocks
    num_k = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k):
        cur_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base \
                 + offs_i[:, None] * stride_l_i \
                 + cur_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)   # half -> float32

        # right tile transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base \
                 + cur_k[:, None] * stride_r_k \
                 + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # write out result
    out_base = out_ptr + b * stride_o_b + h * stride_o_h
    c_ptrs = out_base \
             + offs_i[:, None] * stride_o_i \
             + offs_j[None, :] * stride_o_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype  # expected float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].view(1, 1, 1, -1).to(dtype)
    norm_b = weights["norm.bias"].view(1, 1, 1, -1).to(dtype)

    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b                         # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Cast to half for the large intermediate tensors
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections and gates (all half‑precision)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)               # [B,N,N,hidden]
    right = F.linear(x_h, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    # ------------------------------------------------------------------
    # Optional mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Apply per‑channel gates
    # ------------------------------------------------------------------
    left = left * left_gate          # still half‑precision
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core O(N³) reduction using Triton (batch*hidden GEMMs)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (PyTorch is row‑major, contiguous)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Triton block sizes – tuned for H100 and half precision
    BLOCK_M = 64   # rows of the output block
    BLOCK_N = 64   # cols of the output block
    BLOCK_K = 32   # reduction tile

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    triamul_gemm_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # to_out LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].view(1, 1, 1, -1).to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].view(1, 1, 1, -1).to(dtype)

    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # Apply the output gate (still float32)
    # ------------------------------------------------------------------
    out_gate_f = out_gate.to(dtype)                     # [B,N,N,hidden]
    out_norm = out_norm * out_gate_f

    # ------------------------------------------------------------------
    # Final linear projection back to 'dim'
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(dtype)
    out_final = F.linear(out_norm, to_out_w)            # [B,N,N,dim]

    return out_final
