    B, N, H,

    stride_b_batch, stride_bj, stride_bk, stride_bh,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(axis=0)  # combined batch*head
    batch_idx = pid_bh // H
    head_idx = pid_bh % H
    
    # compute block row and col ids via axis1 and axis2? Actually we need 2D grid for i,j.
    pid_m = tl.program_id(axis=1)  # block row i
    pid_n = tl.program_id(axis=2)  # block col j

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # compute pointers offsets for a
    a_ptrs = a_ptr + batch_idx * stride_a_batch + head_idx * stride_ah + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak

    # compute pointers for b transposed (k,j)
    b_ptrs = b_ptr + batch_idx * stride_b_batch + head_idx * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        k = start_k * BLOCK_K + offs_k
        # but using static loop across k blocks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k < N)[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)
        # increment pointers for next k block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    # store accumulator
    c_ptrs = c_ptr + batch_idx * stride_c_batch + head_idx * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_ai, stride_ak, stride_ah,
    stride_b_batch, stride_bj, stride_bk, stride_bh,
    stride_c_batch, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    batch_idx = pid_bh // H
    head_idx = pid_bh % H

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch/head
    a_ptrs = a_ptr + batch_idx * stride_a_batch + head_idx * stride_ah
    b_ptrs = b_ptr + batch_idx * stride_b_batch + head_idx * stride_bh
    # offsets for i (m) and j (n) within ptrs
    # We'll compute inner loop over k using pointer offset increments

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k-blocks
    for k_offset in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_offset * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load a block: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptrs + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # load b block transposed: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptrs + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # convert to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)

    # final store
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + batch_idx * stride_c_batch + head_idx * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj

def layer_norm(x, weight, bias, eps=1e-5):
    # x shape [..., dim]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    inv = torch.rsqrt(var + eps)
    return (x - mean) * inv * weight + bias

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True) # default True
    # extract weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. LayerNorm on input (dim)
    x = layer_norm(input_tensor, norm_weight, norm_bias)  # shape [B,N,N,dim]

    # 2. Projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # 3. Apply mask if any
    if not nomask:
        # mask shape [B,N,N]
        mask_ = mask.unsqueeze(-1).to(left.dtype)  # broadcast
        left = left * mask_
        right = right * mask_

    # 4. Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # 5. Apply gates to left and right
    left = left * left_gate
    right = right * right_gate

    # 6. Cast to fp16 for matmul kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Output tensor
    out_h = torch.empty_like(left_h)

    # 7. Compute batched matmul via Triton kernel
    B, N, _, H = left_h.shape  # H == hidden_dim

    # Strides (in element counts)
    stride_a = left_h.stride()
    stride_b = right_h.stride()
    stride_c = out_h.stride()

    # Kernel launch config
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batch_matmul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B, N, H,
        stride_a[0], stride_a[1], stride_a[2], stride_a[3],
        stride_b[0], stride_b[1], stride_b[2], stride_b[3],
        stride_c[0], stride_c[1], stride_c[2], stride_c[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Cast result back to float32
    out = out_h.to(torch.float32)

    # 8. to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5
    )

    # 9. Apply out_gate
    out = out * out_gate

    # 10. Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) forward pass implemented with Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension (dim).
2. Compute left/right linear projections and their gates.
3. Apply an optional mask and the gates element‑wise.
4. Perform the core “einsum”:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is a batched matrix multiplication for every head (hidden_dim).
   The heavy‑weight work is done by a Triton kernel that evaluates
   C = A @ Bᵀ for each (batch,head) pair, where A/L = left,
   B = right and the transpose is on the k‑dimension.
5. Layer‑norm the intermediate result (hidden_dim) and modulate with
   a third gate (out_gate).
6. Final linear projection back to the original feature dimension (dim).

Only the N³ accumulation (step 4) is performed on the GPU via Triton;
all other cheap operations stay in PyTorch.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul where each batch*head computes
#   C = A @ Bᵀ   (A: [i,k], B: [j,k] -> C: [i,j])
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # tensors (fp16)
    B, N, H,                            # runtime sizes
    stride_a_batch, stride_ai, stride_ak, stride_ah,
    stride_b_batch, stride_bj, stride_bk, stride_bh,
    stride_c_batch, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    a_ptr : LEFT  [B, N, N, H]  (i, k, h)
    b_ptr : RIGHT [B, N, N, H]  (j, k, h)
    c_ptr : OUT   [B, N, N, H]  (i, j, h)
    For each (b, h) we compute C = A @ Bᵀ over the k‑axis.
    """
    pid_bh = tl.program_id(0)        # combined batch * head index
    pid_m  = tl.program_id(1)        # block row (i)
    pid_n  = tl.program_id(2)        # block col (j)

    # -------------------------------------------------------------
    # Decode batch and head from the combined id
    # -------------------------------------------------------------
    batch_idx = pid_bh // H
    head_idx  = pid_bh % H

    # -------------------------------------------------------------
    # Offsets for the M and N dimensions (i and j)
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (batch,head) pair
    a_base = a_ptr + batch_idx * stride_a_batch + head_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_b_batch + head_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_c_batch + head_idx * stride_ch

    # -------------------------------------------------------------
    # Accumulator in FP32 for numerical stability
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)

    # -------------------------------------------------------------
    # Loop over the reduction axis k
    # -------------------------------------------------------------
    for kb in range(num_k_blocks):
        # global k offsets
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block transposed (k, j): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_base
            + offs_k[:, None] * stride_bk   # k‑dimension as rows
            + offs_n[None, :] * stride_bj,  # j‑dimension as columns
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Convert to FP32, accumulate
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # -------------------------------------------------------------
    # Write out the result (convert back to FP16)
    # -------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_base \
        + offs_m[:, None] * stride_ci \
        + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper: simple LayerNorm (no bias for speed‑up)
# ----------------------------------------------------------------------
def layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    inv = torch.rsqrt(var + eps)
    return (x - mean) * inv * weight + bias

# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask         : torch.Tensor [B, N, N]    (bool/float) or None
        weights      : dict of torch.Tensors (model parameters)
        config       : dict containing "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_weight        = weights["norm.weight"]
    norm_bias          = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = layer_norm(input_tensor, norm_weight, norm_bias)   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left  = left  * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply gates to the projected tensors
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 and allocate output buffer for the heavy einsum
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)
    out_h   = torch.empty_like(left_h)   # [B, N, N, hidden_dim] FP16

    B, N, _, H = left_h.shape   # H == hidden_dim

    # ------------------------------------------------------------------
    # 6. Launch the Triton kernel (batched matmul for every head)
    # ------------------------------------------------------------------
    # Strides expressed in elements (not bytes)
    stride_a = left_h.stride()   # (batch, i, k, h)
    stride_b = right_h.stride()
    stride_c = out_h.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                         # one program per (batch,head)
        (N + BLOCK_M - 1) // BLOCK_M, # i‑blocks
        (N + BLOCK_N - 1) // BLOCK_N, # j‑blocks
    )

    tri_mul_matmul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B, N, H,
        stride_a[0], stride_a[1], stride_a[2], stride_a[3],
        stride_b[0], stride_b[1], stride_b[2], stride_b[3],
        stride_c[0], stride_c[1], stride_c[2], stride_c[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to FP32 for the remaining lightweight operations
    out = out_h.to(torch.float32)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 8. Apply the final gate and linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate                     # gating
    out = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
