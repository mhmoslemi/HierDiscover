```python
import torch, torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_bl, stride_bi, stride_bk, stride_bh,
                   stride_rl, stride_rj, stride_rk, stride_rh,
                   stride_ol, stride_oi, stride_oj, stride_oh,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # grid: (num_blocks_i, num_blocks_j, B*H)
    pid_bh = tl.program_id(2)
    b = pid_bh // H
    h = pid_bh % H

    pid_i = tl.program_id(0)  # block_i
    pid_j = tl.program_id(1)  # block_j

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers
        left_ptrs = left_ptr + b * stride_bl + h * stride_bh \
                    + offs_i[:, None] * stride_bi + offs_k[None, :] * stride_bk
        right_ptrs = right_ptr + b * stride_rl + h * stride_rh \
                    + offs_j[:, None] * stride_rj + offs_k[None, :] * stride_rk

        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # tl.dot expects (M,K) and (N,K) -> (M,N)
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + b * stride_ol + h * stride_oh \
               + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

```python
def custom_kernel(data):
    """
    Triton-accelerated outgoing TriMul forward.
    """
    inp, mask, weights, config = data
    bs, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    # fetch weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on dim
    x = F.layer_norm(inp, (dim,), weight=norm_weight, bias=norm_bias)

    # linear proj
    left = F.linear(x, left_proj_weight)   # [bs,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # mask handling
    if not config.get("nomask", False):
        # mask shape [bs,N,N], broadcast to hidden_dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # cast to fp16 for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)

    # stride info
    stride_bl = left_fp16.stride(0)
    stride_bi = left_fp16.stride(1)
    stride_bk = left_fp16.stride(2)
    stride_bh = left_fp16.stride(3)

    stride_rl = right_fp16.stride(0)
    stride_rj = right_fp16.stride(1)
    stride_rk = right_fp16.stride(2)
    stride_rh = right_fp16.stride(3)

    stride_ol = out_fp32.stride(0)
    stride_oi = out_fp32.stride(1)
    stride_oj = out_fp32.stride(2)
    stride_oh = out_fp32.stride(3)

    # choose tile size heuristics based on N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # grid
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             bs * hidden_dim)

    tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp32,
        bs, N, hidden_dim,
        stride_bl, stride_bi, stride_bk, stride_bh,
        stride_rl, stride_rj, stride_rk, stride_rh,
        stride_ol, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # out now shape [bs,N,N,hidden_dim]
    out = out_fp32  # float32

    # to_out_norm (layernorm over hidden_dim)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply out_gate
    out_gate = out_gate.unsqueeze(-1)  # shape [bs,N,N,hidden_dim]
    out = out * out_gate

    # final linear to original dim
    out = F.linear(out, to_out_weight)

    return out

"""
TriMul outgoing kernel (AlphaFold3 style)

- LayerNorm on the input channel dimension.
- Linear projections + element‑wise masking + sigmoid gating.
- Heavy part:   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
  This is a batched matrix‑multiply (N×N) for every hidden channel `h`.
  Implemented with a fused Triton kernel (fp16 inputs, fp32 accumulation).
- Followed by a second LayerNorm, output gating and a final linear
  projection back to the original dimensionality.

The kernel works for any batch size ≤2, sequence length up to 1024,
and hidden dimension 128 (the values used in the tests).  The Python
wrapper handles all surrounding Torch operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,         # fp16, shape (B,N,N,H)
    right_ptr,        # fp16, shape (B,N,N,H)
    out_ptr,          # fp32, shape (B,N,N,H)
    B, N, H,          # problem sizes
    stride_bl, stride_bi, stride_bk, stride_bh,   # left strides
    stride_rl, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ol, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute for a single (batch, hidden) pair:
        out[i, j] = Σ_k left[i, k] * right[j, k]
    using a classic tiled GEMM approach.
    Grid dimensions: (num_blocks_i, num_blocks_j, B*H)
    """

    # ---------- identify which batch/hidden channel we are working on ----------
    bh_id = tl.program_id(2)                     # 0 … B*H-1
    b = bh_id // H                               # batch index
    h = bh_id % H                                # hidden channel index

    # ---------- block indices in the output matrix ----------
    pid_i = tl.program_id(0)                     # block row (i)
    pid_j = tl.program_id(1)                     # block col (j)

    # ---------- compute global offsets ----------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------- loop over K dimension ----------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < N

        # ----- pointers for the current tile -----
        # left:  (b, i, k, h)
        left_ptrs = (
            left_ptr
            + b * stride_bl
            + h * stride_bh
            + offs_i[:, None] * stride_bi
            + offs_k[None, :] * stride_bk
        )
        # right: (b, j, k, h)   note the order (j,k) for the second matrix
        right_ptrs = (
            right_ptr
            + b * stride_rl
            + h * stride_rh
            + offs_j[:, None] * stride_rj
            + offs_k[None, :] * stride_rk
        )

        # load with boundary masks
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # tl.dot does (M,K)·(N,K) -> (M,N) with fp32 accumulation
        acc += tl.dot(left, right)

    # ---------- write out ----------
    out_ptrs = (
        out_ptr
        + b * stride_ol
        + h * stride_oh
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.
    Input:
        data = (inp_tensor, mask_tensor, weights_dict, config_dict)
    Output:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # unpack arguments
    # --------------------------------------------------------------------- #
    inp, mask, weights, config = data
    bs, seq_len, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # fetch model weights (all tensors are already on the correct device)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1) Input LayerNorm (over the channel dimension)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(inp, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = F.linear(x, left_proj_weight)   # (bs, N, N, hidden_dim)
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------------- #
    # 3) Optional masking (broadcast over hidden_dim)
    # --------------------------------------------------------------------- #
    if not nomask:
        # mask shape: (bs, N, N).  Convert to float and broadcast.
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (bs, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    # 4) Gating (sigmoid of linear transforms of the normalized input)
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5) Heavy O(N³) contraction – performed by Triton kernel
    # --------------------------------------------------------------------- #
    # Cast to fp16 for higher arithmetic throughput (H100 has fast fp16)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    # Output buffer in fp32 (final accumulation precision)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)

    # Strides (in elements, not bytes)
    stride_bl = left_fp16.stride(0)
    stride_bi = left_fp16.stride(1)
    stride_bk = left_fp16.stride(2)
    stride_bh = left_fp16.stride(3)

    stride_rl = right_fp16.stride(0)
    stride_rj = right_fp16.stride(1)
    stride_rk = right_fp16.stride(2)
    stride_rh = right_fp16.stride(3)

    stride_ol = out_fp32.stride(0)
    stride_oi = out_fp32.stride(1)
    stride_oj = out_fp32.stride(2)
    stride_oh = out_fp32.stride(3)

    # Tile sizes – chosen to fit the H100 shared‑memory and warp size
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,
        (seq_len + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        bs,
        seq_len,
        hidden_dim,
        stride_bl,
        stride_bi,
        stride_bk,
        stride_bh,
        stride_rl,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ol,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 6) Post‑processing: LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------------- #
    # out is now (bs, N, N, hidden_dim) in fp32
    out = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    # Apply the output gate (same shape)
    out = out * out_gate

    # Final linear back to original dim (no bias)
    out = F.linear(out, to_out_weight)

    return out
