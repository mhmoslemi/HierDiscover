import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_n, stride_left_k, stride_left_h,
    stride_right_b, stride_right_n, stride_right_k, stride_right_h,
    stride_out_b, stride_out_n, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Identify block indices
    batch_h = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # Decompose batch_h to batch and h
    batch_idx = batch_h // H
    h_idx = batch_h % H

    # Compute starting offsets i, j
    i0 = block_i * BLOCK_M
    j0 = block_j * BLOCK_N

    # Create offsets within block
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k0 in range(0, N, BLOCK_K):
        # offsets for k
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (BLOCK_M, BLOCK_K)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(left_ptr, right_ptr, out_ptr,
                        B, N, H,
                        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
                        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_h = tl.program_id(0)   # combined batch and hidden dim index
    pid_i = tl.program_id(1)     # i tile
    pid_j = tl.program_id(2)     # j tile

    # decompose batch and hidden index
    batch_idx = batch_h // H
    h_idx = batch_h % H

    # compute start offsets for i and j within the N dimension
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # offsets within a tile
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    # boundary checks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k_base in range(0, N, BLOCK_K):
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for left: [batch, i, k, h]
        left_ptrs = left_ptr + \
            batch_idx * stride_left_b + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            h_idx * stride_left_h

        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # pointers for right: [batch, j, k, h]
        right_ptrs = right_ptr + \
            batch_idx * stride_right_b + \
            offs_j[:, None] * stride_right_j + \
            offs_k[None, :] * stride_right_k + \
            h_idx * stride_right_h

        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # dot-product across K; treat right as transposed (K,N)
        acc += tl.dot(left, right, trans_b=True)

    # store result
    out_ptrs = out_ptr + \
        batch_idx * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        h_idx * stride_out_h

    tl.store(out_ptrs,
             acc.to(tl.float32),
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton implementation of the outgoing triangle multiplicative update.
    Performs LayerNorm, linear projections, gating, the O(N^3) pairwise
    multiplication (batched GEMM across the sequence dimension), a second
    LayerNorm, gating and final linear projection.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    bs, N, N2, dim = input_tensor.shape
    assert N == N2
    hidden_dim = config['hidden_dim']

    # ----- LayerNorm on input -----
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights['norm.weight'],
                     bias=weights['norm.bias'],
                     eps=1e-5)

    # ----- Linear projections (no bias) -----
    left = F.linear(x, weights['left_proj.weight'])
    right = F.linear(x, weights['right_proj.weight'])

    # ----- Mask (optional) -----
    if not config.get('nomask', True):
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----- Gates -----
    left_gate = torch.sigmoid(F.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(F.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(F.linear(x, weights['out_gate.weight']))

    left = left * left_gate
    right = right * right_gate

    # ----- Batched GEMM (the N^3 part) -----
    # ensure fp32 for better numeric stability
    left = left.to(torch.float32)
    right = right.to(torch.float32)

    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # strides
    left_strides = left.stride()
    right_strides = right.stride()
    out_strides = out.stride()

    # constants for tiling
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (bs * hidden_dim,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_gemm_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=bs,
        N=N,
        H=hidden_dim,
        stride_left_b=left_strides[0],
        stride_left_i=left_strides[1],
        stride_left_k=left_strides[2],
        stride_left_h=left_strides[3],
        stride_right_b=right_strides[0],
        stride_right_j=right_strides[1],
        stride_right_k=right_strides[2],
        stride_right_h=right_strides[3],
        stride_out_b=out_strides[0],
        stride_out_i=out_strides[1],
        stride_out_j=out_strides[2],
        stride_out_h=out_strides[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Second LayerNorm -----
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights['to_out_norm.weight'],
                       bias=weights['to_out_norm.bias'],
                       eps=1e-5)

    # ----- Output gate -----
    out = out * out_gate

    # ----- Final linear projection -----
    out = F.linear(out, weights['to_out.weight'])

    return out
```

"""
TriMul (outgoing) – Triton implementation
-----------------------------------------

The function implements the forward pass of the “outgoing” triangle
multiplicative update from AlphaFold‑3.  The heavy O(N³) part

    out[..., i, j, d] = Σₖ left[..., i, k, d] * right[..., j, k, d]

is realised as a batched GEMM where each hidden channel `d` is treated as an
independent batch.  A custom Triton kernel computes

    C[t,i,j] = Σₖ A[t,i,k] * B[t,j,k]          (t = batch * hidden_dim)

with explicit tiling (BLOCK_M × BLOCK_N tiles, BLOCK_K reduction).  All
remaining operations (LayerNorms, linear projections, gating) are performed
with regular PyTorch kernels.

The kernel works for any batch size (1‑2), sequence length up to 1024 and
hidden dimension up to 384.  It respects optional masking, uses fp32
accumulation and returns a fp32 tensor compatible with the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM kernel for the TriMul outer product.
    For each (batch, hidden) pair we compute:
        out[t,i,j] = Σₖ left[t,i,k] * right[t,j,k]
    where t = batch * H + hidden.
    """
    # ------------------------------------------------------------------
    # program ids
    # ------------------------------------------------------------------
    batch_h = tl.program_id(0)          # combined batch * hidden index
    pid_i = tl.program_id(1)            # i‑tile
    pid_j = tl.program_id(2)            # j‑tile

    # split combined index into batch and hidden channel
    batch_idx = batch_h // H
    h_idx = batch_h % H

    # ----- tile offsets -------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_j = j0 + tl.arange(0, BLOCK_N)          # [BLOCK_N]
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- reduction over k --------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)      # [BLOCK_K]
        mask_k = offs_k < N

        # left[t,i,k]   (shape BLOCK_M × BLOCK_K)
        left_ptrs = (
            left_ptr
            + batch_idx * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h_idx * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right[t,j,k]  (shape BLOCK_N × BLOCK_K)
        right_ptrs = (
            right_ptr
            + batch_idx * stride_right_b
            + offs_j[:, None] * stride_right_i
            + offs_k[None, :] * stride_right_k
            + h_idx * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # dot across K, treat right as transposed (K × N)
        acc += tl.dot(left, right, trans_b=True)

    # ----- write result -------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch_idx * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h_idx * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float32),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Outgoing Triangle Multiplicative Update (TriMul) – forward pass.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        * input_tensor : torch.Tensor   [B, N, N, dim]
        * mask_tensor  : torch.Tensor   [B, N, N] (optional)
        * weights      : dict of torch.Tensors (model parameters)
        * config       : dict with keys
                         - "dim"         : int
                         - "hidden_dim"  : int
                         - "nomask"      : bool (True → ignore mask)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    B, N, N2, dim = input_tensor.shape
    assert N == N2, "input must be square in the sequence dimensions"
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask is [B, N, N] → [B, N, N, 1] and cast to the proj dtype
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched GEMM – the O(N³) part
    #    We compute for each hidden channel independently.
    # ------------------------------------------------------------------
    # Ensure we work in fp32 (more stable & matches reference)
    left = left.to(torch.float32)
    right = right.to(torch.float32)

    # Output placeholder: shape [B, N, N, hidden_dim]
    out = torch.empty(
        (B, N, N, hidden_dim),
        dtype=torch.float32,
        device=device,
    )

    # Strides (int64) – needed for pointer arithmetic inside the kernel
    left_str = left.stride()      # (B, i, k, h)
    right_str = right.stride()    # (B, j, k, h)
    out_str = out.stride()        # (B, i, j, h)

    # Tiling parameters – chosen to fit well on H100
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # Grid layout: (batch*hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Launch kernel
    tri_mul_gemm_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=left_str[0],
        stride_left_i=left_str[1],
        stride_left_k=left_str[2],
        stride_left_h=left_str[3],
        stride_right_b=right_str[0],
        stride_right_i=right_str[1],
        stride_right_k=right_str[2],
        stride_right_h=right_str[3],
        stride_out_b=out_str[0],
        stride_out_i=out_str[1],
        stride_out_j=out_str[2],
        stride_out_h=out_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # (kernel is asynchronous; the next CUDA operation will synchronize.)

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 7) Output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    # Ensure the output dtype matches the module's original dtype
    out = out.to(input_tensor.dtype)

    return out
