import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    total_batches,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Compute masks
        mask_m = offs_m < M
        mask_n = offs_n < N
        mask_k = k_base < K

        # Compute pointers
        a_ptrs = A_ptr + batch * stride_a_batch + offs_m[:, None] * stride_a_m + k_base[None, :] * stride_a_k
        b_ptrs = B_ptr + batch * stride_b_batch + k_base[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # Load tiles
        a = tl.load(a_ptrs, mask = mask_m[:, None] & mask_k[None, :], other = 0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask = mask_k[:, None] & mask_n[None, :], other = 0.0).to(tl.float32)

        # Compute dot product
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = C_ptr + batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask = mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False) # maybe not needed

    # LayerNorm: x_norm (float32)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    ) # shape B,N,N,dim

    # Cast to half
    x_half = x_norm.to(torch.float16)

    # Linear proj
    left = torch.nn.functional.linear(x_half, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_half, weights["right_proj.weight"].to(torch.float16))

    if mask is not None:

        left = left * mask_f16
        right = right * mask_f16

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_half, weights["left_gate.weight"].to(torch.float16)))

    out_gate = torch.sigmoid(torch.nn.functional.linear(x_half, weights["out_gate.weight"].to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # Compute out = left @ right^T per hidden
    B, N, _, _ = left.shape
    # Permute to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()

import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_segments, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel: C = A @ B, where A shape (batch, M, K), B shape (batch, K, N).
    All tensors are dense and contiguous. Accumulates in fp32, operates on fp16 inputs.
    """
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for OOB in the output tile
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile [M,K]
        a_ptrs = A_ptr + batch_idx * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0)).to(tl.float32)

        # Load B tile [K,N]
        b_ptrs = B_ptr + batch_idx * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0)).to(tl.float32)

        # Compute dot product for this k‑block
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = C_ptr + batch_idx * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """TriMul outgoing update (no gradient) using a fused Triton batched GEMM.
    Args:
        data: tuple (input_tensor, mask_tensor, weight_dict, config_dict)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    # ------------------------------------------------------------
    # 1. Layer‑norm on the input (float32)
    # ------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------
    # 2. Cast to fp16 and compute linear projections + gates
    # ------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(
        x_fp16, weights["left_proj.weight"].to(torch.float16)
    )
    right = torch.nn.functional.linear(
        x_fp16, weights["right_proj.weight"].to(torch.float16)
    )

    # Optional mask (broadcast on hidden dim)
    if mask is not None and not config.get("nomask", False):
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # Gates (sigmoid)
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x_fp16, weights["left_gate.weight"].to(torch.float16)
        )
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x_fp16, weights["right_gate.weight"].to(torch.float16)
        )
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x_fp16, weights["out_gate.weight"].to(torch.float16)
        )
    )

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 3. Batched matrix multiplication: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    This equals (left) @ (right)^T for each (b,d) pair.
    # ------------------------------------------------------------
    B, N, _, _ = left.shape

    # Bring the hidden dimension next to batch and flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    total_batches = B * hidden

    left_flat = left_perm.view(total_batches, N, N)            # [B*H, N, N]
    right_flat_T = right_perm.view(total_batches, N, N).transpose(1, 2).contiguous()  # [B*H, N, N]

    # Output buffer for the GEMM (fp16)
    out_flat = torch.empty_like(left_flat)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat_T,
        out_flat,
        total_batches,
        N,
        N,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat_T.stride(0),
        right_flat_T.stride(1),
        right_flat_T.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] and convert to fp32 for the rest
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------
    # 4. Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)                     # element‑wise gate

    # Final projection back to `dim`
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with a fused Triton batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Cast to fp16 and compute the four linear projections
   (left/right) and the three gate vectors.
3. Apply the mask (if present) and the left/right gates.
4. The core operation is
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batched matrix multiplication:
       out_{(b,d)} = left_{(b,d)} @ right_{(b,d)}ᵀ .
   A custom Triton kernel performs this GEMM for all
   (batch × hidden) pairs, accumulating in fp32.
5. Reshape back to the original 4‑D layout, run the
   second layer‑norm, multiply by the output gate and
   finish with a linear projection to the original `dim`.

The kernel works on fp16 inputs, accumulates in fp32 and
produces fp16 results; the surrounding PyTorch code handles
the remaining (layer‑norm / sigmoid / linear) steps.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_cnt, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM (C = A @ B) for the outgoing TriMul update.

    A : (batch, M, K)  fp16
    B : (batch, K, N)  fp16
    C : (batch, M, N)  fp16  (accumulated in fp32 internally)
    """
    batch_id = tl.program_id(0)          # which (batch × hidden) pair
    pid_m = tl.program_id(1)             # output‑tile row block
    pid_n = tl.program_id(2)             # output‑tile column block

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / columns
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K‑sized chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

            A_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )

            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        ).to(tl.float32)

        # Load B block [K, N]
        b_ptrs = (
            B_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0),
        ).to(tl.float32)

        acc += tl.dot(a, b)

    c_ptrs = (
        C_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing Triangle‑Multiplicative‑Update (TriMul).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim]
            - mask_tensor  : torch.Tensor of shape [B, N, N] (may be None)
            - weights_dict : dict with all model weights
            - config_dict  : dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16, project and compute gates
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left = torch.nn.functional.linear(
        x_fp16, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]
    right = torch.nn.functional.linear(
        x_fp16, weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, H]

    if mask is not None and not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x_fp16, weights["left_gate.weight"].to(torch.float16)
        )
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x_fp16, weights["right_gate.weight"].to(torch.float16)
        )
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x_fp16, weights["out_gate.weight"].to(torch.float16)
        )
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Core update: batched GEMM (outgoing version)
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Move hidden dimension next to batch and flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    total_batches = B * hidden_dim

    left_flat = left_perm.view(total_batches, N, N)                # [B*H, N, N]
    right_flat_T = right_perm.view(total_batches, N, N).transpose(1, 2).contiguous()  # [B*H, N, N]

    # Allocate output buffer (fp16)
    out_flat = torch.empty_like(left_flat)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat_T,
        out_flat,
        total_batches,
        N,
        N,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat_T.stride(0),
        right_flat_T.stride(1),
        right_flat_T.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] and convert to fp32 for the remaining ops
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4) Second LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)

    # Final linear mapping back to original `dim`
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
