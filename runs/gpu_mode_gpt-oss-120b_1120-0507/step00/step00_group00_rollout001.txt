    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Optional mask
    if mask is not None:
        # Ensure mask dtype matches x dtype
        mask = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    right = right * right_gate

    B, N, _, _ = left.shape

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)

    out_flat = torch.empty_like(left_flat)

    # Launch Triton kernel
    M = N
    K = N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

"""
TriMul (outgoing) forward implementation.

Algorithm
---------
1. Layer‑norm the input tensor on the channel axis (`dim`).
2. Linear projections (`left_proj`, `right_proj`) from `dim` → `hidden_dim`.
3. Optional mask multiplication (broadcast over the hidden dimension).
4. Compute three gating tensors (`left_gate`, `right_gate`, `out_gate`) via a
   bias‑free linear + sigmoid and apply them to `left`, `right`.
5. Core operation: for each batch and hidden channel compute
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix‑multiply
        L_d @ R_dᵀ   for every hidden channel `d`.
   This is realised by a custom Triton kernel that fuses the reduction over
   `k` for many batches (batch = B * hidden_dim).
6. Layer‑norm the result on the hidden dimension (`to_out_norm`), gate it with
   `out_gate` and finally project back to `dim` with `to_out`.
All heavy arithmetic (step 5) runs on the GPU via Triton; the remaining
operations use PyTorch functional calls.

The kernel works for arbitrary batch size (1‑2), sequence length up to 1024
and hidden dimension up to 384.  The output is returned as a float32 tensor.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication L @ Rᵀ
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,                     # *ptr, shape (batch, M, K)
    b_ptr,                     # *ptr, shape (batch, K, N)  (Rᵀ view of original right)
    c_ptr,                     # *ptr, shape (batch, M, N)
    M, N, K,                   # matrix dimensions (int32 scalars)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,   # b is already transposed
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B   (A.shape = [B,M,K], B.shape = [B,K,N])"""
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Grid layout: each program processes one (M,N) tile of one batch.
    # --------------------------------------------------------------
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    batch_id = pid // (num_pid_m * num_pid_n)
    tile_id  = pid % (num_pid_m * num_pid_n)
    pid_m = tile_id // num_pid_n
    pid_n = tile_id % num_pid_n

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # Pointers to the current tile (M×K and K×N)
    a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    # Accumulator (float32 for numerical safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in blocks of size BLOCK_K
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)                     # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)                     # (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)                       # (BLOCK_M, BLOCK_N)

        # Move pointers to next K‑block
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ------------------------------------------------------------------
    # Write the result back to C
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask)


# ----------------------------------------------------------------------
# Public entry point used by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensor containing model parameters
        - config       : dict with keys "dim", "hidden_dim" (and optional others)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, on the same device as inputs)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the channel dimension (`dim`)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)  # same shape

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N]  →  [B, N, N, 1]
        mask = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (linear → sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: out = Σ_k left[...,k,:] * right[...,k,:]
    #    Implemented as batch matrix multiplication L @ Rᵀ
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Move hidden dimension to the batch axis for a simple batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_dim = B * hidden_dim
    # shape (batch_dim, N, N)
    left_flat = left_perm.view(batch_dim, N, N)
    right_flat = right_perm.view(batch_dim, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid_M = (N + BLOCK_M - 1) // BLOCK_M
    grid_N = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = batch_dim * grid_M * grid_N
    grid = (total_programs,)

    # Strides (int64)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_flat.stride()
    # For the transposed view of `right` we swap the last‑two strides
    stride_b_k = stride_b_k          # original column stride (==1 for contiguous)
    stride_b_n = stride_b_m          # original row stride (==N)

    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    _batched_matmul_kernel[grid](
        left_flat,                      # a_ptr
        right_flat,                     # b_ptr (will be read as transposed)
        out_flat,                       # c_ptr
        N, N, N,                        # M, N, K
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Post‑processing: layer‑norm, output gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate

    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)   # [B, N, N, dim]

    return out
```
