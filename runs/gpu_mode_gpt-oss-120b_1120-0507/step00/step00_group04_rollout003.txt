```
@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    eps,
    stride_xm, stride_xc,
    stride_om, stride_oc,
    stride_w, stride_b,
    C: tl.constexpr, BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)  # row index
    # compute offsets
    row_offset = pid * stride_xm
    # load vector of C elements in blocks
    c_offset = tl.arange(0, BLOCK_SIZE)
    # mask for c < C
    mask = c_offset < C
    x = tl.load(x_ptr + row_offset + c_offset * stride_xc, mask=mask, other=0.0).to(tl.float32)
    # compute mean
    sum1 = tl.sum(x, axis=0)  # sum across block? Actually we need sum across all C; but if C > BLOCK_SIZE we need multiple loads. We can assume C <= BLOCK_SIZE (max 384). Choose BLOCK_SIZE = 1024; so okay.
    # mean = sum1 / C
    mean = sum1 / C
    # compute var
    var = tl.sum((x - mean) * (x - mean), axis=0) / C
    # normalized
    inv_std = 1.0 / tl.sqrt(var + eps)
    y = (x - mean) * inv_std
    # apply weight and bias
    w = tl.load(weight_ptr + c_offset * stride_w, mask=mask, other=0.0)
    b = tl.load(bias_ptr + c_offset * stride_b, mask=mask, other=0.0)
    y = y * w + b
    # store
    tl.store(out_ptr + row_offset + c_offset * stride_oc, y, mask=mask)
```

```
def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    eps,
    stride_xm, stride_xc,
    stride_om, stride_oc,
    stride_w, stride_b,
```

```
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    right_gate_pre = torch.nn.functional.linear(x_norm, right_gate_weight)

    right_gate = torch.sigmoid(right_gate_pre)
```

"""
TriMul forward (outgoing) – Triton‑accelerated implementation.

The operator works on a 4‑D tensor x ∈ ℝ^{B×N×N×C} (C = dim) and
produces y ∈ ℝ^{B×N×N×C}.  The steps are:

1️⃣ Layer‑Norm over the last dimension (dim).  Implemented with a small
   Triton kernel that reduces each row of size C (C ≤ 384) to a mean
   and variance, normalises, and applies the learned weight/bias.

2️⃣ Linear projections to a hidden dimension H (hidden_dim):
   * left  = x·W_left   (no bias)
   * right = x·W_right

3️⃣ Optional masking (mask ∈ {0,1}^{B×N×N}) applied to left/right.

4️⃣ Gating:
   * left_gate  = σ(x·W_left_gate)
   * right_gate = σ(x·W_right_gate)
   * out_gate   = σ(x·W_out_gate)
   left  ← left  * left_gate
   right ← right * right_gate

5️⃣ Multiplicative update (the expensive O(N³) part):
   For each hidden slice h, compute
       out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]
   i.e. a batched matrix multiplication A @ Bᵀ where A and B are
   reshaped to (B·H, N, N).  We use torch.bmm with FP16 tensors,
   which on an H100 maps to Tensor‑Core GEMM.

6️⃣ Layer‑Norm over the hidden dimension (weights/bias of to_out_norm).

7️⃣ Multiply by out_gate and a final linear projection to the original
   dim (W_to_out).

Only step 1 (the first Layer‑Norm) is performed by a custom Triton
kernel; the rest leverages highly‑optimised PyTorch/cuBLAS kernels.
The final output is returned as a float‑32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: LayerNorm over the last dimension (C)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                     # const * input (M, C)
    out_ptr,                   # * output (M, C)
    weight_ptr,                # const * weight (C,)
    bias_ptr,                  # const * bias   (C,)
    eps,                       # float32 epsilon
    stride_xm, stride_xc,      # strides for input (row, col)
    stride_om, stride_oc,      # strides for output
    stride_w, stride_b,        # strides for weight / bias
    C: tl.constexpr,           # hidden size (dim)
    BLOCK_SIZE: tl.constexpr,  # compile‑time block size (≥ max C)
):
    """Row‑wise LayerNorm (no mean/var across rows)."""
    pid = tl.program_id(0)                     # one program per row (M)
    row_off = pid * stride_xm                   # offset to start of the row

    # column indices (0 … BLOCK_SIZE‑1) and mask for valid entries
    col = tl.arange(0, BLOCK_SIZE)
    col_mask = col < C

    # ---------- load the vector ----------
    x = tl.load(x_ptr + row_off + col * stride_xc,
                mask=col_mask, other=0.0).to(tl.float32)

    # ---------- compute mean ----------
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / C

    # ---------- compute variance (mask padded positions) ----------
    diff = tl.where(col_mask, x - mean, 0.0)
    var = tl.sum(diff * diff, axis=0) / C

    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---------- normalise ----------
    y = (x - mean) * inv_std
    y = tl.where(col_mask, y, 0.0)

    # ---------- apply affine parameters ----------
    w = tl.load(weight_ptr + col * stride_w, mask=col_mask, other=0.0)
    b = tl.load(bias_ptr + col * stride_b,   mask=col_mask, other=0.0)
    y = y * w + b

    # ---------- write back ----------
    tl.store(out_ptr + pid * stride_om + col * stride_oc, y, mask=col_mask)


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm (over the last dimension) using the Triton kernel.
    Input/Output shape: (..., C)
    """
    assert x.is_contiguous()
    M = x.numel() // x.shape[-1]       # number of rows = product of leading dims
    C = x.shape[-1]

    # flatten to (M, C) for the kernel
    x_flat = x.view(M, C)
    out_flat = torch.empty_like(x_flat)

    # kernel launch configuration
    BLOCK_SIZE = 512                      # > max C (384)
    grid = (M,)

    layernorm_kernel[grid](
        x_flat,
        out_flat,
        weight,
        bias,
        eps,
        x_flat.stride(0), x_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1),
        weight.stride(0), bias.stride(0),
        C=C,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return out_flat.view_as(x)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation.
    Args:
        data: Tuple (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # config parameters
    dim = config["dim"]                     # C  (input/output feature dim)
    hidden_dim = config["hidden_dim"]       # H  (intermediate hidden dimension)
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # load model weights (all are assumed to be on the correct device)
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton)
    # ------------------------------------------------------------------
    x_norm = layernorm_triton(input_tensor, norm_weight, norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    # Shape after projection: [B, N, N, hidden_dim]
    left  = F.linear(x_norm, left_proj_weight)   # (B,N,N,H)
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1]
        mask_exp = mask.to(dtype).unsqueeze(-1)
        left  = left  * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Multiplicative update (batched matmul)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    → treat each (b, h) slice as a separate matrix.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) with row‑major layout for cuBLAS
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # FP16 GEMM (Tensor‑cores).  H100 works extremely well with FP16.
    left_fp16  = left_mat.to(torch.float16)
    right_fp16 = right_mat.to(torch.float16)

    # BMM with transposed right matrix (A @ Bᵀ)
    out_fp16 = torch.bmm(left_fp16, right_fp16.transpose(1, 2))

    # Back to FP32 and reshape to original layout
    out = out_fp16.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
