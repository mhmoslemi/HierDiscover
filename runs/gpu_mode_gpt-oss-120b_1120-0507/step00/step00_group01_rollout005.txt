def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Compute program ID indices for batch (b,d), block row(i), block col(j).
    pid = tl.program_id(0)
    # total batch*D dimension flatten
    batch_d = pid // (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_block = pid % (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    # Now determine block row and col within matrix.
    num_block_i = tl.cdiv(N, BLOCK_M)
    block_i = pid_block // num_block_i
    block_j = pid_block % num_block_i

    # Compute offsets
    i_offset = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Setup mask for out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Load pointers:
    # left: (i,k,d) where d is in this batch dimension index.
    # we need to isolate d index: For each batch_d, we need to compute b = batch_d // D, d = batch_d % D
    b = batch_d // D
    d = batch_d % D

    # compute base pointers for this b,d
    left_base = left_ptr + b * stride_l_batch + d * stride_l_d
    right_base = right_ptr + b * stride_r_batch + d * stride_r_d
    out_base = out_ptr + b * stride_o_batch + d * stride_o_d

    # Allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left tile (M x K)
        left_ptrs = left_base + i_offset[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Use right_ptr = right_base + j_offset[:, None] * stride_r_j + k_offset[None, :] * stride_r_k
        right_ptrs = right_base + j_offset[:, None] * stride_r_j + k_offset[None, :] * stride_r_k
        # shape (BLOCK_N, BLOCK_K)
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Compute multiplication: we need sum over k: left[i,k] * right[j,k]
        # So we need to compute outer product of left_tile and right_tile across k dimension.
        # We can do: for each k, acc[i,j] += left_tile[i,k] * right_tile[j,k]
        # Use tl.dot for each k? Not efficient.
        # Instead compute using broadcasting: left_tile (M,K), right_tile (N,K). We need sum over k of left_tile[:,k] * right_tile[:,k], where right_tile's rows correspond to N.
        # Compute elementwise multiplication of left_tile and right_tile with appropriate transposition.
        # Could perform left_tile * right_tile.T? Wait right_tile is (N,K). We need product left[i,k] * right[j,k], so we can multiply left_tile (M,K) with right_tile (N,K) after broadcasting across i and j dims: we want shape (M,N) = sum_k left_tile[i,k] * right_tile[j,k].
        # In Triton we can compute using:
        # acc += tl.sum(left_tile[:, None, :] * right_tile[None, :, :], axis=2)
        # However broadcasting will create a (M,N,K) tensor; can we sum over K efficiently? Might be memory heavy but M,N sizes small tile (like 32 each) and K tile (e.g., 32). This is manageable.

        # So:
        # prod = left_tile[:, None, :] * right_tile[None, :, :]   # shape (M,N,K)
        # acc += tl.sum(prod, axis=2)

        # Implement:
        prod = left_tile[:, None, :] * right_tile[None, :, :]  # broadcast
        acc += tl.sum(prod, axis=2)

    # Store result
    out_ptrs = out_base + i_offset[:, None] * stride_o_i + j_offset[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16) if out_ptr.dtype == tl.float16 else acc, mask=mask_i[:, None] & mask_j[None, :])

def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_l_batch, stride_l_d, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_d, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_d, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)

    # Compute offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Get mask for i and j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Compute batch and hidden_dim indices
    batch_idx = pid_batch // hidden_dim
    d_idx = pid_batch % hidden_dim

    # Compute base pointers offset to correct batch & channel
    left_ptrs_base = left_ptr + batch_idx * stride_l_batch + d_idx * stride_l_d
    right_ptrs_base = right_ptr + batch_idx * stride_r_batch + d_idx * stride_r_d
    out_ptrs_base = out_ptr + batch_idx * stride_o_batch + d_idx * stride_o_d

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left tile (M, K): left[i,k]
        left_ptrs = left_ptrs_base + i_offset[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

def trmul_outgoing_kernel(left_ptr, right_ptr, out_ptr,
                          B, N, hidden_dim,
                          stride_l_batch, stride_l_d, stride_l_i, stride_l_k,
                          stride_r_batch, stride_r_d, stride_r_j, stride_r_k,
                          stride_o_batch, stride_o_d, stride_o_i, stride_o_j,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    # compute batch index and hidden dim index
    batch_idx = pid_batch // hidden_dim
    d_idx = pid_batch % hidden_dim

    # base pointers for this batch & channel
    left_base = left_ptr + batch_idx * stride_l_batch + d_idx * stride_l_d
    right_base = right_ptr + batch_idx * stride_r_batch + d_idx * stride_r_d
    out_base = out_ptr + batch_idx * stride_o_batch + d_idx * stride_o_d

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # load left tile (M, K)
        left_ptrs = left_base + i[:, None] * stride_l_i + k_off[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile (K, N) -> transposed version
        right_ptrs = right_base + k_off[:, None] * stride_r_k + j[None, :] * stride_r_j
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # matmul
        acc += tl.dot(left_tile, right_tile)

    # store
    out_ptrs = out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if needed
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Permute for kernel
    left_reshape = left.permute(0,3,1,2).contiguous()   # (B, hidden_dim, N, N)
    right_reshape = right.permute(0,3,1,2).contiguous()
    out_reshape = torch.empty_like(left_reshape)

    B, hidden_dim_check, N, _ = left_reshape.shape
    assert hidden_dim_check == hidden_dim

    # Compute strides
    stride_l_batch = left_reshape.stride(0)
    stride_l_d = left_reshape.stride(1)
    stride_l_i = left_reshape.stride(2)
    stride_l_k = left_reshape.stride(3)

    stride_r_batch = right_reshape.stride(0)
    stride_r_d = right_reshape.stride(1)
    stride_r_j = right_reshape.stride(2)
    stride_r_k = right_reshape.stride(3)

    stride_o_batch = out_reshape.stride(0)
    stride_o_d = out_reshape.stride(1)

"""
TriMul (outgoing) kernel for AlphaFold3 style pairwise updates.

The forward pass consists of:
1. Layer‑norm over the last dimension of the input.
2. Two linear projections (left / right) + optional mask.
3. Gated projections using sigmoid‑gated linear layers.
4. The heavy O(N³) operation:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   implemented as a batched GEMM (B·D independent matrix multiplies)
   using a Triton kernel (no Python loops, fused loads / stores).
5. Layer‑norm + output‑gate + final linear projection.

All heavy arithmetic (step 4) is performed in the Triton kernel;
all other ops are cheap PyTorch calls.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trmul_outgoing_kernel(
    left_ptr,              # *Pointer* to left tensor   (B, D, N, N)
    right_ptr,             # *Pointer* to right tensor  (B, D, N, N)
    out_ptr,               # *Pointer* to output tensor (B, D, N, N)
    B, N, D,               # static problem sizes
    stride_l_batch,        # stride for batch dimension in left
    stride_l_d,            # stride for channel (hidden) dim in left
    stride_l_i,            # stride for i   dimension in left
    stride_l_k,            # stride for k   dimension in left
    stride_r_batch,        # stride for batch dimension in right
    stride_r_d,            # stride for channel dim in right
    stride_r_j,            # stride for j   dimension in right
    stride_r_k,            # stride for k   dimension in right
    stride_o_batch,        # stride for batch dimension in out
    stride_o_d,            # stride for channel dim in out
    stride_o_i,            # stride for i   dimension in out
    stride_o_j,            # stride for j   dimension in out
    BLOCK_M: tl.constexpr,  # tile size in i dimension
    BLOCK_N: tl.constexpr,  # tile size in j dimension
    BLOCK_K: tl.constexpr,  # tile size in k dimension
):
    """
    Compute a batched “outgoing” TriMul:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    The batch dimension is flattened (B*D independent GEMMs).
    """
    pid_i = tl.program_id(0)          # block index along the i axis
    pid_j = tl.program_id(1)          # block index along the j axis
    pid_batch = tl.program_id(2)      # flattened batch*channel index

    # ------------------------------
    #   Offsets for current tile
    # ------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # (BLOCK_M,)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # (BLOCK_N,)
    mask_i = i < N
    mask_j = j < N

    # decode flattened batch*channel index
    b = pid_batch // D      # original batch index
    d = pid_batch % D       # hidden‑channel index

    # base pointers that already point at the correct batch & channel
    left_base  = left_ptr  + b * stride_l_batch + d * stride_l_d
    right_base = right_ptr + b * stride_r_batch + d * stride_r_d
    out_base   = out_ptr   + b * stride_o_batch + d * stride_o_d

    # accumulator for the (BLOCK_M × BLOCK_N) sub‑matrix
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------
    #   Loop over the shared k axis
    # ------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)   # (BLOCK_K,)
        mask_k = k_off < N

        # left tile: shape (BLOCK_M, BLOCK_K)  – indices (i, k)
        left_ptrs = left_base + i[:, None] * stride_l_i + k_off[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N) – indices (k, j)
        right_ptrs = right_base + k_off[:, None] * stride_r_k + j[None, :] * stride_r_j
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # GEMM on the tile
        acc += tl.dot(left_tile, right_tile)   # (BLOCK_M, BLOCK_N)

    # ------------------------------
    #   Store the result
    # ------------------------------
    out_ptrs = out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Entry point used by the evaluation harness.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask:        torch.Tensor [B, N, N]   (may be None)
            - weights:     dict of torch.Tensors holding all model weights
            - config:      dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)           # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    #   Triton kernel:    out = Σₖ left[...,i,k,:] * right[...,j,k,:]
    # ------------------------------------------------------------------
    #   Prepare tensors in (B, D, N, N) layout for easy striding.
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, D, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, D, N, N]
    out_t = torch.empty_like(left_t)                  # placeholder for result

    B = left_t.shape[0]
    N = left_t.shape[2]  # N (seq_len)

    # strides in element units (torch gives element strides)
    stride_l_batch, stride_l_d, stride_l_i, stride_l_k = left_t.stride()
    stride_r_batch, stride_r_d, stride_r_j, stride_r_k = right_t.stride()
    stride_o_batch, stride_o_d, stride_o_i, stride_o_j = out_t.stride()

    # Tile sizes – tuned for H100 (fits in registers & shared memory)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # Grid: (num_i_tiles, num_j_tiles, B*hidden_dim)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trmul_outgoing_kernel[grid](
        left_t,
        right_t,
        out_t,
        B,
        N,
        hidden_dim,
        stride_l_batch,
        stride_l_d,
        stride_l_i,
        stride_l_k,
        stride_r_batch,
        stride_r_d,
        stride_r_j,
        stride_r_k,
        stride_o_batch,
        stride_o_d,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout: [B, N, N, hidden_dim]
    out = out_t.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # post‑processing: LayerNorm → out‑gate → final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final projection to `dim`

    return out
