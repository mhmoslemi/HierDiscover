"""
custom_kernel
==============

Implements the forward pass of the Triangular Multiplicative Update (TriMul)
operator (outgoing version) used in AlphaFold‑3 and related models.

Algorithm
---------
1. **LayerNorm** on the input ``x`` (dim‑last) using the supplied weight/bias.
2. **Linear projections** (left/right) from ``dim`` → ``hidden_dim``.
3. Optional element‑wise masking of the projected tensors.
4. **Gated projections** – sigmoid gates (left/right) are computed from ``x``
   and applied to the projected tensors.
5. **Core cubic operation** – for each batch element and each hidden channel
   we compute

        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

   which is equivalent to a batched matrix multiplication
   ``left @ rightᵀ`` per hidden dimension.
   This step is performed by a custom Triton kernel that
   loads tiles of size 64×64×64, accumulates in FP32 and writes the result
   as FP32.
6. **Post‑processing** – LayerNorm over the hidden dimension,
   element‑wise multiplication with the ``out_gate`` (computed from ``x``),
   and a final linear projection back to ``dim``.

Only the cubic accumulation (step 5) is written in Triton; the remaining
operations use PyTorch (FP16 where possible) to keep the implementation
compact while still attaining the required speed on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul with transposed right operand
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    batch_hidden, N,                    # runtime scalars
    stride_A_batch, stride_A_i, stride_A_k,
    stride_B_batch, stride_B_k, stride_B_j,
    stride_C_batch, stride_C_i, stride_C_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes, for each batch*hidden slice ``h``:

        C[h, i, j] = Σₖ A[h, i, k] * B[h, k, j]

    A shape : (batch_hidden, N, N)   (i, k)
    B shape : (batch_hidden, N, N)   (k, j)  <-- we treat the original
                                            shape (j, k) as transposed
    C shape : (batch_hidden, N, N)   (i, j)
    """
    pid_m = tl.program_id(0)               # block row   (i)
    pid_n = tl.program_id(1)               # block col   (j)
    pid_z = tl.program_id(2)               # batch*hidden index

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for the i/j dimensions (handle N not a multiple of block size)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the current batch*hidden slice
    A_batch_ptr = A_ptr + pid_z * stride_A_batch
    B_batch_ptr = B_ptr + pid_z * stride_B_batch
    C_batch_ptr = C_ptr + pid_z * stride_C_batch

    # Initialise accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles of size BLOCK_K
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for the k dimension
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load A tile:   A[h, i, k]   (shape BLOCK_M x BLOCK_K)
        # ------------------------------------------------------------------
        a_ptrs = (A_batch_ptr
                  + offs_m[:, None] * stride_A_i
                  + offs_k[None, :] * stride_A_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # ------------------------------------------------------------------
        # Load B tile:   B[h, k, j]   (shape BLOCK_K x BLOCK_N)
        # Note: the original right tensor is (j, k); we treat it as transposed.
        # ------------------------------------------------------------------
        b_ptrs = (B_batch_ptr
                  + offs_k[:, None] * stride_B_k
                  + offs_n[None, :] * stride_B_j)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Accumulate
        acc += tl.dot(a, b)      # fp16 * fp16 → fp32 accumulation

    # ------------------------------------------------------------------
    # Write result tile
    # ------------------------------------------------------------------
    c_ptrs = (C_batch_ptr
              + offs_m[:, None] * stride_C_i
              + offs_n[None, :] * stride_C_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True => no mask (ignore mask tensor)

    # ------------------------------------------------------------------
    # Extract weight tensors (they are stored as float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (over last dimension ``dim``)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy linear/gate path
    # ------------------------------------------------------------------
    x_h = x.half()   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (left/right)  dim → hidden_dim
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_weight.half())   # [B, N, N, hidden]
    right = F.linear(x_h, right_proj_weight.half())

    # ------------------------------------------------------------------
    # 3) Optional mask (mask shape [B, N, N])
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype=left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gated projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_weight.half()))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight.half()))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))   # [B,N,N,hidden]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul = batched matmul left @ rightᵀ per hidden channel
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) -> flatten (B*hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, N_check = left_perm.shape
    assert N == N_check, "Inconsistent sequence length."

    batch_hidden = B * H
    left_flat = left_perm.view(batch_hidden, N, N)      # [BH, N, N]
    right_flat = right_perm.view(batch_hidden, N, N)    # [BH, N, N]

    # Allocate output buffer (float32 accumulator)
    out_flat = torch.empty((batch_hidden, N, N), dtype=torch.float32, device=device)

    # Gather stride information (in element units)
    stride_A_batch = left_flat.stride(0)
    stride_A_i = left_flat.stride(1)
    stride_A_k = left_flat.stride(2)

    stride_B_batch = right_flat.stride(0)
    stride_B_k = right_flat.stride(1)   # k dimension (second)
    stride_B_j = right_flat.stride(2)   # j dimension (third)

    stride_C_batch = out_flat.stride(0)
    stride_C_i = out_flat.stride(1)
    stride_C_j = out_flat.stride(2)

    # Tunable block sizes – 64 works well on H100 for the problem sizes.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_hidden,
    )
    # Launch kernel
    tri_mul_kernel[grid](
        left_flat.data_ptr(),
        right_flat.data_ptr(),
        out_flat.data_ptr(),
        batch_hidden,
        N,
        stride_A_batch, stride_A_i, stride_A_k,
        stride_B_batch, stride_B_k, stride_B_j,
        stride_C_batch, stride_C_i, stride_C_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # Reshape back to [B, N, N, hidden]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6) Post‑processing
    # ------------------------------------------------------------------
    # LayerNorm over hidden dimension
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # Apply output gate (already fp16 → cast to fp32 for multiplication)
    out = out * out_gate.float()

    # Final linear projection back to ``dim`` (float32)
    out = F.linear(out, to_out_weight)

    return out
