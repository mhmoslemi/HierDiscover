@triton.jit
def triu_matmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_bn, stride_left_bi, stride_left_bk, stride_left_bc,
    stride_right_bn, stride_right_bi, stride_right_bk, stride_right_bc,
    stride_out_bn, stride_out_bi, stride_out_bj, stride_out_bc,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    num_pair = B * H

    # compute pair index (b,h)
    pair_id = pid // (num_m * num_n)
    block_id = pid % (num_m * num_n)
    block_m = block_id // num_n
    block_n = block_id % num_n

    # compute batch and hidden indices
    b = pair_id // H
    h = pair_id % H

    # compute block offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left block ptr: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr \
            + b * stride_left_bn \
            + h * stride_left_bc \
            + offs_m[:, None] * stride_left_bi \
            + k_offset[None, :] * stride_left_bk

        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right block ptr: we need right[j,k] transposed => shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr \
            + b * stride_right_bn \
            + h * stride_right_bc \
            + k_offset[:, None] * stride_right_bk \
            + offs_n[None, :] * stride_right_bi

        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # compute block multiplication: left (M x K) * right (K x N)
        acc += tl.dot(left, right)

    # write out
    out_ptrs = out_ptr \
        + b * stride_out_bn \
        + h * stride_out_bc \
        + offs_m[:, None] * stride_out_bi \
        + offs_n[None, :] * stride_out_bj

    tl.store(out_ptrs, acc.to(out_ptr.dtype), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel – Triton‑accelerated outgoing triangular multiplicative update (TriMul)

Algorithm
---------
1. Layer‑norm the input tensor over the last channel (dim).
2. Linear projections:
   * left  = Linear(x, left_proj.weight)      → [B,N,N,hidden]
   * right = Linear(x, right_proj.weight)    → [B,N,N,hidden]
   * left_gate  = sigmoid(Linear(x, left_gate.weight))
   * right_gate = sigmoid(Linear(x, right_gate.weight))
   * out_gate   = sigmoid(Linear(x, out_gate.weight))
3. Optional masking (mask shape [B,N,N]) is applied to *left* and *right*.
4. Gate the projections: left  = left  * left_gate,
                         right = right * right_gate.
5. Compute the outgoing TriMul:
   out[i,j] = Σ_k left[i,k]·right[j,k]   (batched matrix‑multiply for each
   hidden channel).  Implemented as a custom Triton kernel that performs
   B·hidden independent GEMMs of size N×N.
6. Apply a second LayerNorm, multiply by out_gate and a final linear
   projection back to the original channel dimension (dim).

Only the heavy N³ matrix multiplication (step 5) is performed with Triton;
all other operations use standard PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: per‑batch‑and‑hidden‑channel matrix multiplication
# ----------------------------------------------------------------------
@triton.jit
def _triu_matmul_outgoing_kernel(
    left_ptr,               # *float32   [B,N,N,hidden]
    right_ptr,              # *float32   [B,N,N,hidden]
    out_ptr,                # *float32   [B,N,N,hidden]
    B, N, H,                # int32
    stride_left_bn, stride_left_bi, stride_left_bk, stride_left_bc,   # int32
    stride_right_bn, stride_right_bi, stride_right_bk, stride_right_bc, # int32
    stride_out_bn, stride_out_bi, stride_out_bj, stride_out_bc,       # int32
    BLOCK_M: tl.constexpr,  # tile size along the i‑dimension
    BLOCK_N: tl.constexpr,  # tile size along the j‑dimension
    BLOCK_K: tl.constexpr   # tile size along the reduction k‑dimension
):
    """
    Compute, for every (batch, hidden) pair:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Interpreted as a batched GEMM:   (N×N) @ (N×N).T   for each hidden channel.
    """

    pid = tl.program_id(0)                     # 1‑D grid id
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    pair_cnt = B * H                            # number of (batch, hidden) pairs

    # decode program id into (batch, hidden, block_i, block_j)
    pair_id = pid // (num_m * num_n)            # which (b, h)
    block_id = pid % (num_m * num_n)            # which (i‑tile, j‑tile)
    pid_m = block_id // num_n                   # i‑tile index
    pid_n = block_id % num_n                    # j‑tile index

    b = pair_id // H
    h = pair_id % H

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k‑tiles
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- load left tile  (M × K) -------
        # left layout: (b, i, k, h)
        left_ptrs = left_ptr \
            + b * stride_left_bn \
            + h * stride_left_bc \
            + offs_i[:, None] * stride_left_bi \
            + offs_k[None, :] * stride_left_bk
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # ------- load right tile (K × N) -------
        # we need right[j,k] transposed → treat as (k, j)
        # right layout: (b, j, k, h)
        right_ptrs = right_ptr \
            + b * stride_right_bn \
            + h * stride_right_bc \
            + offs_k[:, None] * stride_right_bk \
            + offs_j[None, :] * stride_right_bi
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # matrix multiply the two tiles
        acc += tl.dot(left_tile, right_tile)   # (M×K)·(K×N) → (M×N)

    # ------- write result -------
    out_ptrs = out_ptr \
        + b * stride_out_bn \
        + h * stride_out_bc \
        + offs_i[:, None] * stride_out_bi \
        + offs_j[None, :] * stride_out_bj
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def _triu_matmul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel defined above.
    left/right: [B, N, N, H]  (contiguous, same dtype)
    Returns:    out    : [B, N, N, H]  (same dtype)
    """
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, H = left.shape

    out = torch.empty_like(left)

    # strides (in elements, not bytes)
    sL = left.stride()
    sR = right.stride()
    sO = out.stride()

    # Tile sizes – tuned for H100 FP32
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = B * H * num_m * num_n

    grid = (total_tiles,)

    _triu_matmul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        sL[0], sL[1], sL[2], sL[3],
        sR[0], sR[1], sR[2], sR[3],
        sO[0], sO[1], sO[2], sO[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Perform the outgoing TriMul operation using a Triton kernel for the
    N³ reduction step.

    Parameters
    ----------
    data : tuple
        (input, mask, weights, config)
        - input : torch.Tensor of shape [bs, seq_len, seq_len, dim]
        - mask  : torch.Tensor of shape [bs, seq_len, seq_len] (may be omitted)
        - weights : dict of tensors (see code for required keys)
        - config  : dict containing at least ``dim`` and ``hidden_dim``.
                  ``nomask`` (bool) indicates whether to ignore the mask.
                  ``layernorm_eps`` may be supplied; defaults to 1e‑5.

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] – the result of the
        outgoing triangular multiplicative update.
    """

    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2. Linear projections
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)          # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 3. Optional masking
    # --------------------------------------------------------------
    if not config.get("nomask", False):
        # mask: [B,N,N] → [B,N,N,1] broadcast over hidden dim
        mask = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Apply element‑wise gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triangular multiplicative update (outgoing) – Triton kernel
    # --------------------------------------------------------------
    # out[i,j] = Σ_k left[i,k] * right[j,k]   (per batch & hidden dim)
    out = _triu_matmul_outgoing(left, right)   # shape [B,N,N,hidden]

    # --------------------------------------------------------------
    # 6. Post‑norm, gate and final projection back to ``dim``
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate                     # gate after norm

    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)            # back to ``dim``

    return out
