"""
TriMul (outgoing) implementation for AlphaFold3 style pairwise updates.

The forward pass consists of:
1. Layer‑norm over the channel dimension.
2. Two linear projections (`left_proj`, `right_proj`) from dim → hidden_dim.
3. Optional elementwise mask (broadcasted on the hidden dimension).
4. Gating (sigmoid) on the projected tensors.
5. The heavy N³ operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batch of independent matrix‑multiplications per hidden
   channel.  The implementation launches a lightweight Triton kernel that
   computes one channel at a time (batched over the batch dimension) using a
   classic tiled GEMM with `tl.dot`.  Although it processes each channel
   separately, the kernel is fully GPU‑resident and avoids Python‑level loops.
6. Layer‑norm on the hidden dimension, gating, and a final linear projection
   back to the original dimension.

Only the N³ reduction is performed in Triton – the remaining operations stay
in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # scalars
    B, N,
    # strides for left (shape [B, N, N])
    stride_left_b, stride_left_i, stride_left_k,
    # strides for right (shape [B, N, N])
    stride_right_b, stride_right_i, stride_right_k,
    # strides for out (shape [B, N, N])
    stride_out_b, stride_out_i, stride_out_j,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication for a single hidden channel:
       out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k].
    """
    pid_b = tl.program_id(2)          # batch dimension
    pid_m = tl.program_id(0)          # i‑tile
    pid_n = tl.program_id(1)          # j‑tile

    # ----- tile offsets -------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the start of the current block (k = 0)
    a_ptrs = left_ptr  + pid_b * stride_left_b \
                      + offs_i[:, None] * stride_left_i \
                      + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    b_ptrs = right_ptr + pid_b * stride_right_b \
                      + tl.arange(0, BLOCK_K)[:, None] * stride_right_k \
                      + offs_j[None, :] * stride_right_i

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K dimension ----------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        # actual k offset for this iteration
        k_offset = k * BLOCK_K
        cur_k = tl.arange(0, BLOCK_K) + k_offset
        mask_k = cur_k < N                # valid k indices in this tile

        # masked loads (float32)
        a = tl.load(a_ptrs,
                     mask=mask_i[:, None] & mask_k[None, :],
                     other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                     mask=mask_k[:, None] & mask_j[None, :],
                     other=0.0).to(tl.float32)

        # GEMM core: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # ----- write back ----------------------------------------------------
    out_ptrs = out_ptr + pid_b * stride_out_b \
                         + offs_i[:, None] * stride_out_i \
                         + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of torch.Tensors with the model parameters
        - config       : dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Config & weight extraction
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 2. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 3. Linear projections (dim -> hidden_dim)
    # ------------------------------------------------------------------
    # Linear: out = x @ weight.T   (bias=False)
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", True):
        mask = mask.unsqueeze(-1).to(dtype)       # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid) on the original input and apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. TriMul core (N³ reduction) – Triton implementation
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)  # shape [B, N, N, hidden_dim]

    # Tiling parameters – these work well for the provided sizes on H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Loop over hidden channels; each launch processes a single channel.
    # This keeps the kernel simple and stays within register limits.
    for h in range(hidden_dim):
        # Views for a single hidden channel (shape [B, N, N])
        left_h = left[..., h]
        right_h = right[..., h]
        out_h = out[..., h]

        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            B,
        )
        trimul_matmul_kernel[grid](
            left_h,
            right_h,
            out_h,
            B,
            N,
            left_h.stride(0),
            left_h.stride(1),
            left_h.stride(2),
            right_h.stride(0),
            right_h.stride(1),
            right_h.stride(2),
            out_h.stride(0),
            out_h.stride(1),
            out_h.stride(2),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=4,
        )

    # ------------------------------------------------------------------
    # 7. Post‑processing: hidden‑dim layer norm, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B, N, N, hidden_dim]

    out = out * out_gate                      # gating
    output = F.linear(out, to_out_weight)     # final projection to dim

    return output
