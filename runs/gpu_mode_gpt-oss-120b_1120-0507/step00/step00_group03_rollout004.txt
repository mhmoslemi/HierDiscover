@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_ab, stride_ai, stride_ak, stride_ah,
    stride_bb, stride_bj, stride_bk, stride_bh,
    stride_cb, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decode batch and hidden
    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    # compute base pointers for this batch and hidden
    a_base = a_ptr + batch_idx * stride_ab + hidden_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_bb + hidden_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_cb + hidden_idx * stride_ch

    # compute offsets for i, j
    i_off = pid_m * BLOCK_M
    j_off = pid_n * BLOCK_N

    # create range vectors
    offs_i = i_off + tl.arange(0, BLOCK_M)
    offs_j = j_off + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute masks for i,j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_start = k * BLOCK_K
        # compute actual k offsets
        cur_k = k_start + offs_k

        # mask for k dimension (for load)
        mask_k = cur_k < N

        # pointers for A tile (i,k)
        a_ptrs = a_base + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak
        # load A tile
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

"""
TriMul outgoing implementation (AlphaFold‑3 style).

Algorithm
---------
1. Layer‑norm the input tensor x ∈ [B,N,N,C] (C = dim).
2. Project x to a hidden space (size H = hidden_dim) with two linear layers:
      left  = x @ W_left   (no bias)
      right = x @ W_right
3. Compute three gating scalars (sigmoid of linear projections of x):
      left_gate  = sigmoid(x @ W_left_gate)
      right_gate = sigmoid(x @ W_right_gate)
      out_gate   = sigmoid(x @ W_out_gate)
4. Apply the gates and an optional pair mask (mask ∈ [B,N,N]):
      left  = left  * left_gate  * mask[...,None]
      right = right * right_gate * mask[...,None]
5. **Heavy part – the 3‑tensor contraction**
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix‑multiply for every hidden slice h.
   A custom Triton kernel computes all B·H independent N×N products
   in a single launch (program id 0 = batch·hidden, id 1 = i‑block,
   id 2 = j‑block).  The kernel operates on FP16 inputs (to halve
   memory traffic) and accumulates in FP32 for numerical stability.
6. Layer‑norm the contraction result over the hidden dimension.
7. Multiply by the output gate and project back to the original channel
   dimension with a final linear layer.

Only the N³ contraction is performed on Triton; all surrounding
operations use PyTorch (they are cheap compared with the N²·H matmul).
The kernel works for any batch size B∈{1,2}, sequence length N up to 1024
and hidden dimensions H∈{128,384,768}.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # tensors: a=left, b=right, c=out (fp16, fp16, fp32)
    B, N, H,                           # runtime sizes
    stride_ab, stride_ai, stride_ak, stride_ah,   # a: (batch, i, k, h) strides
    stride_bb, stride_bj, stride_bk, stride_bh,   # b: (batch, j, k, h) strides
    stride_cb, stride_ci, stride_cj, stride_ch,   # c: (batch, i, j, h) strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute one hidden‑slice of the contraction:
        C_i_j = Σ_k A_i_k  * B_j_k   (B_j_k is accessed as B_T_k_j)
    The kernel is launched with a 3‑D grid:
        pid_bh = program_id(0)   → combined batch & hidden index
        pid_m  = program_id(1)   → i‑block
        pid_n  = program_id(2)   → j‑block
    """
    pid_bh = tl.program_id(0)   # 0 … B*H-1   (batch*hidden)
    pid_m  = tl.program_id(1)   # 0 … ceil(N/BLOCK_M)-1
    pid_n  = tl.program_id(2)   # 0 … ceil(N/BLOCK_N)-1

    # decode batch and hidden index
    batch_idx  = pid_bh // H
    hidden_idx = pid_bh %  H

    # starting offsets for the current tile
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    # range vectors
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # boundary masks for i and j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers that already include batch and hidden offsets
    a_base = a_ptr + batch_idx * stride_ab + hidden_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_bb + hidden_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_cb + hidden_idx * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(0, num_k):
        k_start = k_iter * BLOCK_K
        cur_k = k_start + offs_k
        mask_k = cur_k < N

        # ------------------- load A tile (i,k) -------------------
        # shape: (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)   # FP16 → FP32

        # ------------------- load B_T tile (k,j) -----------------
        # we load B as transposed: rows = k, cols = j
        # shape: (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + cur_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)   # FP16 → FP32

        # ------------------- block‑level dot ----------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # ------------------- store result ----------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nominal_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input (layer‑norm over the last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # x: [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Optional pair mask (shape [B,N,N] → [B,N,N,1])
    # ------------------------------------------------------------------
    if nominal_mask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left  * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # Allocate output for the contraction (float32)
    # ------------------------------------------------------------------
    out = torch.empty(
        (bs, seq_len, seq_len, hidden_dim),
        dtype=torch.float32,
        device=device,
    )

    # ------------------------------------------------------------------
    # Triton launch parameters (tuned for H100)
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows per block
    BLOCK_N = 64   # cols per block
    BLOCK_K = 32   # reduction block

    grid = (
        bs * hidden_dim,                           # program_id(0) → batch*hidden
        (seq_len + BLOCK_M - 1) // BLOCK_M,        # program_id(1) → i‑tiles
        (seq_len + BLOCK_N - 1) // BLOCK_N,        # program_id(2) → j‑tiles
    )

    # stride extraction (all tensors are contiguous in the last dim)
    stride_alb, stride_ali, stride_alk, stride_alh = left_f16.stride()
    stride_blb, stride_blj, stride_blk, stride_blh = right_f16.stride()
    stride_clb, stride_cli, stride_clj, stride_clh = out.stride()

    # launch kernel
    _tri_mul_matmul_kernel[
        grid
    ](
        left_f16,
        right_f16,
        out,
        bs,
        seq_len,
        hidden_dim,
        stride_alb, stride_ali, stride_alk, stride_alh,
        stride_blb, stride_blj, stride_blk, stride_blh,
        stride_clb, stride_cli, stride_clj, stride_clh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure the kernel finishes before we use `out` in later ops
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden dim, gate, final projection
    # ------------------------------------------------------------------
    # Layer‑norm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )   # still [B,N,N,hidden]

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])   # → [B,N,N,dim]

    return out
