def trimul_einsum_kernel( left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program_id axes
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2) # combine batch and hidden

    b = pid_bh // H
    h = pid_bh % H

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers for L and R slices:
    # left: ... i, k, h
    # right: ... j, k, h
    # stride dims:
    # left: (b,i,k,h)
    # right: (b,j,k,h)
    # out: (b,i,j,h)

    # offsets for k dimension range
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left block shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + offs_i[:,None]*stride_left_i + offs_k[None,:]*stride_left_k

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_einsum_kernel(left_ptr, right_ptr, out_ptr,
                         B, N, H,
                         stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                         stride_right_b, stride_right_j, stride_right_k, stride_right_h,
                         stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)
    
    b = pid_bh // H
    h = pid_bh % H
    
    # Offsets in i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # Base pointer offsets for batch and hidden dim
    base_left = left_ptr + b * stride_left_b + h * stride_left_h
    base_right = right_ptr + b * stride_right_b + h * stride_right_h
    
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N
        
        # Load left block (M,K)
        ptrs_left = base_left + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(ptrs_left, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        
        # Load right block transposed (K,N)
        ptrs_right = base_right + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right = tl.load(ptrs_right, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        
        acc += tl.dot(left, right)
    
    # Store the result
    ptrs_out = out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(ptrs_out, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel – Triton‑accelerated outgoing TriMul

The core of the TriMul module is the tensor contraction

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

for a batch of size B, a sequence length N, and a hidden dimension H.
The operation is a batched matrix‑multiply over the third axis (k) for
each hidden channel h.  A GEMM‑style Triton kernel tiles the i, j and k
dimensions (BLOCK_M, BLOCK_N, BLOCK_K) and accumulates in FP32 while
loading the inputs in FP16 – this gives a large reduction in memory
traffic compared with the naïve Python loops.

All surrounding linear projections, gates and LayerNorms are performed
with PyTorch (using FP16 where possible).  The kernel implements only
the O(N³) part of the forward pass, satisfying the requirement to have
“part of the operations in Triton”.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_einsum_kernel(
    left_ptr,          # *fp16
    right_ptr,         # *fp16
    out_ptr,           # *fp16
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b,   stride_out_i,   stride_out_j,   stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Tile‑wise kernel for   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_i = tl.program_id(0)        # block row   (i)
    pid_j = tl.program_id(1)        # block col   (j)
    pid_bh = tl.program_id(2)       # combined batch‑hidden index

    b = pid_bh // H                 # batch index
    h = pid_bh % H                  # hidden‑channel index

    # i / j coordinates for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for a given (b, h)
    base_left  = left_ptr  + b * stride_left_b  + h * stride_left_h
    base_right = right_ptr + b * stride_right_b + h * stride_right_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k, h]   – shape (M, K)
        ptrs_left = base_left + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(ptrs_left,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right[j, k, h] transposed to (K, N) so that dot works
        # (we load as rows = k, cols = j)
        ptrs_right = base_right + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right = tl.load(ptrs_right,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # accumulate Σ_k left * right  (dot: (M,K) @ (K,N) → (M,N))
        acc += tl.dot(left, right)

    # write result  out[b,i,j,h]
    ptrs_out = out_ptr + b * stride_out_b + h * stride_out_h \
               + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(ptrs_out,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _trimul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    left/right:  [B, N, N, H]  (FP16, contiguous)
    Returns out of same shape & dtype.
    """
    assert left.shape == right.shape, "left/right must have identical shapes"
    B, N, _, H = left.shape
    out = torch.empty_like(left)          # FP16 output

    # strides in element units (not bytes)
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,     # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,     # j‑tiles
        B * H                             # one program per (batch, hidden)
    )

    _trimul_einsum_kernel[grid](
        left, right, out,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for
    the cubic contraction.  The function expects:

        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------
    #   Extract configuration and weights
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # LayerNorm parameters
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # Linear weights (FP16 for the heavy work)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)
    to_out_w     = weights["to_out.weight"]            # keep FP32 for final proj
    # Output LayerNorm
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]

    # ------------------------------------------------------------------
    #   1) Input LayerNorm (FP32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    #   2) Project to hidden dimension (FP16)
    # ------------------------------------------------------------------
    x_half = x.to(torch.float16)                 # [B, N, N, dim] -> FP16
    left  = F.linear(x_half, left_proj_w)       # [B, N, N, hidden_dim]
    right = F.linear(x_half, right_proj_w)

    # ------------------------------------------------------------------
    #   3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        mask = mask_tensor.unsqueeze(-1).to(x_half.dtype)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    #   4) Gating (sigmoid) – still in FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_half, left_gate_w)).to(x_half.dtype)
    right_gate = torch.sigmoid(F.linear(x_half, right_gate_w)).to(x_half.dtype)
    out_gate = torch.sigmoid(F.linear(x_half, out_gate_w)).to(x_half.dtype)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    #   5) Core cubic contraction via Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _trimul_einsum(left, right)           # FP16, shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    #   6) Output LayerNorm, gating and final projection (FP32)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)                 # switch to FP32 for LayerNorm
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate.to(out.dtype)          # apply output gate

    # final linear back to original dimension (FP32)
    out = F.linear(out, to_out_w)               # [B, N, N, dim]

    return out.to(torch.float32)
