@triton.jit
def layernorm_kernel(out_ptr, inp_ptr, weight_ptr, bias_ptr,
                     stride_ib, stride_ii, stride_ij, stride_ic,
                     stride_ob, stride_oi, stride_oj, stride_oc,
                     B, N, C,
                     eps: tl.constexpr,
                     BLOCK_C: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # compute base offset for input and out.
    inp_base = b * stride_ib + i * stride_ii + j * stride_ij
    out_base = b * stride_ob + i * stride_oi + j * stride_oj

    # reduction loop
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)
    c = 0
    while c < C:
        cur_block = tl.minimum(BLOCK_C, C - c)
        offs_c = tl.arange(0, BLOCK_C)
        mask = offs_c < cur_block
        inp = tl.load(inp_ptr + inp_base + (c + offs_c) * stride_ic,
                      mask=mask,
                      other=0.0)
        sum_val += tl.sum(inp)
        sumsq_val += tl.sum(inp * inp)
        c += BLOCK_C

    mean = sum_val / C
    var = sumsq_val / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # second pass to write normalized values
    c = 0
    while c < C:
        cur_block = tl.minimum(BLOCK_C, C - c)
        offs_c = tl.arange(0, BLOCK_C)
        mask = offs_c < cur_block
        x = tl.load(inp_ptr + inp_base + (c + offs_c) * stride_ic,
                    mask=mask,
                    other=0.0)
        w = tl.load(weight_ptr + c + offs_c, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + c + offs_c, mask=mask, other=0.0)
        y = (x - mean) * inv_std * w + b_
        tl.store(out_ptr + out_base + (c + offs_c) * stride_oc, y, mask=mask)
        c += BLOCK_C
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The kernel implements the heavy LayerNorm (normalisation across the last
dimension) in Triton while the O(N³) “pairwise” update is performed with a
batched GEMM (`torch.bmm`) which already uses highly tuned CUTLASS kernels.
The algorithm follows the reference implementation:

1. LayerNorm on the input tensor (shape [B, N, N, C]).
2. Linear projections → left / right.
3. Optional mask applied to the projections.
4. Gating (sigmoid of three linear projections) applied to left/right/out.
5. Pairwise update:  out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
   which can be expressed as a batch of H matrix‑multiplications
   (H = hidden_dim):   out_h = left_h @ right_hᵀ.
6. LayerNorm on the pairwise result, gating, and final linear projection.

Only the LayerNorms are written in Triton; all other ops use PyTorch for
simplicity and to guarantee numerical parity with the reference code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for LayerNorm over the last dimension of a 4‑D tensor.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    out_ptr,                     # *mut f32
    inp_ptr,                     # *const f32
    weight_ptr,                  # *const f32   (γ)
    bias_ptr,                    # *const f32   (β)
    stride_ib, stride_ii, stride_ij, stride_ic,   # int64 strides for input
    stride_ob, stride_oi, stride_oj, stride_oc,   # int64 strides for output
    B, N, C,                                            # compile‑time ints
    eps: tl.constexpr,                                   # compile‑time scalar
    BLOCK_C: tl.constexpr,                               # compile‑time tile size
):
    # one program instance per (b, i, j) coordinate
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base offsets for the 3‑D index (b,i,j)
    inp_base = b * stride_ib + i * stride_ii + j * stride_ij
    out_base = b * stride_ob + i * stride_oi + j * stride_oj

    # ------------------------------------------------------------------
    # First reduction pass: compute mean and variance.
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sumsq_val = tl.float32(0.0)

    num_blocks = tl.cdiv(C, BLOCK_C)        # compile‑time constant
    for block in tl.static_range(num_blocks):
        offs = tl.arange(0, BLOCK_C) + block * BLOCK_C
        mask = offs < C
        x = tl.load(inp_ptr + inp_base + offs * stride_ic, mask=mask, other=0.0)
        sum_val += tl.sum(x)
        sumsq_val += tl.sum(x * x)

    mean = sum_val / C
    var = sumsq_val / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write the normalized values.
    # ------------------------------------------------------------------
    for block in tl.static_range(num_blocks):
        offs = tl.arange(0, BLOCK_C) + block * BLOCK_C
        mask = offs < C
        x = tl.load(inp_ptr + inp_base + offs * stride_ic, mask=mask, other=0.0)
        w = tl.load(weight_ptr + offs, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + offs, mask=mask, other=0.0)

        y = (x - mean) * inv_std * w + b_
        tl.store(out_ptr + out_base + offs * stride_oc, y, mask=mask)


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     block_c: int = 64) -> torch.Tensor:
    """
    Triton‑based LayerNorm.
    x    : [B, N, N, C]    (float32)
    weight, bias : [C]   (float32)
    Returns a tensor of the same shape as ``x``.
    """
    B, N, N2, C = x.shape
    assert N == N2, "Tensor must be square in the two middle dimensions"
    out = torch.empty_like(x)

    # strides in element units (Triton works with element offsets)
    stride_ib, stride_ii, stride_ij, stride_ic = x.stride()
    stride_ob, stride_oi, stride_oj, stride_oc = out.stride()

    total = B * N * N
    grid = (total,)

    # launch kernel
    layernorm_kernel[grid](
        out,
        x,
        weight,
        bias,
        stride_ib, stride_ii, stride_ij, stride_ic,
        stride_ob, stride_oi, stride_oj, stride_oc,
        B, N, C,
        eps=eps,
        BLOCK_C=block_c,
    )
    return out


# ----------------------------------------------------------------------
# Custom TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]
        - mask         : torch.Tensor  [B, N, N]   (may be None)
        - weights      : dict of torch.Tensors containing all linear &
                         layer‑norm parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` and
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm  (via Triton)
    # ------------------------------------------------------------------
    x = layernorm_triton(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
        block_c=64,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])        # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])      # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast along the hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask is [B,N,N]; we need [B,N,N,1] for broadcasting
        m = mask.to(dtype=dtype).unsqueeze(-1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gating signals (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Pairwise update: batched GEMM across the hidden dimension
    #    The operation is equivalent to:
    #        out_h = left_h @ right_hᵀ   for each head h ∈ [hidden_dim]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*hidden, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # batched matrix multiplication
    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(2, 1))

    # reshape back to [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (via Triton)
    # ------------------------------------------------------------------
    out = layernorm_triton(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
        block_c=64,
    )

    # ------------------------------------------------------------------
    # 7. Final gating and linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
