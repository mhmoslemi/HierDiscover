@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid = tl.program_id(0) # i tiles
    pid_j = tl.program_id(1) # j tiles
    pid_bd = tl.program_id(2) # combined batch and hidden idx

    # Extract batch idx and hidden dim idx
    b = pid_bd // H
    h = pid_bd % H

    # compute offsets
    offs_i = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers for left: shape [B,N,N,H]
    # left[b, i, k, h]
    left_ptrs = left_ptr + b * stride_bl + h * stride_hl + offs_i[:, None] * stride_il + offs_k[None, :] * stride_kl

    # For right, we need right[b, j, k, h] where we later use j as row and k as col.
    # Right is not transposed, we need to load right[b, j, k, h] and multiply with left[i,k,h].
    # It matches dims: left (i,k) * right (j,k). So we load right similarly.
    right_ptrs = right_ptr + b * stride_br + h * stride_hr + offs_j[:, None] * stride_jr + offs_k[None, :] * stride_kr

    # Init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for k_start in range(0, N, BLOCK_K):
        # compute current k offset
        cur_k = k_start + offs_k
        k_mask = cur_k < N
        # load left tile
        left_tile = tl.load(left_ptrs + k_start * stride_kl, mask=mask_i[:, None] & k_mask[None, :], other=0.0).to(tl.float32)
        # load right tile
        right_tile = tl.load(right_ptrs + k_start * stride_kr, mask=mask_j[:, None] & k_mask[None, :], other=0.0).to(tl.float32)
        # multiply and accumulate using dot product across K dimension
        # Compute outer product: left_tile[:, k] * right_tile[:, k] (with broadcasting)
        # Efficiently compute using tl.dot? We can compute sum over k of left_tile[i,k] * right_tile[j,k]
        # This is essentially a matrix multiplication: left_tile * right_tile.T
        # So we can do:
        acc += tl.dot(left_tile, right_tile, trans_b=True) # left MxK, right NxK (transposed) -> MxN

    # Store result to out
    out_ptrs = out_ptr + b * stride_bo + h * stride_ho + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) – fast Triton implementation.

The kernel computes, for every batch b and hidden channel h,
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where `left` and `right` are already masked and gated.
The operation is a batched matrix multiplication of
    left[..., :, :, h] (N×N) with right[..., :, :, h].T.
The kernel tiles the i‑ and j‑ dimensions (BLOCK_M × BLOCK_N) and
processes the reduction over k in tiles of size BLOCK_K.
All arithmetic is performed in float32 while the tensors are stored
in float16 to reduce memory traffic.  The rest of the TriMul block
(LayerNorms, linear projections, gating and the final output projection)
are carried out with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: per‑batch‑and‑channel matrix multiply (left @ right.T)
# ----------------------------------------------------------------------
@triton.jit
def triton_tri_mul_kernel(
    left_ptr,          # [B, N, N, H]  (fp16)
    right_ptr,         # [B, N, N, H]  (fp16)
    out_ptr,           # [B, N, N, H]  (fp16) – result
    B, N, H,           # runtime sizes
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = left @ right.T for every (b, h) pair."""
    pid_i = tl.program_id(0)               # tile row
    pid_j = tl.program_id(1)               # tile col
    pid_bd = tl.program_id(2)              # combined batch & hidden index

    # ------------------------------------------------------------------
    # Decode batch and hidden‑channel from pid_bd
    # ------------------------------------------------------------------
    b = pid_bd // H
    h = pid_bd % H

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the current batch & hidden channel
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        cur_k = k_start + tl.arange(0, BLOCK_K)          # shape (BLOCK_K,)
        mask_k = cur_k < N

        # Pointers to the current k‑slice
        left_tile_ptr  = left_base \
                         + offs_i[:, None] * stride_li \
                         + cur_k[None, :] * stride_lk
        right_tile_ptr = right_base \
                         + offs_j[:, None] * stride_rj \
                         + cur_k[None, :] * stride_rk

        # Load tiles (fp16 -> float32 for accumulation)
        left_tile = tl.load(left_tile_ptr,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)
        right_tile = tl.load(right_tile_ptr,
                             mask=mask_j[:, None] & mask_k[None, :],
                             other=0.0).to(tl.float32)

        # Perform the dot product: left_tile (M×K) * right_tile.T (K×N)
        acc += tl.dot(left_tile, right_tile, trans_b=True)

    # ------------------------------------------------------------------
    # Store the result (cast back to fp16)
    # ------------------------------------------------------------------
    out_tile_ptr = out_base \
                   + offs_i[:, None] * stride_oi \
                   + offs_j[None, :] * stride_oj
    tl.store(out_tile_ptr,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Python entry point – reproduces the forward pass of the reference module
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: Tensor[B, N, N, dim]  (float32, on CUDA)
            - mask:        Tensor[B, N, N]        (bool/float, optional)
            - weights:     dict of weight & bias tensors
            - config:      dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        Tensor[B, N, N, dim] – the TriMul forward result (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # same shape

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if (mask is not None) and (not nomask):
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to fp16 and launch Triton kernel for the heavy N³ reduction
    # ------------------------------------------------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, _ = left_f16.shape  # (B, N, N, hidden_dim)

    # Destination tensor for the reduction
    out_f16 = torch.empty_like(left_f16)

    # Strides (in elements) – needed by Triton
    stride_lb, stride_li, stride_lk, stride_lh = left_f16.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right_f16.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_f16.stride()

    # Tiling parameters – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,                 # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,                 # j‑tiles
        B * hidden_dim,                               # (batch, hidden) combos
    )

    # Launch kernel
    triton_tri_mul_kernel[grid](
        left_f16, right_f16, out_f16,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,  # H100 performs well with 8 warps per block
    )

    # ------------------------------------------------------------------
    # 6. Convert back to FP32 for the remaining LayerNorm / gating / projection
    # ------------------------------------------------------------------
    out = out_f16.to(torch.float32)  # [B, N, N, hidden_dim]

    # 7. Final LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # 8. Apply the output gate
    out = out * out_gate

    # 9. Final linear projection back to `dim`
    output = F.linear(out, weights["to_out.weight"])   # shape [B, N, N, dim]

    return output
