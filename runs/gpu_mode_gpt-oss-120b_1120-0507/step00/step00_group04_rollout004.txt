"""
TriMul (outgoing) forward pass fused with Triton.

The heavy O(N³·H) reduction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is performed by a custom Triton kernel.  The surrounding
LayerNorms, Linear projections and gating are kept in PyTorch for
convenience.  The kernel tiles the output over (i,j) and a tile of the
hidden dimension (h) while iterating over k‑tiles, using a warp‑level
matrix multiply (tl.dot) for each hidden slice.  Optional masking
simply zero‑out the projected left/right tensors before the reduction.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel for the triple‑product reduction
# ------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,           # pointers
    B, N, H,                                # batch, seq_len, hidden
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Each program instance works on a tile (i0:i0+BLOCK_I, j0:j0+BLOCK_J)
    for a specific batch and a slice of the hidden dimension.
    """
    # ------------------------------
    # block indices
    # ------------------------------
    pid = tl.program_id(axis=2)               # fused (batch, hidden_tile)
    batch_id   = pid % B                       # batch index
    hid_tile   = pid // B                      # which hidden‑tile we are processing

    i_tile = tl.program_id(axis=0)
    j_tile = tl.program_id(axis=1)

    i0 = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j0 = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
    k0 = tl.arange(0, BLOCK_K)                # will be shifted inside the loop

    h0 = hid_tile * BLOCK_H + tl.arange(0, BLOCK_H)

    # bounds
    mask_i = i0 < N
    mask_j = j0 < N
    mask_h = h0 < H

    # accumulator for the output tile (i,j,h)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # number of k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    # ------------------------------
    # loop over k tiles
    # ------------------------------
    for k_tile in range(num_k_tiles):
        k = k_tile * BLOCK_K + k0               # global k positions
        mask_k = k < N

        # ------------ load left -------------------------------------------------
        left_ptrs = (
            left_ptr
            + batch_id * stride_l_b
            + (i0[:, None, None] * stride_l_i)
            + (k[None, :, None] * stride_l_k)
            + (h0[None, None, :] * stride_l_h)
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ------------ load right ------------------------------------------------
        right_ptrs = (
            right_ptr
            + batch_id * stride_r_b
            + (j0[:, None, None] * stride_r_i)
            + (k[None, :, None] * stride_r_k)
            + (h0[None, None, :] * stride_r_h)
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ------------ multiply‑accumulate over k for each hidden slice -----------
        # unroll over the hidden tile (static since BLOCK_H is a compile‑time const)
        for h in tl.static_range(0, BLOCK_H):
            # left_h : (BLOCK_I, BLOCK_K), right_h : (BLOCK_J, BLOCK_K)
            left_h  = left[:, :, h]
            right_h = right[:, :, h]
            # left_h @ right_hᵀ  → (BLOCK_I, BLOCK_J)
            acc[:, :, h] += tl.dot(left_h, right_h, trans_b=True)

    # ------------------------------
    # write the result
    # ------------------------------
    out_ptrs = (
        out_ptr
        + batch_id * stride_o_b
        + (i0[:, None, None] * stride_o_i)
        + (j0[None, :, None] * stride_o_j)
        + (h0[None, None, :] * stride_o_h)
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# ------------------------------------------------------------
# Python entry‑point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel
    for the core reduction.  Arguments are packed into a tuple:

        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on the same
        device as the inputs).
    """
    input_tensor, mask_tensor, weights, config = data

    # ------------------------------------------------------------------
    # configuration
    # ------------------------------------------------------------------
    dim        = config["dim"]          # input & output channel size
    hidden_dim = config["hidden_dim"]   # size of the intermediate projection
    no_mask    = config.get("nomask", True)

    # ------------------------------------------------------------------
    # unpack weights (all stored as float32, no bias unless explicitly given)
    # ------------------------------------------------------------------
    norm_w          = weights["norm.weight"]
    norm_b          = weights["norm.bias"]
    left_proj_w     = weights["left_proj.weight"]
    right_proj_w    = weights["right_proj.weight"]
    left_gate_w     = weights["left_gate.weight"]
    right_gate_w    = weights["right_gate.weight"]
    out_gate_w      = weights["out_gate.weight"]
    to_out_norm_w   = weights["to_out_norm.weight"]
    to_out_norm_b   = weights["to_out_norm.bias"]
    to_out_w        = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)          # (B,N,N,hidden)
    right = F.linear(x, right_proj_w)         # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not no_mask and mask_tensor is not None:
        mask = mask_tensor.to(dtype=x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core reduction via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # make sure the tensors are contiguous for simple stride logic
    left  = left.contiguous()
    right = right.contiguous()
    out   = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=left.device)

    # strides (in elements) of the 4‑D tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tile sizes – chosen to give good occupancy on an H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 8   # hidden dimension is processed in slices of 8

    grid_i = (N + BLOCK_I - 1) // BLOCK_I
    grid_j = (N + BLOCK_J - 1) // BLOCK_J
    grid_h = (hidden_dim + BLOCK_H - 1) // BLOCK_H   # number of hidden tiles
    grid = (grid_i, grid_j, B * grid_h)             # fuse batch & hidden‑tile in axis‑2

    _trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7. Output layer‑norm, gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    output = F.linear(out, to_out_w)   # (B,N,N,dim)

    return output
