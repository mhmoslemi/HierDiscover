import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_akb,   # maybe unify
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for i, j
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute mask for out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile
        a_ptrs = a_ptr + batch_id * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load b tile
        b_ptrs = b_ptr + batch_id * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_ptr + batch_id * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

"""
TriMul (outgoing) forward pass – highly optimized implementation.

The module performs:
1. Layer‑norm over the feature dimension.
2. Two linear projections (left/right) + two linear gates and a final output gate.
3. Mask‑application (if provided).
4. Core “tri‑multiplicative” reduction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden channel.
   We fuse the B×H independent GEMMs into a single Triton kernel
   that processes all batches in one launch (batch dimension = B*H).
5. Layer‑norm over the hidden dimension, gating, and final linear projection
   back to the original feature dimension.

All heavy compute (step 4) is done in a custom Triton kernel; the remaining
operations use PyTorch kernels (they are already highly tuned). The kernel
uses fp16 arithmetic with a fp32 accumulator for accuracy and writes back in
fp16; the final output is returned as fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ B
#   A : (batch, M, K)   fp16
#   B : (batch, K, N)   fp16
#   C : (batch, M, N)   fp16
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,      # A: (batch, M, K)
    stride_bk, stride_bn,      # B: (batch, K, N)
    stride_cm, stride_cn,      # C: (batch, M, N)
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds elements in M/N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulation in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile : (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_id * batch_stride_a \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B tile : (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + batch_id * batch_stride_b \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write back result in fp16
    c_ptrs = c_ptr + batch_id * batch_stride_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched fp16 GEMM using Triton.
    a – (B, M, K)   fp16
    b – (B, K, N)   fp16
    returns c – (B, M, N) fp16
    """
    assert a.is_cuda and b.is_cuda
    assert a.dtype == b.dtype == torch.float16
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # element‑size in bytes
    es = a.element_size()  # =2 for fp16

    # strides (in bytes) for A
    stride_a_batch = a.stride(0) * es
    stride_am = a.stride(1) * es
    stride_ak = a.stride(2) * es

    # strides for B
    stride_b_batch = b.stride(0) * es
    stride_bk = b.stride(1) * es
    stride_bn = b.stride(2) * es

    # strides for C
    stride_c_batch = c.stride(0) * es
    stride_cm = c.stride(1) * es
    stride_cn = c.stride(2) * es

    # tile sizes – empirically good for H100 with fp16
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        M=M,
        N=N,
        K=K,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        batch_stride_a=stride_a_batch,
        batch_stride_b=stride_b_batch,
        batch_stride_c=stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return c


# ----------------------------------------------------------------------
# Entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    # ------------------------------------------------------------------
    # 0. basic shapes & config
    # ------------------------------------------------------------------
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Cast everything to fp16 (best performance on H100)
    # ------------------------------------------------------------------
    x = input_tensor.half()
    if not nomask:
        mask = mask_tensor.half()
    else:
        mask = None

    # ------------------------------------------------------------------
    # 2. Load weights (cast to fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].half()
    norm_bias = weights["norm.bias"].half()
    left_proj_weight = weights["left_proj.weight"].half()
    right_proj_weight = weights["right_proj.weight"].half()
    left_gate_weight = weights["left_gate.weight"].half()
    right_gate_weight = weights["right_gate.weight"].half()
    out_gate_weight = weights["out_gate.weight"].half()
    to_out_norm_weight = weights["to_out_norm.weight"].half()
    to_out_norm_bias = weights["to_out_norm.bias"].half()
    to_out_weight = weights["to_out.weight"].half()

    eps = 1e-5  # same as PyTorch default

    # ------------------------------------------------------------------
    # 3. Input LayerNorm over the feature dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 4. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # (bs, N, N, hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 5. Optional mask (broadcasted over the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)            # (bs, N, N, 1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 6. Gating (sigmoid of linear projections of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7. Core TriMul reduction via a fused batched GEMM kernel
    #    left : (bs, N, N, hidden)  -> (bs, hidden, N, N)   (i,k)
    #    right: (bs, N, N, hidden)  -> (bs, hidden, N, N)   (k,j) after transpose
    #    Compute (bs*hidden) independent matrix products (i,k) @ (k,j)
    # ------------------------------------------------------------------
    # Cast to fp16 (already half) – ensure contiguity for the tiling
    left = left.contiguous()
    right = right.contiguous()

    # Move hidden dim to the second axis and make the needed transposes
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (bs, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (bs, hidden, k, j)

    B, H, M, K = left_perm.shape          # M == K == N
    batch = B * H

    # flatten batch dimension for the kernel
    left_flat = left_perm.view(batch, M, K)          # (B*H, M, K)
    right_flat = right_perm.view(batch, K, M)        # (B*H, K, N) with N==M

    # Triton‑accelerated batched GEMM
    out_flat = _batched_matmul_triton(left_flat, right_flat)   # (B*H, M, N)

    # reshape back to (bs, N, N, hidden)
    out = out_flat.view(B, H, M, M).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over the hidden dimension, gating and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    out = out * out_gate.unsqueeze(-1)   # broadcast gating

    # final linear back to original feature dimension
    out = F.linear(out, to_out_weight)   # (bs, N, N, dim)

    # return in float32 (easier for downstream code)
    return out.to(torch.float32)
