def trimul_outgoing_kernel(left_ptr, right_ptr, out_ptr,
                           B, N, H,
                           stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                           stride_right_b, stride_right_j, stride_right_k, stride_right_h,
                           stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                           BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr):

    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    b = pid_b

    # compute mask for out-of-bounds dims
    off_i = i0 + tl.arange(0, BLOCK_I)
    off_j = j0 + tl.arange(0, BLOCK_J)
    off_h = tl.arange(0, BLOCK_H)

    # allocate accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k blocks
    for k0 in range(0, N, BLOCK_K):
        # compute offset ranges for k
        off_k = k0 + tl.arange(0, BLOCK_K)

        # mask for loading
        mask_i = off_i < N
        mask_j = off_j < N
        mask_k = off_k < N
        mask_h = off_h < H

        # Compute pointers for left and right
        # left shape (I, K, H)
        left_ptrs = left_ptr + b * stride_left_b \
                    + off_i[:, None, None] * stride_left_i \
                    + off_k[None, :, None] * stride_left_k \
                    + off_h[None, None, :] * stride_left_h

        # right shape (J, K, H)
        right_ptrs = right_ptr + b * stride_right_b \
                     + off_j[:, None, None] * stride_right_j \
                     + off_k[None, :, None] * stride_right_k \
                     + off_h[None, None, :] * stride_right_h

        left = tl.load(left_ptrs, mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)
        right = tl.load(right_ptrs, mask= mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)

        # compute contribution: sum_{k} left[i,k,h] * right[j,k,h]
        # Broadcast left over j dimension and right over i dimension:
        # left_expand shape (I, 1, K, H)
        # right_expand shape (1, J, K, H)
        # multiply and sum over K axis (axis=2)
        # Since we cannot directly sum axis with tl, we loop over K dimension inside tile:
        # But we can compute using matmul across K dimension
        # Let's compute outer product via inner loop over K: accumulate across K
        # For each offset within BLOCK_K:
        for kk in range(0, BLOCK_K):
            mask_kk = off_k[kk] < N
            # We can load left slice and right slice for this k index using partial pointers, but can't do dynamic index from Python; we can slice:
            # left_slice = left[:, kk, :] shape (I,H)
            # right_slice = right[:, kk, :] shape (J,H)
            left_slice = left[:, kk, :]  # (I,H)
            right_slice = right[:, kk, :]  # (J,H)
            # Expand dims
            # left_expand: (I,1,H), right_expand: (1,J,H)
            left_exp = left_slice[:, None, :]  # shape (I,1,H)
            right_exp = right_slice[None, :, :]  # shape (1,J,H)
            # Multiply and accumulate
            acc += left_exp * right_exp

    # after K loop, store accumulator to output
    out_ptrs = out_ptr + b * stride_out_b \
               + off_i[:, None, None] * stride_out_i \
               + off_j[None, :, None] * stride_out_j \
               + off_h[None, None, :] * stride_out_h

    # store with mask for bounds
    mask_store = (off_i[:, None, None] < N) & (off_j[None, :, None] < N) & (off_h[None, None, :] < H)
    tl.store(out_ptrs, acc, mask=mask_store)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape B,N,N,H
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if needed
    if not nomask:
        # mask shape (B,N,N)
        mask_ = mask.unsqueeze(-1).type_as(x)  # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous and proper dtype
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape  # shape (B,N,N,hidden_dim)

    # Allocate output
    out = torch.empty_like(left)  # (B,N,N,hidden_dim)

    # Compute strides
    stride_left = left.stride()

    stride_out = out.stride()
    # left: (b,i,k,h)

    # right: (b,j,k,h) - same layout (b,i,k,h) but we treat second dim as j
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = stride_right

import torch
import triton
import triton.language as tl

# Triton kernel definition
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    offset_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_b = tl.program_id(2)      # batch dimension
    pid_i = tl.program_id(0)      # i‐tile
    pid_j = tl.program_id(1)      # j‐tile

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    b  = pid_b

    # offsets for i, j and hidden channels
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_h = offset_h + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # iterate over the k dimension in tiles
    for k0 in range(0, N, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for the current tile
        left_ptrs = left_ptr + b * stride_left_b \
                    + offs_i[:, None, None] * stride_left_i \
                    + offs_k[None, :, None] * stride_left_k \
                    + offs_h[None, None, :] * stride_left_h

        right_ptrs = right_ptr + b * stride_right_b \
                     + offs_j[:, None, None] * stride_right_j \
                     + offs_k[None, :, None] * stride_right_k \
                     + offs_h[None, None, :] * stride_right_h

        # load with masking – out‑of‑range elements are zero
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0
        )

        # accumulate over the K tile.
        #   left : (I,K,H)   right : (J,K,H)
        #   for each k we add left[i,k,h] * right[j,k,h] to acc[i,j,h]
        for kk in range(0, BLOCK_K):
            # slices of shape (I,H) and (J,H)
            left_slice  = left[:, kk, :]      # (I, H)
            right_slice = right[:, kk, :]     # (J, H)

            # broadcast to (I,J,H) and accumulate
            acc += left_slice[:, None, :] * right_slice[None, :, :]

    # write back the result
    out_ptrs = out_ptr + b * stride_out_b \
               + offs_i[:, None, None] * stride_out_i \
               + offs_j[None, :, None] * stride_out_j \
               + offs_h[None, None, :] * stride_out_h

    store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=store_mask)

def custom_kernel(data):
    """
    Triton implementation of the “outgoing” Triangle Multiplicative Update.
    The heavy O(N³·C) reduction  out[b,i,j,c] = Σ_k left[b,i,k,c]·right[b,j,k,c]
    is performed in a custom Triton kernel; everything else (LayerNorm,
    linear projections and gating) is done with PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----- Linear + LayerNorm pre‑processing (PyTorch) -----
    # Normalisation of the pair representation
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear projections (bias‑free)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Optional masking (the mask is broadcast over the channel dimension)
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)          # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # Gating scalars
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Make tensors contiguous – required for the pointer arithmetic in Triton
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape

    # ----- Triton kernel: the O(N³·C) reduction -----
    out = torch.empty_like(left)   # (B,N,N,hidden_dim), dtype matches `left`

    # Strides (in elements, not bytes)
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out.stride()
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = stride_l

            BLOCK_J=BLOCK_J,

            BLOCK_H=cur_h,
        )

    # Normalisation of the hidden channel dimension

        out,
        (hidden_dim,),

"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The heavy part of the TriMul forward pass is the three‑dimensional reduction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where
    left  = Linear(x)  * left_gate  * mask
    right = Linear(x)  * right_gate * mask

x has shape [B, N, N, D] and the hidden dimension is H (``hidden_dim``).
The reduction is O(B·N³·H) and dominates the runtime.  The kernel below
tiles the i, j and k dimensions (BLOCK_I × BLOCK_J × BLOCK_K) and
processes the hidden channels in small chunks (BLOCK_H).  For each (i,j)
tile we load a (BLOCK_I, BLOCK_K, BLOCK_H) block of *left* and a
(BLOCK_J, BLOCK_K, BLOCK_H) block of *right*, then accumulate the outer
product over the k‑dimension:

    acc[i,j,h] += Σ_k left[i,k,h] * right[j,k,h]

All arithmetic is performed in FP32; the kernel is launched repeatedly
over the hidden‑channel axis so that any hidden dimension is supported.
The surrounding LayerNorms, linear projections and sigmoids are kept in
PyTorch for simplicity – only the cubic reduction is Triton‑fused.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    offset_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    The kernel processes a tile of size (BLOCK_I, BLOCK_J, BLOCK_H) and
    iterates over the k‑dimension in chunks of BLOCK_K.
    """
    pid_b = tl.program_id(2)      # batch dimension
    pid_i = tl.program_id(0)      # i‑tile index
    pid_j = tl.program_id(1)      # j‑tile index

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    b = pid_b

    # ----- offsets for the current tile -----
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_h = offset_h + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ----- loop over k -----
    for k0 in range(0, N, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for the (I,K,H) and (J,K,H) tiles
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_h[None, None, :] * stride_left_h
        )
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_j[:, None, None] * stride_right_j
            + offs_k[None, :, None] * stride_right_k
            + offs_h[None, None, :] * stride_right_h
        )

        # masked loads – out‑of‑bounds entries are zero
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # accumulate over the K‑tile (unrolled over the compile‑time dimension BLOCK_K)
        for kk in range(0, BLOCK_K):
            # left_slice: (BLOCK_I, BLOCK_H), right_slice: (BLOCK_J, BLOCK_H)
            left_slice = left[:, kk, :]          # (I, H)
            right_slice = right[:, kk, :]        # (J, H)

            # broadcast to (I, J, H) and add to the accumulator
            acc += left_slice[:, None, :] * right_slice[None, :, :]

    # ----- write back the result -----
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_h[None, None, :] * stride_out_h
    )
    store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the “outgoing” Triangle Multiplicative Update.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, D]
            - mask:        torch.Tensor of shape [B, N, N] (may be None)
            - weights:     dict with the module’s parameters
            - config:      dict with keys ``dim`` and ``hidden_dim``.  May also
                           contain ``nomask`` (bool).

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ---------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ---------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ---------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------
    # 3) Optional mask (broadcast over channel dim)
    # ---------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)  # (B, N, N, 1)
        left = left * mask_
        right = right * mask_

    # ---------------------------------------------------------
    # 4) Gating scalars (sigmoid of a bias‑free linear)
    # ---------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # Ensure tensors are contiguous for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape

    # ---------------------------------------------------------
    # 5) Triton kernel: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ---------------------------------------------------------
    # Allocate output buffer (same dtype as left/right)
    out = torch.empty_like(left)

    # Strides (in number of elements)
    stride_l = left.stride()   # (b, i, k, h)
    stride_r = right.stride()
    stride_o = out.stride()
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = stride_l
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = stride_r
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = stride_o

    # Tile sizes – chosen to fit well in H100 shared memory
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_H = 32          # hidden‑channel chunk size processed per kernel launch

    grid = (triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J), B)

    # Launch the kernel for each hidden‑channel chunk
    for h_start in range(0, hidden_dim, BLOCK_H):
        cur_h = min(BLOCK_H, hidden_dim - h_start)

        trimul_outgoing_kernel[grid](
            left,
            right,
            out,
            B,
            N,
            hidden_dim,
            stride_left_b,
            stride_left_i,
            stride_left_k,
            stride_left_h,
            stride_right_b,
            stride_right_j,
            stride_right_k,
            stride_right_h,
            stride_out_b,
            stride_out_i,
            stride_out_j,
            stride_out_h,
            h_start,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=cur_h,   # compile‑time constant for this launch
        )

    # ---------------------------------------------------------
    # 6) Post‑processing in PyTorch
    # ---------------------------------------------------------
    # LayerNorm over the hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to the original dimension
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
