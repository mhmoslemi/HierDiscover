@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0) # i tile
    pid_n = tl.program_id(1) # j tile
    pid_batch = tl.program_id(2) # batch*hidden

    # compute starting offsets
    block_i = pid * BLOCK_M
    block_j = pid_n * BLOCK_N

    # offsets for rows and cols
    offs_i = block_i + tl.arange(0, BLOCK_M)
    offs_j = block_j + tl.arange(0, BLOCK_N)

    # compute mask: off_i < M, off_j < N
    mask_i = offs_i < M
    mask_j = offs_j < N

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_batch * stride_ah + offs_i[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask = mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B transposed: B tile shape (BLOCK_K, BLOCK_N)
        # We need B as K x N: original B is N x K shape; we need to load as transposed.
        # So pointer: b_ptr + pid_batch * stride_bh + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn
        b_ptrs = b_ptr + pid_batch * stride_bh + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn
        b = tl.load(b_ptrs, mask = mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # write output
    c_ptrs = c_ptr + pid_batch * stride_ch + offs_i[:, None] * stride_cm + offs_j[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16) if dtype_out is fp16 else acc, mask = mask_i[:, None] & mask_j[None, :])

    to_out_weight = weights["to_out.weight"]

    # layer_norm expects normalized_shape = dim
    # Input shape [B,N,N,dim] -> normalized over last dimension.
    # So:
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]

    right = right * mask_unsq

    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()

    right = right * right_gate

    left = left.to(torch.float16)

    # optionally make contiguous memory for permute

    right = right.permute(0, 3, 1, 2).contiguous()  # same shape

def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bm, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_m = tl.program_id(0)  # i block
    pid_n = tl.program_id(1)  # j block
    pid_batch = tl.program_id(2) # batch*hidden

    # Compute offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy part of the module is the 4‑D tensor contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left`/`right` have shape (B, N, N, H) (H = hidden_dim).
We fold the batch dimension and the hidden dimension, obtaining a
batched matrix‑multiplication of shape (B*H, N, N) × (B*H, N, N)ᵀ.
The kernel below implements this batched GEMM on H100 using fp16 inputs
and fp32 accumulation.  All other operations (layer‑norms, linear
projections, gating) are performed in PyTorch.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many independent matrices.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (M=N=seq_len, K=N)
    stride_am, stride_ak, stride_ah,   # strides for A (batch, row, col)
    stride_bk, stride_bn, stride_bh,   # strides for B (batch, row=k, col=j)
    stride_cm, stride_cn, stride_ch,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for all batches.
    A: (batch, M, K)   (fp16)
    B: (batch, N, K)   (fp16)  <-- we read it transposed as (K, N)
    C: (batch, M, N)   (fp32)
    The batch dimension folds the original batch * hidden_dim.
    """
    pid_m = tl.program_id(0)          # tile row (i)
    pid_n = tl.program_id(1)          # tile column (j)
    pid_b = tl.program_id(2)          # batch*hidden index

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_b * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile transposed: we need Bᵀ (K, N)
        # Original B layout is (batch, N, K); we address it as (K, N)
        b_ptrs = (
            b_ptr
            + pid_b * stride_bh
            + offs_k[:, None] * stride_bk   # k dimension (original col)
            + offs_n[None, :] * stride_bn   # n dimension (original row)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate: fp16 * fp16 -> fp32 promotion happens automatically
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store the result
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Custom forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]   (float32)
            mask         : [B, N, N]        (float32 or bool)
            weights      : dict of model parameters
            config       : dict with keys "dim" and "hidden_dim"
    Returns:
        out : [B, N, N, dim] (float32)
    """
    # -------------------------- unpack ------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------- weights ----------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # -------------------------- layernorm -------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_w, bias=norm_b, eps=eps
    )  # [B,N,N,dim]

    # -------------------------- projections ------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # -------------------------- mask -------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # -------------------------- gates ------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------- cast to fp16 for kernel ------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ---------------------------------------------------------------
    # Rearrange to [B, hidden, N, N] and flatten batch*hidden
    # ---------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    batch_hidden = B * H

    left_flat = left.view(batch_hidden, N, N)    # [B*H, N, N]  fp16
    right_flat = right.view(batch_hidden, N, N)  # [B*H, N, N]  fp16

    # Output buffer (fp32)
    out_flat = torch.empty((batch_hidden, N, N), dtype=torch.float32, device=device)

    # ---------------------------------------------------------------
    # Triton kernel launch
    # ---------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch_hidden,
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),  # note transposed strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion (optional but safe)
    torch.cuda.synchronize()

    # ---------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim]
    # ---------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H], fp32

    # -------------------------- to_out_norm ------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps
    )

    # -------------------------- gate -------------------------------
    out = out * out_gate

    # -------------------------- final linear ------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # weight shape (dim, hidden_dim)

    return out
