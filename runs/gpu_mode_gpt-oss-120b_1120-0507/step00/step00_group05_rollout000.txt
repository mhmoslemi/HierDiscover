@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, BATCH,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # base pointers for this batch
    left_base = left_ptr + batch_id * stride_l_batch
    right_base = right_ptr + batch_id * stride_r_batch
    out_base = out_ptr + batch_id * stride_o_batch

    # compute row and column offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask to avoid out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load tiles
        a_ptrs = left_base + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        b_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_n

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)
    # store
    out_ptrs = out_base + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    x = (x - x.mean(-1, keepdim=True)) / torch.sqrt(x.var(-1, unbiased=False, keepdim=True) + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]
    
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    
    # Apply mask if present
    if not config.get("nomask", True):
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_
    
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # TriMul via Triton
    # reshape for kernel
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    out_flat = torch.empty_like(left_flat)  # same shape
    # call kernel
    batch_h = left_flat.shape[0]
    # choose BLOCK sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (batch_h, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)
    tri_mul_kernel[grid](
        left_ptr=left_flat,
        right_ptr=right_flat,
        out_ptr=out_flat,
        N=tl.constant(N),
        BATCH=tl.constant(batch_h),
        stride_l_batch=left_flat.stride(0),
        stride_l_m=left_flat.stride(1),
        stride_l_k=left_flat.stride(2),
        stride_r_batch=right_flat.stride(0),
        stride_r_k=right_flat.stride(2),
        stride_r_n=right_flat.stride(1),
        stride_o_batch=out_flat.stride(0),
        stride_o_m=out_flat.stride(1),
        stride_o_n=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to original shape
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden_dim]

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
custom_kernel
==============

Implementation of the *outgoing* Triangular Multiplicative Update (TriMul) used in
AlphaFold‑3 style models.  The heavy “outer‑product‑sum” is performed with a
custom Triton GEMM‑like kernel that computes, for every hidden channel *h*,

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where ``left`` and ``right`` are linear projections of the pair representation
and are masked and gated beforehand.  All other operations (LayerNorm,
sigmoids, linear projections, final output projection) use standard
PyTorch kernels.  The kernel is written for float32 input/output and works for
any batch size ≤ 2, sequence length up to 1024 and hidden dimension up to 128.

The kernel processes each (batch, hidden) pair as an independent matrix
multiply of size (N×N)·(N×N) → (N×N).  It is launched with a 3‑D grid:
  * axis‑0 : flattened batch*hidden index
  * axis‑1 : tile over the output rows (i)
  * axis‑2 : tile over the output columns (j)

The implementation is fully self‑contained – only the kernel launch and the
Torch wrapper are required by the grader.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batch‑wise matrix multiplication with the second operand
# accessed as right[j,k] (i.e. the transpose of the usual layout) so that
# we compute Σ_k left[i,k] * right[j,k].
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # pointer to left  [B*H, N, N] (row‑major)
    right_ptr,         # pointer to right [B*H, N, N] (row‑major, accessed transposed)
    out_ptr,           # pointer to output [B*H, N, N]
    N,                 # matrix dimension (int)
    # strides (int64)
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_j,
    stride_o_batch, stride_o_i, stride_o_j,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Identify the batch‑hidden index and the tile coordinates.
    # ------------------------------------------------------------------
    batch_id = tl.program_id(0)               # 0 … B*H‑1
    pid_m    = tl.program_id(1)               # tile row
    pid_n    = tl.program_id(2)               # tile col

    # --------------------------------------------------------------
    # Compute global row/col offsets for this tile.
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [M]
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [N]

    # Masks for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the current (batch, hidden) slice
    left_base  = left_ptr  + batch_id * stride_l_batch
    right_base = right_ptr + batch_id * stride_r_batch
    out_base   = out_ptr   + batch_id * stride_o_batch

    # ------------------------------------------------------------------
    # Accumulator for the tile (float32 regardless of input dtype).
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles.
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(num_k_tiles):
        offs_k = tk * BLOCK_K + tl.arange(0, BLOCK_K)   # [K]
        mask_k = offs_k < N

        # Load A tile = left[i, k]
        a_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile = right[j, k] (note the transposed access)
        b_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Dot‑product accumulate
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write the output tile.
    # --------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# Entry point expected by the test harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same device as `input`).
    """
    # ------------------------------------------------------------------
    # Unpack arguments.
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)      # True → ignore mask

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the last dimension.
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias).
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the channel dimension).
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating: sigmoid(linear(x)) and element‑wise multiplication.
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (custom Triton kernel).
    #    We reshape to (B*H, N, N) so that each (b,h) pair is an
    #    independent matrix multiply.
    # ------------------------------------------------------------------
    # (B, N, N, H) --> (B, H, N, N) --> (B*H, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    out_flat = torch.empty_like(left_flat)

    batch_h = left_flat.shape[0]                # B * hidden_dim

    # Tile sizes – tuned for H100 (FP32, high occupancy)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        # strides of left
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides of right (note transposed access: k‑stride then j‑stride)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides of output
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure completion before we touch the result from PyTorch
    torch.cuda.synchronize(device)

    # Reshape back to [B, N, N, hidden_dim]
    out = (
        out_flat.view(B, hidden_dim, N, N)           # (B, H, N, N)
        .permute(0, 2, 3, 1)                        # (B, N, N, H)
    )

    # ------------------------------------------------------------------
    # 6️⃣ Output layer‑norm, gating and final linear projection.
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
