def custom_kernel(data):
    x, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5 # maybe from config else default

    # LayerNorm for x (last dimension = dim)
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x_norm = (x - x.mean(-1, keepdim=True)) / torch.sqrt(x.var(-1, unbiased=False, keepdim=True) + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Linear projs
    left = torch.nn.functional.linear(x_norm, weights['left_proj.weight'])  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x_norm, weights['right_proj.weight'])

        right = right * mask[..., None]

    # Gate computations
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['right_gate.weight']))

def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_head, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    
    # program IDs
    pid_batch_head = tl.program_id(2)  # combined batch*hidden
    pid_m = tl.program_id(0)  # i block
    pid_n = tl.program_id(1)  # j block

    # compute batch and hidden indices
    b = pid_batch_head // hidden_dim
    d = pid_batch_head % hidden_dim

    # Determine offsets for a block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Boundary mask:
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # pointers for left: [b, i, k, d]
    # pointer base
    left_base = left_ptr + b * stride_left_b + d * stride_left_d
    
    # pointers for right: [b, j, k, d]
    right_base = right_ptr + b * stride_right_b + d * stride_right_d

    # iterate over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension using the provided
   `norm.weight`/`bias`.
2. Project the normalized tensor to the hidden dimension `H` with two linear
   layers (`left_proj`, `right_proj`).  If a mask is supplied it is applied
   (broadcasted on the hidden dim) after projection.
3. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) from the
   same normalized tensor and multiply them element‑wise with the projected
   tensors.
4. Core operation: for each batch `b` and hidden channel `h`
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. a batch of `B*H` matrix multiplications (N×N) with the reduction
   over the shared `k` dimension.  This is executed by a custom Triton kernel
   that treats each `(b,h)` pair as an independent GEMM and tiles the `i`,
   `j`, and `k` dimensions.
5. Apply a second Layer‑norm on the result, multiply by the `out_gate`,
   and finally linearly project back to the original dimension `dim`.
6. Return the final tensor of shape `[B, N, N, dim]` (float32).

The Triton kernel is deliberately written for mixed‑precision (float32
operands, fp32 accumulation) but can be switched to fp16 by casting the
inputs before the launch if needed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Triton kernel: batched GEMM for the einsum
# ---------------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # *const float* left tensor [B,N,N,H]
    right_ptr,              # *const float* right tensor [B,N,N,H]
    out_ptr,                # *mut   float* output tensor [B,N,N,H]
    B, N, H,                # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # tile size in i (row)
    BLOCK_N: tl.constexpr,  # tile size in j (col)
    BLOCK_K: tl.constexpr,  # tile size in k (reduction)
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    # -----------------------------------------------------------------------
    # 1) Identify which (batch, hidden) pair we are processing
    # -----------------------------------------------------------------------
    pid_batch_head = tl.program_id(2)          # linear index over B*H
    h = pid_batch_head % H                     # hidden channel
    b = pid_batch_head // H                    # batch index

    # -----------------------------------------------------------------------
    # 2) Tile indices for the i (row) and j (col) dimensions
    # -----------------------------------------------------------------------
    pid_i = tl.program_id(0)                   # block row
    pid_j = tl.program_id(1)                   # block col

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # -----------------------------------------------------------------------
    # 3) Pointers to the beginning of the (b,h) slice
    # -----------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # -----------------------------------------------------------------------
    # 4) Accumulator (fp32) for the tile
    # -----------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------------
    # 5) Loop over the reduction dimension k
    # -----------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape (BLOCK_M, BLOCK_K)  --->  left[b,i,k,h]
        left_ptrs = left_base \
                    + (offs_i[:, None] * stride_left_i) \
                    + (offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right: shape (BLOCK_N, BLOCK_K) --->  right[b,j,k,h]
        right_ptrs = right_base \
                     + (offs_j[:, None] * stride_right_j) \
                     + (offs_k[None, :] * stride_right_k)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0)

        # dot over k → adds (BLOCK_M, BLOCK_N) to the accumulator
        acc += tl.dot(left, right)   # fp32 accumulation

    # -----------------------------------------------------------------------
    # 6) Store the result back to global memory
    # -----------------------------------------------------------------------
    out_ptrs = out_base \
               + (offs_i[:, None] * stride_out_i) \
               + (offs_j[None, :] * stride_out_j)

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ---------------------------------------------------------------------------
# Helper that launches the Triton kernel
# ---------------------------------------------------------------------------
def trimul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the 'outgoing' TriMul einsum using the Triton GEMM kernel.

    Args:
        left (torch.Tensor):   [B, N, N, H] (fp32)
        right (torch.Tensor):  [B, N, N, H] (fp32)

    Returns:
        torch.Tensor: out of shape [B, N, N, H] (fp32)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides in elements (not bytes)
    s_l = left.stride()   # (b,i,k,h)
    s_r = right.stride()
    s_o = out.stride()

    # Choose tile sizes – empirically good for H100 & the given workloads
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks over i dimension
        triton.cdiv(N, BLOCK_N),   # blocks over j dimension
        B * H                      # one program per (batch, hidden) pair
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ---------------------------------------------------------------------------
# Entry point required by the problem statement
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (AlphaFold3).
    The heavy pairwise reduction is performed by a Triton kernel.

    Args:
        data: Tuple (input, mask, weights, config)
            - input : torch.Tensor of shape [B, N, N, dim]
            - mask  : torch.Tensor of shape [B, N, N] or None
            - weights: dict of torch.Tensors containing all linear / layernorm
                       parameters (biases for the LayerNorms are included).
            - config : dict with keys “dim”, “hidden_dim”, “nomask” (optional).

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    # -----------------------------------------------------------------------
    # 0) Unpack arguments
    # -----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # -----------------------------------------------------------------------
    # 1) First LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------------
    # weights for the first norm
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # torch.nn.functional.layer_norm expects normalized_shape as a tuple
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]

    # -----------------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # -----------------------------------------------------------------------
    left  = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------------
    # 3) Optional mask handling (mask broadcasted on hidden dim)
    # -----------------------------------------------------------------------
    if not config.get("nomask", True) and mask_tensor is not None:
        # mask shape: [B, N, N] -> [B, N, N, 1]
        mask = mask_tensor.to(dtype=input_tensor.dtype).unsqueeze(-1)
        left  = left * mask
        right = right * mask
    # else: no mask → nothing to do

    # -----------------------------------------------------------------------
    # 4) Gating values (sigmoid after linear)
    # -----------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left  = left  * left_gate   # element‑wise gating
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 5) Core einsum using the fused Triton kernel
    # -----------------------------------------------------------------------
    # Result shape: [B, N, N, hidden_dim]
    out = trimul_einsum(left, right)

    # -----------------------------------------------------------------------
    # 6) Second LayerNorm (over hidden_dim)
    # -----------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # -----------------------------------------------------------------------
    # 7) Apply the output gate and final linear projection back to dim
    # -----------------------------------------------------------------------
    out = out * out_gate               # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
