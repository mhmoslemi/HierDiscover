Standard algorithm:

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

        max_conv = conv.max()
        return 2.0 * n * max_conv  # sum(seq) is 1

    uniform_seq = np.full(n, 1.0/n, dtype=np.float64)
    best_seq = uniform_seq.copy()

    print(f"[init] uniform score={best_score:.6f}")

    exponents = np.concatenate([
        np.linspace(0.05, 0.8, 15),
        np.linspace(0.8, 1.4, 10)

    for p in exponents:
        seq = (np.arange(1, n+1, dtype=np.float64) ** (-p))
        seq = seq / seq.sum()
        score = compute_score(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

        eps = 1e-12
        seq = np.power(x + eps, p-1) * np.power(1 - x + eps, p-1)

        if total <= 0:

        term1_full = conv_full(best_seq, v[::-1])
        term1 = term1_full[2*n - 2 - np.arange(n)]

        grad = term1 + term2

        # step size schedule
        eta = eta0 / np.sqrt(it + 1)

        # update and project
        a_new = best_seq - eta * grad
        a_new = project_simplex(a_new)

            a_new = project_simplex(a_new)

        # compute score of new candidate (exact)
        conv_new = conv_full(a_new, a_new)
        new_score = 2.0 * n * conv_new.max()

        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            print(f"[improve] it={it} score={best_score:.6f}")

        # assign iteration counter
        it += 1

    # final evaluation using evaluate_sequence for safety
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[done] final best score={final_score:.6f}")

"""
Propose a candidate step‑function sequence.

The evaluation metric is
    score = 2·n·max(conv(a,a)) / (sum(a))²

Because the score is invariant under scaling (as long as no value is
clipped at 1000), we work with a probability distribution a (a_i ≥ 0,
∑a_i = 1).  The problem then becomes:
    minimise   max_k  (a ⋆ a)_k
where “⋆” is the linear convolution.

Approach
--------
1.   Determine the required length n (taken from the global
     `height_sequence_1` when available, otherwise a safe default).
2.   Initialise with a collection of analytically‑motivated shapes:
     * uniform,
     * power‑law a_i ∝ (i+1)^(‑p),
     * Beta‑family a_i ∝ x^{p‑1}(1‑x)^{p‑1}  (x = (i+½)/n),
     * the discrete arcsine law a_i ∝ 1/√(x(1‑x)).
     The best of these (measured with the exact score) seeds the search.
3.   Refine the seed by projected gradient descent on a smooth
     approximation of the max‑convolution.
     * Use the soft‑max surrogate
           f(a) = (1/α)·log Σ_k exp(α·c_k) ,  c = a⋆a .
       Its gradient is  ∇f = Σ_k v_k ∂c_k/∂a with
       v_k = exp(α·c_k)/Σ_j exp(α·c_j) .
       The derivative of a convolution term is
           ∂c_k/∂a_i = a_{k‑i} + a_{i‑k}   (zero when indices are out of range).
       Hence
           grad_i = Σ_k v_k (a_{k‑i}+a_{i‑k})
                  = term1_i + term2_i,
       where
           term1 = a * v   (convolution)  → first n entries,
           term2 = a * rev(v) shifted appropriately.
     * Convolutions are performed with FFT (`np.fft.rfft`) for speed
       (O(n log n) per iteration).
     * After each gradient step a projection onto the probability simplex
       keeps the vector non‑negative and of unit sum.
     * The step size decays as η₀/√t, a small amount of Gaussian noise is
       injected every few hundred steps to escape plateaus.
4.   The loop runs until the time budget expires, keeping the best
   sequence (according to the true `evaluate_sequence` function).
5.   The final sequence is returned as a plain Python list.

The method explores a richer family of shapes than the reference
(power‑law + block perturbations) and uses a principled gradient‑based
optimiser that directly targets the max‑convolution, which typically
pushes the score well below the previous 1.578 → target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_simplex(v):
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x_i ≥ 0, sum_i x_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u > cssv / np.arange(1, n + 1))[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full(x, y):
    """
    Full (linear) convolution using FFT.  Returns a real array of length
    len(x)+len(y)-1.
    """
    size = len(x) + len(y) - 1
    fft_len = 1 << (size - 1).bit_length()    # next power of two
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    Z = X * Y
    out = np.fft.irfft(Z, fft_len)
    return out[:size]


def exact_score(a):
    """
    Exact evaluation of a (assumed to sum to 1) without calling the external
    `evaluate_sequence`.  This is safe because a_i will be far below the
    clipping bound 1000.
    """
    n = a.shape[0]
    c = conv_full(a, a)
    return 2.0 * n * c.max()

# ----------------------------------------------------------------------

    `evaluate_sequence`.  Returns the best sequence found before the
    time limit expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    score = exact_score(uniform)

    print(f"[init] uniform → score={score:.6f}")

    except Exception:

    # power‑law shapes a_i ∝ (i+1)^(-p)
    power_exps = np.concatenate([
        np.linspace(0.05, 0.45, 9),
        np.linspace(0.45, 0.85, 17),

    x = (i + 0.5) / n

    print(f"[info] Best initial score = {best_score:.6f}")

    # --------------------------------------------------------------

    #     soft‑max surrogate of max(conv)
    # --------------------------------------------------------------
    # hyper‑parameters
    eta0 = 0.12               # initial step size
    alpha_start = 5.0         # soft‑max sharpness at the beginning
    alpha_end = 30.0          # final sharpness
    max_iters = 10_000        # safety upper bound (loop will stop earlier)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    a = best_seq.copy()       # current iterate (already sum‑to‑1)

    it = 0

        # linear schedule for α (more aggressive as time goes on)
        frac = (time.monotonic() - start_time) / budget_s
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # convolution a⋆a
        conv = conv_full(a, a)
        conv_max = conv.max()
        if conv_max == 0.0:      # degenerate – should not happen
            break

        # soft‑max weights, normalised and numerically stable
        # we use conv / conv_max - 1 to bring the maximum to zero.
        weights = np.exp(alpha * (conv / conv_max - 1.0))
        weights /= weights.sum()

        # term2 = (a * weights)  → first n entries of the convolution
        term2_full = conv_full(a, weights)
        term2 = term2_full[:n]

        # term1 = (a * rev(weights)) shifted
        rev_weights = weights[::-1]
        term1_full = conv_full(a, rev_weights)
        term1 = term1_full[2 * n - 2 - np.arange(n)]

        grad = term1 + term2                       # sub‑gradient of the surrogate

        # step size decays like η₀/√(it+1)
        eta = eta0 / np.sqrt(it + 1.0)

        # gradient step + projection onto the simplex
        a_new = a - eta * grad
        a_new = project_simplex(a_new)

        # occasional small random shake to avoid stagnation
        if it % 250 == 0 and it > 0:

            a_new = a_new + noise
            a_new = project_simplex(a_new)

        it += 1
