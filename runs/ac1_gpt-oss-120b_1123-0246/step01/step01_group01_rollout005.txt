"""
Propose a step‑function sequence that improves on the binomial (π/2) bound.

Algorithm
---------
1. Initialise a sequence `f` with the classic binomial coefficients
   a_i = C(2i,i) / 4^i (the coefficients of (1‑z)^‑1/2).  This shape yields a
   self‑convolution peak M = 1 and a score ≈ π/2.

2. Iteratively improve `f` by solving a linear programme:
      maximise   Σ g_i
      subject to g_i ≥ 0 , g_i ≤ 1000
                 (f * g)_k ≤ M      for all k = 0 … 2 n‑2
   The convolution (f * g) is linear in the unknown vector `g`,
   therefore the constraints are linear.  The solution `g` puts as much
   mass as possible on positions where the current `f` is small,
   thereby increasing the total ℓ¹‑norm without raising the worst
   convolution value.

3. After the LP is solved, `g` is rescaled so that its ℓ¹‑norm equals the
   original sum of `f`.  The sequence is then replaced by this normalised
   `g`; the new peak `M = max(f*f)` is recomputed.

4. Steps 2‑3 are repeated until the time budget expires or the peak stops
   decreasing.  The best sequence seen during the run is kept and returned.

The optimisation works directly on the quantity that matters for the
objective (the maximum convolution value) and therefore finds
sub‑π/2‑scores much faster than blind hill‑climbing.  The programme
uses only NumPy, SciPy and the provided `evaluate_sequence` function,
runs under the required time limit and prints progress information.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def _binomial_sequence(length: int) -> np.ndarray:
    """Truncated binomial coefficients a_i = C(2i,i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(length - 1):
        a[i + 1] = a[i] * (2.0 * i + 1.0) / (2.0 * (i + 1))
    return np.clip(a, 0.0, 1000.0)


def _build_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (shape (2*n-1, n)) such that
        (f * g)_k = (A @ g)[k]
    for linear convolution with zero padding.
    """
    n = f.size
    diags = []
    offsets = []
    # column i contributes the constant f[i] to rows i … i+n-1
    for d in range(n):
        # diagonal with offset d has length n-d and constant f[:n-d]
        diags.append(f[: n - d])
        offsets.append(d)
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def propose_candidate(seed: int = 42, budget_s: float = 900, **kwargs) -> list[float]:
    """Return a non‑negative height sequence that (hopefully) beats the π/2 bound."""
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # 1. Choose a working length.
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = max(200, base.size)        # ensure we have enough room for optimisation
    else:
        n = 800                         # a comfortable size for the LP

    # -----------------------------------------------------------------
    # 2. Initialise with the binomial sequence.
    # -----------------------------------------------------------------
    f = _binomial_sequence(n)
    f = np.clip(f, 0.0, 1000.0)

    # Keep the ℓ¹‑norm fixed – scaling does not change the final score.
    sum_target = f.sum()
    if sum_target < 0.01:            # defensive – should never happen
        sum_target = np.sqrt(2.0 * n)

    # -----------------------------------------------------------------
    # 3. Main LP‑driven improvement loop.
    # -----------------------------------------------------------------
    conv_f = np.convolve(f, f)
    M = conv_f.max()
    best = f.copy()
    best_score = 2.0 * n * M / (sum_target * sum_target)
    print(f"[init] n={n}, initial score={best_score:.6f}, M={M:.6f}")

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # Build the linear constraints (f * g)_k ≤ M
        A = _build_constraint_matrix(f)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # Linear programme: maximise Σ g_i  ⇔ minimise -Σ g_i
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, 1000.0) for _ in range(n)]

        # Solve the LP (HiGHS is fast and accepts CSR matrices)
        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            print(f"[{iteration}] LP failed: {res.message}")
            break

        g = res.x
        # Rescale to keep the ℓ¹‑norm constant.
        g_sum = g.sum()
        if g_sum == 0.0:
            print(f"[{iteration}] Zero‑sum LP solution – aborting.")
            break
        scale = sum_target / g_sum
        g *= scale
        # Ensure we stay inside the allowed [0,1000] box after scaling.
        np.clip(g, 0.0, 1000.0, out=g)

        # New peak and score.
        conv_g = np.convolve(g, g)
        M_new = conv_g.max()
        score = 2.0 * n * M_new / (sum_target * sum_target)

        if score < best_score - 1e-7:
            best = g.copy()
            best_score = score
            print(f"[{iteration:5d}] improvement → score={best_score:.6f}, M={M_new:.6f}")

        # Convergence test – if the peak does not move appreciably we stop.
        if abs(M_new - M) < 1e-12:
            print(f"[{iteration}] convergence: peak unchanged (M={M_new:.6f}).")
            break

        # Prepare next iteration.
        f = g
        M = M_new

    print(f"[done] iterations={iteration}, final best score={best_score:.6f}")

    # Return the best sequence as a plain Python list.
    return [float(v) for v in best]
