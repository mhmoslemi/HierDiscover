"""
Iterative refinement of a non‑negative step sequence.

The objective is to minimise

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the ordinary (linear) convolution.
The algorithm proceeds as follows:

1.  Initialise ``a`` with the Catalan‑type coefficients
    a_i = C_i = binom(2*i, i) / 4**i.
    If a global variable ``height_sequence_1`` exists we also try a truncated
    copy of it and keep the best of the two starts.

2.  For the current sequence ``a`` compute the worst convolution value
    ``M = max(conv(a, a))``.

3.  Solve the linear program

        maximise   sum(g_i)
        subject to (a * g)_k ≤ M   for all k
                   g_i ≥ 0

    The constraints are linear in ``g``; we build the dense Toeplitz matrix
    that represents the convolution operator and feed it to ``scipy.optimize.linprog``.
    The LP finds a non‑negative vector ``g`` that can be added to ``a`` without
    raising the current worst convolution bound ``M`` and that has a larger
    ℓ₁–norm than ``a``.

4.  Mix the current sequence with the LP solution:
    ``candidate = (1-t) * a + t * g`` for a few step sizes
    ``t ∈ {0.4, 0.3, 0.2, 0.1, 0.05}``.
    The mixed sequence is clipped to the admissible range [0, 1000] and
    evaluated with ``evaluate_sequence``.  The first improvement is accepted,
    ``a`` is updated and the loop restarts.

5.  If the LP does not yield an improvement we perform a cheap random
    multiplicative perturbation (log‑normal noise) and keep it if it improves
    the score.

6.  The process repeats until the allotted time budget expires or a maximum
    number of iterations is reached.  The best sequence seen is returned.

The method is a pure‑Python implementation that only relies on NumPy and
SciPy (no external solvers).  It is designed to run comfortably within the

def catalan_type_sequence(n: int) -> np.ndarray:
    """
    Return the first ``n`` Catalan‑type coefficients
    C_i = binom(2*i, i) / 4**i  (i ≥ 0).

    The sequence is decreasing and satisfies (C * C)_k = 1 for all
    k < n (up to rounding errors).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence a[i] = a[i-1] * (2*i-1)/(2*i)
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    # safety clipping – the true values never exceed 1, but we respect the
    # limits imposed by ``evaluate_sequence``.
    return np.clip(a, 0.0, 1000.0)


def build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix ``A`` such that ``A @ g = np.convolve(a, g)``.
    ``A`` has shape (2*len(a)-1, len(a)) and is a Toeplitz matrix whose
    i‑th column is ``a`` shifted down by i rows.
    """
    n = a.size
    # First column = a followed by (n-1) zeros
    first_col = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
    # First row = a[0] followed by (n-1) zeros
    first_row = np.concatenate([np.array([a[0]], dtype=np.float64),
                                np.zeros(n - 1, dtype=np.float64)])
    return toeplitz(first_col, first_row)


def solve_lp_for_g(a: np.ndarray, M: float, eps: float = 1e-9):
    """
    Solve the LP that maximises the ℓ₁‑norm of ``g`` under the constraint
    ``np.convolve(a, g) ≤ M`` (element‑wise) and ``g ≥ 0``.
    Returns the optimal ``g`` as a NumPy array, or ``None`` on failure.
    """
    n = a.size
    A = build_convolution_matrix(a)

    # Upper bound for every convolution entry
    b = np.full(A.shape[0], M + eps, dtype=np.float64)

    # Objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None)] * n

    # Use HiGHS – fast interior‑point method for dense problems
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method="highs", options={"presolve": True})

    if res.success and res.x is not None:
        return np.maximum(0.0, res.x)  # numerical safety
    return None

    array and returns a Python float.  The function is deliberately tiny so
    that failures propagate as ``np.inf`` (the judge treats that as a loss).
    """
    return float(evaluate_sequence(seq.tolist()))

        # -------------------------------------------------------------
        g = solve_lp_for_g(best, M, eps=eps)

        if g is None:
            print(f"Iteration {iteration}: LP failed – switching to random tweaks.")
            # fall back to a cheap random perturbation
            perturb = np.exp(rng.normal(0.0, 0.06, size=n))
            cand = np.clip(best * perturb, 0.0, 1000.0)
            try:
                cand_score = evaluate(cand)
            except Exception:
                cand_score = np.inf
            if np.isfinite(cand_score) and cand_score < best_score:
                best = cand
                best_score = cand_score
                print(f"Iter {iteration:02d}, random improvement → score {best_score:.6f}")
            iteration += 1
            continue

        # -------------------------------------------------------------
        # 3b.  Try mixing with several step sizes; accept the first
        #      improvement (this mimics the original “t‑grid”).
        # -------------------------------------------------------------
        improved = False
        for t in (0.4, 0.3, 0.2, 0.1, 0.05):
            candidate = (1.0 - t) * best + t * g
            candidate = np.clip(candidate, 0.0, 1000.0)
            try:
                cand_score = evaluate(candidate)
            except Exception:
                continue
            if np.isfinite(cand_score) and cand_score < best_score - 1e-12:
                best = candidate
                best_score = cand_score
                improved = True
                print(f"Iter {iteration:02d}, mix t={t:.2f} → score {best_score:.6f}")
                break   # accept the first successful step size

        # -------------------------------------------------------------
        # 3c.  If mixing did not improve, fall back to a random tweak.
        # -------------------------------------------------------------
        if not improved:
            perturb = np.exp(rng.normal(0.0, 0.06, size=n))
            candidate = np.clip(best * perturb, 0.0, 1000.0)
            try:
                cand_score = evaluate(candidate)
            except Exception:
                cand_score = np.inf
            if np.isfinite(cand_score) and cand_score < best_score - 1e-12:
                best = candidate
                best_score = cand_score
                print(f"Iter {iteration:02d}, random tweak → score {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 4.   Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    # Convert to plain Python list of floats as required by the harness
    return [float(v) for v in best]
