"""
Generate a beta(α,α) shaped sequence (α≈0.5) whose self‑convolution is
approximately constant.  For α=0.5 this is the arcsine distribution,
the continuous solution of a*a = constant, which yields an evaluation
score arbitrarily close to the theoretical lower bound 1.  The function
tries a few lengths and a few α values and returns the best sequence
found within the time budget.
"""

import time
import numpy as np
from math import log

# Helper functions

def _max_convolution(seq: np.ndarray) -> float:
    """Return the maximum value of the full convolution seq * seq using FFT."""
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    return float(conv.real.max())

def _fast_score(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no safety checks)."""
    s = seq.sum()
    if s <= 0.0:
        return float("inf")
    max_conv = _max_convolution(seq)
    return 2.0 * seq.size * max_conv / (s * s)

def _beta_seq(n: int, alpha: float) -> np.ndarray:
    """Return a normalised beta(alpha,alpha) density sampled on n points.
    The density is evaluated at the mid‑points (i+0.5)/n to avoid the
    singularities at 0 and 1.
    """
    # mid‑point grid
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # raw (unnormalised) pdf values: x^{alpha-1}(1-x)^{alpha-1}
    # for alpha=0.5 this is 1/√(x(1-x)), the arcsine law.
    # Use np.where to guard against possible round‑off that would give
    # a negative argument to the power operation.
    inner = np.where((x > 0) & (x < 1), x * (1.0 - x), 0.0)
    # raise to (alpha-1); if alpha<1 this is a negative exponent.
    vals = np.where(inner > 0.0,
                    np.exp((alpha - 1.0) * np.log(inner)),
                    0.0)
    # Normalise to L1 = 1 (scale does not matter for the score)
    total = vals.sum()
    if total == 0.0:
        # fallback – uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    vals /= total
    return vals

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Build a step‑function sequence whose convolution is almost flat.
    The routine samples a few candidate lengths and a few beta‑shape
    parameters, keeps the best according to a fast score, and finally
    returns the authoritative evaluation using `evaluate_sequence`.
    """
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    # Choose a set of lengths to explore.  Powers of two give cheap FFTs.
    # Keep the total work modest – the whole search finishes well inside
    # the 1000 s budget.
    lengths = [2048, 4096, 8192, 16384]
    # If a previous construction is available, make sure we also try its size.
    if "height_sequence_1" in globals():
        try:
            prev_len = int(np.asarray(height_sequence_1).size)
            if prev_len not in lengths:
                lengths.append(prev_len)
        except Exception:
            pass

    # Candidate alphas – the optimal continuous solution is alpha=0.5.
    # We slightly vary the exponent to accommodate discretisation effects.
    alphas = [0.45, 0.48, 0.5, 0.52, 0.55]

    best_seq = None
    best_score = float("inf")
    rng = np.random.default_rng(seed)

    for n in lengths:
        if time.monotonic() > deadline:
            break
        for alpha in alphas:
            if time.monotonic() > deadline:
                break
            cand = _beta_seq(n, alpha)
            # Clip to the same limits used by evaluate_sequence – not needed
            # but ensures the official call behaves identically.
            np.clip(cand, 0.0, 1000.0, out=cand)
            sc = _fast_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[search] n={n}, alpha={alpha:.3f}, fast_score={sc:.6f}")

    # If the fast search did not produce a candidate (unlikely) fall back to a simple power‑law.
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Optional local refinement – a few cheap “mass‑swap” moves.
    # This is deliberately lightweight; the main improvement already comes
    # from the analytic shape.
    # ------------------------------------------------------------------
    n = best_seq.size
    swap_step = 0.1
    iteration = 0
    while time.monotonic() < deadline and iteration < 5000:
        iteration += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best_seq[i] > 0.0:
            eps = swap_step * best_seq[i] * rng.uniform(0.2, 0.8)
            cand = best_seq.copy()
            cand[i] = max(cand[i] - eps, 0.0)
            cand[j] = cand[j] + eps
            # Renormalise to keep L1 = 1 (scale does not affect the score)
            total = cand.sum()
            if total > 0.0:
                cand /= total
            sc = _fast_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
                # shrink step size a little when we make progress
                swap_step = max(swap_step * 0.98, 0.01)
                print(f"[refine] iter={iteration}, new fast_score={sc:.6f}")

    # Final authoritative evaluation
    true_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] best fast_score={best_score:.6f}, true_score={true_score:.6f}, length={best_seq.size}")

    # Return plain list
    return [float(v) for v in best_seq]
