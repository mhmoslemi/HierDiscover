Algorithm details:

def _project_simplex(v):
    # v is np.ndarray
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function sequence.

The algorithm is a *projected sub‑gradient descent* on the quantity  

    F(a) = 2·n·max_k (a * a)_k   (the sum of a is kept = 1)

Because the objective is invariant under scaling we may keep the
vector on the probability simplex.  At each iteration we

1. compute the full convolution a*a (FFT, O(n log n));
2. locate the (possibly several) indices where the convolution attains
   its maximum;
3. build a mask w that is 1 on those indices and 0 elsewhere;
4. the sub‑gradient of ``max`` is the convolution of a with the
   reversed mask (again via FFT);
5. take a step –step·grad, project the result onto the simplex,
   and accept the step only if the fast score improves (back‑tracking).

Occasionally a small random perturbation is added to avoid getting trapped
in a flat region.  The best sequence seen so far is evaluated with the
authoritative ``evaluate_sequence`` function and returned at the end of
the allotted time budget.

The method needs only two FFTs per iteration and therefore comfortably
fits into the 1000 s budget even for a few‑thousand‑element vectors.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Full convolution of ``a`` with itself (real FFT) and return its maximum.
    """
    n = a.size
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def _fast_score(a: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` – scale invariant.
    """
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    max_conv = _max_convolution_fft(a)
    return 2.0 * n * max_conv / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    # find the greatest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolution_with_mask(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Compute g = a * mask_rev   (full convolution) and return the slice
    that yields the sub‑gradient:  grad_j = 2 * g[L‑1‑j] where L = len(mask).
    """
    n = a.size
    L = mask.size               # = 2·n‑1
    size = _next_pow2(n + L - 1)   # ≥ 3·n‑2
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(mask[::-1], n=size)   # reversed mask
    conv = np.fft.irfft(A * W, n=size)
    # extract the required entries
    idx = L - 1 - np.arange(n)
    grad = 2.0 * conv[idx]
    return grad


def _make_random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random start on the simplex – a Dirichlet(1) draw.
    """
    x = rng.exponential(1.0, size=n)
    x /= x.sum()
    return x


def _power_law_start(n: int, exponent: float = 0.5, shift: float = 0.0) -> np.ndarray:
    """Analytic 1/√(i+shift) start, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power(i + shift + 1.0, exponent)
    vals /= vals.sum()
    return vals

    """
    Search for a step‑function (non‑negative vector) that minimises
    `evaluate_sequence`.  A projected sub‑gradient descent with back‑tracking
    line‑search is employed; the best vector found within the time budget
    is returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # Choose problem size (length of the vector)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(height_sequence_1, dtype=np.float64)
        if cand.ndim != 1 or cand.size == 0 or not np.isfinite(cand).all():
            n = 3000
        else:
            n = int(cand.size)
    else:
        n = 3000

    # --------------------------------------------------------------
    # Initialise the current iterate
    # --------------------------------------------------------------
    # Start from a smooth analytic shape; fall back to random if needed.
    a = _power_law_start(n, exponent=0.5, shift=0.0)
    a = np.clip(a, 0.0, 1000.0)
    a /= a.sum()                     # simplex

    # keep a copy of the best sequence seen so far
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial authoritative score={best_score:.6f}")

    # parameters for the descent
    step = 0.5                       # initial step size
    min_step = 1e-6
    decay_factor = 0.9995            # very slow global decay
    it = 0

    return [float(v) for v in best]
