def _targeted_move(base, rng, step_mass):
    n = base.size
    conv = _full_convolution(base) # size 2n-1
    k = int(np.argmax(conv))
    # Determine contributions array
    # For i in [0,n-1], j = k - i
    # compute product if 0 <= j < n else 0
    # store product
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    prod = np.zeros_like(base)
    prod[mask] = base[idx[mask]] * base[j[mask]]
    total = prod.sum()
    if total == 0:
        # fallback to random move
        return base.copy()
    # sample i with probability proportional to prod
    probs = prod / total
    i = int(rng.choice(n, p=probs))
    j = k - i
    # compute epsilon
    if i == j:
        max_eps = base[i] * step_mass
        epsilon = max_eps * rng.uniform(0.5, 1.0)  # some random fraction

def _targeted_move(base, rng, step_factor):
    n = base.size
    # compute full conv once
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(base, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    k = int(np.argmax(conv))
    # compute contributions
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    # product contributions
    prod = np.zeros_like(base)
    prod[mask] = base[idx[mask]] * base[j[mask]]
    total = prod.sum()
    if total == 0.0:
        # fallback small random perturbation
        cand = base.copy()
        i = rng.integers(0, n)
        epsilon = step_factor * cand[i] * rng.uniform(0.1, 0.5)
        cand[i] = max(cand[i] - epsilon, 0.0)
        # add epsilon to random index
        r = rng.integers(0, n)
        cand[r] += epsilon
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand /= cand.sum()
        return cand

    # sample index i with probability prop to prod
    probs = prod / total
    i = int(rng.choice(n, p=probs))
    j = k - i
    # ensure within range
    if j < 0 or j >= n:
        # fallback
        i = int(rng.integers(0, n))
        j = i
    # compute epsilon up to step_factor * min(base[i], base[j] if i!=j else base[i])
    if i == j:
        max_eps = base[i] * step_factor
        epsilon = max_eps * rng.uniform(0.2, 0.8)
        cand = base.copy()
        cand[i] = max(cand[i] - epsilon, 0.0)
    else:
        max_eps = min(base[i], base[j]) * step_factor
        epsilon = max_eps * rng.uniform(0.2, 0.8)
        cand = base.copy()
        cand[i] = max(cand[i] - epsilon, 0.0)
        cand[j] = max(cand[j] - epsilon, 0.0)
    # redistribute removed mass: total_removed = epsilon * (2 if i!=j else 1)
    total_removed = epsilon * (2 if i != j else 1)
    # add mass to two random positions (could be same)
    r1, r2 = rng.choice(n, size=2, replace=False)
    cand[r1] += total_removed / 2
    cand[r2] += total_removed / 2
    np.clip(cand, 0.0, 1000.0, out=cand)
    # renormalize to unit sum
    s = cand.sum()
    if s > 0:
        cand /= s
    return cand

"""
Propose a step‑function sequence that minimises
    2 · n · max(conv(seq,seq)) / (sum(seq)**2)

The algorithm combines three ideas:

1. **Good analytical starter** – the coefficients of the series
   (1‑z)^‑½, i.e. a_i ∝ 1/√(i+1), are known to give a value ≈ π/2 ≈ 1.5708.
   This is used as the initial candidate (normalised to L1 = 1).

2. **Diverse stochastic moves** – at every iteration a new candidate is
   generated by one of four operators:
   * mixture of a uniform vector and a power‑law vector,
   * block‑wise multiplicative perturbation,
   * a tiny mass‑swap between two positions,
   * a *targeted* move that explicitly reduces the currently maximal
     convolution entry (the shift k where max(conv) occurs).  The latter
     removes a small amount of probability mass from the pair (i, k‑i)
     that contributes most to the peak and redistributes it elsewhere.

3. **Fast pre‑filtering & classic hill‑climbing** – the candidate is first
   evaluated with a cheap FFT‑based “fast” score; only if it improves the
   best known value the expensive, fully‑checked ``evaluate_sequence``
   is called.  The step size for the targeted move is slowly annealed.

The search runs for the allotted time (default 1000 s, leaving a few
seconds margin) and finally returns the best sequence found as a plain
list of floats.  All numbers are non‑negative and respect the clipping
rules used inside ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core convolution helpers (FFT based – O(n log n))
# ----------------------------------------------------------------------


def _full_convolution(seq: np.ndarray) -> np.ndarray:
    """Return the full discrete convolution seq * seq (length 2·n‑1)."""
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    return conv.real


def _max_convolution(seq: np.ndarray) -> float:
    """Maximum value of the full self‑convolution of ``seq``."""
    return float(_full_convolution(seq).max())


def _fast_score(seq: np.ndarray) -> float:
    """
    The same quantity returned by ``evaluate_sequence`` but without the
    safety checks.  Scale‑invariant, therefore we can use the raw sum.
    """
    n = seq.size
    s = float(seq.sum())
    if s < 1e-12:
        return float("inf")
    max_conv = _max_convolution(seq)
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Simple generators for “parameter sweep” moves
# ----------------------------------------------------------------------


def _power_law_seq(n: int, exponent: float = 0.5, shift: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+shift)^{‑exponent}  (normalised to L1 = 1)."""
    idx = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power(idx + shift + 1.0, exponent)
    vals /= vals.sum()
    return vals


def _mixed_seq(
    n: int, exponent: float, shift: float, alpha: float
) -> np.ndarray:
    """Uniform ↔ power‑law mixture.  α = weight of the uniform part."""
    uniform = np.full(n, 1.0 / n)
    power = _power_law_seq(n, exponent=exponent, shift=shift)
    mixed = alpha * uniform + (1.0 - alpha) * power
    mixed /= mixed.sum()
    return mixed

# Local moves
# ----------------------------------------------------------------------


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    w_low = max(1, n // 200)
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    scale = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= scale
    if rng.random() < 0.35:               # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _targeted_move(base: np.ndarray,
                   rng: np.random.Generator,
                   step_factor: float) -> np.ndarray:
    """
    Reduce the currently maximal convolution entry.
    * Identify the shift k where max(conv) occurs.
    * Pick an index i with probability proportional to the contribution
      a_i·a_{k‑i}.
    * Remove a small amount ε (controlled by ``step_factor``) from i
      (and from its partner j = k‑i if j≠i) and redistribute the removed
      mass to two random locations.
    * Renormalise to L1 = 1.
    """
    n = base.size
    conv = _full_convolution(base)
    k = int(np.argmax(conv))

    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)

    # contributions of each index to the peak
    contrib = np.zeros_like(base)
    contrib[mask] = base[idx[mask]] * base[j[mask]]
    total = contrib.sum()
    if total == 0.0:
        # fall back to a tiny random perturbation
        cand = base.copy()
        i = int(rng.integers(0, n))
        eps = step_factor * cand[i] * rng.uniform(0.2, 0.8)
        cand[i] = max(cand[i] - eps, 0.0)
        r = int(rng.integers(0, n))
        cand[r] += eps
        np.clip(cand, 0.0, 1000.0, out=cand)
        cand /= cand.sum()
        return cand

    probs = contrib / total
    i = int(rng.choice(n, p=probs))
    j = k - i
    if not (0 <= j < n):
        # safety – treat as a single‑index move
        j = i

    if i == j:
        max_eps = base[i] * step_factor
        eps = max_eps * rng.uniform(0.2, 0.8)
        cand = base.copy()
        cand[i] = max(cand[i] - eps, 0.0)
        removed = eps
    else:
        max_eps = min(base[i], base[j]) * step_factor
        eps = max_eps * rng.uniform(0.2, 0.8)
        cand = base.copy()
        cand[i] = max(cand[i] - eps, 0.0)
        cand[j] = max(cand[j] - eps, 0.0)
        removed = 2.0 * eps

    # redistribute the removed mass to two random positions
    r1, r2 = rng.choice(n, size=2, replace=False)
    cand[r1] += removed / 2.0
    cand[r2] += removed / 2.0
    np.clip(cand, 0.0, 1000.0, out=cand)

    # renormalise (preserves the scale‑invariant objective)
    s = cand.sum()
    if s > 0.0:
        cand /= s
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a sequence that minimises ``evaluate_sequence``.
    The function respects a time budget (seconds) and returns the best
    sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    # leave a few seconds spare for finalisation / printing
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Choose a working length and an initial reference sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base_seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = base_seq.size
        if n == 0 or not np.isfinite(base_seq).all():
            n = 6000
            base_seq = np.ones(n, dtype=np.float64)
    else:
        n = 6000
        base_seq = np.ones(n, dtype=np.float64)

    # Clip to the limits that ``evaluate_sequence`` would enforce
    np.clip(base_seq, 0.0, 1000.0, out=base_seq)

    # Normalise to L1 = 1 (scale does not matter for the objective)
    base_seq = base_seq / base_seq.sum()

    # ------------------------------------------------------------------
    # Initial candidates
    # ------------------------------------------------------------------
    # analytic starter – the (1‑√i) shape
    current = _power_law_seq(n, exponent=0.5, shift=0.0)

    best = current.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # compare with the provided global sequence (if any)
    try:
        orig_score = float(evaluate_sequence(base_seq.tolist()))
        if orig_score < best_score:
            best = base_seq.copy()
            best_score = orig_score
    except Exception:
        pass

    print(f"[init] length={n}, initial best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main stochastic hill‑climbing loop
    # ------------------------------------------------------------------
    iteration = 0
    step_factor = 0.15          # controls the aggressiveness of the targeted move
    swap_step = 0.12            # for the simple mass‑swap move

    while time.monotonic() < deadline:
        iteration += 1

        # pick a move type
        move_type = rng.choice(
            ["param", "block", "swap", "target"],
            p=[0.22, 0.38, 0.20, 0.20],
        )

        if move_type == "param":
            # random mixture of uniform and a power‑law vector
            exponent = float(rng.uniform(0.30, 0.75))
            shift = float(rng.exponential(5.0))
            alpha = float(rng.uniform(0.0, 0.4))
            cand = _mixed_seq(n, exponent=exponent, shift=shift, alpha=alpha)

        elif move_type == "block":
            cand = _make_block_perturbation(best, rng)

        elif move_type == "swap":
            cand = best.copy()
            i, j = rng.choice(n, size=2, replace=False)
            if cand[i] > 0.0:
                eps = swap_step * cand[i] * rng.uniform(0.2, 0.8)
                cand[i] = max(cand[i] - eps, 0.0)
                cand[j] = cand[j] + eps
            cand /= cand.sum()

        else:   # "target"
            cand = _targeted_move(best, rng, step_factor)

        # cheap check – skip if the fast proxy is not better
        fast_sc = _fast_score(cand)
        if fast_sc >= best_score:
            continue

        # authoritative evaluation (includes clipping, NaN checks, …)
        try:
            true_sc = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue

        if np.isfinite(true_sc) and true_sc < best_score:
            best = cand.copy()
            best_score = true_sc
            print(f"[{iteration:6d}] new best {best_score:.6f} (move={move_type})")

        # schedule – slowly shrink the aggressive step sizes
        if iteration % 250 == 0:
            step_factor = max(step_factor * 0.92, 0.008)
            swap_step = max(swap_step * 0.94, 0.015)

    print(f"[final] best score = {best_score:.6f} after {iteration} iterations")
    # Return as a plain Python list (the harness expects that)
    return [float(v) for v in best]
