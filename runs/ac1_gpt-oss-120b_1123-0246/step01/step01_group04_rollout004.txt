def _project_simplex(v, s=1):
    """Project onto simplex {x>=0, sum x = s}."""
    # as per algorithm by Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

The evaluation function  

\[
E(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is *scale‑invariant*: multiplying the whole vector by a constant does not change the
score.  Consequently we may keep the ℓ₁‑norm fixed, e.g.  

\[
\sum_i a_i = \sqrt{2n}\;.
\]

If we denote  

\[
r_k(a)=\sum_i a_i\,a_{k-i},
\]

the objective is `max_k r_k(a)`.  For a fixed `k` the map `a → r_k(a)` is a
quadratic form and a sub‑gradient of `max_k r_k` at a point `a` is simply the
gradient of the *active* lag – the index `k*` where the maximum is attained.
Hence a classic sub‑gradient descent with projection onto the simplex
`{a≥0, ∑a = √(2n)}` is a natural deterministic improvement over a purely random
search.

The algorithm is a hybrid of three ingredients

1. **Deterministic sub‑gradient step** – reduces the current worst
   convolution value.
2. **Stochastic neighbourhood moves** – block perturbations, tiny mass‑swaps
   and a mixture of a uniform vector with a power‑law vector.  
   They help to escape shallow local minima.
3. **Fast evaluation** – the convolution max is obtained by an FFT in
   `O(n log n)`.  Only when a candidate beats the best known value the
   official `evaluate_sequence` is called.

The whole search runs for the allotted time (default ≈ 1000 s).  
All helper functions are top‑level, there are no closures or lambdas, and the
code respects the required API (`propose_candidate`).  

```python
"""
Hybrid deterministic / stochastic optimisation for the step‑function
evaluation problem.

The algorithm keeps the ℓ₁‑norm fixed to √(2 n) and repeatedly applies
  * a sub‑gradient descent step that directly reduces the currently
    maximal autocorrelation entry,
  * random block perturbations,
  * tiny mass swaps,
  * mixtures of a uniform vector with a power‑law vector.

Fast FFT‑based convolution supplies the objective and its active lag.
Every candidate that improves the best known score is checked with the
authoritative ``evaluate_sequence``.  The programme respects a time budget
(≈1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Fast convolution utilities
# ------------------------------------------------------------

def _conv_full(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of ``seq`` with itself, using FFT."""
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    return conv.real


def _max_conv_and_index(seq: np.ndarray) -> tuple[float, int]:
    """Return (max(conv), index_of_max) for the self‑convolution of ``seq``."""
    conv = _conv_full(seq)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _fast_score(seq: np.ndarray) -> float:
    """Score identical to ``evaluate_sequence`` but without the safety checks."""
    n = seq.size
    s = float(seq.sum())
    if s == 0.0:
        return float("inf")
    max_conv = _max_conv_and_index(seq)[0]   # only the value is needed
    return 2.0 * n * max_conv / (s * s)


# ------------------------------------------------------------
# Geometry: projection onto the simplex {x≥0, sum(x)=S}
# ------------------------------------------------------------

def _project_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=S}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if S < 0:
        raise ValueError("Simplex radius must be non‑negative")

    # work on a copy to avoid side‑effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.nonzero(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Simple analytic families – used for initialisation and random
# ------------------------------------------------------------

def _power_law_seq(n: int, exponent: float = 0.5, shift: float = 0.0) -> np.ndarray:
    """a_i ∝ (i + shift + 1)^{-exponent}, normalised to ℓ₁‑norm = 1."""
    idx = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power(idx + shift + 1.0, exponent)
    vals /= vals.sum()
    return vals


def _mixed_seq(n: int, exponent: float, shift: float, alpha: float) -> np.ndarray:
    """alpha·uniform + (1‑alpha)·power‑law, ℓ₁‑norm = 1."""
    uniform = np.full(n, 1.0 / n)
    power = _power_law_seq(n, exponent=exponent, shift=shift)
    mixed = alpha * uniform + (1.0 - alpha) * power
    mixed /= mixed.sum()
    return mixed


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """Random contiguous block multiplied by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    w_low = max(1, n // 200)
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    scale = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= scale
    if rng.random() < 0.35:                # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return cand


# ------------------------------------------------------------
# Sub‑gradient descent on the maximal autocorrelation entry
# ------------------------------------------------------------

def _subgradient_step(seq: np.ndarray, step: float,
                      target_sum: float) -> np.ndarray:
    """
    Perform one sub‑gradient step that reduces the current maximal
    autocorrelation value.
    """
    n = seq.size
    # identify the active lag
    _, k_star = _max_conv_and_index(seq)

    # gradient of r_{k*}(a) = a * a at lag k*
    g = np.zeros_like(seq)
    i_start = max(0, k_star - (n - 1))
    i_end = min(k_star, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=int)
    g[i] = seq[k_star - i]               # a_{k*-i}
    g *= 2.0                             # ∂r/∂a_i = 2·a_{k*-i}

    # gradient descent step
    cand = seq - step * g

    # project back onto the simplex (keeps sum = target_sum, non‑negative)
    cand = _project_simplex(cand, target_sum)
    return cand


# ------------------------------------------------------------
# Main entry point required by the evaluation harness
# ------------------------------------------------------------

def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The function respects the supplied time budget
    (default ≈ 500 s, the caller may increase it up to the full 1000 s) and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------
    # Length and initial vector
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        n = prev.size if prev.ndim == 1 and prev.size > 0 else 2000
    else:
        n = 2000

    target_sum = np.sqrt(2.0 * n)                 # ℓ₁‑norm we keep fixed

    # start from a classic 1/√i shape (scaled to the target sum)
    cur = _power_law_seq(n, exponent=0.5, shift=0.0) * target_sum
    cur = _project_simplex(cur, target_sum)      # safety projection

    # initial best (authoritative evaluation)
    best = cur.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------
    # Optional seeding with an external construction
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand_ext = np.asarray(height_sequence_1, dtype=np.float64)
            if cand_ext.ndim == 1 and cand_ext.size == n:
                cand_ext = _project_simplex(cand_ext, target_sum)
                sc_ext = float(evaluate_sequence(cand_ext.tolist()))
                if sc_ext < best_score:
                    best = cand_ext.copy()
                    best_score = sc_ext
        except Exception:
            pass

    print(f"[init] n={n}, start score={best_score:.6f}")

    # --------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------
    it = 0
    swap_step = 0.12          # fraction of source mass moved in a swap
    subgrad_initial_step = 0.7

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---------- choose a move ----------
        move_type = rng.choice(["subgrad", "block", "swap", "mix"],
                               p=[0.35, 0.40, 0.15, 0.10])

        if move_type == "subgrad":
            # deterministic descent on the active lag
            step = subgrad_initial_step / np.sqrt(it)
            cand = _subgradient_step(cur, step, target_sum)

        elif move_type == "block":
            # random block perturbation of the current state
            cand = _make_block_perturbation(cur, rng)
            cand = _project_simplex(cand, target_sum)

        elif move_type == "swap":
            # tiny mass move between two positions
            cand = cur.copy()
            i, j = rng.choice(n, size=2, replace=False)
            if cand[i] > 0.0:
                eps = swap_step * cand[i] * rng.uniform(0.2, 0.8)
                cand[i] -= eps
                cand[j] += eps
            cand = _project_simplex(cand, target_sum)

        else:   # "mix"
            # mixture of uniform and a power‑law vector
            exponent = float(rng.uniform(0.20, 0.80))
            shift = float(rng.exponential(5.0))
            alpha = float(rng.uniform(0.0, 0.4))
            cand = _mixed_seq(n, exponent=exponent, shift=shift, alpha=alpha)
            cand = cand * target_sum               # enforce the ℓ₁‑norm
            # projection is cheap but guarantees the exact sum
            cand = _project_simplex(cand, target_sum)

        # ---------- cheap score ----------
        fast_sc = _fast_score(cand)

        # ---------- update the global best ----------
        if fast_sc < best_score:
            # authoritative check only when we have a chance to improve
            true_sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(true_sc) and true_sc < best_score:
                best = cand.copy()
                best_score = true_sc
                print(f"[{it:6d}] new best score {best_score:.6f} (type={move_type})")

        # ---------- decide whether to keep the candidate as the new current ----------
        cur_fast = _fast_score(cur)
        if fast_sc <= cur_fast:
            cur = cand.copy()
        else:
            # occasional uphill move to avoid stagnation
            if rng.random() < 0.001:
                cur = cand.copy()

        # ---------- adapt auxiliary parameters ----------
        if it % 400 == 0:
            swap_step = max(swap_step * 0.85, 0.01)

    print(f"[final] iterations={it}, best score={best_score:.6f}")
    # Return a plain Python list as required by the harness
    return [float(v) for v in best]
```
