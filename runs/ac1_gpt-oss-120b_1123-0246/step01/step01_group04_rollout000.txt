def _generate_sequence(n, params):
    alpha, exponent, shift, beta = params
    alpha = np.clip(alpha, 0, 1)
    exponent = np.clip(exponent, 0.05, 2.0)
    shift = np.clip(shift, 0.0, 500.0)
    beta = max(beta, 0.0)
    uniform = np.full(n, 1.0/n)
    idx = np.arange(n, dtype=np.float64)
    power = 1.0 / np.power(idx + shift + 1.0, exponent)  # shape
    exp_factor = np.exp(-beta * idx)
    seq = alpha * uniform + (1.0 - alpha) * power * exp_factor
    seq = np.clip(seq, 0.0, 1000.0)
    seq_sum = seq.sum()
    if seq_sum > 0:
        seq /= seq_sum
    else:
        seq = uniform
    return seq
```

def propose_candidate(seed=123, budget_s=500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-2.0)

    # Pick n
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.shape[0]
    else:
        n = 5000

    # initialize best with power law shape.
    best = _power_law_seq(n, exponent=0.5, shift=0.0)
    best_score = float(evaluate_sequence(best.tolist()))
    # also evaluate base if exists
    if "height_sequence_1" in globals():
        base_seq = np.clip(base,0.0,1000.0)
        # pad/trim to length n
        if base_seq.size != n:
            base_seq = np.resize(base_seq, n) # maybe pad/truncate
        base_score = float(evaluate_sequence(base_seq.tolist()))
        if base_score < best_score:
            best = base_seq.copy()
            best_score = base_score

    print(f"Starting search length {n}, initial best {best_score}")

    # define objective for DE
    def objective(params):
        seq = _generate_sequence(n, params)
        sc = _fast_score(seq)
        return sc

    # DE bounds
    bounds = [(0.0, 1.0),   # alpha
              (0.05, 2.0), # exponent
              (0.0, 500.0),# shift
              (0.0, 0.02)] # beta

    # Use SciPy DE with limited iterations
    try:
        from scipy.optimize import differential_evolution
        result = differential_evolution(objective, bounds,
                                        seed=seed,
                                        maxiter=30,
                                        popsize=10,
                                        polish=False,
                                        updating='deferred',  # to parallelize?
                                        workers=2)
    except Exception as e:
        print("SciPy DE not available or error:", e)
        result = None

    if result is not None and result.success:
        seq_de = _generate_sequence(n, result.x)
        score_de = float(evaluate_sequence(seq_de.tolist()))
        if score_de < best_score:
            best = seq_de.copy()
            best_score = score_de
            print(f"DE improved to {best_score}")

    # Now local search with block perturbations and mass swaps similar to previous.
    # Parameter for block width adapt.
    iteration = 0
    swap_step = 0.1
    while time.monotonic() < deadline:
        iteration += 1
        move = rng.choice(["block","swap","param"], p=[0.45,0.35,0.20])
        if move =="block":
            cand = _make_block_perturbation(best, rng)
        elif move=="swap":
            cand = best.copy()
            i,j = rng.choice(n, size=2, replace=False)
            if cand[i] > 0:
                eps = swap_step * cand[i] * rng.uniform(0.2,0.8)
                cand[i] = max(cand[i] - eps, 0)
                cand[j] += eps
            cand /= cand.sum()
        else: # param: random fresh param generation
            alpha = float(rng.random())
            exp = float(rng.uniform(0.2,1.2))
            shift = float(rng.exponential(10.0))
            beta = float(rng.uniform(0.0,0.02))
            cand = _generate_sequence(n, [alpha, exp, shift, beta])
        # fast score prune
        fast_sc = _fast_score(cand)
        if fast_sc >= best_score:
            continue
        # evaluate authoritatively
        try:
            true_sc = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue
        if true_sc < best_score:
            best = cand.copy()
            best_score = true_sc
            print(f"[{iteration}] new best {best_score:.6f} via {move}")
        # adapt swap step
        if iteration % 200 ==0:
            swap_step *= 0.9
            swap_step = max(swap_step, 0.01)

    print(f"Finished with best {best_score} after {iteration} iterations")
    return [float(v) for v in best]

"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  

Algorithm (≈ 2‑stage hybrid):
-------------------------------------------------
1.  **Parameterised shape search (global).**  
    The sequence is generated from four real parameters  
       *α* – weight of a uniform component,  
       *p* – exponent of a power‑law,  
       *s* – additive shift in the denominator,  
       *β* – exponential tail factor.  

    a_i ∝ α·(1/n) + (1‑α)·(i+1+s)^{-p}·exp(-β·i) ,  i=0,…,n‑1  

    For a fixed length *n* the shape is normalised to L¹‑norm = 1, so the
    score reduces to `2·n·max(conv)` and does not depend on the absolute
    scale.  A low‑dimensional DE (Differential Evolution) optimiser
    searches the 4‑dimensional parameter space, using a fast FFT based
    objective (`_fast_score`).  The DE runs only a few hundred
    function‑evaluations, which is cheap even for n≈7000.

2.  **Local refinement (stochastic hill‑climbing).**  
    Starting from the best DE candidate (or from the previously stored
    `height_sequence_1` if it is better) we perform cheap random moves
    until the time budget expires:

    * *block perturbation* – multiply a randomly chosen contiguous block
      by a log‑normal factor, optionally symmetrise the whole vector.
    * *mass‑swap* – move a fraction of the mass from one index to another.
    * *parameter perturbation* – generate a fresh random parameter set
      (same form as in step 1) and evaluate it.

    The fast score is used as a cheap filter; only improving candidates
    are checked with the authoritative `evaluate_sequence`.  The step
    size of the mass‑swap is slowly decreased to allow fine‑tuning.

The two stages are combined inside `propose_candidate`.  The code is
fully self‑contained, uses only NumPy (and SciPy if available for DE) and
conforms to the required interface.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core numeric helpers (top‑level, no closures)
# ----------------------------------------------------------------------


def _max_convolution(seq: np.ndarray) -> float:
    """
    Compute max value of the discrete convolution seq * seq.
    Uses zero‑padded FFT (real‑valued) for O(n log n) speed.
    """
    n = seq.size
    # length of full convolution = 2*n-1, round up to next power of two
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    return float(np.max(conv.real))


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast analogue of `evaluate_sequence` (no safety checks).
    The sequence is assumed to be non‑negative; the sum is used as the
    denominator.  The score is scale‑invariant, therefore we do not
    re‑scale to any particular L¹‑norm.
    """
    s = float(seq.sum())
    if s < 1e-12:
        return float("inf")
    max_conv = _max_convolution(seq)
    n = seq.size
    return 2.0 * n * max_conv / (s * s)


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """
    Return a copy of `base` where a random contiguous block is multiplied
    by a log‑normal factor.  Occasionally the whole sequence is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    w_low = max(1, n // 200)          # a few percent of n
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal perturbation
    scale = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= scale
    # occasional symmetrisation (helps reduce peak of autocorrelation)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # clip to the same bounds that `evaluate_sequence` enforces
    np.clip(cand, 0.0, 1000.0, out=cand)
    # renormalise to L¹ = 1 (scale does not affect the score)
    cand_sum = cand.sum()
    if cand_sum > 0:
        cand /= cand_sum
    return cand


def _power_law_seq(n: int,
                   exponent: float = 0.5,
                   shift: float = 0.0) -> np.ndarray:
    """Return a normalised power‑law sequence a_i ∝ (i+shift)^{-exponent}."""
    idx = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power(idx + shift + 1.0, exponent)  # +1 avoids div‑by‑0
    if vals.sum() == 0.0:
        vals = np.ones_like(vals)
    vals /= vals.sum()
    return vals


def _generate_sequence(n: int, params: np.ndarray) -> np.ndarray:
    """
    Build a candidate sequence from a 4‑vector of parameters:

        params = [alpha, p, shift, beta]

    The formula is
        a_i ∝ α·(1/n) + (1‑α)·(i+1+shift)^{-p}·exp(-β·i)

    The result is clipped to [0,1000] and normalised to L¹ = 1.
    """
    alpha, p, shift, beta = params
    # enforce bounds manually (DE may generate values slightly outside)
    alpha = float(np.clip(alpha, 0.0, 1.0))
    p = float(np.clip(p, 0.05, 2.0))
    shift = float(max(shift, 0.0))
    beta = float(max(beta, 0.0))

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    power = 1.0 / np.power(idx + shift + 1.0, p)
    exp_factor = np.exp(-beta * idx)
    seq = alpha * uniform + (1.0 - alpha) * power * exp_factor
    # Clip to the bounds used by `evaluate_sequence`
    np.clip(seq, 0.0, 1000.0, out=seq)
    total = seq.sum()
    if total > 0.0:
        seq /= total
    else:
        # fallback – uniform distribution
        seq = uniform.copy()
    return seq


class _DEObjective:
    """Callable wrapper for differential evolution (no closure)."""

    def __init__(self, n: int):
        self.n = n

    def __call__(self, params: np.ndarray) -> float:
        seq = _generate_sequence(self.n, params)
        return _fast_score(seq)

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The function respects the supplied time budget
    (seconds) and returns a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (length of the sequence)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base_seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if base_seq.ndim != 1 or base_seq.size == 0 or not np.isfinite(base_seq).all():
            n = 2000
            base_seq = np.empty(0, dtype=np.float64)
        else:
            n = base_seq.size
    else:
        n = 2000
        base_seq = np.empty(0, dtype=np.float64)

    # ------------------------------------------------------------------
    # Initial candidate(s)
    # ------------------------------------------------------------------
    # analytic start: pure power law with exponent 0.5
    current = _power_law_seq(n, exponent=0.5, shift=0.0)
    best = current.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # also test the previously stored sequence (if any)
    if base_seq.size == n:
        try:
            base_score = float(evaluate_sequence(base_seq.tolist()))
            if base_score < best_score:
                best = base_seq.copy()
                best_score = base_score
        except Exception:
            pass

    print(f"[init] length={n}, initial best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – global DE on the 4‑parameter family
    # ------------------------------------------------------------------
    de_obj = _DEObjective(n)
    bounds = [
        (0.0, 1.0),    # alpha
        (0.05, 2.0),   # exponent p
        (0.0, 500.0),  # shift
        (0.0, 0.02),   # beta
    ]

    # Set a safe deadline a few seconds before the absolute limit
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    try:
        from scipy.optimize import differential_evolution

        # DE is cheap for this low‑dimensional problem; we cap
        # the total number of generations so that it does not dominate the time.
        de_result = differential_evolution(
            de_obj,
            bounds,
            seed=seed,
            maxiter=80,          # generations
            popsize=15,          # population size (default 15)
            polish=False,        # we do our own polishing later
            updating='deferred',  # allows parallel workers without extra memory copies
            workers=2,
            atol=1e-12,
            tol=1e-12,
        )
        if de_result.success:
            seq_de = _generate_sequence(n, de_result.x)
            # Authoritative evaluation
            true_sc = float(evaluate_sequence(seq_de.tolist()))
            if true_sc < best_score:
                best = seq_de.copy()
                best_score = true_sc
                print(f"[DE ] improved to {best_score:.6f}")
    except Exception as e:
        # SciPy may be unavailable in the execution environment
        print(f"[DE ] skipping (SciPy not usable): {e}")

    # ------------------------------------------------------------------
    # Stage 2 – stochastic local improvement
    # ------------------------------------------------------------------
    iteration = 0
    swap_step = 0.10  # initial fraction of source mass to move

    while time.monotonic() < deadline:
        iteration += 1
        move_type = rng.choice(["block", "swap", "param"], p=[0.45, 0.35, 0.20])

        if move_type == "block":
            cand = _make_block_perturbation(best, rng)

        elif move_type == "swap":
            cand = best.copy()
            i, j = rng.choice(n, size=2, replace=False)
            if cand[i] > 0.0:
                eps = swap_step * cand[i] * rng.uniform(0.2, 0.8)
                cand[i] = max(cand[i] - eps, 0.0)
                cand[j] += eps
            # re‑normalise (scale does not affect the score)
            total = cand.sum()
            if total > 0.0:
                cand /= total

        else:  # "param" – draw a fresh random param vector
            alpha = float(rng.random())
            exponent = float(rng.uniform(0.2, 1.2))
            shift = float(rng.exponential(10.0))
            beta = float(rng.uniform(0.0, 0.02))
            cand = _generate_sequence(n, np.array([alpha, exponent, shift, beta]))

        # Fast pruning: if the cheap score is not better, skip the expensive call
        fast_sc = _fast_score(cand)
        if fast_sc >= best_score:
            continue

        # Authoritative check (includes clipping, sum‑check, etc.)
        try:
            true_sc = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue  # treat as invalid

        if np.isfinite(true_sc) and true_sc < best_score:
            best = cand.copy()
            best_score = true_sc
            print(f"[{iteration:6d}] new best {best_score:.6f} via {move_type}")

        # Gradually tighten the swap step so we can fine‑tune later
        if iteration % 200 == 0:
            swap_step = max(swap_step * 0.9, 0.01)

    print(f"[final] best score = {best_score:.6f} after {iteration} iterations")
    # Return a plain Python list (evaluate_sequence expects a list of floats)
    return [float(v) for v in best]
