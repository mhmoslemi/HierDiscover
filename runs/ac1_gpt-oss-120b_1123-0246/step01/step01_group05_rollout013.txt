function project_simplex(v):
    - Input 1D np.ndarray of real numbers.
    - Return projected onto probability simplex sum=1, all >= 0.

Algorithm: sort v descending to get u, compute cumulative sum, find rho = max {j | u_j + (1/(j+1)) * (1 - sum_{i<=j} u_i) > 0}. Or known O(n log n) algorithm.

Implementation:

def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    """Project a 1-D array onto the probability simplex."""
    # https://arxiv.org/abs/1101.6081
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) that tries to minimise
`evaluate_sequence`.  The search consists of three stages:

1. **Coarse grid search** over a two‑parameter family
      a[i] ∝ (i + 1 + offset)^{-α}
   where the exponent α∈[0.35,0.8] and the integer offset shifts the whole
   power‑law away from the first index.  The best candidate of this cheap
   sweep is kept.

2. **Targeted local refinement** – the current maximal convolution value
   occurs at some shift k*.  The contribution of index i to that peak is
   c_i = a_i·a_{k*−i}.  We repeatedly move a small fraction of the mass
   from the index with the largest contribution to an index with a tiny
   (or zero) contribution, re‑projecting onto the probability simplex.
   Because only the pair that creates the current maximum is weakened,
   this phase can lower the bound substantially.

3. **Randomised perturbations** as a fallback when the targeted move
   cannot find an improvement.  Small random transfers of mass between
   two random positions keep the search stochastic and help to escape
   tiny plateaus.

All operations respect the required constraints (non‑negative, sum ≥0.01)
and the whole routine stops a little before the time budget (default
≈ 950 s) to guarantee a return.  The final sequence is returned as a list
of floats; its overall scale is irrelevant because the evaluation function
is scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – projection onto the probability simplex {x≥0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of vector v onto the simplex."""
    # reference: algorithm of Wang & Carreira‑Perpiñán (2013)
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are (almost) zero – return uniform distribution
        return np.full_like(v, 1.0 / n, dtype=np.float64)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Power‑law family with an optional integer offset
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """Return a normalised (sum = 1) power‑law vector of length n.

    a[i] ∝ (i+1+offset)^{-α}.
    """
    idx = np.arange(n, dtype=np.float64) + 1.0 + float(offset)
    w = np.power(idx, -alpha)
    w_sum = w.sum()
    if w_sum == 0:
        # should never happen for the chosen ranges
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / w_sum


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950.0) -> list[float]:
    """Search for a step‑function with a low evaluation score.

    The routine may use the global variable `height_sequence_1` (if present)
    as a starting point, otherwise it builds candidates from a parametrised
    power‑law family and refines them with a simple hill‑climbing scheme.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a tiny safety margin for the final return
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Choose a working length.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name‑defined]
        n = max(256, min(n, 8000))
    except Exception:
        n = 4096

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0️⃣  Try the previously supplied construction (if any)
    # --------------------------------------------------------------
    try:
        cand = [float(x) for x in height_sequence_1]   # type: ignore[name‑defined]
        sc = evaluate_sequence(cand)
        if np.isfinite(sc) and sc < best_score:
            best_seq = np.array(cand, dtype=np.float64)
            best_score = float(sc)
            print(f"starting from global seq → score={best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1️⃣  Coarse grid over (α, offset)
    # --------------------------------------------------------------
    alphas = np.linspace(0.35, 0.80, 23)          # 0.35 … 0.80
    offset_fracs = np.linspace(0.0, 0.12, 7)     # up to ~12 % of n
    for alpha in alphas:
        if time.monotonic() >= deadline:
            break
        for frac in offset_fracs:
            offset = int(frac * n)
            seq = _power_law_sequence(n, alpha, offset)
            # evaluate – the evaluator is scale‑invariant, keep sum = 1
            sc = evaluate_sequence(seq.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_seq = seq.copy()
                print(f"grid α={alpha:.3f} off={offset:4d}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  If nothing succeeded, fall back to a uniform vector
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"fallback uniform → score={best_score:.6f}")

    # Work with a probability vector (sum = 1)
    a = best_seq.copy()
    a = a / a.sum()

    # --------------------------------------------------------------
    # 3️⃣  Targeted local refinement – move mass away from the
    #      dominant convolution pair.
    # --------------------------------------------------------------
    # control parameters
    step_frac = 0.20          # fraction of source mass to move
    no_improve = 0
    max_no_improve = 200      # after many failures switch to random moves

    while time.monotonic() < deadline:
        # current convolution and the worst shift
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))

        # contribution of each index to that maximal entry
        idx_arr = np.arange(n)
        j = k_max - idx_arr
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[j[mask]]

        src = int(np.argmax(contrib))
        # indices that hardly contribute to the current peak
        low_thr = contrib[src] * 0.01   # 1 % of the worst contribution
        low_cands = np.where(contrib <= low_thr)[0]

        if low_cands.size == 0:
            # fall back to a random target different from src
            tgt = rng.integers(0, n)
            while tgt == src:
                tgt = rng.integers(0, n)
        else:
            tgt = rng.choice(low_cands)
            if tgt == src:
                tgt = (tgt + 1) % n

        delta = a[src] * step_frac
        if delta <= 1e-12:
            break

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        # numerical noise may create tiny negatives – clip and re‑project
        cand = np.clip(cand, 0.0, None)
        cand = _project_simplex(cand)

        sc = evaluate_sequence(cand.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            a = cand
            no_improve = 0
            # slowly increase the step – we were successful
            step_frac = min(0.5, step_frac * 1.05)
            print(f"targeted src={src:4d}→tgt={tgt:4d}  score={best_score:.6f}")
        else:
            no_improve += 1
            # on failure shrink the step size
            step_frac = max(0.01, step_frac * 0.93)

        # ------------------------------------------------------------------
        # Switch to a purely random local move after a while of stagnation.
        # ------------------------------------------------------------------
        if no_improve >= max_no_improve:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = a[i] * 0.15
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            sc = evaluate_sequence(cand.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                a = cand
                print(f"random shake i={i:4d}→j={j:4d}  score={best_score:.6f}")
            no_improve = 0   # reset counter after a random trial

    # --------------------------------------------------------------
    # 4️⃣  Return the best sequence (any positive scale works)
    # --------------------------------------------------------------
    final_seq = (a * 100.0).tolist()            # keep the same scale as the
                                                # original code for readability
    final_score = evaluate_sequence(final_seq)
    print(f"final score={final_score:.6f}   length={len(final_seq)}")
    return [float(x) for x in final_seq]
