    n = len(height_sequence_1) if defined else 256
    # start with arcsine shape
    a = build_arcsine_sequence(n)

    best = a.copy()

    start_time = time.time()
    deadline = start_time + budget_s - 2

    # define utility to evaluate quickly for array
    # We'll reuse evaluate_sequence

    # main loop: alternating LP + stochastic moves
    while time.time() < deadline:
        # compute conv and M
        conv = np.convolve(best, best)
        M = np.max(conv)
        # Solve LP to find g maximizing sum under same M
        g = solve_lp_max_sum(best, M)
        # Compare sum
        sum_best = best.sum()
        sum_g = g.sum()
        # If g sum > best sum, mix
        if sum_g > sum_best:
            # mixing factor t maybe 0.5
            t = 0.3
            cand = (1-t) * best + t * g
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best = cand
                best_score = score
                print(f"[LP-improve] new best {best_score:.6f}")
                continue

        # else do random move
        if rng.random() < 0.5:
            cand = block_perturbation(best, rng)

        score = evaluate_sequence(cand.tolist())

            best_score = score
            print(f"[rand-improve] new best score {best_score:.6f}")
        # occasional symmetrize
        if rng.random() < 0.1:
            best = 0.5*(best + best[::-1])
    # final best
    return best.tolist()

def max_reduction_move(a, conv, rng):
    k = np.argmax(conv)
    # Find i where a_i * a_{k-i} maximal
    start = max(0, k-(len(a)-1))
    end = min(len(a)-1, k)
    # create array of indices i in [start..end]
    idxs = np.arange(start, end+1)
    partners = k - idxs
    valid = (partners >=0) & (partners < len(a))
    idxs = idxs[valid]
    partners = partners[valid]
    products = a[idxs] * a[partners]
    max_idx = int(idxs[np.argmax(products)])
    # choose corresponding j
    j = int(k - max_idx)
    # if i==j adjust to avoid double reduction? we can treat as single.
    # Determine delta = fraction of smaller of a[max_idx], a[j]
    delta = rng.random() * 0.2 * min(a[max_idx], a[j])
    # decrement both by delta, add 2*delta to another index l
    l = int(rng.integers(0, len(a)))
    new = a.copy()
    new[max_idx] -= delta
    new[j] -= delta
    new[l] += 2*delta
    # Clip
    return np.clip(new, 0.0, 1000.0)

"""
A simulated‑annealing / targeted‑move optimizer for the convolution‑bound
problem.

The evaluator is scale‑invariant, therefore the algorithm works only with the
shape of the step function.  The routine starts from the analytically optimal
“arcsine” (inverse‑square‑root) sequence, then iteratively applies cheap
stochastic moves:

* block_perturbation – multiplicative scaling of a random contiguous block,
* transfer_move      – move a fraction of the mass from one index to another,
* swap_move          – exchange two entries,
* targeted_move      – identify the shift k that currently attains the maximal
  convolution, pick the pair (i, k‑i) with the largest product a[i]·a[k‑i] and
  transfer a small amount of mass away from that pair.

All moves keep the total sum unchanged, i.e. they preserve the denominator of
the evaluation function.  Moves are accepted according to a simple
Metropolis rule with a linearly‑cooled temperature, which allows temporary
worsening steps and helps escape local minima.

The best sequence found before the time budget expires is returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Analytic start – the arcsine (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i  (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    # recurrence: a[i] = a[i‑1] * (2i‑1)/(2i)
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# 2.  Simple stochastic moves.
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrisation (helps stability)
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another (sum unchanged)."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    # move at most 30 % of the source
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def swap_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Exchange two entries – useful for reshuffling large values."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    cand = base.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand


def targeted_move(base: np.ndarray,
                  conv: np.ndarray,
                  k_max: int,
                  rng: np.random.Generator) -> np.ndarray:
    """
    Reduce the dominant convolution entry.
    - Find the pair (i, j) with i+j = k_max that gives the largest product.
    - Move a small amount of mass away from that pair to two random positions.
    The total sum stays unchanged.
    """
    n = base.size
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idxs = np.arange(i_start, i_end + 1)
    partners = k_max - idxs
    products = base[idxs] * base[partners]
    # index of the maximal contribution
    max_pos = int(np.argmax(products))
    i = int(idxs[max_pos])
    j = int(partners[max_pos])

    # choose two distinct target positions
    l1, l2 = rng.choice(n, size=2, replace=False)

    # move a small fraction of the smaller participant
    # (the fraction itself is random to avoid getting stuck)
    max_move = min(base[i], base[j])
    if max_move == 0:
        return base.copy()
    delta = rng.random() * 0.20 * max_move   # at most 20 % of the smaller entry

    cand = base.copy()
    if i == j:
        # both indices identical – we remove 2*delta from the same slot
        cand[i] -= 2.0 * delta
        cand[l1] += delta
        cand[l2] += delta
    else:
        cand[i] -= delta
        cand[j] -= delta
        cand[l1] += delta
        cand[l2] += delta
    return np.clip(cand, 0.0, 1000.0)

    lower than the baseline arcsine construction.
    The routine runs a simulated‑annealing loop for up to *budget_s* seconds.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size – use the global reference if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = int(len(globals()["height_sequence_1"]))
        # use the provided sequence as an alternative start if it is valid
        try:
            # quick sanity check – the harness already applied clipping,
            # but we still make sure the length matches.
            init_seq = np.array(height_sequence_1, dtype=np.float64)
            if init_seq.shape[0] != n:
                raise ValueError
        except Exception:
            init_seq = build_arcsine_sequence(n)
    else:
        n = 256
        init_seq = build_arcsine_sequence(n)

    # Ensure the start respects the evaluator's clipping rules.
    base_seq = np.clip(init_seq, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Initial evaluation.
    # ------------------------------------------------------------------
    best = base_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur = best.copy()
    cur_score = best_score
    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # a tiny safety margin
    it = 0

    # Temperature schedule – start relatively warm and cool linearly.
    T_start = 0.02
    T_end = 0.001

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1.  Choose a move.
        # --------------------------------------------------------------
        move_type = rng.random()
        if move_type < 0.25:
            cand = block_perturbation(cur, rng)
        elif move_type < 0.50:
            cand = transfer_move(cur, rng)
        elif move_type < 0.70:
            cand = swap_move(cur, rng)
        else:
            # targeted reduction of the current maximum
            conv = np.convolve(cur, cur)
            k_max = int(np.argmax(conv))
            cand = targeted_move(cur, conv, k_max, rng)

        # --------------------------------------------------------------
        # 2.  Evaluate the candidate.
        # --------------------------------------------------------------
        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
            # safety fallback – ignore illegal candidates
            continue

        # --------------------------------------------------------------
        # 3.  Acceptance test (simulated annealing).
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        frac = elapsed / float(budget_s)
        T = T_start * (1.0 - frac) + T_end * frac   # linear cooling

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis accept probability
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                print(f"[{it:>6}] new best = {best_score:.6f}")

                # stop early if we already beat the target
                if best_score < 1.503:
                    print("[early] target reached")
                    break

        # occasional global perturbation to avoid deep stagnation
        if it % 5000 == 0 and rng.random() < 0.5:
            # re‑seed from the arcsine baseline (still respects the best seen)
            cur = np.clip(build_arcsine_sequence(n), 0.0, 1000.0)

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    return [float(x) for x in best]
