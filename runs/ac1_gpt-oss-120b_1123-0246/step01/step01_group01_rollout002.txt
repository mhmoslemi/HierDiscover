def fast_evaluation(seq: np.ndarray) -> float:
    n = seq.size
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    # Next power of 2 >= 2*n - 1
    m = 1 << ((2*n - 1).bit_length())
    # compute rfft
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)
    max_b = np.max(conv[:2*n-1])
    return 2.0 * n * max_b / (sum_a * sum_a)
```

def fast_score(seq):
    n = seq.size
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (sum_a * sum_a)

"""
Propose a step‑function sequence with a low evaluation score.

The search uses a fast Fourier‑transform based evaluation of the metric
(avoiding the costly Python‑list conversion of `evaluate_sequence`).  The
algorithm starts from the analytically optimal binomial coefficients
(a_i = C(2i,i)/4^i) or from a previously found global sequence if it
exists.  It then performs a timed simulated‑annealing / hill‑climb that
combines three move types:

1. **Gradient move** – identifies the convolution index where the bound
   is maximal, builds the corresponding gradient (which is just the
   reversed slice of the current sequence) and transfers a fraction of
   mass from an index whose gradient is large to one whose gradient is
   small.  This directly reduces the dominant convolution entry.

2. **Block perturbation** – multiplies a random contiguous block by a
   log‑normal factor, allowing larger‑scale exploration.

3. **Random redistribution** – moves a small amount of mass between two
   arbitrary positions.

The current state is accepted according to a simple temperature schedule
(linear decay) so that occasional uphill moves are possible early on.
The best sequence seen so far is kept and returned at the end of the
budget.  All values are kept inside the allowed range [0, 1000] and are
non‑negative, satisfying the security checks of `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _binomial_sequence(length: int) -> np.ndarray:
    """Truncated binomial sequence a_i = C(2i,i) / 4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(length - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def _full_convolution(seq: np.ndarray) -> np.ndarray:
    """
    Linear convolution of seq with itself using FFT.
    Returns the 2·n‑1 length result (real numbers).
    """
    n = seq.size
    size = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    return conv


def fast_score(seq: np.ndarray) -> float:
    """
    Fast replica of `evaluate_sequence` that works directly on a NumPy array.
    """
    sum_a = float(np.sum(seq))
    if sum_a < 0.01:
        return float("inf")
    n = seq.size
    conv = _full_convolution(seq)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random block."""
    cand = base.copy()
    n = cand.size
    # block width between ~0.5% and ~6% of the length
    w_min = max(1, n // 200)
    w_max = max(2, n // 15)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.10)))  # moderate log‑normal spread
    cand[start : start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def gradient_move(seq: np.ndarray, idx_max: int, rng: np.random.Generator) -> np.ndarray:
    """
    Transfer a fraction of mass from a donor (large gradient) to a recipient
    (small gradient) for the convolution index `idx_max`.  This tends to lower
    the dominant convolution entry.
    """
    n = seq.size
    # build gradient = reversed slice of seq that participates in idx_max
    lo = max(0, idx_max - (n - 1))
    hi = min(idx_max, n - 1)
    grad = np.zeros_like(seq)
    # range of i where both i and idx_max-i are valid
    i_range = np.arange(lo, hi + 1, dtype=int)
    grad[i_range] = seq[idx_max - i_range]

    # donors: positions with a positive value and relatively large gradient
    donor_mask = (seq > 0) & (grad > 0)
    if not np.any(donor_mask):
        # fallback – just return a copy
        return seq.copy()
    donor = int(np.argmax(grad * donor_mask))

    # recipients: positions with smallest gradient (could be zero)
    recipient = int(np.argmin(grad))

    # avoid a no‑op
    if donor == recipient:
        recipient = (donor + 1) % n

    # how much to move? up to a random fraction of the donor's mass
    max_transfer = seq[donor] * rng.random() * rng.uniform(0.1, 0.7)
    # respect the 1000 ceiling on the recipient
    delta = min(max_transfer, 1000.0 - seq[recipient])

    new_seq = seq.copy()
    new_seq[donor] -= delta
    new_seq[recipient] += delta
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq


def random_redistribute(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random amount of mass from one index to another."""
    n = seq.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j or seq[i] <= 0.0:
        return seq.copy()
    # move at most 30 % of the source value
    delta = seq[i] * rng.random() * 0.30
    delta = min(delta, 1000.0 - seq[j])
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 900.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    `evaluate_sequence`.  The function respects the given time budget and
    always returns a valid list; the best sequence found within the budget
    is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise the starting sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.size == 0:
            n = 256
            cur = _binomial_sequence(n)
        else:
            n = base.size
            cur = base.copy()
    else:
        n = 256
        cur = _binomial_sequence(n)

    # Ensure all entries respect the hard caps
    np.clip(cur, 0.0, 1000.0, out=cur)

    # Best‑so‑far bookkeeping
    best_seq = cur.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # Current state for the stochastic walk
    cur_seq = cur.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # 2. Timed stochastic search (simulated annealing style)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 3.0)  # leave a margin
    it = 0

    while time.time() < deadline:
        it += 1

        # compute convolution of the current state once (used for gradient)
        conv_cur = _full_convolution(cur_seq)
        idx_max = int(np.argmax(conv_cur))

        # ------------------------------------------------------------------
        # choose a move type
        # ------------------------------------------------------------------
        r = rng.random()
        if r < 0.45:
            cand_seq = gradient_move(cur_seq, idx_max, rng)
        elif r < 0.75:
            cand_seq = make_block_perturbation(cur_seq, rng)
        else:
            cand_seq = random_redistribute(cur_seq, rng)

        # evaluate the candidate
        cand_score = fast_score(cand_seq)

        # ------------------------------------------------------------------
        # Acceptance logic (always keep the globally best)
        # ------------------------------------------------------------------
        if cand_score < best_score:
            best_seq = cand_seq.copy()
            best_score = cand_score
            print(f"[{it:6d}] new global best: {best_score:.6f}")

        # Decide whether to move the current walk state
        if cand_score < cur_score:
            cur_seq = cand_seq
            cur_score = cand_score
        else:
            # simulated‑annealing acceptance with a linear temperature schedule
            time_left = deadline - time.time()
            fraction = max(0.0, time_left / (budget_s - 3.0))
            # temperature starts around 0.05 and decays to ~0
            temperature = fraction * 0.05
            # avoid division by zero
            if temperature > 0.0 and rng.random() < math.exp(-(cand_score - cur_score) / temperature):
                cur_seq = cand_seq
                cur_score = cand_score

        # occasional cheap progress report
        if it % 5000 == 0:
            print(f"[{it:6d}] iter, cur_score={cur_score:.6f}, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    print(f"[done] iterations={it}, time={elapsed:.1f}s, final best score={best_score:.6f}")
    return [float(x) for x in best_seq]
