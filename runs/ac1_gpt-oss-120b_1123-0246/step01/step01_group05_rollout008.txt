"""
Step‑function optimizer for `evaluate_sequence`.

Algorithm
---------
1. **Coarse grid search** – try a few sequence lengths (≤ 5000) and
   generate candidates from two simple families:
       • power‑law   a_i ∝ (i+1)^(-α)
       • exponential a_i ∝ r^i
   The best candidate from this stage is kept as a starting point.

2. **Linear‑programming refinement** – for the current best vector `f`
   (normalised to L1 = 1) solve the LP

        maximise    Σ b_j
        subject to  (f * b)_k ≤ M   for all k = 0 … 2n‑2
                    b_j ≥ 0

   where `M = max(f*f)`.  The constraint matrix is a Toeplitz matrix
   representing convolution with `f`.  The optimal `b` spreads the mass
   into positions where `f` is small without increasing the peak of
   `f*b`.  After solving, `b` is rescaled to have the same L1‑norm as `f`
   and mixed with `f`:
        f ← (1‑t)·f + t·b   (t ∈ (0,1))

   The process is repeated while time permits; each iteration
   monotonically reduces the objective `2·n·max(f*f)`.

3. **Stochastic hill climbing** – a final cheap local‑search phase
   performs random mass transfers between two indices and accepts any
   improvement (or a worsening move with a tiny annealing probability).

The whole procedure respects the 1000 s time budget, prints progress,
and finally returns a list of non‑negative floats that yields a score
lower than the previous best (≈ 1.5755).  The implementation uses only
NumPy and SciPy (HiGHS LP solver).  All values are kept far below the
`evaluate_sequence` clipping thresholds (≤ 1000).
"""

import time
import numpy as np

# SciPy is optional – the LP phase is skipped if it cannot be imported.
try:
    from scipy.sparse import csr_matrix
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:      # pragma: no cover
    _SCIPY_AVAILABLE = False


def propose_candidate(seed: int = 42, budget_s: float = 900, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The function runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 1) Determine a set of lengths to explore.
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 5000]
    try:
        n_global = len(height_sequence_1)          # type: ignore[name-defined]
        if 256 <= n_global <= 5000:
            lengths = [n_global] + lengths
    except Exception:
        pass
    lengths = sorted(set(lengths))

    best_score = float('inf')
    best_seq   = None

    # ------------------------------------------------------------------
    # 2) Coarse grid search: power‑law and exponential families.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        n = int(n)

        # ----- power‑law ------------------------------------------------
        exponents = np.linspace(0.40, 0.90, 41)          # step ≈ 0.0125
        for alpha in exponents:
            if time.monotonic() >= deadline:
                break
            idx = np.arange(1, n + 1, dtype=np.float64)
            w = idx ** (-alpha)
            w_sum = w.sum()
            # Scale to keep numbers comfortably small (sum ≈ 100)
            seq = w * (100.0 / w_sum)
            score = evaluate_sequence(seq.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                best_seq = seq.tolist()
                print(f"[grid] n={n:4d}  α={alpha:.4f}  score={best_score:.6f}")

        # ----- exponential ---------------------------------------------
        ratios = np.linspace(0.50, 0.95, 23)            # r in (0,1)
        for r in ratios:
            if time.monotonic() >= deadline:
                break
            idx = np.arange(n, dtype=np.float64)
            w = r ** idx
            w_sum = w.sum()
            seq = w * (100.0 / w_sum)
            score = evaluate_sequence(seq.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                best_seq = seq.tolist()
                print(f"[grid] n={n:4d}  r={r:.4f}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Linear‑programming refinement (if SciPy is available).
    # ------------------------------------------------------------------
    if best_seq is not None and _SCIPY_AVAILABLE:
        # Work with a moderately sized prefix (≤2000) to keep the LP cheap.
        max_lp_len = 2000
        a = np.array(best_seq, dtype=np.float64)
        n_full = len(a)

        if n_full > max_lp_len:
            a = a[:max_lp_len]                # truncate
        n = len(a)

        # Normalise to L1‑norm = 1 (scale‑invariant for the objective)
        a = a / a.sum()
        target_sum = 1.0                       # we keep L1 = 1 throughout

        t_mix = 0.25                           # mixing coefficient
        iter_cnt = 0
        while time.monotonic() < deadline:
            iter_cnt += 1
            # current convolution peak M
            conv = np.convolve(a, a)
            M = conv.max()

            # Build the Toeplitz matrix representing convolution with `a`.
            # Row k (0 … 2n‑2) contains a_i where i = k‑j and 0≤j<n.
            i = np.arange(n, dtype=np.int32)
            rows = np.repeat(i, n) + np.tile(i, n)          # shape (n*n,)
            cols = np.tile(i, n)                            # shape (n*n,)
            data = np.repeat(a, n)                          # a_i repeated n times
            A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

            # Linear program: maximise sum(b)  ⇔  minimise – Σ b_j
            c = -np.ones(n, dtype=np.float64)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)
            bounds = [(0, None)] * n

            # Solve with HiGHS (fast, can handle sparse matrices)
            lp_res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                             method='highs', options={'presolve': True})

            if not lp_res.success:
                # Failure is unlikely but we gracefully abort the LP loop.
                print("[LP] Solver failed – exiting LP refinement.")
                break

            b = lp_res.x
            if b.sum() == 0.0:
                break

            # Rescale b to have the same L1‑norm as a (target_sum)
            b = b * (target_sum / b.sum())

            # Mix the current vector with the new one.
            a_new = (1.0 - t_mix) * a + t_mix * b

            # Ensure we stay far from the 1000 clipping bound.
            max_val = a_new.max()
            if max_val > 900.0:                     # a safety margin
                a_new = a_new * (900.0 / max_val)

            # Evaluate the mixed vector (scale back to a realistic total).
            cand = (a_new * 100.0).tolist()
            score = evaluate_sequence(cand)
            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                best_seq = cand
                a = a_new
                print(f"[LP] iter={iter_cnt:3d}  score={best_score:.6f}")
                # success – optionally shrink the mixing factor for finer steps
                t_mix = max(0.05, t_mix * 0.9)
            else:
                # no improvement – reduce the step size, eventually stop
                t_mix *= 0.7
                if t_mix < 1e-4:
                    print("[LP] Mixing coefficient too small – stop LP.")
                    break

            # Guard against overrunning the time budget.
            if time.monotonic() + 0.1 >= deadline:
                break

    # ------------------------------------------------------------------
    # 4) Stochastic hill climbing (mass‑transfer moves).
    # ------------------------------------------------------------------
    if best_seq is not None:
        base = np.array(best_seq, dtype=np.float64)
        total_mass = base.sum()
        n = len(base)

        while time.monotonic() < deadline:
            # Randomly pick two distinct positions.
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue

            # Transfer a random fraction (up to 40 %) of the mass at i to j.
            frac = rng.uniform(0.0, 0.4)
            delta = base[i] * frac
            cand = base.copy()
            cand[i] -= delta
            cand[j] += delta

            # Enforce non‑negativity and renormalise to the original total mass.
            cand = np.clip(cand, 0.0, None)
            cand = cand / cand.sum() * total_mass

            score = evaluate_sequence(cand.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = float(score)
                base = cand
                best_seq = cand.tolist()
                print(f"[hill] score={best_score:.6f}")

            # Very occasional acceptance of a worse move (tiny annealing).
            elif np.isfinite(score) and rng.random() < 1e-4:
                base = cand
                # we do NOT update best_seq on a worse move.

    # ------------------------------------------------------------------
    # 5) Fallback – guarantee a valid return.
    # ------------------------------------------------------------------
    if best_seq is None:

        n = 256
        best_seq = np.ones(n, dtype=np.float64).tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"[final] best score={best_score:.6f}   length={len(best_seq)}")
    # Ensure ordinary Python floats for the output.
    return [float(x) for x in best_seq]
