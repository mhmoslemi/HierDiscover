import numpy as np, time, scipy.optimize

def lp_optimize(f, M):
    n = len(f)
    # Build matrix A such that A[k,j] = f[k-j] if 0 <= k-j < n else 0
    # Use convolution representation: for each j, the convolution kernel f placed at offset j.
    # But easiest: we can compute convolution f * b via Toeplitz (full).
    # Build A as shape (2*n-1, n)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # for each b_j coefficient, contributes f shifted by j
        # Convolution term at lag k = f_{k-j} * b_j
        # So fill A[k,j] = f_{k-j} if that index in range
        # Equivalent: A[:,j] = np.pad(f, (j, (2*n-1) - n - j), 'constant')
        # But simpler: shift f by j positions.
        start = j
        end = j + n
        A[start:end, j] = f
    # constraints: A @ b <= M
    # note: M must be >= 0
    c = -np.ones(n) # maximize sum(b) => minimize -sum(b)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        return None, None
    b_opt = res.x
    sum_b = b_opt.sum()
    return b_opt, sum_b
```

"""...
Softmax gradient descent with annealed temperature, periodic reordering
and spectral flattening, using a fixed mass normalization.
"""
import time
import numpy as np

def normalize_to_target(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip and scale seq to [0,1000] and enforce sum == target_sum."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s > 0.0:
        seq = seq * (target_sum / s)
    else:
        # If everything became zero, fall back to uniform
        seq = np.full_like(seq, target_sum / seq.size)
    seq = np.clip(seq, 0.0, 1000.0)
    return seq

def weighted_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the gradient of a smoothed ``max`` of a*a with respect
    to ``a``.
    ``alpha`` is the temperature parameter; larger ``alpha`` puts
    more weight on the true maximum.
    """
    # linear convolution a*a
    b = np.convolve(a, a)
    M = b.max()
    # weights that concentrate on the largest entries of b
    # diff <= 0, so we exponentiate a negative number
    diff = b - M               # non‑positive
    # avoid under‑flow for very negative values
    w = np.exp(alpha * diff)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum
    # gradient = 2 * (w * a_rev) sliding dot‑product
    conv_rev = np.convolve(w, a[::-1])
    grad = 2.0 * conv_rev[a.size - 1 : a.size - 1 + a.size]
    # Remove the component that would change the total mass
    grad -= grad.mean()
    return grad

def spectral_flatten(a: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project ``a`` onto the set of sequences whose DFT magnitude is constant.
    The projection is performed by replacing the magnitude of the FFT with a
    constant equal to sqrt(sum_i a_i^2) and leaving the phases unchanged.
    The result is clipped to the admissible box and finally renormalised.
    """
    n = a.size
    # Choose a length that accommodates a linear convolution without wrap‑around.
    L = int(2 ** np.ceil(np.log2(2 * n)))
    padded = np.zeros(L, dtype=np.float64)
    padded[:n] = a
    A = np.fft.fft(padded)
    # Constant magnitude that respects Parseval's identity.
    const_mag = np.sqrt(np.sum(a * a))
    A_new = const_mag * np.exp(1j * np.angle(A))
    a_new_full = np.fft.ifft(A_new).real
    a_new = a_new_full[:n]
    return normalize_to_target(a_new, target_sum)

def propose_candidate(seed: int = 123, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm combines:
      * Soft‑max based gradient descent with an annealing schedule for the
        temperature parameter.
      * Periodic random swaps and deterministic descending sort to explore
        different permutations.
      * A spectral‑flattening projection (constant‑magnitude FFT) applied
        every few hundred iterations.
    The total sum of the sequence is kept fixed to ``sqrt(2·n)`` which makes
    the evaluation score equal to the maximal autocorrelation value.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise a feasible sequence.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except NameError:
        # default: uniform vector of length 512
        base = np.ones(512, dtype=np.float64)

    # limit length to a manageable size (large lengths make the LP costly)
    max_len = 1024
    if base.size > max_len:
        # keep the first max_len entries – they already contain a lot of
        # information – and discard the rest.
        base = base[:max_len]

    n = base.size
    target_sum = np.sqrt(2.0 * n)

    base = normalize_to_target(base, target_sum)

"""
Softmax‑gradient descent with annealing, occasional re‑ordering and a
spectral‑flattening projection.

The search works on a fixed‑mass sequence (∑a = √(2·n)).  Under this
normalisation the evaluation score equals the maximal value of
the linear convolution a*a, so minimising the score is equivalent to
flattening the autocorrelation.

Algorithm outline
-----------------
1. Initialise a feasible sequence – either the provided ``height_sequence_1``
   (truncated to ≤1024 elements) or a uniform vector of length 512.
2. Perform a quick global scan: sample a few parametrised shapes
   (exponential, power‑law, linear) and keep the best one.
3. Main loop (time‑bounded):
   * Compute a soft‑max of the convolution values and obtain a
     gradient that points away from the large autocorrelation lags.
   * Take a step opposite to the gradient, re‑clip to the box
     [0,1000] and renormalise the sum.
   * Occasionally try a convex combination with the candidate
     (different mixing factors) and keep any improvement.
   * Every few hundred iterations apply cheap heuristic moves:
       – random swap of two entries,
       – deterministic descending sort (large values paired with
         small ones, which is optimal for the dominant lag by the
         rearrangement inequality),
       – a spectral‑flattening projection that forces the DFT
         magnitude to be constant.
4. Return the best sequence found.

All helper functions are defined at top level; the only external call is
``evaluate_sequence`` which is assumed to be available in the execution
environment.
"""

import time
import numpy as np

def _normalize(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and rescale the vector so that its sum equals *target_sum*."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s > 0.0:
        seq = seq * (target_sum / s)
    else:                     # extreme corner – fall back to a uniform vector
        seq = np.full_like(seq, target_sum / seq.size)
    seq = np.clip(seq, 0.0, 1000.0)
    return seq


def _weighted_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of ``max(a*a)``.
    ``alpha`` plays the role of an inverse temperature: larger values
    concentrate the weighting on the true maximum.
    """
    # convolution a*a
    conv = np.convolve(a, a)          # length 2n‑1
    m = conv.max()
    # weights – a smooth version of the arg‑max operator
    diff = conv - m                    # ≤ 0
    w = np.exp(alpha * diff)           # w[i]∈(0,1], w[max]=1
    w_sum = w.sum()
    w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

    # grad_i = 2 * Σ_k w_k * a_{k‑i}
    # This is a sliding dot‑product of w with a; the most efficient way
    # is to convolve w with the reversed a and extract the centre segment.
    conv_rev = np.convolve(w, a[::-1])
    n = a.size
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # Project onto the hyper‑plane of zero sum to keep the total mass unchanged.
    grad -= grad.mean()
    return grad


def _spectral_flatten(a: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Projection onto the set of sequences with a constant‑magnitude DFT.
    The constant magnitude that respects Parseval's identity is
    sqrt(∑a_i²).  After the inverse transform the result is clipped and
    renormalised to the required sum.
    """
    n = a.size
    L = int(2 ** np.ceil(np.log2(2 * n)))      # power‑of‑two length for a fast FFT
    padded = np.zeros(L, dtype=np.float64)
    padded[:n] = a

    A = np.fft.fft(padded)
    const_mag = np.sqrt(np.sum(a * a))
    A_new = const_mag * np.exp(1j * np.angle(A))

    a_new_full = np.fft.ifft(A_new).real
    a_new = a_new_full[:n]
    return _normalize(a_new, target_sum)

    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The method combines random shape sampling,
    soft‑max gradient descent, occasional permutations and a spectral
    flattening projection.  It respects the box constraints
    (0 ≤ a_i ≤ 1000) and keeps the total mass fixed to √(2·n).

    Returns
    -------
    list[float]
        The best sequence found within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise a feasible sequence
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except NameError:
        base = np.ones(512, dtype=np.float64)      # fallback length

    max_len = 1024
    if base.size > max_len:
        base = base[:max_len]                      # truncate overly long vectors

    n = base.size
    target_sum = np.sqrt(2.0 * n)

    a = _normalize(base, target_sum)

    # ------------------------------------------------------------------
    # 2. Quick global scan over a few parametric families
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)   # 1 s safety margin
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # allocate ~15 % of the budget for this coarse sampling stage
    sampling_deadline = start + 0.15 * (deadline - start)
    while time.monotonic() < sampling_deadline:
        if rng.random() < 0.5:
            # exponential decay a_i ∝ exp(-λ·i)
            lam = rng.uniform(0.0, 5.0 / n)
            raw = np.exp(-lam * np.arange(n, dtype=np.float64))
        else:
            # power‑law decay a_i ∝ (i+1)^{-p}
            p = rng.uniform(0.0, 3.0)
            raw = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)

        cand = _normalize(raw, target_sum)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            a = cand                      # hand the improved point to the next stage
            print(f"[sample] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop (gradient + occasional heuristics)
    # ------------------------------------------------------------------
    iteration = 0
    init_eta = 0.03 * target_sum                     # initial step magnitude
    alpha_start, alpha_end = 2.0, 12.0                # soft‑max temperature schedule
    mix_factors = (0.02, 0.05, 0.1, 0.2, 0.4)          # candidate convex‑mix weights

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        # anneal the soft‑max temperature
        alpha = alpha_start + (alpha_end - alpha_start) * (elapsed / (deadline - start))
        grad = _weighted_gradient(a, alpha)

        # linearly decreasing step size
        eta = init_eta * max(0.0, 1.0 - elapsed / (deadline - start))
        eta = max(eta, 1e-9)

        # candidate obtained by a gradient step
        cand = a - eta * grad
        cand = _normalize(cand, target_sum)

        # try a few convex mixtures between the current point and the candidate
        improved = False
        for t in mix_factors:
            mixed = (1.0 - t) * a + t * cand
            mixed = _normalize(mixed, target_sum)
            score = float(evaluate_sequence(mixed.tolist()))
            if np.isfinite(score) and score < best_score:
                best = mixed.copy()
                best_score = score
                a = mixed
                improved = True
                print(f"[{iteration:6d}] mix t={t:.2f} → new best {best_score:.6f}")
                break

        if not improved:
            a = cand   # continue walking even without improvement

        # ------------------------------------------------------------------
        # 3.1 heuristic moves (every few hundred iterations)
        # ------------------------------------------------------------------
        if iteration % 250 == 0 and iteration > 0:
            # a) random swap of two entries
            i, j = rng.integers(0, n, size=2)
            swapped = a.copy()
            swapped[i], swapped[j] = swapped[j], swapped[i]
            swapped = _normalize(swapped, target_sum)
            score_swapped = float(evaluate_sequence(swapped.tolist()))
            if np.isfinite(score_swapped) and score_swapped < best_score:
                best = swapped.copy()
                best_score = score_swapped
                a = swapped
                print(f"[{iteration:6d}] random swap → new best {best_score:.6f}")

            # b) deterministic descending sort (large values paired with small ones)
            sorted_desc = np.sort(a)[::-1]
            sorted_desc = _normalize(sorted_desc, target_sum)
            score_sorted = float(evaluate_sequence(sorted_desc.tolist()))
            if np.isfinite(score_sorted) and score_sorted < best_score:
                best = sorted_desc.copy()
                best_score = score_sorted
                a = sorted_desc
                print(f"[{iteration:6d}] descending sort → new best {best_score:.6f}")

        # ------------------------------------------------------------------
        # 3.2 spectral‑flattening projection (every 600 iterations)
        # ------------------------------------------------------------------
        if iteration % 600 == 0 and iteration > 0:
            proj = _spectral_flatten(a, target_sum)
            score_proj = float(evaluate_sequence(proj.tolist()))
            if np.isfinite(score_proj) and score_proj < best_score:
                best = proj.copy()
                best_score = score_proj
                a = proj
                print(f"[{iteration:6d}] spectral flatten → new best {best_score:.6f}")

        iteration += 1

    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    # Provide a plain Python list as the required output format.
    return [float(v) for v in best]
