def propose_candidate(seed=..., budget_s=..., **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    # determine length n
    try:
        n = len(height_sequence_1)
        n = max(256, min(n, 5000))
    except Exception:
        n = 2048
    # start from height_sequence_1 if exists else power-law
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
        if len(a) != n:
            # adjust size maybe cut or pad
            if len(a) > n: a = a[:n]
            else: a = np.pad(a, (0,n-len(a)), 'constant')
    else:
        a = np.array(power_law_sequence(n, 0.6), dtype=np.float64)
    # Normalize to sum 1
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    # start gradient descent
    step_frac = 0.05  # fraction of source mass to move each iteration
    min_step = 1e-6
    while time.monotonic() < deadline:
        # compute conv
        b = np.convolve(a, a)
        max_idx = int(b.argmax())
        max_val = b[max_idx]
        # compute gradient from max_idx
        g = np.zeros_like(a)
        # for i where j = max_idx - i in [0, n-1]:
        lo = max(0, max_idx - (n-1))
        hi = min(n-1, max_idx)
        # i range [lo, hi]
        i_range = np.arange(lo, hi+1)
        j = max_idx - i_range
        g[i_range] = 2.0 * a[j]
        # choose source idx (high gradient) weighted by a_i*g_i maybe
        # compute contribution = a_i * g_i
        contrib = a * g
        # if all contributions zero, break
        if contrib.sum() == 0:
            break
        src = int(np.argmax(contrib))  # most responsible
        # choose target idx with minimal gradient
        # but prefer location where a is small (to keep non-negative)
        # let's pick dest as argmin of g (or small a)
        dest_candidates = np.where(g == g.min())[0]
        # if multiple, select random among them
        dest = rng.choice(dest_candidates)
        if src == dest:
            # find alternative dest
            other = np.where(g < g.max())[0]
            if len(other)==0:
                break
            dest = rng.choice(other)
        # delta to move
        delta = min(a[src] * step_frac, a[src])
        if delta < min_step:
            # shrink step_frac
            step_frac *= 0.5
            if step_frac < 1e-8:
                break
            continue
        a[src] -= delta
        a[dest] += delta
        # ensure non-negative and sum remains 1
        # small numerical errors: renormalize
        a = np.clip(a, 0.0, None)
        a = a / a.sum()
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = float(score)
            best_a = a.copy()
            print(f"improved score={best_score:.6f} step_frac={step_frac:.4e}")
            # optionally increase step_frac if improvement
            step_frac = min(step_frac*1.2, 0.2)
        else:
            # occasionally accept? for now simple
            pass
        # occasional random perturbation
        if rng.random() < 0.001:
            i, j = rng.integers(0,n,2)
            if i != j:
                delta2 = a[i] * rng.uniform(0,0.1)
                a[i] -= delta2
                a[j] += delta2
                a = np.clip(a,0,None); a = a / a.sum()
    # finalize
    final_seq = best_a.tolist()
    print(f"final best score={best_score:.6f} length={len(final_seq)}")

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def power_law_sequence(length: int, exponent: float) -> np.ndarray:
    """
    Generate a non‑negative power‑law sequence a_i ∝ (i+1)^(-exponent)
    and normalise it to have unit sum.
    """
    idx = np.arange(length, dtype=np.float64) + 1.0
    w = np.power(idx, -exponent)
    w /= w.sum()
    return w


def conv_score(a: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(a*a) / (sum(a)^2) without calling
    ``evaluate_sequence`` (the evaluator uses the same formula after its
    clipping steps, which are irrelevant for the small numbers we work with).
    """
    b = np.convolve(a, a)
    max_b = b.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.size * max_b / (s * s)


def gradient_for_indices(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """
    Return the (sub‑)gradient of the maximum convolution entries whose
    indices are given in ``idxs``.  For each k∈idxs the contribution to the
    gradient at position i is 2·a_{k‑i} whenever the pair (i, k‑i) is inside
    the vector.
    The total gradient is the sum over all selected indices.
    """
    n = a.size
    g = np.zeros_like(a)
    for k in idxs:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        j = k - i
        g[i] += 2.0 * a[j]
    return g

        mass from indices that contribute most to the current maximal
        convolution entry to indices that contribute least.  The descent is

        is returned.

    (the objective is scale‑invariant).
    """
