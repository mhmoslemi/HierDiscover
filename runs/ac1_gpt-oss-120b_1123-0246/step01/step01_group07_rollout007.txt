def _project_simplex(v):
    # Projects v onto the probability simplex: nonnegative entries sum to 1.
    if np.all(v <= 0):

        return np.full_like(v, 1.0 / v.size)

"""
Hybrid projected‑gradient / simulated‑annealing search for the
self‑convolution L∞ minimisation problem.

The algorithm starts from the analytically motivated power‑law
a_i ∝ 1/√(i+1) (normalised to unit L1‑norm).  It then repeatedly
applies three cheap stochastic moves:

* a projected gradient step – reduces the currently maximal
  convolution entry,
* a pairwise “smoothing’’ move – redistributes the mass of two
  randomly chosen coordinates while keeping the total sum,
* a multiplicative block perturbation (the same idea used in the
  previous baseline).

After each move the candidate is projected onto the probability
simplex (non‑negative entries, sum = 1).  The best sequence found
so far is kept.  Simulated‑annealing style acceptance lets the
search escape shallow local minima.  The loop runs until the time
budget (≈ budget_s seconds) is exhausted and finally returns the
best list of heights.

All heavy operations (convolution and objective) use FFT‑based
O(n log n) kernels, allowing thousands of iterations on vectors of
a few thousand entries within the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – required by the task description)
# ----------------------------------------------------------------------


def _evaluate_fast(seq: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for non‑negative vectors.
    The function is homogeneous, so we keep the raw sum (no clipping).
    """
    n = seq.size
    # FFT based full convolution (real input → real output)
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))   # next power of two
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_b = conv.max()
    total = seq.sum()
    if total <= 0.0:
        return float("inf")
    return float(2.0 * n * max_b / (total * total))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, ∑x = 1}.  Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v <= 0):
        # Uniform distribution is the projection of a non‑positive vector
        return np.full_like(v, 1.0 / v.size)

    u = np.sort(v)[::-1]          # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    # Find the largest rho satisfying the KKT condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(seq: np.ndarray) -> np.ndarray:
    """Full convolution of ``seq`` with itself using a real FFT."""
    n = seq.size
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    return conv


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation (helps flatten the convolution)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # keep the vector on the simplex
    cand = _project_simplex(cand)
    return cand


def _pairwise_smoothing(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Redistribute the mass of two random coordinates (Beta‑biased)."""
    cand = base.copy()
    n = cand.size
    i, j = rng.choice(n, size=2, replace=False)
    total = cand[i] + cand[j]
    if total == 0.0:
        return cand
    # Beta(2,2) is peaked at 0.5 → a gentle smoothing operation
    r = rng.beta(2.0, 2.0)
    cand[i] = total * r
    cand[j] = total * (1.0 - r)
    cand = _project_simplex(cand)
    return cand


def _gradient_step(base: np.ndarray, eta: float) -> np.ndarray:
    """
    One step of projected gradient descent on the objective
    ``max(conv(base, base))``.  The sub‑gradient is taken from a
    currently maximal convolution index.
    """
    n = base.size
    conv = _conv_full_fft(base)
    k_max = int(np.argmax(conv))

    # compute sub‑gradient of the active convolution entry
    grad = np.zeros_like(base)
    i_start = max(0, k_max - (n - 1))
    i_end = min(k_max, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=np.int64)
    grad[idx] = 2.0 * base[k_max - idx]

    cand = base - eta * grad
    cand = _project_simplex(cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of non‑negative step heights that (hopefully) improves
    on the previous best bound (~1.5807).  The routine performs a
    hybrid stochastic/gradient search for about ``budget_s`` seconds
    and returns the best sequence encountered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a working length.  Prefer the length of a previously
    # discovered construction if it is available.
    # ------------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        if base_seq.ndim != 1 or base_seq.size == 0:
            raise ValueError
        n = int(base_seq.size)
    except Exception:
        n = 3000                     # a comfortable size for the time budget

    # ------------------------------------------------------------------
    # Analytic starting point a_i ∝ 1/√(i+1)  (normalised to L1–norm = 1)
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(idx)
    a = a / a.sum()                # project onto the simplex

    best = a.copy()
    best_score = _evaluate_fast(best)
    cur = a.copy()
    cur_score = best_score

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    eta = 0.5                      # initial step size for gradient moves
    it = 0
    # Simulated‑annealing temperature schedule (linear decay)
    T0 = 0.5

    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # Choose a move type
        # ------------------------------------------------------------------
        move = rng.choice(["gradient", "pair", "block"], p=[0.4, 0.3, 0.3])

        if move == "gradient":
            cand = _gradient_step(cur, eta)
        elif move == "pair":
            cand = _pairwise_smoothing(cur, rng)
        else:  # block perturbation
            cand = _make_block_perturbation(cur, rng)

        # ------------------------------------------------------------------
        # Evaluate candidate (fast version)
        # ------------------------------------------------------------------
        cand_score = _evaluate_fast(cand)

        # ------------------------------------------------------------------
        # Accept / reject logic (simple simulated‑annealing)
        # ------------------------------------------------------------------
        improve = cand_score < cur_score
        if improve:
            cur, cur_score = cand, cand_score
        else:
            # temperature decays linearly with the remaining budget
            remaining = deadline - time.monotonic()
            T = T0 * remaining / (budget_s - 2.0)
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-9))
            if rng.random() < prob:
                cur, cur_score = cand, cand_score

        # ------------------------------------------------------------------
        # Keep the overall best solution seen so far
        # ------------------------------------------------------------------
        if cand_score < best_score:
            best, best_score = cand, cand_score
            print(f"[{it:05d}] new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Simple schedules for step size and iteration counter
        # ------------------------------------------------------------------
        eta = max(eta * 0.9997, 1e-5)   # slowly shrink gradient step
        it += 1

    print(f"[final] best score = {best_score:.6f}")
    # The external evaluator will perform clipping; returning a simple list suffices.
    return best.tolist()
