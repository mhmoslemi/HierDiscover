"""
Propose a step‑function sequence.

The algorithm consists of two phases:

1. **Linear‑programming boost** – starting from the binomial
   (C(2i,i)/4^i) sequence we solve a small LP:
   maximize the total sum of a new non‑negative vector g
   under the linear constraints (f * g)[k] ≤ M,
   where M = max(f * f) and * denotes discrete convolution.
   The LP pushes mass to indices where f is tiny, increasing the
   overall sum while keeping the convolution peak bounded.
   We then move the current sequence towards the LP solution with a
   convex combination (parameter t) and repeat a few times.

2. **Stochastic hill‑climb** – after the LP stage we run a fast
   block‑perturbation / mirror‑swap local search (similar to the
   original baseline) to polish the result.

Both phases respect the wall‑clock budget (default 500 s; the whole
function will be terminated after 1000 s by the judge).  The best
sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _binomial_sequence(length: int) -> np.ndarray:
    """
    Truncated coefficients a_i = C(2i,i)/4^i.
    They satisfy (a*a)[k] ≈ 1 for interior k, giving a good initial score.
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(length - 1):
        a[i + 1] = a[i] * (2.0 * i + 1.0) / (2.0 * (i + 1))
    return np.clip(a, 0.0, 1000.0)


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and 8 % of n
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation – the optimum is often near‑symmetric
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _mirror_swap(candidate: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a small amount of mass between a position and its mirror."""
    n = candidate.size
    i = rng.integers(0, n // 2)
    j = n - 1 - i
    delta = min(candidate[i], candidate[j]) * rng.random() * 0.1
    if rng.random() < 0.5:
        candidate[i] += delta
        candidate[j] -= delta
    else:
        candidate[i] -= delta
        candidate[j] += delta
    return np.clip(candidate, 0.0, 1000.0)


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the (2n‑1) × n matrix A such that A @ g == np.convolve(f, g).
    Row k contains the coefficients f[i] for all i+j=k.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j contributes f[i] to row i+j
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max sum(g)   s.t.   A @ g <= M,   g >= 0,
    where A is the convolution matrix of f.
    Returns the optimal g (numpy array) or None if the solver fails.
    """
    try:
        import cvxpy as cp
    except Exception:      # pragma: no cover
        # cvxpy not available – skip the LP phase
        return None

    n = f.size
    A = _build_convolution_matrix(f)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # ECOS works well for dense LPs of this size
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    if g.value is None or prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    # Numerical noise may produce tiny negatives – clip them
    return np.clip(np.asarray(g.value, dtype=np.float64), 0.0, 1000.0)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s))

    # ------------------------------------------------------------------
    # Phase 1 – a few LP updates.
    # Allocate roughly 30 % of the budget (but never longer than 120 s)
    # ------------------------------------------------------------------
    lp_deadline = start_time + min(120.0, 0.3 * budget_s)
    lp_iter = 0
    while time.monotonic() < lp_deadline:
        # current convolution peak
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())
        g_opt = _solve_lp(f, M)
        if g_opt is None:
            break               # LP failed or cvxpy unavailable
        # convex update towards the LP solution
        t = 0.5                # step size; 0.5 works well in practice
        f = (1.0 - t) * f + t * g_opt
        f = np.clip(f, 0.0, 1000.0)

        # evaluate the new candidate
        score = float(evaluate_sequence(f.tolist()))
        lp_iter += 1
        if score < best_score:
            best = f.copy()
            best_score = score
            print(f"[LP {lp_iter:03d}] new best: {best_score:.6f}")

        # a tiny safeguard: if the LP does not improve after several steps,
        # we can stop early.
        if lp_iter >= 20 and best_score < 1.57:
            # we have already crossed the original binomial barrier – stop LP
            break

    # ------------------------------------------------------------------
    # Phase 2 – stochastic hill‑climbing (block perturbations + swaps).
    # Use the remaining time budget.
    # ------------------------------------------------------------------
    iter_cnt = 0
    while time.monotonic() < deadline:
        iter_cnt += 1
        # choose move type
        if rng.random() < 0.7:
            candidate = _make_block_perturbation(best, rng)
        else:
            candidate = _mirror_swap(best.copy(), rng)

        # occasional small global scaling of the first element (helps reduce a0)
        if rng.random() < 0.05:
            factor = float(np.exp(rng.normal(0.0, 0.04)))
            candidate[0] = np.clip(candidate[0] * factor, 0.0, 1000.0)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"[{iter_cnt:06d}] new best: {best_score:.6f}")

        # occasional re‑centering to avoid drift to the boundary
        if iter_cnt % 5000 == 0:
            # blend a tiny amount of the current best with a fresh binomial
            fresh = _binomial_sequence(n)
            best = 0.99 * best + 0.01 * fresh
            best = np.clip(best, 0.0, 1000.0)

    print(f"[done] iterations={iter_cnt}, LP steps={lp_iter}, final best={best_score:.6f}")
    return [float(x) for x in best]
