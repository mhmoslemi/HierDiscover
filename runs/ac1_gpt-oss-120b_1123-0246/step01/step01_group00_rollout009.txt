"""
Soft‑max gradient descent for the step‑function minimisation problem.

The algorithm builds on the previous gradient‑descent approach but replaces the
hard “max” with a smooth log‑sum‑exp approximation.  The gradient of the
soft‑max (∇ log ∑ exp(β·bₖ)) is a weighted correlation between the weight
vector and the current sequence, which we compute efficiently with FFTs.
β is increased during the run so that the optimisation first moves the mass
smoothly and then focuses on the true maximum convolution value.
A few different starting points (uniform, random, U‑shaped) are tried;
the best sequence found (according to the exact evaluation function) is
returned.  The total running time respects the supplied budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the full linear convolution a * a using real FFTs.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)
    return conv_full[: 2 * a.size - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the full correlation (w ⋆ a) = convolution(w[::-1], a)
    using FFTs.  Returns an array of length (len(w) + len(a) - 1).
    """
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, fft_len)
    fa = np.fft.rfft(a, fft_len)
    corr_full = np.fft.irfft(fw * fa, fft_len)
    # length = len(w_rev) + len(a) - 1 = len(w) + len(a) - 1
    return corr_full[: len(w) + len(a) - 1]


def _objective_from_selfconv(b: np.ndarray, target_sum: float) -> float:
    """
    Given the full self‑convolution b = a*a and the normalisation
    target_sum = √(2 n), return the evaluation score.
    For this normalisation the score equals max(b).
    """
    max_b = b.max()
    # The exact evaluate_sequence would compute 2*n*max_b/target_sum**2,
    # but target_sum**2 = 2*n, therefore the score is simply max_b.
    return float(max_b)

                cand3 = a * pert
                cand3 = np.clip(cand3, 0.0, 1000.0)
                s3 = cand3.sum()
                if s3 > 0.0:
                    cand3 *= (target_sum / s3)
                b3 = _self_convolution_fft(cand3, fft_len)
                max3 = b3.max()
                if max3 < cur_max:
                    a = cand3
                    cur_max = max3
                    if max3 < best_score:
                        best_score = max3
                        best_seq = cand3.copy()
                        print(f"[{total_iters:6d}] perturbed improvement = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 4. Final exact evaluation of the best candidate before returning
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector
        best_seq = np.full(n, target_sum / n)

    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] best exact score = {final_score:.6f}")
    return [float(v) for v in best_seq]
