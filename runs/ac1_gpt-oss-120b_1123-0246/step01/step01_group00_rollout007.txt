def adam_optimize(a, steps, lr, beta1=0.9, beta2=0.999, eps=1e-8, target_sum=..., deadline):
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    for t in range(1, steps+1):
        if time.monotonic() >= deadline: break
        b = np.convolve(a, a)
        k_max = np.argmax(b)
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n-1))
        hi = min(k_max, n-1)
        i_idx = np.arange(lo, hi+1)
        j_idx = k_max - i_idx
        grad[i_idx] = a[j_idx]   # derivative proportional to a[j]
        # project sum zero
        grad -= grad.mean()
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = a + step
        # clipping:
        cand = np.clip(cand, 0, 1000)
        # renorm to target_sum
        s = cand.sum()
        if s > 0:
            cand = cand * (target_sum / s)
        cand = np.clip(cand, 0, 1000)
        # evaluate:
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            a = cand
        else:
            a = cand # continue moving
        # optionally random perturbation every some steps
```

Define w_extended = w (length L). Extend a to length L by zero padding after n elements? For convolution we want (w * a)[i] where (w * a)[i] = ∑_j w_i_minus_j a_j (i-j). That's not our case.

But we can compute g as cross-correlation of w with a: cross-correlation xcorr_{i} = ∑_j w_{i+j} a_j = (w ⋆ a)[i] (where ⋆ denotes cross-correlation). Indeed, cross-correlation at lag i is sum_j w_{i+j} a_j. So we need cross-correlation of w and a for i=0..n-1.

Cross-correlation can be computed via convolution with reversed version: (w ⋆ a)[i] = (w_rev * a)[i + (L-1)], where w_rev is reversed w. Because (w_rev * a)[k] = ∑_j w_rev_j a_{k-j}, and if we set k = i + (L-1), then term is ∑_j w_rev_j a_{i+L-1 - j}. Since w_rev_j = w_{L-1-j}, thus above becomes ∑_j w_{L-1-j} a_{i+L-1 - j} = ∑_{j'} w_{j'} a_{i + j'} where j' = L-1-j, sum over j' so indeed cross-correlation.

Thus to compute g_i: compute conv = np.convolve(w[::-1], a, mode='full'); then g = 2 * conv[(L-1) : (L-1 + n)]. Since conv length = L + n - 1 = (2n-1) + n - 1 = 3n-2. So conv indices L-1 to L-1+n-1 = L-1 to L+n-2 correspond to i = 0..n-1.

```
def gradient_softmax(a, beta):
    b = np.convolve(a, a)
    # compute softmax weights
    max_b = np.max(b)
    # compute weights stable: w = exp(beta*(b - max_b))   # avoid overflow
    w = np.exp(beta * (b - max_b))
    w_sum = np.sum(w)
    w = w / w_sum
    # compute gradient: g = 2 * cross-correlation(w, a)
    w_rev = w[::-1]
    conv = np.convolve(w_rev, a, mode='full')
    g = 2 * conv[(len(w)-1) : (len(w)-1 + len(a))]
    # project to sum-zero: g = g - mean(g)
    g -= g.mean()
    return g
```

import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    # convolution f*g <= M for each lag
    for k in range(2*n-1):
        # compute sum_{i} f_i * g_{k-i}
        # include indices where 0 <= i < n and 0 <= k-i < n
        # g_{k-i} is variable g_{j} where j = k-i
        # we can write linear expression using cp.sum(cp.multiply(f[i], g[j]))
        # Use loops to build.
        expr = 0
        lo_i = max(0, k - (n-1))
        hi_i = min(k, n-1)
        for i in range(lo_i, hi_i+1):
            j = k - i
            expr += f[i] * g[j]
        constraints.append(expr <= M)
    # objective maximize sum(g)
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        g_val = g.value
        return np.maximum(g_val, 0.0)
    else:
        return None
```

Hybrid simulated‑annealing + Adam optimizer for the step‑function
minimisation problem.

The algorithm starts from the best known global sequence (``height_sequence_1``)
or, if unavailable, a simple U‑shaped initialization.
During the main phase a *soft‑max* of the autocorrelation is used as a smooth
approximation of the “max’’ in the objective.  The resulting weighted gradient
is obtained via a cross‑correlation (implemented with a single convolution)
and is fed to an Adam optimiser with a linearly decaying learning rate.
The soft‑max temperature (``beta``) is annealed from a very small value up to
a larger one, allowing the optimiser to first move in a smooth landscape and
later focus on the true peak.

Every few hundred iterations a random multiplicative perturbation is applied
to escape plateaus, and every few thousand iterations a fresh U‑shaped
candidate is tried (restart).  After the time budget expires a short
max‑lag gradient descent (the classic “max’’ gradient) is performed as a
final refinement.

All values are kept in the feasible box ``[0,1000]`` and the total mass is
renormalised to ``sqrt(2·n)`` (scale‑invariant normalisation used by the
reference solution).  The best sequence found is returned as a list of
floats.
"""

import time

# ----------------------------------------------------------------------


def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """Return soft‑max weights w_k ∝ exp(beta·b_k) stabilised by max(b)."""
    max_b = np.max(b)
    # subtraction of max avoids overflow, division normalises
    exp_term = np.exp(beta * (b - max_b))
    w = exp_term / np.sum(exp_term)
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)_k  w.r.t. a.

    Using the identity  ∂/∂a_i Σ_k w_k b_k = 2·Σ_k w_k a_{k‑i},
    which is a cross‑correlation of w and a.  Implemented via one convolution.
    """
    # cross‑correlation = convolution of reversed w with a
    w_rev = w[::-1]
    conv = np.convolve(w_rev, a, mode="full")
    n = a.size
    # slice the n entries that correspond to i = 0 … n‑1
    grad = 2.0 * conv[(len(w) - 1) : (len(w) - 1 + n)]
    # projection onto the hyper‑plane Σ a_i = const  (zero‑sum gradient)
    grad -= grad.mean()
    return grad


def _max_lag_gradient(a: np.ndarray) -> (np.ndarray, np.ndarray):
    """
    Gradient of the *max* autocorrelation term.
    Returns (grad, b) where b = a*a.
    """
    n = a.size
    b = np.convolve(a, a)                # full autocorrelation, length 2n‑1
    k_max = int(np.argmax(b))            # lag with the current maximal value

        grad[i_idx] = a[j_idx]           # ∂max/∂a_i = a_{k_max‑i}
    grad -= grad.mean()                  # keep total mass unchanged
    return grad, b

    best = best_seq.copy()
