"""
Propose a step‑function minimizing 2·n·max(a*a)/(sum a)^2.

Algorithm
---------
* Choose a few candidate lengths (the known length, and a few powers of two).
* For each length run a cheap projected sub‑gradient descent on the
  non‑negative vector, keeping the sum normalised to one.
  – The sub‑gradient of the maximal convolution entry is computed
    analytically (two shifted copies of the current vector).
  – A simple back‑tracking line‑search adapts the step‑size.
* Occasionally apply a random block‑wise multiplicative perturbation
  (the “hill‑climber” from the previous solution) to escape shallow
  minima.
* Keep the best vector seen globally and return it as a plain Python list.

The routine respects the security checks of ``evaluate_sequence``:
all entries are non‑negative, finite and well below the 1000‑limit.
The search runs until the supplied time budget (default 1000 s) is
exhausted.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Fast evaluation – identical to evaluate_sequence but without
# the security wrappers.  It assumes a non‑negative vector.
def _eval_fast(a: np.ndarray) -> float:
    """Return the objective value for a non‑negative vector a."""
    n = a.size
    if n == 0:
        return float("inf")
    # FFT based full convolution (real‑valued)
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ------------------------------------------------------------
# Gradient of max‑convolution entry.
def _grad_max_conv(a: np.ndarray, act_idx: np.ndarray) -> np.ndarray:
    """Projected sub‑gradient of max_k (a*a)[k] w.r.t. a."""
    n = a.size
    grad = np.zeros_like(a)
    for k in act_idx:
        # contribution from a_{k-j}
        if k < n:
            # grad[0:k+1] += a[k::-1]
            grad[: k + 1] += a[k::-1]
        # contribution from a_{j-k}
        if k > 0:
            # grad[k:] += a[:n-k]
            grad[k:] += a[: n - k]
    return grad


# ------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """Cheap multiplicative log‑normal perturbation of a random block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # at least 1 element, up to 0.5 % of n
    hi = max(2, n // 12)           # up to about 8 % of n
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation (helps flatten the autocorrelation)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ------------------------------------------------------------
def _initial_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised power‑law vector  a_i ∝ (i+1)^{-p} .
    The exponent p is chosen randomly in a sensible range."""
    p = rng.uniform(0.35, 0.85)          # random exponent for diversity
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a = np.clip(a, 0.0, 1000.0)
    a /= a.sum()
    return a


# ------------------------------------------------------------
def _optimize_length(n: int,
                     time_budget: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run the projected sub‑gradient / random‑block search for a fixed length."""
    deadline = time.monotonic() + time_budget
    # start from a random power‑law; keep the best exponent seen
    a = _initial_powerlaw(n, rng)

    # a quick scan of a few exponents to get a better start point
    for p in np.linspace(0.30, 0.90, 13):
        cand = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        cand = np.clip(cand, 0.0, 1000.0)
        cand /= cand.sum()
        if _eval_fast(cand) < _eval_fast(a):
            a = cand

    best = a.copy()
    best_score = _eval_fast(best)

    # step‑size handling
    step = 0.2
    no_improve = 0

    while time.monotonic() < deadline:
        # 20 % of the time we do a random block perturbation,
        # the rest we perform a sub‑gradient step.
        if rng.random() < 0.20:
            cand = _make_block_perturbation(a, rng)
        else:

            nfft = 1 << ((2 * n - 1).bit_length())
            A = np.fft.rfft(a, n=nfft)
            conv = np.fft.irfft(A * A, n=nfft)[: 2 * n - 1]
            max_b = conv.max()

            grad = _grad_max_conv(a, active)

                s = cand.sum()
            cand /= s

            best_score = cand_score
            a = cand                         # continue from the improvement
            step = min(step * 1.3, 1.0)      # modestly enlarge step after success

    return best, best_score
