def project_onto_simplex(v):

    # returns w: non-negative, sum=1.

    # if sum(v) == 1 and v>=0, return v.

    # else:

    u = np.sort(v)[::-1]   # descending

    cssv = np.cumsum(u) - 1

    ind = np.arange(len(v)) + 1

    # find rho = max i s.t. u_i - cssv_i / i > 0

    cond = u - cssv / ind > 0

    rho = np.where(cond)[0][-1]

    theta = cssv[rho] / (rho + 1)

    w = np.maximum(v - theta, 0)

    return w

def project_simplex(v):

   v = np.asarray(v, dtype=np.float64)

   n = v.shape[0]

   u = np.sort(v)[::-1]

   cssv = np.cumsum(u) - 1

   ind = np.arange(1, n+1)

   cond = u - cssv / ind > 0

   if not np.any(cond):

       # all entries zero? return uniform? default

       return np.full(n, 1.0/n)

   rho = np.where(cond)[0][-1]

   theta = cssv[rho] / (rho + 1)

   w = np.maximum(v - theta, 0)

   return w

def gradient_step(seq, beta, step, m=None, v=None, t=1, beta1=0.9, beta2=0.999, eps=1e-8):

   # compute conv
   conv = full_convolution(seq, seq)
   max_conv = conv.max()
   # compute weights w as softmax
   # shift to improve stability:
   shifted = conv - max_conv
   expb = np.exp(beta * shifted)
   w = expb / expb.sum()

   # compute gradient via convolution w * seq_rev
   seq_rev = seq[::-1]
   g_full = full_convolution(w, seq_rev)
   # slice indices n-1 .. n-1+n-1
   g = 2.0 * g_full[len(seq)-1 : len(seq)-1 + len(seq)]   # length n

   # Adam update
   if m is None:
       m = np.zeros_like(seq)
       v = np.zeros_like(seq)
   m = beta1 * m + (1 - beta1) * g
   v = beta2 * v + (1 - beta2) * (g * g)
   m_hat = m / (1 - beta1**t)
   v_hat = v / (1 - beta2**t)
   seq_new = seq - step * m_hat / (np.sqrt(v_hat) + eps)

   # project onto simplex
   seq_proj = project_simplex(seq_new)

   return seq_proj, conv, m, v

"""
Propose a step‑function with a low self‑convolution L∞‑norm.

The evaluation is   2·n·max(conv(f,f)) / (sum(f)^2) .
Because the objective is homogeneous, we keep the vector on the
probability simplex (non‑negative, sum = 1).  Starting from the
analytic a_i ∝ 1/√(i+1) we run a projected Adam optimiser on a smooth
approximation of the max‑convolution (log‑sum‑exp).  The smoothness
parameter β is increased during the run so that the optimiser first
sees a well‑behaved landscape and finally concentrates on the true
maximum.  Periodically a cheap random block perturbation (the same
mechanism used in the previous baseline) is tried to escape possible
local minima.  The best vector seen within the time budget is returned.

All helper functions are defined at top level; no closures or file I/O
are used.
"""

import time
import numpy as np


def _full_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full convolution a*b using a single FFT (real input)."""
    n = a.size + b.size - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)[:n]
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]                      # descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fast_score(seq: np.ndarray) -> float:
    """Fast replica of evaluate_sequence for a non‑negative vector."""
    n = seq.size
    conv = _full_convolution(seq, seq)
    max_conv = conv.max()
    # sum(seq) is 1 after we keep the vector on the simplex
    return float(2.0 * n * max_conv)


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

                     budget_s: float = 900.0,

    """
    Search for a low‑L∞ self‑convolution step function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget (seconds).  The function stops slightly
        before this limit to give the caller time to finish.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a length.  Prefer the length of the previous best sequence
    # if it exists, otherwise fall back to a moderate size.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
        seq = np.maximum(base, 0.0)
        seq = _project_onto_simplex(seq)         # normalise to sum = 1
    except Exception:
        n = 2000                                 # a safe default
        idx = np.arange(1, n + 1, dtype=np.float64)
        seq = 1.0 / np.sqrt(idx)                # analytic power‑law start
        seq = _project_onto_simplex(seq)

    best_seq = seq.copy()
    best_score = _fast_score(best_seq)
    print(f"initial length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(seq)          # first moment
    v = np.zeros_like(seq)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- compute the full self‑convolution -----------------------
        conv = _full_convolution(seq, seq)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv          # sum(seq)=1

        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"iter {it:5d} – new best: {best_score:.6f}")

        # ----- decide whether to take a gradient step or a random jump -
        if rng.random() < 0.70:                 # gradient‑based move
            # smooth‑max temperature (increases with time)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** it)

            seq = seq - step                     # gradient descent
        else:                                    # random block perturbation
            cand = _make_block_perturbation(seq, rng)
            cand = _project_onto_simplex(cand)
            cand_score = _fast_score(cand)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"iter {it:5d} – random improve: {best_score:.6f}")
                seq = cand.copy()
                m.fill(0.0)
                v.fill(0.0)
                continue
            # occasionally accept a worse random move (simulated‑annealing flavour)
            if rng.random() < 0.03:
                seq = cand.copy()
                m.fill(0.0)
                v.fill(0.0)

        # ----- always enforce feasibility (simplex) --------------------
        seq = _project_onto_simplex(seq)

    # final sanity check: make sure the returned list obeys the limits
    final = np.clip(best_seq, 0.0, 1000.0).tolist()
    return final
