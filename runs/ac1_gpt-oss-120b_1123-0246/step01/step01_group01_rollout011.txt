def propose_candidate(seed=0, budget_s=800, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.1
    best_seq = None
    best_score = float('inf')
    # Parameter search
    while time.time() < deadline:
        # choose length randomly from a set
        n = int(rng.integers(100, 2000))
        # choose exponent p:
        p = rng.uniform(0.45, 0.80)
        # offset shift >=0:
        offset = rng.uniform(0.0, 10.0)
        idx = np.arange(1, n+1, dtype=np.float64) + offset
        seq = np.power(idx, -p)
        # optionally random perturbation:
        if rng.random() < 0.2:
            # apply scaling to small random block
            seq = make_block_perturbation(seq.tolist(), rng)
        # evaluate
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[search] n={n} p={p:.4f} off={offset:.2f} score={score:.6f}")
        # occasional refinement of best: small random tweaks
        if best_seq is not None and rng.random() < 0.05:
            cand = make_block_perturbation(best_seq.tolist(), rng)
            sc = float(evaluate_sequence(cand))
            if sc < best_score:
                best_score = sc
                best_seq = np.array(cand)
    # final refinement: hill climb using block and mirror moves similar to earlier program
    while time.time() < deadline:
        # Mirror swap move on best_seq
        n = best_seq.size
        if rng.random() < 0.5:
            candidate = best_seq.copy()
            i = rng.integers(0, n // 2)
            j = n - 1 - i
            delta = min(candidate[i], candidate[j]) * rng.random() * 0.2
            if rng.random() < 0.5:
                candidate[i] += delta
                candidate[j] -= delta
            else:
                candidate[i] -= delta
                candidate[j] += delta
            candidate = np.clip(candidate, 0.0, 1000.0)
        else:
            candidate = make_block_perturbation(best_seq.tolist(), rng)
        sc = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"[refine] new best score={best_score:.6f}")
    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()

Propose a step‑function sequence that improves on the binomial (π/2) baseline.

Strategy
--------
*  The evaluation metric is   score = 2·n·max(a*a) / (sum(a))² .
*  For a decreasing power‑law a_i ≈ (i+shift)^(–p) the maximum of the
   self‑convolution is usually b₀ = a₀² = (1+shift)^(–2p) while the sum
   grows like n^{1‑p}.  For moderate lengths (n≈50…300) this yields scores
   well below 1.5.
*  We therefore explore three parameters:
      - n      : length of the sequence (integer)
      - p      : exponent of the power‑law (float, 0.45 ≤ p ≤ 0.75)
      - shift  : additive shift that reduces the first element (float, 0…10)
*  A cheap grid‑search (deterministic + random sampling) is performed
   inside the time budget.  Every candidate is evaluated with the provided
   ``evaluate_sequence`` function, the best is kept, and a short stochastic
   hill‑climb (block perturbations and mirror swaps) is run on the best
   candidate until the time limit expires.
*  The algorithm never uses closures or lambdas and only depends on the
   standard library and NumPy.
"""

import time
import numpy as np

def make_block_perturbation(base, rng):
    """Apply a multiplicative log‑normal perturbation to a random block."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width: between 0.5 % and 10 % of the length
    w_min = max(1, n // 200)
    w_max = max(2, n // 10)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # log‑normal multiplicative noise (σ≈0.1)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.10)))
    # occasional symmetrisation – helps when the optimum is roughly symmetric
    if rng.random() < 0.3:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def _powerlaw_sequence(length, exponent, shift):
    """
    Generate a decreasing power‑law sequence:
        a_i = (i + 1 + shift)⁻ᵉˣᵖ
    ``exponent`` > 0, ``shift`` ≥ 0.
    The sequence is returned as a NumPy float64 array.
    """
    idx = np.arange(1, length + 1, dtype=np.float64) + shift
    seq = np.power(idx, -exponent)
    return np.clip(seq, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=900, **kwargs):
    """
    Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.

    The search consists of:
      1. a deterministic grid over length / exponent / shift,
      2. a random sampling of the same parameter space,
      3. a short stochastic hill‑climb (block perturbations + mirror swaps)
    limited by the supplied wall‑clock budget (seconds).

    Returns
    -------
    list[float]
        The best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Deterministic grid – quick coarse scan
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    start = time.time()
    deadline = start + max(0.0, budget_s - 2.0)   # keep a little slack for final hill‑climb

    # Lengths that are cheap to evaluate yet large enough to exhibit the effect
    length_grid = [64, 96, 128, 160, 192, 224, 256, 320, 384, 448, 512]
    # Exponents around the theoretical optimum (0.5) – a bit larger helps for short n
    exponent_grid = np.linspace(0.45, 0.75, 31)   # step ≈0.01
    shift_grid = [0.0, 0.5, 1.0, 2.0, 5.0]

    for n in length_grid:
        for p in exponent_grid:
            for sh in shift_grid:
                seq = _powerlaw_sequence(n, p, sh)
                score = float(evaluate_sequence(seq.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"[grid] n={n:4d} p={p:.3f} shift={sh:4.1f} score={score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Randomised sampling – broaden the search
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # random length (50 … 600)
        n = int(rng.integers(50, 601))
        # exponent drawn uniformly in a useful interval
        p = rng.uniform(0.45, 0.75)
        # small shift to soften the first element
        sh = rng.uniform(0.0, 6.0)

        seq = _powerlaw_sequence(n, p, sh)

        # occasional extra block perturbation to escape the pure power‑law family
        if rng.random() < 0.15:
            seq = make_block_perturbation(seq.tolist(), rng)

        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[rand] n={n:4d} p={p:.3f} shift={sh:4.2f} score={score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Local stochastic hill‑climb starting from the current best
    # ------------------------------------------------------------------
    # If for some reason no candidate was generated we fall back to a tiny constant.
    if best_seq is None:
        best_seq = np.ones(64, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Hill‑climb loop (uses the same time deadline as above)
    while time.time() < deadline:
        n = best_seq.size
        # choose a move type
        if rng.random() < 0.65:       # block perturbation (most frequent)
            cand = make_block_perturbation(best_seq.tolist(), rng)
        else:                         # mirror‑swap – moves mass between symmetric indices
            cand = best_seq.copy()
            i = rng.integers(0, n // 2)
            j = n - 1 - i
            # move up to 15 % of the smaller side
            delta = min(cand[i], cand[j]) * rng.random() * 0.15
            if rng.random() < 0.5:
                cand[i] += delta
                cand[j] -= delta
            else:
                cand[i] -= delta
                cand[j] += delta
            cand = np.clip(cand, 0.0, 1000.0)

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[hill] new best = {best_score:.6f} (len={n})")

    print(f"[done] final best score = {best_score:.6f} (length={best_seq.size})")
    # Return plain Python list, as required by the evaluation harness
    return [float(x) for x in best_seq]
