from scipy.optimize import linprog
from scipy.sparse import diags

def propose_candidate(seed=42, budget_s=950):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)
    
    # load global start sequence if exists
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.shape[0]
    except NameError:
        n = 2000
        base = None

    # create Catalan start if not present
    def catalan(n):
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1,n):
            a[i] = a[i-1] * (2*i-1) / (2*i)
        return np.clip(a, 0.0, 1000.0)

    if base is None or base.size != n:
        f = catalan(n)
    else:
        # evaluate which is better
        score_base = float(evaluate_sequence(base.tolist()))
        score_catalan = float(evaluate_sequence(catalan(n).tolist()))
        if score_base < score_catalan:
            f = base.copy()
        else:
            f = catalan(n)
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Start: n={n}, score={best_score:.6f}")

    iteration = 0
    # parameters
    eps = 1e-12
    max_iter = 30
    # step sizes decreasing each iteration
    step_sizes = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]

    while time.monotonic() < deadline and iteration < max_iter:
        f = best  # current sequence
        # compute convolution and max
        conv_f_f = np.convolve(f, f)
        M = float(conv_f_f.max())
        # Build LP matrix
        # Using sparse diags
        # each diagonal i has constant f[i] values.
        # Build list of data arrays
        data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
        A_ub = diags(data, offsets=range(n), shape=(2*n-1, n), format='csr')
        b_ub = np.full(2*n-1, M + eps, dtype=np.float64)

        # Solve LP
        c = -np.ones(n, dtype=np.float64)  # maximize sum(g) -> minimize -sum
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n,
                          method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.monotonic())})
        except Exception as exc:
            print(f"Linprog exception: {exc}")
            break

        if not res.success:
            print(f"LP failed at iteration {iteration}: status {res.message}")
            break

        g = np.maximum(0.0, res.x)  # ensure nonneg
        # mixing
        improved = False
        for t in step_sizes:
            candidate = (1-t) * best + t * g
            candidate = np.clip(candidate, 0.0, 1000.0)
            try:
                cand_score = float(evaluate_sequence(candidate.tolist()))
            except Exception:
                continue
            if np.isfinite(cand_score) and cand_score < best_score - 1e-8:
                best = candidate
                best_score = cand_score
                improved = True
                print(f"Iter {iteration}, t={t:.3f} -> score {best_score:.6f}")
                break
        if not improved:
            # random perturb for exploration
            perturb = np.exp(rng.normal(0.0, 0.03, size=n))
            candidate = best * perturb
            candidate = np.clip(candidate, 0.0, 1000.0)
            try:
                cand_score = float(evaluate_sequence(candidate.tolist()))
            except Exception:
                cand_score = np.inf
            if np.isfinite(cand_score) and cand_score < best_score:
                best = candidate
                best_score = cand_score
                print(f"Iter {iteration}, random perturb -> score {best_score:.6f}")
        iteration += 1

    print(f"Done after {iteration} iter, best score {best_score:.6f}")
    return [float(v) for v in best]

import time, numpy as np
from math import sqrt
try:
    from scipy.optimize import linprog
    from scipy.sparse import diags
    SCIPY_AVAILABLE = True
except Exception:
    SCIPY_AVAILABLE = False

def _catalan_sequence(n):
    """Return Catalan-type coefficients C_i = binom(2i,i)/4**i for i=0..n-1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i-1) / (2*i)
    # enforce bounds that evaluate_sequence will clamp anyway
    np.clip(a, 0.0, 1000.0, out=a)
    return a

def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """
    Search for a non‑negative step sequence with a low score.
    The routine starts from the Catalan distribution (or a provided global)
    and repeatedly solves a linear program that adds mass without raising the
    current worst convolution.  Mixing the current sequence with the LP
    solution yields a strictly larger total sum and typically a lower score.
    The loop stops after the time budget or after a modest number of iterations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # ----------------------------------------------------------------------
    # 1. Initialise the base sequence
    # ----------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.shape[0]
    except Exception:
        # fallback size – 2 000 gives a reasonable trade‑off between speed
        # and quality for the optimisation loop.
        n = 2000
        base = None

    # Catalan start (always available)
    catalan = _catalan_sequence(n)

    if base is not None and base.size == n:
        try:
            score_base = float(evaluate_sequence(base.tolist()))
            score_catalan = float(evaluate_sequence(catalan.tolist()))
            if score_base < score_catalan:
                a = base.copy()
                best_score = score_base
                print(f"Using provided global sequence as start, score={best_score:.6f}")
            else:
                a = catalan.copy()
                best_score = score_catalan
                print(f"Using Catalan start, score={best_score:.6f}")
        except Exception:
            a = catalan.copy()
            best_score = float(evaluate_sequence(a.tolist()))
            print(f"Start with Catalan, score={best_score:.6f}")
    else:
        a = catalan.copy()
        best_score = float(evaluate_sequence(a.tolist()))
        print(f"Start with Catalan, score={best_score:.6f}")

    best = a.copy()
    n = best.size

    # ----------------------------------------------------------------------
    # 2. Iterative refinement
    # ----------------------------------------------------------------------
    iteration = 0
    eps = 1e-12          # tiny slack for numerical safety
    max_iters = 30       # a safe bound that fits within the time budget
    # step sizes tried in each iteration – larger steps first.
    step_sizes = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]

    while time.monotonic() < deadline and iteration < max_iters:
        f = best
        # convolution and current worst value
        conv_self = np.convolve(f, f)
        M = float(conv_self.max())

        # ------------------------------------------------------------------
        #   2a – build the linear program  max Σ g_i  s.t. (f * g) ≤ M
        # ------------------------------------------------------------------
        if not SCIPY_AVAILABLE:
            # No LP solver – fall back to a cheap random tweak
            print("SciPy not available – using random perturbation only.")
            break

        # Build sparse constraint matrix: each diagonal i contains f[i].
        # This creates a (2n‑1) × n matrix with the exact structure of the
        # linear convolution operator.
        # The construction is O(n²) but for n≈2000–5000 it still fits in memory.
        data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
        A_ub = diags(data, offsets=range(n), shape=(2 * n - 1, n), format='csr')
        b_ub = np.full(2 * n - 1, M + eps, dtype=np.float64)

        # Solve the LP (linprog minimises, so we maximise –Σ g_i)
        c = -np.ones(n, dtype=np.float64)
        try:
            res = linprog(c,
                          A_ub=A_ub,
                          b_ub=b_ub,
                          bounds=[(0.0, None)] * n,
                          method='highs',
                          options={'presolve': True,
                                   'time_limit': max(0.0, deadline - time.monotonic())})
        except Exception as exc:
            print(f"linprog raised an exception: {exc}")
            break

        if not res.success:
            print(f"LP not solved (iteration {iteration}): {res.message}")
            break

        g = np.maximum(0.0, np.asarray(res.x, dtype=np.float64))

        # ------------------------------------------------------------------
        #   2b – mix the current sequence with the LP solution
        # ------------------------------------------------------------------
        improved = False
        for t in step_sizes:
            candidate = (1.0 - t) * f + t * g
            candidate = np.clip(candidate, 0.0, 1000.0)
            try:
                cand_score = float(evaluate_sequence(candidate.tolist()))
            except Exception:
                continue
            if np.isfinite(cand_score) and cand_score < best_score - 1e-9:
                best = candidate
                best_score = cand_score
                improved = True
                print(f"Iter {iteration:02d}, mix t={t:.2f} → score {best_score:.6f}")
                break   # accept first improving step size

        # ------------------------------------------------------------------
        #   2c – if the LP mix gave nothing, try a cheap random perturbation
        # ------------------------------------------------------------------
        if not improved:
            perturb = np.exp(rng.normal(0.0, 0.04, size=n))
            candidate = best * perturb
            candidate = np.clip(candidate, 0.0, 1000.0)
            try:
                cand_score = float(evaluate_sequence(candidate.tolist()))
            except Exception:
                cand_score = np.inf
            if np.isfinite(cand_score) and cand_score < best_score:
                best = candidate
                best_score = cand_score
                print(f"Iter {iteration:02d}, random tweak → score {best_score:.6f}")
                # keep looping – the new best may open fresh slack

        iteration += 1

    # ----------------------------------------------------------------------
    # 3. Return the best sequence found
    # ----------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return [float(v) for v in best]

            try:
                cand_score = float(evaluate_sequence(cand.tolist()))
            except Exception:
                continue
            if np.isfinite(cand_score) and cand_score < best_score - 1e-9:
                best = cand
                best_score = cand_score
                improved = True
                print(f"Iter {iteration:02d}, mix t={t:.2f} → score {best_score:.6f}")
                break   # accept the first improving step size

        # ----------------------------------------------------------
        # 2c – if nothing improved, try a cheap random perturbation.
        # ----------------------------------------------------------
        if not improved:
            pert = np.exp(rng.normal(0.0, 0.04, size=n))
            cand = best * pert
            cand = np.clip(cand, 0.0, 1000.0)
            try:
                cand_score = float(evaluate_sequence(cand.tolist()))
            except Exception:
                cand_score = np.inf
            if np.isfinite(cand_score) and cand_score < best_score:
                best = cand
                best_score = cand_score
                print(f"Iter {iteration:02d}, random tweak → score {best_score:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # 3.  Return the best sequence discovered.
    # --------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return [float(v) for v in best]
