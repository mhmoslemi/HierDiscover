"""
Propose a candidate step function using a fast gradient‑descent / simulated‑annealing
scheme.  The objective to minimise is

    score = 2·n·max(a∗a) / (∑a)²

where “∗” is the ordinary (full) convolution.  The score is scale‑invariant,
so we keep the total mass equal to 1 during the optimisation.
The algorithm works on the sequence a (non‑negative, sum = 1) and repeatedly:

1. Computes the autocorrelation by means of an FFT (O(n log n)).
2. Finds the lag k where the convolution is maximal.
3. Forms the sub‑gradient of the term (a∗a)[k] :  g_i = a_{k−i} (zero if out of range).
4. Takes a small step a ← a − α·g, clips to non‑negative and renormalises.
5. Occasionally adds random Gaussian noise and/or symmetrises the sequence.
6. Accepts the new point if it improves the score or with a temperature‑dependent
   probability (simulated annealing) to escape local minima.

A simple block‑perturbation hill‑climber is run in the remaining time as a
polish‑step.  All operations are vectorised and use only NumPy, so thousands of
iterations fit comfortably into the 1000 s budget.

The function returns the best sequence found (as a plain Python list).  The
returned values are non‑negative and well below the 0‑1000 limits, thus
`evaluate_sequence` will give exactly the same score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def _fft_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """
    Full convolution a * a using a power‑of‑two FFT length.
    Returns the (2·n‑1) real result.
    """
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * a.size - 1]
    return conv


def _fast_score(a: np.ndarray, size: int) -> float:
    """
    Compute the evaluation score without calling evaluate_sequence.
    a is assumed to be non‑negative; it is not clipped here.
    """
    n = a.size
    conv = _fft_convolution(a, size)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def _gradient_from_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the term (a∗a)[k] w.r.t. a.
    For each i we have  ∂(a∗a)[k]/∂a_i = a_{k-i}  (zero when the index is out of range).
    """
    n = a.size
    g = np.zeros_like(a)
    if k < n:
        # indices 0 … k are involved
        g[: k + 1] = a[k::-1]
    else:
        # k ≥ n
        length = 2 * n - 1 - k
        start = k - (n - 1)
        g[start : start + length] = a[:length][::-1]
    return g


def _block_perturb(a: np.ndarray, rng: np.random.Generator,
                   min_width: int, max_width: int) -> np.ndarray:
    """
    One non‑negative block perturbation (multiplicative log‑normal factor)
    similar to the reference solution.
    """
    cand = a.copy()
    n = a.size
    width = int(rng.integers(min_width, max_width + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)

    # multiplicative log‑normal perturbation on the block
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))

    # occasional symmetrisation – useful for autocorrelation objectives
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])

    # enforce hard limits of evaluate_sequence (they do not affect the score)
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs) -> list:
    """
    Search for a non‑negative step function that minimises the provided
    evaluate_sequence() score.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence (as a Python list) found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 1. Determine problem size and initialise the starting sequence
    # ------------------------------------------------------------------
    try:
        height_sequence_1  # may raise NameError
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_seq.size
    except Exception:
        # fallback – a moderately large random Dirichlet vector
        n = 4096
        init_seq = rng.dirichlet(np.ones(n)).astype(np.float64)

    # Clip to the hard limits (they never become active after normalisation)
    np.clip(init_seq, 0.0, 1000.0, out=init_seq)

    # Normalise to unit total mass – the score is scale‑invariant
    total = init_seq.sum()
    if total < 1e-12:
        init_seq = np.full(n, 1.0 / n, dtype=np.float64)
        total = 1.0
    a = init_seq / total

    # Pre‑compute an FFT length that is a power of two (fast for all calls)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2. Evaluate the initial point
    # ------------------------------------------------------------------
    cur_score = _fast_score(a, fft_len)
    best_seq = a.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop – gradient descent + annealing
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    it = 0
    # Annealing / step‑size schedules (linear decay)
    STEP_START = 0.6
    STEP_END = 0.001
    TEMP_START = 0.12
    TEMP_END = 1e-5

    while time.monotonic() < deadline:
        # --- compute convolution and the worst lag ---------------------------------
        conv = _fft_convolution(a, fft_len)
        k_star = int(conv.argmax())                 # lag where a∗a is maximal

        # --- gradient of the active quadratic term ---------------------------------
        g = _gradient_from_max(a, k_star)

        # --- schedule parameters ----------------------------------------------------
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / (budget_s - 0.5))
        alpha = STEP_START * (1.0 - frac) + STEP_END * frac
        temperature = TEMP_START * (1.0 - frac) + TEMP_END * frac

        # --- propose a new point (gradient step) ----------------------------------
        cand = a - alpha * g
        # clip to non‑negative (nothing else can become negative)
        np.clip(cand, 0.0, np.inf, out=cand)

        # --- occasional random exploration -----------------------------------------
        if rng.random() < 0.12:
            # tiny Gaussian noise, scaled to the current mean magnitude
            noise = rng.normal(0.0, 0.015, size=n) * cand.mean()
            cand += noise
            np.clip(cand, 0.0, np.inf, out=cand)

        # --- optional symmetrisation (helps flatten the autocorrelation) ----------
        if rng.random() < 0.18:
            cand = 0.5 * (cand + cand[::-1])

        # --- renormalise to unit sum ------------------------------------------------
        s = cand.sum()
        if s < 1e-12:                     # fallback in the (unlikely) event of total collapse
            cand = a.copy()
            s = cand.sum()
        cand /= s

        # --- evaluate quickly -------------------------------------------------------
        cand_score = _fast_score(cand, fft_len)

        # --- simulated‑annealing acceptance -----------------------------------------
        improvement = cand_score < cur_score
        if improvement or (rng.random() < np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))):
            a = cand
            cur_score = cand_score

        # --- keep the best ever -----------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[iter {it:05d}] new best score = {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 4. Final polishing with the block‑perturbation hill‑climber
    # ------------------------------------------------------------------
    # Use any remaining time (if any) for a cheap hill‑climber that was
    # successful in the reference solution.
    min_width = max(1, n // 200)
    max_width = max(2, n // 12)
    while time.monotonic() < deadline:
        cand = _block_perturb(best_seq, rng, min_width, max_width)
        cand_score = _fast_score(cand, fft_len)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[block] improved best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the result (as a plain list)
    # ------------------------------------------------------------------
    # Scale to respect the 0‑1000 bounds (the values are far below that already)
    if best_seq.max() > 1000.0:
        best_seq = best_seq * (1000.0 / best_seq.max())

    print(f"[done] final best score ≈ {best_score:.6f}")
    return best_seq.tolist()
