def simplex_proj(v, z=1):
    """Projects a vector onto the simplex {x >= 0, sum x = z}."""
    v = np.asarray(v)
    if v.sum() == z and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

The method consists of three stages:

1. **Broad sampling** – try several families of analytic shapes
   (uniform, power‑law, mixtures of two power‑laws, discrete beta‑like
   shapes and Gaussians).  The best candidate is kept as a warm start.
2. **Targeted stochastic refinement** – repeatedly locate the index of
   the maximal convolution value, sample a contributing entry with
   probability proportional to its contribution and move a random
   fraction of its mass to a location that currently carries little
   mass.  This directly attacks the part of the sequence that creates
   the worst peak.
3. **Optional simulated‑annealing acceptance** – occasionally accept
   non‑improving moves with a small temperature to escape flat local
   minima.

All scores are computed directly (the same formula used inside
``evaluate_sequence``) on a unit‑sum normalisation; the final sequence
is scaled by 100 to stay safely above the evaluator’s lower bound.
The routine respects the 1000 s time limit and never returns an empty
or invalid list.
"""

def power_law_sequence(length: int, exponent: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-exponent), normalised to sum = 1."""
    idx = np.arange(1, length + 1, dtype=np.float64)
    w = np.power(idx, -exponent)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones(length, dtype=np.float64)
        w_sum = length
    return w / w_sum


def mixture_power_law_sequence(
    length: int, alpha1: float, alpha2: float, mix: float
) -> np.ndarray:
    """convex combination of two power‑laws, normalised to sum = 1."""
    idx = np.arange(1, length + 1, dtype=np.float64)
    w1 = np.power(idx, -alpha1)
    w2 = np.power(idx, -alpha2)
    w = mix * w1 + (1.0 - mix) * w2
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones(length, dtype=np.float64)
        w_sum = length
    return w / w_sum


def beta_sequence(length: int, a_param: float, b_param: float) -> np.ndarray:
    """
    Discrete analogue of a Beta(a_param,b_param) density on {1,…,length},
    normalised to sum = 1.
    """
    i = np.arange(1, length + 1, dtype=np.float64)
    w = np.power(i, a_param - 1.0) * np.power(length - i + 1, b_param - 1.0)
    w = np.clip(w, 0.0, None)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones(length, dtype=np.float64)
        w_sum = length
    return w / w_sum


def gaussian_sequence(length: int, sigma: float) -> np.ndarray:
    """Symmetric discrete Gaussian (mean at centre), normalised to sum = 1."""
    centre = (length - 1) / 2.0
    idx = np.arange(length, dtype=np.float64) - centre
    w = np.exp(-0.5 * (idx / sigma) ** 2)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones(length, dtype=np.float64)
        w_sum = length
    return w / w_sum


def linf_score(vector: np.ndarray) -> float:
    """
    Compute the objective used by ``evaluate_sequence``.
    The vector is assumed to be a 1‑D numpy array of non‑negative numbers.
    """
    s = vector.sum()
    if s < 0.01:
        return np.inf
    n = len(vector)
    conv = np.convolve(vector, vector)
    max_conv = conv.max()
    return float(2.0 * n * max_conv / (s * s))

            best_score, best_vec = sc, cand.copy()
            print(f"power‑law α={alpha:.3f}  score={best_score:.6f}")

    # Random mixtures of two power‑laws
    for _ in range(300):
        if time.monotonic() >= deadline:
            break
        a1 = rng.uniform(0.1, 2.0)
        a2 = rng.uniform(0.1, 2.0)
        mix = rng.uniform(0.0, 1.0)
        cand = mixture_power_law_sequence(n, a1, a2, mix)
        sc = linf_score(cand)
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()
            print(f"mix α1={a1:.3f} α2={a2:.3f} mix={mix:.3f}  score={best_score:.6f}")

    # Random discrete Beta‑like shapes
    for _ in range(200):
        if time.monotonic() >= deadline:
            break
        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        cand = beta_sequence(n, a, b)
        sc = linf_score(cand)
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()
            print(f"beta a={a:.2f} b={b:.2f}  score={best_score:.6f}")

    # Random centred Gaussians
    for _ in range(150):
        if time.monotonic() >= deadline:
            break
        sigma = rng.uniform(0.05 * n, 0.35 * n)
        cand = gaussian_sequence(n, sigma)
        sc = linf_score(cand)
        if sc < best_score:
            best_score, best_vec = sc, cand.copy()
            print(f"gaussian σ={sigma:.1f}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Targeted stochastic refinement.
    # --------------------------------------------------------------
    a = best_vec.copy()          # current working vector (sum = 1)
    total_mass = a.sum()         # stays equal to 1 throughout the loop

    # Simulated‑annealing parameters
    temp = 0.01
    temp_decay = 0.9999
    # Minimal time left for a batch of refinements
    while time.monotonic() < deadline:
        # ----- locate the worst convolution index -----
        conv = np.convolve(a, a)
        k_max = int(conv.argmax())
        # ----- compute contributions of each index to this peak -----
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k_max - i_vals
        contrib = a[i_vals] * a[j_vals]
        # ----- pick a source index weighted by its contribution -----
        if contrib.sum() > 0:
            probs_src = contrib / contrib.sum()
            src = rng.choice(i_vals, p=probs_src)
        else:
            src = rng.integers(0, n)

        # ----- pick a destination index that currently has little mass -----
        inv_mass = 1.0 / (a + 1e-15)
        inv_mass[src] = 0.0                     # never pick the same index
        probs_dst = inv_mass / inv_mass.sum()
        dst = rng.choice(n, p=probs_dst)

        # ----- perform a mass transfer -----
        frac = rng.uniform(0.10, 0.45)           # move 10‑45 % of the source mass
        delta = a[src] * frac
        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta                       # sum stays exactly 1

        # ----- evaluate the candidate -----
        cand_score = linf_score(cand)

        # ----- accept if improvement, otherwise possibly accept by annealing -----
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_vec = cand.copy()
            a = cand
            print(f"improve score={best_score:.6f}")
        else:
            # Metropolis criterion (temperature slowly decays)
            if rng.random() < np.exp(-(cand_score - best_score) / max(temp, 1e-12)):
                a = cand  # accept as the new state, but best remains unchanged

        # ----- anneal temperature -----
        temp *= temp_decay

    # --------------------------------------------------------------
    # 3️⃣  Final scaling and return.
    # --------------------------------------------------------------
    # Scale to a comfortable magnitude (the evaluator is scale‑invariant)
    final_seq = (best_vec * 100.0).tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"final best score={final_score:.6f}   length={len(final_seq)}")
    return [float(x) for x in final_seq]
