"""
A simple “soft‑max‑gradient‑descent” optimizer for the step‑function
minimisation problem.

The objective (up to a constant factor) is to make the self‑convolution
a*a as flat as possible.  The algorithm works with a normalised
sequence (∑a_i = 1) and repeatedly:

1. computes the full linear convolution a*a (FFT, O(n log n));
2. builds a smooth weighting w_k = exp(alpha·(c_k / max(c) – 1)) that
   concentrates on the largest convolution entries (alpha grows from
   ≈2 to ≈12 during the run);
3. obtains a sub‑gradient of the weighted max by convolving w with the
   reversed a;
4. projects the gradient onto the zero‑sum hyper‑plane, applies an
   Adam‑style update, clips to the allowed interval and re‑normalises;
5. keeps the best sequence seen so far.

Occasional random multiplicative perturbations (≈2 % log‑normal) are
applied to escape plateaus.  The whole routine stops after the supplied
time budget and returns the best feasible sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast linear convolution using an FFT (real input → real output)
# ----------------------------------------------------------------------
def conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution x⋆y (length len(x)+len(y)-1)."""
    n = len(x) + len(y) - 1
    # next power of two for efficient FFT
    n_fft = 1 << (n - 1).bit_length()
    fx = np.fft.rfft(x, n_fft)
    fy = np.fft.rfft(y, n_fft)
    conv = np.fft.irfft(fx * fy, n_fft)[:n]
    # numerical noise can create tiny negative values – clip safely
    return np.clip(conv, 0.0, None)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 900, **kwargs):
    """
    Search for a low‑score height sequence within the allotted time.
    Returns a list of floats (the best sequence found).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a starting length and initialise the sequence
    # --------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:          # height_sequence_1 may not exist
        base = None

    # Use the length of the known construction if possible,
    # otherwise fall back to a moderate size.
    if base is not None and base.ndim == 1 and base.size > 0:
        n = base.size
        a = base
        # normalise to ∑a_i = 1 (objective is scale‑invariant)
        s = a.sum()
        if s == 0.0:
            a = np.ones(n, dtype=np.float64) / n
        else:
            a = a / s
    else:
        n = 6000                     # a reasonable default
        a = np.ones(n, dtype=np.float64) / n

    # hard limits from the evaluation function
    a = np.clip(a, 0.0, 1000.0)

    # --------------------------------------------------------------
    # 2. Optimisation parameters
    # --------------------------------------------------------------
    # Adam optimiser parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Learning‑rate schedule (linear decay to zero)
    lr_start = 0.12                # relatively aggressive start
    # Soft‑max temperature (alpha) – grows from low to high
    alpha_start, alpha_end = 2.0, 12.0

    # Normalisation target (∑a_i = 1)
    target_sum = 1.0

    # Timing
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # reserve 1 s margin

    # --------------------------------------------------------------
    # 3. Initial score
    # --------------------------------------------------------------
    conv = conv_full(a, a)
    max_conv = conv.max()
    best_score = 2.0 * n * max_conv / (target_sum ** 2)
    best = a.copy()
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    iteration = 0
    # Main optimisation loop
    while time.monotonic() < deadline:
        # ----------------------------------------------------------
        # schedule for learning rate and soft‑max temperature
        # ----------------------------------------------------------
        elapsed = time.monotonic() - start_time
        prog = elapsed / (deadline - start_time)          # ∈[0,1]
        lr = lr_start * max(0.0, 1.0 - prog)               # linear decay
        alpha = alpha_start + prog * (alpha_end - alpha_start)

        # ----------------------------------------------------------
        # 3‑a. Full convolution a⋆a  (needed for both objective & w)
        # ----------------------------------------------------------
        conv = conv_full(a, a)
        max_conv = conv.max()
        if max_conv <= 0.0:               # should never happen, but guard
            break

        # ----------------------------------------------------------
        # 3‑b. Soft‑max weighting of convolution entries
        # ----------------------------------------------------------
        # w_k = exp(alpha * (c_k / max(c) - 1)) ∈ (0,1]
        w = np.exp(alpha * (conv / max_conv - 1.0))

        # ----------------------------------------------------------
        # 3‑c. Gradient of the weighted max (sub‑gradient of the objective)
        #     ∂/∂a_i  Σ_k w_k (a⋆a)_k = 2 * Σ_k w_k a_{k‑i}
        # ----------------------------------------------------------
        grad_full = conv_full(w, a[::-1])                 # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]           # central n entries
        # keep the total mass unchanged → project onto zero‑sum subspace
        grad -= grad.mean()

        # ----------------------------------------------------------
        # 3‑d. Adam update
        # ----------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.clip(a_new, 0.0, 1000.0)

        # re‑normalise to keep ∑a_i = target_sum
        s = a_new.sum()
        if s > 0.0:
            a_new = a_new * (target_sum / s)

        # ----------------------------------------------------------
        # 3‑e. Evaluate the candidate (fast, no extra FFT)
        # ----------------------------------------------------------
        conv_new = conv_full(a_new, a_new)
        max_new = conv_new.max()
        score = 2.0 * n * max_new / (target_sum ** 2)

        # ----------------------------------------------------------
        # 3‑f. Keep the best solution seen so far
        # ----------------------------------------------------------
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new                     # move the current point to the improvement
            if iteration % 1000 == 0:
                print(f"[{iteration:06d}] new best  score = {best_score:.6f} (α={alpha:.2f}, lr={lr:.5f})")
        else:
            a = a_new                     # continue walking even if not improved

        # ----------------------------------------------------------
        # 3‑g. Occasional random perturbation (helps escaping plateaus)
        # ----------------------------------------------------------
        if iteration % 5000 == 0:
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            a_pert = a * perturb
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            s = a_pert.sum()
            if s > 0.0:
                a_pert = a_pert * (target_sum / s)

            conv_pert = conv_full(a_pert, a_pert)
            max_pert = conv_pert.max()
            score_pert = 2.0 * n * max_pert / (target_sum ** 2)

            if score_pert < best_score:
                best_score = score_pert
                best = a_pert.copy()
                a = a_pert
                print(f"[{iteration:06d}] perturb improved score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Final report
    # --------------------------------------------------------------
    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    # convert to plain Python list as required by the harness
    return [float(v) for v in best]
