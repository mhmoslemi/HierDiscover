import numpy as np
import time

def power_law_weights(length, exponent):
    idx = np.arange(length, dtype=np.float64) + 1.0
    w = np.power(idx, -exponent)
    w_sum = w.sum()
    if w_sum == 0:
        return np.ones(length) / length
    return w / w_sum

import numpy as np
import time

def power_law_weights(length: int, exponent: float) -> np.ndarray:
    """
    Return a normalised (sum == 1) non‑negative vector of the given length
    whose entries follow a power‑law a_i ∝ (i+1)^(-exponent).
    """
    idx = np.arange(length, dtype=np.float64) + 1.0
    w = np.power(idx, -exponent)
    total = w.sum()
    if total == 0:
        w[:] = 1.0 / length
    else:
        w /= total
    return w
