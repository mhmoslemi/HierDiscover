Define gradient_step function:

def gradient_step(a, step_factor):
    n = a.shape[0]
    Lc = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, Lc)
    conv = np.fft.irfft(A * A, Lc)[:2*n-1]
    sum_a = np.sum(a)
    target = sum_a * sum_a / (2*n - 1)  # S^2/(2n-1)
    diff = conv - target
    # convolution diff with a (full)
    Lg = 1 << ((len(diff) + n - 1).bit_length())
    D = np.fft.rfft(diff, Lg)
    B = np.fft.rfft(a, Lg)
    grad_full = np.fft.irfft(D * B, Lg)[:len(diff) + n - 1]
    # extract central n entries:
    start = n - 1
    grad = grad_full[start : start + n]
    # step size
    max_abs = np.max(np.abs(grad))
    if max_abs < 1e-12:
        return a
    step = step_factor / (max_abs + 1e-12)
    a_new = a - step * grad
    # enforce nonneg, clip
    a_new = np.clip(a_new, 0.0, 1000.0)
    # optionally enforce symmetry
    a_new = 0.5 * (a_new + a_new[::-1])
    return a_new

"""
propose_candidate
-----------------
A two‑stage optimiser for the convolution‑max objective.

Stage 1 – fast global sweep:
    * Generate several families of “U‑shaped’’ or monotone sequences
      (power‑law, exponential, mixed‐power) of the required length.
    * Scale each to the allowed height (≤ 1000) and keep the best one
      according to a cheap implementation of ``evaluate_sequence``.

Stage 2 – projected gradient descent on the L₂‑error
    * Let a be the current non‑negative sequence (symmetrised to reduce
      the search space).  The full linear convolution c = a∗a is obtained
      by an FFT of length 2ⁿ (a power of two ≥ 3·N).  The target for a
      perfectly flat autocorrelation is  S²/(2N‑1)  with  S = Σa .
    * The gradient of ½‖c‑target‖² w.r.t. a is (c‑target)∗a, which we also
      obtain with FFTs.  A step in the opposite direction flattens the
      convolution.
    * After each step we enforce the hard constraints:
            – aᵢ ≥ 0,
            – aᵢ ≤ 1000,
            – a is symmetric (aᵢ = a_{N‑1‑i}),
      and we keep the best sequence seen so far.
    * Occasionally a random block‑wise log‑normal perturbation (the same
      mechanism that was useful in the reference solution) is applied
      to help escape shallow local minima and to diversify the search.

The algorithm runs for the supplied time budget (default ≈ 990 s) and
returns the best sequence found.  All heavy work is done with FFTs,
so even for N≈7000 the inner loop is fast enough to perform many
thousands of iterations within the allowed time.
"""

import time
import numpy as np

# Helper utilities – all at module level (no closures)
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _full_convolve_fft(a: np.ndarray, b: np.ndarray, nfft: int) -> np.ndarray:
    """Full linear convolution of two real vectors using a fixed FFT size."""
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(A * B, n=nfft)
    # keep only the first len(a)+len(b)-1 entries (full mode)
    return conv[: a.size + b.size - 1]

def _cheap_score(a: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` (used for the search loop).
    The function respects the clipping to [0,1000] and the sum‑threshold.
    """
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    n = a.shape[0]
    nfft = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def _gradient_step(a: np.ndarray, step_factor: float, L: int) -> np.ndarray:
    """
    One projected gradient descent step.
    The gradient of ½‖a∗a - target‖² is (a∗a - target) ∗ a.
    ``step_factor`` controls the aggressiveness of the update.
    ``L`` is a power‑of‑two >= 3·N, used for all FFTs in this step.
    """
    n = a.shape[0]

    # ----- a ∗ a ------------------------------------------------
    A = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

    # ----- target (flat value) ----------------------------------
    s = a.sum()
    target = (s * s) / (2 * n - 1)

    # ----- (a∗a - target) --------------------------------------
    diff = conv_full - target

    # ----- gradient = (diff) ∗ a -------------------------------
    B = np.fft.rfft(diff, n=L)
    grad_full = np.fft.irfft(A * B, n=L)[: (2 * n - 1) + n - 1]  # length 3n‑2
    start = n - 1
    grad = grad_full[start : start + n]

    # ----- step size -------------------------------------------------
    # Scale the step so that the largest absolute update is ``step_factor``.
    max_abs = np.max(np.abs(grad))
    if max_abs < 1e-16:
        return a
    step = step_factor / (max_abs + 1e-16)

    # ----- apply update, project onto feasible set --------------------
    a_new = a - step * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    # enforce exact symmetry (reduces the effective dimensionality)
    a_new = 0.5 * (a_new + a_new[::-1])
    return a_new

def _make_block_perturbation(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Random multiplicative log‑normal perturbation of a contiguous block."""
    n = seq.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand = seq.copy()
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation – useful for autocorrelation objectives
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

        cand = _gradient_step(best_seq, step_factor, L)
        cand_score = _cheap_score(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            last_improve = iteration
            step_factor = min(5.0, step_factor * 1.07)   # accelerate a bit
            print(f"[grad {iteration:5d}] improved -> {best_score:.6f}")

        else:
            # no improvement – slowly decay the step factor
            if iteration - last_improve > 30:
                step_factor = max(0.001, step_factor * 0.72)

        # occasional random block perturbation (helps escape plateaus)
        if iteration % 97 == 0:
            pert = _make_block_perturbation(best_seq, rng)
            pert_score = _cheap_score(pert)
            if pert_score < best_score:
                best_score = pert_score
                best_seq = pert
                print(f"[pert {iteration:5d}] improved -> {best_score:.6f}")
                step_factor = max(0.001, step_factor * 0.9)

        # keep step_factor within reasonable bounds
        if step_factor < 0.001:
            step_factor = 0.001
        if step_factor > 5.0:
            step_factor = 5.0

    # ------------------------------------------------------------------
    # 4. Final evaluation with the official function (just to be sure)
    # ------------------------------------------------------------------
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] best cheap score = {best_score:.6f}, official score = {final_score:.6f}")

    return best_seq.tolist()
