"""
Propose a candidate step‑function sequence.

The algorithm starts from the analytically optimal “arcsine” sequence
a[i] = C·binom(2i,i)/4**i (C chosen so that Σa = √(2n)).  Its self‑
convolution is flat, giving the classic bound π/2 ≈ 1.571.  To improve
upon this we iteratively solve a linear program:

    maximise   Σ b_j
    subject to   (a * b)[k] ≤ M   for all k,
                 b_j ≥ 0,

where M = max(a*a) and * denotes the discrete convolution.
The LP finds a second non‑negative sequence b that can be added
without violating the convolution bound.  After normalising b to the
same total mass as a we replace a by a convex combination
    a ← (1‑t)·a + t·b   (t small, e.g. 0.1).

Because Σa is preserved, the evaluation
    2·n·max(a*a) / (Σa)²
decreases monotonically as the maximum of the convolution shrinks.
The loop is repeated while time permits; occasional cheap random
perturbations (block scaling / mass transfer) are applied as a final
refinement step.

If SciPy is not available the routine falls back to the simple
stochastic search from the reference implementation.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Optional import of SciPy – used for the linear programme.
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _HAS_SCIPY = True
except Exception:                 # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Helper: build the “arcsine” (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2*i,i)/4**i .  The values lie in (0,1]."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i-1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Small random moves – used as a final local‑search polish.
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random fraction of mass from one index to another."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Linear‑programming step – one iteration of the Frank‑Wolfe style update.
# ----------------------------------------------------------------------
def lp_step(current: np.ndarray, M: float) -> np.ndarray | None:
    """Given the current sequence `current` and its max convolution M,
    solve  maximise Σb   s.t. (current * b) ≤ M , b ≥ 0.
    Returns the optimiser b (unnormalised) or None on failure."""
    n = current.size
    rows = 2 * n - 1

    # Build the Toeplitz‑like constraint matrix A where
    #   (A @ b)[k] = Σ_i current[i] * b_{k-i}
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = current

    # Right‑hand side is the bound M for every convolution index.
    b_ub = np.full(rows, M, dtype=np.float64)

    # Objective: maximise Σ b  → minimise -Σ b.
    c = -np.ones(n, dtype=np.float64)

    # Solve with HiGHS (fast interior‑point/simplex hybrid).
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:                     # pragma: no cover
        return None

    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of floats – the best step‑function heights found within
    ``budget_s`` seconds.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a workable length.  Use the provided reference sequence if
    # it exists, but cap the size to keep the LP tractable.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        ref_len = len(globals()["height_sequence_1"])
    else:
        ref_len = 0
    max_n = 1200                # safe upper bound for the dense LP
    n = max(64, min(ref_len, max_n))
    if n == 0:
        n = 1024                # fallback size

    # ------------------------------------------------------------------
    # 1.  Starting point – the analytically optimal arcsine shape,
    #     scaled so that Σa = √(2 n) (the usual normalisation).
    # ------------------------------------------------------------------
    a = build_arcsine_sequence(n)
    target_sum = math.sqrt(2.0 * n)
    a *= target_sum / a.sum()                    # normalise
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Iterative Frank‑Wolfe / LP improvement.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # leave slack
    iteration = 0
    t = 0.15                     # mixing coefficient (small, for stability)

    while time.monotonic() < deadline:
        iteration += 1

        # ---- current max convolution ---------------------------------
        conv = np.convolve(best, best)
        M = float(conv.max())

        # ---- linear programme (if available) -------------------------
        if _HAS_SCIPY:
            b_raw = lp_step(best, M)
        else:
            b_raw = None

        # ---- fallback: if LP failed, just break and use local search --
        if b_raw is None:
            print("[lp] LP step failed or unavailable – exiting LP loop")
            break

        # ---- normalise the new direction to the same total mass -------
        sum_b = float(b_raw.sum())
        if sum_b <= 0.0:
            print("[lp] LP returned zero vector – exiting LP loop")
            break
        scale = target_sum / sum_b
        b = b_raw * scale

        # ---- convex update --------------------------------------------
        candidate = (1.0 - t) * best + t * b
        candidate = np.clip(candidate, 0.0, 1000.0)

        # ---- evaluate -------------------------------------------------
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"[{iteration:>4}] new best = {best_score:.6f}")

        # keep iterating even if this particular mix did not improve;
        # the convex combination still reduces the max bound, so we
        # replace the working point.
        best = candidate

        # ---- optional schedule tweaks ---------------------------------
        # shrink the step size slowly to allow finer adjustments later.
        if iteration % 20 == 0 and t > 0.02:
            t *= 0.9

        # safety break if the score is already very good
        if best_score < 1.48:
            print("[early] target reached, breaking")
            break

    # ------------------------------------------------------------------
    # 3.  Final cheap stochastic polish (uses the same moves as the
    #    reference code).  Run as long as time permits.
    # ------------------------------------------------------------------
    polishing_deadline = start_time + float(budget_s) - 0.5
    it = 0
    while time.monotonic() < polishing_deadline:
        it += 1
        if rng.random() < 0.6:
            cand = block_perturbation(best, rng)
        else:
            cand = transfer_move(best, rng)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[polish {it:>5}] improved = {best_score:.6f}")

            if best_score < 1.48:
                print("[polish] excellent score reached – stop polishing")
                break

    print(f"[final] best score {best_score:.6f} after {iteration} LP iters + {it} polish iters")
    return [float(x) for x in best]
