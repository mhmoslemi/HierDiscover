"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  The search consists of three phases:

1. **Broad parametric sweep** – generate candidates from several
   families (power‑law, exponential, beta‑like, block‑uniform,
   triangular, arcsine).  For each family a grid of parameters is tried
   and the best sequence so far is kept.

2. **Targeted local optimisation** – repeatedly identify the index
   `k_max` where the self‑convolution attains its maximum.
   The contribution of each entry `i` to this peak is
   `w_i = a_i * a_{k_max-i}` (zero if the partner is out of range).
   A small fraction of mass is moved from an index with *large* `w_i`
   to one with *small* `w_j`.  Moves are only kept if they improve the
   objective; this directly attacks the current bottleneck.

3. **Stochastic annealing fallback** – if the directed moves stop making
   progress, occasional random redistributions of mass are performed.
   They are accepted according to a simple exponential cooling schedule,
   allowing the search to escape shallow local minima.

The routine respects the supplied time budget (default ≈ 950 s) and
returns the best sequence found within that time.  All numbers are
non‑negative and the total mass is kept far above the 0.01 lower bound
required by the evaluator.

    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function must return before this time).

    Returns
    -------
    list[float]
        The best sequence found (list of Python floats).

    TARGET_SUM = 100.0

    best_score = float('inf')

    # ----------------------------------------------------------------------

        return evaluate_sequence(arr.tolist())

        idx = np.arange(n, dtype=np.float64) + 1.0
        w = np.power(idx, -alpha)
        w /= w.sum()
        return w * TARGET_SUM

    def make_exp(lam: float) -> np.ndarray:
        idx = np.arange(n, dtype=np.float64)
        w = np.exp(-lam * idx)
        w /= w.sum()
        return w * TARGET_SUM

    def make_beta(p: float, q: float) -> np.ndarray | None:
        # Beta‑like density on {0,…,n‑1}: (i+1)^{p-1} (n-i)^{q-1}
        i = np.arange(n, dtype=np.float64)
        # keep the exponents > -1 to avoid singularities at the borders
        if p <= 0.0 or q <= 0.0:
            return None
        w = (i + 1.0) ** (p - 1.0) * (n - i) ** (q - 1.0)
        total = w.sum()
        if total == 0.0 or not np.isfinite(total):
            return None
        w /= total
        return w * TARGET_SUM

    def make_block(m: int) -> np.ndarray | None:
        if m <= 0 or m > n:
            return None
        w = np.zeros(n, dtype=np.float64)
        w[:m] = 1.0 / m
        return w * TARGET_SUM

        # Symmetric triangle, peak at middle, zero at the ends
        i = np.arange(n, dtype=np.float64)
        half = n // 2
        w = np.where(i <= half, i + 1.0, n - i)
        w /= w.sum()
        return w * TARGET_SUM

        i = np.arange(n, dtype=np.float64) + 0.5   # avoid zero

        top_k = max(1, int(0.02 * n))

        # indices with smallest w (but not zero‑mass to keep numerical stability)
        dst_candidates = np.argpartition(w, low_k)[:low_k]

        src = int(rng.choice(src_candidates))
        dst = int(rng.choice(dst_candidates))
        if src == dst:
            continue

        # Amount to move: a modest fraction of the source mass.
        max_move = a[src] * rng.uniform(0.05, 0.20)
        if max_move <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= max_move
        cand[dst] += max_move
        # Guard against accidental negative rounding errors.
        cand = np.clip(cand, 0.0, None)
        # Renormalise to keep total sum unchanged (scale‑invariant objective).
        cand = cand / cand.sum() * total_mass

        cand_score = eval_seq(cand)
        if np.isfinite(cand_score) and cand_score < best_score:
            best_score = float(cand_score)
            best_seq = cand.copy()
            a = cand
            no_improve_counter = 0
            print(f"[Iter {iteration:06d}] Directed improve → {best_score:.6f}")
            continue

        # If not better, keep the current a (but still count attempts)
        no_improve_counter += 1

        # ------------------------------------------------------------------
        # 3️⃣  Stochastic annealing fallback if we have stalled too long.
        # ------------------------------------------------------------------
        if no_improve_counter >= 500:
            # generate a purely random redistribution move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = a[i] * rng.uniform(0.01, 0.15)
            cand2 = a.copy()
            cand2[i] -= delta
            cand2[j] += delta
            cand2 = np.clip(cand2, 0.0, None)
            cand2 = cand2 / cand2.sum() * total_mass
            cand2_score = eval_seq(cand2)

            # Simple exponential cooling schedule
            temp = max(1e-4, best_score * 0.05 * math.exp(-0.0005 * iteration))
            accept = False
            if np.isfinite(cand2_score):
                if cand2_score < best_score:
                    accept = True
                else:
                    prob = math.exp(-(cand2_score - best_score) / temp)
                    if rng.random() < prob:
                        accept = True
            if accept:
                a = cand2
                if cand2_score < best_score:
                    best_score = float(cand2_score)
                    best_seq = cand2.copy()
                    print(f"[Iter {iteration:06d}] Annealed accept → {best_score:.6f}")
            no_improve_counter = 0   # reset stall counter after a random attempt

    # ----------------------------------------------------------------------
    # Final sanity check – ensure we return a plain Python list of floats.
    # ----------------------------------------------------------------------
    if best_seq is None:
        # Should never happen; as a final safeguard return uniform.
        best_seq = np.full(n, TARGET_SUM / n, dtype=np.float64)
        best_score = eval_seq(best_seq)
        print("Fallback (post‑search) uniform sequence:", best_score)

    print(f"Search finished. Best score = {best_score:.6f} (length={len(best_seq)})")
    return [float(x) for x in best_seq]
