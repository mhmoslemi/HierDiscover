def _evaluate_fast(seq: np.ndarray) -> float:
    n = seq.size
    conv = np.convolve(seq, seq)          # full convolution
    max_b = conv.max()
    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (total * total))

"""
Propose a candidate step‑function for the convolution minimisation problem.

The algorithm consists of three stages:

1. **Parametric exploration** –  a simple one‑parameter family
   `a_i = (i+1)^(-α)` (α ∈ (0,1)) is searched on a coarse grid and then
   refined with a finer grid.  The best exponent gives a very strong
   reduction of the objective (often < 1.5 for the default length).

2. **Local refinement** –  starting from the best power‑law vector we run a
   stochastic hill‑climber that performs cheap multiplicative block
   perturbations (log‑normal noise on a random sub‑segment) and occasional
   small updates of the exponent α.  The perturbations keep the vector
   non‑negative and respect the ≤ 1000 bound required by
   `evaluate_sequence`.

3. **Result** –  the best sequence found before the time budget expires is
   returned as a plain Python list.  The routine is fully self‑contained,
   uses only NumPy, and respects the 1000 second time limit (default
   900 s to leave a safety margin).
"""

import time
import numpy as np


def _evaluate_fast(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` for non‑negative vectors."""
    n = seq.size
    conv = np.convolve(seq, seq)          # full linear convolution
    max_b = conv.max()
    total = seq.sum()
    if total < 1e-12:                     # sum too close to zero – invalid
        return float("inf")
    return float(2.0 * n * max_b / (total * total))


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator,
                             sigma: float = 0.12) -> np.ndarray:
    """
    Apply a cheap O(n) multiplicative perturbation on a randomly chosen block.
    The block width is chosen between n/200 and n/12 (inclusive) and multiplied
    by a log‑normal factor exp(N(0, sigma)).  The result is clipped to the
    allowed domain [0, 1000].
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    # multiplicative log‑normal noise on the selected block
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, sigma)))
    # occasional symmetrisation – helps avoid a single dominant peak
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def propose_candidate(seed: int = 42,
                     budget_s: float = 900,
                     **kwargs) -> list[float]:
    """
    Generate a sequence of step heights.

    1. Determine the vector length (use the provided ``height_sequence_1`` if
       available, otherwise fall back to 2048).
    2. Perform a grid search over the power‑law exponent α to obtain a strong
       initial candidate.
    3. Run a stochastic hill‑climber (block perturbations + tiny α‑updates)
       until the allotted time runs out.
    4. Return the best list encountered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # Choose the problem dimension.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        n = 2048                     # reasonable default if no reference seq

    idx = np.arange(1, n + 1, dtype=np.float64)   # i = 1 … n

    # --------------------------------------------------------------
    # --- 1. Coarse grid search over the exponent α -----------------
    best_seq = None
    best_score = float("inf")
    best_alpha = 0.5

    # coarse grid (step ≈ 0.01)
    coarse_alphas = np.linspace(0.40, 0.90, 51)
    for alpha in coarse_alphas:
        if time.monotonic() > deadline:
            break
        seq = idx ** (-alpha)
        seq /= seq.sum()                 # normalise to unit sum (any scale works)
        score = _evaluate_fast(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            best_alpha = alpha
            print(f"grid  coarse: α={alpha:.4f}  score={score:.6f}")

    # --------------------------------------------------------------
    # --- 2. Refine the exponent around the current best -------------
    if time.monotonic() < deadline:
        lo = max(0.10, best_alpha - 0.05)
        hi = min(0.99, best_alpha + 0.05)
        fine_alphas = np.linspace(lo, hi, 101)    # step ~0.001
        for alpha in fine_alphas:
            if time.monotonic() > deadline:
                break
            seq = idx ** (-alpha)
            seq /= seq.sum()
            score = _evaluate_fast(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_alpha = alpha
                print(f"grid  fine  : α={alpha:.4f}  score={score:.6f}")

    # --------------------------------------------------------------
    # --- 3. Stochastic local optimisation (hill climbing) ----------
    # Mixing two move types:
    #   * block‑wise multiplicative noise
    #   * tiny change of the global exponent α
    while time.monotonic() < deadline:
        # decide move type (70 % block, 30 % exponent tweak)
        if rng.random() < 0.70:
            cand = _make_block_perturbation(best_seq, rng)
        else:
            # small Gaussian perturbation of the exponent
            delta = rng.normal(0.0, 0.003)            # keep changes tiny
            new_alpha = float(np.clip(best_alpha + delta, 0.10, 0.95))
            cand = idx ** (-new_alpha)
            cand /= cand.sum()
            # keep a little bit of the previous vector to stay near the basin
            cand = 0.9 * cand + 0.1 * best_seq
            cand = np.clip(cand, 0.0, 1000.0)

        # occasional global symmetrisation (helps stability)
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])

        score = _evaluate_fast(cand)
        if score < best_score:
            best_score = score
            best_seq = cand
            best_alpha = new_alpha if 'new_alpha' in locals() else best_alpha
            print(f"new best: α={best_alpha:.4f}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final report
    print(f"final best score = {best_score:.6f}  (α ≈ {best_alpha:.4f}, length = {n})")
    # Return as a plain Python list – the external evaluator will clip again.
    return best_seq.tolist()
