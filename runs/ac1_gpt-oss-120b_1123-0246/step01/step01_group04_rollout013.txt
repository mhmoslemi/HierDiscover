def _u_shape_seq(n, exponent=0.5, shift=0.0, uniform_weight=0.0):
    idx = np.arange(n, dtype=float)
    left = np.power(idx + shift + 1.0, -exponent)  # (i+1+shift)^-p
    right = np.power((n-1 - idx) + shift + 1.0, -exponent)
    base = left + right
    if uniform_weight > 0:
        base = (1 - uniform_weight) * base + uniform_weight * np.ones(n) / n
    base = np.clip(base, 0.0, 1000.0)
    base = base / base.sum()
    return base

    side = rng.choice(['left','right'])
    if side == 'left':
        w_low = max(1, n//200)
        w_high = max(2, n//20)
        width = int(rng.integers(w_low, w_high+1))
        start = 0

def _max_convolution(seq: np.ndarray) -> float:
    """Maximum entry of the linear convolution ``seq * seq``."""
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution ``seq * seq`` (real array)."""
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    return conv


def _fast_score(seq: np.ndarray) -> float:
    """Fast version of the evaluation (no safety checks)."""
    s = seq.sum()
    if s <= 1e-12:
        return float("inf")
    m = _max_convolution(seq)
    return 2.0 * seq.size * m / (s * s)

    """Return (score, full convolution) – useful for gradient moves."""
    s = seq.sum()
    if s <= 1e-12:
        return float("inf"), None
    conv = _full_convolution(seq)
    m = float(conv.max())
    return 2.0 * seq.size * m / (s * s), conv


# ----------------------------------------------------------------------
# Analytic families of candidates
# ----------------------------------------------------------------------


def _u_shape_seq(n: int, exponent: float = 0.5,
                 shift: float = 0.0, uniform_weight: float = 0.0) -> np.ndarray:
    """
    Symmetric “U” shape:
        a_i ∝ (i+shift+1)^{-exponent} + (n-1-i+shift+1)^{-exponent}
    optionally mixed with a uniform vector.
    """
    idx = np.arange(n, dtype=np.float64)
    left = np.power(idx + shift + 1.0, -exponent)
    right = np.power((n - 1 - idx) + shift + 1.0, -exponent)
    base = left + right
    if uniform_weight > 0.0:
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        base = (1.0 - uniform_weight) * base + uniform_weight * uniform
    # Clip to the limits used by evaluate_sequence and normalise
    np.clip(base, 0.0, 1000.0, out=base)
    total = base.sum()
    if total == 0.0:
        base[:] = 1.0 / n
    else:
        base /= total
    return base


def _random_u_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random member of the U‑shape family."""
    exponent = float(rng.uniform(0.2, 0.9))
    shift = float(rng.exponential(3.0))
    uniform_weight = float(rng.uniform(0.0, 0.3))
    return _u_shape_seq(n, exponent, shift, uniform_weight)


# ----------------------------------------------------------------------
# Local move operators
# ----------------------------------------------------------------------


def _reverse_swap(base: np.ndarray, rng: np.random.Generator,
                  frac: float = 0.07) -> np.ndarray:
    """Move a fraction of mass from i to its mirror n‑1‑i."""
    cand = base.copy()
    n = cand.size
    i = int(rng.integers(0, n // 2))
    j = n - 1 - i
    if cand[i] > 0.0:
        eps = frac * cand[i] * rng.uniform(0.1, 0.9)
        cand[i] = max(cand[i] - eps, 0.0)
        cand[j] += eps
    # mass is preserved → no renormalisation needed
    return cand


def _edge_swap(base: np.ndarray, rng: np.random.Generator,
               frac: float = 0.07) -> np.ndarray:
    """Transfer a small amount from the centre to a random edge block."""
    cand = base.copy()
    n = cand.size
    # choose a source in the central half
    src = int(rng.integers(n // 4, 3 * n // 4))
    # choose a destination near one of the edges
    if rng.random() < 0.5:
        dst = int(rng.integers(0, n // 8 + 1))
    else:
        dst = int(rng.integers(7 * n // 8, n))
    if cand[src] > 0.0:
        eps = frac * cand[src] * rng.uniform(0.1, 0.9)
        cand[src] = max(cand[src] - eps, 0.0)
        cand[dst] += eps
    return cand


def _block_scale(base: np.ndarray, rng: np.random.Generator,
                sigma: float = 0.12) -> np.ndarray:
    """Scale a random contiguous block (often near an edge) and renormalise."""
    cand = base.copy()
    n = cand.size
    w_low = max(1, n // 200)
    w_high = max(2, n // 20)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    scale = float(np.exp(rng.normal(0.0, sigma)))
    cand[start:stop] *= scale
    # keep values in the allowed range and renormalise
    np.clip(cand, 0.0, 1000.0, out=cand)
    cand /= cand.sum()
    return cand


def _gradient_swap(base: np.ndarray, rng: np.random.Generator,
                   step: float = 0.05):
    """
    Compute the shift k where the convolution is maximal,
    evaluate the gradient ∂(a*a)[k]/∂a_i = a_{i+k}+a_{i‑k},
    and move a small amount from a high‑gradient index to a low‑gradient
    one.
    """
    n = base.size
    score, conv = _full_score(base)
    if conv is None:
        return base.copy()
    k_max = int(np.argmax(conv))

    # gradient contributions
    grad = np.zeros(n, dtype=np.float64)

    # contribution from a_{i+k}
    if k_max < n:
        grad[: n - k_max] += base[k_max:]
        grad[k_max:] += base[: n - k_max]

    # for very large k (k ≥ n) the overlapping region is small; the
    # above already handles the possible indices, the rest stays zero.

    # pick source (large gradient) and destination (small gradient)
    i_high = int(np.argmax(grad))
    i_low = int(np.argmin(grad))

    # if both indices coincide nothing to do
    if i_high == i_low:
        return base.copy()

    cand = base.copy()
    if cand[i_high] > 0.0:
        eps = step * cand[i_high] * rng.uniform(0.1, 0.9)
        cand[i_high] = max(cand[i_high] - eps, 0.0)
        cand[i_low] += eps
        # renormalise to keep the sum exactly 1 (scale invariance)
        cand /= cand.sum()
    return cand


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 800, **kwargs):
    """
    Perform a time‑limited stochastic search for a sequence that minimises
    ``evaluate_sequence``.  The function returns the best list of floats
    discovered before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # Choose the problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base_seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = max(2000, base_seq.size)               # keep a reasonable lower bound
    else:
        n = 12000                                    # a size that still fits comfortably

    # Ensure we have exactly n entries (pad or truncate)
    if base_seq.size != n:
        if base_seq.size > n:
            base_seq = base_seq[:n]
        else:
            pad = np.zeros(n - base_seq.size, dtype=np.float64)
            base_seq = np.concatenate([base_seq, pad])

    # Clip to the limits enforced by ``evaluate_sequence``
    np.clip(base_seq, 0.0, 1000.0, out=base_seq)

    # --------------------------------------------------------------
    # Initialise a good starting point
    # --------------------------------------------------------------
    # 1) a symmetric U‑shape (no uniform mixing → fairly aggressive)
    current = _u_shape_seq(n, exponent=0.5, shift=0.0, uniform_weight=0.0)

    # 2) maybe the previous best (if provided)
    try:
        cand_score = float(evaluate_sequence(base_seq.tolist()))
        if cand_score < float(evaluate_sequence(current.tolist())):
            current = base_seq.copy()
    except Exception:
        pass

    best = current.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters for the stochastic walk
    # --------------------------------------------------------------
    iteration = 0
    temperature = 0.02                     # annealing temperature
    swap_frac = 0.07                       # fraction used by simple swaps
    block_sigma = 0.12                     # std‑dev for block scaling
    gradient_step = 0.05                   # step size for gradient swaps

    move_weights = np.array([30, 20, 15, 20, 15], dtype=float)   # percentages
    move_weights /= move_weights.sum()
    move_names = ["reverse", "edge", "block", "gradient", "random"]

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        move_type = rng.choice(move_names, p=move_weights)

        # ----------------------------------------------------------
        # generate a candidate
        # ----------------------------------------------------------
        if move_type == "reverse":
            cand = _reverse_swap(best, rng, frac=swap_frac)

        elif move_type == "edge":
            cand = _edge_swap(best, rng, frac=swap_frac)

        elif move_type == "block":
            cand = _block_scale(best, rng, sigma=block_sigma)

        elif move_type == "gradient":
            cand = _gradient_swap(best, rng, step=gradient_step)

        else:  # random – fresh sample from the analytic family
            cand = _random_u_shape(n, rng)

        # -----------------------------------------------------------------
        # safety – clip and renormalise (preserves scale‑invariance)
        # -----------------------------------------------------------------
        np.clip(cand, 0.0, 1000.0, out=cand)
        total = cand.sum()
        if total == 0.0:
            continue
        cand /= total

        # ----------------------------------------------------------
        # cheap evaluation first
        # ----------------------------------------------------------
        fast_sc = _fast_score(cand)

        # ----------------------------------------------------------
        # accept if it looks promising
        # ----------------------------------------------------------
        accept = False
        if fast_sc < best_score:
            accept = True
        else:
            # simulated‑annealing acceptance of worse moves
            delta = fast_sc - best_score
            if temperature > 0.0 and rng.random() < np.exp(-delta / temperature):
                accept = True

        if not accept:
            continue

        # ----------------------------------------------------------
        # authoritative evaluation (only for promising candidates)
        # ----------------------------------------------------------
        try:
            true_sc = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue

        if np.isfinite(true_sc) and true_sc < best_score:
            best = cand.copy()
            best_score = true_sc
            print(f"[{iteration:6d}] new best {best_score:.6f} (move={move_type})")

        # ----------------------------------------------------------
        # schedule updates
        # ----------------------------------------------------------
        if iteration % 2000 == 0:
            temperature = max(temperature * 0.90, 1e-4)
            swap_frac = max(swap_frac * 0.95, 0.01)
            block_sigma = max(block_sigma * 0.95, 0.02)
            gradient_step = max(gradient_step * 0.95, 0.01)

    # ------------------------------------------------------------------
    # final report
    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} after {iteration} iterations")
    # expose the result for possible reuse in a later run
    global height_sequence_1
    height_sequence_1 = best.copy()
    return [float(v) for v in best]
