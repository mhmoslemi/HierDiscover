"""
Hybrid local‑search for the convolution‑bound problem.

The algorithm starts from the analytically optimal “arcsine” sequence.
It then performs a mixture of cheap stochastic moves:
    – block scaling (as in the reference code)
    – simple mass transfers
    – targeted transfers from indices that contribute most to the current
      maximum convolution value toward indices that contribute least
    – early‑to‑late transfers (moving mass from the first part of the
      sequence to the last part)

All moves keep the total ℓ₁‑norm (the sum of the coefficients) fixed,
so the objective `2·n·max(b)/(sum(a)²)` is invariant under a global
scale‑factor and only the shape of the sequence matters.
The search accepts only improving moves (a simple hill‑climber); the
most promising candidate seen so far is remembered and returned.
The loop runs until the time budget expires (or the target score
1.503 is reached).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: arcsine (inverse‑square‑root) sequence
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i  (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence: a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Scoring routine (identical to evaluate_sequence but without the safety
# checks; we keep the numbers inside the allowed range ourselves).
# ----------------------------------------------------------------------
def sequence_score(seq: np.ndarray) -> float:
    """Return the objective value 2·n·max(b)/(sum(a)²) for a given sequence."""
    s = float(seq.sum())
    if s < 0.01:
        return np.inf
    b = np.convolve(seq, seq)
    max_b = float(b.max())
    n = seq.shape[0]
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Simple building blocks (all keep the ℓ₁‑norm fixed via renormalisation)
# ----------------------------------------------------------------------
def renormalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale seq so that its sum equals target_sum (clipping first)."""
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur == 0.0:
        return seq
    seq *= target_sum / cur
    return seq


def block_perturbation(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                 # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return renormalise(cand, target_sum)


def transfer_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """Move a random fraction (≤30 %) of mass from one index to another."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1                 # guarantee i != j
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


def early_to_late_move(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Move a fraction of mass from the early part of the sequence to its
    late part (keeps the total sum unchanged)."""
    n = base.size
    early_cut = max(1, n // 10)          # first 10 % of positions
    late_start = max(early_cut, n - early_cut)   # last 10 %
    i = int(rng.integers(0, early_cut))
    j = int(rng.integers(late_start, n))
    if i == j:
        return base.copy()
    frac = rng.random() * 0.40           # up to 40 % of the chosen early entry
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


def gradient_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """
    Reduce the current maximum of the convolution by moving mass from the
    index that contributes most to the maximal entries toward an index that
    contributes little.  The contribution of index i to the set of maximal
    convolution positions is approximated by
        a_i * (a * mask)[i],
    where mask marks the positions of the convolution attaining the global
    maximum.
    """
    n = base.size
    # convolution of the current sequence with itself
    b = np.convolve(base, base)
    max_b = b.max()
    # a tiny tolerance to collect the truly maximal positions
    tol = max_b * (1.0 - 1e-12)
    mask = (b >= tol).astype(np.float64)          # length = 2n‑1
    # contributions of each coefficient to those maximal positions
    contrib_conv = np.convolve(base, mask)        # length = 2n‑1
    contrib = base * contrib_conv[:n]              # one contribution per index
    # heavy (most responsible) and light (least responsible) indices
    heavy_idx = int(np.argmax(contrib))
    # pick a light index – we prefer a small contribution, but avoid the
    # same index as the heavy one
    light_candidates = np.where(contrib < (contrib.mean()))[0]
    if light_candidates.size == 0:
        # fall back to a random index different from heavy_idx
        light_idx = int(rng.integers(0, n - 1))
        if light_idx >= heavy_idx:
            light_idx += 1
    else:
        # ensure we do not pick the heavy index
        light_candidates = light_candidates[light_candidates != heavy_idx]
        if light_candidates.size == 0:
            # all low-contribution indices are the heavy one → pick random
            light_idx = int(rng.integers(0, n - 1))
            if light_idx == heavy_idx:
                light_idx = (light_idx + 1) % n
        else:
            light_idx = int(rng.choice(light_candidates))
    # transfer a fraction of the heavy mass to the light index
    frac = rng.random() * 0.30               # up to 30 %
    delta = frac * base[heavy_idx]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[heavy_idx] -= delta
    cand[light_idx] += delta
    return renormalise(cand, target_sum)

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of floats – the best step‑function heights found within
    ``budget_s`` seconds.  The algorithm:
        1. Initialise with the arcsine sequence (almost optimal analytically).
        2. Repeatedly apply a mixture of stochastic moves, including a
           gradient‑inspired transfer that explicitly reduces the current
           maximal convolution value.
        3. Keep the best sequence seen (according to the exact objective
           2·n·max(b)/(sum(a)²)).
    """
    rng = np.random.default_rng(seed)

    # Length of the problem – fall back to 256 if the global reference is missing.
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256

    # ------------------------------------------------------------------
    # 1.  Starting point – the analytically derived arcsine shape.
    # ------------------------------------------------------------------
    base_seq = build_arcsine_sequence(n)
    target_sum = float(base_seq.sum())
    cur_seq = renormalise(base_seq, target_sum)
    cur_score = sequence_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"[init] length={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Search loop
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)  # small slack
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # choose a move type (probabilities tuned empirically)
        r = rng.random()
        if r < 0.35:
            cand = block_perturbation(cur_seq, rng, target_sum)
        elif r < 0.55:
            cand = transfer_move(cur_seq, rng, target_sum)
        elif r < 0.80:
            cand = early_to_late_move(cur_seq, rng, target_sum)
        else:
            cand = gradient_move(cur_seq, rng, target_sum)

        cand_score = sequence_score(cand)

        # accept only improving moves (simple hill‑climber)
        if np.isfinite(cand_score) and cand_score < cur_score:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                print(f"[{it:>6}] new best = {best_score:.6f}")

                # early exit if we already beat the target comfortably
                if best_score < 1.503:
                    break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return as plain Python list – the harness expects a list of floats.
    return [float(x) for x in best_seq]
