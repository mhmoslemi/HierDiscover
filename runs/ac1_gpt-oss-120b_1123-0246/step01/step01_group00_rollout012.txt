"""
Propose a low‑score sequence for the step‑function problem.

Algorithm
---------
The score is   score = 2·n·max(a*a) / (sum(a)**2) .
With the conventional normalisation sum(a)=sqrt(2·n) the score equals
max(a*a).  Hence we want to minimise the largest entry of the autocorrelation
(a*a).

The core of the search is an *iterative linear‑programming* scheme inspired
by the paper that produced the previous state‑of‑the‑art.  From a current
candidate `a` we compute its worst autocorrelation value `M`.  Keeping `M`
fixed we solve the LP

    maximise    ∑ b_i
    subject to  (a * b)_k ≤ M   for every lag k = 0 … 2n‑2
                0 ≤ b_i ≤ 1000

which finds a new non‑negative sequence `b` that puts as much mass as
possible into positions where the current `a` is small (the constraints keep
the convolution with `a` below the current worst value `M`).  After normalising
`b` to the same total sum as `a` we move a fraction `τ` towards `b`:

        a←(1‑τ)·a + τ·b .

Because mass is shifted from “dangerous” positions to “safe” positions,
`max(a*a)` tends to drop.  The procedure is repeated until the time budget
expires.  Occasional random perturbations (log‑normal noise) are applied to
escape possible stagnation, and a simple gradient step is used as a cheap
fallback when the LP is infeasible.

The implementation works for sequences up to a few thousand entries
(we limit the dimension to ≤1500 to keep the LP size reasonable).  All
operations respect the security checks performed by `evaluate_sequence`.

The function prints progress information and finally returns the best
sequence found (as a list of floats).
"""

import time

def _downsample(arr: np.ndarray, new_len: int) -> np.ndarray:
    """
    Linearly interpolate a 1‑D array to a new length.
    If new_len >= len(arr) the original array is returned unchanged.
    """
    old_len = arr.shape[0]
    if new_len >= old_len or old_len == 0:
        return arr.copy()
    x_old = np.linspace(0, 1, old_len, dtype=np.float64)
    x_new = np.linspace(0, 1, new_len, dtype=np.float64)
    return np.interp(x_new, x_old, arr).astype(np.float64)


def _build_conv_matrix(a: np.ndarray) -> csr_matrix:
    """
    Build the sparse matrix representing the linear operator
        b ↦ a * b   (full convolution, length 2·n‑1)

    For each column j (the variable b_j) the entries a_i are placed on rows
    k = i + j.  The resulting matrix has shape (2·n‑1, n) and can be used as
    A_ub in ``scipy.optimize.linprog``.
    """
    n = a.shape[0]
    # row indices = i + j  for all i,j in 0..n-1
    i = np.arange(n, dtype=np.int32).reshape(-1, 1)      # shape (n,1)
    j = np.arange(n, dtype=np.int32).reshape(1, -1)      # shape (1,n)
    rows = (i + j).ravel()                               # length n*n
    cols = np.repeat(np.arange(n, dtype=np.int32), n)    # length n*n
    data = np.tile(a, n)                                 # a_i repeated for each j
    return coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def _objective_max_convolution(seq: np.ndarray) -> float:
    """
    Compute the raw objective value ``max(convolution)`` (no normalisation).
    """
    return np.max(np.convolve(seq, seq))


def _normalise_to_target_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale a non‑negative sequence so that its L1‑norm equals ``target_sum``.
    The sequence is first clipped to the allowed range [0, 1000] (the same
    limits used by ``evaluate_sequence``) and then rescaled.
    """
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur <= 0.0:
        return np.full_like(seq, target_sum / seq.size)
    return np.clip(seq * (target_sum / cur), 0.0, 1000.0)

            grad -= grad.mean()                 # preserve the total sum
            # Small step in opposite direction
            step = 0.02 * target_sum * max(1.0 - (iteration / 2000), 0.1)
            candidate = a - step * grad
            candidate = np.clip(candidate, 0.0, 1000.0)
            candidate = _normalise_to_target_sum(candidate, target_sum)

            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = candidate.copy()
                a = candidate.copy()
                print(f"[{iteration:6d}] grad improve score = {best_score:.6f}")

        # ---- (g) occasional random perturbation to escape plateaus -----
        if iteration % 120 == 0 and iteration > 0:
            # multiplicative log‑normal noise
            noise = rng.normal(0.0, 0.07, size=n)
            perturbed = a * np.exp(noise)
            perturbed = _normalise_to_target_sum(perturbed, target_sum)
            score = float(evaluate_sequence(perturbed.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = perturbed.copy()
                a = perturbed.copy()
                print(f"[{iteration:6d}] random perturb improved score = {best_score:.6f}")

        iteration += 1

    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    # Convert to plain Python list of floats as required by the evaluator.
    return [float(v) for v in best]
