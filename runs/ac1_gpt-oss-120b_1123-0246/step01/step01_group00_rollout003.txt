"""
Softmax‑gradient descent for minimizing the maximal autocorrelation of a
non‑negative step function.  The algorithm starts from the best known
construction (`height_sequence_1` if present) and performs a temperature‑
annealed gradient descent on a smooth approximation of
max_k (a*a)_k.  At each step the update direction is projected onto the
hyper‑plane of zero total mass, the step size decays linearly and periodic
multiplicative perturbations are applied to escape plateaux.  The sequence
with the smallest observed peak convolution (i.e. the smallest value of
evaluate_sequence) is returned.
"""

import time
import numpy as np

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*y using a power‑of‑two FFT."""
    n = x.size + y.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv

def _grad_and_maxb(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of the soft‑max approximation to max_k (a*a)_k
    and return it together with the exact maximum value of the autocorrelation.
    """
    n = a.size
    # Exact autocorrelation (needed for the score)
    b = _fft_convolve(a, a)
    maxb = b.max()

    # Soft‑max weights over all lags
    # subtract maxb for numerical stability
    exp_term = np.exp(beta * (b - maxb))
    w = exp_term / exp_term.sum()

    # Gradient: ∂/∂a_i  Σ_k w_k (a*a)_k = 2 Σ_k w_k a_{k‑i}
    w_rev = w[::-1]
    conv_rev = _fft_convolve(a, w_rev)               # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)                 # indices n‑1 … 2n‑2
    grad = 2.0 * conv_rev[idx]
    # Project onto the hyper‑plane Σ a_i = const (i.e. zero‑sum gradient)
    grad -= grad.mean()
    return grad, maxb

        Random seed for reproducibility.

    # ------------------------------------------------------------------

            base = base[lo:lo + 12000]
            n = base.size

        n = 4000
        base = np.ones(n, dtype=np.float64)

    a = np.clip(base, 0.0, 1000.0)

    # equal to the maximal convolution value.

    a = np.clip(a, 0.0, 1000.0)

    a *= (target_sum / a.sum())

    # 2. Initialise bookkeeping
    # ------------------------------------------------------------------
    # current best
    best_seq = a.copy()
    _, best_maxb = _grad_and_maxb(a, beta=1.0)   # we only need maxb
    best_score = best_maxb                      # because sum is fixed → score = maxb
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # algorithmic parameters
    beta_start = 0.5
    beta_end = 12.0                     # increase temperature gradually
    init_lr = target_sum * 0.02         # base step size (≈ 2 % of total mass)
    min_lr = init_lr * 0.02

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:

        now = time.monotonic()

        beta = beta_start + (beta_end - beta_start) * frac

        grad, _ = _grad_and_maxb(a, beta)

    # 5. Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    return [float(v) for v in best_seq]

Soft‑max‑gradient descent for the “max‑autocorrelation” step‑function problem.

The algorithm starts from the best known construction (`height_sequence_1`
if it exists, otherwise a uniform vector).  The sum of the coefficients is
kept fixed to √(2 n) – under this normalisation the evaluation function
reduces to the maximal value of the convolution a∗a.  We minimise a smooth
approximation of this maximum using a temperature‑annealed soft‑max.  At
each iteration

    • the gradient of the soft‑max objective is obtained analytically
      (∂/∂a_i = 2 ∑_k w_k a_{k‑i}) and is projected onto the
      zero‑sum hyper‑plane so that the total mass stays constant,
    • the update direction is normalised and a linearly decaying step‑size
      is applied,
    • after the update the vector is clipped to [0, 1000] and renormalised
      to the target sum,
    • the true objective (the maximal convolution) is evaluated; the best
      sequence ever seen is kept,
    • every few hundred iterations a multiplicative log‑normal perturbation
      is tried to escape plateaux, and if progress stalls a random restart
      is performed.

The convolution and the gradient are computed by FFT‑based convolution,
which is fast even for several‑thousand‑element vectors.  The function
returns the best sequence found before the time budget expires.
"""
