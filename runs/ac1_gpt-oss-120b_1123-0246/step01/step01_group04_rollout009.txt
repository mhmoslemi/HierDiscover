The theoretical lower bound of this expression is 2 (attained by the uniform
distribution).  Nevertheless the function is highly non‑convex and stochastic
search can be useful when the evaluation implementation is noisy (e.g. the
FFT‑based fast score used for pre‑filtering).  The algorithm below:

* Starts from a uniform vector (or from the previously discovered
  ``height_sequence_1`` if it exists).
* Repeatedly applies three cheap neighbourhood moves:
    – *mass swap*: move a fraction of the mass from one index to another.
    – *block scaling*: pick a random contiguous block and multiply it by a
      log‑normal factor.
    – *random mixture*: blend the current best with a power‑law profile.
* Uses a fast FFT‑based estimator to discard clearly worse candidates.
* Calls the authoritative ``evaluate_sequence`` only when a candidate improves
  the fast estimate.
* Stops when the allocated time budget expires (default 950 s, leaving a few
  seconds for clean‑up).

The routine returns a plain ``list[float]`` that respects all security checks
of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _max_convolution_fft(seq: np.ndarray) -> float:
    """
    Maximum entry of the full linear convolution seq * seq using FFT.
    The routine works for any length; the zero‑padding size is the next
    power of two that accommodates the full convolution result.
    """
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)[: 2 * n - 1]
    return float(np.max(conv))


def _fast_score(seq: np.ndarray) -> float:
    """Fast (FFT‑based) version of the objective used for pre‑filtering."""
    s = float(seq.sum())
    if s < 1e-12:
        return float("inf")
    return 2.0 * seq.size * _max_convolution_fft(seq) / (s * s)


def _uniform_seq(n: int) -> np.ndarray:
    """Uniform vector of length n (exactly satisfies the sum‑independent bound)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power_law_seq(n: int, exponent: float = 0.5, shift: float = 0.0) -> np.ndarray:
    """Generate a normalised power‑law profile a_i ∝ (i+shift+1)^(-exponent)."""
    idx = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power(idx + shift + 1.0, exponent)
    vals /= vals.sum()
    return vals


def _mix_with_powerlaw(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a candidate by mixing ``base`` with a fresh power‑law shape.
    ``alpha`` controls the amount of the original kept (0 → pure power‑law).
    """
    n = base.size
    exponent = float(rng.uniform(0.35, 0.65))
    shift = float(rng.exponential(5.0))
    power = _power_law_seq(n, exponent=exponent, shift=shift)
    alpha = float(rng.uniform(0.0, 0.4))
    cand = alpha * base + (1.0 - alpha) * power
    cand /= cand.sum()
    return cand


def _block_perturb(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Scale a randomly chosen contiguous block by a log‑normal factor.
    Occasionally symmetrises the whole vector to keep it balanced.
    """
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    scale = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= scale
    if rng.random() < 0.35:               # optional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    cand /= cand.sum()
    return cand


def _mass_swap(base: np.ndarray, rng: np.random.Generator, step: float) -> np.ndarray:
    """
    Transfer a small fraction (controlled by ``step``) of mass from a random
    source index to a random target index.
    """
    n = base.size
    i, j = rng.choice(n, size=2, replace=False)
    if base[i] == 0.0:
        return base
    eps = step * base[i] * rng.uniform(0.2, 0.8)
    cand = base.copy()
    cand[i] = max(cand[i] - eps, 0.0)
    cand[j] += eps
    cand /= cand.sum()
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 950.0, **kwargs):
    """
    Search for a sequence minimising `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Wall‑clock time budget (seconds).  The function stops a few seconds
        before the limit to avoid being killed.
    **kwargs : dict
        Ignored – kept for compatibility with the surrounding harness.

    Returns
    -------
    list[float]
        The best sequence found (plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the outer harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine a sensible length and initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        # use the length of the previously discovered best if available
        base = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        n = base.size if base.size > 0 else 2000
    else:
        n = 2000
        base = None

    # Start from a uniform distribution – it is provably optimal for the
    # exact objective, but we still run a local search because the evaluation
    # function may exhibit numerical noise.
    current = _uniform_seq(n)

    # If a previous candidate exists, evaluate it and keep the better one.
    if base is not None and base.size == n:
        try:
            prev_score = float(evaluate_sequence(base.tolist()))
            cur_score = float(evaluate_sequence(current.tolist()))
            if prev_score < cur_score:
                current = base.copy()
                cur_score = prev_score
        except Exception:  # pragma: no cover – safety net
            cur_score = float(evaluate_sequence(current.tolist()))
    else:
        cur_score = float(evaluate_sequence(current.tolist()))

    best = current.copy()
    best_score = cur_score
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main stochastic optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    swap_step = 0.12                       # initial step size for mass‑swap

    while time.monotonic() < deadline:
        iteration += 1
        move = rng.choice(["mix", "block", "swap"], p=[0.30, 0.45, 0.25])

        if move == "mix":
            cand = _mix_with_powerlaw(best, rng)

        elif move == "block":
            cand = _block_perturb(best, rng)

        else:  # "swap"
            cand = _mass_swap(best, rng, swap_step)

        # Fast pre‑filter – cheap FFT estimate
        fast = _fast_score(cand)
        if fast >= best_score:
            continue

        # Authoritative evaluation
        try:
            true_score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue

        if np.isfinite(true_score) and true_score < best_score:
            best = cand.copy()
            best_score = true_score
            print(f"[{iteration:6d}] new best: {best_score:.6f} (type={move})")

        # Gradually tighten the swap step to allow fine‑grained improvements
        if iteration % 400 == 0:
            swap_step = max(swap_step * 0.85, 0.005)

    print(f"[final] best score = {best_score:.6f} after {iteration} iterations")
    return [float(v) for v in best]
