    rng= np.random.default_rng(seed)

    # Determine n:

        n = max(base_len, 6000)  # maybe bigger

        if base_len < n:

        else:
            padded = base_seq[:n]
        p = padded / padded.sum() if padded.sum()>0 else np.full(n, 1/n)
    else:
        n = 6000

    # Normalize sum to 1 (since p is probability)

    iteration = 0

    while time.monotonic() < deadline:
        # Temperature schedule linear
        elapsed = time.monotonic() - start_time
        fraction = elapsed / (deadline - start_time)

"""
Propose a low‑score step‑function using a mix of:
1️⃣  Simple parametric families (linear‑decay, logistic‑decay) – cheap global scan.
2️⃣  Simulated‑annealing‐style refinement that moves probability mass from
    indices that increase the worst convolution lag toward indices that
    decrease it (gradient‑guided Monte‑Carlo).
All work is done on the normalised probability vector p (∑p = 1).  The
actual heights are a = √(2·n)·p, clipped to [0,1000] and re‑normalised at the end.
The heavy lifting (max‑convolution) uses an FFT‑based O(n log n) routine,
so thousands of candidates can be evaluated well within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(p: np.ndarray, fft_len: int) -> float:
    """
    Compute max_{k} (p * p)[k] using an FFT of length ``fft_len``.
    ``p`` is assumed to be a probability vector (sum == 1).
    """
    # real‑FFT based convolution (linear, zero‑padded)
    fp = np.fft.rfft(p, fft_len)
    conv = np.fft.irfft(fp * fp, fft_len)[: 2 * p.size - 1]
    return float(conv.max())


def _score_from_p(p: np.ndarray, fft_len: int, n: int) -> float:
    """
    Objective value for a probability vector ``p``.
    Because the final heights are a = √(2·n)·p,
    the evaluate_sequence value equals   2·n·max(p⋆p) .
    """
    return 2.0 * n * _conv_max(p, fft_len)


def _gradient_worst_lag(p: np.ndarray, k_max: int) -> np.ndarray:
    """
    Gradient of the worst convolution lag (ignoring the factor 2).
    Only entries that actually participate in the lag receive a non‑zero
    entry:  grad[i] = p[k_max‑i]  for i where the index is valid.
    """
    n = p.size
    grad = np.zeros_like(p)
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    if lo <= hi:
        i_idx = np.arange(lo, hi + 1, dtype=np.int64)
        j_idx = k_max - i_idx
        grad[i_idx] = p[j_idx]
    return grad


def _normalise_and_scale(p: np.ndarray, n: int) -> np.ndarray:
    """
    Turn a probability vector into a valid height sequence:
    a_i = √(2·n)·p_i, clipped to [0,1000] and re‑normalised to keep the sum
    exactly √(2·n) (required for the evaluate_sequence invariant).
    """
    target_sum = np.sqrt(2.0 * n)
    a = p * target_sum
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.0:
        a *= (target_sum / s)
    a = np.clip(a, 0.0, 1000.0)
    return a

# Main search routine
# ----------------------------------------------------------------------

    list[float]

        p_new = p_cur.copy()
        p_new[i] -= delta
        p_new[j] += delta
        # Numerical safety – renormalise to keep ∑p = 1
        p_new = p_new / p_new.sum()

        # ----------------------------------------------------------
        # 4e) Evaluate the proposal (fast FFT)
        # ----------------------------------------------------------
        fp_new = np.fft.rfft(p_new, fft_len)
        conv_new = np.fft.irfft(fp_new * fp_new, fft_len)[: 2 * n - 1]
        new_score = 2.0 * n * float(conv_new.max())

        # ----------------------------------------------------------
        # 4f) Metropolis acceptance
        # ----------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # uphill move allowed with probability exp(-Δ/Temp)
            if rng.random() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                accept = True

        if accept:
            p_cur = p_new
            cur_score = new_score

            if new_score < best_score:
                best_score = new_score
                best_p = p_new.copy()
                print(f"[{iteration:6d}] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # 4g) Occasionally inject a larger random perturbation
        # ----------------------------------------------------------
        if iteration % 3000 == 0 and iteration > 0:
            pert = np.exp(rng.normal(0.0, 0.15, size=n))
            p_pert = p_cur * pert
            p_pert /= p_pert.sum()
            fp_pert = np.fft.rfft(p_pert, fft_len)
            conv_pert = np.fft.irfft(fp_pert * fp_pert, fft_len)[: 2 * n - 1]
            pert_score = 2.0 * n * float(conv_pert.max())
            if pert_score < best_score:
                best_score = pert_score
                best_p = p_pert.copy()
                p_cur = p_pert
                cur_score = pert_score
                print(f"[{iteration:6d}] perturb improved score = {best_score:.6f}")
            else:
                # occasional acceptance of a worse perturbation
                if rng.random() < np.exp(-(pert_score - cur_score) / max(temp, 1e-12)):
                    p_cur = p_pert
                    cur_score = pert_score

        iteration += 1

    # --------------------------------------------------------------
    # 5️⃣  Convert the best probability vector back to a height sequence
    # --------------------------------------------------------------
    final_a = _normalise_and_scale(best_p, n)

    # Final verification using the provided evaluator (slow but done once)
    final_score = float(evaluate_sequence(final_a.tolist()))
    print(f"[final] n={n}, final score = {final_score:.6f}")

    # Return a plain Python list of floats as required
    return [float(v) for v in final_a]
