def conv_fft(x, y):
    n = len(x) + len(y) - 1
    L = 1 << int(np.ceil(np.log2(n)))
    fx = np.fft.rfft(x, L)
    fy = np.fft.rfft(y, L)
    conv = np.fft.irfft(fx*fy, L)[:n]
    return conv

Thus for conv of a with a: conv_fft(a, a). For conv of w and a_rev: conv_fft(w, a_rev).

"""
Simulated‑annealing + smoothed‑gradient search for the step‑function problem.

The algorithm works on sequences of length *n* (taken from the previous best
`height_sequence_1` if it exists, otherwise a power‑of‑two default).  The sum of
the entries is forced to `sqrt(2·n)`, because with this normalisation the
evaluation function is exactly the maximal autocorrelation value
`max_k (a*a)_k`.

A soft‑max approximation of the max is used to obtain a smooth gradient:
```
w_k = exp(p·(c_k - max(c)))   (p grows from a small to a large value)
grad_i ≈ 2·∑_k w_k·a_{k-i}
```
The gradient is projected onto the hyper‑plane of zero‑sum so that the total
mass stays constant, a step is taken, the result is clipped to
[0, 1000] and renormalised.  An exponential‑cooling temperature yields a
Metropolis acceptance rule, allowing occasional uphill moves.  Every few
hundred iterations a multiplicative random perturbation is applied to escape
plateaus.  The best sequence seen during the run is returned.

All convolutions are performed with FFTs (real‑valued) for speed, and the
search respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper convolution (real FFT, full linear convolution)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.size + y.size - 1
    # next power of two for efficient FFT
    L = 1 << int(np.ceil(np.log2(n)))
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[:n]
    return conv


def _max_autocorr(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute max_k (a*a)_k and return the value together with the full
    autocorrelation vector.
    """
    c = _conv_fft(a, a)          # length 2n‑1
    return float(c.max()), c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a sequence that minimises `evaluate_sequence`.
    The function returns the best list of non‑negative heights found within
    the given time budget (seconds).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Initialise a feasible sequence
    # --------------------------------------------------------------
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = a.size
    except NameError:
        # default length – a power of two makes FFT cheap
        n = 8192
        a = np.ones(n, dtype=np.float64)

    # Clip to the allowed range and enforce non‑negativity
    a = np.clip(a, 0.0, 1000.0)

    # Normalise the total sum to sqrt(2·n)  (makes the objective = max autocorr)
    target_sum = np.sqrt(2.0 * n)
    if a.sum() == 0.0:
        a[:] = target_sum / n
    else:
        a *= target_sum / a.sum()
    a = np.clip(a, 0.0, 1000.0)
    # Re‑normalise after clipping (tiny adjustment)
    a *= target_sum / a.sum()

    # --------------------------------------------------------------
    # 2. Initial evaluation
    # --------------------------------------------------------------
    cur_max, cur_conv = _max_autocorr(a)
    best = a.copy()
    best_score = cur_max
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Search parameters
    # --------------------------------------------------------------
    total_time = float(budget_s)
    deadline = time.monotonic() + max(0.0, total_time - 1.5)  # safety margin
    start = time.monotonic()
    iteration = 0

    # step‑size, smoothness and temperature schedules
    init_eta = 0.02 * target_sum
    init_p = 5.0          # start of soft‑max smoothing
    final_p = 200.0       # final smoothing (close to true max)
    init_T = cur_max * 0.2   # initial temperature for Metropolis rule

    # --------------------------------------------------------------
    # 4. Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        # ----------------------------------------------------------
        # recompute current autocorrelation (needed for soft‑max)
        # ----------------------------------------------------------
        cur_max, cur_conv = _max_autocorr(a)

        # keep the best ever seen
        if cur_max < best_score:
            best = a.copy()
            best_score = cur_max
            print(f"[{iteration:6d}] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # soft‑max weighting
        # ----------------------------------------------------------
        p = init_p + (final_p - init_p) * (elapsed / total_time)
        # exp() is safe because (cur_conv - cur_max) ≤ 0
        w = np.exp(p * (cur_conv - cur_max))

        # ----------------------------------------------------------
        # gradient of the smoothed max (via convolution)
        # ----------------------------------------------------------
        a_rev = a[::-1]

        start_idx = n - 1

        grad -= grad.mean()

        eta = init_eta * max(0.0, 1.0 - elapsed / total_time)

            if s2 > 0.0:

                best_score = pert_max
                a = pert_cand
                cur_max = pert_max

    print(f"[done] iterations={iteration}, best (max autocorr)={best_score:.6f}, "
          f"evaluate_sequence={final_score:.6f}")
    return [float(v) for v in best]
