def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate step‑function sequence that minimises
`evaluate_sequence`.

Algorithm
---------
1. **Problem reduction** – The score is scale‑invariant, therefore we
   normalise every candidate to sum = 1.  The objective becomes
   `2·n·max(conv(a,a))`.

2. **Initial exploration** – A cheap parametric sweep over
   power‑law and exponentially‑tilted power‑law families quickly finds a
   decent starting shape (exponent, shift, exponential decay, mixing with
   the uniform vector).

3. **Iterative improvement** – The main loop runs until the time budget
   expires and repeatedly applies one of four stochastic moves:
   * *Sub‑gradient step*:  compute the index `k*` where the autocorrelation
     attains its maximum, form the sub‑gradient `g_i = 2·a_{k*‑i}` and
     move opposite to it.  After the step the vector is projected back
     onto the probability simplex.
   * *Block perturbation*:  multiply a random contiguous block by a
     random factor (log‑normal scale) and optionally symmetrise.
   * *Mass‑swap*:  move a small fraction of mass from one index to
     another and renormalise.
   * *Parametric mix*:  generate a fresh candidate from the same analytic
     families as in the initial sweep.

   Candidates are first screened with a fast FFT‑based convolution; only
   improvements are verified with the authoritative `evaluate_sequence`.

4. **Step‑size schedules** –  The sub‑gradient step size decays as
   `η = η₀/√t` and the mass‑swap fraction slowly shrinks, allowing
   coarse moves early on and fine tuning later.

5. **Result** – The best sequence seen within the time limit is returned
   as a plain list of floats (already non‑negative and ≤ 1000).

The method combines analytic insight, a cheap gradient‑like descent on a
non‑convex objective, and diversified stochastic moves, and comfortably
fits into the 1000 s budget.
"""

import time
import numpy as np
import math

# Helper functions – top level, no closures

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of `seq` with itself using FFT."""
    n = seq.size
    size = _next_pow2(2 * n - 1)
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    return conv.real


def _fast_score_and_max(seq: np.ndarray):
    """
    Return (score, max_conv, argmax) using the same formula as
    `evaluate_sequence`, but without the safety checks.
    """
    n = seq.size
    s = float(seq.sum())
    if s < 1e-12:
        return float("inf"), None, None
    conv = _conv_full(seq)
    max_conv = float(conv.max())
    argmax = int(conv.argmax())
    score = 2.0 * n * max_conv / (s * s)
    return score, max_conv, argmax


def _fast_score(seq: np.ndarray) -> float:
    """Convenient wrapper returning only the score."""
    sc, _, _ = _fast_score_and_max(seq)
    return sc


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector `v` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u > cssv / ind)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_law_seq(n: int, exponent: float = 0.5, shift: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+shift+1)^(-exponent), normalised to sum = 1."""
    idx = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power(idx + shift + 1.0, exponent)
    vals /= vals.sum()
    return vals


def _exp_tilted_power_law_seq(
    n: int, exponent: float, shift: float, decay: float
) -> np.ndarray:
    """a_i ∝ (i+shift+1)^(-exponent) * exp(-decay * i), normalised."""
    idx = np.arange(n, dtype=np.float64)
    vals = (
        1.0
        / np.power(idx + shift + 1.0, exponent)
        * np.exp(-decay * idx)
    )
    vals /= vals.sum()
    return vals


def _mixed_seq(
    n: int, exponent: float, shift: float, decay: float, alpha: float
) -> np.ndarray:
    """
    Mixture of a uniform vector and an (exponential‑tilted) power‑law.
    alpha is the weight of the uniform part.
    """
    uniform = np.full(n, 1.0 / n)
    power = _exp_tilted_power_law_seq(n, exponent, shift, decay)
    mixed = alpha * uniform + (1.0 - alpha) * power
    mixed /= mixed.sum()
    return mixed


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a random log‑normal scaling to a contiguous block of `base`."""
    cand = base.copy()
    n = cand.size
    # width as a few percent of n
    w_low = max(1, n // 250)
    w_high = max(2, n // 10)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    sigma = rng.uniform(0.08, 0.30)          # occasionally larger moves
    scale = float(np.exp(rng.normal(0.0, sigma)))
    cand[start:stop] *= scale
    # optional symmetrisation
    if rng.random() < 0.30:
        cand = 0.5 * (cand + cand[::-1])
    # enforce the clipping performed by evaluate_sequence
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

    iteration = 0
    swap_step = 0.10          # fraction of source mass to move in a swap
    eta0 = 0.5                # initial sub‑gradient step size

    while time.time() < deadline:
        iteration += 1

        # Choose a move type
        move_type = rng.choice(
            ["subgrad", "block", "swap", "param"],
            p=[0.40, 0.35, 0.15, 0.10],
        )

        if move_type == "subgrad":
            # sub‑gradient step based on the current worst convolution index
            conv = _conv_full(best_seq)
            kmax = int(conv.argmax())
            # Sub‑gradient: g_i = 2 * a_{kmax - i} when the index lies in range
            rev = kmax - np.arange(n)
            mask = (rev >= 0) & (rev < n)
            g = np.zeros_like(best_seq)
            g[mask] = 2.0 * best_seq[rev[mask]]
            # Diminishing step size
            eta = eta0 / math.sqrt(iteration)
            cand = best_seq - eta * g
            cand = _project_to_simplex(cand)

        elif move_type == "block":
            cand = _make_block_perturbation(best_seq, rng)

        elif move_type == "swap":
            cand = best_seq.copy()
            i, j = rng.choice(n, size=2, replace=False)
            if cand[i] > 0.0:
                eps = swap_step * cand[i] * rng.uniform(0.2, 0.8)
                cand[i] = max(cand[i] - eps, 0.0)
                cand[j] = cand[j] + eps
            # renormalise to keep the sum exactly 1
            s = cand.sum()
            if s > 0.0:
                cand /= s

        else:   # "param" – fresh analytic candidate
            exponent = float(rng.uniform(0.30, 0.70))
            shift = float(rng.exponential(4.0))
            decay = float(rng.exponential(200.0))
            alpha = float(rng.uniform(0.0, 0.25))
            cand = _mixed_seq(n, exponent, shift, decay, alpha)

        # Fast pre‑filter
        fast_sc = _fast_score(cand)
        if fast_sc >= best_score:
            # not promising – skip the expensive check
            continue

        # Authoritative evaluation
        try:
            true_sc = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue

        if true_sc < best_score:
            best_seq = cand.copy()
            best_score = true_sc
            print(f"[{iteration:5d}] new best {best_score:.6f} ({move_type})")

        # slowly shrink the swap step size
        if iteration % 400 == 0:
            swap_step = max(0.01, swap_step * 0.85)

    print(f"[final] best score = {best_score:.6f} after {iteration} iterations")
    # Return a plain Python list – all entries already satisfy the checks.
    return [float(v) for v in best_seq]
