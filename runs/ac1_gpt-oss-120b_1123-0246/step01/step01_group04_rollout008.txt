def _simplex_projection(v):
    # ensure v sum to 1 and nonnegative
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step‑function sequence that minimises
`evaluate_sequence`.

Algorithm
---------
1. **Length selection** – If a previous construction is stored in the
   global `height_sequence_1`, its length is reused, otherwise a power‑of‑two
   length `n = 8192` is chosen (large enough to allow improvement while
   keeping FFTs cheap).

2. **Initial candidates** – Uniform, a simple power‑law (`a_i ∝ (i+1)^‑0.5`)
   and the optional previous best are evaluated (fast score).  The best of
   those becomes the starting point.

3. **Iterative optimisation** – A simulated‑annealing loop runs until the
   allocated time budget expires.  Four complementary move types are used
   (probabilities can be tuned):
   * **Exponentiated‑gradient step** – minimise the peak of the
     self‑convolution by updating the current probability vector `p`
     with `p ← p·exp(-η·∇)` where `∇` is the sub‑gradient of the maximum
     convolution entry.  The update is projected onto the simplex by
     normalisation.
   * **Block perturbation** – randomly scale a contiguous block of
     entries (the classic “local” move from the previous solution).
   * **Mix with uniform** – replace the current vector by a convex mixture
     with the uniform distribution.
   * **Random Dirichlet** – draw a fresh vector from a Dirichlet
     distribution with a random concentration parameter (producing
     sparse or dense shapes).

   The candidate’s *fast* score is obtained via an FFT‑based convolution
   (`_fast_score`).  If the fast score improves the global best,
   `evaluate_sequence` (the authoritative, safety‑checked function) is
   called and the best solution is updated.

   The annealing temperature decays slowly; uphill moves are accepted
   with probability `exp(-(Δscore)/T)`, allowing the search to escape
   local minima.

4. **Return** – When the budget is exhausted the best sequence found
   (as a plain list of Python floats) is returned.  All entries are
   non‑negative, bounded by the limits enforced inside `evaluate_sequence`.

The method combines a principled gradient descent on the convex
objective “max convolution” with stochastic perturbations, and fits well
within the 1000 s time limit.
"""

import time
import numpy as np

# Helper functions – top level (no closures)
# ----------------------------------------------------------------------


def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Compute the full convolution x * x using FFT (real input).
    Returns a real array of length 2*len(x)-1.
    """
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[: 2 * n - 1]
    return conv.real


def _max_convolution(x: np.ndarray) -> float:
    """Maximum entry of the full convolution x * x (FFT based)."""
    conv = _full_convolution(x)
    return float(conv.max())


def _fast_score(x: np.ndarray) -> float:
    """
    Fast version of `evaluate_sequence` – scale invariant.
    Assumes x is a 1‑D numpy array of non‑negative floats.
    """
    n = x.size
    s = float(x.sum())
    if s <= 0.0:
        return float("inf")
    max_conv = _max_convolution(x)
    return 2.0 * n * max_conv / (s * s)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { w >= 0, sum(w) = 1 }.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D vectors")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # All entries non‑positive – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = rho_arr[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _active_set(conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Indices where the convolution attains (or almost attains) its maximum."""
    max_val = conv.max()
    return np.where(conv >= max_val - eps)[0]


def _compute_gradient(p: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of `max_k (p*p)_k` at the current point `p`.
    For each active index k the contribution is
        ∂/∂p_i (p_i * p_{k-i}) = 2 * p_{k-i}
    (zero when the index is out of bounds).
    """
    n = p.size
    g = np.zeros(n, dtype=np.float64)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        # p_{k-i} corresponds to p[k - idx]
        g[i_start : i_end + 1] += 2.0 * p[k - idx]
    return g


def _exp_gradient_step(p: np.ndarray, eta: float, rng: np.random.Generator) -> np.ndarray:
    """
    One exponentiated‑gradient update of `p` (which lives on the simplex).
    The gradient is computed from the current active set of the convolution.
    """
    conv = _full_convolution(p)
    active = _active_set(conv)
    grad = _compute_gradient(p, active)
    # multiplicative update, always stays non‑negative
    new = p * np.exp(-eta * grad)
    total = new.sum()
    if total <= 0.0:
        # fallback to uniform in the unlikely event all entries vanish
        return np.full_like(p, 1.0 / p.size)
    return new / total


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block of `base` by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    w_low = max(1, n // 200)          # a few % of the length
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    scale = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= scale
    # occasional symmetrisation (helps for some lengths)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _power_law_seq(n: int, exponent: float = 0.5, shift: float = 0.0) -> np.ndarray:
    """Generate a normalised decreasing power‑law sequence."""
    idx = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power(idx + shift + 1.0, exponent)  # +1 avoids div‑by‑0
    vals /= vals.sum()
    return vals


def _random_dirichlet_seq(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(alpha,…,alpha)."""
    draws = rng.gamma(alpha, 1.0, size=n)
    s = draws.sum()
    if s <= 0.0:
        return np.full(n, 1.0 / n)
    return draws / s


def _mix_with_uniform(p: np.ndarray, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Convex mixture of `p` with the uniform distribution."""
    n = p.size
    uniform = np.full(n, 1.0 / n)
    mixed = alpha * uniform + (1.0 - alpha) * p
    mixed /= mixed.sum()
    return mixed

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a step‑function sequence that minimises `evaluate_sequence`.
    The function respects the given time budget and returns the best sequence
    found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------------
    # Determine problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            h1 = np.asarray(height_sequence_1, dtype=np.float64)
            if h1.ndim == 1 and h1.size > 0 and np.isfinite(h1).all():
                n = h1.size
                init_from_prev = True
            else:
                raise ValueError
        except Exception:
            n = 8192
            init_from_prev = False
    else:
        n = 8192
        init_from_prev = False

    # --------------------------------------------------------------
    # Build a few simple initial candidates and pick the best
    # --------------------------------------------------------------
    candidates = []

    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    candidates.append(uniform)

    # power‑law (the classic analytic start)
    power = _power_law_seq(n, exponent=0.5, shift=0.0)
    candidates.append(power)

    # optional previous best
    if init_from_prev:
        # normalise the previous vector to sum = 1
        vec = np.clip(h1, 0.0, 1000.0)
        if vec.sum() == 0.0:
            vec = uniform.copy()
        else:
            vec = vec / vec.sum()
        candidates.append(vec)

    # evaluate fast scores and keep the best as the start
    best_seq = None
    best_score = float("inf")
    for cand in candidates:
        sc = _fast_score(cand)
        if sc < best_score:
            best_seq = cand.copy()
            best_score = sc

    # authoritative check of the initial best (just in case)
    try:
        true_sc = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(true_sc) and true_sc < best_score:
            best_score = true_sc
    except Exception:
        pass

    print(f"[init] length={n}, initial fast score={best_score:.6f}")

    # --------------------------------------------------------------
    # Simulated‑annealing / gradient descent loop
    # --------------------------------------------------------------
    current_seq = best_seq.copy()
    current_fast = _fast_score(current_seq)

    # annealing parameters
    temperature = 0.02  # initial temperature for uphill moves
    eta0 = 0.5          # initial step size for exponentiated‑gradient
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # ---- choose a move -------------------------------------------------
        r = rng.random()
        if r < 0.50:
            # exponentiated‑gradient step
            eta = max(1e-4, eta0 / np.sqrt(iteration))
            cand = _exp_gradient_step(current_seq, eta, rng)

        elif r < 0.70:
            # block perturbation
            cand = _make_block_perturbation(current_seq, rng)

        elif r < 0.85:
            # blend with uniform
            alpha = rng.uniform(0.0, 0.4)  # mixing weight for uniform part
            cand = _mix_with_uniform(current_seq, rng, alpha)

        else:
            # fresh Dirichlet sample (random concentration)
            alpha_param = rng.uniform(0.05, 2.0)
            cand = _random_dirichlet_seq(n, rng, alpha_param)

        # ---- fast evaluation ------------------------------------------------
        cand_fast = _fast_score(cand)

        # ---- acceptance (simulated annealing) -------------------------------
        if cand_fast < current_fast:
            accept = True
        else:
            # uphill move accepted with probability exp(-Δ/T)
            delta = cand_fast - current_fast
            prob = np.exp(-delta / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            current_seq = cand
            current_fast = cand_fast

        # ---- global best update ---------------------------------------------
        if cand_fast < best_score:
            # authoritative evaluation (includes clipping, safety checks)
            try:
                true_score = float(evaluate_sequence(cand.tolist()))
            except Exception:
                true_score = float("inf")
            if np.isfinite(true_score) and true_score < best_score:
                best_seq = cand.copy()
                best_score = true_score
                print(f"[{iteration:6d}] NEW BEST {best_score:.6f}")

        # ---- temperature & step‑size schedules --------------------------------
        if iteration % 200 == 0:
            temperature *= 0.97
            temperature = max(temperature, 1e-4)

    # --------------------------------------------------------------
    # Final reporting
    # --------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} after {iteration} iterations")
    return [float(v) for v in best_seq]
