"""
Simulated‑annealing refinement of the Catalan (arcsine) step function.

The classical Catalan coefficients give a convolution that is everywhere
equal to 1, which yields the well‑known bound π/2 ≈ 1.5708 for the evaluation
score.  To push the score below 1.503 we alter the shape of the sequence:
mass is transferred from the large early coefficients (especially a₀)
towards the far tail.  This lowers the maximal self‑convolution while
keeping the total sum (≈ 1 after normalisation) essentially unchanged.
The optimisation is performed by a simple temperature‑controlled
Monte‑Carlo (simulated‑annealing) loop that repeatedly proposes a small
transfer of probability mass between two random positions.  The objective
to minimise is

    f(p) = 2·n·max(p∗p)   (p is a probability vector, ∑p_i = 1)

which is exactly the value returned by ``evaluate_sequence`` because the
function is homogeneous of degree 0.  Convolutions are computed with a fast
FFT (real‑valued) of length a power of two ≥ 2·n‑1, allowing many
iterations within the 1000 s budget.

The routine returns the best sequence found (as plain floats).  The global
variable ``height_sequence_1`` is used when available; otherwise the
Catalan start is generated with the same length.  No external files or
network access are required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Catalan‑type coefficients (the classic arcsine distribution)
# ----------------------------------------------------------------------
def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients C_i = binom(2i,i) / 4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence C_{i} = C_{i-1} * (2*i-1)/(2*i)
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    # safety clipping (the values are already ≤ 1)
    return np.clip(a, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Fast max‑convolution using FFT (real‑valued)
# ----------------------------------------------------------------------
def _max_self_convolution(p: np.ndarray, fft_len: int) -> float:
    """
    Return max_k (p * p)_k.
    Convolution is performed via FFT of length ``fft_len`` (≥2·len(p)-1).
    """
    # real FFT is enough because p is real
    ft = np.fft.rfft(p, n=fft_len)
    conv = np.fft.irfft(ft * ft, n=fft_len)
    # true linear convolution occupies the first 2·n-1 entries
    conv = conv[: 2 * p.size - 1]
    # avoid tiny negative round‑off values
    np.maximum(conv, 0.0, out=conv)
    return float(conv.max())


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 800, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence``.  The algorithm starts from the Catalan sequence
    (or the globally supplied ``height_sequence_1``), normalises it to a
    probability vector and then performs a simulated‑annealing search that
    transfers small amounts of mass between indices while keeping the sum
    equal to 1.  The objective being minimised is ``2·n·max(p∗p)``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. initialise the starting sequence (length n is taken from the
    #    provided global if possible, otherwise a default of 4800)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(base.size)
        start = base.copy()
    except Exception:
        n = 4800
        start = None

    if start is None:
        start = _catalan_sequence(n)

    # normalise to a probability vector (sum = 1).  This does not affect
    # the evaluation score because the objective is homogeneous.
    total = float(start.sum())
    if total < 1e-12:
        p = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        p = start.astype(np.float64) / total

    # clip just in case the global contains out‑of‑range numbers
    np.clip(p, 0.0, 1000.0, out=p)

    # pre‑compute FFT length (next power of two ≥ 2·n‑1)
    fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))

    # ------------------------------------------------------------------
    # 2. Evaluate the starting objective
    # ------------------------------------------------------------------
    cur_max = _max_self_convolution(p, fft_len)
    cur_obj = 2.0 * n * cur_max
    best_p = p.copy()
    best_obj = cur_obj

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)
    total_time = deadline - start_time

    print(f"Initial objective: {cur_obj:.6f} (n={n})")

    # annealing schedule parameters
    T0 = 1.0                     # initial temperature
    step_factor_start = 0.5      # initial max transfer = fraction of donor mass
    step_factor_end = 1e-4       # final fraction

    iteration = 0
    while time.monotonic() < deadline:
        # ---- temperature and step‑size (linear cooling) ----
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_time
        T = T0 * (1.0 - frac)          # temperature (decreases to 0)
        step_factor = (step_factor_start *
                       (1.0 - frac) + step_factor_end * frac)

        # ---- propose a move: transfer a small amount from i to j ----
        # donor i chosen with probability proportional to current mass
        i = rng.choice(n, p=p)
        # recipient j chosen uniformly, ensure j != i
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # max amount we may move out of i (keep p_i ≥ 0)
        max_delta = p[i] * step_factor
        if max_delta <= 0.0:
            continue

        delta = rng.random() * max_delta

        # construct the new probability vector
        p_new = p.copy()
        p_new[i] -= delta
        p_new[j] += delta

        # ---- evaluate the new candidate ----
        new_max = _max_self_convolution(p_new, fft_len)
        new_obj = 2.0 * n * new_max

        # ---- Metropolis acceptance ----
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            # probability to accept a worse move
            if T > 0.0:
                prob = np.exp((cur_obj - new_obj) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            p = p_new
            cur_obj = new_obj
            cur_max = new_max
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_p = p.copy()
                print(f"Iter {iteration:6d}: improved objective = {best_obj:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 3. Return the best sequence discovered.
    #    The vector is already a valid list of non‑negative floats; its
    #    sum equals 1, satisfying the security checks in ``evaluate_sequence``.
    # ------------------------------------------------------------------
    best_seq = best_p.tolist()
    # One final sanity check (does not affect the returned value)
    try:
        final_score = float(evaluate_sequence(best_seq))
        print(f"Final evaluated score: {final_score:.6f}")
    except Exception:
        # The sequence is still valid – the evaluation function will be called
        # by the grader later.
        print("Final evaluation failed internally; returning best found sequence.")

    return best_seq
