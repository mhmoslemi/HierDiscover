def conv_fft(x,y):
    n = len(x) + len(y) - 1
    L = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    res = np.fft.irfft(X * Y, L)[:n]
    return res

k<n, yielding the well‑known bound π/2≈1.5708 for the evaluation metric.
To beat this bound we start from the arcsine shape, normalise the sum to 1,
and then iteratively minimise the L∞‑norm of the convolution by a
soft‑max (smooth max) surrogate.  The gradient of the surrogate can be
written as a convolution, so each iteration needs only a few FFTs (O(n log n)
time).  A multiplicative (exponentiated‑gradient) update together with a
simple projection onto the probability simplex keeps the coefficients
non‑negative and their sum equal to 1.  Occasionally a random block
perturbation is injected to escape shallow local minima.  The best sequence
found before the time budget expires is returned.

The algorithm runs entirely in RAM, uses only NumPy and the provided
evaluate_sequence (only for sanity‑checking the final answer), and respects
the 0 ≤ a_i ≤ 1000 limits.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: build the arcsine (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """
    Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i) / 4**i  (exact up to floating‑point noise).
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    # recurrence: a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# FFT‑based convolution (real input)
# ----------------------------------------------------------------------
def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Return the (full) discrete convolution of real 1‑D arrays x and y.
    Uses zero‑padded FFT of length a power of two for speed.
    """
    n = x.size + y.size - 1
    L = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    conv = np.fft.irfft(X * Y, L)[:n]
    return conv


# ----------------------------------------------------------------------
# Score evaluation (identical to evaluate_sequence but cheaper)
# ----------------------------------------------------------------------
def compute_score(seq: np.ndarray) -> float:
    """
    Return the exact value of evaluate_sequence(seq) for a non‑negative
    sequence (no clipping needed).  The sum of the sequence is used as the
    normalisation.
    """
    b = conv_fft(seq, seq)
    max_b = b.max()
    sum_a = seq.sum()
    # protect against division by zero (should never happen)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simple random block perturbation (helps to escape plateaus)
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0, σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrisation (keeps the shape well‑behaved)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # enforce the hard 0‑1000 bounds from evaluate_sequence
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of floats – the best step‑function heights found within
    ``budget_s`` seconds.

    The routine:
      1. Takes the problem size from the global ``height_sequence_1``
         (fallback to 256).
      2. Starts from the analytically optimal arcsine sequence (normalised
         to sum = 1).
      3. Repeatedly:
           * computes a soft‑max approximation of the current maximum
             convolution value,
           * obtains its gradient via a second convolution,
           * performs an exponentiated‑gradient update,
           * occasionally applies a random block perturbation,
           * keeps the best sequence seen.
      4. Stops when the time budget is exhausted or a score below 1.5 is
         reached.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine the problem length.
    # ------------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256

    # ------------------------------------------------------------------
    # 2. Initialise with the arcsine sequence and normalise its sum.
    # ------------------------------------------------------------------
    a = build_arcsine_sequence(n).astype(np.float64)
    a /= a.sum()                     # enforce sum == 1  (scale‑invariant)
    best_seq = a.copy()
    best_score = compute_score(a)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Parameters for the soft‑max/gradient descent.
    # ------------------------------------------------------------------
    beta = 5.0                       # temperature for soft‑max (low → smooth)
    beta_max = 150.0
    eta = 100.0                      # step size for exponentiated gradient
    min_eta = 5.0
    decay_eta = 0.9999               # gradual reduction of η
    iteration = 0

    # ------------------------------------------------------------------
    # 4. Main optimisation loop.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)  # keep 1 s slack
    while time.monotonic() < deadline:
        iteration += 1

        # ---- convolution a*a (used for the soft‑max) -----------------
        b = conv_fft(a, a)
        b_max = b.max()

        # ---- soft‑max weights w_k ≈ 1_{b_k ≈ max} --------------------
        # shift by the maximum to avoid overflow
        w = np.exp(beta * (b - b_max))
        w_sum = w.sum()
        if w_sum == 0.0:                     # should not happen, but be safe
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the soft‑max (2 * a * w) --------------------
        g_full = conv_fft(a, w)               # length = n + (2n‑1) – 1 = 3n‑2
        g = 2.0 * g_full[:n]                  # keep only the first n entries

        # ---- exponentiated‑gradient update ---------------------------
        new_a = a * np.exp(-eta * g)
        s = new_a.sum()
        if s == 0.0:                          # all entries vanished -> revert
            new_a = a.copy()
            s = new_a.sum()
        a = new_a / s                          # project back onto the simplex

        # ---- occasional random block perturbation --------------------
        if rng.random() < 0.05:               # 5 % chance per iteration
            a = block_perturbation(a, rng)
            a /= a.sum()

        # ---- evaluate the new candidate -----------------------------
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{iteration:>6}] new best = {best_score:.6f}")

            # early stop if we have already crossed the target threshold
            if best_score < 1.5:
                break

        # ---- schedule updates ----------------------------------------
        # increase the soft‑max sharpness slowly
        if beta < beta_max:
            beta = min(beta_max, beta * 1.001)

        # slowly decay the learning rate to fine‑tune at the end
        eta = max(min_eta, eta * decay_eta)

    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    # ------------------------------------------------------------------
    # 5. Return the best sequence as a plain Python list.
    # ------------------------------------------------------------------
    # Guard against the tiny clipping performed by evaluate_sequence.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
