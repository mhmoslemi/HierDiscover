"""
search_step_function.py
-----------------------

Search for a non‑negative step‑function (a list of heights) that minimises
the provided ``evaluate_sequence`` objective.

The algorithm consists of three phases:

1. **Initialisation**
   - Use the length of a previously found sequence (``height_sequence_1``)
     if it exists, otherwise fall back to a default length of 6000.
   - Start from a simple analytic shape: a power‑law
     `a_i ∝ (i + shift + 1)^{-α}` (the “continuous optimum” for this
     problem).  This shape already yields a score ≈ 1.57.

2. **Parameter sweep**
   - Perform a cheap grid search over the exponent `α` (≈ 0.45 … 0.70)
     and a small integer shift (0 … 10).  The fast score
     ``_fast_score`` (FFT based, O(n log n)) is used to prune the
     candidates.
   - The best few candidates are re‑evaluated with the authoritative
     ``evaluate_sequence`` (which contains clipping and safety checks).

3. **Local refinement**
   - Around the best parameters found in the sweep we run a finer
     search (step ≈ 0.001 for `α` and step ≈ 0.1 for `shift`), again
     using the fast score and finally confirming the best one with
     ``evaluate_sequence``.

All helper functions are defined at module level; there are no closures
or lambda expressions.  The routine respects a time budget (default
500 s, you can request up to the full 1000 s) and prints progress
information.

The returned value is a plain Python list of floats ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _max_convolution(seq: np.ndarray) -> float:
    """
    Compute the maximum entry of the full convolution ``seq * seq``.
    FFT based O(n log n) implementation.
    """
    n = seq.size
    # zero‑pad to next power of two for efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    return float(np.max(conv.real))


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` – the same quantity without the
    security checks.  The output is exactly the value returned by
    ``evaluate_sequence`` for a correctly normalised sequence.
    """
    n = seq.size
    s = float(seq.sum())
    if s < 1e-12:
        return float("inf")
    max_conv = _max_convolution(seq)
    return 2.0 * n * max_conv / (s * s)


def _power_law_seq(n: int, exponent: float = 0.5, shift: float = 0.0) -> np.ndarray:
    """
    Generate a normalised power‑law sequence:
        a_i ∝ (i + shift + 1)^{-exponent},
    then scale so that the sum equals 1.
    """
    idx = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power(idx + shift + 1.0, exponent)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 500.0, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns a plain list of floats (the best sequence found before the
    time budget expires).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)   # small safety margin

    # ------------------------------------------------------------------
    # Choose length and initialise the current best
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size if base.ndim == 1 and base.size > 0 else 6000
    else:
        n = 6000

    # start from the classical 1/√(i+1) shape
    best_seq = _power_law_seq(n, exponent=0.5, shift=0.0)
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] length = {n}, initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – coarse grid search on (exponent, shift)
    # ------------------------------------------------------------------
    exp_candidates = np.linspace(0.45, 0.70, 51)   # step ≈ 0.005
    shift_candidates = np.arange(0, 11, 1)        # integer shifts 0…10

    # keep a small list of the top‑K fast scores to refine later
    K = 10
    best_fast = []   # entries: (fast_score, exponent, shift)

    for α in exp_candidates:
        for sh in shift_candidates:
            if time.monotonic() >= deadline:
                break
            cand = _power_law_seq(n, exponent=α, shift=float(sh))
            fsc = _fast_score(cand)
            # keep only the most promising candidates
            if len(best_fast) < K:
                best_fast.append((fsc, α, sh, cand))
                best_fast.sort(key=lambda x: x[0])
            elif fsc < best_fast[-1][0]:
                best_fast[-1] = (fsc, α, sh, cand)
                best_fast.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Phase 2 – precise refinement around the best candidates
    # ------------------------------------------------------------------
    refined_best_seq = None
    refined_best_score = best_score

    for fsc, α, sh, _ in best_fast:
        # fine neighbourhood: exponent +-0.005, shift +-1 with finer steps
        α_low, α_high = max(0.3, α - 0.006), min(1.5, α + 0.006)
        sh_low, sh_high = max(0.0, sh - 1.2), sh + 1.2

        fine_exps = np.linspace(α_low, α_high, 13)   # step ≈ 0.001
        fine_shifts = np.linspace(sh_low, sh_high, 13)  # step ≈ 0.2

        for αf in fine_exps:
            for shf in fine_shifts:
                if time.monotonic() >= deadline:
                    break
                cand = _power_law_seq(n, exponent=αf, shift=shf)
                fast_sc = _fast_score(cand)
                if fast_sc >= refined_best_score:
                    continue
                # authoritative check only when we see a genuine improvement
                true_sc = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(true_sc) and true_sc < refined_best_score:
                    refined_best_seq = cand.copy()
                    refined_best_score = true_sc
                    print(f"[refine] new best score {refined_best_score:.6f} "
                          f"(α={αf:.5f}, shift={shf:.3f})")

    # ------------------------------------------------------------------
    # If refinement found something better, adopt it
    # ------------------------------------------------------------------
    if refined_best_seq is not None:
        best_seq = refined_best_seq
        best_score = refined_best_score

    # ------------------------------------------------------------------
    # Phase 3 – optional lightweight stochastic perturbations
    # (helps to escape tiny local plateaus)
    # ------------------------------------------------------------------
    swap_step = 0.1
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # choose a simple local move
        cand = best_seq.copy()
        i, j = rng.choice(n, size=2, replace=False)
        if cand[i] > 0.0:
            eps = swap_step * cand[i] * rng.uniform(0.2, 0.8)
            cand[i] = max(cand[i] - eps, 0.0)
            cand[j] += eps
        cand /= cand.sum()                     # re‑normalise to keep sum = 1
        fast_sc = _fast_score(cand)
        if fast_sc >= best_score:
            continue
        true_sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(true_sc) and true_sc < best_score:
            best_seq = cand.copy()
            best_score = true_sc
            print(f"[{iteration:05d}] local‑swap improved to {best_score:.6f}")

        # slowly shrink the swap size for fine‑tuning
        if iteration % 500 == 0:
            swap_step = max(0.01, swap_step * 0.9)

    print(f"[final] best score = {best_score:.6f} after {iteration} iterations")
    # Return a plain Python list – values already lie in the required range
    return [float(v) for v in best_seq]
