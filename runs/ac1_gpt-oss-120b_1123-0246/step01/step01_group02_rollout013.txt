def solve_lp_direction(f, M):
    n = len(f)
    # number of constraints: 2*n-1
    K = 2*n - 1
    rows = []
    cols = []
    data = []
    for k in range(K):
        # For each j in [0, n-1], coefficient = f_{k - j} if within bounds.
        # iterate i where i = k - j
        # Equivalent to j = k - i where i in [0,n-1]; j ∈ [0,n-1]
        # So for each i in [max(0, k-(n-1)), min(n-1,k)]; set coeff f_i for g_{k-i}.
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # For i from i_start to i_end, j = k - i
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])  # coefficient is f_i (since g_j multiplies f_i)
    # convert to csr
    A = csr_matrix((data, (rows, cols)), shape=(K, n))
    # variable bounds: (0, None)
    b = np.full(K, M)
    # Use linprog to maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not result.success:
        # fallback to current f perhaps
        return None
    g0 = result.x
    sum_g0 = g0.sum()
    return g0, sum_g0

"""
Propose a candidate step‑function using Simulated Annealing.

The algorithm starts from the analytically optimal "arcsine" sequence
(a_i = C·binom(2i,i)/4**i).  This sequence already gives a score close
to π/2 ≈ 1.5708.  A simple stochastic local search that only accepts
improvements gets stuck at this value.  To escape the flat plateau we
use a Simulated Annealing scheme:

* The temperature is decreased linearly during the allotted time.
* Three complementary moves are used
    – transfer a random fraction of mass from an early index to a late index,
    – multiply a random contiguous block by a random factor,
    – rebalance the prefix/suffix mass (scale early part down, late part up)
      while keeping the total sum roughly constant.
* A candidate is always accepted if it improves the score; otherwise it
  can be accepted with probability exp(-(Δscore)/T).

Because the evaluation function is scale–invariant we do not enforce a
fixed sum – the moves directly optimise the true objective
score = 2·n·max(a*a)/(sum(a)²).

The routine runs until the time budget expires (default 500 s, the
harness may give up to 1000 s).  The best sequence found is returned as
a plain list of floats.
"""

import numpy as np
import time
from math import exp

# ----------------------------------------------------------------------
# Helper to build the analytically optimal arcsine sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i  (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Move primitives
# ----------------------------------------------------------------------
def transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from an early index to a later index."""
    n = base.size
    # pick early index (first quarter) and late index (last quarter)
    i = int(rng.integers(0, n // 4))
    j = int(rng.integers(3 * n // 4, n))
    if base[i] == 0.0:
        return base.copy()
    # fraction of the source that we move (up to 90 %)
    frac = rng.random() * 0.9
    delta = frac * base[i]
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.13)))
    cand[start:stop] *= factor
    # occasional symmetrisation – helps keep the shape regular
    if rng.random() < 0.30:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def rebalance_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Shift mass from an early prefix to a later suffix while roughly
    preserving the total sum."""
    n = base.size
    # cut point somewhere between 10 % and 90 % of the sequence
    cut = int(rng.integers(n // 10, 9 * n // 10))
    prefix = base[:cut]
    suffix = base[cut:]

    sum_prefix = prefix.sum()
    sum_suffix = suffix.sum()
    if sum_prefix == 0.0:
        return base.copy()

    # amount of mass we want to move (up to half of the prefix mass)
    amount = rng.random() * 0.5 * sum_prefix

    # scale the prefix down
    factor_pre = max(0.0, (sum_prefix - amount) / sum_prefix)
    new_prefix = prefix * factor_pre

    # scale the suffix up (if suffix sum is zero we distribute evenly)
    if sum_suffix > 0.0:
        factor_suf = (sum_suffix + amount) / sum_suffix
        new_suffix = suffix * factor_suf
    else:
        new_suffix = np.full_like(suffix, amount / suffix.size)

    cand = np.concatenate([new_prefix, new_suffix])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Simulated‑Annealing search for a step‑function that minimises
    ``evaluate_sequence``.  The routine works inside the given time budget
    and returns the best sequence seen.
    """
    rng = np.random.default_rng(seed)

    # Determine the length to use.  If the harness provides a reference
    # sequence we match its length, otherwise we fall back to 256.
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256

    # ------------------------------------------------------------------
    # 1. Starting point – the analytically optimal arcsine shape.
    # ------------------------------------------------------------------
    cur = build_arcsine_sequence(n)
    cur = np.clip(cur, 0.0, 1000.0)

    cur_score = float(evaluate_sequence(cur.tolist()))
    best = cur.copy()
    best_score = cur_score

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Simulated annealing loop.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # keep a second margin
    it = 0

    # Temperature schedule (linear cooling)
    T_start = 0.03          # relatively high to escape the flat plateau
    T_end = 1e-5

    while time.monotonic() < deadline:
        it += 1
        # linearly interpolate temperature
        elapsed = (time.monotonic() - start_time) / budget_s
        T = T_start * (1.0 - elapsed) + T_end * elapsed

        # pick a move type
        r = rng.random()
        if r < 0.45:
            cand = transfer_move(cur, rng)
        elif r < 0.80:
            cand = block_perturbation(cur, rng)
        else:
            cand = rebalance_move(cur, rng)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
            continue

        # acceptance rule (Metropolis)
        if cand_score < cur_score:
            accept = True
        else:
            prob = exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                print(f"[{it:>6}] new best = {best_score:.6f}")

        # occasional resets to keep diversity
        if it % 30000 == 0:
            cur = best.copy()
            cur_score = best_score

        # early exit if we already beat the target comfortably
        if best_score < 1.50:
            break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return a plain Python list as required by the harness.
    return [float(x) for x in best]
