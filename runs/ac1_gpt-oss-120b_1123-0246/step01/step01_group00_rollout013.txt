def gradient_step(a, target_sum, step_scale):

    b = np.convolve(a, a) # compute conv
    k_max = int(np.argmax(b))
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, k_max - (n-1))
    hi = min(k_max, n-1)
    if lo <= hi:
        i_idx = np.arange(lo, hi+1, dtype=np.int64)
        j_idx = k_max - i_idx
        grad[i_idx] = a[j_idx]
        # double for diagonal entries (i=j)
        mask = (i_idx == j_idx)
        if mask.any():
            grad[i_idx[mask]] += a[i_idx[mask]]
    # Project onto sum-zero hyperplane
    grad -= grad.mean()

    # Determine step magnitude: step_scale * target_sum [like earlier init_eta]
    eta = step_scale * target_sum

    cand = a - eta * grad
    cand = np.clip(cand, 0.0, 1000.0)
    # renorm to target sum
    s = cand.sum()
    if s > 0:
        cand *= target_sum / s
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

def random_perturb(a, target_sum, rng, magnitude=0.05):
    # Choose between additive transfer or multiplicative scaling
    if rng.random() < 0.5:
        # transfer
        i, j = rng.integers(0, a.size, size=2)
        while i == j:
            j = rng.integers(0, a.size)
        # Choose amount to move: fraction of a[i]
        amount = rng.random() * magnitude * a[i]
        cand = a.copy()
        cand[i] -= amount
        cand[j] += amount
        # ensure non-negative (if amount > a[i], but we bounded by a[i]*mag)
    else:
        # multiplicative noise: each element multiplied by exp(N(0, sigma))
        sigma = magnitude * 0.5
        factor = np.exp(rng.normal(0.0, sigma, size=a.shape))
        cand = a * factor
    cand = np.clip(cand, 0.0, 1000.0)
    s = cand.sum()
    if s > 0:
        cand *= target_sum / s
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

def conv_max_and_index(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    # Use rfft for real input
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    conv = conv[:2*n - 1]  # linear conv part
    # rounding errors may cause tiny negative values for near-zero; set clip
    # we consider max
    max_val = conv.max()
    max_idx = int(conv.argmax())
    return max_val, max_idx

def gradient_of_max(a, k_max):
    n = len(a)
    grad = np.zeros_like(a)
    lo = max(0, k_max - (n-1))
    hi = min(k_max, n-1)
    if lo <= hi:
        idx = np.arange(lo, hi+1)
        jdx = k_max - idx
        grad[idx] = a[jdx]
        # Add second term for diagonal
        mask = (idx == jdx)
        if mask.any():
            grad[idx[mask]] += a[idx[mask]]
    # Project to zero sum
    grad -= grad.mean()
    return grad

def perturb_sequence(a, target_sum, rng, sigma_rel=0.05):
    # choose random transfer fraction:
    if rng.random() < 0.5:
        i, j = rng.integers(0, a.size, size=2)
        while i == j:
            j = rng.integers(0, a.size)
        # amount = fraction of a[i]
        max_transfer = a[i]
        amount = rng.random() * sigma_rel * max_transfer
        cand = a.copy()
        cand[i] -= amount
        cand[j] += amount
    else:
        # multiplicative perturb
        sigma = sigma_rel # standard deviation on log scale
        factor = np.exp(rng.normal(0.0, sigma, size=a.shape))
        cand = a * factor
    # clip and renorm
    cand = np.clip(cand, 0.0, 1000.0)
    total = cand.sum()
    if total > 0:
        cand *= target_sum / total
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

"""
Hybrid Simulated‑Annealing + Projected Gradient Search for the step‑function
minimisation problem.

The algorithm works with a fixed length `n` (taken from the global
`height_sequence_1` if it exists, otherwise a default size).  The sum of the
coefficients is normalised to √(2 n) – this makes the evaluation score equal
to the maximal entry of the (linear) convolution of the sequence with itself.
A fast FFT based convolution is used to obtain the maximum and its lag.

Search strategy
----------------
* Initialise several simple families (uniform, linear increase, exponential)
  and keep the best of them.
* Maintain a current point `cur` and a global best `best`.
* At each iteration either
    – take a **projected gradient step** that reduces the dominant convolution
      entry (the lag `k_max` where the max occurs), or
    – perform a **random perturbation** (mass transfer between two bins or a
      multiplicative log‑normal noise).
* Acceptance follows a Simulated Annealing rule:
    better solutions are always accepted, worse ones with prob exp(−Δ/T).
  The temperature `T` and the step size decay linearly with the elapsed time.
* Every few hundred iterations a stronger random “kick’’ is applied to avoid
  stagnation.
* The best sequence seen so far is returned (as a plain list of floats).

All operations respect the problem constraints (non‑negative, ≤ 1000, fixed
sum).  The use of FFT makes each convolution O(n log n) which allows thousands
of iterations within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (no closures, no lambdas)
# ----------------------------------------------------------------------


def _fft_max_convolution(a: np.ndarray) -> tuple[float, int]:
    """
    Compute the maximal value of the linear convolution a*a using FFT
    and return (max_value, index_of_max).
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values – they are harmless.
    max_idx = int(conv.argmax())
    return float(conv[max_idx]), max_idx


def _gradient_of_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry w.r.t. the coefficients.
    The gradient is projected onto the hyper‑plane of zero sum (the total mass
    must stay constant).
    """
    n = a.size
    grad = np.zeros_like(a)

    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    if lo <= hi:
        i = np.arange(lo, hi + 1, dtype=np.int64)
        j = k_max - i
        grad[i] = a[j]                     # ∂b_k/∂a_i = a_{k-i}
        # diagonal terms (i == j) contribute twice (a_i^2 → 2 a_i)
        diag = i == j
        if np.any(diag):
            grad[i[diag]] += a[i[diag]]
    # Project to the zero‑sum subspace (preserve total mass)
    grad -= grad.mean()
    return grad


def _project_and_renorm(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Clip to the allowed interval, renormalise the vector so that its sum equals
    `target_sum`, and clip again (to protect against rounding errors).
    """
    v = np.clip(v, 0.0, 1000.0)
    s = v.sum()
    if s > 0.0:
        v *= target_sum / s
    v = np.clip(v, 0.0, 1000.0)
    return v


def _gradient_step(a: np.ndarray,
                   target_sum: float,
                   step_scale: float,
                   k_max: int) -> np.ndarray:
    """One projected gradient descent step aimed at lowering the dominant lag."""
    grad = _gradient_of_max(a, k_max)
    eta = step_scale * target_sum               # absolute step length
    cand = a - eta * grad
    return _project_and_renorm(cand, target_sum)


def _random_perturb(a: np.ndarray,
                    target_sum: float,
                    rng: np.random.Generator,
                    sigma_rel: float = 0.05) -> np.ndarray:
    """
    Randomly modify the sequence.
    With probability 0.5 perform a small transfer of mass between two bins;
    otherwise apply a multiplicative log‑normal noise.
    """
    n = a.size
    if rng.random() < 0.5:
        # transfer a fraction of mass from i to j
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        amount = rng.random() * sigma_rel * a[i]   # never move more than we have
        cand = a.copy()
        cand[i] -= amount
        cand[j] += amount
    else:
        # multiplicative perturbation
        sigma = sigma_rel
        factor = np.exp(rng.normal(0.0, sigma, size=n))
        cand = a * factor
    return _project_and_renorm(cand, target_sum)


def _initial_sequence(n: int,
                     mode: str,
                     rng: np.random.Generator,
                     target_sum: float) -> np.ndarray:
    """
    Produce a simple initial guess.
    mode ∈ {"uniform", "linear", "exp"}.
    """
    if mode == "uniform":
        a = np.ones(n, dtype=np.float64)
    elif mode == "linear":
        a = np.linspace(1.0, n, n, dtype=np.float64)
    elif mode == "exp":
        # exponential decay, parameter tuned so that the sum is not dominated by
        # the first few entries.
        r = 0.9995                     # very mild decay
        a = np.power(r, np.arange(n, dtype=np.float64))
    else:
        raise ValueError(f"unknown mode {mode}")
    return _project_and_renorm(a, target_sum)


def _accept(old: float, new: float, temperature: float,
            rng: np.random.Generator) -> bool:
    """Metropolis acceptance rule."""
    if new < old:
        return True
    if temperature <= 0.0:
        return False
    prob = np.exp(-(new - old) / temperature)
    return rng.random() < prob


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 900,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) with a low evaluation
    score.  The routine runs for at most ``budget_s`` seconds and returns the
    best sequence found (as a plain ``list`` of floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine the length n and the normalisation target sum.
    # ------------------------------------------------------------------
    try:
        # use the historic best if it is available
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = base.size
    except NameError:
        # fallback: a relatively large default length
        n = 5600
        base = None

    target_sum = np.sqrt(2.0 * n)          # the usual normalisation

    # ------------------------------------------------------------------
    # 2. Build a few starter vectors and keep the best one.
    # ------------------------------------------------------------------
    starter_modes = ["uniform", "linear", "exp"]
    best_seq = None
    best_score = np.inf

    for mode in starter_modes:
        seq = _initial_sequence(n, mode, rng, target_sum)
        # Exact evaluation – cheap for a few candidates
        score = float(evaluate_sequence(seq.tolist()))
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    # If a historic vector exists, also try it (after normalising)
    if base is not None:
        base = np.clip(base, 0.0, 1000.0)
        base = _project_and_renorm(base, target_sum)
        score = float(evaluate_sequence(base.tolist()))
        if score < best_score:
            best_score = score
            best_seq = base.copy()

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing / gradient descent loop.
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_max, cur_k = _fft_max_convolution(cur_seq)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # small safety margin
    total_budget = deadline - start_time

    # algorithmic hyper‑parameters
    prob_gradient = 0.6                # likelihood of a gradient step
    init_step_scale = 0.05             # relative step size (times target_sum)
    init_temperature = max(0.5, best_score * 0.3)  # start temperature
    iteration = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_budget

        # linear decay of step size and temperature
        step_scale = init_step_scale * max(0.0, 1.0 - frac)
        temperature = init_temperature * max(0.0, 1.0 - frac)

        # --------------------------------------------------------------
        # Propose a new candidate
        # --------------------------------------------------------------
        if rng.random() < prob_gradient:
            # gradient descent step aimed at the current max lag
            cand = _gradient_step(cur_seq, target_sum, step_scale, cur_k)
        else:
            # random perturbation
            cand = _random_perturb(cur_seq, target_sum, rng,
                                   sigma_rel=0.07 if iteration % 400 == 0 else 0.05)

        # evaluate (fast) using the FFT based convolution
        cand_max, cand_k = _fft_max_convolution(cand)

        # --------------------------------------------------------------
        # Acceptance test (simulated annealing)
        # --------------------------------------------------------------
        if _accept(cur_max, cand_max, temperature, rng):
            cur_seq = cand
            cur_max = cand_max
            cur_k = cand_k

        # --------------------------------------------------------------
        # Update the global best if we have improved (full evaluation)
        # --------------------------------------------------------------
        if cand_max + 1e-12 < best_score:          # tiny tolerance to avoid noise
            # verify with the reference evaluator (expensive but rare)
            true_score = float(evaluate_sequence(cand.tolist()))
            if true_score < best_score:
                best_score = true_score
                best_seq = cand.copy()
                print(f"[{iteration:6d}] new global best  score = {best_score:.6f}")

        # occasional heavy random kick to escape deep plateaus
        if iteration % 1000 == 0 and iteration > 0:
            kick = np.exp(rng.normal(0.0, 0.25, size=n))
            cand_kick = _project_and_renorm(cur_seq * kick, target_sum)
            cand_kick_max, _ = _fft_max_convolution(cand_kick)
            if cand_kick_max < best_score:
                true_score = float(evaluate_sequence(cand_kick.tolist()))
                if true_score < best_score:
                    best_score = true_score
                    best_seq = cand_kick.copy()
                    cur_seq = cand_kick.copy()
                    cur_max = cand_kick_max
                    print(f"[{iteration:6d}] kick improved best score = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Final polishing: a few deterministic gradient steps from the best.
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    for _ in range(20):
        max_val, k_max = _fft_max_convolution(cur)
        cur = _gradient_step(cur, target_sum, init_step_scale * 0.3, k_max)
    final_score = float(evaluate_sequence(cur.tolist()))
    if final_score < best_score:
        best_seq = cur
        best_score = final_score

    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}, n = {n}")
    # Return a plain Python list as required by the grader
    return [float(v) for v in best_seq]
