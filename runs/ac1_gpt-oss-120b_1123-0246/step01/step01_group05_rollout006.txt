Implementation of projection: use Duchi algorithm:

def project_onto_simplex(v, s):
    if s <= 0:
        return np.zeros_like(v)
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - css / ind > 0
    rho = np.where(cond)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, s):
    if s <= 0:
        return np.zeros_like(v)
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    # find rho where u_i > css_i / i
    cond = u > css / ind
    rho = np.where(cond)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function with a very small self‑convolution maximum.

The algorithm consists of three stages:

1. **Fine search over pure power‑law exponents**  
   For a fixed length `n` (taken from the supplied global
   `height_sequence_1` or bounded to 5000) we generate
   `a_i ∝ (i+1)^-α`, normalise the sum to the usual total mass
   `100.0`, and evaluate the maximum entry of `a*a` using an
   FFT based convolution (≈O(n log n), ~10 µs for n≈5000).  
   The best few candidates are stored.

2. **Random convex combinations of the best power‑laws**  
   By linearly mixing two or three of the top candidates we obtain
   richer shapes without leaving the simplex, keeping the total
   mass unchanged.  Each mixed candidate is evaluated and kept if it
   improves the best score.

3. **Targeted local mass‑transfer guided by the current worst
   convolution index**  
   While time permits we repeatedly:
   * compute the full convolution,
   * locate the index `k_max` where the maximum occurs,
   * move a small fraction of mass from a position that participates
     heavily in that entry to a random low‑weight position,
   * re‑normalise to the original total mass,
   * accept the move if it strictly lowers the maximum.
   This is a cheap greedy descent that quickly removes the few
   dominant contributions.

All steps respect the evaluator’s constraints (non‑negative,
finite numbers, total‑mass invariance).  The routine stops a few
seconds before the time limit and returns the best sequence found.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions (no closures, top‑level as required)
# ----------------------------------------------------------------------


def power_law_sequence(length: int, exponent: float) -> np.ndarray:
    """Return a power‑law vector a_i ∝ (i+1)^{-exponent},
    normalised to sum = 100 (the evaluator’s common scale)."""
    idx = np.arange(length, dtype=np.float64) + 1.0
    w = np.power(idx, -exponent)
    w /= w.sum()
    return w * 100.0


def fast_max_convolution(seq: np.ndarray, L: int) -> float:
    """Maximum entry of the full convolution of `seq` with itself,
    computed via FFT (real‑to‑real) for speed."""
    # rfft length L (>= 2*len(seq))
    F = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(F * F, n=L)          # length L (real)
    # The genuine linear convolution occupies the first 2*n-1 entries.
    return float(np.max(conv[: 2 * len(seq) - 1]))


def proj_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of vector `v` onto the simplex {x≥0, sum x = s}.
    Implements the algorithm of Duchi et al. (2008)."""
    if s <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    css = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u > css / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 900,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The function returns a list of floats that
    respects all security checks of the evaluator.
    """
    rng = np.random.default_rng(seed)
    # --------------------------------------------------------------
    # Length & total mass (the evaluator is scale‑invariant, we keep 100)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)                 # global, if available
        n = max(256, min(n, 5000))
    except Exception:
        n = 5000

    total_mass = 100.0

    # FFT length (next power of two >= 2*n)
    L = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # 1️⃣ Fine grid search over pure power‑law exponents
    # --------------------------------------------------------------
    best_score = float("inf")
    best_seq = None
    top_candidates = []          # store (score, seq) tuples
    # a relatively fine grid – the extra work is cheap thanks to FFT
    exponents = np.linspace(0.30, 1.10, 81)   # step ≈0.01
    for alpha in exponents:
        seq = power_law_sequence(n, float(alpha))
        score = fast_max_convolution(seq, L)   # raw max of self‑convolution
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[grid] α={alpha:.3f}  score={best_score:.6f}")
        # keep the 15 best for later mixing
        top_candidates.append((score, seq.copy()))
    # keep the 15 smallest scores
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:15]

    # --------------------------------------------------------------
    # 2️⃣ Random convex combinations of the best power‑laws
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)
    while time.monotonic() < deadline:
        # pick 2 or 3 distinct candidates from the pool
        k = rng.choice([2, 3])
        idxs = rng.choice(len(top_candidates), size=k, replace=False)
        weights = rng.random(k)
        weights /= weights.sum()                     # convex coefficients
        mixed = np.zeros(n, dtype=np.float64)
        for w, cand_idx in zip(weights, idxs):
            mixed += w * top_candidates[cand_idx][1]
        # total mass stays 100 because each component already sums to 100
        score = fast_max_convolution(mixed, L)
        if score < best_score:
            best_score = score
            best_seq = mixed.copy()
            print(f"[mix] k={k}  score={best_score:.6f}")
        # occasional break if we are very close to the target
        if best_score < 1.51:
            break

    # --------------------------------------------------------------
    # 3️⃣ Greedy local improvement that attacks the current worst sum
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (never used in practice)
        best_seq = np.full(n, total_mass / n, dtype=np.float64)
        best_score = fast_max_convolution(best_seq, L)

    seq = best_seq.copy()
    it = 0
    # parameters for the local moves
    max_iter = 200_000   # safety cap; will usually stop earlier by time
    while it < max_iter and time.monotonic() < deadline:
        # full convolution (FFT) – also gives the index of the maximal entry
        F = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(F * F, n=L)[: 2 * n - 1]
        cur_max = float(conv.max())
        if cur_max < best_score:
            best_score = cur_max
            best_seq = seq.copy()
            print(f"[local] iter={it}  score={best_score:.6f}")

        k_max = int(conv.argmax())

        # Identify a pair (i, j) with i+j = k_max that contributes most.
        # The contribution is a_i * a_j.
        start = max(0, k_max - (n - 1))
        end = min(n - 1, k_max)
        i_range = np.arange(start, end + 1, dtype=np.int64)
        contrib = seq[i_range] * seq[k_max - i_range]
        max_pair_idx = int(i_range[np.argmax(contrib)])
        j_pair_idx = k_max - max_pair_idx

        # Move a small random fraction of the larger of the two entries
        # to a random low‑weight position.
        if seq[max_pair_idx] > seq[j_pair_idx]:
            src = max_pair_idx
        else:
            src = j_pair_idx
        dst = rng.integers(0, n)
        if src != dst and seq[src] > 1e-12:
            delta = seq[src] * rng.uniform(0.01, 0.15)   # 1‑15 % of the source mass
            cand = seq.copy()
            cand[src] -= delta
            cand[dst] += delta
            # re‑normalise to the original total mass (projection onto simplex)
            cand = proj_to_simplex(cand, total_mass)
            # cheap evaluation – same FFT routine as above
            cand_score = fast_max_convolution(cand, L)
            if cand_score < cur_max - 1e-7:   # accept only genuine improvement
                seq = cand
        it += 1

    # --------------------------------------------------------------
    # Final sanity check using the official evaluator (slow but safe)
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"final best score (official) = {final_score:.6f}   length = {len(best_seq)}")
    return [float(x) for x in best_seq.tolist()]
