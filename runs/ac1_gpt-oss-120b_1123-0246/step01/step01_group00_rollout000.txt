"""
Simulated‑annealing / biased‑transfer optimiser.

The routine starts from the best known construction (the global variable
`height_sequence_1` if it exists, otherwise a uniform vector) and keeps the
total mass fixed to √(2 n).  The search repeatedly moves a small amount of mass
from a “central’’ region of the step‑function to the periphery (or performs a
completely random transfer).  Moves are accepted with a standard SA
Metropolis rule, so the algorithm can climb out of local minima.  Every few
thousand iterations a stronger multiplicative perturbation and a lightweight
gradient‑step (using the top‑few convolution lags) are performed, which helps
to accelerate convergence.  The procedure stops after the allotted time and
returns the best sequence seen.

The method is deliberately different from the pure gradient descent that
was used before, while still exploiting the structure of the worst‑lag
convolution.
"""

import time
import numpy as np

def _initial_sequence(seed: int, n: int) -> np.ndarray:
    """
    Produce a feasible starting vector of length ``n``.
    If the global ``height_sequence_1`` exists it is used (clipped and
    rescaled), otherwise a uniform vector of ones is returned.
    The vector is scaled so that its sum equals sqrt(2 n).
    """
    rng = np.random.default_rng(seed)
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise ValueError
    except (NameError, ValueError):
        base = np.ones(n, dtype=np.float64)

    base = np.clip(base, 0.0, 1000.0)
    if base.sum() == 0.0:
        base[:] = 1.0

    target_sum = np.sqrt(2.0 * n)
    base *= (target_sum / base.sum())
    base = np.clip(base, 0.0, 1000.0)
    return base


def _biased_transfer(a: np.ndarray,
                     rng: np.random.Generator,
                     central_idx: np.ndarray,
                     outer_idx: np.ndarray) -> tuple[np.ndarray, bool]:
    """
    Propose a transfer of mass that is biased from the centre of the vector
    towards the periphery.  The function returns the new candidate and a flag
    indicating whether the proposal was actually performed (``False`` when
    no admissible transfer is possible).
    """
    n = a.size
    # Choose a direction: most of the time we move mass outward,
    # otherwise a completely random pair.
    if rng.random() < 0.7:
        i = int(rng.choice(central_idx))   # donor (typically centre)
        j = int(rng.choice(outer_idx))     # receiver (typically edge)
    else:
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

    if a[i] <= 0.0:
        return a, False

    # Respect the hard bounds [0, 1000] while preserving total mass.
    max_delta = min(a[i], 1000.0 - a[j])
    if max_delta <= 0.0:
        return a, False

    # Transfer a random fraction of the admissible amount.
    delta = rng.random() * max_delta
    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta
    return cand, True


def _gradient_step(a: np.ndarray,
                   target_sum: float,
                   rng: np.random.Generator,
                   temperature_frac: float) -> np.ndarray:
    """
    A cheap gradient‑like step that reduces the contribution of the
    largest few convolution lags.  ``temperature_frac`` (∈[0,1]) scales the
    step size – larger when the algorithm is still warm.
    """
    n = a.size
    # Compute the convolution and pick the few worst lags.
    b = np.convolve(a, a)
    # Take the top 5 lags (indices).  ``np.argpartition`` is O(n).
    top_k = np.argpartition(b, -5)[-5:]

    grad = np.zeros_like(a)
    for k_max in top_k:
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        if lo <= hi:
            i_idx = np.arange(lo, hi + 1)
            j_idx = k_max - i_idx
            grad[i_idx] += a[j_idx]

    # Zero‑sum projection (preserve total mass).
    grad -= grad.mean()

    # Step magnitude – modest fraction of a typical entry, scaled by temperature.
    step_size = temperature_frac * 0.02 * target_sum
    cand = a - step_size * grad
    cand = np.clip(cand, 0.0, 1000.0)

    # Renormalise to keep the exact sum.
    s = cand.sum()
    if s > 0.0:
        cand *= (target_sum / s)
    cand = np.clip(cand, 0.0, 1000.0)
    return cand


def propose_candidate(seed: int = 42,
                     budget_s: float = 950.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that minimises
    ``evaluate_sequence``.  Returns the best sequence found within the
    allotted time budget (seconds).

    Parameters
    ----------
    seed: int
        Random seed for reproducibility.
    budget_s: float
        Total time budget in seconds.  A small safety margin is kept
        internally.

    Returns
    -------
    List[float]
        The best height vector discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a problem size and initialise a feasible sequence.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 5000                             # a moderately large size

    a = _initial_sequence(seed, n)

    # ------------------------------------------------------------------
    # 2. Basic bookkeeping.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    current_score = best_score

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Region definitions for biased transfers.
    central_lo = int(0.35 * n)
    central_hi = int(0.65 * n)
    central_idx = np.arange(central_lo, central_hi, dtype=np.int64)
    outer_idx = np.concatenate([np.arange(0, central_lo, dtype=np.int64),
                                np.arange(central_hi, n, dtype=np.int64)])

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # keep 2 s safety margin
    total_budget = deadline - start_time

    # Temperature schedule (simple linear cooling).
    T0 = max(1e-3, best_score * 0.08)   # initial temperature proportional to score

    iteration = 0
    last_improvement_iter = 0

    # ------------------------------------------------------------------
    # 3. Main simulated‑annealing loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        # Linear cooling: T decays from T0 to ~0.
        T = T0 * max(0.0, 1.0 - elapsed / total_budget)
        T = max(T, 1e-7)

        # ----- propose a biased transfer ---------------------------------
        cand, ok = _biased_transfer(a, rng, central_idx, outer_idx)
        if not ok:
            # fall back to a tiny random jitter to keep the loop alive
            cand = a + rng.normal(0.0, 1e-4, size=n)
            cand = np.clip(cand, 0.0, 1000.0)
            cand_sum = cand.sum()
            if cand_sum > 0.0:
                cand *= (target_sum / cand_sum)

        # ----- evaluate --------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(score):
            # Should not happen because we respect bounds, but skip if it does.
            iteration += 1
            continue

        # ----- Metropolis acceptance --------------------------------------
        if score < current_score or rng.random() < np.exp(-(score - current_score) / (T + 1e-12)):
            a = cand
            current_score = score
            if score < best_score:
                best = cand.copy()
                best_score = score
                last_improvement_iter = iteration
                print(f"[{iteration:6d}] new best  score = {best_score:.6f}")

        # ----- occasional strong perturbations ----------------------------
        if iteration % 2000 == 0 and iteration > 0:
            # Multiplicative noise (larger than in the previous code).
            pert = np.exp(rng.normal(0.0, 0.12, size=n))
            cand2 = a * pert
            cand2 = np.clip(cand2, 0.0, 1000.0)
            s2 = cand2.sum()
            if s2 > 0.0:
                cand2 *= (target_sum / s2)
            score2 = float(evaluate_sequence(cand2.tolist()))
            if score2 < best_score:
                best = cand2.copy()
                best_score = score2
                a = cand2
                current_score = score2
                last_improvement_iter = iteration
                print(f"[{iteration:6d}] strong perturb improved score = {best_score:.6f}")

        # ----- cheap gradient‑like refinement (every few thousand iters) ---
        if iteration % 5000 == 0 and iteration > 0:
            # Temperature fraction for scaling the gradient step.
            temp_frac = max(0.0, 1.0 - elapsed / total_budget)
            cand3 = _gradient_step(a, target_sum, rng, temp_frac)
            score3 = float(evaluate_sequence(cand3.tolist()))
            if score3 < current_score:
                a = cand3
                current_score = score3
                if score3 < best_score:
                    best = cand3.copy()
                    best_score = score3
                    last_improvement_iter = iteration
                    print(f"[{iteration:6d}] gradient refinement new best = {best_score:.6f}")

        # ----- stagnation handling ---------------------------------------
        # If we have not improved for a long while, inject a random jitter.
        if iteration - last_improvement_iter > 15000:
            jitter = rng.normal(0.0, 0.02, size=n)
            cand4 = a + jitter
            cand4 = np.clip(cand4, 0.0, 1000.0)
            s4 = cand4.sum()
            if s4 > 0.0:
                cand4 *= (target_sum / s4)
            score4 = float(evaluate_sequence(cand4.tolist()))
            if score4 < best_score:
                best = cand4.copy()
                best_score = score4
                a = cand4
                current_score = score4
                last_improvement_iter = iteration
                print(f"[{iteration:6d}] jitter recovered improvement = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Final brief gradient polishing (remaining time permitting).
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 5.0:  # a few seconds left → quick polish
        polish_steps = 2000
        a = best.copy()
        for _ in range(polish_steps):
            # same gradient step but with a fixed small step size
            cand = _gradient_step(a, target_sum, rng, temperature_frac=0.05)
            score = float(evaluate_sequence(cand.tolist()))
            if score < best_score:
                best = cand.copy()
                best_score = score
                a = cand
        print(f"[polish] final best score after polishing = {best_score:.6f}")

    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    return [float(v) for v in best]
