def propose_candidate(seed=0, budget_s=900):
    """Find a step-function minimizing the given metric.
    Starts from the binomial sequence, then iterates a linear-programming
    update: for the current sequence f, solve max sum(g) s.t. f*g <= M,
    mix the solution into f, and repeat.  Falls back to random block
    perturbations if the LP solver fails or time runs low.
    """
    import time, numpy as np
    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)
    # Determine length n
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.size > 0:
            n = base.size
        else:
            n = 300
    else:
        n = 300

    # Build binomial start
    def binomial_seq(length):
        a = np.empty(length, dtype=np.float64)
        a[0] = 1.0
        for i in range(length-1):
            a[i+1] = a[i] * (2*i + 1) / (2.0 * (i+1))
        return a

    cur = binomial_seq(n)
    best = cur.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n} start score={best_score:.6f}")

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    iter_num = 0
    t = 0.3  # mixing coefficient
    # main loop
    while time.monotonic() < deadline:
        iter_num += 1

        # compute current max convolution
        conv_cur = np.convolve(cur, cur)
        M = conv_cur.max()
        # Build LP matrix
        # A shape (2n-1, n)
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for j in range(n):
            A[j:j+n, j] = cur
        # solve LP: maximize sum(g) -> minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(rows, M, dtype=np.float64)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        except Exception as e:
            print(f"[iter {iter_num}] LP failed: {e}")
            # fallback to random perturbation
            candidate = random_perturb(cur, rng)
            score = float(evaluate_sequence(candidate.tolist()))
            if score < best_score:
                best = candidate
                best_score = score
                cur = candidate
                print(f"[{iter_num}] improved by random to {best_score:.6f}")
            continue

        if not res.success:
            # fallback
            print(f"[iter {iter_num}] LP not successful")
            # just break out to avoid infinite loop
            break

        g = res.x
        # Mix current and g
        # Optionally scale g to have same ℓ1 norm as cur (optional)
        # Here we just mix
        new = (1.0 - t) * cur + t * g
        # Clip to allowed range (evaluate_sequence also clips)
        new = np.clip(new, 0.0, 1000.0)

        score = float(evaluate_sequence(new.tolist()))
        if score < best_score:
            best = new
            best_score = score
            cur = new
            print(f"[{iter_num}] LP mix improved: score={best_score:.6f}")
            # adjust mixing coefficient adaptively
            t = min(0.5, t * 1.05)
        else:
            # if no improvement, try smaller t or random perturbation
            t = max(0.05, t * 0.7)
            # occasional random perturb
            if rng.random() < 0.2:
                candidate = random_perturb(cur, rng)
                score2 = float(evaluate_sequence(candidate.tolist()))
                if score2 < best_score:

                    best_score = score2
                    cur = candidate
                    print(f"[{iter_num}] random perturb improved to {best_score:.6f}")

    print(f"[done] iterations={iter_num}, best score={best_score:.6f}")
    return [float(x) for x in best]

def random_perturb(seq, rng):
    candidate = seq.copy()
    n = candidate.size
    # choose a region (block) width 1% of n
    w = max(1, n // 100)
    start = rng.integers(0, n - w + 1)
    factor = np.exp(rng.normal(0.0, 0.1))
    candidate[start:start+w] *= factor
    # optionally symmetrize
    if rng.random() < 0.3:
        candidate = 0.5 * (candidate + candidate[::-1])
    candidate = np.clip(candidate, 0.0, 1000.0)
    return candidate

"""
Propose a step‑function sequence that improves on the binomial baseline.

The algorithm works in two phases:

1. **Linear‑programming update** –  
   For the current sequence `f` we compute `M = max(f*f)`.  
   We then solve the LP  

        maximise   Σ g_i
        subject to f * g ≤ M      (convolution constraints, linear in g)
                   g_i ≥ 0

   The solution `g` has a larger ℓ¹‑norm while still respecting the same
   convolution bound.  By mixing the old sequence with `g` we obtain a new
   sequence with a smaller peak `max(f*f)` and therefore a smaller
   evaluation score.  The LP is built with ``scipy.optimize.linprog`` using
   the modern HiGHS solver.

2. **Exploratory perturbations** –  
   If the LP does not improve the score (or if the remaining time is too
   short for another LP) we fall back to cheap stochastic block
   perturbations (multiplicative log‑normal noise on a random block and an
   optional symmetrisation).  This keeps the search alive and can escape
   tiny local minima.

The loop alternates between these two moves until the time budget expires.
The best sequence seen is returned.  The code respects all constraints of
`evaluate_sequence` (non‑negative values, clipping to [0,1000]) and never
uses closures or lambda expressions.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _binomial_sequence(length: int) -> np.ndarray:
    """Truncated coefficients of (1‑z)^‑1/2 : a_i = C(2i,i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(length - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    return a.clip(0.0, 1000.0)


def _random_block_perturbation(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) stochastic perturbation of ``seq``."""
    cand = seq.copy()
    n = cand.size
    # block width: roughly 1‑2 % of the length
    width = max(1, n // 150)
    start = rng.integers(0, n - width + 1)
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation – many good solutions are (almost) symmetric
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return cand.clip(0.0, 1000.0)


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1) × n) such that ``A @ g`` equals the
    convolution ``f * g`` (valid for non‑negative vectors).
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j contains f shifted by j positions
    for j in range(n):
        A[j:j + n, j] = f
    return A


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 900.0, **kwargs) -> list[float]:
    """
    Find a step‑function with a low value of `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Wall‑clock budget in seconds (the function must return before this
        time runs out).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Decide on the length of the candidate.
    #    If a previous construction exists we reuse its length, but never
    #    exceed a size that would make the LP prohibitively large.
    # ------------------------------------------------------------------
    max_len_for_lp = 800        # safe upper bound for the LP matrix
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(min(max_len_for_lp, base.size)) if base.size > 0 else 300
    else:
        n = 300

    # ------------------------------------------------------------------
    # 2. Initialise with the analytically optimal binomial sequence.
    # ------------------------------------------------------------------
    cur = _binomial_sequence(n)
    best = cur.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop – LP updates + cheap perturbations.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    iteration = 0
    # mixing coefficient (how much of the LP solution we incorporate)
    mix_coef = 0.35

    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        #   a) Linear‑programming update (the core of the algorithm)
        # --------------------------------------------------------------
        conv_cur = np.convolve(cur, cur)
        M = float(conv_cur.max())          # current convolution peak

        # Build matrix for the constraints f * g ≤ M
        A = _build_conv_matrix(cur)
        b_ub = np.full(A.shape[0], M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)   # maximise Σ g_i  →  minimise -Σ g_i
        # solve the LP; the HiGHS backend works well for moderate dimensions
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': 5.0})
        except Exception as exc:            # unexpected solver failure
            print(f"[{iteration}] LP exception: {exc}")
            res = None

        # --------------------------------------------------------------
        #   b) If the LP succeeded, create a new candidate by mixing.
        # --------------------------------------------------------------
        if res is not None and res.success:
            g = res.x
            # Mix the current sequence with the LP solution.
            cand = (1.0 - mix_coef) * cur + mix_coef * g
            cand = np.clip(cand, 0.0, 1000.0)

            cand_score = float(evaluate_sequence(cand.tolist()))
            if cand_score < best_score - 1e-9:          # strict improvement
                best = cand
                cur = cand
                best_score = cand_score
                # slowly increase mixing strength when we are making progress
                mix_coef = min(0.5, mix_coef * 1.07)
                print(f"[{iteration:6d}] LP‑mix improved → {best_score:.6f}")
                continue
            else:
                # no improvement – become more conservative
                mix_coef = max(0.05, mix_coef * 0.85)

        # --------------------------------------------------------------
        #   c) Fall back to a cheap stochastic perturbation.
        # --------------------------------------------------------------
        cand = _random_block_perturbation(cur, rng)
        cand_score = float(evaluate_sequence(cand.tolist()))
        if cand_score < best_score - 1e-9:
            best = cand
            cur = cand
            best_score = cand_score
            print(f"[{iteration:6d}] random‑perturb improved → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. End of search – report the best found sequence.
    # ------------------------------------------------------------------
    print(f"[done] iterations={iteration}, final best score={best_score:.6f}")
    # Convert to plain Python list of floats as required.
    return [float(x) for x in best]
