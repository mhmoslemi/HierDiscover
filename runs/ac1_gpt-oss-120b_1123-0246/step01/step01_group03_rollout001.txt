```python
import time
import math
import numpy as np
from scipy.optimize import linprog
import scipy.sparse

def catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan-type coefficients C_i = binom(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    # Clip to evaluator bounds
    np.clip(a, 0.0, 1000.0, out=a)
    return a

def build_conv_matrix(a: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build a sparse matrix A such that A @ g == np.convolve(a, g)
    (where the convolution has length 2*len(a)-1).
    """
    n = a.shape[0]
    # column indices repeat each column i for n rows
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    # row indices: for each column i, rows i, i+1, ..., i+n-1
    row_idx = np.arange(n, dtype=np.int32).repeat(n) + np.tile(np.arange(n, dtype=np.int32), n)
    # data = a repeated for each column i
    data = np.tile(a, n)
    # shape (2n-1, n)
    A = scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2*n - 1, n))
    return A

def linear_program_step(a: np.ndarray, M: float, max_time: float) -> np.ndarray:
    """
    Solve   max Σ g_i   s.t.   conv(a, g) ≤ M,  g_i ≥ 0.
    Returns the optimal g (numpy array) or None on failure.
    """
    n = a.shape[0]
    # Build constraint matrix once (its entries depend on a)
    A = build_conv_matrix(a)
    b = np.full(2*n - 1, M, dtype=np.float64)
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)
    # linprog expects dense b and A_ub; we give sparse A
    # Set a reasonable time limit for the LP: at most remaining budget
    # SciPy's HiGHS backend respects a "time_limit" option in seconds.
    options = {"presolve": True}
    # If max_time specified, pass it (may be ignored on older versions)
    if max_time is not None:
        options["time_limit"] = max_time
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method="highs", options=options)
    if res.success and res.x is not None:
        g = np.maximum(0.0, res.x.astype(np.float64))
        return g
    return None
```

The goal is to find a non‑negative step‑function (a list of heights) that minimises  

\[
\text{score}= \frac{2\,n\;\max\limits_k\;(a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

where `(a*a)` is the discrete self‑convolution.  
For the Catalan‑type sequence the score is ≈ π/2 ≈ 1.5708.  
To beat this we repeatedly:

1. **Compute the worst convolution value** `M = max(a*a)`.
2. **Solve a linear program**  

   \[
   \begin{aligned}
   \max\;&\sum_i g_i \\
   \text{s.t.}&\;(a*g)_k \le M \quad\forall k,\\
               &\;g_i\ge 0 .
   \end{aligned}
   \]

   The constraints are linear in the unknown vector `g`.  
   `scipy.optimize.linprog` (HiGHS) can solve this LP efficiently with a
   sparse Toeplitz matrix that encodes the convolution `a*g`.
3. **Mix the LP solution into the current sequence**  

   \[
   a \leftarrow (1-t)\,a + t\,g,
   \]

   trying a range of step sizes `t`.  The mixing never increases the
   convolution bound because `a*g ≤ M`; a small `t` also keeps the omitted
   quadratic term `t² g*g` harmless.
4. **Fallback random perturbation** – if the LP does not improve the score,
   multiply the current sequence by a log‑normal factor and keep the change
   when it yields a better score.
5. **Iterate until the time budget expires**.

The procedure starts from the provided global `height_sequence_1` (truncated
to a tractable length) or, if that is missing, from the Catalan sequence.
All numbers are clipped to the range `[0, 1000]` as required by
`evaluate_sequence`.

The whole search runs inside a 1000 s wall‑clock budget (the function receives
the remaining budget in `budget_s`).  The best sequence found is returned as
a plain Python list.

```python
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def build_conv_matrix(a: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build a sparse matrix A such that A @ g == np.convolve(a, g)
    (convolution length = 2*len(a)-1).  The matrix is Toeplitz with the
    vector `a` on each column.
    """
    n = a.shape[0]
    # Column index repeats each column i for the n rows belonging to it
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    # Row index for column i runs from i to i+n-1
    row_idx = np.arange(n, dtype=np.int32).repeat(n) + np.tile(np.arange(n, dtype=np.int32), n)
    # Data are the entries of `a` repeated for every column
    data = np.tile(a, n)
    return scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))


def linear_program_step(a: np.ndarray, M: float, max_time: float) -> np.ndarray | None:
    """
    Solve  max Σ g_i  s.t.  conv(a, g) ≤ M , g ≥ 0.
    Returns the optimal g (numpy array) or None on failure.
    """
    n = a.shape[0]
    A = build_conv_matrix(a)                     # shape (2n-1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)  # right‑hand side
    c = -np.ones(n, dtype=np.float64)            # minimise –sum(g)  → maximise sum(g)
    # HiGHS accepts a sparse A_ub.  The `time_limit` option is honoured
    # by recent SciPy versions; if unsupported it is ignored.
    options = {"presolve": True}
    if max_time is not None:
        options["time_limit"] = max_time
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method="highs",
                  options=options)
    if res.success and res.x is not None:
        return np.maximum(0.0, res.x.astype(np.float64))
    return None


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The algorithm starts from the Catalan sequence (or a provided global
    ``height_sequence_1``), repeatedly solves a linear program that finds a
    direction which can increase the total mass without raising the worst
    self‑convolution, mixes the direction with a small step size and falls
    back to a random log‑normal perturbation when necessary.  The routine
    respects the wall‑clock budget and returns the best sequence seen.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # ------------------------------------------------------------------
    # 1. Choose length and initialise `a`
    # ------------------------------------------------------------------
    try:
        base_global = np.asarray(height_sequence_1, dtype=np.float64)
        max_len = 1024            # safe upper bound for the LP size
        n = min(base_global.size, max_len)
        a = base_global[:n].copy()
        source = "global"
    except Exception:
        n = 1024
        a = catalan_sequence(n)
        source = "catalan"

    np.clip(a, 0.0, 1000.0, out=a)

    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = float("inf")
    best = a.copy()
    print(f"Initial sequence from {source} (n={n}), score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Iterative improvement
    # ------------------------------------------------------------------
    iteration = 0
    step_sizes = (0.4, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005)

    while time.monotonic() < deadline and iteration < 100:
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            break

        # Current self‑convolution and its maximum value
        conv_self = np.convolve(a, a)
        M = float(conv_self.max())

        # Solve the LP (give it at most 10 seconds of the remaining budget)
        lp_time = min(10.0, remaining)
        g = linear_program_step(a, M, max_time=lp_time)

        improved = False
        if g is not None:
            # Try several mixing coefficients
            for t in step_sizes:
                cand = (1.0 - t) * a + t * g
                np.clip(cand, 0.0, 1000.0, out=cand)
                try:
                    cand_score = float(evaluate_sequence(cand.tolist()))
                except Exception:
                    continue
                if np.isfinite(cand_score) and cand_score < best_score:
                    a = cand
                    best = cand.copy()
                    best_score = cand_score
                    improved = True
                    print(f"Iter {iteration:03d}, LP mix t={t:.3f} → score {best_score:.6f}")
                    break
        else:
            print(f"Iter {iteration:03d}, LP failed – trying random tweak.")

        if not improved:
            # Random log‑normal perturbation as a cheap explorer
            perturb = np.exp(rng.normal(0.0, 0.06, size=n))
            cand = a * perturb
            np.clip(cand, 0.0, 1000.0, out=cand)
            try:
                cand_score = float(evaluate_sequence(cand.tolist()))
            except Exception:
                cand_score = float("inf")
            if np.isfinite(cand_score) and cand_score < best_score:
                a = cand
                best = cand.copy()
                best_score = cand_score
                print(f"Iter {iteration:03d}, random tweak → score {best_score:.6f}")

        iteration += 1

    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return [float(v) for v in best]
```

**Explanation of the key ideas**

* **Linear‑programming direction** – The LP searches for a non‑negative vector
  `g` that can be added to the current sequence without breaking the worst
  convolution bound `M`.  Because the constraints are linear in `g`,
  `scipy.optimize.linprog` (HiGHS) solves them quickly even for a few
  thousand variables.

* **Mixing** – A convex combination `(1‑t)a + t g` keeps the bound `M`
  (the quadratic term `t² g*g` is tiny for the small step sizes we try)
  while strictly increasing the total mass `∑ a_i`.  A larger sum while
  `M` stays the same directly reduces the evaluation score.

* **Random perturbations** – When the LP cannot improve the score (because
  the current shape is already optimal for the given `M`) a small random
  log‑normal scaling explores neighbouring regions of the search space.

* **Tractable problem size** – The LP matrix is built once per iteration as a
  sparse Toeplitz matrix; with `n = 1024` its size (`≈10⁶` non‑zeros) is well
  within memory limits and solves in a fraction of a second.

The algorithm respects the 1000 s time limit, continuously prints progress,
and finally returns the best sequence discovered.
