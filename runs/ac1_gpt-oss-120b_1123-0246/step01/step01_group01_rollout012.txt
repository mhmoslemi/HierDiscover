"""
Optimising the step‑function evaluation metric.

The main idea is to start from the *binomial* sequence
    a_i = C(2i,i) / 4^i
(which gives a flat self‑convolution = 1).  By convolving this
sequence with a long flat kernel (a uniform distribution) we
*smooth* the coefficients: the sum of the coefficients stays the
same, while the peak of the self‑convolution is reduced roughly as
(b / m) where ``b`` is the length of the binomial block and ``m`` is
the length of the uniform kernel.  The evaluation score
    2·n·max(b)/sum(a)^2   ≈   π·b / m
can therefore be pushed arbitrarily low by taking ``m`` a few
times larger than ``b``.  The algorithm samples several
(b,m)‑pairs inside the allotted time, keeps the best construction,
and finally runs a short stochastic hill‑climb (block perturbations
and mirror swaps) to fine‑tune the result.

The function ``propose_candidate`` returns the best list of heights
found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: binomial (1‑z)^‑1/2 coefficients, length = n
# ----------------------------------------------------------------------
def _binomial_sequence(n: int) -> np.ndarray:
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(n - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    # clipping is not really needed – the values stay ≤ 1
    return np.clip(a, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Simple block perturbation used during the final local improvement.
# ----------------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    cand = base.copy()
    n = cand.size
    # block width: 0.5 % – 5 % of the length
    w_min = max(1, n // 200)
    w_max = max(2, n // 20)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.125)))
    # occasional symmetrisation – helps when a symmetric optimum exists
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Mirror‑swap move – transfers a tiny amount of mass between symmetric
# positions (i ↔ n‑1‑i).  Useful for polishing the peak of the convolution.
# ----------------------------------------------------------------------
def _mirror_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = seq.size
    i = rng.integers(0, n // 2)
    j = n - 1 - i
    # move at most 5 % of the smaller side
    delta = min(seq[i], seq[j]) * rng.random() * 0.05
    if rng.random() < 0.5:
        seq[i] += delta
        seq[j] -= delta
    else:
        seq[i] -= delta
        seq[j] += delta
    return np.clip(seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Build a candidate by smoothing a binomial block with a uniform kernel.
# ----------------------------------------------------------------------
def _build_candidate(b_len: int, m_factor: float) -> np.ndarray:
    """
    b_len  – length of the truncated binomial block (>= 200)
    m_factor – multiplier for the length of the uniform kernel (>= 2.0)
    """
    # 1. binomial block
    binom = _binomial_sequence(b_len)

    # 2. uniform kernel – sum = 1 (any scaling does not change the metric)
    m_len = max(1, int(b_len * m_factor))
    uniform = np.full(m_len, 1.0 / m_len, dtype=np.float64)

    # 3. convolution (this is the new step‑function)
    candidate = np.convolve(binom, uniform)
    # the convolution may introduce very small negative floats due to rounding;
    # evaluate_sequence will clip them anyway.
    return candidate


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 900, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    target bound 1.503.  The search uses the binomial‑smoothing construction
    described above, samples several (b_len, m_factor) pairs, keeps the best,
    and finally performs a short stochastic hill‑climb.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0. keep a fallback – the classic binomial sequence (length 256)
    # ------------------------------------------------------------------
    fallback = _binomial_sequence(256)
    best = fallback.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] fallback score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. randomised generation of smoothed binomial candidates
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 30.0)   # 30 s reserved for final polishing
    iter_no = 0
    while time.monotonic() < deadline:
        iter_no += 1

        # choose a reasonable block length (larger blocks give smaller scores)
        b_len = int(rng.integers(800, 3000))                 # 800 … 3000 (tunable)
        # choose a smoothing factor – the larger the factor, the smaller the peak
        m_factor = float(rng.uniform(2.5, 5.0))              # 2.5 … 5.0

        cand = _build_candidate(b_len, m_factor)

        # quick sanity – avoid absurdly long sequences (the runtime of evaluate_sequence grows ~n log n)
        if cand.size > 50000:
            continue

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[{iter_no:5d}] new best from smoothing: b={b_len} m_factor={m_factor:.2f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. short stochastic hill‑climb around the best candidate found so far
    # ------------------------------------------------------------------
    deadline = time.monotonic() + 30.0          # remaining 30 s for polishing
    n = best.size
    while time.monotonic() < deadline:
        move = rng.random()
        if move < 0.6:
            cand = _make_block_perturbation(best, rng)
        elif move < 0.9:
            cand = _mirror_swap(best.copy(), rng)
        else:
            # tiny random jitter of the whole vector (helps escape flat plateaus)
            jitter = np.exp(rng.normal(0.0, 0.03, size=n))
            cand = np.clip(best * jitter, 0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[polish] improved to {best_score:.6f}")

    print(f"[done] iterations={iter_no}, final best score={best_score:.6f}, length={best.size}")
    return [float(x) for x in best]
