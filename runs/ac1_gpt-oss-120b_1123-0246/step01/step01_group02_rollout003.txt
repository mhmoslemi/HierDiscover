"""
search_optimal_step.py

Stochastic “early‑to‑late” mass‑transfer search.

The arcsine sequence (a[i] = binom(2i,i)/4**i) is known to give the
asymptotic bound π/2 ≃ 1.5708.  To beat this we keep the current
maximum of the self‑convolution (M) fixed and try to *increase the total
mass* Σa_i while preserving the bound M.  Any move that reduces M can be
followed by a global scaling ‑ √(Mold/Mnew) – which restores the bound
but raises the sum, thus lowering the evaluation score
   score = 2·n·M / (Σa)² .
The search concentrates on moving mass from the early (high‑impact)
indices to the far tail where it contributes much less to the convolution.
Fast convolution is performed with an FFT (O(n log n)), allowing millions
of candidates within the time budget.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# 1. Helper: build the analytical “arcsine’’ sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative arcsine sequence a[i] = C(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i))/(4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# 2. Fast convolution → maximal value.
# ----------------------------------------------------------------------
def conv_max(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)[k] using an FFT (full length 2·n‑1)."""
    n = seq.size
    m = 2 * n - 1
    # next power‑of‑two ≥ m for a clean circular convolution
    L = 1 << ((m - 1).bit_length())
    # real FFT (rfft) is a little faster than full complex FFT
    fft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[:m]
    # Numerical noise may create tiny negative values – guard against it.
    return float(np.max(conv))


# ----------------------------------------------------------------------
# 3. Move generators
# ----------------------------------------------------------------------
def move_transfer_early_late(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from an early index to a late one."""
    n = base.size
    # early quarter, late quarter
    i = int(rng.integers(0, max(1, n // 4)))
    j = int(rng.integers(3 * n // 4, n))
    # fraction up to 90 % of the source mass
    frac = rng.random() * 0.90
    delta = frac * base[i]
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return cand


def move_block_early_late(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a small early block down and a distant late block up."""
    n = base.size
    # choose a modest block width (≈1 % of n)
    width = max(1, n // 100)
    # early block somewhere in the first quarter
    start_e = int(rng.integers(0, max(1, n // 4 - width + 1)))
    # late block somewhere in the last quarter, guaranteed non‑overlap
    start_l = int(rng.integers(3 * n // 4, n - width + 1))
    # reduction factor (<1) drawn log‑normally around 1
    factor_e = np.exp(rng.normal(0.0, 0.15))
    if factor_e > 1.0:
        factor_e = 1.0 / factor_e          # ensure reduction
    factor_l = 1.0 / factor_e               # compensating increase
    cand = base.copy()
    cand[start_e:start_e + width] *= factor_e
    cand[start_l:start_l + width] *= factor_l
    return cand


def move_global_noise(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a tiny independent multiplicative perturbation to every entry."""
    # σ chosen small so we stay close to the current point
    sigma = 0.04
    factor = np.exp(rng.normal(0.0, sigma, size=base.shape))
    cand = base * factor
    return cand


# ----------------------------------------------------------------------
# 4. Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beat the
    π/2 bound.  The algorithm:
        1. Initialise with the analytic arcsine sequence.
        2. Repeatedly apply mass‑transfer moves that move weight from
           early, high‑impact indices to distant low‑impact ones.
        3. Whenever a move decreases the current maximal convolution M,
           rescale the whole sequence by √(M_old/M_new) – this restores
           the bound but strictly increases the total sum, lowering the
           evaluation score.
        4. Keep the best sequence seen until the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # Length of the problem – use the global reference if present.
    if "height_sequence_1" in globals():
        n = int(len(globals()["height_sequence_1"]))
    else:
        n = 256
    if n == 0:
        n = 256

    # --------------------------------------------------------------
    # 1. Starting point – arcsine shape.
    # --------------------------------------------------------------
    best_seq = build_arcsine_sequence(n)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    best_M = conv_max(best_seq)                # current max of a*a
    sum_best = np.sum(best_seq)
    best_score = 2.0 * n * best_M / (sum_best * sum_best)
    print(f"[init] n={n}, init score={best_score:.6f}, max={best_M:.6f}, sum={sum_best:.6f}")

    # --------------------------------------------------------------
    # 2. Stochastic improvement loop.
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)   # guard a couple of seconds
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Choose a move variant.
        r = rng.random()
        if r < 0.45:
            cand = move_transfer_early_late(best_seq, rng)
        elif r < 0.80:
            cand = move_block_early_late(best_seq, rng)
        else:
            cand = move_global_noise(best_seq, rng)

        # ----------------------------------------------------------
        # 3. Evaluate candidate – fast path.
        # ----------------------------------------------------------
        cand_M = conv_max(cand)

        # If the move reduced the convolution bound we can safely
        # scale the whole vector to restore the old bound,
        # thereby strictly increasing the total mass.
        if cand_M < best_M * (1.0 - 1e-12):
            # scaling factor that brings the maximal convolution back
            # to the previous level (sqrt because a*a scales quadratically).
            scale = np.sqrt(best_M / cand_M)
            cand = cand * scale
            cand_M = conv_max(cand)          # recompute for safety
        # If the move increased the bound we keep the candidate as‑is
        # (the score may still improve because the sum may have risen).

        sum_cand = np.sum(cand)
        if sum_cand < 1e-8:
            continue  # degenerate – ignore

        score = 2.0 * n * cand_M / (sum_cand * sum_cand)

        # ----------------------------------------------------------
        # 4. Accept improvement.
        # ----------------------------------------------------------
        if np.isfinite(score) and score < best_score - 1e-9:
            best_seq = cand
            best_M = cand_M
            best_score = score
            print(f"[{it:>7}] new best score={best_score:.6f}, max={best_M:.6f}, sum={sum_cand:.6f}")

            # Early exit if we already beat the target comfortably.
            if best_score < 1.503:
                break

    print(f"[final] best score={best_score:.6f} after {it} iterations")
    # Return a plain Python list as required by the harness.
    return [float(x) for x in best_seq]
