def shift_mass_early_to_late(base, rng):
    cand = base.copy()
    n = cand.size
    # Choose early block
    max_early_len = n // 3
    early_len = rng.integers(1, max_early_len)
    early_start = rng.integers(0, n - early_len - max_early_len)  # ensure later room
    early_end = early_start + early_len
    # Choose later block same length maybe near end
    late_start = rng.integers(n - max_early_len, n - early_len + 1)   # ensures fit
    late_end = late_start + early_len

    # scaling factor less than 1
    factor = rng.uniform(0.5, 0.95)
    early_removed = np.sum(cand[early_start:early_end]) - factor * np.sum(cand[early_start:early_end])
    cand[early_start:early_end] *= factor

    # distribute removed mass uniformly to later block
    add_per = early_removed / early_len
    cand[late_start:late_end] += add_per
    # clip
    cand = np.clip(cand, 0.0, 1000.0)
    return cand

"""
Propose a step‑function that beats the classical binomial construction.

The algorithm starts from the (truncated) binomial coefficients – the
sequence that gives the classic bound π/2 ≈ 1.57086.  To improve on this,
the search uses a simulated‑annealing loop with three complementary
move types:

1. **Block perturbation** – a multiplicative log‑normal jitter on a
   randomly‑chosen contiguous block (the idea from the reference code).

2. **Early‑to‑late mass shift** – a dedicated move that transfers mass
   from an early part of the sequence (where the coefficients dominate
   many convolution entries) to a later part.  By decreasing the early
   coefficients while keeping the total sum roughly constant we can
   lower the peak of the self‑convolution without increasing the sum,
   which tends to reduce the evaluation score.

3. **Mirror‑swap** – the original “swap a little mass between i and
   n‑1‑i” move, useful for fine‑tuning the symmetry.

Moves are accepted if they improve the score, otherwise with a
temperature‑dependent probability (simulated annealing) so that the
search can escape shallow plateaus.  The temperature is linearly cooled
to near zero over the allotted wall‑clock budget.  The best sequence
encountered is returned.

The implementation uses only NumPy and the standard library and obeys
the required “no closures / no lambdas” rule.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _binomial_sequence(length: int) -> np.ndarray:
    """
    Truncated binomial coefficients a_i = C(2i,i) / 4**i.
    They satisfy (a*a)[k] = 1 for all k < length, giving the classic
    π/2 bound after the evaluation function.
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(length - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    return np.clip(a, 0.0, 1000.0)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a randomly‑chosen block."""
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and ~8 % of the length
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation – helps if the optimum is (approximately)
    # symmetric
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def shift_mass_early_to_late(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Transfer mass from an early block to a later block while keeping the
    total sum (almost) unchanged.  The early block is shrunk by a factor
    <1, the liberated mass is distributed uniformly over the later block.
    This reduces the contribution of the early coefficients to the
    convolution peak.
    """
    cand = base.copy()
    n = cand.size

    # size of the blocks – a few percent of the total length
    max_block = max(2, n // 12)
    block_len = int(rng.integers(1, max_block + 1))

    # early block near the front
    early_start = int(rng.integers(0, max(1, n // 3 - block_len + 1)))
    early_end = early_start + block_len

    # later block near the back, disjoint from the early block
    late_start = int(rng.integers(n - n // 3, max(1, n - block_len + 1)))
    late_end = late_start + block_len

    # shrink factor for the early block (0.5 … 0.95)
    factor = rng.uniform(0.5, 0.95)

    early_sum_before = np.sum(cand[early_start:early_end])
    cand[early_start:early_end] *= factor
    early_sum_after = np.sum(cand[early_start:early_end])

    # mass that has been removed
    delta = early_sum_before - early_sum_after
    if delta <= 0.0:
        return cand  # nothing to shift

    # distribute uniformly to the later block
    add_per = delta / block_len
    cand[late_start:late_end] += add_per

    return np.clip(cand, 0.0, 1000.0)


def _score(sequence: np.ndarray) -> float:
    """
    Fast wrapper around evaluate_sequence – reproduces exactly the same
    formula but avoids Python‑level list conversion on the inner loop.
    """
    # the public evaluate_sequence already does all the clipping and
    # safety checks, but invoking it millions of times is costly.
    # Re‑implement the formula directly (the same semantics).
    seq = np.clip(sequence, 0.0, 1000.0)
    n = seq.size
    b = np.convolve(seq, seq)
    max_b = np.max(b)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 900.0, **kwargs) -> list[float]:
    """
    Simulated‑annealing search for a non‑negative step‑function that
    minimises `evaluate_sequence`.  The routine respects the time budget
    (default 900 s) and always returns a valid list; the best sequence
    found during the run is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Initialise – use the globally‑available best construction if
    #    present; otherwise start from the analytic binomial sequence.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        cur = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = cur.size
        if n == 0:
            n = 256
            cur = _binomial_sequence(n)
    else:
        n = 256
        cur = _binomial_sequence(n)

    # make sure we start from a reasonably good point (binomial is fine)
    cur = np.clip(cur, 0.0, 1000.0)
    best = cur.copy()
    best_score = _score(best)
    cur_score = best_score

    print(f"[init] length={n}, start score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Simulated annealing loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a few seconds margin
    total_time = deadline - start_time

    # temperature schedule – linear cooling
    temp_start = best_score * 0.1   # a modest initial temperature
    temp_end = 1e-7

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # linear temperature decay
        elapsed = time.monotonic() - start_time
        T = max(temp_end, temp_start * (1.0 - elapsed / total_time))

        # ----------------------------------------------------------
        # pick a move type
        # ----------------------------------------------------------
        r = rng.random()
        if r < 0.45:
            # 45 %: early‑to‑late mass shift
            cand = shift_mass_early_to_late(cur, rng)
        elif r < 0.80:
            # 35 %: block perturbation (the classic move)
            cand = make_block_perturbation(cur, rng)
        else:
            # 20 %: mirror‑swap (small local adjustment)
            cand = cur.copy()
            i = rng.integers(0, n // 2)
            j = n - 1 - i
            delta = min(cand[i], cand[j]) * rng.random() * 0.1
            if rng.random() < 0.5:
                cand[i] += delta
                cand[j] -= delta
            else:
                cand[i] -= delta
                cand[j] += delta
            cand = np.clip(cand, 0.0, 1000.0)

        # ----------------------------------------------------------
        # evaluate the candidate
        # ----------------------------------------------------------
        try:
            score = _score(cand)
        except Exception:      # safety – should never happen
            continue
        if not math.isfinite(score):
            continue

        # ----------------------------------------------------------
        # Acceptance test (simulated annealing)
        # ----------------------------------------------------------
        if score < cur_score or rng.random() < math.exp(-(score - cur_score) / T):
            cur = cand
            cur_score = score

        # keep the global best
        if score < best_score:
            best = cand
            best_score = score
            print(f"[{it:7d}] new best: {best_score:.6f}")

    print(f"[done] iterations={it}, final best score={best_score:.6f}")
    # Return plain Python floats as required by the competition harness
    return [float(x) for x in best]
