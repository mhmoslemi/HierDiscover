"""
Propose a candidate step‑function for the convolution‐minimisation task.

The algorithm consists of three stages:

1. **Parameterized start** – generate a family of power‑law vectors
   a_i ∝ (i+β)^(-α) and keep the best one on a small grid of
   (α,β).  When α≈0.5 this reproduces the classic √‑law which yields
   a bound ≈π/2, but slight shifts of β can improve the discrete case.

2. **Fast simulated‑annealing / gradient moves** – repeatedly locate the
   index `k_max` where the current convolution attains its maximum.
   The contribution of each coefficient to that peak is
   `c_i = a_i * a_{k_max-i}`.  The algorithm moves a small amount of
   mass from a coefficient with a large contribution to a randomly
   chosen other coefficient, keeping the L¹‑norm fixed (scaling does not
   affect the objective).  Occasionally a random “block perturbation”
   (as used in the baseline) is tried to escape local minima.
   Acceptance follows a simple temperature schedule so that upward
   moves are allowed early on.

3. **Return the best vector** – after the time budget (≈ 1000 s) the best
   sequence found is clipped to the [0, 1000] range and returned as a
   plain Python list.  The external `evaluate_sequence` will be used to
   obtain the final score.

All heavy work (convolution and max‑search) is performed via an FFT
based routine `score_fft` which runs in O(n log n) and is therefore
fast enough for thousands of iterations on a few‑thousand‑element
vector.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation using FFT (identical to evaluate_sequence up to
# the harmless scaling invariance).  Returns the exact same quantity as
# evaluate_sequence for any non‑negative vector.
def score_fft(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq))/(sum(seq)²) computed with FFT."""
    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    n = seq.size
    # length for convolution: next power of two ≥ 2·n‑1 for speed
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (total * total)


# ----------------------------------------------------------------------
def power_law(alpha: float, beta: float, n: int) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+β)^(-α)."""
    idx = np.arange(n, dtype=np.float64) + beta
    seq = idx ** (-alpha)
    # enforce problem bounds
    seq = np.clip(seq, 0.0, 1000.0)
    if seq.sum() == 0:
        seq[0] = 1.0
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray,
                       rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) multiplicative perturbation of a vector."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # log‑normal multiplicative noise on a contiguous block
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation (helps the convolution peak)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

        # If a previous construction is present, reuse its length.
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        n = 2500                     # a comfortable size for the FFT

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1️⃣  Grid initialisation – explore (α,β) around the classical ½‑law.
    best_seq = power_law(alpha=0.5, beta=1.0, n=n)
    best_score = score_fft(best_seq)
    print(f"initial length={n}, score={best_score:.6f}")

    alphas = np.linspace(0.40, 0.60, 9)          # 0.40 … 0.60
    betas = np.linspace(0.0, 5.0, 11)           # 0 … 5
    for a in alphas:
        for b in betas:
            cand = power_law(alpha=a, beta=b, n=n)
            sc = score_fft(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
                print(f"grid improvement: α={a:.3f}, β={b:.3f}, score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Simulated‑annealing style refinement.
    cur_seq = best_seq.copy()
    cur_score = best_score
    iter_count = 0
    # Simple linear cooling schedule – start relatively “hot”.
    T0 = 0.02

    while time.monotonic() < deadline:
        # Compute convolution and locate the current maximum.
        n = cur_seq.size
        m = 1 << ((2 * n - 1).bit_length())
        f = np.fft.rfft(cur_seq, n=m)
        conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # Contribution of each coefficient to this peak.
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idxs = np.arange(i_start, i_end + 1)
        contrib = cur_seq[idxs] * cur_seq[k_max - idxs]
        if contrib.sum() == 0:
            prob = np.full_like(contrib, 1.0 / contrib.size)
        else:
            prob = contrib / contrib.sum()

        # Pick a “heavy” index i (large contribution) and a random partner j.
        i = int(rng.choice(idxs, p=prob))
        j = int(rng.integers(0, n))
        if j == i:
            continue

        # Small mass transfer from i to j (keeps the L¹‑norm unchanged).
        eps = rng.uniform(0.001, 0.015)          # step size
        delta = eps * cur_seq[i]
        if delta <= 0.0:
            continue
        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0:               # safety check
            continue

        cand_score = score_fft(cand)

        # Acceptance test (always accept improvement, otherwise Metropolis).
        if cand_score < cur_score:
            accept = True
        else:
            # Temperature decreases linearly with elapsed time.
            elapsed = time.monotonic() + (budget_s - deadline) - (budget_s - 5.0)
            T = T0 * (1.0 - elapsed / budget_s)
            accept = rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-8))

        if accept:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"new best {best_score:.6f} (iter {iter_count})")

        # Occasionally inject a random block perturbation to diversify the search.
        if rng.random() < 0.015:
            blk = block_perturbation(cur_seq, rng)
            blk_score = score_fft(blk)
            if blk_score < cur_score:
                cur_seq = blk
                cur_score = blk_score
                if blk_score < best_score:
                    best_score = blk_score
                    best_seq = blk.copy()
                    print(f"block new best {best_score:.6f}")

        iter_count += 1

    # ------------------------------------------------------------------
    # 3️⃣  Return the best candidate (clipped to the allowed range).
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # Normalisation is unnecessary because the objective is scale‑invariant,
    # but keeping a moderate scale avoids pathological under‑/overflow.
    total = best_seq.sum()
    if total == 0.0:
        best_seq[0] = 1.0
    else:
        best_seq *= (1.0 / total)      # optional normalisation to unit L¹

    print(f"final best score ≈ {best_score:.6f} after {iter_count} iterations")
    return best_seq.tolist()
