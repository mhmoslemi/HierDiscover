import numpy as np
import time

def propose_candidate(seed: int = 42,
                     budget_s: float = 900,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.

    The algorithm consists of three stages:

    1. **Coarse grid search** over a power‑law exponent α and an optional
       integer offset (i → i+offset).  For each (α,offset) a candidate
       sequence a_i ∝ (i+offset+1)^{-α} is built, normalised to sum 1,
       scaled by 100 (the evaluator is scale‑invariant) and scored.

    2. **Fine local optimisation** around the best (α,offset) found in the
       coarse phase, using a denser grid on α and a small neighbourhood
       of the offset.

    3. **Stochastic pairwise mass transfer** – while time permits we move a
       random fraction of the mass from one index to another and keep the
       improvement if the score decreases.

    The search works with a small ``n_search`` (≈2000) for speed and finally
    builds the returned sequence with the full length ``n_final`` (≈5000).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time budgeting – keep a small safety margin for the final return.
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Lengths: a fast length for the exploratory phase and the final length.
    # ------------------------------------------------------------------
    n_final = 5000                     # final construction size
    n_search = 2000                    # used while searching parameters

    # ------------------------------------------------------------------
    # Helper to build a power‑law (with optional offset) and evaluate it.
    # ------------------------------------------------------------------
    def eval_power(alpha: float, offset: int, n: int) -> tuple[float, list[float]]:
        """Return (score, sequence) for the given parameters."""
        # indices start at 1+offset, i.e. i ∈ {offset+1,…,offset+n}
        idx = np.arange(offset + 1, offset + n + 1, dtype=np.float64)
        # a_i ∝ idx^{-alpha}   →   a_i = exp(-alpha * log(idx))
        w = np.exp(-alpha * np.log(idx))
        # normalise to sum 1, then scale by 100 (any positive factor works)
        seq = (w / w.sum() * 100.0).tolist()
        score = evaluate_sequence(seq)
        return float(score), seq

    best_score = float("inf")
    best_alpha = 0.5
    best_offset = 0

    # ------------------------------------------------------------------
    # 1️⃣  Coarse grid: α ∈ [0.45,0.65] (step≈0.005), offset ∈ {0,2,…,20}
    # ------------------------------------------------------------------
    alphas_coarse = np.linspace(0.45, 0.65, 41)    # 41 values
    offsets_coarse = np.arange(0, 21, 2)          # 0,2,…,20

    for off in offsets_coarse:
        # pre‑compute log‑indices for this offset (speed‑up inner loop)
        log_idx = np.log(np.arange(off + 1, off + n_search + 1, dtype=np.float64))
        for a in alphas_coarse:
            if time.monotonic() >= deadline:
                break
            # build the candidate using the pre‑computed logs
            w = np.exp(-a * log_idx)
            cand = (w / w.sum() * 100.0).tolist()
            sc = evaluate_sequence(cand)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_alpha = a
                best_offset = off
                print(f"coarse new best α={a:.4f} off={off}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Fine search around the current best parameters.
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        # narrow the offset window (±2) and the exponent window (±0.02)
        off_low = max(0, best_offset - 2)
        off_high = best_offset + 2
        al_low = max(0.4, best_alpha - 0.02)
        al_high = min(0.8, best_alpha + 0.02)

        offsets_fine = np.arange(off_low, off_high + 1, 1)
        alphas_fine = np.linspace(al_low, al_high, 31)   # finer α grid

        for off in offsets_fine:
            log_idx = np.log(np.arange(off + 1, off + n_search + 1, dtype=np.float64))
            for a in alphas_fine:
                if time.monotonic() >= deadline:
                    break
                w = np.exp(-a * log_idx)
                cand = (w / w.sum() * 100.0).tolist()
                sc = evaluate_sequence(cand)
                if np.isfinite(sc) and sc < best_score:
                    best_score = float(sc)
                    best_alpha = a
                    best_offset = off
                    print(f"fine new best α={a:.5f} off={off}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Build the final (large) sequence with the chosen parameters.
    # ------------------------------------------------------------------
    _, best_seq = eval_power(best_alpha, best_offset, n_final)
    best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # 4️⃣  Optional stochastic local improvement (pairwise mass transfer).
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        base = np.asarray(best_seq, dtype=np.float64)
        total_mass = base.sum()

        while time.monotonic() < deadline:
            i, j = rng.integers(0, n_final, size=2)
            if i == j:
                continue
            # move up to 15 % of the mass from i to j
            delta = base[i] * rng.uniform(0.0, 0.15)
            cand = base.copy()
            cand[i] -= delta
            cand[j] += delta
            # keep numbers non‑negative (the move already guarantees this)
            cand = np.clip(cand, 0.0, None)
            # renormalise to the original total mass (scale‑invariant)
            cand = cand / cand.sum() * total_mass
            sc = evaluate_sequence(cand.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                base = cand
                best_seq = cand.tolist()
                print(f"local improve  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Fallback – ensure a valid return even if everything failed.
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = np.ones(n_final, dtype=np.float64).tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"final best score={best_score:.6f}   length={len(best_seq)} "
          f"α={best_alpha:.5f} offset={best_offset}")
    return [float(x) for x in best_seq]
