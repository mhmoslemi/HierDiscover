"""
Propose a high‑quality step‑function sequence.

The algorithm starts from the classic binomial sequence (the
coefficients of (1‑z)^‑1/2).  From this point we run a *Frank‑Wolfe*
style iteration:

1.  Compute M = max(f⊛f).
2.  Solve the linear program
        maximise  Σ g_i
        subject  f⊛g ≤ M   (component‑wise)
                  g_i ≥ 0.
    This finds a direction g that would increase the ℓ1‑norm of
    the coefficients while still respecting the current convolution
    bound.
3.  Rescale g so that Σg = Σf (the ℓ1‑norm stays constant – scaling
    does not change the evaluation metric).
4.  Take a small convex combination f←(1‑t)·f + t·g.
    The mixing reduces the peak of the auto‑convolution; the process
    is repeated until the time budget is exhausted.

Occasionally a random block perturbation is tried as a fallback
exploration step.  The best sequence seen during the run is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _binomial_sequence(length: int) -> np.ndarray:
    """Truncated binomial coefficients a_i = C(2i,i)/4^i (i≥0)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(length - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    return np.clip(a, 0.0, 1000.0)


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a multiplicative log‑normal perturbation to a random block."""
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and 8 % of n
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A for the constraints A @ g ≤ M,
    where (f⊛g)[k] = Σ_i f_i·g_{k-i}.
    The matrix has shape (2n‑1, n).
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes f[i] on rows i … i+n‑1
    for i, val in enumerate(f):
        A[i:i + n, i] = val
    return A


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the given
    `evaluate_sequence` metric.  The routine respects the wall‑clock
    budget and always returns a valid sequence.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Choose a working length.
    # ------------------------------------------------------------------
    # Use a moderate length (fast LP, still high‑quality).
    # If a previous construction is available we keep the same length,
    # otherwise we default to 512.
    if 'height_sequence_1' in globals():
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = max(128, min(int(len(base_seq)), 1024))
    else:
        n = 512

    # ------------------------------------------------------------------
    # 2.  Initialise with the binomial (analytic) sequence.
    # ------------------------------------------------------------------
    f = _binomial_sequence(n)
    sum_f = f.sum()
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)
    iteration = 0
    # mixing parameter (small enough to keep the peak drop monotone)
    t_mix = 0.2
    # probability of trying an LP step versus a random perturbation
    p_lp = 0.75

    while time.monotonic() < deadline:
        iteration += 1

        # --- compute the current convolution peak --------------------------------
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())

        # ----------------------------------------------------------------------
        # Decide which move to apply
        # ----------------------------------------------------------------------
        use_lp = rng.random() < p_lp
        if use_lp:
            # ----- linear‑programming step ------------------------------------
            # Build constraint matrix A for f⊛g ≤ M
            A = _constraint_matrix(f)
            rows = A.shape[0]

            # linprog solves  minimize  c·g  with  A_ub·g ≤ b_ub
            c = -np.ones(n, dtype=np.float64)            # maximise Σg_i
            b = np.full(rows, M, dtype=np.float64)

            # bounds: g_i ≥ 0 (no explicit upper bound)
            bounds = [(0.0, None)] * n

            # solve
            try:
                res = linprog(
                    c,
                    A_ub=A,
                    b_ub=b,
                    bounds=bounds,
                    method="highs",
                    options={"presolve": True, "time_limit": 30.0},
                )
                if not res.success:
                    raise RuntimeError("LP failed")
                g = res.x
            except Exception as e:
                # fall back to a simple perturbation if LP fails
                print(f"[{iteration:06d}] LP failed ({e}); using block perturbation")
                candidate = _make_block_perturbation(f, rng)
                # evaluate and possibly accept
                score = float(evaluate_sequence(candidate.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = candidate
                    best_score = score
                    print(f"[{iteration:06d}] new best (fallback) {best_score:.6f}")
                continue

            # ----- rescale g to keep the ℓ1‑norm unchanged --------------------
            sum_g = g.sum()
            if sum_g <= 0.0:
                # degenerate, skip this iteration
                continue
            scale = sum_f / sum_g
            g_scaled = g * scale

            # ----- convex combination with the old iterate --------------------
            f_new = (1.0 - t_mix) * f + t_mix * g_scaled
            f_new = np.clip(f_new, 0.0, 1000.0)

        else:
            # ----- random block perturbation ---------------------------------
            f_new = _make_block_perturbation(f, rng)

        # ------------------------------------------------------------------
        # 4.  Evaluate the new candidate
        # ------------------------------------------------------------------
        score = float(evaluate_sequence(f_new.tolist()))
        if np.isfinite(score) and score < best_score:
            best = f_new.copy()
            best_score = score
            print(f"[{iteration:06d}] new best score {best_score:.6f}")

        # Accept the update if it does not increase the peak; otherwise keep old f.
        conv_new = np.convolve(f_new, f_new)
        M_new = float(conv_new.max())
        if M_new < M - 1e-9:
            f = f_new
        # (if the peak grew we stay with the previous f – this guarantees
        # a monotone decrease of the bound).

    # ----------------------------------------------------------------------
    # 5.  Return the best sequence found.
    # ----------------------------------------------------------------------
    print(f"[done] iterations={iteration}, final best score={best_score:.6f}")
    return [float(x) for x in best]
