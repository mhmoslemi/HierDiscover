"""
Propose a step-function sequence using a hybrid of
  - exponentiated gradient descent (subgradient of max‑autocorrelation)
  - simulated‑annealing acceptance
  - block scaling, mass‑swap, uniform mixing and random rotation moves.
The sequence is kept on the probability simplex (non‑negative, sum=1) and
symmetrised with a moderate probability.  Fast FFT‑based convolution is
used to evaluate the objective `2·n·max(a*a)/(sum a)^2`.  The best
candidate found within the time budget is returned as a plain list of
floats.
"""

import time
import math
import numpy as np

# ---------- helper functions -----------------------------------------

def _conv_info(seq: np.ndarray):
    """
    Compute full convolution of seq with itself using FFT,
    return (conv_array, max_conv, argmax, score).
    """
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    max_conv = float(conv.max())
    k = int(conv.argmax())
    s = float(seq.sum())
    if s < 1e-12:
        score = float('inf')
    else:
        score = 2.0 * n * max_conv / (s * s)
    return conv, max_conv, k, score


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Scale to L1‑norm 1, keep non‑negative."""
    s = seq.sum()
    if s <= 0.0:
        return np.full_like(seq, 1.0 / seq.size)
    seq = seq / s
    # Clip tiny negatives caused by rounding
    np.clip(seq, 0.0, None, out=seq)
    return seq


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry."""
    return 0.5 * (seq + seq[::-1])


def _exp_gradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    """
    One exponentiated‑gradient step that tries to diminish the
    maximal autocorrelation value.
    """
    # compute convolution to locate the current worst index
    _, _, k, _ = _conv_info(seq)

    n = seq.size
    grad = np.zeros_like(seq)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    if i_min <= i_max:
        i_range = np.arange(i_min, i_max + 1, dtype=np.int64)
        j = k - i_range
        grad[i_range] = 2.0 * seq[j]

    # exponentiated update, keep positivity
    upd = np.exp(-eta * grad)
    new_seq = seq * upd
    new_seq = _normalize(new_seq)
    return new_seq


def _block_perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float) -> np.ndarray:
    """Scale a random contiguous block by exp(N(0, sigma))."""
    n = seq.size
    w_low = max(1, n // 200)         # ~0.5 % of n
    w_high = max(2, n // 12)         # ~8 % of n
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand = seq.copy()
    cand[start:start + width] *= factor
    # keep inside the allowed range (the simplex will later renormalise)
    cand = np.maximum(cand, 0.0)
    cand = _normalize(cand)
    return cand


def _mass_swap(seq: np.ndarray, rng: np.random.Generator, frac: float) -> np.ndarray:
    """Move a fraction of the mass from one index to another."""
    n = seq.size
    i, j = rng.choice(n, size=2, replace=False)
    if seq[i] <= 0.0:
        return seq.copy()
    epsilon = frac * seq[i] * rng.uniform(0.2, 0.9)
    cand = seq.copy()
    cand[i] = max(cand[i] - epsilon, 0.0)
    cand[j] += epsilon
    # renormalise to compensate for rounding errors
    cand = _normalize(cand)
    return cand


def _mix_uniform(seq: np.ndarray, rng: np.random.Generator, max_alpha: float) -> np.ndarray:
    """Convex combination of the current sequence and the uniform distribution."""
    n = seq.size
    alpha = float(rng.random() * max_alpha)          # amount of uniform part
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    cand = alpha * uniform + (1.0 - alpha) * seq
    cand = _normalize(cand)
    return cand


def _rotate(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cyclically shift the whole sequence by a random offset."""
    n = seq.size
    shift = int(rng.integers(0, n))
    return np.roll(seq, shift)


def _power_law_seq(n: int, exponent: float = 0.5, shift: float = 0.0) -> np.ndarray:
    """Generate a normalised power‑law shape a_i ∝ (i+shift+1)^{‑exponent}."""
    idx = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power(idx + shift + 1.0, exponent)
    return _normalize(vals)


# ---------- main entry point -----------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 950, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The algorithm mixes exponentiated‑gradient
    descent on the maximum autocorrelation with a simulated‑annealing
    style stochastic neighbourhood search (block scalings, mass swaps,
    uniform mixing and random rotations).  The best sequence found before
    the time budget expires is returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for final printing/evaluation
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.5)

            if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
                raise ValueError
            n = init.size
            base_seq = _normalize(np.clip(init, 0.0, 1000.0))
        except Exception:
            init = None
    else:
        init = None

    if init is None:
        # If no good starting point is available, fall back to a power‑law.
        n = 6000   # a size comparable to the historic best
        base_seq = _power_law_seq(n, exponent=0.5, shift=0.0)

    # Ensure the sequence lives on the simplex (sum = 1)
    base_seq = _normalize(base_seq)

    # Evaluate the initial candidate
    _, _, _, best_score = _conv_info(base_seq)
    best_seq = base_seq.copy()
    current_seq = base_seq.copy()
    current_score = best_score

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ----------------------------------------------------------------
    # Hyper‑parameters

    eta = 5.0               # step size for exponentiated gradient
    sigma = 0.5             # std for block scaling (log‑normal factor)
    swap_frac = 0.2         # proportion of source mass moved in a swap
    temperature = best_score * 0.05   # initial SA temperature
    temp_decay = 0.99985
    eta_decay = 0.9994
    sigma_decay = 0.9995
    swap_decay = 0.9996

    iteration = 0

    # Main optimisation loop
    # ----------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

            p=[0.35, 0.25, 0.20, 0.10, 0.10]

        if move_type == "grad":
            cand_seq = _exp_gradient_step(current_seq, eta)

        elif move_type == "block":
            cand_seq = _block_perturb(current_seq, rng, sigma)

        elif move_type == "swap":
            cand_seq = _mass_swap(current_seq, rng, swap_frac)

        elif move_type == "mix":
            cand_seq = _mix_uniform(current_seq, rng, max_alpha=0.4)

        else:  # rotate
            cand_seq = _rotate(current_seq, rng)

        # Occasionally enforce mirror symmetry (helps the autocorrelation)
        if rng.random() < 0.45:
            cand_seq = _symmetrize(cand_seq)

        # Normalise – tiny numerical errors may have accumulated
        cand_seq = _normalize(cand_seq)

        # Fast evaluation
        _, _, _, cand_score = _conv_info(cand_seq)

        # Simulated‑annealing acceptance
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # probability to accept a worse move
            prob = math.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            current_seq = cand_seq
            current_score = cand_score

            if cand_score < best_score:
                best_seq = cand_seq.copy()
                best_score = cand_score
                print(f"[{iteration:6d}] new best score {best_score:.6f} (type={move_type})")

        # Parameter cooling – performed every 100 iterations to keep the loop cheap
        if iteration % 100 == 0:
            temperature *= temp_decay
            eta = max(0.001, eta * eta_decay)

"""
Hybrid stochastic optimisation for the min‑max autocorrelation problem.

The objective to minimise is
        f(a) = 2·n·max(a∗a) / (∑a)²
where “∗” denotes (self‑)convolution.  The value is scale‑invariant, so the
search is performed on the probability simplex (all entries ≥0, sum = 1).

Algorithm
---------
1. Initialise with the best known sequence (`height_sequence_1`) if it exists,
   otherwise with a ½‑power‑law shape (a_i ∝ (i+1)^{-½}) normalised to sum = 1.
2. While time permits, repeatedly apply one of several inexpensive moves:
   * **Exponentiated‑gradient step** – computes the index k at which the
     convolution attains its maximum and performs a multiplicative update
     a_i ← a_i·exp(‑η·2·a_{k‑i}) (sub‑gradient descent on the max‑convolution).
   * **Block perturbation** – multiplies a random contiguous block by a
     log‑normally distributed factor.
   * **Mass swap** – moves a fraction of the mass from one entry to another.
   * **Uniform mixing** – convex combination of the current vector with the
     uniform distribution.
   * **Random rotation** – cyclic shift of the whole vector.
3. After each move the candidate is (optionally) symmetrised, renormalised
   to sum = 1 and evaluated via a fast FFT‑based convolution.
4. Acceptance follows a simulated‑annealing rule:
   * always accept improvements,
   * otherwise accept with probability exp(‑Δ/temperature).
   Temperature and all step‑size parameters are slowly decayed.
5. The best candidate encountered is finally evaluated with the authoritative
   `evaluate_sequence` and returned as a plain list of floats.

The search runs completely inside the supplied time budget (default 950 s)
and uses only NumPy, `time`, and `math`.  All helper functions are defined
at the top level as required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _conv_info(seq: np.ndarray):
    """
    Compute the full self‑convolution of `seq` using FFT,
    returning (conv_array, max_value, argmax, score) where
    score = 2·n·max(conv)/(sum(seq)²).
    """
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    max_conv = float(conv.max())
    k = int(conv.argmax())
    s = float(seq.sum())
    if s < 1e-12:
        score = float("inf")
    else:
        score = 2.0 * n * max_conv / (s * s)
    return conv, max_conv, k, score


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Project `seq` onto the simplex (non‑negative, L¹‑norm = 1)."""
    total = seq.sum()
    if total <= 0.0:
        return np.full_like(seq, 1.0 / seq.size)
    seq = seq / total
    np.clip(seq, 0.0, None, out=seq)           # guard against tiny negatives
    return seq


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Force mirror symmetry: a_i ← (a_i + a_{n‑1‑i}) / 2."""
    return 0.5 * (seq + seq[::-1])


def _exp_gradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient descent step on the maximal autocorrelation.
    The sub‑gradient of max(conv) at the worst index k is
        g_i = 2·a_{k‑i}  (if 0 ≤ k‑i < n, else 0).
    The update is a_i ← a_i·exp(‑η·g_i) followed by simplex projection.
    """
    # locate the current worst index
    _, _, k, _ = _conv_info(seq)

    n = seq.size
    grad = np.zeros_like(seq)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    if i_min <= i_max:
        i_range = np.arange(i_min, i_max + 1, dtype=np.int64)
        j = k - i_range
        grad[i_range] = 2.0 * seq[j]

    # multiplicative update, keep positivity
    update = np.exp(-eta * grad)
    new_seq = seq * update
    return _normalize(new_seq)


def _block_perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float) -> np.ndarray:
    """
    Multiply a random contiguous block by exp(N(0, sigma)).
    The block width is chosen between 0.5 % and ≈8 % of the total length.
    """
    n = seq.size
    w_low = max(1, n // 200)            # ~0.5 %
    w_high = max(2, n // 12)            # ~8 %
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand = seq.copy()
    cand[start:start + width] *= factor
    cand = np.maximum(cand, 0.0)
    return _normalize(cand)


def _mass_swap(seq: np.ndarray, rng: np.random.Generator, frac: float) -> np.ndarray:
    """
    Transfer a fraction (controlled by `frac`) of the mass from a random
    source index to a random destination index.
    """
    n = seq.size
    i, j = rng.choice(n, size=2, replace=False)
    if seq[i] <= 0.0:
        return seq.copy()
    epsilon = frac * seq[i] * rng.uniform(0.2, 0.9)
    cand = seq.copy()
    cand[i] = max(cand[i] - epsilon, 0.0)
    cand[j] += epsilon
    return _normalize(cand)


def _mix_uniform(seq: np.ndarray, rng: np.random.Generator, max_alpha: float) -> np.ndarray:
    """
    Convex combination with the uniform distribution.
    `alpha` (the weight of the uniform part) is drawn uniformly from
    [0, max_alpha].
    """
    n = seq.size
    alpha = float(rng.random() * max_alpha)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    cand = alpha * uniform + (1.0 - alpha) * seq
    return _normalize(cand)


def _rotate(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cyclically shift the whole vector by a random offset."""
    n = seq.size
    shift = int(rng.integers(0, n))
    return np.roll(seq, shift)


def _power_law_seq(n: int, exponent: float = 0.5, shift: float = 0.0) -> np.ndarray:
    """Generate a normalised 1/(i+shift+1)^{exponent} shape."""
    idx = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power(idx + shift + 1.0, exponent)
    return _normalize(vals)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 950, **kwargs):
    """
    Search for a step‑function sequence that minimises `evaluate_sequence`.
    The search mixes exponentiated‑gradient descent with a simulated‑annealing
    neighbourhood (block scaling, mass swaps, uniform mixing and rotations).
    The best sequence found before the time budget expires is returned as a
    plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # Reserve a couple of seconds for final printing / evaluation
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.5)

    # ------------------------------------------------------------------
    # 1️⃣ Initialise the working vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
                raise ValueError
            n = init.size
            base_seq = _normalize(np.clip(init, 0.0, 1000.0))
        except Exception:
            init = None
    else:
        init = None

    if init is None:
        # No usable historic solution – fall back to a power‑law shape
        n = 6000                     # a size comparable to the historic best
        base_seq = _power_law_seq(n, exponent=0.5, shift=0.0)

    # Ensure we are on the simplex
    base_seq = _normalize(base_seq)

    # Evaluate the starting point (fast version)
    _, _, _, best_score = _conv_info(base_seq)
    best_seq = base_seq.copy()
    current_seq = base_seq.copy()
    current_score = best_score

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Hyper‑parameters
    # ------------------------------------------------------------------
    eta = 5.0                     # step for exponentiated gradient
    sigma = 0.5                   # std. dev. for block perturbations
    swap_frac = 0.2               # fraction of source mass moved in a swap
    temperature = best_score * 0.05
    temp_decay = 0.99985
    eta_decay = 0.9994
    sigma_decay = 0.9995
    swap_decay = 0.9996

    iteration = 0

    # ------------------------------------------------------------------
    # 3️⃣ Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # Select a neighbourhood move (probabilities tuned empirically)
        move_type = rng.choice(
            ["grad", "block", "swap", "mix", "rotate"],
            p=[0.35, 0.25, 0.20, 0.10, 0.10]
        )

        if move_type == "grad":
            cand_seq = _exp_gradient_step(current_seq, eta)

        elif move_type == "block":
            cand_seq = _block_perturb(current_seq, rng, sigma)

        elif move_type == "swap":
            cand_seq = _mass_swap(current_seq, rng, swap_frac)

        elif move_type == "mix":
            cand_seq = _mix_uniform(current_seq, rng, max_alpha=0.4)

        else:   # "rotate"
            cand_seq = _rotate(current_seq, rng)

        # Optional symmetry enforcement (helps flatten the autocorrelation)
        if rng.random() < 0.45:
            cand_seq = _symmetrize(cand_seq)

        # Re‑project onto the simplex (removes accumulated round‑off)
        cand_seq = _normalize(cand_seq)

        # Fast objective evaluation
        _, _, _, cand_score = _conv_info(cand_seq)

        # Simulated‑annealing acceptance
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            current_seq = cand_seq
            current_score = cand_score

            if cand_score < best_score:
                best_seq = cand_seq.copy()
                best_score = cand_score

        # --------------------------------------------------------------

            sigma = max(0.001, sigma * sigma_decay)
            swap_frac = max(0.01, swap_frac * swap_decay)

    # ------------------------------------------------------------------

        print(f"[final] authoritative score = {auth_score:.6f} after {iteration} iterations")
    except Exception as exc:
        print(f"[final] evaluate_sequence raised an exception: {exc!r}")

    # Return a plain Python list of floats as required
    return [float(v) for v in best_seq]
