def _compute_gradient(seq, conv):
    n = seq.size
    max_val = conv.max()
    # tolerance: relative tolerance maybe 1e-12 absolute?
    # We'll treat near-equal values within 1e-12 of max.
    ks = np.where(conv >= max_val - 1e-12)[0]
    g = np.zeros_like(seq)
    for k in ks:
        # add contributions
        idx = np.arange(n)
        mask1 = (k - idx >= 0) & (k - idx < n)

        g[mask2] += seq[idx[mask2] - k]
    # average over number of ks
    if ks.size > 0:

    return g, max_val

* evaluates the full self‑convolution with an FFT,

authoritative ``evaluate_sequence`` to obtain the official score.

def _convolution(seq: np.ndarray) -> np.ndarray:
    """
    Compute the full discrete convolution of ``seq`` with itself using an
    FFT of the next power‑of‑two length.  Returns a real 1‑D array of length
    ``2*len(seq)-1``.
    """
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values; clamp to zero.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n log n) algorithm of
    Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]

    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative – return the uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].sum() / rho
    w = v - theta
    w[w < 0] = 0.0
    return w


def _gradient_max_conv(seq: np.ndarray, conv: np.ndarray) -> tuple:
    """
    Return a sub‑gradient of ``max(conv)`` with respect to ``seq`` and the
    maximal convolution value.  If the maximum is attained at several
    indices the gradients are averaged.
    """
    n = seq.size
    max_val = float(conv.max())
    # Consider indices whose value is within a tiny tolerance of the maximum.
    # The tolerance is absolute; 1e-12 is safe for double precision.
    active = np.where(conv >= max_val - 1e-12)[0]
    g = np.zeros_like(seq)
    # Vectorised computation of the gradient for all active k
    idx = np.arange(n)
    for k in active:
        # first term: a_{k-i} if 0 ≤ k-i < n
        mask1 = (k - idx >= 0) & (k - idx < n)
        g[mask1] += seq[k - idx[mask1]]
        # second term: a_{i-k} if 0 ≤ i-k < n
        mask2 = (idx - k >= 0) & (idx - k < n)
        g[mask2] += seq[idx[mask2] - k]
    if active.size > 1:
        g /= float(active.size)
    return g, max_val


def _power_law_seq(n: int, exponent: float = 0.5, shift: float = 0.0) -> np.ndarray:
    """
    Produce a normalised power‑law sequence a_i ∝ (i+shift+1)^(-exponent).
    The result is already scaled to have ℓ1‑norm equal to 1.
    """
    idx = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power(idx + shift + 1.0, exponent)
    vals_sum = vals.sum()
    if vals_sum == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals_sum
    return vals


def _mix_uniform_power(n: int, exponent: float, shift: float, alpha: float) -> np.ndarray:
    """
    Convex mixture of the uniform distribution and a power‑law distribution.
    ``alpha`` is the weight of the uniform component.
    The result is normalised to ℓ1‑norm 1.
    """
    uniform = np.full(n, 1.0 / n)
    power = _power_law_seq(n, exponent=exponent, shift=shift)
    mixed = alpha * uniform + (1.0 - alpha) * power
    mixed_sum = mixed.sum()
    if mixed_sum == 0:
        mixed = uniform
    else:
        mixed /= mixed_sum
    return mixed

    expression is homogeneous of degree 0).

simplex.  For a fixed a, let conv = a*a be the full self‑convolution.
If K = {k : conv[k] = max(conv)} the sub‑gradient of max(conv) is
