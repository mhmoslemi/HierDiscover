import time, numpy as np, scipy.optimize as opt, scipy.sparse as sp

def propose_candidate(seed=0, budget_s=900):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    global height_sequence_1
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        n = 1024
        base = None

    # catalan start
    a = catalan_sequence(n)
    best = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    print(f"Initial score {best_score:.8f}")

    # maybe replace if base better
    if base is not None and base.size == n:
        sbase = float(evaluate_sequence(base.tolist()))
        if sbase < best_score:
            best = base.copy()
            best_score = sbase
            print(f"Using provided base with score {best_score:.8f}")

    iteration = 0
    while time.monotonic() < deadline - 0.5 and iteration < 20:
        n = best.size
        conv_self = np.convolve(best, best)
        M = float(conv_self.max())
        # Solve LP
        # Build A_ub
        m = 2*n - 1
        # Build sparse matrix using diags
        diags = []
        offsets = []
        for i in range(n):
            length = n - i
            diags.append(np.full(length, best[i]))
            offsets.append(i)
        A_ub = sp.diags(diags, offsets, shape=(m, n), format='csc')
        b_ub = np.full(m, M + 1e-12) # small epsilon

        c = -np.ones(n)  # maximize sum g

        # Use highs
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n,
                          method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.monotonic() - 0.2)})

        if not res.success:
            print(f"LP failed iter {iteration}: {res.message}")
            break

        g = np.maximum(0.0, res.x)
        # try mixing
        improved = False
        # choose a set of t values
        for t in (0.5, 0.3, 0.2, 0.1, 0.05):
            cand = (1 - t) * best + t * g
            cand = np.clip(cand, 0.0, 1000.0)
            score = float(evaluate_sequence(cand.tolist()))
            if score < best_score:
                best = cand
                best_score = score
                improved = True
                print(f"Iter {iteration} mix t={t:.2f} new score {best_score:.8f}")
                break
        if not improved:
            # random perturbation
            perturb = np.exp(rng.normal(0.0, 0.02, size=n))
            cand = best * perturb
            cand = np.clip(cand, 0.0, 1000.0)
            score = float(evaluate_sequence(cand.tolist()))
            if score < best_score:
                best = cand
                best_score = score
                print(f"Iter {iteration} random better score {best_score:.8f}")
        iteration += 1

    print("Final best", best_score)
    return [float(v) for v in best]

"""
Catalan‑type initialization + linear‑programming refinement.

The algorithm starts from the Catalan sequence
    a_i = binom(2i,i)/4**i   (i≥0)
which gives a convolution max value M = 1 and a score ≈ π/2.

We then repeatedly solve the linear program

    maximise   Σ g_i
    s.t.       (a * g)_k ≤ M   for all k = 0 … 2n‑2
                g_i ≥ 0

using SciPy’s high‑performance HiGHS solver.  The LP finds a non‑negative
vector g that can be added to a without increasing the worst convolution
value M.  A convex combination a ← (1‑t)·a + t·g (with a few step sizes)
strictly increases the total sum Σ a_i while keeping M unchanged, which
lowers the evaluation score 2·n·M/Σa².  If the LP does not yield an
improvement we fall back to a cheap multiplicative random perturbation.

The loop runs until the time budget expires (or a small fixed number of
iterations is reached) and the best sequence seen is returned.
All numbers are clipped to the interval [0,1000] as required by
`evaluate_sequence`.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def catalan_sequence(n: int) -> np.ndarray:
    """
    Return the first n Catalan‑type coefficients
        C_i = binom(2i,i) / 4**i   (i≥0)

    The recurrence C_{i+1} = C_i * (2*i+1)/(2*i+2) is used.
    Result is clipped to the interval [0,1000] for safety.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(a, 0.0, 1000.0)


def convolution_matrix(a: np.ndarray) -> sp.csc_matrix:
    """
    Build the (2n‑1) × n sparse matrix A such that
        (A @ g)[k] = (a * g)[k]
    for all k.  Each diagonal k‑j = offset contains the constant a[offset].
    """
    n = a.shape[0]
    m = 2 * n - 1
    diags = []
    offsets = []
    for offset, val in enumerate(a):
        length = n - offset          # number of entries on this diagonal
        if length <= 0:
            continue
        diags.append(np.full(length, val, dtype=np.float64))
        offsets.append(offset)       # row = col + offset  (lower diagonal)
    return sp.diags(diags, offsets, shape=(m, n), format='csc')


def solve_lp_for_g(a: np.ndarray, M: float, time_remaining: float):
    """
    Solve the linear program:
        maximise Σ g_i
        s.t.   (a * g)_k ≤ M  for all k
               g_i ≥ 0

    The constraint matrix is built on the fly using a sparse Toeplitz
    representation of the convolution.  The HiGHS interior‑point solver
    (accessed via scipy.optimize.linprog) is used.

    Returns
        g : np.ndarray of shape (n,)  (the optimal vector,
               or None if the LP failed)
    """
    n = a.size
    A_ub = convolution_matrix(a)               # shape (2n‑1, n)
    b_ub = np.full(A_ub.shape[0], M + 1e-12)   # tiny epsilon for safety
    c = -np.ones(n, dtype=np.float64)          # maximise sum(g)  →  minimise -sum(g)

    # HiGHS respects a `time_limit` option (seconds).  If the remaining time
    # is very small we just skip the LP.
    if time_remaining <= 0.1:
        return None

    # The `highs` method accepts sparse matrices directly.
    res = opt.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0.0, None)] * n,
        method='highs',
        options={'presolve': True, 'time_limit': time_remaining}
    )
    if not res.success or res.x is None:
        return None
    # Numerical noise may produce tiny negatives – clip them.
    return np.maximum(0.0, res.x)


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 900) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence`.  The routine starts from the Catalan sequence,
    repeatedly solves a linear programme that enlarges the total sum without
    increasing the worst convolution value, mixes the result a little, and
    keeps the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # ------------------------------------------------------------------
    # 1. Determine a convenient length and initial sequence.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)  # global if present
        n = base.size
    except Exception:
        # fallback length – 1024 is a good trade‑off between speed and quality
        n = 1024
        base = None

    # Catalan start (deterministic baseline)
    a = catalan_sequence(n)

    # If a previous construction is supplied and strictly better, keep it.
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if base is not None and base.size == n:
        try:
            base_score = float(evaluate_sequence(base.tolist()))
            if base_score < best_score:
                best = base.copy()
                best_score = base_score
                print(f"Using provided global sequence as start, score={best_score:.6f}")
        except Exception:
            pass

    print(f"Initial score: {best_score:.8f} (n={n})")

    # ------------------------------------------------------------------
    # 2. Iterative improvement: LP → mixing → evaluation
    # ------------------------------------------------------------------
    iteration = 0
    max_iterations = 15   # safety guard; early exit if time runs out

    while time.monotonic() < deadline and iteration < max_iterations:
        # Current worst convolution value
        conv_self = np.convolve(best, best)
        M = float(conv_self.max())
        # ------------------------------------------------------------------
        # 2a. Linear program to find a sum‑increasing direction
        # ------------------------------------------------------------------
        time_left = deadline - time.monotonic()
        g = solve_lp_for_g(best, M, time_left * 0.6)  # reserve room for mixing

        if g is None:
            print(f"Iter {iteration}: LP failed or timed out.")
            break

        # ------------------------------------------------------------------
        # 2b. Convex mixing – try a few step sizes and keep the first improvement.
        # ------------------------------------------------------------------
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1, 0.07, 0.04):
            candidate = (1.0 - t) * best + t * g
            candidate = np.clip(candidate, 0.0, 1000.0)
            try:
                cand_score = float(evaluate_sequence(candidate.tolist()))
            except Exception:
                continue
            if np.isfinite(cand_score) and cand_score < best_score:
                best = candidate
                best_score = cand_score
                improved = True
                print(f"Iter {iteration:02d}, mix t={t:.2f} → score {best_score:.8f}")
                break  # accept first improvement

        # ------------------------------------------------------------------
        # 2c. If LP mixing did not improve, try a cheap random tweak.
        # ------------------------------------------------------------------
        if not improved:
            perturb = np.exp(rng.normal(0.0, 0.03, size=n))
            candidate = best * perturb
            candidate = np.clip(candidate, 0.0, 1000.0)
            try:
                cand_score = float(evaluate_sequence(candidate.tolist()))
            except Exception:
                cand_score = np.inf
            if np.isfinite(cand_score) and cand_score < best_score:
                best = candidate
                best_score = cand_score
                print(f"Iter {iteration:02d}, random tweak → score {best_score:.8f}")

        iteration += 1

    print(f"Finished after {iteration} iterations, best score = {best_score:.8f}")
    # Return a plain Python list as required by the harness
    return [float(v) for v in best]
