def gradient_move(base, rng):
    n = base.size
    conv = np.convolve(base, base)
    M = conv.max()
    kstar = conv.argmax()
    S = base.sum()
    # compute contributions g_i = 2 * base[kstar - i] if valid
    indices = np.arange(n)
    js = kstar - indices
    valid = (js >= 0) & (js < n)
    g = np.zeros_like(base)
    g[valid] = 2 * base[js[valid]]
    # compute condition: if g_i * S > 2*M => want to decrease, else increase.
    threshold = 2 * M / S
    # g_i > threshold leads to decreasing; else increasing.
    dec_mask = g > threshold
    inc_mask = ~dec_mask  # might include zeros.
    
    # select candidate i from dec_mask with probability proportional to g (maybe)
    if dec_mask.any():
        dec_indices = np.where(dec_mask)[0]
        # choose i weighted by g_i perhaps
        probs = g[dec_indices] / g[dec_indices].sum()
        i = rng.choice(dec_indices, p=probs)
    else:
        # no decreasing needed? maybe we are near optimum
        i = rng.integers(0, n)
    
    if inc_mask.any():
        inc_indices = np.where(inc_mask)[0]
        # choose j weighted by something like (threshold - g) big.
        # Use inverse weighting maybe
        # just random uniform
        j = rng.choice(inc_indices)
        # ensure i != j
        while j == i:
            j = rng.choice(inc_indices)
    else:
        # fallback random j
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

    # Transfer fraction of source i's mass to j
    # Transfer fraction up to 0.4 of a_i.
    frac = rng.random() * 0.4
    delta = frac * base[i]
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)

def gradient_transfer(base, rng):
    n = len(base)
    conv = np.convolve(base, base)
    M = conv.max()
    k = int(conv.argmax())
    S = base.sum()
    idx = np.arange(n)
    j = k - idx
    valid = (j >= 0) & (j < n)
    g = np.zeros_like(base)
    g[valid] = 2.0 * base[j[valid]]
    thresh = 2.0 * M / S
    dec = np.where(g > thresh)[0]
    inc = np.where(g < thresh)[0]
    if dec.size == 0 or inc.size == 0:
        # fallback: simple random transfer
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n-1))
        if j >= i:
            j += 1
        cand = base.copy()
        frac = rng.random() * 0.3
        delta = frac * cand[i]
        cand[i] -= delta
        cand[j] += delta
        return np.clip(cand, 0.0, 1000.0)
    # Choose donor index weighted by g
    donor_weights = g[dec]
    donor_weights = donor_weights / donor_weights.sum()
    i = int(rng.choice(dec, p=donor_weights))
    j = int(rng.choice(inc))
    # transfer fraction
    frac = rng.random() * 0.25
    delta = frac * base[i]
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    # Ensure nonnegative
    cand = np.clip(cand, 0.0, 1000.0)
    # Optional re-normalize to keep sum equal to S (should be close)
    sum_c = cand.sum()
    if sum_c != 0:
        cand *= (S / sum_c)
    return cand

"""
Optimisation of the step‑function bound.
-------------------------------------------------
We start from the analytically optimal *arcsine* (inverse‑square‑root)
sequence a_i = C·binom(2i,i)/4**i.  After normalising the L¹‑norm to 1,
this construction is known to attain the continuous lower bound
    2·n·max(a∗a) / (∑a)^2 = π/2
as n → ∞.  For finite n the actual score is a little larger because of the
truncation at index n‑1.

To improve on this we perform a fast gradient‑style local search that
preserves the total mass:

* The full convolution of the current candidate is computed.
* Let k* be a location where the convolution attains its maximum M.
* For each index i the (sub‑)gradient of M w.r.t a_i is
        g_i = 2·a_{k*‑i}   (if 0 ≤ k*‑i < n, otherwise 0)
  (derived from ∂(a∗a)_k/∂a_i = 2·a_{k‑i}).

  The derivative of the objective
        f(a) = 2·n·M / (∑a)²
  shows that we should *decrease* a_i when g_i·∑a > 2·M
  and *increase* it otherwise.

* A single move therefore transfers a small fraction of the mass from a
  “high‑gradient” index to a “low‑gradient” index.  This reduces M while
  keeping the L¹‑norm unchanged, which strictly improves the score.
* To avoid getting stuck in a flat region we interleave occasional
  random block‑scaling perturbations (as in the reference code) and a
  simple simulated‑annealing acceptance rule.

The routine runs while the time budget permits, always keeping the best
sequence seen.  It works for any length supplied via the global variable
`height_sequence_1`; if that is missing a default length of 256 is used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Analytic start – the “arcsine” sequence
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the inverse‑square‑root (arcsine) coefficients
    a[i] = binom(2i,i) / 4**i   (i = 0,…,length‑1)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    # recurrence relation: a[i] = a[i‑1] * ((2i‑1)*(2i)) / (4·i²)
    for i in range(1, length):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    return a


# ----------------------------------------------------------------------
# 2.  Small random block perturbation (exploration)
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ)).  The total mass is
    re‑normalised to the original L¹‑norm."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width

    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor

    # occasional symmetric averaging – keeps the shape well‑behaved
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])

    # re‑scale to keep the L¹‑norm unchanged
    total = cand.sum()
    if total > 0:
        cand *= (base.sum() / total)

    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# 3.  Gradient‑driven mass transfer
# ----------------------------------------------------------------------
def gradient_transfer(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a fraction of the mass from an index that contributes a lot to
    the current maximal convolution entry to an index that contributes
    little.  The move preserves the L¹‑norm."""
    n = base.size
    conv = np.convolve(base, base)                # full (2n‑1) convolution
    M = conv.max()
    k_star = int(conv.argmax())
    S = base.sum()

    # g_i = 2·a_{k*-i} for indices where the partner exists
    idx = np.arange(n)
    partner = k_star - idx
    valid = (partner >= 0) & (partner < n)
    g = np.zeros_like(base)
    g[valid] = 2.0 * base[partner[valid]]

    # Decision threshold from the sub‑gradient of the objective
    thresh = 2.0 * M / S
    dec_idx = np.where(g > thresh)[0]   # indices we would like to decrease
    inc_idx = np.where(g < thresh)[0]   # indices we would like to increase

    # Fallback – simple random transfer if one side is empty
    if dec_idx.size == 0 or inc_idx.size == 0:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n - 1))
        if j >= i:
            j += 1
        cand = base.copy()
        frac = rng.random() * 0.25
        delta = frac * cand[i]
        cand[i] -= delta
        cand[j] += delta
        return np.clip(cand, 0.0, 1000.0)

    # Choose a donor weighted by its contribution to the max
    donor_weights = g[dec_idx]
    donor_weights /= donor_weights.sum()
    i = int(rng.choice(dec_idx, p=donor_weights))

    # Choose a receiver uniformly from the low‑contribution set
    j = int(rng.choice(inc_idx))

    # Transfer a modest fraction of the donor's mass
    frac = rng.random() * 0.25          # up to 25 % of the donor’s value
    delta = frac * base[i]

    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    cand = np.clip(cand, 0.0, 1000.0)

    # Re‑normalise to keep the L¹‑norm (numerical drift may appear)
    total = cand.sum()
    if total > 0:
        cand *= (S / total)

    return cand

        3. Accept a candidate if it improves the score, otherwise accept
           it with a small probability that decays with time (simulated
           annealing).  The best sequence ever seen is retained.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the length and the starting sequence.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, list):
        start = np.array(height_sequence_1, dtype=np.float64)
        if start.size == 0:
            n = 256
            start = build_arcsine_sequence(n)
    else:
        n = 256
        start = build_arcsine_sequence(n)

    n = int(start.size)
    # Normalise the L¹‑norm to 1 – the objective is scale‑invariant.
    total = start.sum()
    if total == 0:
        total = 1.0
    cur_seq = start / total
    cur_seq = np.clip(cur_seq, 0.0, 1000.0)

    best_seq = cur_seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    cur_score = best_score

    print(f"[init] length={n}, initial L1={cur_seq.sum():.6f}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # keep a 2 s buffer
    it = 0
    # Simple linear cooling schedule for the acceptance temperature
    while time.monotonic() < deadline:
        it += 1
        # Choose a move type
        if rng.random() < 0.55:
            cand = gradient_transfer(cur_seq, rng)
        else:
            cand = block_perturbation(cur_seq, rng)

        # Evaluate – this also enforces all the security checks
        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
            continue

        # Keep the global optimum
        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score
            print(f"[{it:>6}] new best = {best_score:.6f}")

        # Simulated‑annealing acceptance for the current walk
        if cand_score < cur_score:
            cur_seq, cur_score = cand, cand_score
        else:
            # temperature falls from ~0.02 to 0 over the run
            elapsed = time.monotonic() - start_time
            temperature = 0.02 * (1.0 - elapsed / (deadline - start_time))
            accept_prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < accept_prob:
                cur_seq, cur_score = cand, cand_score

        # Very early stop if we already beat a generous target
        if best_score < 1.50:
            print("[stop] reached target < 1.50")
            break

    print(f"[final] best score {best_score:.6f} after {it} iterations, length={n}")
    # Convert to plain Python list for the harness
    return [float(x) for x in best_seq]
