def proj_simplex(v):
   # project onto probability simplex (sum=1, v>=0)
   # Equivalent to l1-norm ball with radius 1 and nonnegativity.
   # Input v (numpy array)
   # if sum(v) == 1 and all >=0 already: return v
   # else:
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - 1
   ind = np.arange(len(v)) + 1
   cond = u - cssv / ind > 0
   rho = ind[cond][-1]
   theta = cssv[cond][-1] / rho
   w = np.maximum(v - theta, 0)
   return w

2. **Multiplicative (exponentiated) gradient descent** – the objective
   ``2·n·max(conv)/sum^2`` is scale‑invariant, so we keep the vector on the
   probability simplex (sum = 1, a_i ≥ 0).  For the index where the
   self‑convolution attains its maximum we compute the sub‑gradient
   (g_i = 2·a_{k−i} for all valid i) and update the vector with a
   multiplicative step ``a_i ← a_i·exp(-η·g_i)`` followed by renormalisation.
   This pushes weight away from entries that contribute most to the peak.

3. **Occasional stochastic block perturbations** – to escape local minima we
   apply the inexpensive block‑wise log‑normal perturbation used in the
   previous solution.

The whole search runs until the supplied time budget expires (default ≈ 1000 s)
and returns the best vector found, as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation (identical to evaluate_sequence except for the
# safety checks).  All vectors are kept non‑negative and normalised to sum=1,
# so the denominator is constant and the objective reduces to 2·n·max(conv).
def _evaluate_fast(seq: np.ndarray) -> float:
    n = seq.size
    conv = np.convolve(seq, seq)          # full convolution, O(n)
    max_b = conv.max()
    # sum(seq) is 1 by construction – if it is not, the caller normalises.
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex (sum = 1, non‑negative)
# using the algorithm of Duchi et al. (2008).  Not used in the main loop
# because the multiplicative update preserves positivity, but kept for
# occasional fall‑backs.
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x | x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    # shift by the largest element to improve numerical stability
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: cheap block‑wise multiplicative perturbation (as in the previous
# solution).  Used sporadically to inject diversity.
def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # multiplicative log‑normal noise on the selected block
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation – helps avoid a single dominant peak
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # clipping respects the security limits of ``evaluate_sequence``
    cand = np.clip(cand, 0.0, 1000.0)
    # renormalise (the outer optimiser works on the simplex)
    total = cand.sum()
    if total == 0:
        cand = np.full_like(cand, 1.0 / n)
    else:
        cand /= total
    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 900,
                     **kwargs) -> list[float]:
    """
    Search for a low‑L∞ self‑convolution step function.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds.  The function stops a few hundred
        milliseconds before the deadline to leave a safety margin.
    **kwargs
        Additional arguments are ignored (kept for compatibility).

    Returns
    -------
    list[float]
        The best sequence found (normalised to sum = 1; the external
        ``evaluate_sequence`` will re‑clip it if necessary).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Determine the length of the vector we are allowed to use.
    #    Prefer the length of the supplied construction if it exists.
    try:
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if base_arr.ndim != 1 or base_arr.size == 0:
            raise ValueError
        n = int(base_arr.size)
    except Exception:
        n = 2600          # a robust default that fits comfortably in the budget

    # ------------------------------------------------------------------
    # 2) Initialise with a good known candidate.
    best_seq = None
    best_score = np.inf

    #   a)  If a previous construction is available, use its normalised form.
    try:
        seq0 = np.clip(base_arr, 0.0, 1000.0)
        seq0 = seq0 / seq0.sum()
        score0 = _evaluate_fast(seq0)
        best_seq = seq0.copy()
        best_score = score0
        print(f"initialising from previous best, score={best_score:.6f}")
    except Exception:
        pass

    #   b)  Simple analytic power‑law (p = 0.5) as a fallback seed.
    if best_seq is None:
        idx = np.arange(1, n + 1, dtype=np.float64)
        seq0 = 1.0 / np.sqrt(idx)
        seq0 = np.clip(seq0, 0.0, 1000.0)
        seq0 /= seq0.sum()
        best_seq = seq0.copy()
        best_score = _evaluate_fast(seq0)
        print(f"fallback analytic seed, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Small random search in the 2‑parameter family
    #    a_i ∝ (i + offset)^{-p}.
    search_deadline = time.monotonic() + 0.15 * float(budget_s)
    while time.monotonic() < search_deadline:
        p = rng.uniform(0.35, 0.65)               # exponent
        offset = rng.integers(0, max(1, n // 10) + 1)  # shift, up to 10 % of n
        idx = np.arange(n, dtype=np.float64) + offset + 1.0
        cand = np.power(idx, -p)

        # occasional Gaussian smoothing (helps to reduce localized spikes)
        if rng.random() < 0.3:
            sigma = rng.uniform(0.5, 3.0)
            kern_len = int(min(n, max(3, sigma * 6)))
            x = np.arange(kern_len) - kern_len // 2
            kernel = np.exp(-0.5 * (x / sigma) ** 2)
            kernel /= kernel.sum()
            cand = np.convolve(cand, kernel, mode="same")

        cand = np.clip(cand, 0.0, 1000.0)
        cand /= cand.sum()
        score = _evaluate_fast(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"param‑search new best: {best_score:.6f}  p={p:.4f} off={offset}")

    # ------------------------------------------------------------------
    # 4) Multiplicative (exponentiated) gradient descent on the simplex.
    cur_seq = best_seq.copy()
    cur_score = best_score
    step = 0.05                     # initial learning rate η
    no_improve = 0
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    while time.monotonic() < deadline:
        # --- compute convolution and locate the current maximum(s) ---
        conv = np.convolve(cur_seq, cur_seq)
        max_val = conv.max()
        max_idxs = np.where(conv == max_val)[0]

        # --- sub‑gradient of the max‑convolution (sum of contributions) ---
        grad = np.zeros_like(cur_seq)
        for k in max_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # vector of indices i that actually appear in the sum
            i_range = np.arange(i_start, i_end + 1, dtype=np.intp)
            grad[i_range] += 2.0 * cur_seq[k - i_range]

        # --- multiplicative update (exponentiated gradient) ---
        cand = cur_seq * np.exp(-step * grad)
        total = cand.sum()
        if total == 0.0:                     # numerical safety net
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / total
        cand_score = _evaluate_fast(cand)

        # --- accept or reject the step ---
        if cand_score < cur_score:
            cur_seq = cand
            cur_score = cand_score
            # keep the global best
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"gradient descent new best: {best_score:.6f}")
            # success → slightly increase step size
            step = min(step * 1.08, 0.5)
            no_improve = 0
        else:
            # failure → shrink step size and possibly inject noise
            step *= 0.72
            no_improve += 1
            if no_improve >= 30:
                # random log‑normal perturbation to shake out of a plateau
                noise = np.exp(rng.normal(0.0, 0.05, size=n))
                pert = cur_seq * noise
                pert /= pert.sum()
                pert_score = _evaluate_fast(pert)
                if pert_score < cur_score:
                    cur_seq = pert
                    cur_score = pert_score
                no_improve = 0

        # --- occasional block‑wise random perturbation (global exploration) ---
        if rng.random() < 0.07:
            blk = _make_block_perturbation(cur_seq, rng)
            blk_score = _evaluate_fast(blk)
            if blk_score < best_score:
                best_score = blk_score
                best_seq = blk.copy()
                print(f"block‑perturb new best: {best_score:.6f}")

    print(f"search finished – best score ≈ {best_score:.6f}")
    # Return a plain Python list; the external evaluator will apply its own
    # clipping and safety checks.
    return best_seq.tolist()
```
