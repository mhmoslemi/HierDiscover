"""
Propose a new step‑function sequence.

The previous solution used a simple block‑wise multiplicative hill‑climber.
Below we implement an *iterated linear‑programming* refinement that follows
the scheme described in the statement:

1.  Start from a reasonable analytic shape (1/√i) or from the supplied
    `height_sequence_1` if it exists.
2.  For the current candidate `f` compute the current worst auto‑convolution
    value `M = max(f * f)`.
3.  Solve the linear programme

        maximise    Σ b_j
        subject to  (f * b)_k ≤ M   for every convolution index k
                    b_j ≥ 0
                    b_j ≤ 1000

   This finds a vector `b` that is as “large” as possible while keeping the
   convolution with `f` below the current bound `M`.
4.  Rescale `b` to unit L¹ norm (the score is scale‑invariant) and blend it
   with the current iterate

        f ← (1‑t)·f + t·b          (t is a slowly decreasing step‑size)

   The blend guarantees monotonic non‑increase of the objective.
5.  Occasionally perform a cheap random block perturbation (as in the
   reference implementation) to help escape plateaus.
6.  Keep the best sequence seen and stop when the time budget expires.

The linear programme is small (O(n) variables, O(2n) constraints) and can be
solved very quickly with SciPy’s HiGHS interior‑point solver.  With a modest
length (n≈256…512) the whole optimisation finishes well within the 1000 s
budget, but the code automatically falls back to the length of
`height_sequence_1` (which may be a few thousand) if it is provided.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper functions – top level, no closures or lambdas as required.
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that A @ b = np.convolve(f, b).

    For each column i the value f[i] appears on the diagonal that starts at
    row i and ends at row i + n‑1.
    """
    n = f.size
    rows = np.arange(2 * n - 1).repeat(n)          # placeholder, will be sliced
    cols = np.tile(np.arange(n), 2 * n - 1)       # placeholder, will be sliced
    data = []
    row_idx = []
    col_idx = []

    # fill column by column – this is fast enough for n up to a few thousand
    for i in range(n):
        start = i
        stop = i + n
        length = n
        row_idx.extend(range(start, stop))
        col_idx.extend([i] * length)
        data.extend([f[i]] * length)

    mat = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return mat


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear programme described in the docstring.
    Returns the optimal b (as a dense NumPy array) or None if the solver fails.
    """
    n = f.size
    # Build the convolution matrix A (sparse for speed and memory)
    A = _build_conv_matrix(f)

    # Upper bounds for every convolution entry:   (f * b)_k ≤ M
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear objective: maximise Σ b_j  → minimise -Σ b_j
    c = -np.ones(n, dtype=np.float64)

    # Bounds for the variables: 0 ≤ b_j ≤ 1000   (the outer clipping in evaluate)
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # HiGHS is the default for recent SciPy versions; it accepts sparse matrices.
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method="highs", options={"presolve": True})

    if res.success:
        return res.x
    return None


def _conv_max(seq: np.ndarray) -> float:
    """Return max of the full self‑convolution."""
    return np.max(np.convolve(seq, seq))


def _random_block_perturbation(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap multiplicative perturbation of a random block, plus optional symmetrisation.
    Mirrors the strategy used in the reference code but is kept separate here.
    """
    n = seq.size
    # block width between 1 % and 8 % of the length (clamped)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width

    cand = seq.copy()
    # log‑normal multiplicative noise
    cand[start:stop] *= np.exp(rng.normal(0.0, 0.12))
    # occasional symmetrisation (helps to flatten the convolution)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # enforce domain constraints
    return np.clip(cand, 0.0, 1000.0)


def _normalise(seq: np.ndarray) -> np.ndarray:
    """Normalise to unit L¹ sum (the objective is scale‑invariant)."""
    total = seq.sum()
    if total < 1e-12:
        # fallback to a tiny uniform vector to avoid division by zero
        return np.full_like(seq, 1.0 / seq.size)
    return seq / total
