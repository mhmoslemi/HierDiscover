"""
Propose a step‑function that beats the analytical arcsine baseline.

The algorithm starts from the arcsine (inverse‑square‑root) sequence,
which gives a score ≈ π/2 ≈ 1.5708.  From there a simulated‑annealing
loop explores several simple, cheap moves that deliberately push mass
towards the borders and smooth the peak at the first element.
Typical moves:

* block scaling (multiplying a random contiguous block)
* mass transfer (moving a fraction from one index to another)
* edge‑biased transfer (source in the middle, target near the ends)
* small multiplicative jitter
* mixing the current shape with a uniform background
* a 3‑point smoothing (kernel [0.25,0.5,0.25])
* reducing the highest coefficient (the “peak‑reduction” move)

All candidates are clipped to the allowed [0, 1000] range and
renormalised to total sum = 1 (the evaluation is homogeneous, any
normalisation is fine).  The best sequence seen so far is kept,
and an annealing temperature gradually decays to allow occasional
worse moves early on.  The search runs for the allotted time budget
(~ 950 s) and stops early if the target score < 1.503 is reached.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: the analytical arcsine (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative arcsine sequence a[i] = binom(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence   a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Simple stochastic moves
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:start + width] *= factor
    # occasional symmetrisation (helps stability)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return cand


def transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction (≤ 30 %) of mass from one index to another."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return cand


def edge_transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer mass from a middle region toward the boundaries."""
    n = base.size
    # source: middle half, target: outer quarters
    src_lo = n // 4
    src_hi = 3 * n // 4
    i = int(rng.integers(src_lo, src_hi))
    # pick a target in the left or right quarter
    if rng.random() < 0.5:
        j = int(rng.integers(0, n // 4))
    else:
        j = int(rng.integers(3 * n // 4, n))
    frac = rng.random() * 0.40   # up to 40 % of the source value
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return cand


def mix_with_uniform(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Convex combination with a uniform background (mix‑parameter λ)."""
    n = base.size
    lam = rng.random() * 0.5          # λ ∈ [0,0.5]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return (1.0 - lam) * base + lam * uniform


def smooth_sequence(base: np.ndarray) -> np.ndarray:
    """Three‑point smoothing with kernel [0.25,0.5,0.25]."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    # mode='same' keeps the length unchanged
    return np.convolve(base, kernel, mode='same')


def peak_reduction(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Shrink the largest coefficient (the “peak”)."""
    idx = int(np.argmax(base))
    # reduce it to a random factor between 0.5 and 1.0
    factor = 0.5 + 0.5 * rng.random()
    cand = base.copy()
    cand[idx] *= factor
    return cand


def jitter_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal jitter on each entry (small σ)."""
    n = base.size
    noise = np.exp(rng.normal(0.0, 0.03, size=n))
    return base * noise


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950, **kwargs):
    """
    Search for a step‑function that minimises
    evaluate_sequence(...).  Returns the best list of heights found.
    """
    rng = np.random.default_rng(seed)

    # Length of the problem – use the provided construction if present.
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256

    # ------------------------------------------------------------------
    # 1.  Starting point – the analytically optimal arcsine shape.
    # ------------------------------------------------------------------
    base_seq = build_arcsine_sequence(n)
    # Normalise to sum = 1 (the objective is scale‑invariant)
    base_seq = base_seq / np.sum(base_seq)

    best = base_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # Current state (used for the annealing walk)
    cur = best.copy()
    cur_score = best_score

    # Simulated‑annealing parameters
    temperature = 0.025          # initial temperature
    min_temp = 1e-3
    decay = 0.9997               # exponential decay per iteration

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)  # a couple of seconds slack
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # 2.  Propose a new candidate using a randomly chosen move.
        # ------------------------------------------------------------------
        move = rng.choice(
            ["block", "transfer", "edge", "jitter", "mix", "smooth", "peak"]
        )
        cand = cur.copy()

        if move == "block":
            cand = block_perturbation(cand, rng)
        elif move == "transfer":
            cand = transfer_move(cand, rng)
        elif move == "edge":
            cand = edge_transfer_move(cand, rng)
        elif move == "jitter":
            cand = jitter_move(cand, rng)
        elif move == "mix":
            cand = mix_with_uniform(cand, rng)
        elif move == "smooth":
            cand = smooth_sequence(cand)
        elif move == "peak":
            cand = peak_reduction(cand, rng)

        # Clip to the hard bounds required by evaluate_sequence
        np.clip(cand, 0.0, 1000.0, out=cand)

        # Renormalise to keep the scale stable (any non‑zero sum works)
        total = cand.sum()
        if total < 1e-12:
            continue
        cand /= total

        # ------------------------------------------------------------------
        # 3.  Evaluate the candidate.
        # ------------------------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))

        # ------------------------------------------------------------------
        # 4.  Acceptance rule (simulated annealing).
        # ------------------------------------------------------------------
        accept = False
        if score < cur_score:
            accept = True
        else:
            # probability to accept a worse move
            prob = np.exp(-(score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = score

        # ------------------------------------------------------------------
        # 5.  Keep the best sequence ever seen.
        # ------------------------------------------------------------------
        if score < best_score:
            best = cand
            best_score = score
            print(f"[{it:>6}] new best = {best_score:.6f}")

            # Early exit if we already beat the target comfortably
            if best_score < 1.503:
                break

        # ------------------------------------------------------------------
        # 6.  Temperature schedule.
        # ------------------------------------------------------------------
        temperature = max(min_temp, temperature * decay)

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return a plain Python list – the harness expects a list of floats.
    return [float(x) for x in best]
