"""
Propose a candidate step‑function sequence.

The algorithm combines several ideas that differ from the reference
implementation:

*   Multiple diverse initial seeds (uniform, power‑law, exponential).
*   Simulated‑annealing style acceptance: occasional uphill moves
    allow escape from local minima.
*   Two complementary perturbation operators:
    – block‑wise multiplicative log‑normal changes (as in the reference);
    – global multiplicative noise applied to every entry.
*   Optional symmetrisation and random swaps to explore the permutation
    space.
*   A simple exponential cooling schedule and a hard time‑budget
    (1000 s total, leaving ~10 s margin for the harness).

All candidates are clipped to the limits required by ``evaluate_sequence``
(0 ≤ value ≤ 1000).  The best sequence found before the deadline is
returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# The function ``evaluate_sequence`` is assumed to be already imported.


def _clip(sequence: np.ndarray) -> np.ndarray:
    """Clamp to the domain accepted by ``evaluate_sequence``."""
    # negatives become 0, values >1000 become 1000
    return np.clip(np.maximum(sequence, 0.0), 0.0, 1000.0)


def _score(seq: np.ndarray) -> float:
    """Convenient wrapper returning a Python float."""
    # ``evaluate_sequence`` expects a plain list.
    return float(evaluate_sequence(seq.tolist()))


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a pool of diverse starting points."""
    cand = []

    # 1. Uniform (mid‑range, safe baseline)
    cand.append(np.full(n, 500.0, dtype=np.float64))

    # 2. Power‑law families (the shape that performed well previously)
    exponents = np.concatenate(
        [
            np.linspace(0.10, 0.45, 8, endpoint=False),
            np.linspace(0.45, 0.80, 15, endpoint=False),
            np.linspace(0.80, 0.95, 8),
        ]
    )
    for p in exponents:
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        seq = seq * (1000.0 / seq.max())
        cand.append(seq)

    # 3. Exponential decay families
    lambdas = np.linspace(0.0005, 0.0100, 12)
    for lam in lambdas:
        seq = np.exp(-lam * np.arange(n, dtype=np.float64))
        seq = seq * (1000.0 / seq.max())
        cand.append(seq)

    # 4. Randomised modest perturbations of the uniform baseline
    for _ in range(5):
        noise = np.exp(rng.normal(0.0, 0.25, size=n))
        seq = np.full(n, 500.0) * noise
        cand.append(_clip(seq))

    return cand


def _perturb_block(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """One O(n) block perturbation (multiplicative log‑normal)."""
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12 (inclusive)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    return cand


def _perturb_global(base: np.ndarray, rng: np.random.Generator,
                    sigma: float = 0.08) -> np.ndarray:
    """Multiply every entry by an independent log‑normal factor."""
    factors = np.exp(rng.normal(0.0, sigma, size=base.shape))
    return base * factors


def propose_candidate(seed: int = 42, budget_s: float = 950.0, **kwargs):
    """Return a list of non‑negative heights that (hopefully) beats the
    previous best score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Allowed running time in seconds (a small safety margin is
        subtracted internally).

    Returns
    -------
    list[float]
        The best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Determine the working length (use the previously discovered one if
    # available, otherwise fall back to a modest size).
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        if n == 0:
            raise Exception
    except Exception:
        n = 256
    print(f"[info] Working with sequence length n={n}")

    # ------------------------------------------------------------------
    # Evaluate a collection of diversified seeds.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    for seq in _initial_candidates(n, rng):
        seq = _clip(seq)
        sc = _score(seq)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[init] score={sc:.6f}")

    # If something went terribly wrong fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 500.0, dtype=np.float64)
        best_score = _score(best_seq)
        print("[fallback] uniform sequence used")

    print(f"[info] Starting simulated‑annealing, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing loop (time‑bounded)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Temperature schedule
    temperature = 0.5          # initial temperature
    cooling = 0.9998           # exponential cooling factor
    it = 0

    while time.monotonic() < deadline:
        # ----- choose a perturbation -----------------------------------
        cand = best_seq.copy()
        r = rng.random()
        if r < 0.55:
            cand = _perturb_block(cand, rng)
        elif r < 0.85:
            # global multiplicative noise, sigma shrinks with temperature
            sigma = max(0.01, temperature * 0.2)
            cand = _perturb_global(cand, rng, sigma=sigma)
        else:
            # simple swap of two entries (changes the permutation)
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        # occasional symmetry enforcement (helps many autocorrelation
        # problems)
        if rng.random() < 0.25:
            cand = 0.5 * (cand + cand[::-1])

        cand = _clip(cand)

        # ----- evaluate -------------------------------------------------
        sc = _score(cand)
        if not math.isfinite(sc):
            continue

        delta = sc - best_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis rule – allows uphill moves early on
            prob = math.exp(-delta / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[improve] it={it:06d} new best={best_score:.6f}")

        # ----- update temperature & iteration counter -------------------
        temperature *= cooling
        it += 1

    print(f"[done] final best score = {best_score:.6f}")
    return best_seq.tolist()
