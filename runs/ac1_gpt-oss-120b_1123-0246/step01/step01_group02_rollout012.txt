"""
Convolution‑bound optimiser.

The optimiser starts from the analytically optimal “arcsine’’ sequence,
then explores a low‑dimensional family of perturbations (tilt,
power‑law and a mixture with the reversed arcsine) to get a better
objective value.  After the cheap parameter sweep it continues with a
directed stochastic local search that repeatedly moves a small amount
of mass from indices that contribute most to the current maximal
convolution coefficient towards indices that contribute the least.
Both the parameter sweep and the local search respect the security
checks of ``evaluate_sequence`` (non‑negative, ≤1000).

The whole routine respects a time budget (default 500 s, the harness
gives 1000 s) and returns the best sequence found.
"""
import numpy as np
import time

# ----------------------------------------------------------------------
# 1.  Helper – build the “arcsine’’ (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i  (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    # recurrence: a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a  # values already lie in [0,1]


# ----------------------------------------------------------------------
# 2.  Simple stochastic moves (kept from the reference code)
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0, σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# 3.  Directed transfer – move mass from a “high‑impact’’ index
#    to a “low‑impact’’ index.
# ----------------------------------------------------------------------
def directed_transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small fraction of mass from the index that contributes most
    to the current maximal convolution coefficient to an index that
    contributes the least."""
    n = base.size
    conv = np.convolve(base, base)

    # use only the first n entries – they contain the interior maximum
    max_idx = int(np.argmax(conv[:n]))

    # contributions to the active convolution entry
    # for i in [0..max_idx] we have pair (i, max_idx-i)
    idx = np.arange(max_idx + 1)
    contrib = base[idx] * base[max_idx - idx]

    src = int(np.argmax(contrib))               # most harmful index
    # pick a target among the 20 % smallest values (far end, low impact)
    low_thr = np.percentile(base, 20)
    low_candidates = np.where(base <= low_thr)[0]
    if low_candidates.size == 0:
        # fall back to a random index that is not src
        tgt = int(rng.integers(0, n - 1))
        if tgt >= src:
            tgt += 1
    else:
        tgt = int(rng.choice(low_candidates))
    if tgt == src:
        return base.copy()

    # fraction to move – slowly decreasing with time
    frac = rng.random() * 0.20 + 0.01          # between 1 % and 21 %
    delta = frac * base[src]
    if delta <= 0.0:
        return base.copy()

    cand = base.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)

# 4.  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of floats – the best step‑function heights found
    within the time budget.  The optimisation proceeds in three phases:

    1. Build the exact arcsine sequence (the analytical baseline).
    2. Perform a cheap grid search over a few low‑dimensional
       perturbations (tilt, power‑law, mixture with the reversed
       arcsine) and keep the best result.
    3. Run a directed stochastic local search that repeatedly moves
       mass from indices that contribute most to the current maximal
       convolution coefficient towards low‑impact indices.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size (length of the known sequence)
    # --------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256                     # a harmless fallback

    # ------------------------------------------------------------------
    # Phase 1 – analytical starting point
    # ------------------------------------------------------------------
    base_seq = build_arcsine_sequence(n)
    best = base_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, arcsine score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – low‑dimensional parameter sweep
    # ------------------------------------------------------------------
    rev_seq = base_seq[::-1]

    # tilt with reversed arcsine: a(α) = (1-α)·a + α·a_rev
    for alpha in np.linspace(0.0, 0.40, 41):        # 0 % … 40 %
        cand = (1.0 - alpha) * base_seq + alpha * rev_seq
        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[sweep] tilt α={alpha:.3f} → score={sc:.6f}")

    # power‑law deformation: a(p) = a**p  (p<1 shifts mass to the tail)
    for p in np.linspace(0.60, 1.00, 9):           # 0.60 … 1.00
        cand = np.power(base_seq, p)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[sweep] power p={p:.3f} → score={sc:.6f}")

    # combined tilt + power
    for alpha in np.linspace(0.0, 0.30, 21):
        mixed = (1.0 - alpha) * base_seq + alpha * rev_seq
        for p in np.linspace(0.70, 0.95, 7):
            cand = np.power(mixed, p)
            cand = np.clip(cand, 0.0, 1000.0)
            sc = float(evaluate_sequence(cand.tolist()))
            if sc < best_score:
                best, best_score = cand, sc
                print(f"[sweep] α={alpha:.2f}, p={p:.2f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – directed stochastic local search
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)   # keep a small slack
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # 70 % directed transfer, 30 % generic block perturbation
        if rng.random() < 0.7:
            cand = directed_transfer_move(best, rng)
        else:
            cand = block_perturbation(best, rng)

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc
            print(f"[{it:>6}] new best = {best_score:.6f}")

            # early stop if we have already beaten the target
            if best_score < 1.503:
                print("[stop] target reached")
                break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    return [float(x) for x in best]
