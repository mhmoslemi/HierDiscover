"""
Propose a step‑function (a non‑negative sequence) that minimises
`evaluate_sequence`.  

The algorithm works in three stages:

1. **Initialisation** – a handful of simple families (uniform,
   several power‑laws, Dirichlet random vectors) are sampled and the
   best of them (according to the true evaluator) is kept.

2. **Peak‑flattening hill‑climbing** – the current sequence is repeatedly
   examined via FFT–based convolution.  The shift `k_max` where the
   convolution attains its maximum is identified and the pair of indices
   `(i, j=k_max-i)` that contributes most to this peak is moved to
   random locations that do **not** belong to the same shift.  This
   directly reduces the maximal convolution value `max_b`.  The move is
   accepted whenever it improves the objective; a tiny simulated‑annealing
   fallback occasionally accepts a worse move to escape plateaus.

3. **Final random fine‑tuning** – after the main loop a fast stochastic
   local‑search (similar to the reference implementation) is performed
   for the remaining time budget.

All operations respect the non‑negativity constraint, keep the total
mass (≈ 100) constant (the evaluation is scale‑invariant) and stay well
below the evaluator’s clipping limits.  The routine returns the best
sequence found before the allotted time expires.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def power_law_sequence(length: int, exponent: float) -> np.ndarray:
    """Return a non‑negative power‑law sequence of given length.
    The vector is normalised to sum to 1 and then scaled by 100 (any
    positive scale works because the objective is homogeneous).
    """
    idx = np.arange(length, dtype=np.float64) + 1.0
    w = np.power(idx, -exponent)
    w /= w.sum()
    return w * 100.0


def fast_score(seq: np.ndarray, n: int, total_mass: float) -> float:
    """Return 2 n * max(conv(seq,seq)) / total_mass².
    The convolution is computed by an FFT of size a power of two.
    """
    fft_len = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (total_mass * total_mass)


def convolution_max(seq: np.ndarray, n: int) -> float:
    """Return only the maximal convolution value."""
    fft_len = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    return conv.max()


def convolution_full(seq: np.ndarray, n: int) -> np.ndarray:
    """Full convolution (real part) of seq with itself."""
    fft_len = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(f * f, n=fft_len)[: 2 * n - 1]
    return conv

    The function runs for up to ``budget_s`` seconds and returns the best
    sequence discovered.  All numbers are non‑negative; the total sum is
    kept around 100 so that the internal clipping of ``evaluate_sequence``
    never activates.

    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        n = max(256, min(n, 8000))
    except Exception:
        n = 4096

    # --------------------------------------------------------------
    candidates = []

    # --------------------------------------------------------------

        k_max = int(np.argmax(conv))

        cand_score = fast_score(cand, n, total_mass)

        # Accept if we improved; otherwise accept with a tiny probability that
        # decays with elapsed time (simulated annealing flavour).
        if cand_score < cur_score - 1e-12:
            seq = cand
        else:
            # annealing probability
            elapsed = time.monotonic() - (deadline - budget_s + 0.5)
            temperature = max(0.01, 1.0 - elapsed / budget_s)
            prob = np.exp((cur_score - cand_score) / (temperature + 1e-12))
            if prob > rng.random():
                seq = cand

        # Occasionally shuffle a bit if we have been stuck for a while.
        if no_improve_counter > 3000:
            # Small random perturbation around the current best.
            jitter = rng.gamma(0.5, size=n)
            jitter = jitter / jitter.sum() * total_mass
            seq = best_seq * 0.85 + jitter * 0.15
            seq = np.maximum(seq, 0.0)
            no_improve_counter = 0
            print(f"[iter {iteration}] jitter reset")

    # --------------------------------------------------------------
    # 3️⃣  Final brief stochastic local refinement (as a safety net).
    # --------------------------------------------------------------
    base = best_seq.copy()
    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move at most 15 % of the source mass
        delta = base[i] * rng.uniform(0.0, 0.15)
        cand = base.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)

        cand_score = fast_score(cand, n, total_mass)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            base = cand
            print(f"[local] improved score={best_score:.6f}")

    print(f"final best score={best_score:.6f}   length={len(best_seq)}")
    # Convert to plain Python list of floats for the caller.
    return [float(x) for x in best_seq.tolist()]
