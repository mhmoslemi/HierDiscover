def _evaluate_fft(seq):
    n = seq.shape[0]
    total = seq.sum()
    if total < 1e-12:
        return float('inf')
    # pad to length >= 2n-1 for convolution
    size = 1 << ((2*n - 1).bit_length())
    freq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(freq * freq, n=size)
    # conv length size; but real convolution of length 2n-1 is first 2n-1 elements
    max_b = np.max(conv[:2*n-1])
    return float(2.0 * n * max_b / (total * total))
```

def _evaluate_fast(seq):
    n = seq.shape[0]
    total = seq.sum()
    if total < 1e-12:
        return float('inf')
    size = 1 << ((2*n - 1).bit_length())
    freq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(freq * freq, n=size)
    max_b = conv[:2*n-1].max()
    return float(2.0 * n * max_b / (total * total))
```

def _pair_transfer(seq: np.ndarray, rng: np.random.Generator, max_frac: float) -> np.ndarray:
    n = seq.size

import numpy as np
import time

def _evaluate_fast(seq: np.ndarray) -> float:
    n = seq.shape[0]
    if n == 0:
        return float('inf')
    total = seq.sum()
    if total < 1e-12:
        return float('inf')
    size = 1 << ((2*n - 1).bit_length())
    # rfft returns n/2+1 values for real transform
    freq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(freq * freq, n=size)
    max_b = conv[:2*n-1].max()
    return float(2.0 * n * max_b / (total * total))
```

def _block_multiply(seq, rng, sigma):
    n = seq.size
    lo = max(1, n // 300)
    hi = max(2, n // 20)
    w = int(rng.integers(lo, hi+1))
    start = int(rng.integers(0, n - w + 1))
    factor = np.exp(rng.normal(0.0, sigma))
    out = seq.copy()
    out[start:start+w] *= factor
    np.clip(out, 0.0, 1000.0, out=out)

def _transfer_mass(seq, rng, max_frac=0.3):
    n = seq.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    while j == i:
        j = int(rng.integers(0, n))
    delta = rng.random() * max_frac * seq[i]
    out = seq.copy()
    out[i] -= delta
    out[j] += delta
    if out[i] < 0.0:
        out[i] = 0.0
    np.clip(out, 0.0, 1000.0, out=out)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 500, **kwargs):
    """
    Simulated-annealing + random‑parameter search for a step‑function that
    minimises ``evaluate_sequence``.  The algorithm first explores a few
    parametric families (power‑law, shifted, exponential) to obtain a good
    starting point, then performs a long annealed hill‑climb using block
    multiplicative perturbations and targeted mass‑transfer moves.  All
    candidates are kept normalised (sum = 1) and are evaluated via an
    FFT‑based fast version of ``evaluate_sequence``.
    """
    import numpy as np, time
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # Helper to evaluate quickly (FFT version)
    def eval_fast(v):
        return _evaluate_fast(v)

    # ------------------------------------------------------------------
    # 1️⃣  Parameteric random search (≈15 % of the budget)
    param_budget = 0.15 * budget_s
    param_deadline = start_time + param_budget
    best_seq = None
    best_score = float('inf')

    length_options = [800, 1200, 1600, 2000, 2400, 2800, 3200]
    # parameter ranges

        p = rng.uniform(*p_range)
        shift = rng.uniform(*shift_range)
        lam = rng.uniform(*lam_range)

"""
Step‑function optimizer.

The algorithm works in two phases.

1️⃣  A short “parameter sweep’’ randomly explores a three‑parameter
    family a_i ∝ (i+shift)^{-p}·exp(-λ·i).  Each candidate is normalised
    (sum = 1) and evaluated with an FFT‑based replica of
    ``evaluate_sequence``.  The best candidate of this stage seeds the
    second phase.

2️⃣  A long simulated‑annealing hill‑climber refines the seed.
    Perturbations consist of:
        – multiplicative block noise,
        – targeted mass‑transfer from early to late indices,
        – occasional symmetrisation.
    Candidates are always re‑normalised; a Metropolis rule with a
    temperature that decays linearly with elapsed time decides acceptance.
    The best sequence found is returned as a plain Python list.

All helpers are defined at top level (no closures) and the evaluation uses
FFT convolution for speed, allowing many thousands of iterations within the
time budget.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Fast evaluation (FFT version of ``evaluate_sequence``)
def _evaluate_fast(seq: np.ndarray) -> float:
    """Return the objective value for a non‑negative sequence.

    The function assumes ``seq`` is a one‑dimensional NumPy array of floats.
    It uses an FFT‑based convolution to compute max(f*f) efficiently.
    """
    n = seq.shape[0]
    if n == 0:
        return float("inf")
    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    # length of the full convolution is 2*n‑1; pad to next power of two
    size = 1 << ((2 * n - 1).bit_length())
    freq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(freq * freq, n=size)
    max_b = conv[: 2 * n - 1].max()
    return float(2.0 * n * max_b / (total * total))


# ----------------------------------------------------------------------
# Family of analytic shapes
def _shape_power_law(n: int, p: float, shift: float) -> np.ndarray:
    """Generate a raw power‑law sequence a_i = (i+shift+1)^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    seq = 1.0 / np.power(idx + shift + 1.0, p)
    return seq


# ----------------------------------------------------------------------
# Local perturbations
def _block_multiply(seq: np.ndarray,
                    rng: np.random.Generator,
                    sigma: float) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0, sigma))."""
    n = seq.size
    lo = max(1, n // 300)
    hi = max(2, n // 20)
    w = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - w + 1))
    factor = np.exp(rng.normal(0.0, sigma))
    out = seq.copy()
    out[start:start + w] *= factor
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def _transfer_mass(seq: np.ndarray,
                   rng: np.random.Generator,
                   max_frac: float) -> np.ndarray:
    """Move a random fraction (≤max_frac) of mass from i to j."""
    n = seq.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    while j == i:
        j = int(rng.integers(0, n))
    delta = rng.random() * max_frac * seq[i]
    out = seq.copy()
    out[i] -= delta
    out[j] += delta
    if out[i] < 0.0:
        out[i] = 0.0
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def _rightward_transfer(seq: np.ndarray,
                        rng: np.random.Generator,
                        max_frac: float) -> np.ndarray:
    """Preferentially shift mass from an early region to a late region."""
    n = seq.size
    # early quarter, late quarter
    early_end = max(1, n // 4)
    late_start = max(early_end, 3 * n // 4)
    i = int(rng.integers(0, early_end))
    j = int(rng.integers(late_start, n))
    delta = rng.random() * max_frac * seq[i]
    out = seq.copy()
    out[i] -= delta
    out[j] += delta
    if out[i] < 0.0:
        out[i] = 0.0
    np.clip(out, 0.0, 1000.0, out=out)
    return out


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Replace the sequence by (seq + reversed(seq))/2."""
    return 0.5 * (seq + seq[::-1])


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 500,
                     **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds).  The function returns the best sequence
        discovered before the deadline.

    Returns
    -------
    list[float]
        The best sequence (plain Python list) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # safety margin for the outer harness
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # 1️⃣  Random parametric sweep (≈15 % of the total budget)
    param_budget = 0.15 * budget_s
    param_deadline = start_time + param_budget

    best_seq = None
    best_score = float("inf")

    # candidate lengths (odd/even does not matter)
    length_options = [800, 1200, 1600, 2000, 2400, 2800, 3200]

    # parameter ranges
    p_range = (0.30, 0.80)          # power‑law exponent
    shift_range = (0.0, 8.0)        # additive shift inside the power
    lam_range = (0.0, 0.0015)       # exponential damping

    while time.monotonic() < param_deadline:
        n = int(rng.choice(length_options))
        p = rng.uniform(*p_range)
        shift = rng.uniform(*shift_range)
        lam = rng.uniform(*lam_range)

        seq = _shape_power_law(n, p, shift)
        if lam > 0.0:
            seq *= np.exp(-lam * np.arange(n))
        np.clip(seq, 0.0, 1000.0, out=seq)

        s = seq.sum()
        if s <= 0.0:
            continue
        seq /= s                          # normalise to sum = 1
        score = _evaluate_fast(seq)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[param] n={n:4d} p={p:.4f} shift={shift:.2f} lam={lam:.6f} → {score:.6f}")

    # ------------------------------------------------------------------
    # If the sweep failed (unlikely), fall back to a known construction.
    if best_seq is None:
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
                best_seq = base.copy()
                best_seq /= best_seq.sum()
                best_score = _evaluate_fast(best_seq)
                print(f"[fallback] using provided height_sequence_1, score={best_score:.6f}")
        except Exception:
            pass

    if best_seq is None:
        # Classic analytic shape a_i = 1/√(i+1)
        n = 2500
        idx = np.arange(1, n + 1, dtype=np.float64)
        best_seq = 1.0 / np.sqrt(idx)
        best_seq /= best_seq.sum()
        best_score = _evaluate_fast(best_seq)
        print(f"[fallback] analytic shape, n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Simulated‑annealing hill‑climber (the remaining budget)
    cur_seq = best_seq.copy()
    cur_score = best_score
    best_seq = cur_seq.copy()
    best_score = cur_score

    # annealing schedule (linear decay)
    max_temp = 0.025
    min_temp = 0.0002
    sigma0 = 0.25

    # logging
    next_log = time.monotonic() + 5.0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        progress = elapsed / budget_s
        temp = max_temp * (1.0 - progress) + min_temp * progress
        sigma = sigma0 * (1.0 - progress) + 0.02 * progress

        # choose a perturbation
        r = rng.random()
        if r < 0.45:
            cand = _block_multiply(cur_seq, rng, sigma)
        elif r < 0.80:
            cand = _rightward_transfer(cur_seq, rng, max_frac=0.30)
        elif r < 0.95:
            cand = _transfer_mass(cur_seq, rng, max_frac=0.30)
        else:
            cand = cur_seq.copy()
            if rng.random() < 0.35:
                cand = _symmetrize(cand)

        # re‑normalise (the objective is scale‑invariant)
        total = cand.sum()
        if total <= 1e-12:
            continue
        cand /= total

        cand_score = _evaluate_fast(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(temp, 1e-12)):

            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                print(f"[{elapsed:6.1f}s] new best {best_score:.6f}")

            next_log = time.monotonic() + 5.0

    print(f"final best score {best_score:.6f}")
    # return plain Python list as required by the external harness
    return best_seq.tolist()
