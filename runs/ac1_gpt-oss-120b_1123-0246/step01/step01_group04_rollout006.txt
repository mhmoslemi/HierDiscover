def _expgrad_step(seq, step, rng):

    conv = _conv(seq)   # length 2n-1

    kmax = int(np.argmax(conv))

    # compute gradient weighting vector
    n = seq.size

    indices = np.arange(n)

    partner = kmax - indices

    mask = (partner >= 0) & (partner < n)

    g = np.zeros_like(seq)

    g[mask] = 2.0 * seq[partner[mask]]

    # apply exponentiated update
    factor = np.exp(-step * g)

    new_seq = seq * factor

    # optionally add small noise to avoid stagnation
    if rng.random() < 0.2:
        new_seq *= np.exp(rng.normal(0.0, 0.02, size=n))

    # renorm
    s = new_seq.sum()
    if s > 0:
        new_seq /= s
    else:
        # fallback uniform
        new_seq = np.full(n, 1.0/n)

    return new_seq

"""
propose_candidate
=================

Search for a non‑negative step‑function (a list of heights) that minimizes the
`evaluate_sequence` objective:

    score = 2 * n * max(conv(a,a)) / (sum(a)^2)

The algorithm combines three ideas:

1. **Rich parametric families** – random draws from power‑law,
   exponential, sinusoidal, Hann/Blackman windows and beta‑shaped
   distributions give a diverse set of starting shapes.

2. **Exponentiated‑gradient move** – for the current best sequence we locate
   the convolution index where the maximum occurs, compute the gradient of
   that maximum with respect to the coefficients and perform an
   exponentiated‑gradient step (keeps positivity and stays on the simplex).

3. **Local stochastic perturbations** – block scaling, small mass swaps and
   occasional uniform mixing help to escape plateaus.

A simple schedule adapts the gradient step size and the move probabilities
as the time budget (default 1000 s) is consumed.  The best sequence found
within the allotted time is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (fast, FFT‑based)
# ----------------------------------------------------------------------


def _full_convolution(seq: np.ndarray) -> np.ndarray:
    """Return the full discrete convolution seq * seq using an FFT."""
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    return conv.real


def _max_convolution(seq: np.ndarray) -> float:
    """Maximum entry of the full convolution of ``seq`` with itself."""
    conv = _full_convolution(seq)
    return float(conv.max())


def _fast_score(seq: np.ndarray) -> float:
    """Score computed without the safety checks of `evaluate_sequence`."""
    s = seq.sum()
    if s <= 0.0:
        return float("inf")
    max_conv = _max_convolution(seq)
    return 2.0 * seq.size * max_conv / (s * s)


# ----------------------------------------------------------------------
# Parameterised families of candidate sequences
# ----------------------------------------------------------------------


def _power_law_seq(n: int, exponent: float = 0.5, shift: float = 0.0) -> np.ndarray:
    """a_i ∝ (i + shift + 1)^{-exponent}, normalised to unit L1."""
    idx = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power(idx + shift + 1.0, exponent)
    vals /= vals.sum()
    return vals


def _exponential_seq(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * idx)
    vals /= vals.sum()
    return vals


def _sine_window_seq(n: int) -> np.ndarray:
    """a_i ∝ sin(π (i+0.5) / n)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * idx / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _hann_window_seq(n: int) -> np.ndarray:
    """Standard Hann window (raised‑cosine)."""
    idx = np.arange(n, dtype=np.float64)
    vals = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
    vals /= vals.sum()
    return vals


def _blackman_window_seq(n: int) -> np.ndarray:
    """Standard Blackman window."""
    a0, a1, a2 = 0.42, 0.5, 0.08
    idx = np.arange(n, dtype=np.float64)
    vals = (
        a0
        - a1 * np.cos(2.0 * np.pi * idx / (n - 1))
        + a2 * np.cos(4.0 * np.pi * idx / (n - 1))
    )
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped density on {0,…,n‑1} (a_i ∝ (i+½)^{α‑1}(n‑½‑i)^{β‑1})."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    vals = np.power(idx, alpha - 1.0) * np.power(n + 0.5 - idx, beta - 1.0)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= s
    return vals


def _random_param_seq(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random candidate from the rich family of shapes."""
    choice = rng.integers(0, 6)

    if choice == 0:  # power‑law
        exponent = rng.uniform(0.2, 1.2)
        shift = rng.exponential(5.0) - 2.0  # can be slightly negative
        seq = _power_law_seq(n, exponent=exponent, shift=shift)

    elif choice == 1:  # exponential
        lam = rng.uniform(0.0005, 0.02)
        seq = _exponential_seq(n, lam)

    elif choice == 2:  # sine window
        seq = _sine_window_seq(n)

    elif choice == 3:  # Hann window
        seq = _hann_window_seq(n)

    elif choice == 4:  # Blackman window
        seq = _blackman_window_seq(n)

    else:  # beta distribution
        alpha = rng.uniform(0.1, 5.0)
        beta = rng.uniform(0.1, 5.0)
        seq = _beta_seq(n, alpha, beta)

    # occasional mixture with the uniform vector
    if rng.random() < 0.35:
        uniform = np.full(n, 1.0 / n)
        alpha_mix = rng.uniform(0.0, 0.4)
        seq = alpha_mix * uniform + (1.0 - alpha_mix) * seq
        seq /= seq.sum()

    return seq


# ----------------------------------------------------------------------
# Local moves
# ----------------------------------------------------------------------


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block (±12 % of the length)."""
    cand = base.copy()
    n = cand.size
    w_low = max(1, n // 200)
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    scale = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= scale

    # occasional symmetrisation (helps for even‑odd interference)
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])

    cand = np.clip(cand, 0.0, 1000.0)
    s = cand.sum()
    if s > 0.0:
        cand /= s
    else:
        cand = np.full(n, 1.0 / n)
    return cand


def _expgrad_step(seq: np.ndarray, step: float, rng: np.random.Generator) -> np.ndarray:
    """
    One exponentiated‑gradient update aimed at reducing the current
    maximal convolution entry.
    """
    conv = _full_convolution(seq)
    kmax = int(np.argmax(conv))

    n = seq.size
    idx = np.arange(n)
    partner = kmax - idx
    mask = (partner >= 0) & (partner < n)

    # gradient of the *maximum* convolution entry (sub‑gradient)
    g = np.zeros_like(seq)
    g[mask] = 2.0 * seq[partner[mask]]

    # exponentiated‑gradient step (always stays non‑negative)
    factor = np.exp(-step * g)
    new_seq = seq * factor

    # tiny multiplicative noise prevents premature stagnation
    if rng.random() < 0.2:
        new_seq *= np.exp(rng.normal(0.0, 0.015, size=n))

    new_seq_sum = new_seq.sum()
    if new_seq_sum > 0.0:
        new_seq /= new_seq_sum
    else:
        new_seq = np.full(n, 1.0 / n)

    return new_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises
    `evaluate_sequence`.  The routine respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine the working length ``n`` – try to reuse a previously
    # found construction if it exists, otherwise fall back to a safe
    # default (8000 entries give enough freedom without being too slow).
    # ------------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        if base_seq.ndim != 1 or base_seq.size == 0:
            base_seq = None
    except NameError:
        base_seq = None

    min_n = 8000
    if base_seq is not None and base_seq.size >= min_n:
        n = base_seq.size
    else:
        n = min_n

    # ------------------------------------------------------------------
    # Initialise the current best candidate
    # ------------------------------------------------------------------
    # start from a smooth power‑law – this is known to be a good baseline
    current = _power_law_seq(n, exponent=0.5, shift=0.0)

    # If we have a previous construction of the same length, prefer it
    if base_seq is not None and base_seq.size == n:
        candidate = np.clip(base_seq, 0.0, 1000.0)
        s = candidate.sum()
        if s > 0.0:
            candidate /= s
        else:
            candidate = np.full(n, 1.0 / n)

        cand_score = float(evaluate_sequence(candidate.tolist()))
        cur_score = float(evaluate_sequence(current.tolist()))
        if cand_score < cur_score:
            current = candidate.copy()

    # Normalise (scale does not affect the objective)
    current = np.clip(current, 0.0, 1000.0)
    current_sum = current.sum()
    if current_sum == 0.0:
        current = np.full(n, 1.0 / n)
    else:
        current /= current_sum

    best = current.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    iteration = 0
    step_size = 0.22                     # initial gradient step
    move_weights = {
        "expgrad": 0.35,
        "block":   0.25,
        "param":   0.25,
        "swap":    0.15,
    }
    move_types = list(move_weights.keys())
    move_probs = np.array(list(move_weights.values()), dtype=np.float64)
    move_probs /= move_probs.sum()

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        move = rng.choice(move_types, p=move_probs)

        # --------------------------------------------------------------
        # Produce a candidate according to the chosen move
        # --------------------------------------------------------------
        if move == "expgrad":

            i, j = rng.choice(n, size=2, replace=False)
            if cand[i] > 0.0:
                eps = 0.1 * cand[i] * rng.uniform(0.2, 0.8)
                cand[i] = max(cand[i] - eps, 0.0)
                cand[j] += eps
                s = cand.sum()
                if s > 0.0:
                    cand /= s
                else:
                    cand = np.full(n, 1.0 / n)

        else:  # param
            cand = _random_param_seq(n, rng)

        # --------------------------------------------------------------
        # Fast pre‑screen (no safety checks)
        # --------------------------------------------------------------
        fast_sc = _fast_score(cand)
        if fast_sc >= best_score:
            # not promising – skip the expensive official evaluation
            continue

        # --------------------------------------------------------------
        # Authoritative evaluation
        # --------------------------------------------------------------
        try:
            true_sc = float(evaluate_sequence(cand.tolist()))
        except Exception:          # pragma: no cover – safety net
            continue

        if np.isfinite(true_sc) and true_sc < best_score:
            best = cand.copy()
            best_score = true_sc

        # --------------------------------------------------------------
        # Simple schedules for step‑size & move probabilities
        # --------------------------------------------------------------

    # ------------------------------------------------------------------
