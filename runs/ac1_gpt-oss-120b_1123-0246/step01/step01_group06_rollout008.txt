"""
propose_candidate
-----------------
Greedy + simulated‑annealing search for a low‑score step function.

The algorithm works in three stages:

1. **Coarse shape search** –  a family of power‑law sequences
   a_i = (i + offset)^{-p}  (offset defaults to 1, i.e. (i+1)^{-p})
   is scanned over a fine grid of exponents p∈[0.40,0.80] and a few
   small offsets.  The best shape becomes the starting point.

2. **Local optimisation** –  a simulated‑annealing loop tries three
   simple perturbations of the current sequence:
       • block multiplicative perturbation (as used in the reference)
       • pairwise mass transfer between two positions
       • convex combination with a freshly generated power‑law shape
   Improvements are always kept; worsening moves may be accepted with
   a temperature that linearly cools to zero, allowing the search to
   escape shallow local minima.

3. **Return** the best sequence seen before the time budget expires.

All sequences are clipped to the limits enforced by ``evaluate_sequence``:
0 ≤ a_i ≤ 1000, and the objective value is invariant under a global
scaling, so we simply normalise each candidate to have a maximum of 1000.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: block perturbation (identical to the reference implementation)
# ----------------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply one multiplicative log‑normal perturbation to a random block of `base`.
    The block width is chosen between n/200 and n/12 (rounded to at least 1).
    With probability 0.35 the result is symmetrised.
    """
    candidate = base.copy()
    n = candidate.size
    # choose a width – occasionally allow a very small block for fine tuning
    if rng.random() < 0.2:
        width = int(rng.integers(1, max(2, n // 400)))
    else:
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # log‑normal multiplier
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetrisation (often beneficial for auto‑correlation tasks)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # enforce hard bounds of evaluate_sequence
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Helper: pairwise mass transfer
# ----------------------------------------------------------------------
def pairwise_transfer(base: np.ndarray, rng: np.random.Generator,
                      sigma_frac: float = 0.02) -> np.ndarray:
    """
    Move a small random amount of mass from one index to another.
    The transferred amount is a Gaussian fraction of the source entry
    (scaled by `sigma_frac`).  Both entries stay inside [0,1000].
    """
    cand = base.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1

    # draw transferred amount – keep it small and respect positivity
    delta = float(rng.normal(0.0, sigma_frac)) * cand[i]
    if delta > 0.0:
        delta = min(delta, cand[i])
        cand[i] -= delta
        cand[j] = min(cand[j] + delta, 1000.0)
    else:
        delta = -delta
        delta = min(delta, cand[j])
        cand[j] -= delta
        cand[i] = min(cand[i] + delta, 1000.0)

    return cand

# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 950.0, seed: int = 43, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The function runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin

    # ------------------------------------------------------------------
    # 1. Determine sequence length (use the provided reference if any)
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        if n == 0:
            raise Exception
    except Exception:
        n = 8192                     # a safe default if the global is missing
    print(f"[info] Working with sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2. Coarse grid search over power‑law exponents (and a few offsets)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    best_p = None
    best_offset = 1.0   # default offset corresponds to (i+1)^{-p}

    # exponents densely sampled around the region that previously worked well
    p_grid = np.concatenate([
        np.linspace(0.40, 0.55, 12, endpoint=False),
        np.linspace(0.55, 0.75, 20, endpoint=False),
        np.linspace(0.75, 0.90, 8)
    ])

    # small set of offsets – 1.0 is the original (i+1) variant
    offset_grid = np.concatenate([
        np.array([1.0]),
        np.geomspace(0.5, 3.0, 5)   # a few fractional offsets to tilt the shape
    ])

    for off in offset_grid:
        for p in p_grid:
            # build the raw power‑law shape
            idx = np.arange(n, dtype=np.float64) + off            # i + offset
            seq = idx ** (-p)

            # scale so that the largest entry equals 1000 (scaling does not change the score)
            seq = seq * (1000.0 / seq.max())

            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_p = p
                best_offset = off
                print(f"[grid] p={p:.3f}, off={off:.2f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # 3. Fallback – uniform sequence (score = 2) if grid failed
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 500.0, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] started from uniform sequence (score = 2.0)")

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing hill‑climbing
    # ------------------------------------------------------------------
    # current point that we move around (starts at the best found so far)
    cur_seq = best_seq.copy()
    cur_score = best_score

    # initial temperature – a small fraction of the current score
    init_temp = max(0.01, best_score * 0.02)

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # ------------------------------------------------------------------
        #   Choose a perturbation type
        # ------------------------------------------------------------------
        r = rng.random()
        if r < 0.45:
            cand = make_block_perturbation(cur_seq, rng)
        elif r < 0.70:
            cand = pairwise_transfer(cur_seq, rng, sigma_frac=0.025)
        else:
            # convex combination with a fresh power‑law shape
            p_new = float(rng.uniform(0.45, 0.80))
            off_new = float(rng.choice(offset_grid))
            shape = ((np.arange(n, dtype=np.float64) + off_new) ** (-p_new))
            shape = shape * (1000.0 / shape.max())
            t = float(rng.random() ** 2)          # bias towards small steps
            cand = (1 - t) * cur_seq + t * shape
            cand = np.clip(cand, 0.0, 1000.0)

        # ------------------------------------------------------------------
        #   Evaluate and possibly accept
        # ------------------------------------------------------------------
        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
            continue

        # linear cooling schedule
        elapsed = time.monotonic() - start_time
        progress = elapsed / (deadline - start_time)
        temperature = init_temp * (1.0 - progress)

        # acceptance rule (Metropolis)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability of accepting a worse move
            prob = np.exp(-(cand_score - cur_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_seq = cand
            cur_score = cand_score

        # ------------------------------------------------------------------
        #   Keep the best ever seen
        # ------------------------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[improve] iter={iteration:5d}  best={best_score:.6f}")

    print(f"[done] final best score = {best_score:.6f}")
    return best_seq.tolist()
