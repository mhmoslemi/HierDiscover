import cvxpy as cp

def solve_lp(f, M):
    n = f.size
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = f
    # define variable

    # if not solved, fallback to zero vector
    if g.value is None:
        return None
    else:
        return np.array(g.value, dtype=np.float64)

def perturb(seq, rng):
    cand = seq.copy()
    n = len(cand)
    # pick random block width between 1% and 10% of n
    min_width = max(1, n // 100)
    max_width = max(2, n // 10)
    width = rng.integers(min_width, max_width+1)
    start = rng.integers(0, n - width + 1)
    # apply lognormal multiplicative factor
    factor = np.exp(rng.normal(loc=0.0, scale=0.05))
    cand[start:start+width] *= factor
    # optional global scaling to maintain sum sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    cur_sum = cand.sum()
    cand = cand * (target_sum / cur_sum)
    # optionally clip
    cand = np.clip(cand, 0.0, 1000.0)
    # optional symmetrization
    if rng.random() < 0.1:
        cand = 0.5*(cand + cand[::-1])
    return cand

"""
Propose a step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a))²

The algorithm is a two‑phase optimiser.

1) **Alternating linear programme (ALP) phase**  
   – Normalise the current vector `f` so that Σf = √(2n).  
   – Let M = max(conv(f,f)).  Solve the LP  
        maximise Σg    subject to   g ≥ 0,  conv(f,g) ≤ M.  
     The LP is tiny (≈n variables, 2n−1 linear constraints) and can be
     solved with CVXPY/ECOS.  The optimum `g` is then rescaled to the same
     normalisation as `f`.  A convex combination `f ← (1‑t)·f + t·g`
     (with a modest step size `t`) strictly reduces `M`.  Repeating this
     step drives the maximal self‑convolution toward the conjectured
     optimum (~1.5).

2) **Fine‑grained stochastic refinement**  
   After the LP phase terminates (or time runs out) a cheap
   block‑wise multiplicative perturbation hill‑climber runs for the
   remaining budget, keeping the best sequence found.

Both phases respect the problem’s domain restrictions
(`0 ≤ a_i ≤ 1000`).  The final vector is returned as a plain Python list.
"""
import time
import numpy as np

# Try to import CVXPY; if unavailable we will fall back to the pure
# stochastic hill‑climber used previously.
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:          # pragma: no cover
    _HAS_CVX = False


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A such that A @ g = conv(f, g) (full convolution).
    Shape of A is (2·n‑1, n).
    """
    n = f.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    # Each column i contributes f[i] on the rows i … i+n‑1.
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear programme:
        maximise Σ g_i
        subject to g_i ≥ 0 and conv(f, g) ≤ M (pointwise).
    Returns the optimal vector g, or None if the solver fails.
    """
    n = f.size
    A = _build_conv_matrix(f)

    # CVXPY variables / problem
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # Solver options – ECOS is a fast LP solver available in the default CVXPY build.
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:      # pragma: no cover
        return None

    if g.value is None:
        return None
    return np.asarray(g.value, dtype=np.float64).reshape(-1)


def _conv_max(seq: np.ndarray) -> float:
    """Return max value of the full auto‑convolution of `seq`."""
    return np.convolve(seq, seq).max()


def _normalise_to_target(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale `seq` so that its L¹‑norm equals `target_sum`."""
    cur = seq.sum()
    if cur == 0:
        return seq
    return seq * (target_sum / cur)


def _perturb_block(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a cheap multiplicative perturbation of `seq`.
    A random contiguous block is multiplied by a log‑normal factor;
    occasional symmetrisation is applied.
    The result is re‑normalised to the fixed target sum.
    """
    n = seq.size
    cand = seq.copy()

    # Choose a block width between 1 % and 10 % of the vector.
    lo = max(1, n // 100)
    hi = max(2, n // 10)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = np.exp(rng.normal(0.0, 0.07))   # modest log‑normal noise
    cand[start:start + width] *= factor

    # Occasionally enforce a mirror symmetry – this often flattens the peak.
    if rng.random() < 0.12:
        cand = 0.5 * (cand + cand[::-1])

    # Re‑normalise to the fixed L¹‑norm required by the evaluation.
    target = np.sqrt(2.0 * n)
    cand = _normalise_to_target(cand, target)

    # Clip to the admissible range (the optimiser never needs the 1000‑cap,
    # but we obey the external checks).
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed: int = 0,
                     budget_s: float = 900,
                     **kwargs) -> list[float]:
    """
    Generate a candidate step‑function.

    The routine prefers an LP‑based alternating optimisation; if CVXPY
    is unavailable it falls back to a pure stochastic hill‑climber.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)   # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  800 is a sweet‑spot: large enough to beat
    # the power‑law bound, yet small enough for many LP solves within the budget.
    # If a historic solution `height_sequence_1` exists and is not too big,
    # we reuse its length (up to 1500) to leverage prior work.
    try:
        legacy = np.asarray(height_sequence_1, dtype=np.float64)
        if legacy.ndim == 1 and legacy.size >= 400 and legacy.size <= 1500:
            n = int(legacy.size)
        else:
            n = 800
    except Exception:
        n = 800

    # ------------------------------------------------------------------
    # Initialise a smooth decreasing shape (1/√i) and normalise to
    # the canonical sum Σa = √(2·n).  This guarantees that the
    # evaluation score equals the maximal auto‑convolution.
    idx = np.arange(1, n + 1, dtype=np.float64)
    f = 1.0 / np.sqrt(idx)               # power‑law seed
    f = np.clip(f, 0.0, 1000.0)
    target_sum = np.sqrt(2.0 * n)
    f = _normalise_to_target(f, target_sum)

    best_f = f.copy()
    best_M = _conv_max(best_f)
    print(f"[INIT] n={n}, initial max‑conv = {best_M:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Alternating Linear Programme (if CVXPY is available)
    if _HAS_CVX:
        iter_counter = 0
        while time.monotonic() < deadline:
            iter_counter += 1
            M = _conv_max(f)                # current bound
            # Solve the LP: maximise Σg under conv(f,g) ≤ M.
            g_raw = _solve_lp_for_g(f, M)
            if g_raw is None:
                print("[LP] Solver failed – exiting ALP loop.")
                break
            sum_g_raw = g_raw.sum()
            if sum_g_raw < 1e-12:
                print("[LP] Trivial solution – exiting ALP loop.")
                break

            # Rescale to the fixed L¹‑norm.
            g = _normalise_to_target(g_raw, target_sum)

            # Try a few step‑sizes; the first one that yields a strict improvement
            # is accepted.
            improved = False
            for t in (0.5, 0.3, 0.15, 0.07, 0.03):
                cand = (1.0 - t) * f + t * g
                # occasional symmetrisation (helps with edge effects)
                if rng.random() < 0.15:
                    cand = 0.5 * (cand + cand[::-1])
                cand_M = _conv_max(cand)
                if cand_M + 1e-12 < M:            # strict improvement
                    f = cand
                    M = cand_M
                    improved = True
                    if M < best_M:
                        best_f = f.copy()
                        best_M = M
                    print(f"[LP] iter {iter_counter:3d}, t={t:.2f}, M={M:.6f}")
                    break
            if not improved:
                # No improvement with any step size – stop the ALP loop.
                print("[LP] No improving step found – terminating ALP phase.")
                break
    else:
        print("[INFO] CVXPY not available – skipping LP phase.")

    # ------------------------------------------------------------------
    # Phase 2 – Stochastic block‑wise refinement for the remaining time.
    # This is a cheap hill‑climber that can still shave a few 1e‑4 off the bound.
    while time.monotonic() < deadline:
        cand = _perturb_block(best_f, rng)
        cand_M = _conv_max(cand)
        if cand_M + 1e-12 < best_M:
            best_f = cand
            best_M = cand_M
            print(f"[HC] new best M = {best_M:.6f}")

    # Final sanity: clip to the allowed range before returning.
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    return best_f.tolist()
