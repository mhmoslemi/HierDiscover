Strategy
--------
1. Use the length of the supplied `height_sequence_1` (fallback 256).
2. Scan pure power‑law shapes a_i = 1000·(i+1)^{-p} – this reproduces the

3. Generate *piecewise* candidates: a descending power‑law for the first
   `m` positions, followed by a flat tail of constant height `c`.
   For a given exponent `p` and cut‑off `m` we compute `c`

4. Keep the best few analytic candidates and evaluate them with the real

5. Refine the current best sequence with a fast block‑perturbation

The combination of an analytically‑guided shape and local hill‑climbing

# Helper: a single non‑negative multiplicative block perturbation
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """One O(n) multiplicative log‑normal perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width is chosen log‑uniformly between n/200 and n/12 (≥1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation on the block
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation (often helpful for autocorrelation)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # enforce the hard limits of evaluate_sequence
    return np.clip(candidate, 0.0, 1000.0)

            total_sum = early_sum + (n - m) * c
            approx_ratio = 2.0 * n * (a0 ** 2) / (total_sum ** 2)
            analytic_candidates.append((approx_ratio, p, m, c))

    # keep the best few based on the cheap approximation
    analytic_candidates.sort(key=lambda x: x[0])
    top_analytic = analytic_candidates[:30]

    for approx_ratio, p, m, c in top_analytic:
        # build the concrete sequence
        early = (np.arange(1, m + 1, dtype=np.float64) ** (-p)) * a0
        tail = np.full(n - m, c, dtype=np.float64)
        seq = np.concatenate([early, tail])
        # evaluate the true objective
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[piecewise] p={p:.3f} m={m} c={c:.2f} "
                  f"approx={approx_ratio:.4f} → score={score:.6f}")
        if time.monotonic() > deadline:
            break

    # ------------------------------------------------------------------
    # 4. Local refinement with block perturbations (hill‑climbing)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence (score will be 2.0)
        best_seq = np.full(n, 500.0, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] started from uniform sequence")

    print(f"[info] Starting local search, best score so far = {best_score:.6f}")

    while time.monotonic() < deadline:
        cand = make_block_perturbation(best_seq, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            print(f"[improve] new best = {best_score:.6f}")

    print(f"[done] final best score = {best_score:.6f}")
    return best_seq.tolist()
