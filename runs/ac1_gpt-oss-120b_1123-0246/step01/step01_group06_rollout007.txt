import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level, no closures)
# ----------------------------------------------------------------------
def _fft_convolution(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of `seq` with itself using a power‑of‑two FFT."""
    n = seq.size
    fft_len = 1
    while fft_len < 2 * n:
        fft_len <<= 1
    F = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(F * F, fft_len)[: 2 * n - 1]
    return conv


def _score_and_conv(seq: np.ndarray):
    """Return (score, convolution) for a clipped non‑negative sequence."""
    s = seq.sum()
    if s < 0.01:
        return float("inf"), None
    conv = _fft_convolution(seq)
    max_b = conv.max()
    n = seq.size
    score = float(2.0 * n * max_b / (s * s))
    return score, conv


def _make_power_law(n, p, offset=0.0, lam=0.0, cutoff_frac=1.0):
    """Generate a clipped power‑law / exponential shape."""
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0 + offset) ** (-p) * np.exp(-lam * i)
    # optional hard cut‑off of the tail
    if cutoff_frac < 1.0:
        cut = int(cutoff_frac * n)
        seq[cut:] = 0.0
    if seq.max() == 0:
        return seq
    seq = seq * (1000.0 / seq.max())
    return np.clip(seq, 0.0, 1000.0)


def _pairwise_transfer(seq, max_idx, rng):
    """
    Propose a new sequence by moving a small amount of mass from an index
    that contributes heavily to the current maximal convolution entry to an
    index that does not participate in that entry.
    """
    n = seq.size
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    idxs = np.arange(i_start, i_end + 1, dtype=int)

    # contribution of each i to the peak b_{max_idx}
    contrib = seq[idxs] * seq[max_idx - idxs]
    if contrib.sum() > 0:
        probs = contrib / contrib.sum()
        i_sel = rng.choice(idxs, p=probs)
    else:
        i_sel = rng.choice(idxs)

    # pick a destination index outside the contributing window
    for _ in range(5):
        j_sel = rng.integers(0, n)
        if j_sel < i_start or j_sel > i_end:
            break
    else:
        j_sel = rng.integers(0, n)

    max_transfer = min(seq[i_sel], 1000.0 - seq[j_sel])
    if max_transfer <= 1e-8:
        return None, None

    # move a random fraction (up to 50 %) of the transferable amount
    delta = max_transfer * rng.random() * 0.5
    cand = seq.copy()
    cand[i_sel] -= delta
    cand[j_sel] += delta
    cand = np.clip(cand, 0.0, 1000.0)
    return cand, (i_sel, j_sel, delta)


def _block_perturb(seq, rng):
    """Same perturbation style as the reference solution (log‑normal block)."""
    n = seq.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)

    cand = seq.copy()
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Multi‑phase optimiser for the step‑function evaluation problem.

    Phases
    ------
    1.  Exhaustive grid search over a *parameterised* family of
        power‑law / exponential shapes (exponent, offset, decay, tail cut‑off).
    2.  Targeted pair‑wise mass transfers that reduce the contribution
        to the currently maximal convolution entry.
    3.  General random block perturbations (as used in the reference code).

    The algorithm keeps the best sequence seen so far and stops when the
    supplied time budget expires (default 500 s).  All numbers are kept
    inside the [0, 1000] range; the score is computed analytically from
    the convolution (no call to ``evaluate_sequence`` is required during
    the search, only for the final answer verification).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Initialise best candidate (use the provided construction if possible)
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        tmp = np.asarray(height_sequence_1, dtype=np.float64)
        tmp = np.clip(tmp, 0.0, 1000.0)
        if tmp.size == n:
            score, _ = _score_and_conv(tmp)
            best_seq = tmp.copy()
            best_score = score
            print(f"[init] start from height_sequence_1  → score = {score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – systematic exploration of a compact parametric family
    # --------------------------------------------------------------
    exp_grid = np.linspace(0.35, 0.85, 21)        # exponent p
    offset_grid = [0.0, 0.5, 1.0, 2.0, 4.0]       # additive shift inside the power law
    lam_grid = np.linspace(0.0, 0.008, 9)         # exponential decay factor
    cutoff_grid = [1.0, 0.9, 0.8, 0.7, 0.6, 0.5]  # tail truncation (1.0 → no truncation)

    # Randomise order to avoid systematic bias and to respect the time budget
    combos = [(p, off, lam, cut) for p in exp_grid
                                 for off in offset_grid
                                 for lam in lam_grid
                                 for cut in cutoff_grid]
    rng.shuffle(combos)

    for p, off, lam, cut in combos:
        if time.monotonic() >= deadline:
            break
        seq = _make_power_law(n, p, offset=off, lam=lam, cutoff_frac=cut)
        if seq.max() == 0:
            continue
        score, _ = _score_and_conv(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[grid] p={p: .3f} off={off: .1f} lam={lam: .4f} cut={cut:.1f} → {score:.6f}")

    # a few pure exponentials (they sometimes beat the mixed family)
    for lam in np.linspace(0.0005, 0.005, 10):
        if time.monotonic() >= deadline:
            break
        seq = np.exp(-lam * np.arange(n, dtype=np.float64))
        seq = seq * (1000.0 / seq.max())
        seq = np.clip(seq, 0.0, 1000.0)
        score, _ = _score_and_conv(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[exp]  lam={lam: .5f} → {score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – targeted pairwise mass transfers
    # --------------------------------------------------------------
    iter_cnt = 0
    while time.monotonic() < deadline:
        # obtain current convolution and the index of its maximum
        cur_score, conv = _score_and_conv(best_seq)
        if conv is None:               # should never happen
            break
        k_max = int(np.argmax(conv))

        cand, info = _pairwise_transfer(best_seq, k_max, rng)
        if cand is None:
            continue

        cand_score, _ = _score_and_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            i_sel, j_sel, delta = info
            print(f"[pair] iter {iter_cnt:5d}  i={i_sel}→j={j_sel} Δ={delta:.2f}  score={cand_score:.6f}")
        iter_cnt += 1

    # --------------------------------------------------------------
    # Phase 3 – random block perturbations (global exploration)
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = _block_perturb(best_seq, rng)
        cand_score, _ = _score_and_conv(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[block] improved score = {cand_score:.6f}")

    print(f"[done] final best score = {best_score:.6f}")
    return best_seq.tolist()
