"""
Propose a step‑function (non‑negative list of heights) that improves on the
Catalan construction.  The algorithm is:

1.  Start from the Catalan sequence C_i = binom(2i,i)/4**i (truncated to a
    manageable length, default 800).  If a global variable ``height_sequence_1``
    exists it is truncated and used instead.

2.  Repeatedly try to increase the ℓ1‑norm (the sum of the heights) while
    keeping the maximal self‑convolution value M = max(a*a) unchanged.
    For a fixed sequence ``a`` we solve the *linear* program

        maximise   Σ g_i
        subject to (a * g)_k ≤ M  for all k,
                   g_i ≥ 0.

    The convolution (a*g) is linear in the unknown vector ``g``, therefore a
    standard LP suffices.  The LP is solved with ``scipy.optimize.linprog``.
    The solution ``g`` has a larger sum than ``a`` but does not increase the
    bound ``M`` when mixed with ``a``.

3.  Mix the current sequence and the LP solution:
        a_new = (1‑t)·a + t·g            (t chosen from a short list)
    The quadratic term t²·(g*g) is harmless for the modest step sizes we use,
    so the maximal convolution stays ≤ M and the score 2·n·M/Σa² strictly
    decreases.

4.  If the LP does not give an improvement we fall back to a cheap
    multiplicative random perturbation (log‑normal noise) and keep the best
    candidate seen.

The loop runs until the allotted time (≈ 950 s) is exhausted.  The best
sequence found is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Catalan (binomial) coefficients
# ----------------------------------------------------------------------
def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑type numbers C_i = binom(2i,i)/4**i (i≥0)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence a[i] = a[i‑1] * (2*i‑1)/(2*i)
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(a, 0.0, 1000.0)

    ``evaluate_sequence``.  The routine starts from a Catalan (or global)

    returns the best sequence found before the time budget expires.
    """
    import numpy as np
    import time

    # ------------------------------------------------------------------
    # optional import of a fast LP solver
    # ------------------------------------------------------------------

        base = np.asarray(height_sequence_1, dtype=np.float64)      # type: ignore
        if base.size >= DEFAULT_N:

            n = DEFAULT_N

            a = base.copy()
            n = base.size

    except Exception:
        n = DEFAULT_N
        a = _catalan_sequence(n)

    a = np.clip(a, 0.0, 1000.0)

    # current best
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial score = {best_score:.6f}")

    # 2. iterative improvement
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

        conv_self = np.convolve(best, best)

        # --------------------------------------------------------------
        if _have_lp:

            # For out‑of‑range indices the term is omitted.
            A = np.zeros((rows, n), dtype=np.float64)
