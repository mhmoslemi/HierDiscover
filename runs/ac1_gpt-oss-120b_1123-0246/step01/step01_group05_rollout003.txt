def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid optimization for the step‑function attack.

The algorithm works in three stages:

1. **Coarse grid search** – try several families of analytic
   sequences (simple power‑law and a symmetric U‑shaped power‑law) and
   keep the one with the smallest `evaluate_sequence` score.

2. **Projected sub‑gradient descent** – the objective is
   `max_k (a * a)[k]` (the ∞‑norm of the auto‑convolution).  The
   gradient of the active (max) indices is easily computed:
   for a given k, ∂b_k/∂a_i = a_{k‑i} (and an extra a_i when i=k‑i).
   We take a step opposite to this sub‑gradient, then project the
   vector back onto the probability simplex (non‑negative entries with
   unit sum).  The projection guarantees the scale‑invariance of the
   original evaluator.  Step‑size is adapted on the fly and random
   restarts are performed if the step becomes too small.

3. **Final polishing** – a tiny stochastic local‑move phase (the same
   idea as in the reference implementation) is run while time permits
   to squeeze out any remaining improvement.

The whole routine stops before the given time budget (default ~950 s)
and returns the best sequence found.  All numbers are non‑negative and
well below the evaluator’s 1000‑limit; the sum is at least 1, far above
the 0.01 lower bound, so the returned sequence is always valid.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – must be defined at the top level (no closures)
# ----------------------------------------------------------------------


def power_law_sequence(length: int, exponent: float) -> list[float]:
    """
    a_i ∝ (i+1)^(-exponent).  Returned list is normalised to sum=1.
    """
    idx = np.arange(length, dtype=np.float64) + 1.0
    w = np.power(idx, -exponent)
    w /= w.sum()
    return (w * 100.0).tolist()          # any positive factor works


def symmetric_power_law_sequence(length: int, exponent: float) -> list[float]:
    """
    Symmetric U‑shaped version: a_i ∝ (i+1)^(-exponent) + (length-i)^(-exponent).
    Normalised to sum=1.
    """
    i = np.arange(length, dtype=np.float64)
    w = (np.power(i + 1.0, -exponent) + np.power(length - i, -exponent))
    w /= w.sum()
    return (w * 100.0).tolist()


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implementation follows Duchi et al. (2008).
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive → return the uniform vertex.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full (linear) convolution a * a using an FFT of length `fft_len`.
    Returns the array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def max_convolution_info(a: np.ndarray, fft_len: int):
    """
    Returns (max_value, list_of_indices_where_max_is_attained, full_convolution).
    """
    b = conv_full_fft(a, fft_len)
    max_val = b.max()
    max_idxs = np.where(b == max_val)[0]
    return max_val, max_idxs, b


def max_subgradient(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution functional.
    For each active index k we add a_{k-i} to coordinate i,
    and an extra a_i when i == k-i (the diagonal term).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in max_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        g[i_range] += a[j_range]               # a_{k-i} part
        diag = i_range == j_range               # diagonal contributions
        if np.any(diag):
            g[i_range[diag]] += a[i_range[diag]]   # extra a_i on the diagonal
    return g
