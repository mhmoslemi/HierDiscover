import scipy.sparse as sp

def build_constraint_matrix(a):
    n = len(a)
    m = 2*n - 1
    diags = []
    offsets = np.arange(n)   # 0..n-1
    # for each offset d, diagonal values are a[d] repeated length n.
    for d in range(n):
        diags.append(np.full(n, a[d], dtype=np.float64))
    A = sp.diags(diags, offsets, shape=(m, n), format='csr')
    return A

"""
Propose a step‐function that improves on the Catalan construction.
The algorithm starts from the Catalan sequence (a_i = binom(2i,i)/4**i),
then repeatedly solves a linear program

    max   Σ g_i
    s.t.  (a * g)_k ≤ M   (M = max(a*a))
          g_i ≥ 0

to find a direction g that can increase the total sum of the coefficients
without raising the worst self‑convolution value M.  The new sequence is
a convex combination a←(1‑t)·a + t·g; several step‑sizes are tried and the
best improvement is kept.  If the LP gives no gain a cheap random perturbation
is attempted.  The loop runs until the time budget (≈1000 s) expires.
SciPy’s “highs” LP solver is used; a fall‑back to CVXPY + SCS is provided
if SciPy is unavailable.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# optional imports – SciPy is preferred, otherwise fall back to CVXPY
# ----------------------------------------------------------------------
try:
    import scipy.sparse as _sp
    import scipy.optimize as _opt
    _HAS_SCIPY = True
except Exception:          # pragma: no cover
    _HAS_SCIPY = False
    try:
        import cvxpy as cp
        _HAS_CVXPY = True
    except Exception:      # pragma: no cover
        _HAS_CVXPY = False


def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients C_i = binom(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    # Clip only for safety – the true values are ≤1.
    return np.clip(a, 0.0, 1000.0)


def _build_convolution_matrix(a: np.ndarray) -> 'sp.spmatrix':
    """
    Build the (2n‑1)×n matrix A such that (A @ g) = a * g (linear convolution).
    Row k of A contains a_{k‑j} for j=0..n‑1 where the index is in range,
    otherwise 0.  The matrix is lower‑triangular Toeplitz with n non‑zero
    diagonals; each diagonal d (≥0) is filled with a[d].
    """
    n = a.shape[0]
    m = 2 * n - 1
    # each diagonal has exactly n entries (because m‑d ≥ n for d≤n‑1)
    diags = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=int)          # 0,1,…,n‑1
    return _sp.diags(diags, offsets, shape=(m, n), format='csr')


def _solve_lp_scipy(a: np.ndarray, M: float, eps: float = 1e-9):
    """
    Solve  max Σ g_i   s.t.   (a * g)_k ≤ M+eps  ,   g_i ≥ 0
    using SciPy’s HiGHS interior‑point optimizer.
    Returns the optimal g (or None on failure).
    """
    n = a.shape[0]
    # build constraint matrix once per iteration
    A = _build_convolution_matrix(a)
    m = A.shape[0]

    c = -np.ones(n, dtype=np.float64)                 # maximize Σ g_i  →  minimize –Σ g_i
    b_ub = (M + eps) * np.ones(m, dtype=np.float64)   # right‑hand side
    bounds = [(0, None)] * n

    res = _opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                       method='highs', options={'maxiter': 2000})
    if not res.success:
        return None
    return np.maximum(0.0, res.x)


def _solve_lp_cvxpy(a: np.ndarray, M: float, eps: float = 1e-9):
    """
    Same LP as above, solved with CVXPY+SCS (fallback if SciPy missing).
    """
    n = a.shape[0]
    g = cp.Variable(n, nonneg=True)
    conv_expr = cp.conv(a, g)          # length 2n‑1
    constraints = [conv_expr <= M + eps]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver='SCS', eps=1e-6, verbose=False, max_iters=5000)
    except Exception:                  # pragma: no cover
        return None
    if g.value is None:
        return None
    return np.maximum(0.0, np.asarray(g.value, dtype=np.float64))

    """
