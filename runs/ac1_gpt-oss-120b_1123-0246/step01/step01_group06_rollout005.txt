def fast_score(seq):
    arr = np.clip(seq,0,1000.0)
    s = arr.sum()
    if s < 1e-3:
        return np.inf
    # convolve via FFT
    # Next power of 2 for length
    m = len(arr)*2-1
    Nfft = 1 << (m-1).bit_length()
    arr_fft = np.fft.rfft(arr, n=Nfft)
    conv = np.fft.irfft(arr_fft * arr_fft, n=Nfft)[:m]
    max_conv = conv.max()
    return 2.0 * len(arr) * max_conv / (s*s)

"""
Propose a step‑function (non‑negative height sequence) that minimizes the
evaluation metric

    2 * n * max(conv(a,a)) / (sum(a)**2)

The implementation combines three ideas:

1. **Fast surrogate score** – the same objective is computed with an
   FFT‑based convolution which is O(n log n) and cheap enough for thousands
   of evaluations.

2. **Global shape search** – a few parametric families (beta‑like,
   power‑law, exponential, sin‑shaped) are sampled at random.  Each sampled
   shape is normalised (the scale does not affect the score) and its fast
   score is measured.  The best shape from this stage becomes the start
   point for the next phase.

3. **Simulated‑annealing + greedy refinement** – the current best sequence
   is repeatedly perturbed by multiplicative log‑normal block updates
   (the same perturbation used in the reference solution).  A simple
   simulated‑annealing acceptance rule lets the search occasionally accept
   worse candidates, helping it escape local minima.  After the annealing
   phase a final greedy hill‑climber runs until the time budget expires.

All candidates honour the constraints enforced by evaluate_sequence
(≥0, ≤1000).  The final sequence is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast surrogate evaluation (FFT based)
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray) -> float:
    """Return the same objective as evaluate_sequence but using FFT.
    seq is assumed to be a 1‑D numpy array of non‑negative numbers.
    """
    # enforce the same hard limits as evaluate_sequence
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 1e-6:
        return np.inf
    n = seq.shape[0]
    m = 2 * n - 1                     # length of linear convolution
    Nfft = 1 << (m - 1).bit_length()  # next power of two
    fft_seq = np.fft.rfft(seq, n=Nfft)
    conv = np.fft.irfft(fft_seq * fft_seq, n=Nfft)[:m]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Helper: block perturbation (same idea as the reference solution)
# ----------------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a copy of ``base`` with a random contiguous block multiplied
    by a log‑normal factor.  Occasionally the result is symmetrised."""
    cand = base.astype(np.float64, copy=True)
    n = cand.size
    # block width between n/200 and n/12, at least 1
    w_low = max(1, n // 200)
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetrisation (helps many autocorrelation problems)
    if rng.random() < 0.45:  # a bit more aggressive than the reference
        cand = 0.5 * (cand + cand[::-1])
    # hard clipping – prevents evaluate_sequence from discarding the candidate
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Shape generators (parameterised families)
# ----------------------------------------------------------------------
def generate_beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete analogue of a Beta(α,β) density on {1,…,n}."""
    k = np.arange(1, n + 1, dtype=np.float64)
    # (k)^{α-1} * (n-k+1)^{β-1}
    seq = (k ** (alpha - 1.0)) * ((n - k + 1) ** (beta - 1.0))
    return seq


def generate_powerlaw_seq(n: int, p: float, offset: float) -> np.ndarray:
    """a_i = (i+1+offset)^{-p}  (monotonically decreasing)."""
    i = np.arange(1, n + 1, dtype=np.float64) + offset
    return i ** (-p)


def generate_exponential_seq(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)  (fast decay)."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-lam * i)


def generate_sine_seq(n: int, gamma: float) -> np.ndarray:
    """a_i = sin(π·(i+1)/(n+1))^{γ}  – zero at the ends, peaked in the middle."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return np.sin(np.pi * i / (n + 1)) ** gamma


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 4242, budget_s: float = 900.0, **kwargs) -> list[float]:
    """
    Search for a sequence that minimises evaluate_sequence().
    The routine works in three stages:
      1. Sample a few parametric families (beta, power‑law, exponential, sin)
         to obtain a good initial shape.
      2. Simulated‑annealing style exploration with block perturbations.
      3. Greedy hill‑climbing refinement until the time budget expires.
    The total wall‑clock time is bounded by ``budget_s`` seconds.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size (number of steps)
    # --------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))          # type: ignore[name-defined]
        if n == 0:
            raise Exception
    except Exception:
        n = 256
    print(f"[info] Working with n = {n}")

    # --------------------------------------------------------------
    # Baseline: start from the previously discovered construction if
    # it exists – it is often already a good candidate.
    # --------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_seq = np.clip(base_seq, 0.0, 1000.0)
        best_seq = base_seq / base_seq.max() if base_seq.max() > 0 else np.full(n, 1.0)
        best_score = fast_score(best_seq)
        print(f"[init] Score of height_sequence_1 = {best_score:.6f}")
    except Exception:
        best_seq = np.full(n, 1.0, dtype=np.float64)
        best_score = fast_score(best_seq)
        print("[init] No previous construction – start from uniform")

    # --------------------------------------------------------------
    # Phase 1 – global search over a few low‑dimensional families
    # --------------------------------------------------------------
    phase1_time = budget_s * 0.15
    deadline_phase1 = time.monotonic() + phase1_time
    families = ["beta", "power", "exp", "sine"]
    while time.monotonic() < deadline_phase1:
        fam = rng.choice(families)

        if fam == "beta":
            # log‑uniform sampling of α,β in [0.2, 5]
            log_alpha = rng.uniform(np.log10(0.2), np.log10(5.0))
            log_beta = rng.uniform(np.log10(0.2), np.log10(5.0))
            alpha, beta = 10 ** log_alpha, 10 ** log_beta
            seq = generate_beta_seq(n, alpha, beta)

        elif fam == "power":
            p = rng.uniform(0.1, 1.8)
            offset = rng.uniform(0.0, 8.0)
            seq = generate_powerlaw_seq(n, p, offset)

        elif fam == "exp":
            # λ chosen so that the decay length is roughly between 0.2·n and n
            lam_max = np.log(1000.0) / (0.2 * n)
            lam = rng.uniform(0.0, lam_max)
            seq = generate_exponential_seq(n, lam)

        else:  # sine
            gamma = rng.uniform(0.2, 5.0)
            seq = generate_sine_seq(n, gamma)

        # optional symmetrisation – many good shapes are symmetric
        if rng.random() < 0.3:
            seq = 0.5 * (seq + seq[::-1])

        # normalise – scaling does NOT affect the objective
        maxv = seq.max()
        if maxv == 0:
            continue
        seq = seq / maxv

        score = fast_score(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[global] New best via {fam} → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – simulated annealing with block perturbations
    # --------------------------------------------------------------
    phase2_time = budget_s * 0.60
    deadline_phase2 = time.monotonic() + phase2_time
    cur_seq = best_seq.copy()
    cur_score = best_score
    start_sa = time.monotonic()
    # temperature schedule parameters
    T_start = 0.03
    T_end = 5e-5

    while time.monotonic() < deadline_phase2:
        cand = make_block_perturbation(cur_seq, rng)
        cand_score = fast_score(cand)

        if cand_score < cur_score:
            accept = True
        else:
            # linear cooling
            elapsed = time.monotonic() - start_sa
            frac = elapsed / phase2_time
            T = T_start * (1.0 - frac) + T_end * frac
            # Metropolis acceptance
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA]    New global best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – greedy refinement (only accept improvements)
    # --------------------------------------------------------------
    phase3_time = max(0.0, budget_s - (phase1_time + phase2_time) - 2.0)
    deadline_phase3 = time.monotonic() + phase3_time
    print(f"[info] Starting greedy refinement, time left ≈ {phase3_time:.1f}s")
    while time.monotonic() < deadline_phase3:
        cand = make_block_perturbation(best_seq, rng)
        cand_score = fast_score(cand)
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            print(f"[refine] Improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Final verification with the official evaluate_sequence()
    # --------------------------------------------------------------
    try:
        # evaluate_sequence is assumed to be present in the execution env
        final_score = float(evaluate_sequence(best_seq.tolist()))  # type: ignore[name-defined]
        print(f"[final] evaluate_sequence → {final_score:.6f}")
    except Exception as exc:  # safety – if the function is missing, ignore
        print("[final] evaluate_sequence not available:", exc)

    # Return a plain Python list (the grader will convert it as needed)
    return best_seq.tolist()
