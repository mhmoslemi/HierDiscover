"""
Smooth‑max gradient descent for the self‑convolution upper‑bound problem.

The algorithm works on a non‑negative step sequence a of length n.
The objective to minimise is

    score = 2 * n * max_k (a * a)[k] / (sum(a)^2)

which is invariant under scaling of a.  By fixing the total mass to
sqrt(2 n) the score becomes exactly the maximum entry of the
convolution (the “worst lag”).  The optimisation proceeds as follows:

1. Initialise a uniform vector (or reuse the global `height_sequence_1`
   if it exists) and scale it to the required total mass.
2. At each iteration compute the full convolution b = a * a via FFT.
   Instead of using only the single maximal lag we smooth the
   maximum with a soft‑max:
        smooth_max = (1/α)·log Σ_k exp(α·b_k)
   The gradient of this smooth objective is
        ∇ = 2·conv(p, a[::-1])
   where p_k = exp(α·b_k) / Σ_j exp(α·b_j) are the soft‑max weights.
   The gradient is obtained efficiently with another FFT‐based
   convolution.
3. An Adam‑style update with a linearly decaying learning rate moves
   the current point, after which we:
   * clip to non‑negative values,
   * renormalise to keep sum(a)=√(2 n) (the objective is scale‑invariant),
   * clip to the allowed range [0,1000].
4. The best sequence (lowest actual score) seen so far is stored.
   Every few hundred steps a random multiplicative perturbation is tried
   to escape plateaus.
5. The loop runs until the allotted time budget (≈ 1000 s) is exhausted.
   The returned list of floats is the best candidate discovered.

The use of the smooth‑max allows the optimiser to see many large
lags simultaneously, helping it to flatten the whole convolution curve
instead of focusing on a single lag as in a naïve sub‑gradient method.
"""

import time
import numpy as np


def _conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of real vectors x and y using FFT of length L."""
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)
    return conv[: len(x) + len(y) - 1]


def propose_candidate(seed: int = 42, budget_s: float = 900.0, **kwargs):
    """
    Search for a low‑score height sequence within the allotted time.
    Returns a list of floats (the best sequence found).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose length and initialise the sequence
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = base.size
    except NameError:
        # fall back to a moderate length if no previous construction exists
        n = 2000
        base = None

    # enforce the invariant sum(a) = sqrt(2*n)
    target_sum = np.sqrt(2.0 * n)

    if base is None or base.ndim != 1 or base.size != n or not np.isfinite(base).all():
        a = np.full(n, target_sum / n, dtype=np.float64)   # uniform start
    else:
        a = base.astype(np.float64)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s == 0.0:
            a[:] = target_sum / n
        else:
            a *= (target_sum / s)

    # ------------------------------------------------------------------
    # 2. Pre‑compute FFT length (must accommodate the longest linear conv)
    # ------------------------------------------------------------------
    L = 1 << ((3 * n - 2).bit_length())   # power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    # 3. Optimisation state
    # ------------------------------------------------------------------
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # smoothing parameter for the soft‑max
    alpha = 1.0
    alpha_max = 150.0
    alpha_inc = 1.001  # geometric growth

    # learning‑rate schedule (initial value relative to typical entry size)
    init_lr = 0.2 * target_sum
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # tiny safety margin
    iteration = 0
    best = a.copy()
    # initial exact score (max entry of convolution)
    b = _conv_fft(a, a, L)
    best_score = float(np.max(b))   # equal to evaluate_sequence(..) for our scaling
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- 4.1  Compute convolution and soft‑max weights ----
        b = _conv_fft(a, a, L)                     # length 2n‑1
        max_b = b.max()

        # soft‑max (stable)
        shifted = b - max_b
        exp_vals = np.exp(alpha * shifted)
        sum_exp = exp_vals.sum()
        p = exp_vals / sum_exp                     # probabilities, sum to 1

        # ---- 4.2  Gradient of smooth‑max ----
        a_rev = a[::-1]
        grad_full = _conv_fft(p, a_rev, L)        # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]  # slice of length n

        # ---- 4.3  Adam update with decaying step size ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # linear decay of the base learning rate
        elapsed = time.monotonic() - start_time
        decay = max(0.01, 1.0 - elapsed / budget_s)
        lr = init_lr * decay

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.clip(a_new, 0.0, 1000.0)

        # ---- 4.4  Renormalise to keep the total mass fixed ----
        s = a_new.sum()
        if s == 0.0:
            a_new[:] = target_sum / n
        else:
            a_new *= (target_sum / s)

        # ---- 4.5  Evaluate the new candidate (exact max) ----
        b_new = _conv_fft(a_new, a_new, L)
        cand_score = float(b_new.max())

        if np.isfinite(cand_score) and cand_score < best_score:
            best = a_new.copy()
            best_score = cand_score
            print(f"[{iteration:7d}] new best  score = {best_score:.6f}")

        # move to the new point irrespective of improvement (helps exploration)
        a = a_new

        # ---- 4.6  Occasional random perturbation ----
        if iteration % 400 == 0:
            perturb = np.exp(rng.normal(0.0, 0.04, size=n))
            a_pert = a * perturb
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            s_pert = a_pert.sum()
            if s_pert > 0.0:
                a_pert *= (target_sum / s_pert)
            b_pert = _conv_fft(a_pert, a_pert, L)
            pert_score = float(b_pert.max())
            if pert_score < best_score:
                best = a_pert.copy()
                best_score = pert_score
                a = a_pert
                print(f"[{iteration:7d}] perturb improved to {best_score:.6f}")

        # ---- 4.7  Increase smoothing parameter slowly ----
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_inc)

    # ------------------------------------------------------------------
    # 5. Final reporting
    # ------------------------------------------------------------------
    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    # Return the best sequence found (as plain Python floats)
    return [float(v) for v in best]
