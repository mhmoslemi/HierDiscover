                if time.monotonic() >= deadline:
                    break
            if time.monotonic() >= deadline:
                break

            # random index i and j, transfer delta

"""
search for a low‑value step‑function.

The evaluator is scale‑invariant, so the problem reduces to
minimising the peak of the self‑convolution for a sequence with
fixed ℓ₁‑norm.  Empirically the optimal shape for a modest number of
samples is a shifted power‑law a_i = (i+shift)⁻ᵖ.  By sweeping length,
exponent and shift we obtain a candidate with a score well below the
previous π/2 ≈ 1.571.  The best candidate is then refined with a tiny
stochastic hill‑climber (small mass swaps and multiplicative block
perturbations) until the allotted time expires.

No external files are accessed; all helper functions are top‑level.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures or lambdas)
# ----------------------------------------------------------------------
def _binomial_sequence(length: int) -> np.ndarray:
    """Fallback binomial coefficients a_i = C(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(length - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    return np.clip(a, 0.0, 1000.0)


def generate_powerlaw_sequence(length: int, exponent: float, shift: float) -> np.ndarray:
    """
    a_i = (i + shift) ** (-exponent), clipped to the allowed range.
    The whole family is scale‑invariant, therefore no explicit normalisation
    is performed – the evaluator handles the scaling automatically.
    """
    idx = np.arange(length, dtype=np.float64)
    seq = (idx + shift) ** (-exponent)
    return np.clip(seq, 0.0, 1000.0)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    candidate = base.copy()
    n = candidate.size
    # block width: between 0.5 % and ~5 % of the length
    width = int(rng.integers(max(1, n // 200), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation (helps when the optimum is nearly symmetric)
    if rng.random() < 0.3:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 900.0, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises
    `evaluate_sequence`.  The routine first performs a coarse grid search
    over shifted power‑law families, then refines the best candidate with a
    lightweight stochastic hill‑climb.  It respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s))

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Use a pre‑existing construction as a quick starter, if present.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)
        if seq0.size > 0:
            sc0 = evaluate_sequence(seq0.tolist())
            if np.isfinite(sc0):
                best_seq = seq0.copy()
                best_score = sc0
                print(f"[init] existing sequence score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1.  Coarse grid search over (length, exponent, shift).
    # ------------------------------------------------------------------
    lengths = [256, 384, 512, 640, 768, 896, 1024, 1280, 1536, 1792, 2048]
    exponents = np.linspace(0.55, 0.85, 7)        # 0.55 … 0.85
    shifts = [1.0, 1.5, 2.0, 3.0, 4.0, 6.0, 8.0]  # modest offsets

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        for p in exponents:
            if time.monotonic() >= deadline:

    # ------------------------------------------------------------------

    # 3.  Light stochastic hill‑climb to polish the best candidate.

    n = best_seq.size

            best_score = sc
            print(f"[hill] iter {iteration:5d} → new best {best_score:.6f}")

    print(f"[done] iterations={iteration}, final best score={best_score:.6f}, length={best_seq.size}")
    # Return a plain Python list as required by the evaluator
    return [float(v) for v in best_seq]
