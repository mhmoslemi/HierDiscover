def project_to_simplex(v):
    # returns projection of v onto simplex {x >= 0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def cross_corr_weights(a, w):
    # compute g_i = sum_k w_k * a_{k-i}
    n = len(a)
    L = len(w)
    size = 1 << ((n + L - 1).bit_length())
    A = np.fft.rfft(a, size)
    w_rev = w[::-1]
    B = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(A * B, size)
    # conv[t] = sum_j a_j * w_rev[t - j] = sum_j a_j * w[L-1 - (t - j)] = sum_j a_j * w[(L-1 - t) + j]
    # Want g_i = sum_j a_j * w_{j + i}
    # So let idx = L-1 - i => conv[idx] = sum_j a_j * w_{j + i}
    idx_start = L-1 - np.arange(n)
    # ensure idx_start within range
    g = 2.0 * conv[idx_start]
    return g

optimise the maximum of the auto‑convolution.
The max is non‑smooth; we replace it by a smooth *soft‑max* with a temperature
parameter β and minimise that instead.  The gradient of the soft‑max
is a weighted sum of the gradients of the individual convolution entries:
∂b_k/∂a_i = 2·a_{k‑i} (out‑of‑range indices give 0).
All operations (convolution, the weighted gradient) are performed with FFTs
so a full iteration costs O(n log n) and can be repeated thousands of times
within the 1000 s budget.

Each iteration:
    1. compute b = a ∗ a (FFT),
    2. build soft‑max weights w_k,
    3. obtain the gradient g_i = 2·∑_k w_k·a_{k‑i} via one more FFT,
    4. take a step a ← a – η·g,
    5. project the vector back onto the probability simplex
       (non‑negative, sum = 1).

The algorithm starts from the analytically optimal “arcsine” sequence,
normalised to unit sum, and continuously improves it.
Early on β is small (smooth objective) and the step size η is gradually
reduced; β is increased over time to focus on the true maximum.
Whenever a new best score is found it is printed.
The final sequence (still non‑negative) is returned as a plain Python list;
the harness will evaluate it with the supplied `evaluate_sequence` routine.

The method is completely self‑contained, uses only NumPy and runs comfortably
within the allowed time budget.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Utility: projection onto the simplex {x >= 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (L1 = 1, non‑negative)."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution using FFT (real valued)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return a * a using FFT (length 2*len(a)-1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, size)
    # element‑wise product = auto‑convolution in frequency domain
    B = np.fft.irfft(A * A, size)
    return B[: 2 * n - 1]


# ----------------------------------------------------------------------
# Gradient of the soft‑max of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the gradient of  softmax_β(a∗a)  w.r.t. a.
    Returns a vector of length n (gradient of the soft‑max, not yet divided by β).
    """
    n = a.size
    # convolution a∗a
    b = _conv_fft(a)                       # length 2n‑1
    # soft‑max weights (stable version)
    b_max = b.max()
    w_raw = np.exp(beta * (b - b_max))
    w = w_raw / w_raw.sum()                # Σ w_k = 1

    # The gradient g_i = 2 * Σ_k w_k * a_{k‑i}
    # This is a cross‑correlation of a and w.
    L = 2 * n - 1
    # reverse w for convolution
    w_rev = w[::-1]

    size = 1 << ((n + L - 1).bit_length())   # enough for the full linear convolution
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(A * B, size)

    # conv[t] = Σ_j a_j * w_rev[t‑j] = Σ_j a_j * w[(L‑1‑t) + j]
    # We need Σ_j a_j * w[j + i]  for i = 0…n‑1, i.e. set t = L‑1‑i
    idx = (L - 1) - np.arange(n)          # array of positions in conv
    g = 2.0 * conv[idx]                   # real‑valued, shape (n,)
    return g


# ----------------------------------------------------------------------
# Build the analytically optimal “arcsine” sequence (unscaled)
# ----------------------------------------------------------------------
def _build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a[i] = binom(2i,i) / 4**i for i=0..length‑1 (exact in double)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return the best step‑function heights discovered within *budget_s* seconds.
    Uses a projected gradient descent on a smooth approximation of the max‑
    convolution.  The final sequence is a list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size – fall back to 256 if the shared variable is absent.
    # --------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256

    # --------------------------------------------------------------
    # 1.  Initialise with the arcsine shape, normalised to unit sum.
    # --------------------------------------------------------------
    a = _build_arcsine_sequence(n).astype(np.float64)
    a = a / a.sum()                     # now ∑a = 1  (required for the simplex)

    # Initial score (exact formula, no need for evaluate_sequence here)
    b = _conv_fft(a)
    best_score = 2.0 * n * b.max()      # because ∑a = 1
    best = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2.  Projected gradient descent on the soft‑max of the auto‑convolution.
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)  # keep a second slack
    iteration = 0
    eta = 0.05               # initial step size
    beta = 5.0               # initial soft‑max temperature
    beta_max = 80.0
    while time.monotonic() < deadline:
        iteration += 1

        # --- compute current convolution and soft‑max gradient ---
        grad = _softmax_gradient(a, beta)

        # --- gradient step + projection onto simplex ---
        a_candidate = _project_to_simplex(a - eta * grad)

        # --- compute the exact score for the candidate (fast, no evaluate_sequence) ---
        b_candidate = _conv_fft(a_candidate)
        cand_score = 2.0 * n * b_candidate.max()   # ∑a_candidate == 1

        # --- keep the best ever seen ---
        if cand_score < best_score:
            best_score = cand_score
            best = a_candidate.copy()
            print(f"[{iteration:6d}] new best {best_score:.6f} (η={eta:.4f}, β={beta:.1f})")
            # early exit if we already beat the target
            if best_score < 1.503:
                break

        # --- accept the move if it does not increase the true max (helps stability) ---
        if cand_score <= 2.0 * n * b.max():
            a = a_candidate
        else:
            # occasionally still accept a worse move to escape plateaus
            if rng.random() < 0.005:
                a = a_candidate

        # --- schedule for η and β (simple decay / increase) ---
        if iteration % 500 == 0:
            eta *= 0.92                     # slowly shrink step size
            beta = min(beta_max, beta + 0.8)  # make the soft‑max sharper

    # --------------------------------------------------------------
    # 3.  Return the best sequence found (as plain Python floats)
    # --------------------------------------------------------------
    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best]
