    then mix the normalised optimiser g back into f.
* The loop stops when the time budget (≈ 1000 s) is exhausted.  The
  best sequence found is returned as a plain ``list[float]``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags


# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def _max_convolution(seq: np.ndarray) -> float:
    """
    Return the maximum value of the full convolution seq * seq.
    Uses an FFT‑based convolution (O(n log n)).
    """
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    return float(np.max(conv.real))


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast analogue of evaluate_sequence (no safety checks).
    The score is scale‑invariant; we normalise only by the L1 sum.
    """
    s = float(seq.sum())
    if s < 1e-12:
        return float("inf")
    max_conv = _max_convolution(seq)
    return 2.0 * seq.size * max_conv / (s * s)


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """Create a cheap local perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    w_low = max(1, n // 200)
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    scale = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= scale
    if rng.random() < 0.35:          # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _power_law_seq(n: int, exponent: float = 0.5,
                   shift: float = 0.0) -> np.ndarray:
    """Generate a normalised sequence a_i ∝ (i+shift)^{-exponent}."""
    idx = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power(idx + shift + 1.0, exponent)
    vals /= vals.sum()
    return vals


def _conv_matrix(seq: np.ndarray):
    """
    Build the (2n‑1) × n convolution matrix A such that A @ g = conv(seq,g).
    The matrix is stored as a CSR sparse matrix (memory‑efficient).
    """
    n = seq.size
    # each diagonal offset d (0 … n‑1) has constant value seq[d]
    diag_data = [np.full(n, seq[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=np.intc)
    A = diags(diag_data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _lp_optimize(f: np.ndarray, M: float):
    """
    Solve  max Σ g_j   subject to  (f * g)[k] ≤ M  (∀k) , 0 ≤ g_j ≤ 1000.
    Returns the optimiser g (as a dense numpy array).  The LP is solved
    with SciPy's high‑performance HiGHS interior‑point solver.
    """
    n = f.size
    A = _conv_matrix(f)                     # (2n‑1)×n matrix
    b = np.full(A.shape[0], M * 1.000001, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)       # minimise –Σ g_j  → maximise Σ g_j
    bounds = [(0.0, 1000.0)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # In the unlikely event of failure fall back to the current vector
        return f.copy()
    return res.x


def _gradient_of_max(f: np.ndarray, max_indices: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k (f*f)[k]  with respect to f.
    For each active index k we add 2·f_{k‑i} to component i.
    """
    n = f.size
    grad = np.zeros_like(f)
    for k in max_indices:
        # valid i satisfy 0 ≤ i < n and 0 ≤ k‑i < n
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=np.intp)
        grad[i] += 2.0 * f[k - i]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a coefficient sequence that minimises ``evaluate_sequence``.
    The function respects the provided time budget and returns the best
    sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Choose length and initialise the current sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        tmp = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if tmp.ndim != 1 or tmp.size == 0 or not np.isfinite(tmp).all():
            n = 2000
            base = _power_law_seq(n, exponent=0.5)
        else:
            n = tmp.size
            base = tmp.clip(0.0, 1000.0)
    else:
        n = 2000
        base = _power_law_seq(n, exponent=0.5)

    # Normalise to L1 = 1  (scale does not matter for the score)
    if base.sum() == 0.0:
        base = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        base = base[:n]
        if base.size < n:
            base = np.pad(base, (0, n - base.size), constant_values=0.0)
    base /= base.sum()

    # Keep a copy of the best sequence seen so far
    current = base.copy()
    best = current.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Parameters for the stochastic sub‑gradient descent
    # ------------------------------------------------------------------
    step = 0.15                     # initial step size (will be adapted)
    iteration = 0
    no_improve = 0                  # counter for stagnation handling

    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) evaluate the full convolution, locate the (few) maximal k
        # --------------------------------------------------------------
        conv_full = np.convolve(current, current)
        M = float(conv_full.max())
        max_idx = np.where(conv_full == M)[0]          # usually 1‑2 indices

        # --------------------------------------------------------------
        # 2) compute a (sub)‑gradient of the max‑convolution term
        # --------------------------------------------------------------
        grad = _gradient_of_max(current, max_idx)

        # --------------------------------------------------------------
        # 3) take a tentative gradient step, project back to the simplex
        # --------------------------------------------------------------
        cand = current - step * grad
        np.clip(cand, 0.0, None, out=cand)            # keep non‑negative
        tot = cand.sum()
        if tot < 1e-12:                               # degenerate – fallback
            cand = _make_block_perturbation(current, rng)
            tot = cand.sum()
        cand /= tot                                    # renormalise to L1 = 1

        # --------------------------------------------------------------
        # 4) fast evaluation – accept only if it improves
        # --------------------------------------------------------------
        fast_sc = _fast_score(cand)
        if fast_sc + 1e-9 < best_score:               # strict improvement
            try:
                true_sc = float(evaluate_sequence(cand.tolist()))
            except Exception:
                true_sc = float("inf")
            if np.isfinite(true_sc) and true_sc < best_score:
                best = cand.copy()
                best_score = true_sc
                current = cand.copy()
                print(f"[{iteration:6d}] new best {best_score:.6f} (grad step)")
                # enlarge step a little after a success
                step = min(step * 1.05, 0.5)
                no_improve = 0
                continue
        # If not accepted, shrink the step size
        step *= 0.90
        step = max(step, 1e-5)
        no_improve += 1

        # --------------------------------------------------------------
        # 5) occasional diversifications
        # --------------------------------------------------------------
        if iteration % 3000 == 0 or no_improve > 5000:
            # a) a cheap block perturbation around the current best
            cand = _make_block_perturbation(best, rng)
            fast_sc = _fast_score(cand)
            if fast_sc + 1e-9 < best_score:
                try:
                    true_sc = float(evaluate_sequence(cand.tolist()))
                except Exception:
                    true_sc = float("inf")
                if np.isfinite(true_sc) and true_sc < best_score:
                    best = cand.copy()
                    best_score = true_sc
                    current = cand.copy()
                    print(f"[{iteration:6d}] new best {best_score:.6f} (block)")
                    step = max(step, 0.02)      # give the search a fresh push
                    no_improve = 0

        # --------------------------------------------------------------
        # 6) occasional alternating linear‑programming refinement
        # --------------------------------------------------------------
        if iteration % 6000 == 0:
            # solve the LP for the current f
            g_raw = _lp_optimize(current, M)
            # normalise g to L1 = 1
            g = g_raw / g_raw.sum()
            # gently mix the optimiser back into the current sequence
            t = 0.12
            cand = (1 - t) * current + t * g
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand /= cand.sum()
            fast_sc = _fast_score(cand)
            if fast_sc + 1e-9 < best_score:
                try:
                    true_sc = float(evaluate_sequence(cand.tolist()))
                except Exception:
                    true_sc = float("inf")
                if np.isfinite(true_sc) and true_sc < best_score:
                    best = cand.copy()
                    best_score = true_sc
                    current = cand.copy()
                    print(f"[{iteration:6d}] new best {best_score:.6f} (LP mix)")
                    step = max(step, 0.02)
                    no_improve = 0

    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    # Return a plain Python list suitable for the external evaluator
    return [float(v) for v in best]
