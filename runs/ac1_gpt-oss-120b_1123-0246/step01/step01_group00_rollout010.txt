def max_convolution(a):
    n = a.shape[0]

def compute_score(a):
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')

    L = 1 << ((2*n-1).bit_length())

    conv = np.fft.irfft(fa*fa, L)
    max_b = conv[:2*n-1].max()

"""
Simple evolutionary+gradient search for the step‑function minimisation problem.

The algorithm works with a (potentially) large sequence length.
* The sum of the heights is kept fixed at √(2 n) – this makes the
  evaluation function equal to the maximal self‑convolution value.
* A fast FFT based routine is used to obtain the convolution and the
  objective (≈10 µs for n≈3·10⁴ on a modest CPU).
* At each iteration we either
    – take a gradient step that reduces the current worst convolution lag,
    – or perform a small random mass‑transfer move (simulated‑annealing style).
  The temperature and the step size decay linearly with the remaining time.
* Every few thousand iterations a global multiplicative perturbation is
  applied to escape plateaus.
* The routine starts from the best known construction (if a global
  variable `height_sequence_1` exists) padded/centred to the target length,
  otherwise from a simple “U‑shaped’’ deterministic seed.

The function respects the 1000 s budget (the caller passes the allowed time
via the `budget_s` argument) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _max_conv_fft(a: np.ndarray, fft_len: int) -> float:
    """
    Return max_{k} (a*a)[k] using an FFT of length ``fft_len``.
    The linear convolution is contained in the first 2·len(a)-1 entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # only the linear part matters
    return float(conv[: 2 * a.shape[0] - 1].max())


def _full_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the full linear convolution a*a (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a deterministic U‑shaped seed.
    If a global ``height_sequence_1`` exists we centre‑pad it to length n.
    """
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        base = None

    if base is not None and base.size > 0:
        # centre‑pad / truncate to the desired length
        seq = np.zeros(n, dtype=np.float64)
        if base.size <= n:
            start = (n - base.size) // 2
            seq[start : start + base.size] = base
        else:  # very unlikely, but safe
            seq[:] = base[:n]
        return seq

    # deterministic “U‑shape’’ – high at the ends, low in the centre
    i = np.arange(n, dtype=np.float64)
    seq = 1.0 / (i + 1.0) + 1.0 / (n - i)
    return seq

            cand[j_idx] += delta

        # ---------- enforce bounds & re‑normalise ----------
        cand = np.clip(cand, 0.0, 1000.0)
        cand_sum = cand.sum()
        if cand_sum > 0.0:
            cand *= (target_sum / cand_sum)   # exact normalisation
        cand = np.clip(cand, 0.0, 1000.0)

        # ---------- evaluate candidate ----------
        cand_max = _max_conv_fft(cand, fft_len)
        cand_score = cand_max                     # same normalisation as above

        # ---------- accept / reject ----------
        improve = cand_score < cur_score
        if improve or rng.random() < math.exp((cur_score - cand_score) / (temp + 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[{iteration:6d}] new best score = {best_score:.6f}")

        # ---------- occasional global perturbation ----------
        if iteration % 3000 == 0 and iteration > 0:
            pert = np.exp(rng.normal(0.0, 0.025, size=n))
            cand2 = a * pert
            cand2 = np.clip(cand2, 0.0, 1000.0)
            cand2_sum = cand2.sum()
            if cand2_sum > 0.0:
                cand2 *= (target_sum / cand2_sum)
            cand2 = np.clip(cand2, 0.0, 1000.0)
            cand2_score = _max_conv_fft(cand2, fft_len)
            if cand2_score < best_score:
                best = cand2.copy()
                best_score = cand2_score
                a = cand2
                print(f"[{iteration:6d}] global perturb improved score = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 7. Final report
    # ------------------------------------------------------------------
    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")

    # Convert to plain Python floats; final clipping guarantees legal input.
    return [float(v) for v in best]
