    score = 2·n·max(a*a) / (∑a)² .
Hence we can normalise the step‑function to ∑a = 1.  In this
normalisation the score is simply 2·n·max(a*a).  The algorithm
starts from the classic Catalan sequence (which already gives
π/2 ≈ 1.5708) and repeatedly:
  * finds an index k where the convolution attains its maximum,
  * computes the sub‑gradient of max(a*a) at that point (g_i = 2·a_{k‑i}),
  * takes a step a ← a – η·g and projects back onto the probability
    simplex (∑a = 1, a ≥ 0) using the Duchi et al. projection.
The step size η is reduced over time.  Occasionally a cheap random
multiplicative perturbation is tried to escape plateaus.

The loop runs until the time budget expires (default ≈1000 s) and
the best sequence seen is returned.  The returned sequence is
scaled only to keep the sum comfortably above the 0.01 threshold
required by `evaluate_sequence`; scaling does not affect the score.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: Catalan‑type coefficients C_i = binom(2i,i)/4**i
# --------------------------------------------------------------
def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan‑type numbers C_i = binom(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence  a[i] = a[i‑1] * (2*i‑1)/(2*i)
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# --------------------------------------------------------------
# Helper: projection onto the probability simplex (∑x = 1, x ≥ 0)
# --------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x | x≥0, sum(x)=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Main entry point required by the evaluation framework
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 970.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimise
    `evaluate_sequence`.  The routine normalises the vector to sum = 1,
    applies a simple sub‑gradient descent on max(convolution) with
    projection onto the simplex, and finally rescales the vector to a
    moderate magnitude before returning it.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # ------------------------------------------------------------------
    # 1. Determine problem size and initialise the sequence.
    # ------------------------------------------------------------------
    # try to reuse a previously found construction if it exists
    height_seq_1 = globals().get("height_sequence_1")
    if isinstance(height_seq_1, (list, tuple, np.ndarray)):
        init = np.asarray(height_seq_1, dtype=np.float64)
        n = init.size
        if n == 0:
            init = None
    else:
        init = None
        n = 4800                     # a reasonable default length

    # start from the Catalan sequence (good deterministic baseline)
    a = _catalan_sequence(n).astype(np.float64)

    # normalise to unit ℓ₁‑norm (∑a = 1) – required for the sub‑gradient step
    a = a / np.sum(a)

    # possibly replace the start by the provided sequence if it is already better
    if init is not None and init.size == n:
        try:
            # normalise the external candidate as well
            cand = init / np.sum(init)
            score_cand = float(evaluate_sequence(cand.tolist()))
            score_catalan = float(evaluate_sequence(a.tolist()))
            if score_cand < score_catalan:
                a = cand.copy()
                best_score = score_cand
                print(f"Using supplied sequence as start, score={best_score:.6f}")
            else:
                best_score = score_catalan
        except Exception:
            best_score = float(evaluate_sequence(a.tolist()))
    else:
        best_score = float(evaluate_sequence(a.tolist()))

    best = a.copy()
    print(f"Initial score: {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 2. Sub‑gradient descent on max(convolution) with simplex projection.
    # ------------------------------------------------------------------
    iteration = 0
    # step‑size schedule – start relatively large, decay with √iteration
    eta0 = 0.7
    # small array of step‑size multipliers to try a few candidates per iteration
    eta_factors = (1.0, 0.5, 0.2, 0.1, 0.05)

    # keep a short‑term moving average of the max‑convolution to decide when to inject noise
    avg_max = None
    while time.monotonic() < deadline and iteration < 8000:
        # current convolution and its worst value
        conv = np.convolve(a, a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b           # sum(a)=1 → score = 2·n·max_b
        if cur_score < best_score - 1e-12:
            best = a.copy()
            best_score = cur_score
            print(f"Iter {iteration:4d}: new best score {best_score:.6f}")

        # --------------------------------------------------------------
        # 2a. Sub‑gradient of the active convolution entry.
        # --------------------------------------------------------------
        # pick a random index among the (almost) maximal entries to avoid bias
        max_val = max_b
        # tolerance to treat near‑maximal entries as active
        tol = max_val * (1.0 - 1e-12)
        active = np.where(conv >= tol)[0]
        k0 = rng.choice(active) if active.size > 0 else int(np.argmax(conv))

        # gradient g_i = 2·a_{k0‑i}  (zero when index out of range)
        g = np.zeros_like(a)
        idx = k0 - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g[mask] = 2.0 * a[idx[mask]]

        # --------------------------------------------------------------
        # 2b. Try several step sizes, keep the first improvement.
        # --------------------------------------------------------------
        improved = False
        eta = eta0 / np.sqrt(iteration + 1.0)   # decaying base step size
        for factor in eta_factors:
            step = eta * factor
            cand = a - step * g
            cand = _simplex_projection(cand)
            # quick evaluation (no need for the full evaluate_sequence)
            conv_cand = np.convolve(cand, cand)
            cand_score = 2.0 * n * conv_cand.max()
            if cand_score < best_score - 1e-12:
                a = cand
                best_score = cand_score
                best = cand.copy()
                improved = True
                print(f"Iter {iteration:4d}, η={step:.5f} → score {best_score:.6f}")
                break

        # --------------------------------------------------------------
        # 2c. If no improvement, apply a cheap random perturbation.
        # --------------------------------------------------------------
        if not improved:
            # multiplicative log‑normal noise (keeps positivity)
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cand = a * noise
            cand = _simplex_projection(cand)
            conv_cand = np.convolve(cand, cand)
            cand_score = 2.0 * n * conv_cand.max()
            if cand_score < best_score - 1e-12:
                a = cand
                best_score = cand_score
                best = cand.copy()
                print(f"Iter {iteration:4d}, random tweak → score {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 3. Scale the best vector to a comfortable magnitude and return.
    # ------------------------------------------------------------------
    # The score is scale‑invariant; we simply ensure the ℓ₁‑sum is >0.01.
    scale_factor = max(1.0, 0.1 / np.sum(best))   # guarantee sum >= 0.1
    final_seq = best * scale_factor
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")

    return [float(v) for v in final_seq.tolist()]
