def _shift(seq, k):
    n = seq.shape[0]
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    # indices in seq for output positions start:end inclusive
    out = np.zeros_like(seq)
    # indices in seq for values: k - i
    # vectorized: i = np.arange(start, end+1)
    i = np.arange(start, end + 1)
    out[i] = seq[k - i]
    return out

def _gradient_update(seq, eta, top_k=3):
    conv = np.convolve(seq, seq)
    # Determine top indices
    if top_k >= 1:
        idxs = np.argpartition(-conv, top_k)[:top_k]  # unsorted but top k largest
    else:
        idxs = [np.argmax(conv)]
    # Compute average shift
    d = np.zeros_like(seq)
    for k in idxs:
        d += _shift(seq, k)
    d = d / len(idxs)  # average
    # Compute direction preserving sum: subtract its mean and add back
    # direction = -d + d.mean() so sum zero
    direction = -d + np.mean(d)  # ensures sum zero (since sum(direction) = -sum(d) + n*mean(d) = 0)
    new_seq = seq + eta * direction
    # Clip
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    total = new_seq.sum()
    if total > 0:
        new_seq = new_seq / total
    else:
        new_seq = seq.copy()
    return new_seq

"""
Step‑function generator for the convolution minimisation problem.

The algorithm uses three ideas:

1. **Power‑law seed** – a 1/√(i+β) shape (theoretical optimum) is
   searched over a small grid of exponents and offsets, the best
   candidate becoming the initial vector.

2. **Targeted projected gradient descent** – at each iteration the
   indices that contribute most to the current maximal convolution
   value are identified (a few top peaks).  A sub‑gradient that pushes
   probability mass away from those “heavy” indices is applied while
   keeping the total sum fixed (projection onto the simplex).

3. **Stochastic block perturbations** – occasional multiplicative
   log‑normal noise on a random block and optional symmetrisation give
   the optimiser the ability to escape local minima.

Both moves respect the non‑negative, ≤ 1000 constraints required by
``evaluate_sequence``.  An exponential cooling schedule controls the
step size and a Metropolis‑like acceptance rule allows uphill moves
early on.  The best sequence found before the time budget expires is
returned as a plain Python list.
"""

    max_b = conv.max()
    n = seq.size
    return float(2.0 * n * max_b / (total * total))

    Return the vector d_i = a_{k-i} (with out‑of‑range entries set to 0).
    This is the sub‑gradient of the term (seq*seq)[k] w.r.t. ``seq``.

    end = min(k, n - 1)

        out[idx] = seq[k - idx]
    return out

    the sub‑gradients of the ``top_k`` largest convolution entries.
