Define fast evaluation with FFT:

def evaluate_fast(seq):
    n = seq.shape[0]
    # compute conv via FFT with size 2*n -1
    size = 1 << ((2*n - 1).bit_length())  # next power of 2
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[:2*n-1]
    max_b = conv.max()
    total = seq.sum()
    if total < 1e-8:
        return np.inf
    return float(2.0 * n * max_b / (total * total))

But need to be careful about floating errors might produce small negative values for conv; we can do np.maximum(conv,0) before max, but conv may have small negative due rounding but max will still be positive.

    while time.monotonic() < deadline:

        if rng.random() < 0.5:

            n = best_seq.size
            # Choose new n maybe +- 10% within bounds
            n_new = int(max(50, min(8000, n * (1 + rng.normal(0, 0.1)))))

            start = int(rng.integers(0, max(0, seq.size - block_width+1)))

            print(f"new best via mutation: n={seq.size}, score={best_score:.6f}")

    # final result: ensure non-negatives and <=1000 and list
    best_seq = np.clip(best_seq, 0.0, 1000.0).tolist()

def generate_powerlaw_sequence(n, p=0.5, beta=0.0):
    idx = np.arange(n, dtype=np.float64)
    seq = 1.0 / np.power(idx + beta + 1.0, p)  # adding 1 to avoid zero
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total == 0:
        return np.full(n, 0.0)
    seq /= total
    return seq

def lp_iterative_improve(f, max_iters=15, t0=0.5, rng=None):
    """
    Perform a few LP-driven improvement steps on a non‑negative sequence f
    (assumed to have unit L1 sum). Returns the improved sequence.
    """
    import scipy.optimize
    n = f.size
    # precompute convolution matrix once and update in each iteration? Since f changes each iteration,
    # matrix changes. So recompute each iteration.
    for it in range(max_iters):
        M = max_convolution(f)
        # Build constraint matrix
        A = np.zeros((2*n-1, n), dtype=np.float64)
        # fill column-wise
        for j in range(n):
            A[j:j+n, j] = f
        # linear program: maximize sum(g) subject to A @ g <= M, g >= 0
        c = -np.ones(n)  # minimize negative sum = maximize sum
        b = np.full(2*n-1, M, dtype=np.float64)
        # bounds defined by variable g >=0
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            # optimizer failed – abort further LP steps
            break
        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g == 0:
            break
        # scale g to have same L1 as f
        g = g * (sum_f / sum_g)

        # mixing step: decreasing step size
        t = t0 * (0.5 ** it)
        f_new = (1 - t) * f + t * g
        # re‑normalise (should stay close to 1)
        f_new = f_new / f_new.sum()
        # compute new max conv
        M_new = max_convolution(f_new)
        # stop if not improving
        if M_new < M - 1e-12:
            f = f_new
        else:
            # break early if no progress
            break
    return f

def block_perturb(seq, rng):
    cand = seq.copy()
    n = cand.size
    # choose block width
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi+1))
    start = int(rng.integers(0, n - width + 1))
    # multiplicative factor lognormal
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:start+width] *= factor
    # occasional symmetrisation

"""
Step‑function generator for the self‑convolution L∞ optimisation problem.

The algorithm consists of three stages:

1. **Parameter sweep** – generate a few power‑law candidates
   a_i ∝ (i+β)^{-p} (with β≥0, 0.4≤p≤0.6) for several lengths and keep the
   best one according to the fast evaluation routine.

2. **LP‑driven refinement** – for a chosen length n (≤ 1024, small enough for
   a dense linear program) we repeatedly solve a linear program

        maximise   Σ g_j
        subject to g_j ≥ 0
                   (f * g)(k) ≤ M      ∀k

   where M = max(f*f) and f is the current sequence.
   The solution g is renormalised to the same L1‑mass as f and mixed with f
   using a geometrically decreasing step size.  Each iteration can only lower
   the objective value 2·n·max(f*f) and converges quickly.

3. **Stochastic block perturbations** – starting from the best sequence
   obtained above we perform a fast hill‑climber (log‑normal multiplicative
   updates on random blocks) until the time budget expires.  Only improvements
   are kept.

All operations respect the domain required by ``evaluate_sequence``:
non‑negative values, capped at 1000, and a total sum far above the 0.01
threshold.

The function ``propose_candidate`` returns a plain Python list; the external
evaluator will apply the same clipping again.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def _fast_evaluate(seq: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for a non‑negative 1‑D array.
    The sequence is assumed to have a non‑zero L1‑mass.
    """
    n = seq.shape[0]
    # use FFT for speed when the sequence is long
    size = 1 << ((2 * n - 1).bit_length())        # next power of two
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    max_b = conv.max()
    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    # scaling invariance: we keep total as is (normally =1)
    return float(2.0 * n * max_b / (total * total))


def _max_convolution(seq: np.ndarray) -> float:
    """Return max_k (seq * seq)[k] using a direct convolution (small n)."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1) × n matrix A such that (A @ g) = f * g (full convolution)
    for any vector g of the same length as f.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j contains f shifted by j positions
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _lp_improve(f: np.ndarray, max_iters: int = 12, rng: np.random.Generator = None,
                deadline: float = None) -> np.ndarray:
    """
    Perform a few LP‑driven improvement steps on a sequence f.
    The L1‑mass of f is assumed to be 1 (the function re‑normalises after each step).
    """
    try:
        from scipy.optimize import linprog
    except Exception:                     # SciPy not available – abort LP phase
        return f

    n = f.size
    f = f / f.sum()                       # ensure unit L1‑mass
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        M = _max_convolution(f)            # current supremum of f*f
        # Build the linear constraints A @ g ≤ M
        A = _conv_matrix(f)
        b = np.full(A.shape[0], M, dtype=np.float64)

        # Linear programme: maximise Σ g_j  ⇔ minimise −Σ g_j
        c = -np.ones(n, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
        if not res.success:
            # If the solver fails we stop trying further LP steps.
            break

        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g == 0:
            break
        # Scale g so that it has the same total mass as f
        g = g * (sum_f / sum_g)

        # Geometrically decreasing convex combination
        t = 0.5 ** (it + 1)                # step size 0.5, 0.25, 0.125, …
        f_new = (1.0 - t) * f + t * g
        f_new /= f_new.sum()               # re‑normalise to unit L1

        # Accept the update only if it reduces the convolution maximum
        if _max_convolution(f_new) < M - 1e-12:
            f = f_new
        else:
            break                         # no progress – terminate early
    return f


def _powerlaw_sequence(n: int, p: float = 0.5, beta: float = 0.0) -> np.ndarray:
    """
    Return a normalised sequence a_i ∝ (i + β + 1)^{-p}.
    Adding the constant ``+1`` avoids a division by zero at i=0.
    """
    idx = np.arange(n, dtype=np.float64)
    seq = 1.0 / np.power(idx + beta + 1.0, p)
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total == 0:
        return np.full(n, 0.0)
    return seq / total


def _block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random block, with an occasional
    symmetrisation.  The result is re‑normalised to unit L1‑mass.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # modest log‑normal noise
    cand[start : start + width] *= factor
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    cand = np.clip(cand, 0.0, 1000.0)
    total = cand.sum()
    if total == 0:
        return base
    return cand / total

# Main entry point
# ----------------------------------------------------------------------

    """
    import numpy as np
    from numpy.random import default_rng

    rng = default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)   # small safety margin

    # ------------------------------------------------------------------
    # Determine a sensible base length (use the supplied construction if it exists)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)

            raise ValueError
        n0 = int(base.size)
    except Exception:

    # A handful of candidate lengths (powers of two are FFT‑friendly)
    n_candidates = sorted(

    # ------------------------------------------------------------------

    best_seq = None

    return best.tolist()
