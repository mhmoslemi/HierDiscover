import numpy as np, time, math

def propose_candidate(seed=12345, budget_s=600, **kwargs):
    """
    Generates a monotone decreasing sequence (starting from a power‑law) and
    iteratively improves it by moving mass away from the entries that
    contribute most to the current maximal convolution coefficient.
    The optimisation runs until the time budget expires and returns the
    best sequence found (numpy array converted to list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # choose length
    try:
        n = len(height_sequence_1)
        n = max(256, min(n, 6000))
    except Exception:
        n = 3000
    # make n a multiple of 2 for convenience
    if n % 2 != 0:
        n += 1

    # initialise with power‑law exponent 0.5 (the known near‑optimal family)
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(idx, -0.5)
    a /= a.sum()
    a *= 100.0                     # arbitrary scaling – does not affect the score
    total_mass = a.sum()

    # compute initial score using the same formula as evaluate_sequence
    def compute_score(arr):
        nlen = len(arr)
        b = np.convolve(arr, arr)
        maxb = b.max()
        s = arr.sum()
        if s < 0.01:
            return np.inf
        return 2.0 * nlen * maxb / (s * s)

    best_arr = a.copy()
    best_score = compute_score(best_arr)
    print(f"initial score={best_score:.6f}  length={n}")

    iteration = 0
    max_iter = 200000   # safety upper bound
    # step size as fraction of selected entry; will be reduced over time
    step_frac = 0.2
    while time.monotonic() < deadline and iteration < max_iter:
        # convolution and locate the current maximal index
        b = np.convolve(a, a)
        k = int(b.argmax())
        # compute contributions for this k (only where indices are valid)
        i = np.arange(n)
        j = k - i
        valid = (j >= 0) & (j < n)
        if not valid.any():
            # no overlap, break
            break
        contrib = np.zeros_like(a)
        contrib[valid] = a[i[valid]] * a[j[valid]]
        # choose source index with largest contribution
        i_src = int(i[valid][np.argmax(contrib[valid])])
        # choose destination index with smallest contribution (but not the same)
        # to avoid moving mass to the same place use random low‑contrib index
        low_candidates = i[valid][contrib[valid] < np.percentile(contrib[valid], 25)]
        if low_candidates.size == 0:
            i_dst = int(i[valid][np.argmin(contrib[valid])])
        else:
            i_dst = int(rng.choice(low_candidates))
        if i_src == i_dst:
            # fallback: pick a random different index
            i_dst = int(rng.integers(0, n))
            if i_dst == i_src:
                continue

        # transfer a fraction of the source mass
        delta = a[i_src] * step_frac * rng.uniform(0.5, 1.0)   # up‑to step_frac of the source
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i_src] -= delta
        cand[i_dst] += delta
        # keep non‑negative
        if cand[i_src] < 0.0:
            continue
        # evaluate
        score = compute_score(cand)
        if np.isfinite(score) and score < best_score:
            best_score = float(score)
            best_arr = cand.copy()
            a = cand      # keep the improvement
            # slowly shrink the step size to fine‑tune later
            step_frac = max(0.02, step_frac * 0.995)
            print(f"iter {iteration:6d}  improve score={best_score:.6f}")
        else:
            # occasional acceptance of non‑improving move (simulated annealing)
            # temperature decays with iteration count
            temp = max(0.001, 0.01 * (1.0 - iteration / max_iter))
            if np.isfinite(score) and rng.random() < math.exp(-(score - best_score) / temp):
                a = cand      # accept temporary worsening move
        iteration += 1

    print(f"final best score={best_score:.6f}   iterations={iteration}")
    # final conversion to list of floats (ensure python built‑ins)
    return best_arr.tolist()

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm works in three stages:

1. **Initialisation** – start from a monotone decreasing power‑law
   `a_i ∝ (i+1)^{-0.5}` (the known near‑optimal shape) and scale it
   so that the sum is comfortably above the evaluator’s 0.01 lower bound.

2. **Targeted local improvement** – repeatedly locate the convolution
   index `k` that gives the current maximal coefficient `b_k`.  The
   contribution of each element `i` to this coefficient is
   `c_i = a_i * a_{k-i}` (zero when the partner index is out of range).
   The element with the largest contribution is the best source from
   which to move mass; a destination with a small contribution is chosen
   at random among the low‑contribution indices.  A fraction of the
   source mass is transferred, the total mass is kept constant, and the
   new score is evaluated.  Improvements are kept; a simple simulated‑annealing
   acceptance rule occasionally accepts non‑improving moves to escape flats.

3. **Return the best sequence found** – after the time budget expires the
   best sequence (as a plain ``list[float]``) is returned.  All numbers
   are non‑negative and far below the evaluator’s 1000‑per‑entry clamp,
   so the score reported by ``evaluate_sequence`` is the true objective.

The method runs entirely in memory, uses only NumPy (and the provided
``evaluate_sequence`` for the final check) and respects the 1000‑second
wall‑clock limit.
"""
import numpy as np
import time
import math

def propose_candidate(seed: int = 42, budget_s: float = 800, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine follows the strategy described in
    the module docstring and returns the best sequence discovered before
    the allotted time runs out.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a working length.
    # Re‑use the length of the previously found construction if it exists,
    # otherwise fall back to a moderate size that fits comfortably into the
    # time budget.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # global from the platform, if any
        n = max(256, min(n, 6000))          # keep it in a reasonable range
    except Exception:
        n = 3000

    # make n even – convenient for the FFT based convolution used later
    if n % 2:
        n += 1

    # ------------------------------------------------------------------
    # Initialise with the classic power‑law a_i ∝ (i+1)^{-0.5}.
    # The scale does not affect the objective; we use a total mass of 100.
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(idx, -0.5)
    a /= a.sum()               # normalise to a probability vector
    a *= 100.0                 # any positive scale works (kept < 1000)

    total_mass = a.sum()       # constant throughout the optimisation

    # ------------------------------------------------------------------
    # Helper that mirrors the behaviour of ``evaluate_sequence`` but avoids
    # the internal clipping overhead.  All entries stay well below 1000,
    # so the result is identical.
    # ------------------------------------------------------------------
    def compute_score(arr: np.ndarray) -> float:
        nlen = arr.shape[0]
        # FFT based convolution – O(n log n) and much faster than the
        # naïve O(n²) implementation for the thousands of calls we make.
        size = int(2 ** np.ceil(np.log2(2 * nlen - 1)))
        fa = np.fft.rfft(arr, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[: 2 * nlen - 1]
        max_b = conv.max()
        s = arr.sum()
        if s < 0.01:
            return np.inf
        return 2.0 * nlen * max_b / (s * s)

    best_arr = a.copy()
    best_score = compute_score(best_arr)
    print(f"initial score = {best_score:.6f}  length = {n}")

    # ------------------------------------------------------------------
    # Main optimisation loop – runs until the wall‑clock budget is spent.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # adaptive step size (as a fraction of the source entry)
    step_frac = 0.20
    iteration = 0
    max_iter = 50000                       # safety upper bound

    while time.monotonic() < deadline and iteration < max_iter:
        # 1) locate the index of the maximal convolution coefficient
        size = int(2 ** np.ceil(np.log2(2 * n - 1)))
        fa = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
        k = int(conv.argmax())              # shift with the current max

        # 2) compute contributions a_i * a_{k-i} for the active shift
        i = np.arange(n)
        j = k - i
        valid = (j >= 0) & (j < n)
        if not valid.any():                 # should never happen
            break
        contrib = np.zeros_like(a)
        contrib[valid] = a[i[valid]] * a[j[valid]]

        # source = index with the largest contribution to the active max
        src_idx = int(i[valid][np.argmax(contrib[valid])])

        # destination = index with *small* contribution (chosen from the
        # lower quartile to keep the move focussed)
        low_mask = contrib[valid] < np.percentile(contrib[valid], 25)
        low_candidates = i[valid][low_mask]
        if low_candidates.size == 0:
            dst_idx = int(i[valid][np.argmin(contrib[valid])])
        else:
            dst_idx = int(rng.choice(low_candidates))

        if src_idx == dst_idx:
            # fall back to a random different index
            dst_idx = int(rng.integers(0, n))
            if dst_idx == src_idx:
                iteration += 1
                continue

        # 3) transfer a fraction of the source mass to the destination
        delta = a[src_idx] * step_frac * rng.uniform(0.5, 1.0)
        if delta <= 0.0:
            iteration += 1
            continue
        cand = a.copy()
        cand[src_idx] -= delta
        cand[dst_idx] += delta
        if cand[src_idx] < 0.0:            # safety – should never happen
            iteration += 1
            continue

        # 4) evaluate the candidate
        cand_score = compute_score(cand)

        # 5) accept if improved; otherwise occasionally accept a worse
        #    move (simulated‑annealing style) to escape plateaus.
        improved = np.isfinite(cand_score) and cand_score < best_score
        if improved:
            best_score = float(cand_score)
            best_arr = cand.copy()
            a = cand                  # continue from the improved point
            # slowly shrink the step size as we get closer to a local optimum
            step_frac = max(0.02, step_frac * 0.995)
            print(f"iter {iteration:6d}  new best = {best_score:.6f}")
        else:
            # simple temperature schedule
            temperature = max(1e-4, 0.05 * (1.0 - iteration / max_iter))
            prob = math.exp(-(cand_score - best_score) / temperature) if np.isfinite(cand_score) else 0.0
            if rng.random() < prob:
                a = cand               # accept the non‑improving move

        iteration += 1

    print(f"final best score = {best_score:.6f} after {iteration} iterations")
    # ``evaluate_sequence`` is scale‑invariant; we already kept values
    # well within the allowed range, so we can return the list directly.
    return best_arr.tolist()
