"""
Search for a low‑score non‑negative step function.

Strategy
--------
1.  Sample a few modest lengths (1024 … 8192) and a handful of simple
    parametric families (uniform, linear, exponential, power‑law).  Each
    candidate is scaled so that the L¹‑norm equals √(2 n) – this makes the
    objective equal to the maximal convolution value and removes the
    trivial scale invariance.

2.  Keep the best candidate from the coarse sweep.

3.  Refine it with a fast projected gradient descent that attacks **all**
    of the currently largest convolution lags (not only the single
    maximum).  The gradient of a lag *k* w.r.t. aᵢ is 2·a_{k‑i}
    (zero when the index is out of range).  By taking a small step in the
    opposite direction, normalising the sum and clipping to [0,1000] we
    steadily flatten the autocorrelation peak.

    * A handful (K≈5) of top lags are handled simultaneously,
      which prevents the algorithm from cycling around a single dominant
      lag.
    * A simple momentum term (β≈0.9) helps to traverse plateaus.
    * Every few hundred iterations a multiplicative random perturbation
      (log‑normal noise) is applied to escape local minima.

4.  The whole procedure is time‑bounded (default ≈ 1000 s).  Progress
    and intermediate scores are printed; the best sequence found is
    returned as a plain list of Python floats.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Efficient full convolution a*a using FFT for n≥512, otherwise np.convolve."""
    n = a.size
    if n < 512:
        return np.convolve(a, a)
    size = 2 * n - 1
    nfft = 1 << (size - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:size]
    # Numerical noise may produce tiny negatives – clip safely
    conv = np.maximum(conv, 0.0)
    return conv


def _score(a: np.ndarray) -> float:
    """Exact replica of evaluate_sequence (without the input‑validation overhead)."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = a.size
    b = _full_convolution(a)
    max_b = b.max()
    return float(2.0 * n * max_b / (s * s))


def _normalise(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and rescale to reach exactly target_sum."""
    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total > 0.0:
        a = a * (target_sum / total)
    a = np.clip(a, 0.0, 1000.0)
    return a


def _make_exponential(n: int, r: float, target: float) -> np.ndarray:
    """a_i = C * r**i   (decreasing, r∈(0,1]) and then scaled to target L¹‑norm."""
    idx = np.arange(n, dtype=np.float64)
    raw = r ** idx
    return _normalise(raw, target)


def _make_powerlaw(n: int, p: float, target: float) -> np.ndarray:
    """a_i = C * (n-i)**p  (p can be negative)."""
    idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n-1, …, 1
    raw = idx ** p
    return _normalise(raw, target)


def _make_linear(n: int, target: float) -> np.ndarray:
    """Simple decreasing linear profile."""
    raw = np.linspace(1.0, 0.0, n, dtype=np.float64)
    return _normalise(raw, target)


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) attains a low
    evaluate_sequence() score.  The routine respects the time budget
    `budget_s` (seconds) and prints progress information.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # Leave a small safety margin for the host environment
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    best_seq = None
    best_val = float('inf')

    # --------------------------------------------------------
    # 1. Coarse parametric sweep (fast, global exploration)
    # --------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    # If a previous construction exists, also try its length.
    try:
        prev_len = len(height_sequence_1)
        if prev_len not in lengths:
            lengths.append(prev_len)
    except NameError:
        pass

    for n in lengths:
        target_sum = np.sqrt(2.0 * n)

        # Uniform baseline
        cand = np.full(n, target_sum / n, dtype=np.float64)
        val = _score(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[sweep] n={n:5d} uniform      score={val:.6f}")

        # Linear decreasing
        cand = _make_linear(n, target_sum)
        val = _score(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[sweep] n={n:5d} linear       score={val:.6f}")

        # Exponential family (30 points, r ∈ (0.5, 0.9995))
        for r in np.linspace(0.5, 0.9995, 30):
            cand = _make_exponential(n, r, target_sum)
            val = _score(cand)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[sweep] n={n:5d} exp r={r:.4f} score={val:.6f}")

        # Power‑law family (31 points, p ∈ [‑3, 3])
        for p in np.linspace(-3.0, 3.0, 31):
            cand = _make_powerlaw(n, p, target_sum)
            val = _score(cand)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[sweep] n={n:5d} pow p={p:5.2f} score={val:.6f}")

    # Optionally try a few random Dirichlet points (cheap)
    n_rand = best_seq.size if best_seq is not None else 1024
    target_sum = np.sqrt(2.0 * n_rand)
    for _ in range(20):
        if time.monotonic() >= deadline:
            break
        tmp = rng.random(n_rand)
        cand = _normalise(tmp, target_sum)
        val = _score(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[random] n={n_rand:5d}          score={val:.6f}")

    # --------------------------------------------------------
    # 2. Refined gradient‑descent on the best candidate
    # --------------------------------------------------------
    a = best_seq.copy()
    n = a.size
    target_sum = np.sqrt(2.0 * n)          # keep the invariant used in the sweep
    a = _normalise(a, target_sum)

    # Gradient‑descent parameters
    max_iters = 10 ** 9                    # effectively “infinite” – loop breaks on time
    init_eta = 0.03 * target_sum           # initial step size (relative)
    beta = 0.9                             # momentum factor
    momentum = np.zeros_like(a)
    K = 5                                  # how many worst lags to attack simultaneously
    iter_cnt = 0

    while time.monotonic() < deadline and iter_cnt < max_iters:
        # Full convolution and the K largest lags
        conv = _full_convolution(a)
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            # argpartition is O(n); get indices of the K biggest values
            top_idx = np.argpartition(conv, -K)[-K:]

        # Gradient accumulation over the selected lags
        grad = np.zeros_like(a)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1, dtype=np.int64)
            j_range = k - i_range
            # ∂conv[k]/∂a_i = 2·a_{k‑i} (the factor 2 is irrelevant for the descent direction)
            grad[i_range] += a[j_range]

        # Project gradient onto the hyper‑plane ΣΔa_i = 0 (preserve total mass)
        grad -= grad.mean()

        # Simple adaptive step: linearly decay with remaining time
        elapsed = time.monotonic() - start_time
        frac = max(0.0, 1.0 - elapsed / (budget_s - 2.0))
        eta = init_eta * frac
        eta = max(eta, 1e-8)

        # Momentum update and candidate construction
        momentum = beta * momentum - eta * grad
        cand = a + momentum

        # Enforce box constraints and renormalise the L¹‑norm
        cand = np.clip(cand, 0.0, 1000.0)
        cand = _normalise(cand, target_sum)

        # Evaluate
        val = _score(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            a = cand                     # keep the improvement as the new base point
            print(f"[{iter_cnt:7d}] new best  score={val:.6f}")
        else:
            # Still move – this helps to keep the momentum alive
            a = cand

        # Occasional random multiplicative perturbation (escape plateaus)
        if iter_cnt % 400 == 0 and iter_cnt > 0:
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            pert = a * noise
            pert = np.clip(pert, 0.0, 1000.0)
            pert = _normalise(pert, target_sum)
            valp = _score(pert)
            if valp < best_val:
                best_val, best_seq = valp, pert.copy()
                a = pert
                print(f"[{iter_cnt:7d}] perturb  improved score={valp:.6f}")

        iter_cnt += 1

    print(f"[done] iterations={iter_cnt}, best score={best_val:.6f}")
    # Return as plain Python floats (the host will call evaluate_sequence on it)
    return [float(v) for v in best_seq]
