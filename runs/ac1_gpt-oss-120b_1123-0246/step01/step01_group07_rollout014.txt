import numpy as np
import time

def _evaluate_fast(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` for non‑negative vectors."""
    n = seq.size
    conv = np.convolve(seq, seq)          # full linear convolution
    max_b = conv.max()
    total = seq.sum()
    if total < 1e-12:                     # protection – should never happen here
        return float("inf")
    return float(2.0 * n * max_b / (total * total))

def _make_powerlaw(length: int, p: float, alpha: float = 0.0) -> np.ndarray:
    """
    Return a normalised decreasing power‑law sequence:
        a_i ∝ (i+1+α)^{-p}
    """
    idx = np.arange(1, length + 1, dtype=np.float64) + alpha
    seq = np.power(idx, -p)
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s == 0:
        return seq
    return seq / s

def _make_exponential(length: int, gamma: float) -> np.ndarray:
    """Return a normalised exponential decay a_i = exp(-γ·i)."""
    idx = np.arange(length, dtype=np.float64)
    seq = np.exp(-gamma * idx)
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s == 0:
        return seq
    return seq / s

def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) non‑negative multiplicative perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation – helps avoid a single dominant peak
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    cand = np.clip(cand, 0.0, 1000.0)
    total = cand.sum()
    if total > 0:
        cand = cand / total
    return cand

       homogeneous, we normalise each candidate to unit L¹ norm.

    3. **Stochastic hill‑climber** – starting from the best candidate
       found in the sweep, repeatedly apply cheap block‑wise multiplicative
       perturbations.  Occasionally we enforce a monotone‑decreasing order,
       which is known to lower the peak of the self‑convolution.

    exhausted and returns the best sequence encountered.
    """
