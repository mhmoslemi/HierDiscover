"""
Propose a good step‑function.

The algorithm starts from the Catalan‑type sequence (or from the globally
provided ``height_sequence_1``).  It then repeatedly solves a *linear program*
that maximises the total mass that can be added without increasing the current
worst convolution value `M = max(a*a)`.  The LP is solved with
``scipy.optimize.linprog`` (HiGHS).  The obtained direction `g` is mixed with
the current sequence using a few step sizes – the best improvement is kept.
If the LP does not yield a better point, a cheap random perturbation is tried.
The loop runs until the time budget expires (≈ 950 s) or a preset iteration
limit is hit.  The best sequence seen is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# optional imports – we fall back to a purely random search if SciPy is missing
# ----------------------------------------------------------------------
try:
    import scipy.sparse as sp
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:   # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# helper: Catalan‑type coefficients C_i = binom(2i,i)/4**i
# ----------------------------------------------------------------------
def catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑type numbers C_i = binom(2i,i)/4**i (C_0 = 1)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence a[i] = a[i‑1] * (2*i‑1)/(2*i)
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    # Clip to the domain enforced by evaluate_sequence
    return np.clip(a, 0.0, 1000.0)


# ----------------------------------------------------------------------
# build the (2n‑1)×n matrix that encodes the convolution constraints:
#    (a * g)_k = sum_i a_i * g_{k‑i}   for k = 0 … 2n‑2
# ----------------------------------------------------------------------
def build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """Return a sparse matrix A such that A @ g == np.convolve(a, g)."""
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    # each column i contributes a[i] on rows i … i+n‑1
    for i, ai in enumerate(a):
        if ai == 0.0:
            continue
        # vector of row indices for this column
        r = np.arange(i, i + n, dtype=np.int32)  # length n
        rows.append(r)
        cols.append(np.full(n, i, dtype=np.int32))
        data.append(np.full(n, ai, dtype=np.float64))
    if not rows:   # all zero case – should never happen for a valid start
        return sp.csr_matrix((2 * n - 1, n), dtype=np.float64)
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A.tocsr()


# ----------------------------------------------------------------------
# linear‑programming sub‑routine: maximise sum(g) s.t. a * g ≤ M
# ----------------------------------------------------------------------
def lp_maximise_sum(a: np.ndarray, M: float, eps: float = 1e-9):
    """Return a feasible g (numpy array) that maximises Σ g_i.
    Constraints:  conv(a, g) ≤ M + eps ,  g ≥ 0.
    Uses scipy.optimize.linprog (HiGHS).  Returns None on failure."""
    if not _SCIPY_AVAILABLE:          # pragma: no cover
        return None

    n = a.shape[0]
    # Build the constraint matrix once per call
    A = build_conv_matrix(a)                     # shape (2n‑1, n)
    ub = np.full(2 * n - 1, M + eps, dtype=np.float64)

    # Objective: maximise sum(g) -> minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    # bounds: g_i ≥ 0, no upper bound
    bounds = [(0, None)] * n

    # Solve the LP
    try:
        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception:               # pragma: no cover
        return None

    if res.success and res.x is not None:
        g = np.maximum(0.0, res.x)   # clean tiny negatives
        return g
    return None


# ----------------------------------------------------------------------
# main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence``.  The routine follows a Frank‑Wolfe style iteration:
    a linear program finds a feasible direction that increases the L1‑norm
    without worsening the L∞‑norm of the self‑convolution; a convex combination
    of the current point and this direction is accepted if it improves the
    evaluation score.  Random perturbations are used as a fallback.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # ------------------------------------------------------------------
    # 1. initialise the starting sequence
    # ------------------------------------------------------------------
    # Use the globally supplied sequence if it exists, otherwise Catalan.
    try:
        global height_sequence_1
        base = np.asarray(height_sequence_1, dtype=np.float64)
        # We limit the size; too large makes the LP huge.
        max_len = 2000
        if base.size > max_len:
            base = base[:max_len].copy()
        a = base
        n = a.size
    except Exception:
        n = 2000
        a = catalan_sequence(n)

    # Clamp to evaluator’s domain (just in case)
    a = np.clip(a, 0.0, 1000.0)

    # Initial score
    best = a.copy()
    try:
        best_score = float(evaluate_sequence(best.tolist()))
    except Exception:               # pragma: no cover
        best_score = float('inf')
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. iterative improvement
    # ------------------------------------------------------------------
    iteration = 0
    # small tolerance for the LP constraints
    eps = 1e-9
    # step sizes tried during mixing
    step_sizes = (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01)

    while time.monotonic() < deadline and iteration < 50:
        # Re‑compute the worst self‑convolution value
        conv_self = np.convolve(best, best)
        M = float(conv_self.max())

        # --------------------------------------------------------------
        # 2a. linear program – maximise total mass without raising M
        # --------------------------------------------------------------
        g = lp_maximise_sum(best, M, eps)
        improved = False

        if g is not None and np.isfinite(g).all():
            # --------------------------------------------------------------
            # 2b. try convex combinations of best and g
            # --------------------------------------------------------------
            for t in step_sizes:
                cand = (1.0 - t) * best + t * g
                cand = np.clip(cand, 0.0, 1000.0)
                try:
                    cand_score = float(evaluate_sequence(cand.tolist()))
                except Exception:
                    continue
                if cand_score < best_score:
                    best = cand
                    best_score = cand_score
                    improved = True
                    print(f"[iter {iteration:02d}] mix t={t:.2f} → score {best_score:.6f}")
                    break   # take the first improving step size

        # --------------------------------------------------------------
        # 2c. fallback random tweak if LP gave nothing useful
        # --------------------------------------------------------------
        if not improved:
            # multiplicative log‑normal perturbation (small variance)
            perturb = np.exp(rng.normal(0.0, 0.04, size=n))
            cand = best * perturb
            cand = np.clip(cand, 0.0, 1000.0)
            try:
                cand_score = float(evaluate_sequence(cand.tolist()))
            except Exception:
                cand_score = float('inf')
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                improved = True
                print(f"[iter {iteration:02d}] random tweak → score {best_score:.6f}")

        # If nothing improved for a few consecutive iterations, we stop early
        if not improved:
            # count consecutive failures
            if iteration > 0 and iteration % 10 == 0:
                print(f"[iter {iteration}] no improvement for a while – terminating.")
                break

        iteration += 1

    # ------------------------------------------------------------------
    # 3. final result
    # ------------------------------------------------------------------
    print(f"[final] finished after {iteration} iterations, best score = {best_score:.6f}")
    # Ensure plain Python list of floats
    return [float(v) for v in best]
