def compute_objective(a):
    a = np.array(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01: return np.inf
    # convolution via FFT
    n = a.size
    size = 2* n - 1
    # compute convolution with np.fft
    m = 1 << (size-1).bit_length()
    # zero pad a to m
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    max_b = conv.max()
    return 2 * n * max_b / (s * s)

def compute_objective(arr):
    a = np.asarray(arr, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01: return np.inf
    # convolution via FFT
    size = 2*n - 1
    m = 1 << ((size-1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]  # real
    # due to rounding, ensure non-negative
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)

     The sequence is scaled so that Σa_i = √(2n); with this
     normalisation the evaluation function reduces to the maximal
     convolution entry.
3.   The objective (max‑convolution) is computed cheaply with an FFT.
4.   The best (n,α) pair is kept, then a local golden‑section search
     refines α for that n.
5.   Finally the best sequence is evaluated with the provided
     `evaluate_sequence` (which also clips to [0,1000]) and returned.
The whole procedure respects the supplied time budget.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective (max of self‑convolution)
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray) -> float:
    """Return 2·n·max(b)/sum(a)^2, using FFT for the convolution."""
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    # convolution via FFT (next power of two for speed & accuracy)
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Helper: generate a symmetric Beta(α,α)‑shaped sequence for a given n
# ----------------------------------------------------------------------
def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Build a non‑negative sequence of length n with shape
    p^{α‑1}(1‑p)^{α‑1} where p = (i+0.5)/n.
    The sequence is scaled so that its sum equals sqrt(2·n).
    """
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Beta(α,α) density (up to a constant factor)
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, alpha - 1.0)
    target_sum = sqrt(2.0 * n)
    # Normalise
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    a = w * (target_sum / w_sum)
    # Respect the hard bounds enforced by evaluate_sequence
    a = np.clip(a, 0.0, 1000.0)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the runner
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    The search uses Beta‑shaped candidates (including the arcsine case)
    and a fast golden‑section refinement of the shape parameter.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)  # tiny safety margin

    best_seq = None
    best_score = np.inf
    best_n = None
    best_alpha = None

    # ------------------------------------------------------------------
    # 0.  Try the existing global sequence if it exists (fallback)
    # ------------------------------------------------------------------
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        seq0 = np.clip(seq0, 0.0, 1000.0)
        if seq0.ndim == 1 and seq0.size > 0:
            sc = _fast_objective(seq0)
            if sc < best_score:
                best_score = sc
                best_seq = seq0.copy()
                best_n = seq0.size
                best_alpha = None
                print(f"[init] global height_sequence_1 score = {best_score:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Grid search over lengths and shape parameters
    # ------------------------------------------------------------------
    n_candidates = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768]
    # α values – a dense grid around the arcsine value 0.5
    alpha_grid = np.concatenate((
        np.linspace(0.15, 0.9, 16),    # coarse sweep
        np.linspace(0.35, 0.65, 13)   # finer central region
    ))
    alpha_grid = np.unique(alpha_grid)

    for n in n_candidates:
        if time.monotonic() >= deadline:
            break
        target_sum = sqrt(2.0 * n)   # keep the scale that makes the objective = max_b
        for alpha in alpha_grid:
            if time.monotonic() >= deadline:
                break
            a = _beta_sequence(n, alpha)
            # quick (exact) objective – because we keep the sum = sqrt(2n)
            score = _fast_objective(a)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()
                best_n = n
                best_alpha = alpha
                print(f"[grid] n={n:5d} α={alpha:5.3f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Refine the best α for the selected n by golden‑section search
    # ------------------------------------------------------------------
    if best_seq is not None and best_alpha is not None and best_n is not None:
        # restrict refinement to a reasonable interval around the current α
        lo = max(0.05, best_alpha - 0.30)
        hi = min(1.5, best_alpha + 0.30)
        phi = (1.0 + 5 ** 0.5) / 2.0
        # initial interior points
        c = hi - (hi - lo) / phi
        d = lo + (hi - lo) / phi
        # evaluate once
        a_c = _beta_sequence(best_n, c)
        f_c = _fast_objective(a_c)
        a_d = _beta_sequence(best_n, d)
        f_d = _fast_objective(a_d)

        while time.monotonic() < deadline and (hi - lo) > 1e-4:
            if f_c < f_d:
                hi, a_d, f_d = d, a_c, f_c
                d = c
                c = hi - (hi - lo) / phi
                a_c = _beta_sequence(best_n, c)
                f_c = _fast_objective(a_c)
            else:
                lo, a_c, f_c = c, a_d, f_d
                c = d
                d = lo + (hi - lo) / phi
                a_d = _beta_sequence(best_n, d)
                f_d = _fast_objective(a_d)

            # keep the best seen so far
            if f_c < best_score:
                best_score = f_c
                best_seq = a_c.copy()
                best_alpha = c
            if f_d < best_score:
                best_score = f_d
                best_seq = a_d.copy()
                best_alpha = d

        print(f"[refine] n={best_n} α≈{best_alpha:.6f} → score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Optional tiny random perturbation sweep (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        # a quick stochastic fine‑tuning: move a tiny random amount of mass
        a = best_seq.copy()
        n = a.size
        target_sum = sqrt(2.0 * n)
        # step size relative to typical entry magnitude
        step_scale = target_sum / (20.0 * n)  # tiny
        while time.monotonic() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # propose moving a tiny amount from i to j
            delta = rng.random() * step_scale
            if a[i] < delta:
                continue
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            # renormalise (should stay near target sum, but clamp just in case)
            s = cand.sum()
            if s > 0:
                cand *= (target_sum / s)
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _fast_objective(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                a = cand
                # occasional log
                if rng.random() < 0.01:
                    print(f"[perturb] improved → score={best_score:.6f}")
            else:
                a = cand  # keep walking even if no improvement

    # ------------------------------------------------------------------
    # 4.  Final accurate evaluation (and final printout)
    # ------------------------------------------------------------------
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] evaluated with evaluate_sequence → score = {final_score:.6f}")

    # Return as a plain Python list
    return [float(v) for v in best_seq]
