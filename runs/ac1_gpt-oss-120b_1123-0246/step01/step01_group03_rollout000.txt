"""
Propose a step‑function using a Catalan start and a fast linear‑programming
refinement.  The LP maximises the ℓ1‑norm of a non‑negative vector g under the
convolution bound (a∗g)≤M where M is the current worst self‑convolution of a.
We solve the LP with SciPy’s HiGHS solver (fallback to CVXPY/SCS) and then
mix the current sequence a with g using a line‑search over step sizes.  Random
log‑normal perturbations are used when the LP gives no improvement.  The routine
stops when the time budget expires or after a fixed number of iterations,
returning the best sequence seen.
"""

import time
import numpy as np

# try to import the fast LP solver, fall back to cvxpy if needed
try:
    from scipy.optimize import linprog
    from scipy.sparse import diags, csr_matrix
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVXPY_AVAILABLE = False


def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan‑type coefficients C_i = binom(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    # enforce the limits required by `evaluate_sequence`
    return np.clip(a, 0.0, 1000.0)


def _build_conv_matrix(a: np.ndarray) -> csr_matrix:
    """
    Build the sparse matrix A such that A @ g = conv(a,g) (linear convolution).
    Each row k (0 ≤ k < 2n-1) contains the coefficients a_i that multiply g_{k-i}.
    """
    n = a.shape[0]
    # each diagonal offset d contains the constant a[d]
    diag_data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=np.int32)
    A = diags(diag_data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _solve_lp_scipy(a: np.ndarray, M: float) -> np.ndarray | None:
    """Solve max Σ g_i s.t. (a∗g)_k ≤ M, g≥0 using SciPy's HiGHS."""
    n = a.shape[0]
    A = _build_conv_matrix(a)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    # tiny slack to keep the LP feasible numerically
    b += 1e-9
    c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
    if res.success and res.x is not None:
        return np.maximum(0.0, res.x.astype(np.float64))
    return None


def _solve_lp_cvxpy(a: np.ndarray, M: float) -> np.ndarray | None:
    """Solve the same LP with CVXPY using the SCS solver."""
    n = a.shape[0]
    g = cp.Variable(n, nonneg=True)
    # cp.conv accepts a Python list/ndarray as the first argument
    conv_expr = cp.conv(a, g)  # length 2n‑1
    eps = 1e-9
    constraints = [conv_expr <= M + eps]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver="SCS", max_iters=25000, verbose=False)
    if g.value is not None:
        return np.maximum(0.0, np.asarray(g.value, dtype=np.float64))
    return None


def propose_candidate(seed: int = 123, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimise
    `evaluate_sequence`.  The algorithm starts from a Catalan‑type sequence,
    repeatedly solves a linear‑programming bound‑preserving LP and mixes the
    solution with the current sequence.  Random perturbations are used as a
    fallback when the LP yields no progress.  The best sequence found before the
    time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # ------------------------------------------------------------------
    # 1. Determine the working length and initialise the sequence.
    # ------------------------------------------------------------------
    # Prefer a previously discovered sequence if it exists, but cap the size
    # to keep the LP tractable.
    try:
        base_global = np.asarray(height_sequence_1, dtype=np.float64)  # provided globally
        n = min(2000, base_global.size)          # keep n moderate for speed
        base_global = base_global[:n]
    except Exception:
        base_global = None
        n = 1800                                 # a reasonable default

    # Catalan baseline
    catalan = _catalan_sequence(n)

    # Pick the better start between the global sequence (if any) and Catalan.
    if base_global is not None:
        try:
            score_base = float(evaluate_sequence(base_global.tolist()))
        except Exception:                     # safety – shouldn't happen
            score_base = np.inf
        score_cat = float(evaluate_sequence(catalan.tolist()))
        if score_base < score_cat:
            a = base_global.copy()
            best_score = score_base
            print(f"Starting from provided sequence, score={best_score:.6f}")
        else:
            a = catalan.copy()
            best_score = score_cat
            print(f"Starting from Catalan baseline, score={best_score:.6f}")
    else:
        a = catalan.copy()
        best_score = float(evaluate_sequence(a.tolist()))
        print(f"Starting from Catalan baseline, score={best_score:.6f}")

    best = a.copy()
    iteration = 0
    max_iters = 40                      # a safe upper bound
    step_sizes = (0.7, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01)

    # ------------------------------------------------------------------
    # 2. Main optimisation loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline and iteration < max_iters:
        iteration += 1

        # Current worst self‑convolution value
        conv_self = np.convolve(best, best)
        M = float(conv_self.max())

        # --------------------------------------------------------------
        # 2a. Solve the LP to obtain a candidate g with larger ℓ1‑norm.
        # --------------------------------------------------------------
        g = None
        if _SCIPY_AVAILABLE:
            g = _solve_lp_scipy(best, M)
            if g is None:
                print(f"Iter {iteration}: SciPy LP failed, trying CVXPY.")
        if g is None and _CVXPY_AVAILABLE:
            g = _solve_lp_cvxpy(best, M)
        if g is None:
            print(f"Iter {iteration}: LP could not be solved, skipping LP step.")
            # fall back to a pure random perturbation in this iteration
            g = None

        improved = False

        # --------------------------------------------------------------
        # 2b. Try mixing the current sequence with the LP solution.
        # --------------------------------------------------------------
        if g is not None:
            for t in step_sizes:
                candidate = (1.0 - t) * best + t * g
                candidate = np.clip(candidate, 0.0, 1000.0)
                try:
                    score = float(evaluate_sequence(candidate.tolist()))
                except Exception:
                    continue
                if np.isfinite(score) and score < best_score - 1e-9:
                    best = candidate
                    best_score = score
                    improved = True
                    print(f"Iter {iteration:02d}, mix t={t:.2f} → score {best_score:.6f}")
                    break  # accept first improvement

            # If mixing with any t failed, also test the pure LP solution.
            if not improved:
                try:
                    score_g = float(evaluate_sequence(g.tolist()))
                except Exception:
                    score_g = np.inf
                if np.isfinite(score_g) and score_g < best_score - 1e-9:
                    best = g.copy()
                    best_score = score_g
                    improved = True
                    print(f"Iter {iteration:02d}, pure LP g → score {best_score:.6f}")

        # --------------------------------------------------------------
        # 2c. If LP gave no improvement, apply a cheap random tweak.
        # --------------------------------------------------------------
        if not improved:
            # Log‑normal perturbation (multiplicative noise)
            sigma = 0.07
            perturb = np.exp(rng.normal(0.0, sigma, size=n))
            candidate = best * perturb
            candidate = np.clip(candidate, 0.0, 1000.0)
            try:
                score = float(evaluate_sequence(candidate.tolist()))
            except Exception:
                score = np.inf
            if np.isfinite(score) and score < best_score - 1e-9:
                best = candidate
                best_score = score
                improved = True
                print(f"Iter {iteration:02d}, random tweak → score {best_score:.6f}")

        # --------------------------------------------------------------
        # 2d. If nothing improved, we shrink the step size schedule and keep looping.
        # --------------------------------------------------------------
        if not improved:
            # Very small random smoothing – sometimes helps escape plateaus
            sigma_small = 0.02
            perturb = np.exp(rng.normal(0.0, sigma_small, size=n))
            candidate = best * perturb
            candidate = np.clip(candidate, 0.0, 1000.0)
            try:
                score = float(evaluate_sequence(candidate.tolist()))
            except Exception:
                score = np.inf
            if np.isfinite(score) and score < best_score - 1e-9:
                best = candidate
                best_score = score
                print(f"Iter {iteration:02d}, fine tweak → score {best_score:.6f}")

        # tiny progress report every few iterations
        if iteration % 5 == 0:
            print(f"Iter {iteration:02d} best score so far: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return [float(v) for v in best]
