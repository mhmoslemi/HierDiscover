"""
Sub‑gradient optimisation of the ‖f*f‖∞ / ‖f‖₁² objective.

The algorithm starts from the analytically motivated shape
    a_i ∝ 1 / √(i+1)
(which makes the continuous self‑convolution constant).
We normalise the L¹‑norm to √(2 n) (the same normalisation as the reference
paper) and then perform a constrained sub‑gradient descent on the
max‑convolution value.

Key ideas
---------

* The objective depends only on the shape of the sequence (scale‑invariant).
* For a given shape we can compute the full convolution fast with an FFT.
* The max‑operator is approximated by a smooth “log‑sum‑exp’’ weighting:
      w_k ∝ exp(α·(b_k – max(b)))    (α grows during the run)
  which yields a sub‑gradient that simultaneously pushes down all large
  convolution entries, not only the current maximum.
* After each gradient step we project onto the hyper‑plane of zero‑sum
  (to keep the total mass fixed) and re‑normalise to the target sum.
* Occasional multiplicative perturbations and a simple restart strategy
  help to escape plateaus.

The routine is time‑bounded (default 900 s, leaving a safety margin) and
returns the best sequence found.  All constraints required by
`evaluate_sequence` (non‑negative, ≤ 1000, sum ≥ 0.01) are respected.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: linear convolution via FFT (full, real‑valued result)
# ----------------------------------------------------------------------
def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full discrete convolution of two 1‑D real arrays using FFT."""
    n = x.size + y.size - 1
    # next power of two for efficient FFT
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    Z = X * Y
    conv = np.fft.irfft(Z, n=size)[:n]
    return conv


# ----------------------------------------------------------------------
# Helper: fast computation of the objective using the known target sum
# ----------------------------------------------------------------------
def fast_score(a: np.ndarray, target_sum: float) -> float:
    """2 · n · max(b) / target_sum²  (where b = a*a)."""
    n = a.size
    b = fft_convolve(a, a)
    max_b = np.max(b)
    return 2.0 * n * max_b / (target_sum * target_sum)

# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 123,
    budget_s: float = 900.0,
    *,
    n: int | None = None,
) -> list[float]:
    """
    Search for a low‑score step‑height sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function stops a few seconds before this).
    n : int | None
        Length of the sequence.  If omitted, the length of the global
        ``height_sequence_1`` is used when available, otherwise 5600
        (similar to the previous best construction).

    Returns
    -------
    list[float]
        The best sequence discovered (non‑negative, ≤ 1000).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose length and target L1‑norm
    # ------------------------------------------------------------------
    if n is None:
        try:
            n = int(np.asarray(height_sequence_1).size)  # type: ignore[name-defined]
        except Exception:
            n = 5600                       # a safe default, similar to past best
    target_sum = np.sqrt(2.0 * n)           # the normalisation used in the paper

    # ------------------------------------------------------------------
    # 2. Initialise a reasonable shape: a_i ∝ 1/√(i+1)
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(idx)                 # decreasing, large at the left
    # small random jitter (helps breaking symmetry)
    a *= 1.0 + 0.02 * rng.normal(size=n)
    a = np.clip(a, 0.0, 1000.0)

    # Normalise to the required L1‑norm
    s = a.sum()
    if s > 0.0:
        a *= (target_sum / s)
    a = np.clip(a, 0.0, 1000.0)

    best = a.copy()
    best_score = fast_score(best, target_sum)
    print(f"[init] n={n}, initial fast score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # 3. Sub‑gradient loop (smooth max approximation)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)  # safety margin
    start = time.monotonic()
    iteration = 0

    # schedule for the smooth‑max temperature α
    alpha_start = 0.5
    alpha_end = 20.0

    # step‑size schedule (linear decay)
    eta_initial = 0.02 * target_sum          # a few percent of the typical entry size
    eta_min = 1e-10

    while time.monotonic() < deadline:
        # ---- full convolution and smooth‑max weights ----
        b = fft_convolve(a, a)                # length 2n‑1
        max_b = np.max(b)

        # linear interpolation of α between start and end
        elapsed = time.monotonic() - start
        frac = min(1.0, elapsed / budget_s)
        alpha = alpha_start + frac * (alpha_end - alpha_start)

        # smooth‑max weighting (log‑sum‑exp trick for stability)
        # w_k ∝ exp(α·(b_k – max_b))  ∈ (0,1]
        w = np.exp(alpha * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:                      # should never happen, but guard
            w = (b == max_b).astype(np.float64)
            w_sum = w.sum()
        w /= w_sum

        # ---- gradient of the smooth‑max objective ----
        # ∂/∂a_i  Σ_k w_k·b_k = Σ_k w_k·2·a_{k‑i}
        a_rev = a[::-1]
        grad_full = fft_convolve(a_rev, w)           # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # slice the part that matches a

        # ---- enforce the zero‑sum constraint (preserve total mass) ----
        grad -= grad.mean()

        # ---- step size (linear decay) ----
        eta = eta_initial * max(0.0, 1.0 - frac)
        eta = max(eta, eta_min)

        # ---- gradient step, clipping and re‑normalisation ----
        cand = a - eta * grad
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > 0.0:
            cand *= (target_sum / s)
        cand = np.clip(cand, 0.0, 1000.0)

        # evaluate fast score (exact score will be computed only occasionally)
        cur_score = fast_score(cand, target_sum)

        # ---- accept improvements ----
        if cur_score < best_score:
            best = cand.copy()
            best_score = cur_score
            print(f"[{iteration:6d}] new best  score = {best_score:.8f}")

        a = cand

        # ---- occasional random multiplicative perturbation (escape plateaus) ----
        if iteration % 2000 == 0 and iteration > 0:
            pert = np.exp(rng.normal(0.0, 0.04, size=n))
            cand2 = a * pert
            cand2 = np.clip(cand2, 0.0, 1000.0)
            s2 = cand2.sum()
            if s2 > 0.0:
                cand2 *= (target_sum / s2)
            cand2 = np.clip(cand2, 0.0, 1000.0)
            score2 = fast_score(cand2, target_sum)
            if score2 < best_score:
                best = cand2.copy()
                best_score = score2
                a = cand2
                print(f"[{iteration:6d}] perturb improved score = {best_score:.8f}")

        # ---- simple restart if stagnation persists for a long time ----
        if iteration % 15000 == 0 and iteration > 0:
            # re‑initialise with a fresh 1/√ shape plus jitter
            a = 1.0 / np.sqrt(idx) * (1.0 + 0.02 * rng.normal(size=n))
            a = np.clip(a, 0.0, 1000.0)
            s = a.sum()
            if s > 0.0:
                a *= (target_sum / s)
            a = np.clip(a, 0.0, 1000.0)
            print(f"[{iteration:6d}] restart performed")

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Final verification (optional, cheap)
    # ------------------------------------------------------------------
    # The external judge will call evaluate_sequence, but we print a quick check.
    try:
        final_score = float(evaluate_sequence(best.tolist()))   # type: ignore[name-defined]
        print(f"[final] best sequence fast score = {best_score:.8f}, "
              f"official score = {final_score:.8f}")
    except Exception:
        # In case the function is not available in this sandbox
        print("[final] official evaluate_sequence not available; using fast score only")

    return [float(v) for v in best]
