"""
propose_candidate
-----------------
Search for a short power‑law step function a_i = (i+1)^(-p) that gives a
low evaluation score.  The search varies both the length of the
sequence (n) and the exponent (p).  After a cheap grid search a simple
block‑perturbation hill‑climber refines the best candidate until the
time budget runs out.

The main idea is that for p≈0.6 the convolution max grows only like
n^{1‑2p} (≈n^{-0.2}) while the L1‑norm grows like n^{1‑p}
(≈n^{0.4}).  Hence the score 2·n·max_b / (sum_a²) ≈ 0.32·n^{0.2}
decreases when n is reduced.  By trying lengths in the few‑hundred to
few‑thousand range we can beat the previous best (≈1.578) and reach the
target < 1.503.
"""

import time
import numpy as np

def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    One O(n) non‑negative perturbation of ``base``.
    The perturbation multiplies a random block by a log‑normal factor.
    """
    cand = base.copy()
    n = cand.size
    # block width roughly between n/200 and n/12
    wmin = max(1, n // 200)
    wmax = max(2, n // 12)
    width = int(rng.integers(wmin, wmax + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    # multiplicative log‑normal perturbation
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation (often useful for convolution problems)
    if rng.random() < 0.35:
        cand[:] = 0.5 * (cand + cand[::-1])
    # enforce hard limits of evaluate_sequence
    return np.clip(cand, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 500.0, **kwargs):
    """
    Search for a step‑function that minimises the provided evaluation function.
    The function respects the given time budget (in seconds) and returns the
    best sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 1. Length candidates – a few short lengths are typically better
    # ------------------------------------------------------------------
    n_candidates = [512, 768, 1024, 1536, 2048]          # primary grid
    # also try a length derived from the reference construction if it exists
    try:
        n_ref = int(len(height_sequence_1))
        if 200 <= n_ref <= 3000 and n_ref not in n_candidates:
            n_candidates.append(n_ref)
    except Exception:
        pass
    n_candidates = sorted(set(n_candidates))

    best_seq = None
    best_score = float("inf")
    best_info = None   # (n, p) pair that produced the best result

    # ------------------------------------------------------------------
    # 2. Coarse grid search over length n and exponent p
    # ------------------------------------------------------------------
    for n in n_candidates:
        # coarse exponent sweep – denser around the region known to be good
        exponents = np.concatenate([
            np.linspace(0.45, 0.55, 12, endpoint=False),
            np.linspace(0.55, 0.75, 20, endpoint=False),
            np.linspace(0.75, 0.90, 10)
        ])

        for p in exponents:
            # power‑law shape a_i = (i+1)^{-p}, scaled so that the first entry is 1000
            seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
            seq = seq * (1000.0 / seq.max())          # scaling (max becomes 1000)

            sc = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                best_info = (n, p)
                print(f"[grid] n={n:4d} p={p: .4f} → score={sc:.6f}")

        # ------------------------------------------------------------------
        # 3. Small local refinement around the best exponent for this n
        # ------------------------------------------------------------------
        if best_seq is not None and best_info[0] == n:
            # fine‑grained search around the best exponent found for this n
            p_center = best_info[1]
            for p in np.linspace(max(0.1, p_center - 0.03), p_center + 0.03, 7):
                seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
                seq = seq * (1000.0 / seq.max())
                sc = float(evaluate_sequence(seq.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    best_info = (n, p)
                    print(f"[refine] n={n:4d} p={p: .4f} → score={sc:.6f}")

        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. If nothing was found (should not happen) fall back to a tiny uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(256, 500.0, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] using uniform default sequence")

    print(f"[info] Best after grid search: n={best_info[0]}, p={best_info[1]:.4f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Hill‑climbing with block perturbations until time runs out
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = make_block_perturbation(best_seq, rng)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_seq = cand
            best_score = sc
            print(f"[improve] new best score = {best_score:.6f}")

    print(f"[done] final best score = {best_score:.6f}")
    return best_seq.tolist()
