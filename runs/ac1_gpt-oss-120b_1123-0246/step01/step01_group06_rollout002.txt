def perturb(seq, rng, sigma):
    n = len(seq)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0, sigma))
    new = seq.copy()
    new[start:start+width] *= factor
    # maybe symmetry
    if rng.random() < 0.3:
         new = 0.5*(new + new[::-1])
    # Clip
    np.clip(new, 0, 1000, out=new)
    return new
```

"""
propose_candidate
-----------------
A two‑stage optimizer for the step‑function bound.

Stage 1 – Global sweep.
    Randomly samples a few parametric families (power‑law, exponential,
    combined power‑law × exponential and a mirrored decay).  Each candidate
    is scaled so that its maximum entry equals 1000 (scaling does not
    affect the score) and the best one is kept.

Stage 2 – Simulated‑annealing hill‑climber.
    Starts from the best sequence of stage 1 (or from the provided
    `height_sequence_1` if it is available) and repeatedly applies a
    multiplicative log‑normal perturbation on a random block.  The
    perturbation size (σ) and the acceptance temperature decrease
    linearly with the remaining time, allowing large exploratory moves
    early on and fine‑grained greedy improvements later.  The current
    sequence is kept if it improves the evaluation; otherwise it may be
    accepted with a probability exp(‑Δ/​T) to escape local minima.

All numbers stay within the hard limits of ``evaluate_sequence``
(i.e. 0 ≤ aᵢ ≤ 1000).  The algorithm respects the 1000 s wall‑clock
budget and returns the best list it has found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _eval(seq: np.ndarray) -> float:
    """
    Wrapper around the provided ``evaluate_sequence``.
    Returns ``inf`` for non‑finite results.
    """
    try:
        val = evaluate_sequence(seq.tolist())
        return float(val)
    except Exception:               # safety – should never happen
        return float('inf')


def _make_sequence(n: int,
                   shape: str,
                   p: float = 0.6,
                   c: float = 0.0) -> np.ndarray:
    """
    Generate a deterministic sequence of length ``n``.
    Supported ``shape`` values:
        * 'decay'      – a_i = (i+1)^{-p}
        * 'exp'        – a_i = exp(-c·i)
        * 'exp_decay'  – a_i = (i+1)^{-p}·exp(-c·i)
        * 'mirror'     – high values at both ends,
                         a_i = (i+1)^{-p} for i < n/2,
                         a_i = (n-i)^{-p} for i ≥ n/2
    The result is scaled so that its maximum equals 1000.
    """
    idx = np.arange(n, dtype=np.float64)

    if shape == 'decay':
        seq = (idx + 1.0) ** (-p)
    elif shape == 'exp':
        seq = np.exp(-c * idx)
    elif shape == 'exp_decay':
        seq = (idx + 1.0) ** (-p) * np.exp(-c * idx)
    elif shape == 'mirror':
        half = n // 2
        left = (idx[:half] + 1.0) ** (-p)
        right = (np.arange(n - half)[::-1] + 1.0) ** (-p)
        seq = np.concatenate([left, right])
        # If n is odd the centre element would be missing; fill with a small value
        if seq.size < n:
            seq = np.append(seq, 0.0)
    else:
        raise ValueError(f'unknown shape {shape!r}')

    # Scale so that the maximal entry hits the clipping bound.
    # This does not change the evaluation score (the formula is
    # homogeneous of degree 0) but keeps the numbers away from the
    # 0‑to‑1000 limits.
    if seq.max() == 0:
        seq[:] = 0.0
    else:
        seq *= 1000.0 / seq.max()
    return seq


def _perturb_block(base: np.ndarray,
                   rng: np.random.Generator,
                   sigma: float) -> np.ndarray:
    """
    Returns a copy of ``base`` where a random contiguous block has been
    multiplied by a log‑normal factor `exp(N(0, sigma²))`.  With a modest
    probability the whole vector is symmetrised to encourage
    autocorrelation‑friendly shapes.
    """
    cand = base.copy()
    n = cand.size
    # width between 1 and roughly n/10 (adaptable)
    max_width = max(1, n // 10)
    width = int(rng.integers(1, max_width + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand[start:start + width] *= factor

    # occasional symmetrisation (helps for the convolution problem)
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])

    # enforce hard limits of evaluate_sequence
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _local_hill_climb(seq: np.ndarray,
                     rng: np.random.Generator,
                     deadline: float) -> np.ndarray:
    """
    Simple greedy hill‑climbing with decreasing block size.
    Stops when the time budget expires.
    """
    best = seq.copy()
    best_score = _eval(best)
    n = best.size
    while time.monotonic() < deadline:
        # gradually shrink the block width as time passes
        remaining = deadline - time.monotonic()
        total = deadline - start_time
        ratio = max(0.0, min(1.0, (total - remaining) / total))
        sigma = 0.4 * (1.0 - ratio) + 0.02 * ratio   # from 0.4 → 0.02
        cand = _perturb_block(best, rng, sigma)
        sc = _eval(cand)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
            print(f"[hill] new best = {best_score:.6f}")
    return best

# Main entry point required by the grading harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimizes ``evaluate_sequence``.  Returns the best list found before
    the wall‑clock budget elapses.
    """
    rng = np.random.default_rng(seed)
    # leave a short safety margin for the harness that kills the process
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1. Choose a workable length (≤ 2000 to keep O(n²) ops cheap)
    # ------------------------------------------------------------------
    try:
        n_full = int(len(height_sequence_1))          # type: ignore[name‑defined]
        n = min(n_full, 2000)
        from_height = np.array(height_sequence_1, dtype=np.float64)[:n]   # type: ignore[name‑defined]
        np.clip(from_height, 0.0, 1000.0, out=from_height)
        print(f"[info] Using truncated reference length n={n}")
    except Exception:
        n = 1024
        from_height = None
        print(f"[info] No reference sequence – using n={n}")

    # ------------------------------------------------------------------
    # 2. Global parametric sweep (a few thousand cheap candidates)
    # ------------------------------------------------------------------
    shape_options = ['decay', 'exp', 'exp_decay', 'mirror']
    best_seq = None
    best_score = float('inf')

    # include the reference (if any) as a starting point
    if from_height is not None:
        sc = _eval(from_height)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = from_height.copy()
            print(f"[init] reference score = {best_score:.6f}")

    # random sampling of parametrised families
    for _ in range(1500):
        shape = rng.choice(shape_options)
        p = rng.uniform(0.30, 0.90)          # exponent for power laws
        c = rng.uniform(0.0, 0.005)          # exponential decay rate
        seq = _make_sequence(n, shape, p, c)

        # small random symmetry flip (helps a bit for some shapes)
        if rng.random() < 0.15:
            seq = 0.5 * (seq + seq[::-1])

        sc = _eval(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[grid] shape={shape:9s} p={p:.3f} c={c:.5f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing hill‑climbing
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall back to a simple power‑law
        best_seq = _make_sequence(n, 'decay', p=0.6, c=0.0)
        best_score = _eval(best_seq)

    print(f"[info] Starting SA, initial best = {best_score:.6f}")

    # SA schedule (linear cooling)
    T_start = 0.02
    T_final = 1e-4
    sigma_start = 0.4
    sigma_final = 0.02

    current_seq = best_seq.copy()
    current_score = best_score
    start_time = time.monotonic()
    total_time = deadline - start_time

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        prog = min(1.0, elapsed / total_time)

        # linearly decreasing temperature and perturbation magnitude
        T = T_start * (1.0 - prog) + T_final * prog
        sigma = sigma_start * (1.0 - prog) + sigma_final * prog

        cand = _perturb_block(current_seq, rng, sigma)
        cand_score = _eval(cand)
        if not np.isfinite(cand_score):
            continue

        delta = cand_score - current_score
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            # accept
            current_seq = cand
            current_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[improve] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final short greedy hill‑climb (tiny σ) to polish the result
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        # a few seconds of pure greedy improvement
        extra_deadline = min(deadline, time.monotonic() + 2.0)
        best_seq = _local_hill_climb(best_seq, rng, extra_deadline)
        best_score = _eval(best_seq)

    print(f"[done] final best score = {best_score:.6f}")
    return best_seq.tolist()
