def propose_candidate(seed=0, budget_s=950):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    # Determine n
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 2000

    # generate initial shape: power law exponent p = 0.65 (choose)
    p = 0.65
    base = (np.arange(1, n+1) ** (-p))
    # Normalize to max 1000
    base = base * (1000.0 / np.max(base))
    # initial ordering: bit-reversal perm (if n power of 2 else fallback)
    order = np.arange(n)
    if (n & (n-1)) == 0:  # power of two
        bits = int(np.log2(n))
        rev = np.arange(n)
        for i in range(n):
            b = format(i, f'0{bits}b')
            rev[i] = int(b[::-1], 2)
        order = rev
    else:
        # simple high-low interleaving
        sorted_vals = np.sort(base)[::-1]
        order = np.empty(n, dtype=int)
        lo, hi = 0, n-1
        idx = 0
        while lo <= hi:
            order[idx] = lo
            idx += 1
            if lo != hi:
                order[idx] = hi
                idx += 1
            lo += 1
            hi -= 1

    seq = base[order]
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n} init score={best_score:.6f}")

    # Simulated annealing parameters
    temperature = 1.0
    # schedule: temperature decays linearly to 0.001 as time goes
    start_time = time.monotonic()

    iteration = 0
    while time.monotonic() < deadline:
        # update temperature based on elapsed fraction
        t_elapsed = time.monotonic() - start_time
        frac = min(1.0, t_elapsed / (deadline - start_time))
        temperature = max(0.001, 1.0 * (1 - frac))

        # propose move: swap two random positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand = best_seq.copy()
        cand[i], cand[j] = cand[j], cand[i]

        # occasionally apply multiplicative perturbation on a block
        if rng.random() < 0.1:
            width = rng.integers(max(1, n // 200), max(2, n // 12))
            start = rng.integers(0, max(1, n - width + 1))
            cand[start:start+width] *= np.exp(rng.normal(0.0, 0.12))

        # enforce clip bounds (though scaling done later)
        cand = np.clip(cand, 0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(score):
            continue

        # accept if better or by probability
        delta = score - best_score
        if delta < 0 or rng.random() < np.exp(-delta / temperature):
            best_seq = cand
            if score < best_score:
                best_score = score
                print(f"[iter {iteration}] new best {best_score:.6f} temp={temperature:.4f}")

        iteration += 1

    print(f"[done] final best {best_score:.6f}")
    return best_seq.tolist()

"""
propose_candidate
-----------------
Simulated‑annealing optimizer for the “step‑function” bound.

*   Starts from a power‑law shape a_i = (i+1)^(-p) with a fine
    search over the exponent p (the shape is scale‑invariant).
*   The initial ordering is the “alternating extremes” pattern which
    tends to keep large values away from each other when the sequence
    is convolved with itself.
*   A long simulated‑annealing loop then proposes three kinds of
    moves:
        – swap two random positions,
        – reverse a random block,
        – multiplicative log‑normal perturbation of a block.
    Moves are accepted if they improve the score or with a probability
    that falls with a linear temperature schedule.
*   The best sequence seen so far is kept and finally returned
    (as a plain Python list) before the time budget expires.

The score is the value returned by ``evaluate_sequence`` – the lower
the better.  The algorithm is deliberately simple, uses only NumPy,
and respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – compute the evaluation score
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """
    Return evaluate_sequence(seq) as a float.
    seq must be a 1‑D NumPy array of non‑negative numbers.
    """
    # evaluate_sequence expects a *list* and rejects non‑list inputs.
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Helper – generate a power‑law shape and an alternating‑extremes order
# ----------------------------------------------------------------------
def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a good starting sequence of length n.
    1. Grid‑search a power‑law exponent p (very cheap – only a few
       candidates, because the score does not depend on the absolute
       scale).
    2. Apply the “alternating‑extremes” permutation:
       largest, smallest, second‑largest, second‑smallest, …
    """
    # ---- 1. fine grid on p ------------------------------------------------
    # The exponent is searched in a narrow window around the known
    # sweet‑spot (≈0.6).  21 points give a resolution of 0.01.
    p_candidates = np.linspace(0.55, 0.70, 21)
    best_p = p_candidates[0]
    best_val = np.inf
    base = None
    for p in p_candidates:
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        # Scale to the allowed maximum (does not change the score)
        seq = seq * (1000.0 / seq.max())
        v = _score(seq)
        if np.isfinite(v) and v < best_val:
            best_val = v
            best_p = p
            base = seq
    if base is None:
        # fallback – plain uniform sequence
        base = np.full(n, 500.0, dtype=np.float64)

    # ---- 2. alternating‑extremes permutation -----------------------------
    # Sort the values descending, then interleave the largest and the
    # smallest.  This tends to keep large*large products rare in the
    # convolution.
    sorted_vals = np.sort(base)[::-1]            # descending
    lo, hi = 0, n - 1
    order = np.empty(n, dtype=int)
    pos = 0
    while lo <= hi:
        order[pos] = lo
        pos += 1
        if lo != hi:
            order[pos] = hi
            pos += 1
        lo += 1
        hi -= 1
    # apply permutation
    seq = sorted_vals[order]
    return seq

# Main entry point required by the evaluator
# ----------------------------------------------------------------------

    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the problem size (use the previously found sequence if
    # it exists – it is often already near the optimum).
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))

        start_seq = np.asarray(height_sequence_1, dtype=np.float64).copy()

        # make sure it respects the hard limits (the evaluator also does
        # this but we keep the array tidy)

        print(f"[info] Starting from provided sequence, n={n}")
    except Exception:
        # Fallback: moderate length (fast enough to evaluate many candidates)
        n = 4000
        start_seq = _initial_sequence(n, rng)
        print(f"[info] No provided sequence – generated n={n}")

    # ------------------------------------------------------------------
    # Initial scoring
    # ------------------------------------------------------------------
    cur_seq = start_seq
    cur_score = _score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"[init] score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)  # a little safety margin
    start_time = time.monotonic()
    # Temperature will drop linearly from 1.0 to 0.001
    T_initial = 1.0
    T_final = 0.001

    it = 0
    while time.monotonic() < deadline:
        # ----- temperature schedule (linear) -----
        elapsed = time.monotonic() - start_time
        fraction = min(1.0, elapsed / (deadline - start_time))
        temperature = T_initial * (1.0 - fraction) + T_final * fraction

        # ----- pick a move type -----
        move_type = rng.integers(0, 3)   # 0: swap, 1: reverse block, 2: block perturb

        cand = cur_seq.copy()

        if move_type == 0:                     # swap two positions

            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start:stop] *= factor

        # enforce the hard limits of evaluate_sequence
        cand = np.clip(cand, 0.0, 1000.0)

        # ----- evaluate candidate -----
        cand_score = _score(cand)
        if not np.isfinite(cand_score):
            # ignore pathological candidates (should not happen)
            continue

        # ----- acceptance test -----
        delta = cand_score - cur_score
        if delta < 0.0 or rng.random() < np.exp(-delta / temperature):
            cur_seq = cand
            cur_score = cand_score

            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[it {it:06d}] new best = {best_score:.6f}  T={temperature:.4f}")

        it += 1

    print(f"[done] final best score = {best_score:.6f} after {it} iterations")
    return best_seq.tolist()
