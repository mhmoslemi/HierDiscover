```python
def blocks_to_sequence(b, n):
    m = len(b)
    base = n // m
    remainder = n % m
    sizes = np.full(m, base, dtype=int)
    sizes[:remainder] += 1
    a = np.concatenate([np.full(sizes[i], b[i], dtype=np.float64) for i in range(m)])
    return a[:n]  # ensure length
```

```python
import numpy as np
import time

def build_arcsine_sequence(length):
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i-1] * ((2*i-1)*(2*i)) / (4.0*i*i)
    return a

def configure_fft_size(n):
    size = 1
    while size < 3*n:  # to accommodate mask*rev convolution
        size <<= 1
    return size

def convolution_fft(x, fft_size):
    X = np.fft.rfft(x, fft_size)
    c = np.fft.irfft(X * X, fft_size)
    return c[:2*len(x)-1]

def compute_score(a, fft_size):
    conv = convolution_fft(a, fft_size)
    max_val = conv.max()
    sum_val = a.sum()
    if sum_val < 0.01:
        return np.inf, conv, max_val
    score = 2.0 * len(a) * max_val / (sum_val * sum_val)
    return score, conv, max_val
```

def propose_candidate(seed=42, budget_s=500, **kwargs):
    """
    Simulated‑annealing local search starting from the analytically optimal
    arcsine sequence.  Random block scalings break the perfect constant‑convolution
    structure, then a sensitivity‑driven “mass‐transfer” step moves weight from
    indices that contribute most to the current maximal convolution towards
    indices that contribute least.  The combination quickly drives the maximum
    of the self‑convolution below the arcsine bound, improving the score.
    """
    rng = np.random.default_rng(seed)
    n = int(len(globals().get("height_sequence_1", []))) if "height_sequence_1" in globals() else 256
    if n == 0:
        n = 256

    # initialise sequence
    a = build_arcsine_sequence(n)
    a = np.clip(a, 0.0, 1000.0)
    fft_size = configure_fft_size(n)
    best_seq = a.copy()
    best_score, conv_cur, max_cur = compute_score(a, fft_size)
    score_cur = best_score
    print(f"[init] n={n} initial score={best_score:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)
    it = 0
    # hyper‑parameters
    sigma0 = 0.15  # initial std. dev. for multiplicative block perturbation
    T0 = 0.01

    while time.monotonic() < deadline:
        it += 1
        # progress based temperature and sigma
        elapsed = time.monotonic() - start_time
        progress = elapsed / budget_s
        sigma = sigma0 * (1.0 - progress) + 0.02 * progress   # anneal to 0.02
        T = T0 * (1.0 - progress) + 1e-6

        if rng.random() < 0.5:
            # block multiplicative perturbation
            w_min = max(1, n // 200)
            w_max = max(2, n // 12)
            width = int(rng.integers(w_min, w_max + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            cand = a.copy()
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand[start:start + width] *= factor
        else:
            # sensitivity driven transfer
            # build mask of entries that are “almost maximal”
            eps = max(1e-9, 1e-6 * max_cur)    # tolerance
            mask = (conv_cur >= max_cur - eps).astype(np.float64)
            # compute sensitivity g_i = sum_{k in mask} a_{k-i}
            a_rev = a[::-1]
            mask_fft = np.fft.rfft(mask, fft_size)
            a_rev_fft = np.fft.rfft(a_rev, fft_size)
            sens = np.fft.irfft(mask_fft * a_rev_fft, fft_size)
            g = sens[n - 1:n - 1 + n]   # length n
            # pick source (high sensitivity) and target (low)
            src = int(np.argmax(g))
            dst = int(np.argmin(g))
            if src == dst:

            delta = frac * a[src]
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        cand = np.clip(cand, 0.0, 1000.0)

        # evaluate candidate
        cand_score, cand_conv, cand_max = compute_score(cand, fft_size)

        if cand_score < score_cur or rng.random() < np.exp(-(cand_score - score_cur) / T):

            max_cur = cand_max

"""
Simulated‑annealing local search for the self‑convolution bound.

The algorithm:
    1. Build the analytically optimal “arcsine’’ sequence (a_i = C·C(2i,i)/4^i).
    2. Repeatedly apply two kinds of stochastic moves:
          – a multiplicative perturbation of a random contiguous block,
          – a sensitivity‑driven transfer of mass from an index that
            contributes most to the current maximal convolution value to an
            index that contributes least.
    3. Acceptance follows a simple simulated‑annealing schedule,
       allowing occasional uphill steps to escape plateaus.
    4. The best sequence seen is kept and finally returned.
All operations use FFT‑based convolution (O(n log n)) and work directly on
the full‑resolution array (no block‑wise compression), enabling thousands of
iterations within the given time budget.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# 1.  Analytical starting point – the arcsine (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    return a


# ----------------------------------------------------------------------
# 2.  FFT utilities – linear convolution via zero‑padded FFTs.
# ----------------------------------------------------------------------
def _fft_size_for(n: int) -> int:
    """Return a power‑of‑two size big enough for all convolutions we need."""
    size = 1
    # we convolve arrays of length up to 2n‑1 with arrays of length n.
    # The worst case length is (2n‑1) + n‑1 = 3n‑2.
    while size < 3 * n:
        size <<= 1
    return size


def convolution_self(a: np.ndarray, fft_size: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (returns length 2·len(a)‑1)."""
    # rfft / irfft give real‑valued result for real input.
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)
    return c[: 2 * len(a) - 1]


def sensitivity_vector(a: np.ndarray,
                       mask: np.ndarray,
                       fft_size: int) -> np.ndarray:
    """
    For a given mask (shape 2·n‑1) representing the active set of
    constraints, return the vector g where
        g[i] = Σ_{k:mask_k=1} a_{k‑i}
    (indices outside the array are treated as zero).
    """
    a_rev = a[::-1]
    M = np.fft.rfft(mask, fft_size)
    R = np.fft.rfft(a_rev, fft_size)
    s = np.fft.irfft(M * R, fft_size)  # length = fft_size
    # the relevant slice starts at index n‑1
    n = len(a)
    return s[n - 1 : n - 1 + n]  # length n


# ----------------------------------------------------------------------
# 3.  Scoring routine – identical to evaluate_sequence but using FFT.
# ----------------------------------------------------------------------
def _score_of(a: np.ndarray, fft_size: int):
    """
    Compute the evaluation score, together with the full convolution
    and its maximal value.
    """
    conv = convolution_self(a, fft_size)
    max_val = conv.max()
    sum_val = a.sum()
    if sum_val < 0.01:
        return float("inf"), conv, max_val
    score = 2.0 * len(a) * max_val / (sum_val * sum_val)
    return float(score), conv, max_val


# ----------------------------------------------------------------------
# 4.  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs):
    """
    Return a list of floats – the best step‑function heights found within
    ``budget_s`` seconds.

    The search starts from the arcsine sequence, then performs a large
    number of stochastic moves (block scaling and a sensitivity‑driven
    mass‑transfer) accepted according to a simple simulated‑annealing
    schedule.  The best sequence encountered (as measured by the same
    evaluation function) is returned.
    """
    rng = np.random.default_rng(seed)

    # Length of the current instance – fall back to 256 if unknown.
    n = int(len(globals().get("height_sequence_1", []))) if "height_sequence_1" in globals() else 256
    if n == 0:
        n = 256

    # ------------------------------------------------------------------
    # 1.  Starting point
    # ------------------------------------------------------------------
    a = build_arcsine_sequence(n)
    a = np.clip(a, 0.0, 1000.0)

    fft_size = _fft_size_for(n)

    best_seq = a.copy()
    best_score, conv_cur, max_cur = _score_of(a, fft_size)
    cur_score = best_score

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Simulated‑annealing loop
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # a tiny safety margin
    it = 0

    # hyper‑parameters (coarse tuning)
    sigma0 = 0.18      # start standard deviation for block scaling (multiplicative)
    T0 = 0.012         # start temperature for SA

    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start_time
        progress = elapsed / float(budget_s)

        # linearly anneal the temperature and the perturbation scale
        sigma = sigma0 * (1.0 - progress) + 0.02 * progress
        T = max(1e-7, T0 * (1.0 - progress))

        # ------------------------------------------------------------------
        # 2a.  Choose a move type
        # ------------------------------------------------------------------
        if rng.random() < 0.55:
            # ----  Block multiplicative perturbation  ----
            w_min = max(1, n // 200)
            w_max = max(2, n // 12)
            width = int(rng.integers(w_min, w_max + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand = a.copy()
            cand[start : start + width] *= factor
        else:
            # ----  Sensitivity‑driven transfer  ----
            # Build a binary mask of “almost maximal” convolution entries.
            eps = max(1e-12, 1e-6 * max_cur)
            mask = (conv_cur >= max_cur - eps).astype(np.float64)

            # Sensitivity vector g_i = Σ_{k∈mask} a_{k‑i}
            g = sensitivity_vector(a, mask, fft_size)

            src = int(np.argmax(g))
            dst = int(np.argmin(g))
            if src == dst:
                dst = (src + 1) % n

            # move a random fraction (up to 25 %) of the source mass to the target
            frac = rng.random() * 0.25
            delta = frac * a[src]
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        # ------------------------------------------------------------------
        # 2b.  Clip, evaluate and possibly accept
        # ------------------------------------------------------------------
        cand = np.clip(cand, 0.0, 1000.0)

        cand_score, cand_conv, cand_max = _score_of(cand, fft_size)

        # simulated‑annealing acceptance
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / T):
            a = cand
            conv_cur = cand_conv
            max_cur = cand_max
            cur_score = cand_score

        # keep the global best
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{it:>6}] new best score = {best_score:.6f}")

            if best_score < 1.503:          # target achieved – we can stop early
                break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
