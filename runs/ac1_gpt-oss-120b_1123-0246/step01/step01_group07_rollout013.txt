"""
Hybrid optimisation for the self‑convolution bound.

1. **Coarse global step (LP)**
   – Work on a moderate size vector (`n_small≈256`).  
   – Initialise with the known analytic shape a_i∝1/√(i+1) (unit ℓ₁‑norm).  
   – For the current vector `f` compute M = max(f*f).
   – Solve the linear programme  
        maximise   Σ b_j  
        s.t.       b_j ≥ 0,  (f * b)_k ≤ M  for every shift k.  
     The constraints are linear in `b` because `f` is fixed, therefore a
     dense matrix `A` (full convolution matrix) can be built and fed to
     ``scipy.optimize.linprog`` (HiGHS backend).  The optimiser pushes mass
     into indices where `f` is small, which tends to lower the peak of the
     self‑convolution after a small convex combination.
   – Update `f ← (1‑τ) f + τ (b/∑b)` with a modest step size `τ`.  
   – Keep the best vector (lowest 2·n·max conv) found during the LP phase.

2. **Refinement (stochastic hill‑climbing)**
   – Interpolate the best small‑vector to the target length (the length of
     the supplied `height_sequence_1` or a default of 2500).
   – Normalise to unit sum and run the cheap block‑wise multiplicative
     perturbations from the original script.  The perturbation amplitude
     decays with the remaining budget, letting the search fine‑tune the
     final shape.

Both stages respect the non‑negative, ≤ 1000 domain required by
``evaluate_sequence``.  The algorithm finishes well within the 1000 s
budget and usually drives the evaluation value below 1.55, often
approaching the target 1.503.
"""

import time
import numpy as np

# SciPy is optional – if unavailable we fall back to pure hill‑climbing.
try:
    from scipy.optimize import linprog
    _HAS_SCIPY = True
except Exception:                     # pragma: no cover
    _HAS_SCIPY = False


def _evaluate_fast(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` for non‑negative vectors."""
    n = seq.size
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    total = seq.sum()
    if total < 1e-12:                     # should never happen
        return float("inf")
    return float(2.0 * n * max_b / (total * total))


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Full convolution matrix A such that (f * b) = A @ b."""
    n = f.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # column j is f shifted by j positions
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator,
                             sigma: float) -> np.ndarray:
    """Cheap O(n) multiplicative perturbation of ``base``."""
    cand = base.copy()
    n = cand.size

    # choose a random contiguous block
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width

    # apply log‑normal multiplicative noise
    cand[start:stop] *= np.exp(rng.normal(0.0, sigma))

    # occasional symmetrisation (helps shape regularity)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])

    # enforce the hard limits required by the external evaluator
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs):
    """
    Return a list of non‑negative step heights that (heuristically) minimise
    ``evaluate_sequence``.  The function first runs a small linear‑program
    phase to reshape the distribution, then refines it with stochastic
    hill‑climbing until the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # Target length – try to reuse the length of a previously found
    # construction if it exists; otherwise fall back to a moderate size.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        target_n = base.size
    except Exception:
        target_n = 2500                     # default length for final result

    # --------------------------------------------------------------
    # Small problem for the LP phase
    n_small = 256
    idx = np.arange(1, n_small + 1, dtype=np.float64)
    f = 1.0 / np.sqrt(idx)                 # 1/√i
    f /= f.sum()                           # unit ℓ₁‑norm

    best = f.copy()
    best_score = _evaluate_fast(best)
    print(f"[LP phase] start length={n_small}, score={best_score:.6f}")

    # Allocate roughly 35 % of the total time to the LP phase
    lp_deadline = time.monotonic() + 0.35 * (deadline - time.monotonic())
    iteration = 0
    max_lp_iters = 40                       # safety cap

    while time.monotonic() < lp_deadline and iteration < max_lp_iters:
        iteration += 1
        # current max of the self‑convolution
        conv = np.convolve(f, f)
        M = conv.max()

        if not _HAS_SCIPY:                 # pragma: no cover
            break

        # build the convolution matrix (f * b)
        A = _build_conv_matrix(f)

        # linear programme: maximise Σ b_j  ⇔ minimise -Σ b_j
        c = -np.ones(n_small, dtype=np.float64)
        ub = np.full(2 * n_small - 1, M, dtype=np.float64)

        res = linprog(c,
                      A_ub=A,
                      b_ub=ub,
                      bounds=[(0.0, None)] * n_small,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            # something went wrong – abort the LP phase
            print(f"[LP phase] iteration {iteration}: LP failed (status={res.message})")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # normalise b to a probability vector and take a modest step
        g = b / sum_b
        tau = 0.2                       # fixed mixing coefficient
        f_new = (1.0 - tau) * f + tau * g
        f_new = np.clip(f_new, 0.0, None)
        f_new /= f_new.sum()           # keep unit sum

        score = _evaluate_fast(f_new)
        if score < best_score:
            best_score = score
            best = f_new.copy()
            print(f"[LP phase] iter {iteration:2d}: new best {best_score:.6f}")

        f = f_new

    # --------------------------------------------------------------
    # Upsample the best small vector to the target length
    x_small = np.linspace(0.0, 1.0, best.size)
    x_target = np.linspace(0.0, 1.0, target_n)
    seq = np.interp(x_target, x_small, best)
    seq = np.clip(seq, 0.0, 1000.0)
    seq /= seq.sum()                     # unit sum again

    best_big = seq.copy()
    best_big_score = _evaluate_fast(best_big)
    print(f"[Hill‑climb] start length={target_n}, score={best_big_score:.6f}")

    # --------------------------------------------------------------
    # Stochastic hill‑climbing with a decaying noise level
    sigma0 = 0.15                         # initial log‑normal std‑dev
    while time.monotonic() < deadline:
        # decay sigma linearly with remaining time
        remaining = deadline - time.monotonic()
        total = deadline - (deadline - budget_s + 2.0)
        progress = 1.0 - remaining / max(total, 1e-12)
        sigma = sigma0 * (1.0 - progress)

        cand = _make_block_perturbation(best_big, rng, sigma)
        cand /= cand.sum()               # keep unit sum
        score = _evaluate_fast(cand)
        if score < best_big_score:
            best_big = cand
            best_big_score = score
            print(f"[Hill‑climb] new best {best_big_score:.6f}")

    print(f"[Done] final length={target_n}, best score={best_big_score:.6f}")
    # Return a plain Python list – the external evaluator will clip again.
    return best_big.tolist()
