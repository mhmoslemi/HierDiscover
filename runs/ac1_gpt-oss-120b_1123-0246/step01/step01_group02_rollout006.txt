def project_simplex(v):
    if v.sum() == 1 and np.all(v >=0): return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑based simplex projection search for a step‑function minimising the
convolution‑peak objective.

The algorithm works as follows:

1. Initialise the sequence with the analytically known “arcsine’’ shape
   (inverse‑square‑root) or with the provided `height_sequence_1`.
2. Normalise the sequence to the probability simplex (sum = 1, all
   non‑negative).  The evaluation function is scale‑invariant, so this does
   not affect the score.
3. Repeatedly improve the shape:
   * Compute the full self‑convolution `b = a * a`.
   * Build a smooth approximation of the max‑operator by a soft‑max
     weighting `w_k = exp(beta * (b_k - max(b)))`.  Adding an infinitesimal
     random noise to `b` breaks ties when many entries attain the maximum.
   * The gradient of the smoothed objective w.r.t. `a` is
        grad = 2 * (rev(a)  ⊛  w)_[n‑1 : n‑1+n],
     where `⊛` denotes convolution and `rev(a)` is `a` reversed.
     Both convolutions are performed with an FFT (O(n log n)).
   * Take a projected‑gradient step on the simplex:
        a_next = Proj_Simplex(a - gamma * grad),
     where `gamma` is chosen so that the L2‑step size is roughly `step0`.
   * Occasionally (2 % of the iterations) apply a random block scaling
     perturbation to escape flat plateaus.
   * Keep the best sequence seen so far.  The loop runs until the time
     budget expires (or a score below 1.5 is found).

The implementation uses only NumPy and the provided `evaluate_sequence`
function, runs in < O(n²) per iteration, and respects the 1000‑second
budget.  It returns the best heights as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: the “arcsine’’ (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a[i] = binom(2i,i)/4**i (i=0…length‑1)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    return a


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, x_i >= 0).
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution (real inputs → real output).
# ----------------------------------------------------------------------
def conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real arrays a and b using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    c = np.fft.irfft(A * B, n=n_fft)[:n]
    return c


# ----------------------------------------------------------------------
# Small random moves – kept from the previous baseline for exploration.
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0, σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

# ----------------------------------------------------------------------

    ``budget_s`` seconds.

    # ------------------------------------------------------------------

        init_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = init_seq.size
        if n == 0:
            init_seq = None
    else:
        init_seq = None
        n = 256

    if init_seq is None:
        init_seq = build_arcsine_sequence(n)

    # Normalise to the probability simplex (scale does not affect the score)
    total = init_seq.sum()
    if total <= 0.0:
        a = build_arcsine_sequence(n)
        a = a / a.sum()
    else:
        a = init_seq / total

    # Initial evaluation
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    cur = a.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # Parameters for the optimisation loop
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 3.0)   # a few seconds slack
    it = 0

    beta = 20.0          # temperature for the soft‑max
    beta_inc = 0.01
