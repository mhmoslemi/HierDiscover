def project_simplex(v):
    # ensure sum(v) = 1, v_i >=0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def compute_gradient(a, conv, tau):
    maxc = conv.max()
    w = np.exp(tau * (conv - maxc))  # shape 2n-1
    total = w.sum()
    rev_a = a[::-1]
    # convolution w * rev_a
    L = 1 << ((len(w) + len(rev_a) - 1).bit_length())
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(rev_a, n=L)
    conv_w_rev = np.fft.irfft(W * A, n=L)[: len(w) + len(rev_a) - 1]
    # extract slice
    start = len(a) - 1
    end = start + len(a)
    grad = (2.0 * conv_w_rev[start:end]) / total
    return grad
```

"""
Propose a step‑function sequence that beats the classic binomial construction.

The algorithm starts from the analytically optimal binomial (Catalan)
distribution and then performs a fast stochastic local search that
combines three move types:

* **Block perturbation** – multiplicative log‑normal noise on a random
  interval (the same idea used in the reference code).

* **Mirror‑swap** – transfers a small amount of mass between a position
  and its mirror (i ↔ n‑1‑i).  This reduces the product a_i·a_{n‑1‑i} that
  dominates the central convolution peak.

* **Focused mass‑shift** – identifies the pair (i, n‑1‑i) whose product
  a_i·a_{n‑1‑i} is maximal and moves a fraction of that mass to the two
  ends of the sequence (indices 0 and n‑1).  This directly lowers the
  central convolution value while keeping the total sum (and therefore
  the scaling‑invariant evaluation) unchanged.

All candidates are kept on the probability simplex (non‑negative,
summing to 1) by a projection step, so the evaluation metric reduces to

    score = 2·n·max(conv(a,a))

which is exactly what ``evaluate_sequence`` computes (scaling cancels out).

The search runs until the allocated wall‑clock budget expires (default
≈ 900 s) and always returns the best sequence discovered.  Simple
statistics are printed so progress can be monitored.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# helper utilities
# ----------------------------------------------------------------------


def _binomial_sequence(length: int) -> np.ndarray:
    """
    Generate the raw (un‑normalised) binomial coefficients
        a_i = C(2i,i) / 4^i   (i = 0,…,length‑1).
    These satisfy (a*a)[k] = 1 for the infinite series.
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(length - 1):
        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | x_i ≥ 0, Σ x_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero; return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _metric_of(a: np.ndarray) -> float:
    """
    Compute the evaluation metric (without the expensive security checks):
        score = 2·n·max( a * a )
    where ``*`` is the ordinary (linear) convolution.
    The vector ``a`` is assumed to be non‑negative and sum to 1
    (the scaling of the sequence does not affect the score).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max())


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Log‑normal multiplicative perturbation of a random contiguous block."""
    n = base.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand = base.copy()
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    cand = np.clip(cand, 0.0, 1000.0)
    cand = _project_simplex(cand)
    return cand


def _mirror_swap(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a small amount of mass between a position and its mirror."""
    n = base.size
    cand = base.copy()
    i = rng.integers(0, n // 2)
    j = n - 1 - i
    if cand[i] == 0.0 and cand[j] == 0.0:
        return cand
    delta = min(cand[i], cand[j]) * rng.random() * 0.1
    if rng.random() < 0.5:
        cand[i] += delta
        cand[j] -= delta
    else:
        cand[i] -= delta
        cand[j] += delta
    cand = np.clip(cand, 0.0, 1000.0)
    cand = _project_simplex(cand)
    return cand


def _mass_shift(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Identify the pair (i, n‑1‑i) with the largest product a_i·a_{n‑1‑i}
    and move a small fraction of that mass to the two ends of the
    sequence (indices 0 and n‑1).  This directly reduces the central
    convolution peak.
    """
    n = base.size
    half = n // 2
    # product of each mirror pair (only need the first half)
    prod = base[:half] * base[-1:-half - 1:-1]
    i = int(rng.choice(np.flatnonzero(prod == prod.max())))
    j = n - 1 - i
    if base[i] <= 0.0 or base[j] <= 0.0:
        return base.copy()
    # move up to 10 % of the smaller side
    delta = min(base[i], base[j]) * rng.random() * 0.1
    cand = base.copy()
    cand[i] -= delta
    cand[j] -= delta
    cand[0] += delta
    cand[-1] += delta
    cand = np.clip(cand, 0.0, 1000.0)
    cand = _project_simplex(cand)
    return cand


# ----------------------------------------------------------------------
# main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 900.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine returns the best list found
    before the allocated time expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. initialise – use the known construction if it exists,
    #    otherwise fall back to a fresh binomial sequence.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
        if n == 0:
            n = 256
    else:
        n = 3000                         # a comfortable size for the search

    # start from the (normalised) binomial distribution
    raw_binom = _binomial_sequence(n)
    a = raw_binom / raw_binom.sum()      # probability simplex (sum = 1)

    # evaluate the starting point (full, using the provided function)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    cur = a.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # 2. stochastic hill‑climb with three move families.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    it = 0
    # occasional temperature‑like schedule for the acceptance probability
    temp_start = 1.0
    temp_end = 0.001
    while time.monotonic() < deadline:
        it += 1
        r = rng.random()
        if r < 0.5:                      # focussed mass‑shift (most promising)
            cand = _mass_shift(cur, rng)
        elif r < 0.75:                   # mirror‑swap
            cand = _mirror_swap(cur, rng)
        else:                            # block multiplicative noise (exploration)
            cand = _make_block_perturbation(cur, rng)

        # quick metric (no clipping, sum already 1)
        score = _metric_of(cand)

        # simulated‑annealing style acceptance (helps to escape plateaus)
        temperature = temp_start + (temp_end - temp_start) * (it / 50000.0)
        accept = False
        if score < cur_score:
            accept = True
        else:
            # accept with probability exp(-(score‑cur)/temp)
            prob = np.exp(-(score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = score

        # keep the globally best sequence
        if score < best_score:
            best = cand
            best_score = score
            print(f"[{it:6d}] new best: {best_score:.6f}")

        # very cheap occasional restart to avoid long stagnation
        if it % 20000 == 0:
            cur = best.copy()
            cur_score = best_score

    # ------------------------------------------------------------------
    # 3. Return the best sequence discovered.
    # ------------------------------------------------------------------
    # The evaluation function scales the sequence internally, so returning
    # a probability vector is perfectly valid.
    print(f"[done] iterations={it}, final best score={best_score:.6f}")
    return [float(x) for x in best]
