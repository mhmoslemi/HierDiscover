"""
Propose a low‑score height sequence for the step‑function problem.

The algorithm is a projected gradient descent on a smooth approximation of
the L∞‑norm of the autocorrelation (the “max‑convolution” term).  Instead of
optimising only the single dominant lag (as in the baseline), a softmax over
all lags is used, so the gradient pushes down every large convolution value.
A temperature τ is increased during the run, turning the smooth objective
into the true max‑objective as the search proceeds.  The L1‑norm of the
sequence is kept constant (target = √(2·n)), non‑negativity and the upper
bound 1000 are enforced by clipping, and occasional random multiplicative
perturbations help escape plateaus.  The best sequence found before the time
budget expires is returned.
"""

import time
import numpy as np

def _initial_sequence(seed: int, n: int) -> np.ndarray:
    """
    Produce a deterministic U‑shaped start sequence of length *n*.
    The shape has large values at the ends and small values near the centre.
    """
    rng = np.random.default_rng(seed)
    # linear distance from centre
    centre = (n - 1) / 2.0
    dist = np.abs(np.arange(n, dtype=np.float64) - centre)
    # add a tiny constant to avoid exact zeros before normalisation
    seq = dist + 1e-6
    # randomise a little so we do not start from a perfectly symmetric point
    seq *= np.exp(rng.normal(0.0, 0.05, size=n))
    return seq


def _full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution of two 1‑D real arrays using NumPy."""
    return np.convolve(x, y)  # NumPy’s implementation is fast enough for our sizes


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds.  The function returns the best sequence found
        before the budget expires.
    **kwargs
        Ignored; present for compatibility with the evaluation harness.

    Returns
    -------
    List[float]
        The best non‑negative height sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1.  Initialise the starting point
    # ----------------------------------------------------------------------
    try:
        # Use the global construction if it exists (it may be very long).
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if base.ndim != 1 or base.size == 0:
            raise NameError
    except NameError:
        # Fallback: a simple U‑shaped sequence of moderate length.
        n = 2048
        base = _initial_sequence(seed, n)

    # Clip to the allowed interval and make sure we have a positive total mass.
    base = np.clip(base, 0.0, 1000.0)
    if base.sum() == 0.0:
        base[:] = 1.0

    n = base.size
    target_sum = np.sqrt(2.0 * n)               # reproduces the paper’s normalisation
    base = base * (target_sum / base.sum())
    a = np.clip(base, 0.0, 1000.0)

    # ----------------------------------------------------------------------
    # 2.  Evaluate the initial candidate
    # ----------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 3.  Parameters for the optimisation loop
    # ----------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # keep a safety margin
    elapsed = 0.0
    iteration = 0

    # learning‑rate schedule (relative to the typical entry magnitude)
    init_eta = 0.06 * target_sum            # a bit larger than the baseline
    # temperature schedule for the softmax (τ)
    tau_start = 0.5
    tau_end = 12.0

    # momentum variables (simple heavy‑ball momentum)
    velocity = np.zeros_like(a)

    # ----------------------------------------------------------------------
    # 4.  Main optimisation loop
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- 4.1  current convolution and softmax weights -----------------
        b = _full_convolve(a, a)                      # length 2n‑1
        max_b = b.max()

        elapsed = time.monotonic() - start_time
        # linearly increase temperature so we start smooth and end sharp
        tau = tau_start + (tau_end - tau_start) * min(1.0, elapsed / budget_s)

        # softmax over all lags (stable by shifting by max_b)
        weights_raw = np.exp(tau * (b - max_b))
        w_sum = weights_raw.sum()
        if w_sum == 0.0:
            # numerically impossible, but guard against it
            w = np.full_like(weights_raw, 1.0 / weights_raw.size)
        else:
            w = weights_raw / w_sum

        # ---- 4.2  gradient of the smoothed max‑convolution -----------------
        # grad_i = 2 * Σ_k w_k * a_{k‑i}
        conv_wr = _full_convolve(w, a[::-1])                 # length 3n‑2
        grad = 2.0 * conv_wr[n - 1 : n - 1 + n]              # slice of length n

        # Project onto the hyper‑plane Σ a_i = constant (zero‑mean step)
        grad -= grad.mean()

        # ---- 4.3  step‑size (decaying) ------------------------------------
        eta = init_eta * max(0.0, 1.0 - elapsed / budget_s)
        if eta < 1e-8:
            eta = 1e-8

        # simple heavy‑ball momentum
        velocity = 0.9 * velocity - eta * grad
        cand = a + velocity

        # ---- 4.4  enforce the box constraints and renormalise ---------------
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > 0.0:
            cand = cand * (target_sum / s)
        cand = np.clip(cand, 0.0, 1000.0)

        # ---- 4.5  evaluate ------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            a = cand.copy()
            velocity[:] = 0.0               # reset momentum after a genuine improvement
            print(f"[{iteration:6d}] new best  score = {best_score:.6f}")

        else:
            # even if not improved we keep moving – helps exploration
            a = cand

        # ---- 4.6  occasional random multiplicative perturbation ------------
        if iteration % 600 == 0 and iteration > 0:

            if s2 > 0.0:

            cand2 = np.clip(cand2, 0.0, 1000.0)
            score2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best = cand2.copy()
                best_score = score2
                a = cand2.copy()
                velocity[:] = 0.0
                print(f"[{iteration:6d}] random perturb improved score = {best_score:.6f}")

        iteration += 1

    # ----------------------------------------------------------------------
    # 5.  Return the best sequence found
    # ----------------------------------------------------------------------
    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    return [float(v) for v in best]
