def build_constraints(f, M):
    n = len(f)
    # row indices: for each i, rows = np.arange(i, i+n)
    row_parts = [np.arange(i, i+n) for i in range(n)]  # sums produce many arrays
    rows = np.concatenate(row_parts)
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(f, n)   # each f_i repeated n times

    A = coo_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M + eps, dtype=np.float64)

"""
propose_candidate
~~~~~~~~~~~~~~~~~

Search for a non‑negative step‑function that minimizes the provided
`evaluate_sequence` score.

Algorithm
---------

1. **Start point** – Use the Catalan‑type sequence
   a_i = binom(2i,i)/4**i (the coefficients of (1‑x)^‑½).  This
   sequence satisfies (a∗a)_k = 1 for 0 ≤ k < n and gives a good
   initial score (≈ π/2).

2. **Linear‑programming improvement** – With the current sequence f,
   compute M = max(f∗f).  The LP

        maximise   Σ g_i
        subject to (f ∗ g)_k ≤ M   for all k = 0 … 2n‑2
                   g_i ≥ 0

   is linear because (f∗g)_k is a linear form in the unknown vector g.
   Solving this LP (via ``scipy.optimize.linprog`` with a sparse
   constraint matrix) yields a non‑negative vector g that adds the
   largest possible total mass without increasing the worst convolution
   value M.

3. **Mixing step** – Replace the current sequence by a convex combination
   ``f ← (1‑t)·f + t·g`` for a small step size t (we try several t’s and
   keep the best improvement).  The mixing increases the L₁‑norm of the
   sequence while keeping the maximum convolution essentially unchanged,
   which lowers the evaluation score
   `2·n·max(conv)/sum²`.

4. **Iterate** – Repeat steps 2‑3 until the time budget is exhausted or
   no further improvement is found.  If the LP fails for any reason a
   cheap random multiplicative perturbation is tried as a fall‑back.

The whole procedure runs on a moderate sequence length (≈ 800) so that
the LP (roughly n² constraints) is fast while still providing a strong
improvement over the raw Catalan construction.  The final sequence is
returned as a plain Python list; `evaluate_sequence` is scale‑invariant,
so the absolute magnitude of the entries is irrelevant (they are clipped
to the allowed range automatically).

def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan‑type coefficients C_i = binom(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence a[i+1] = a[i] * (2*i+1)/(2*i+2)
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(a, 0.0, 1000.0)


def _conv_matrix_structure(n: int):
    """
    Pre‑compute the row and column index arrays that describe the linear
    convolution constraints (f∗g)_k ≤ M.
    For each variable g_i the coefficient f_i appears in the constraints
    k = i … i+n‑1.
    Returns (rows, cols) of length n*n suitable for a ``coo_matrix``.
    """
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        pos += n
    return rows, cols

                    best = candidate
                    best_score = cand_score
                    improved = True
                    print(f"Iter {iteration:02d}, mix t={t:.2f} → score {best_score:.6f}")
                    break
        else:
            # Linear program unavailable or failed – use a cheap random perturbation
            perturb = np.exp(rng.normal(0.0, 0.08, size=n))
            candidate = best * perturb
            candidate = np.clip(candidate, 0.0, 1000.0)
            try:
                cand_score = float(evaluate_sequence(candidate.tolist()))
            except Exception:  # pragma: no cover
                cand_score = float("inf")
            if np.isfinite(cand_score) and cand_score < best_score:
                best = candidate
                best_score = cand_score
                improved = True
                print(f"Iter {iteration:02d}, random tweak → score {best_score:.6f}")

        # If neither LP nor random tweak improved the score we stop early.
        if not improved:
            # small random walk as a last resort
            perturb = np.exp(rng.normal(0.0, 0.04, size=n))
            candidate = best * perturb
            candidate = np.clip(candidate, 0.0, 1000.0)
            try:
                cand_score = float(evaluate_sequence(candidate.tolist()))
            except Exception:
                cand_score = float("inf")
            if np.isfinite(cand_score) and cand_score < best_score:
                best = candidate
                best_score = cand_score
                print(f"Iter {iteration:02d}, gentle tweak → score {best_score:.6f}")

        iteration += 1

    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    # Return a plain Python list of floats (the evaluator expects a list)
    return [float(v) for v in best]
