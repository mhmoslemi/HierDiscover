def propose_candidate(seed=42, budget_s=500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 5)   # leave margin
    # Determine length n:
    base_seq = np.array(height_sequence_1 if 'height_sequence_1' in globals() else np.ones(256), dtype=float)
    n = base_seq.size

    # Optionally choose length = n or perhaps adjust.
    # Initialize candidates:
    # start from uniform:
    base = np.full(n, 1.0, dtype=float)   # sum = n
    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    # Also keep best from height_sequence_1 for fallback.
    # Random restarts:
    # We'll keep best overall.

    # cooling schedule
    T0 = 1.0
    # loops
    while time.monotonic() < deadline:
        # propose move:
        move_type = rng.choice(['pair_transfer', 'block_scale', 'swap'], p=[0.4,0.4,0.2])
        if move_type == 'pair_transfer':
            # Pick random shift k corresponding to max b or random shift:
            # Choose k = rng.integers(0, 2*n-1)
            # Or choose k = argmax(b)
            candidate = best.copy()
            # compute convolution b
            b = np.convolve(candidate, candidate)
            k_max = np.argmax(b)
            # Choose index i in overlapping region of k_max
            i_min = max(0, k_max - (n-1))
            i_max = min(n-1, k_max)
            i = rng.integers(i_min, i_max+1)
            j = k_max - i
            # Determine delta to transfer
            delta = candidate[i] * rng.random() * 0.2  # up to 20% of a_i
            # Ensure we don't go negative:
            delta = min(delta, candidate[i])
            # If j != i, also adjust candidate[j]; take proportion as min
            if j != i:
                delta = min(delta, candidate[j])
                candidate[i] -= delta
                candidate[j] -= delta
                # Add delta to two random other positions not i,j
                r1, r2 = rng.choice(n, size=2, replace=False)
                # Ensure not i/j maybe but not essential
                candidate[r1] += delta/2
                candidate[r2] += delta/2
            else:
                # self-pair, reduce by delta *2?
                # Keep sum constant: remove delta from a_i, add delta elsewhere
                candidate[i] -= delta
                r = rng.choice([idx for idx in range(n) if idx != i])
                candidate[r] += delta
        elif move_type == 'block_scale':
            candidate = make_block_perturbation(best, rng)
        else:  # swap
            candidate = best.copy()
            i, j = rng.choice(n, size=2, replace=False)
            # swap values partially
            # transfer a fraction from i to j
            fraction = rng.random() * 0.3
            delta = fraction * candidate[i]
            candidate[i] -= delta
            candidate[j] += delta
        # Clip, evaluate
        candidate = np.clip(candidate, 0.0, 1000.0)
        # Ensure sum > 0 else reinitialize
        if np.sum(candidate) < 0.01:
            continue
        score = float(evaluate_sequence(candidate.tolist()))
        # Simulated annealing: accept if better, else accept with prob exp(-(score - best_score)/T)
        if np.isfinite(score):
            if score < best_score:
                best = candidate
                best_score = score
                # Possibly shrink temperature
            else:
                # temperature schedule based on elapsed fraction
                elapsed = (budget_s - (deadline - time.monotonic())) / budget_s
                T = T0 * (1 - elapsed)  # linear cooling
                if T > 0 and rng.random() < np.exp(-(score - best_score) / (T + 1e-12)):
                    best = candidate
                    best_score = score
        # Optionally, periodically reinitialize candidate to uniform to escape
        if rng.random() < 0.001:
            best = np.full(n, 1.0)
    return [float(x) for x in best]

    coefficient of the self‑convolution.  It starts from a uniform step
    function, repeatedly moves a small amount of mass from the two indices
    that contribute to the current maximal convolution coefficient to two
    random positions, and accepts only improving moves (with a tiny
    probability of accepting worsening moves to escape plateaus).  The
    algorithm runs until the time budget expires and returns the best
    sequence found.
    """

    temperature_start = 0.01

    def compute_score(arr):
        # arr is assumed to sum to 1
        max_val, _ = conv_max_and_index(arr)
        return 2.0 * n * max_val  # since denominator sum^2 = 1

    # main loop
    while time.monotonic() < deadline:
        # compute convolution and the shift with maximal coefficient
        max_val, k_max = conv_max_and_index(a)

        # current score
        current_score = 2.0 * n * max_val

        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)

        j = k_max - i

            candidate[i] -= delta

"""Minimize the self‑convolution peak of a non‑negative step function.

The algorithm is a lightweight simulated‑annealing style local search that
operates directly on the quantity that the evaluation function measures.
Starting from a uniform (or a short geometric) sequence we repeatedly

1. compute the convolution a*a and locate the shift `k_max` that attains the
   current maximum;
2. pick a pair of indices `(i, j)` that contribute to that maximum
   (i + j = k_max);
3. move a small amount of mass from the pair to two random other positions,
   re‑normalise the sequence (the objective is scale invariant) and keep the
   change if it improves the score (occasionally we accept worse moves to
   escape plateaus);
4. occasionally mix the current vector with the uniform one to keep diversity.

The convolution is evaluated with an FFT (O(n log n)) to keep the inner loop
fast, while the final score is obtained with the provided `evaluate_sequence`
function.  The search runs until the allotted time expires and returns the
best sequence found.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: fast convolution and maximum index using FFT
# ----------------------------------------------------------------------
def conv_max_and_index(a: np.ndarray):
    """Return (max_value, index_of_max) of the self‑convolution a*a."""
    n = a.size
    # size >= 2*n-1, power of two for efficient FFT
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # numerical noise may create tiny negatives
    conv = np.maximum(conv, 0.0)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=500, **kwargs):
    """Search for a sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget (seconds) for the search.

    Returns
    -------
    list[float]
        The best sequence discovered (non‑negative, any length).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Length of the problem – fall back to 256 if the global variable is missing
    n = int(len(globals().get("height_sequence_1", np.ones(256)))) if "height_sequence_1" in globals() else 256

    # ------------------------------------------------------------------
    # Initialise – try a few geometric decays and keep the best as start
    # ------------------------------------------------------------------
    def fast_score(vec):
        """Score assuming the L1 norm of vec is 1."""
        mx, _ = conv_max_and_index(vec)
        return 2.0 * n * mx   # denominator sum^2 == 1

    # start with uniform
    a = np.full(n, 1.0 / n, dtype=np.float64)
    best_a = a.copy()
    best_score = fast_score(a)

    # a few geometric candidates (a_i ∝ r^i) – cheap sweep
    for r in np.linspace(0.90, 0.9999, 30):
        geom = np.array([r ** i for i in range(n)], dtype=np.float64)
        geom_sum = geom.sum()
        if geom_sum == 0:
            continue
        geom /= geom_sum
        sc = fast_score(geom)
        if sc < best_score:
            best_score = sc
            best_a = geom.copy()
            a = geom.copy()   # use the best geometry as the current point

    # ------------------------------------------------------------------
    # Parameters controlling the local search
    # ------------------------------------------------------------------
    move_scale = 0.25          # fraction of the smaller of the chosen pair to move
    temperature_start = 0.02   # initial temperature for occasional uphill moves
    accept_worse = True        # enable simulated‑annealing acceptance
    mix_prob = 0.02            # probability to mix a little uniform mass each step

    # Main optimisation loop
    while time.monotonic() < deadline:
        # Current convolution maximum and its location
        cur_max, k_max = conv_max_and_index(a)
        cur_score = 2.0 * n * cur_max      # L1 norm is 1 -> denominator = 1

        # ------------------------------------------------------------------
        # Propose a move that reduces the contribution to the maximal shift
        # ------------------------------------------------------------------
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)
        i = rng.integers(i_low, i_high + 1)
        j = k_max - i

        # amount of mass to relocate – never more than the smaller entry
        delta = rng.random() * move_scale * min(a[i], a[j] if i != j else a[i])
        if delta <= 0:
            continue

        cand = a.copy()
        if i == j:
            # only one index contributes (i=j => square term)
            cand[i] -= delta
            oth = rng.integers(0, n)
            cand[oth] += delta
        else:
            cand[i] -= delta
            cand[j] -= delta
            # distribute the removed mass to two random positions
            o1, o2 = rng.choice(n, size=2, replace=False)
            cand[o1] += delta / 2.0
            cand[o2] += delta / 2.0

        # ensure non‑negativity and renormalise (the objective is scale‑invariant)
        np.maximum(cand, 0.0, out=cand)
        s = cand.sum()
        if s == 0:
            continue
        cand /= s

        # Quick score for the candidate
        cand_score = fast_score(cand)

        # ------------------------------------------------------------------
        # Acceptance test
        # ------------------------------------------------------------------
        accept = False
        if cand_score < best_score:
            accept = True
        elif accept_worse:
            # linear cooling schedule
            elapsed = (time.monotonic() - (deadline - budget_s)) / budget_s
            T = temperature_start * (1.0 - elapsed) + 1e-12
            if rng.random() < np.exp(-(cand_score - best_score) / T):
                accept = True

        if accept:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[{time.monotonic():.2f}] new best score: {best_score:.6f}")

        # Occasionally blend a little uniform mass to keep diversity
        if rng.random() < mix_prob:
            alpha = rng.random() * 0.1   # up to 10 % uniform mix
            a = (1.0 - alpha) * a + alpha * (1.0 / n)

    # ------------------------------------------------------------------
    # Return the best sequence discovered – convert to plain Python list
    # ------------------------------------------------------------------
    return [float(x) for x in best_a]
