def propose_candidate(seed=42, budget_s=500, **kwargs):
    import numpy as np, time, math
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # get starting sequence
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = None

    if base is None:
        n = 512  # default size
        # uniform
        a = np.full(n, np.sqrt(2) / np.sqrt(n), dtype=np.float64)
    else:
        # Use provided base but cut to some manageable size if too large
        n = int(min(len(base), 800))  # limit size
        a = np.array(base[:n], dtype=np.float64)
        # Ensure non-negative and clip
        a = np.clip(a, 0.0, 1000.0)

    # Normalize sum to sqrt(2n)
    target_sum = math.sqrt(2.0 * n)
    s = np.sum(a)
    if s > 0:
        a *= target_sum / s
    else:
        a = np.full(n, target_sum / n)

    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Initial length {n}, score {best_score}")

    # parameters
    t = 0.3  # step size
    min_t = 0.02
    decay = 0.99
    last_improve_time = time.monotonic()
    iteration = 0

    while time.monotonic() < deadline:
        # compute current bound M
        conv = np.convolve(a, a)
        M = conv.max()
        if not np.isfinite(M) or M <= 0:
            break

        # Build constraint matrix A_ub for convolution with a
        # A_ub is shape (2n-1, n). Row k: for col j, the entry is a_{k-j} if 0<=k-j<n else 0.
        rows = []
        cols = []
        data = []
        # for each j, iterate i such that row = i + j, value = a[i]
        for i in range(n):
            ai = a[i]
            # This ai contributes to rows i..i+n-1, each at column j = row - i
            start_row = i
            end_row = i + n  # exclusive
            # So for each col j 0..n-1, row = i + j
            # We'll add a vector of length n: rows = np.arange(start_row, end_row), cols = np.arange(0,n)
            rows.extend(range(start_row, end_row))
            cols.extend(range(n))
            data.extend([ai] * n)
        # Convert to numpy arrays
        A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        # Right-hand side: M for each constraint (allow tiny epsilon)
        ub = np.full(2 * n - 1, M * (1.0 + 1e-9), dtype=np.float64)

        # Linear program: maximize sum(b)  => minimize -sum(b)
        c = np.full(n, -1.0, dtype=np.float64)

        # bounds: b_j >= 0, no upper bound
        bounds = [(0, None)] * n

        # Solve using HiGHS
        res = linprog(c, A_ub=A_ub, b_ub=ub, bounds=bounds, method='highs')
        if not res.success:
            # If fails, break
            print(f"LP failed at iteration {iteration}: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            # nothing to do
            break

        # Scale to target sum
        g = b * (target_sum / sum_b)

        # Update a with convex combination
        a = (1.0 - t) * a + t * g

        # Ensure numeric constraints (nonneg) and possibly clip
        a = np.clip(a, 0.0, 1000.0)

        # Evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            last_improve_time = time.monotonic()
            print(f"Iter {iteration}, improved score: {best_score:.6f}")

        # update parameters
        iteration += 1
        t = max(min_t, t * decay)  # slowly reduce step size

        # occasionally add a random perturbation to escape local minima if no improvement for a while
        if iteration % 10 == 0:
            if time.monotonic() - last_improve_time > 5.0:
                # random shuffle small noise
                noise = rng.normal(0.0, 0.05, size=n)
                a = a * np.exp(noise)  # multiplicative noise
                a = np.clip(a, 0.0, 1000.0)
                # renormalize sum
                s = np.sum(a)
                if s > 0:
                    a *= target_sum / s

    # final result: best_seq
    print(f"Finished after {iteration} iterations, best_score {best_score:.6f}")
    # Return as Python list of floats
    return [float(x) for x in best_seq]

"""
Propose a step‑function sequence that beats the uniform bound (score = 2).

Algorithm
---------
1. Start from the provided `height_sequence_1` (or a uniform sequence) and
   truncate / resize it to a modest length (default 800).  The sum of the
   coefficients is normalised to √(2 n) because the evaluation function is
   scale‑invariant and this normalisation makes the score equal to
   max(conv(a,a)).

2. Iteratively improve the sequence with a *linear‑programming lift*:
   given the current a, let M = max(conv(a,a)).  Solve the LP

        maximise   Σ b_j
        subject to conv(a,b) ≤ M  (for every convolution index)
                   b_j ≥ 0

   The LP finds a vector b that puts as much mass as possible into positions
   where the current a does not already saturate the bound M.  Typically b
   concentrates near the ends of the interval, which tends to lower the peak
   of the self‑convolution when mixed with a.

3. Scale b so that its sum again equals √(2 n) and replace
   a ← (1‑t)·a + t·b with a step size `t` (gradually reduced).  After each
   update the true score is evaluated; the best sequence seen is kept.

4. The loop runs until the allotted time budget (minus a small safety margin)
   expires.  If the LP fails or stalls, the algorithm falls back to a simple
   random block perturbation (the method used previously).

The whole procedure respects the 1000 s time limit, uses only O(n²) memory
(≈0.5 M entries for n≈800) and SciPy's HiGHS LP solver, and prints
intermediate scores for monitoring.  The final returned list contains only
non‑negative floats.
"""

import time
import math
import numpy as np

# SciPy is available in the execution environment; fall back gracefully.
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _HAS_SCIPY = True
except Exception:               # pragma: no cover
    _HAS_SCIPY = False

def _build_convolution_matrix(a):
    """Return a CSR matrix representing the convolution operator with vector a.
    For a vector a of length n this matrix has shape (2n‑1, n) and satisfies
    conv(a, x) = A @ x  (where conv is the discrete linear convolution).
    """
    n = a.size
    # rows: for each i (0..n‑1) the contribution a[i] appears in rows i…i+n‑1
    rows = np.concatenate([np.arange(i, i + n, dtype=np.int32) for i in range(n)])
    cols = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.repeat(a, n)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def propose_candidate(seed=42, budget_s=500, **kwargs):
    """Return a candidate sequence of non‑negative heights.

    The function runs a time‑bounded optimisation that tries to minimise the
    evaluation metric defined in `evaluate_sequence`.  It uses a linear‑program
    based convexification step to push mass towards the ends of the interval,
    thereby flattening the self‑convolution.
    """
    rng = np.random.default_rng(seed)
    # Give a little time buffer for clean‑up / printing.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.5)

    # ------------------------------------------------------------------
    # 1. Initialise the working sequence.
    # ------------------------------------------------------------------
    base_seq = globals().get('height_sequence_1')
    if (
        isinstance(base_seq, (list, tuple, np.ndarray))
        and len(base_seq) > 0
    ):
        # Truncate to a manageable size (≈800) – larger sizes make the LP
        # quadratic in memory/time.
        n_target = min(len(base_seq), 800)
        a = np.asarray(base_seq[:n_target], dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
    else:
        # Fallback: start from a uniform distribution.
        n_target = 800
        a = np.full(n_target, 1.0, dtype=np.float64)

    # Normalise the sum to √(2 n) – this makes the score = max(conv) .
    n = a.size
    target_sum = math.sqrt(2.0 * n)
    s = a.sum()
    if s > 0:
        a *= target_sum / s
    else:                                   # pragma: no cover
        a = np.full(n, target_sum / n, dtype=np.float64)

    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Main optimisation loop (LP lift + mixing).
    # ------------------------------------------------------------------
    # LP step size, will be reduced as we progress.
    t = 0.4
    min_t = 0.02
    decay = 0.97

    last_improve = time.monotonic()
    iteration = 0

    # Pre‑compute the static sparsity pattern (rows/cols) once – only the
    # data (the actual a[i] values) change each iteration.
    rows = np.concatenate([np.arange(i, i + n, dtype=np.int32) for i in range(n)])
    cols = np.tile(np.arange(n, dtype=np.int32), n)

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # (a) Current bound M = max(conv(a,a))
        # --------------------------------------------------------------
        conv_aa = np.convolve(a, a)
        M = conv_aa.max()
        if not np.isfinite(M) or M <= 0.0:            # pragma: no cover
            break

        # --------------------------------------------------------------
        # (b) Solve the LP: maximise Σ b_j  subject to conv(a,b) ≤ M , b≥0
        # --------------------------------------------------------------
        if _HAS_SCIPY:
            # Build the convolution matrix with the *current* a.
            data = np.repeat(a, n)                     # a[i] repeated n times
            A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

            # Right‑hand side – a tiny epsilon avoids infeasibility due to
            # rounding errors.
            ub = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=np.float64)

            # Maximise Σ b  → minimise −Σ b
            c = -np.ones(n, dtype=np.float64)

            # No upper bound on b_j, lower bound = 0.
            bounds = [(0, None)] * n

            # Solve with HiGHS (fast and handles sparse matrices).
            lp_res = linprog(
                c,
                A_ub=A_ub,
                b_ub=ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": max(0.0, deadline - time.monotonic())},
            )
        else:                                           # pragma: no cover
            lp_res = None

        if lp_res is None or not lp_res.success:
            # If the LP fails (or SciPy unavailable), fall back to a random
            # block perturbation (the method used in the reference code).
            # This still yields a valid candidate and guarantees progress.
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            cand = a.copy()
            cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
            if rng.random() < 0.35:
                cand = 0.5 * (cand + cand[::-1])
            cand = np.clip(cand, 0.0, 1000.0)
            # Renormalise.
            cand_sum = cand.sum()
            if cand_sum > 0:
                cand *= target_sum / cand_sum
            a = cand
        else:
            # ---------------------------------------------------------
            # (c) Build g from the LP solution, renormalise, mix.
            # ---------------------------------------------------------
            b = lp_res.x
            sum_b = b.sum()
            if sum_b <= 0.0:                           # pragma: no cover
                break
            g = b * (target_sum / sum_b)               # enforce the sum constraint

            # Convex combination with current a.
            a = (1.0 - t) * a + t * g
            # Numerical safety (the evaluation routine also clips).
            a = np.clip(a, 0.0, 1000.0)

        # --------------------------------------------------------------
        # (d) Evaluate the new candidate.
        # --------------------------------------------------------------
        current_score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(current_score) and current_score < best_score:
            best_score = current_score
            best_seq = a.copy()
            last_improve = time.monotonic()
            print(f"[iter {iteration:04d}] improved score = {best_score:.6f}")

        # --------------------------------------------------------------
        # (e) Update loop control parameters.
        # --------------------------------------------------------------
        iteration += 1
        t = max(min_t, t * decay)

        # If we have not seen improvement for a few seconds, inject a tiny
        # random perturbation to escape possible stagnation.
        if time.monotonic() - last_improve > 5.0:
            # Multiplicative log‑normal noise – keeps sum unchanged after renorm.
            noise = rng.normal(0.0, 0.07, size=n)
            a = a * np.exp(noise)
            a = np.clip(a, 0.0, 1000.0)
            s = a.sum()
            if s > 0:
                a *= target_sum / s
            last_improve = time.monotonic()
            print(f"[iter {iteration:04d}] added jitter to escape stall")

    # ------------------------------------------------------------------
    # 3. Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"[done] iterations={iteration}, best_score={best_score:.6f}")
    return [float(x) for x in best_seq]
