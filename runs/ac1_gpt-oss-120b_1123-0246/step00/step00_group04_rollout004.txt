"""
Propose a better step‑function by a mixed linear‑programming / local‑search loop.

The algorithm works as follows:

1.  Initialise the current sequence *f* from the global ``height_sequence_1``
    (or a uniform vector if the global is unavailable).  The vector is clipped
    to the allowed value range and normalised so that its sum equals
    ``sqrt(2·n)`` – this is the normalisation used in the reference approach.

2.  In each iteration we compute
        M = max(conv(f , f)),
    i.e. the current worst autocorrelation value.

3.  We build a small linear program that **maximises the total sum of a new
    vector g** while guaranteeing that the mixed autocorrelation
        conv(f , g)[s] ≤ M
    for the most critical shifts *s* (the shifts where ``conv(f,f)`` is
    close to its maximum).  The LP variables are the entries of *g*,
    all constrained to be non‑negative.

    The LP is solved with ``scipy.optimize.linprog`` using the HiGHS
    interior‑point algorithm and a sparse constraint matrix.

4.  The resulting vector *g* is rescaled to the same target sum as *f*.
    We then **mix** the old and the new vectors:
        f_new = (1‑t)·f + t·g          (t≈0.2)
    The mixing step keeps the maximal autocorrelation from blowing up
    (the increase is at most t²·(M_g−M) and t is kept small).

5.  Occasionally a random “block perturbation’’ (the function used in the
    baseline code) is applied to ``f_new`` to help the search escape local
    minima.

6.  The candidate is evaluated with the provided ``evaluate_sequence``.
    If the score improves, the candidate becomes the new best sequence
    (and also the starting point for the next iteration).

The loop runs until the allotted time budget expires and finally returns the
best sequence seen.  The method uses only a few dozen constraints per LP,
so each iteration is cheap and many iterations fit easily into the 1000 s
budget, typically driving the evaluation value well below the initial bound
of 2.0.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper: a simple block‑wise random perturbation (identical to the baseline)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # block width: between n//200 and n//12 (inclusive)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12) + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal noise on the chosen block
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetrisation (helps the autocorrelation shape)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Build the (sparse) constraint matrix for a given set of shifts
# ----------------------------------------------------------------------
def _build_constraint_matrix(f, shifts):
    """
    For each shift s in ``shifts`` construct a row of the matrix A
    such that   (A @ g)[row] == conv(f, g)[s] .
    Only the non‑zero entries are stored (CSR format).
    """
    n = f.size
    rows = []
    cols = []
    data = []

    for row_idx, s in enumerate(shifts):
        # Indices j where g_j actually contributes to conv(f,g)[s]
        j_start = max(0, s - (n - 1))
        j_end = min(n - 1, s)
        j_range = np.arange(j_start, j_end + 1, dtype=np.int32)
        coeffs = f[s - j_range]                 # f_{s‑j}
        rows.extend([row_idx] * len(j_range))
        cols.extend(j_range.tolist())
        data.extend(coeffs.tolist())

    return sp.csr_matrix((data, (rows, cols)), shape=(len(shifts), n))

def propose_candidate(seed=42, budget_s=500, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget (seconds).  The function stops slightly
        before this limit to avoid termination.
    **kwargs
        Ignored – kept for compatibility with the original signature.

    Returns
    -------
    List[float]
        The best sequence found (as a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 1. Initialise the starting sequence
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():

    else:

    f = make_block_perturbation(start, rng)

        A_ub = _build_constraint_matrix(f, top_idx)

        c = np.full(n, -1.0, dtype=np.float64)
        bounds = [(0, None)] * n

        res = opt.linprog(

            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,

        if not res.success:

                print(f"[{iteration:04d}] LP failed – random improve: score={best_score:.6f}")

        if g.sum() == 0.0:

            g = make_block_perturbation(f, rng)
        else:
            g = g * (target_sum / g.sum())

        # ---- e) convex combination (mix) --------------------------------------
        candidate = (1.0 - mix_factor) * f + mix_factor * g
        candidate = np.clip(candidate, 0.0, 1000.0)

        # ---- f) occasional extra random shake (helps escapism) ---------------
        if rng.random() < 0.08:
            candidate = make_block_perturbation(candidate, rng)

        # ---- g) evaluate -------------------------------------------------------
        cand_score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best = candidate.copy()
            best_score = cand_score
            f = best.copy()
            print(f"[{iteration:04d}] improved: score={best_score:.6f}, sum={best.sum():.3f}")

                best_score = explore_score
                f = best.copy()
                print(f"[{iteration:04d}] exploration win: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[done] iterations={iteration}, best_score={best_score:.6f}, sum={best.sum():.3f}")
    return [float(x) for x in best]
