def propose_candidate(seed=42, budget_s=500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)
    # initialize seq length n from height_sequence_1 else length 256
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 256
    # initial sequence random Dirichlet:
    seq = rng.exponential(scale=1.0, size=n)
    seq = seq / seq.max() * 1000.0  # ensure max <=1000
    # may also normalize sum to 1 for scale invariant, but not needed.
    best = seq.copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"Initial score: {best_score:.6f}")
    # local search loop
    while time.monotonic() < deadline:
        # generate candidate
        typ = rng.integers(0, 4)
        cand = best.copy()
        if typ == 0:
            # block scaling
            width = rng.integers(max(1, n // 200), max(2, n // 12))
            start = rng.integers(0, max(1, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.12))
            cand[start:start+width] *= factor
        elif typ == 1:
            # swap two random positions
            i, j = rng.choice(n, size=2, replace=False)
            cand[i], cand[j] = cand[j], cand[i]
        elif typ == 2:
            # move small amount from i to j
            i, j = rng.choice(n, size=2, replace=False)
            transfer = cand[i] * rng.random() * 0.1
            cand[i] = max(0.0, cand[i] - transfer)
            cand[j] = cand[j] + transfer
        else:
            # random shuffle of whole array before scaling
            rng.shuffle(cand)
        # enforce bounds
        cand = np.clip(cand, 0.0, 1000.0)
        score = evaluate_sequence(cand.tolist())
        if np.isfinite(score) and score < best_score:

"""
Propose a candidate step‑function sequence.

The algorithm performs a simple stochastic local search that starts from the
previous best construction (``height_sequence_1`` if available) and repeatedly
applies a variety of cheap perturbations:

* multiplicative scaling of a randomly chosen block,
* swapping two entries,
* moving a small amount of mass from one index to another,
* a targeted “peak reduction” move that looks at the current worst convolution
  index and transfers a little mass away from the most contributing positions,
* random full shuffling (useful for escaping plateaus).

All candidates are clipped to the allowed range ``[0, 1000]``; the objective
``evaluate_sequence`` is scale‑invariant, so we do not bother with a separate
normalisation step.  The best sequence found before the time budget expires is
returned.

The routine is deliberately written without closures, lambdas or heavy external
dependencies so that it can run under the strict sandbox used for the
challenge.
"""

import time
import numpy as np

def propose_candidate(seed=42, budget_s=500, **kwargs):
    """Return a sequence of non‑negative heights with low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Approximate wall‑clock time budget (seconds).  The function will stop
        slightly before this limit so that the caller can collect the result.
    **kwargs : dict
        Unused – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise the search.
    # ------------------------------------------------------------------
    # Use the previously discovered construction if it exists, otherwise fall
    # back to a simple uniform vector.
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            parent = np.clip(base, 0.0, 1000.0)
        else:
            parent = np.ones(256, dtype=np.float64) * 500.0
    else:
        parent = np.ones(256, dtype=np.float64) * 500.0

    # Normalise roughly to keep numbers away from the clipping limits.
    if parent.max() == 0:
        parent[:] = 1.0
    parent = parent * (1000.0 / parent.max())

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] score={best_score:.6f}  length={best.size}")

    n = best.size

    # Small safety margin so we finish before the external timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = best.copy()
        move_type = rng.integers(0, 5)   # five different perturbation types

        # ------------------------------------------------------------------
        # 1) Multiplicative block perturbation (similar to the reference)
        # ------------------------------------------------------------------
        if move_type == 0:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start:start + width] *= factor

        # ------------------------------------------------------------------
        # 2) Swap two random entries
        # ------------------------------------------------------------------
        elif move_type == 1:
            i, j = rng.choice(n, size=2, replace=False)
            cand[i], cand[j] = cand[j], cand[i]

        # ------------------------------------------------------------------
        # 3) Transfer a small amount of mass from i -> j (preserves sum)
        # ------------------------------------------------------------------
        elif move_type == 2:
            i, j = rng.choice(n, size=2, replace=False)
            # Transfer up to 10 % of the source entry.
            transfer = cand[i] * rng.random() * 0.10
            cand[i] = max(0.0, cand[i] - transfer)
            cand[j] = min(1000.0, cand[j] + transfer)

        # ------------------------------------------------------------------
        # 4) Targeted peak‑reduction move.
        #    Identify the index of the current max convolution value and move
        #    a little mass away from the most responsible positions.
        # ------------------------------------------------------------------
        elif move_type == 3:
            conv = np.convolve(cand, cand)
            peak_idx = int(np.argmax(conv))
            # All i that contribute to this peak satisfy 0 ≤ i < n and
            # 0 ≤ peak_idx‑i < n .
            lo = max(0, peak_idx - (n - 1))
            hi = min(n - 1, peak_idx)
            if lo <= hi:
                # pick a "heavy" contributor (probability ∝ value)
                contributors = np.arange(lo, hi + 1, dtype=int)
                weights = cand[contributors] + 1e-12
                i = rng.choice(contributors, p=weights / weights.sum())
                # pick a random destination (could be the same, harmless)
                j = rng.integers(0, n)
                # move a modest fraction (≤5 %) of the source mass
                amount = cand[i] * rng.random() * 0.05
                cand[i] = max(0.0, cand[i] - amount)
                cand[j] = min(1000.0, cand[j] + amount)

        # ------------------------------------------------------------------
        # 5) Global random shuffle (helps to jump out of local minima)
        # ------------------------------------------------------------------
        else:   # move_type == 4
            rng.shuffle(cand)

        # ------------------------------------------------------------------
        # Apply the hard bounds required by the evaluator.
        # ------------------------------------------------------------------
        cand = np.clip(cand, 0.0, 1000.0)

        # ------------------------------------------------------------------
        # Score the candidate.
        # ------------------------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[improve] score={best_score:.6f}  time={time.monotonic():.2f}")

    # ------------------------------------------------------------------
    # Return the best candidate found.
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f}")
    return [float(x) for x in best]
