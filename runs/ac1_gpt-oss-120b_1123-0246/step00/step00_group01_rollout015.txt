def solve_lp_max_sum(f, M, tol=1e-6, max_iter=5):
    n = len(f)
    # compute conv_f_f
    conv_f_f = np.convolve(f, f)
    # initial active indices: top few largest values, maybe top K or those > (M - small).
    active = np.where(conv_f_f >= M - 1e-6)[0].tolist()
    # Also optionally add some random indices.
    # Build sparse matrix A for these rows.
    # Use SciPy sparse matrix.
    for iteration in range(max_iter):
        rows = len(active)
        data = []
        row_ind = []
        col_ind = []
        for idx, k in enumerate(active):
            # for each shift k, include coefficients f_{k - j} for j within range.
            # j varies from max(0, k - (n-1)) to min(k, n-1)
            j_start = max(0, k - (n-1))
            j_end = min(k, n-1)
            # For each j in this range:
            # coefficient = f[k - j]
            # We'll fill vector
            for j in range(j_start, j_end+1):
                coeff = f[k - j]
                data.append(coeff)
                row_ind.append(idx)
                col_ind.append(j)
        A = sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
        # Solve LP
        c = -np.ones(n)
        b = np.full(rows, M)
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
        if not res.success:
            break
        g = res.x
        # Check constraints
        conv_fg = np.convolve(f, g)
        # find violations > M + tolerance
        viol = np.where(conv_fg > M + 1e-8)[0]
        if len(viol) == 0:
            # done
            return g
        # else add violating indices to active (ensuring uniqueness)
        for v in viol:
            if v not in active:
                active.append(v)
        # loop again
    return g

def solve_max_sum_under_bound(f, M, max_iter=5):
    n = len(f)
    # Determine active constraints: start with indices where conv_f_f >= M - 1e-6
    conv = np.convolve(f,f)
    active_ids = list(np.where(conv >= M - 1e-6)[0])
    # possibly also include indices near high contributions? Not now
    for it in range(max_iter):
        rows = len(active_ids)
        # Build sparse matrix A shape rows x n.
        data = []
        row_ind = []
        col_ind = []
        for r_idx, k in enumerate(active_ids):
            j_start = max(0, k - (n-1))
            j_end = min(k, n-1)
            # will iterate j from j_start to j_end
            # For each j, coeff = f[k - j]
            # as we need column = j
            for j in range(j_start, j_end+1):
                coeff = f[k - j]
                data.append(coeff)
                row_ind.append(r_idx)
                col_ind.append(j)
        A = sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows,n))
        # Solve LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n)
        b = np.full(rows, M)
        bounds = [(0, None)] * n
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        # Check all constraints
        conv_fg = np.convolve(f, g)
        viol = np.where(conv_fg > M + 1e-8)[0]
        if len(viol) == 0:
            return g
        # add new violating indices to active set
        for v in viol:
            if v not in active_ids:
                active_ids.append(v)
    # If not converged, return best we have (maybe g from last solve)
    return g

The algorithm starts from the analytically optimal “binomial” sequence

* random block perturbations (the same idea as the reference code) to
  explore the neighbourhood,

  between a position and its mirror (i ↔ n‑1‑i).  This tends to lower the
  product a_i·a_{n‑1‑i} that dominates the convolution peak.

All candidates are evaluated with the provided `evaluate_sequence`
function; the best ever seen is kept and returned.  The search respects
the allotted wall‑clock budget (default ≈ 500 s) and will always finish

import time
import numpy as np

def make_block_perturbation(base, rng):
    """Return one O(n) non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width: between 0.5% and ~8% of the length
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation of the chosen block
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation – helps when the optimum is symmetric
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def _binomial_sequence(length):
    """
    Generate the truncated binomial sequence a_i = C(2i,i)/4^i.
    This sequence satisfies (a*a)[k] = 1 for the central index k = n‑1,
    giving a very low evaluation score.

    a = np.empty(length, dtype=np.float64)

        a[i + 1] = a[i] * (2 * i + 1) / (2.0 * (i + 1))
    return np.clip(a, 0.0, 1000.0)

    else:
        n = 256

    # Create the binomial starting point of length n
    current = _binomial_sequence(n)

    # Evaluate the starting point
    best = current.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 2. stochastic local search loop – runs until the deadline.
    # -----------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    iter_cnt = 0
    while time.monotonic() < deadline:
        iter_cnt += 1

        # ---- decide which move to make --------------------------------
        if rng.random() < 0.7:                     # 70 % block perturbation
            candidate = make_block_perturbation(best, rng)
        else:                                      # 30 % mirror‑swap move
            candidate = best.copy()
            # pick a random index in the first half and move a bit of mass
            i = rng.integers(0, n // 2)
            j = n - 1 - i
            # transfer up to 10 % of the smaller side
            delta = min(candidate[i], candidate[j]) * rng.random() * 0.1
            if rng.random() < 0.5:
                candidate[i] += delta
                candidate[j] -= delta
            else:
                candidate[i] -= delta
                candidate[j] += delta
            candidate = np.clip(candidate, 0.0, 1000.0)

        # ---- evaluate -------------------------------------------------
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"[{iter_cnt:6d}] new best: {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3. Return the best sequence found.
    # -----------------------------------------------------------------
    print(f"[done] iterations={iter_cnt}, final best score={best_score:.6f}")
    return [float(x) for x in best]
