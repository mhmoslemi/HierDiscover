def propose_candidate(seed=42, budget_s=500, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 0.1)

    # Determine sequence length
    try:
        n = len(height_sequence_1)
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 1024
        base_seq = np.ones(n, dtype=np.float64) / n
    # Ensure we have a valid vector
    if base_seq.ndim != 1 or base_seq.size != n:
        base_seq = np.ones(n, dtype=np.float64)

    # Normalize to sum 1 (since evaluation scale invariant)
    base_seq = np.clip(base_seq, 0.0, 1000.0)
    base_seq = base_seq / base_seq.sum()
    best_seq = base_seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"Start optimization length {n}, initial score {best_score}")

    # Helper: project onto simplex (l1=1)
    def project_simplex(v):
        """Return the Euclidean projection of v onto the probability simplex."""
        # using algorithm by Wang & Carreira-Perpiñán
        n = v.shape[0]
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(n) + 1
        cond = u - cssv / ind > 0
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w

    # Function to evaluate and update best
    def try_seq(seq):
        nonlocal best_seq, best_score
        # ensure numeric domain
        seq = np.clip(seq, 0.0, 1000.0)
        total = seq.sum()
        if total < 1e-12:
            return
        # rescale to avoid extremely small values? Not needed.
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New best: {best_score:.6f} at time {time.monotonic():.2f}")

    # --- Search over parametric families (exponential, power) ---
    # Exponential decay shape
    for r in np.linspace(0.97, 0.9999, 30):
        if time.monotonic() >= deadline:
            break
        idx = np.arange(n)
        min_idx = np.minimum(idx, n-1-idx)
        weights = r ** min_idx
        seq = weights / weights.sum()
        try_seq(seq)

    # Power shape
    for p in np.linspace(0.2, 4.0, 30):
        if time.monotonic() >= deadline:
            break
        idx = np.arange(n)
        min_idx = np.minimum(idx+1, n-idx)  # 1-indexed distance to nearest edge
        weights = (min_idx) ** (-p)
        seq = weights / weights.sum()
        try_seq(seq)

    # Combined exponential+power (maybe logspace)
    for r in np.linspace(0.98, 0.9995, 20):
        for p in np.linspace(0.5, 3.0, 10):
            if time.monotonic() >= deadline:
                break
            idx = np.arange(n)
            min_idx = np.minimum(idx, n-1-idx)
            weights = (r ** min_idx) * (min_idx + 1) ** (-p)
            seq = weights / weights.sum()
            try_seq(seq)

    # Random simplex initialization + exponential gradient descent
    def exp_gradient_descent(a0, steps=200, eta=0.5):
        a = a0.copy()
        for t in range(steps):
            b = np.convolve(a, a)
            k = np.argmax(b)
            # gradient of b_k wrt a
            # g_i = 2 * a_{k-i} if index valid else 0
            idx = np.arange(n)

            mask = (j >= 0) & (j < n)
            g = np.zeros_like(a)

            # multiplicative update
            a = a * np.exp(-eta * g)
            a = project_simplex(a)
            # occasionally evaluate
            if t % 20 == 0:
                try_seq(a)
        return a

    # Run several restarts

        rand_vec = rand_vec / rand_vec.sum()

"""Hill‑climbing optimizer for the max‑auto‑convolution functional.

The algorithm explores simple parametric families (exponential,
power‑law) to obtain good seeds and then runs a greedy stochastic
transfer search that repeatedly moves a small amount of mass from one
position to another.  The objective is evaluated with a fast FFT‑based
convolution, which is equivalent to `evaluate_sequence`.  The
best sequence found before the time budget expires is returned.
"""

import numpy as np
import time

def _fast_score(a):
    """Return the same objective as evaluate_sequence but with FFT."""
    n = a.shape[0]
    # length of the linear convolution
    m = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << ((m).bit_length())
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)[:m]
    max_b = float(conv.max())
    sum_a = float(a.sum())
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed=42, budget_s=500, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 0.05)

    # length of the target vector – use the provided reference if available
    try:
        n = int(len(height_sequence_1))
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 1024
        base = np.ones(n, dtype=np.float64)

    if base.ndim != 1 or base.size != n:
        base = np.ones(n, dtype=np.float64)

    # normalise to a probability vector (the objective is scale invariant)
    base = np.clip(base, 0.0, 1000.0)
    base = base / base.sum()
    best_seq = base.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # helper that tries a candidate and updates the global best if it improves
    def _try(seq):
        nonlocal best_seq, best_score
        seq = np.clip(seq, 0.0, 1000.0)
        if seq.sum() < 1e-12:
            return
        # use the fast score for speed, but keep a reference to the official one
        sc = _fast_score(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            # also compute the official value (cheap when we already have a good one)
            real_sc = float(evaluate_sequence(best_seq.tolist()))
            print(f"[improve] new best fast={best_score:.6f} real={real_sc:.6f} time={time.monotonic():.2f}")

    # -----------------------------------------------------------------
    # 1.  Scan simple parametric families (exponential and power‑law)
    # -----------------------------------------------------------------
    idx = np.arange(n)
    rev = n - 1 - idx
    min_idx = np.minimum(idx, rev)

    # exponential decay towards the centre: a_i ∝ r^{min(i,n‑1‑i)}
    for r in np.linspace(0.970, 0.9999, 35):
        if time.monotonic() >= deadline:
            break
        w = r ** min_idx
        seq = w / w.sum()
        _try(seq)

    # power‑law decay: a_i ∝ (min(i+1,n‑i))^{-p}
    for p in np.linspace(0.2, 4.5, 35):
        if time.monotonic() >= deadline:
            break
        dist = np.minimum(idx + 1, n - idx)  # distance to the nearest end (1‑based)
        w = dist.astype(float) ** (-p)
        seq = w / w.sum()
        _try(seq)

    # a mixed family (exponential × power)
    for r in np.linspace(0.98, 0.998, 20):
        for p in np.linspace(0.5, 3.0, 12):
            if time.monotonic() >= deadline:
                break
            w = (r ** min_idx) * (dist.astype(float) ** (-p))
            seq = w / w.sum()
            _try(seq)

    # -----------------------------------------------------------------
    # 2.  Stochastic greedy transfer search (hill‑climbing)
    # -----------------------------------------------------------------
    # number of moves per restart – will be reduced as we run out of time
    moves_per_restart = 2000

    while time.monotonic() < deadline:
        # start from a fresh random simplex point
        a = rng.exponential(scale=1.0, size=n)
        a /= a.sum()

        current_score = _fast_score(a)
        # keep local best for this restart
        local_best_a = a
        local_best_score = current_score

        step_fraction = 0.5  # typical fraction of mass that may be moved

        for it in range(moves_per_restart):
            if time.monotonic() >= deadline:
                break
            # pick two distinct indices
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] <= 0.0:
                continue
            # amount to transfer – a random fraction of the source mass
            epsilon = a[i] * rng.random() * step_fraction
            if epsilon <= 0.0:
                continue

            # propose the transfer
            a[i] -= epsilon
            a[j] += epsilon

            # evaluate the candidate
            cand_score = _fast_score(a)

            if cand_score < local_best_score:
                # improvement – keep it
                local_best_a = a.copy()
                local_best_score = cand_score
                # occasionally update the global best
                if cand_score < best_score:
                    _try(a)

            else:
                # reject – undo the move
                a[i] += epsilon
                a[j] -= epsilon

            # slowly shrink step size to fine‑tune later in the run
            if (it + 1) % 200 == 0:
                step_fraction *= 0.95
                # also re‑normalise to avoid drift caused by rounding
                a /= a.sum()

        # after a restart keep the best we saw locally
        if local_best_score < best_score:
            _try(local_best_a)

    # final safety check – make sure the vector is still valid for
    # evaluate_sequence (non‑negative, sum>0.01).  If not, fall back to the
    # uniform vector.
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    if final_seq.sum() < 0.01:
        final_seq = np.ones(n, dtype=np.float64) / n

    print(f"[finish] best score {best_score:.6f} time {time.monotonic():.2f}")
    return [float(x) for x in final_seq]

"""Hill‑climbing optimizer for the max‑auto‑convolution functional.

The algorithm works in three stages:

1.  Scan a few simple parametric families (exponential, power‑law and a
    mixed family) to obtain good seed vectors.
2.  Perform a stochastic greedy “mass‑transfer’’ search.  In each move a
    random amount of mass is taken from one index and added to another;
    the move is kept only if the objective improves.  The objective is
    evaluated with a fast FFT‑based convolution that is equivalent to the
    provided ``evaluate_sequence`` function.
3.  The best sequence discovered before the time budget expires is
    returned (clipped to the required range).  The implementation respects
    the 1000 s time limit, uses only NumPy and the built‑in ``time`` module,
    and contains no file or network I/O.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective (identical to evaluate_sequence)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(b)/sum(a)^2 where b = a * a (linear convolution)."""
    n = a.shape[0]
    m = 2 * n - 1                     # length of the linear convolution
    size = 1 << ((m).bit_length())    # next power of two – fast FFT
    A = np.fft.rfft(a, n=size)
    B = A * A                         # element‑wise square = spectrum of a*a
    conv = np.fft.irfft(B, n=size)[:m]
    max_b = float(conv.max())
    sum_a = float(a.sum())
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

# ----------------------------------------------------------------------
# Main entry point required by the challenge framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a non‑negative sequence that minimizes the evaluation
    function.  `seed` controls the RNG, `budget_s` is the time budget
    (seconds).  The function returns the best sequence found as a list
    of floats.
    """
    rng = np.random.default_rng(seed)
    # reserve a small margin so we finish before the hard limit
    deadline = time.monotonic() + max(0.0, budget_s - 0.05)

    # --------------------------------------------------------------
    # Determine the length of the vector to optimise.
    # The reference construction (height_sequence_1) is available globally.
    # --------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 1024
        base = np.ones(n, dtype=np.float64)

    if base.ndim != 1 or base.size != n:
        base = np.ones(n, dtype=np.float64)

    # Normalise to a probability vector (the objective is scale invariant)
    base = np.clip(base, 0.0, 1000.0)
    base = base / base.sum()

    best_seq = base.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Helper that tries a candidate and updates the global best.
    # --------------------------------------------------------------
    def _try(seq: np.ndarray):
        nonlocal best_seq, best_score
        seq = np.clip(seq, 0.0, 1000.0)
        if seq.sum() < 1e-12:
            return
        sc = _fast_score(seq)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            # occasional exact check (cheap because improvements are rare)
            real_sc = float(evaluate_sequence(best_seq.tolist()))
            print(f"[improve] fast={best_score:.6f}  real={real_sc:.6f}  time={time.monotonic():.2f}")

    # ------------------------------------------------------------------
    # 1.  Scan a few low‑dimensional parametric families.
    # ------------------------------------------------------------------
    idx = np.arange(n)
    rev = n - 1 - idx
    min_idx = np.minimum(idx, rev)                         # distance to the nearer edge

    # --- exponential decay towards the centre: a_i ∝ r^{min(i,n‑1‑i)} ----------
    for r in np.linspace(0.970, 0.9999, 35):
        if time.monotonic() >= deadline:
            break
        w = r ** min_idx
        seq = w / w.sum()
        _try(seq)

    # --- power‑law decay: a_i ∝ (min(i+1,n‑i))^{-p} ---------------------------
    for p in np.linspace(0.2, 4.5, 35):
        if time.monotonic() >= deadline:
            break
        dist = np.minimum(idx + 1, n - idx)                 # 1‑based distance to nearest edge
        w = dist.astype(float) ** (-p)
        seq = w / w.sum()
        _try(seq)

    # --- mixed family (exponential × power) -----------------------------------
    for r in np.linspace(0.98, 0.998, 20):
        for p in np.linspace(0.5, 3.0, 12):
            if time.monotonic() >= deadline:
                break
            dist = np.minimum(idx + 1, n - idx)
            w = (r ** min_idx) * (dist.astype(float) ** (-p))
            seq = w / w.sum()
            _try(seq)

    # ------------------------------------------------------------------
    # 2.  Stochastic greedy mass‑transfer hill climbing.
    # ------------------------------------------------------------------
    moves_per_restart = 2000          # number of proposals in one restart

    while time.monotonic() < deadline:
        # start from a fresh random simplex point
        a = rng.exponential(scale=1.0, size=n)
        a /= a.sum()

        local_best_a = a.copy()
        local_best_score = _fast_score(a)

        # step size: fraction of the source mass that may be moved
        step_fraction = 0.5

        for it in range(moves_per_restart):
            if time.monotonic() >= deadline:
                break

            i, j = rng.choice(n, size=2, replace=False)
            if a[i] <= 0.0:
                continue
            # random fraction of the source component
            epsilon = a[i] * rng.random() * step_fraction
            if epsilon <= 0.0:
                continue

            # propose the transfer
            a[i] -= epsilon
            a[j] += epsilon

            cand_score = _fast_score(a)

            if cand_score < local_best_score:
                # improvement – keep the move
                local_best_a = a.copy()
                local_best_score = cand_score
                if cand_score < best_score:
                    _try(a)
            else:
                # reject – revert
                a[i] += epsilon
                a[j] -= epsilon

            # decay the step size periodically to allow fine tuning
            if (it + 1) % 200 == 0:
                step_fraction *= 0.95
                # renormalise to avoid drift caused by floating‑point noise
                a /= a.sum()

        # after the restart keep the locally best vector
        if local_best_score < best_score:
            _try(local_best_a)

    # ------------------------------------------------------------------
    # Final sanity check and return.
    # ------------------------------------------------------------------
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    if final_seq.sum() < 0.01:
        final_seq = np.ones(n, dtype=np.float64) / n

    print(f"[finish] best score={best_score:.6f}  time={time.monotonic():.2f}")
    return [float(x) for x in final_seq]
