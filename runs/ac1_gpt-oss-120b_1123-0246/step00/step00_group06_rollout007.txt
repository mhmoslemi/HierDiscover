def gradient_step(a, eta):
    n = len(a)
    b = np.convolve(a, a)
    k = np.argmax(b)
    # gradient of max_b w.r.t a is 2 * a shifted by k (reverse)
    # define grad = a shifted by k: for i such that 0 <= k - i < n:
    grad = np.zeros_like(a)
    lo = max(0, k - (n-1))
    hi = min(k, n-1)
    # i from lo..hi, then j=k-i is within range
    i_indices = np.arange(lo, hi+1)
    j_indices = k - i_indices
    grad[i_indices] = a[j_indices] * 2  # may double
    # we ignore factor 2; scaling step anyway
    # project gradient onto sum-zero
    avg = np.mean(grad)
    adj = grad - avg
    a_new = a - eta * adj
    a_new = np.clip(a_new, 0, 1000)
    # rescale to keep sum same? We can adjust.
    sum_diff = a_new.sum() - a.sum()
    if sum_diff != 0:
        # subtract equal amount from all entries (or scale)
        a_new -= sum_diff / a.size
        a_new = np.clip(a_new, 0, 1000)
    return a_new

"""
Gradient‑based local search for the step‑function minimisation problem.

The algorithm starts from a feasible sequence (the global `height_sequence_1`
if it exists, otherwise a uniform vector).  The sum of the sequence is kept
fixed (scale‑invariant objective) and the values are restricted to
[0,1000].

At each iteration we:
1. Compute the full convolution a*a and locate the lag k_max where the
   maximum occurs.
2. The gradient of the maximal convolution value w.r.t. the entries of a
   is   ∂max_b/∂a_i = 2·a_{k_max‑i} (zero when the index is out of range).
   Reducing this gradient lowers the dominating lag.
3. Project the gradient onto the hyper‑plane of zero‑sum (so the total
   mass stays constant) and take a small step in the opposite direction.
4. Clip to the allowed interval and renormalise the total sum.
5. Keep the best sequence seen so far.

The step size decays linearly with the remaining budget; every few hundred
iterations a random multiplicative perturbation is tried to escape possible
plateaus.  The whole procedure stops when the time budget (minus a safety
margin) is exhausted and returns the best sequence found.

This method differs from the block‑perturbation baseline by using an
explicit analytic gradient of the worst convolution lag, driving the shape
towards a “U‑shape’’ (small centre, larger ends) that reduces the peak
convolution and therefore the evaluation score.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a low‑score height sequence within the allotted time.
    Returns a list of floats (the best sequence found).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Initialise the starting sequence
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except NameError:
        # fallback: uniform vector of moderate length
        base = np.ones(256, dtype=np.float64)

    # enforce basic constraints
    base = np.clip(base, 0.0, 1000.0)
    if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        base = np.ones(256, dtype=np.float64)

    n = base.size
    # normalise the total mass (any positive constant works because the
    # objective is scale‑invariant).  Using sqrt(2n) reproduces the normalisation
    # from the reference paper.
    target_sum = np.sqrt(2.0 * n)
    if base.sum() == 0:
        base[:] = 1.0
    base *= (target_sum / base.sum())
    a = np.clip(base, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # 2. Evaluate the initial candidate
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Gradient‑descent loop (time‑bounded)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)  # safety margin
    start_time = time.monotonic()
    iteration = 0

    # initial step magnitude (relative to the typical entry size)
    init_eta = 0.05 * target_sum
    # schedule for step decay (linear)
    full_budget = budget_s

    while time.monotonic() < deadline:
        # ---- convolution and dominant lag --------------------------------
        b = np.convolve(a, a)               # length 2n‑1
        k_max = int(np.argmax(b))           # lag with maximal value

        # ---- gradient of the maximal convolution term -------------------
        grad = np.zeros_like(a)

        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        if lo <= hi:
            i_idx = np.arange(lo, hi + 1, dtype=np.int64)
            j_idx = k_max - i_idx
            # ∂max_b/∂a_i = 2 * a_{k_max‑i}; constant factor 2 is irrelevant
            grad[i_idx] = a[j_idx]

        # Project gradient to the sum‑zero subspace (preserve total mass)
        grad -= grad.mean()

        # ---- step size ----------------------------------------------------
        elapsed = time.monotonic() - start_time
        eta = init_eta * max(0.0, 1.0 - elapsed / full_budget)
        if eta < 1e-8:
            eta = 1e-8

        # ---- propose new sequence -----------------------------------------
        cand = a - eta * grad
        cand = np.clip(cand, 0.0, 1000.0)

        # Renormalise to keep the total sum exactly `target_sum`
        s = cand.sum()
        if s > 0.0:
            cand *= (target_sum / s)
        cand = np.clip(cand, 0.0, 1000.0)

        # ---- evaluate ------------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            a = cand                 # move the current point to the improvement
            print(f"[{iteration:6d}] new best  score = {best_score:.6f}")
        else:
            # keep walking even if not improved (helps to escape plateaus)
            a = cand

        # ---- occasional random perturbation to diversify ------------------
        if iteration % 500 == 0 and iteration > 0:
            pert = np.exp(rng.normal(0.0, 0.05, size=n))
            cand2 = a * pert
            cand2 = np.clip(cand2, 0.0, 1000.0)
            s2 = cand2.sum()
            if s2 > 0.0:
                cand2 *= (target_sum / s2)
            cand2 = np.clip(cand2, 0.0, 1000.0)
            score2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best = cand2.copy()
                best_score = score2
                a = cand2
                print(f"[{iteration:6d}] random perturb improved score = {best_score:.6f}")

        iteration += 1

    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    return [float(v) for v in best]
