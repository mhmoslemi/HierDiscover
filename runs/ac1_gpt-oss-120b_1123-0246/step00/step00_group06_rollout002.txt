def propose_candidate(seed=42, budget_s=500, **kwargs):
    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)
    # initial parent:
    parent = np.copy(np.asarray(globals().get('height_sequence_1', np.ones(256)), dtype=np.float64))
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.ndim != 1 or parent.size ==0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    current = parent
    current_score = evaluate_sequence(current.tolist())
    best = current.copy()
    best_score = current_score

    # temperature schedule:
    T0 = 1.0
    # maybe adapt based on initial score
    if np.isfinite(best_score):
        T0 = best_score * 0.1
    else:
        T0 = 0.1

    iteration = 0
    last_print = time.monotonic()
    while time.monotonic() < deadline:
        iteration += 1
        # Choose move
        move_type = rng.integers(0, 5)  # 0..4
        cand = current.copy()
        n = cand.size
        if move_type == 0:
            # block scaling similar to provided
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start:stop] *= factor
        elif move_type == 1:
            # transfer mass between two indices
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                # choose delta as a proportion of smaller
                delta = rng.random() * min(cand[i], cand[i] * 0.2)  # up to 20% of i
                # move from i to j
                cand[i] = max(0.0, cand[i] - delta)
                cand[j] = cand[j] + delta
        elif move_type == 2:
            # add small Gaussian noise to random indices
            idx = rng.integers(0, n)
            noise = rng.normal(0.0, cand[idx] * 0.1 + 0.01)  # relative scale
            cand[idx] = max(0.0, cand[idx] + noise)
        elif move_type == 3:
            # mirror symmetry (enforce near-symmetry)
            if rng.random() < 0.5:
                cand = 0.5 * (cand + cand[::-1])
        elif move_type == 4:
            # global random reinitialization (with same sum)
            # generate new random weights
            m = n
            # use Dirichlet distribution to allocate sum
            # Use gamma draws to produce positive values
            rand = rng.gamma(shape=1.0, scale=1.0, size=m)
            cand = np.clip(rand, 0.0, 1000.0)
        # clip to allowed range
        cand = np.clip(cand, 0.0, 1000.0)

        # evaluate
        score = evaluate_sequence(cand.tolist())
        if not np.isfinite(score):
            continue

        # accept or reject
        if score < current_score or rng.random() < math.exp(-(score - current_score) / max(T0, 1e-12)):
            current = cand
            current_score = score

        # update best
        if score < best_score:
            best_score = score
            best = cand.copy()
            # log
            now = time.monotonic()
            if now - last_print > 10.0:
                print(f"Iter {iteration}, time {now - (deadline - budget_s):.1f}s, best_score {best_score:.6f}")
                last_print = now

        # temperature decay
        elapsed = (deadline - time.monotonic()) / budget_s
        T = T0 * elapsed  # linearly decreases
        T0 = max(T, 1e-9)
    return best.tolist()

"""
Propose a low‑score step‑function using a mixed local‑search / simulated‑annealing
strategy.

The core idea is to flatten the self‑convolution of the sequence.
At each iteration we look at the bin k where (a*a)[k] is maximal and we move a
fraction of the mass that contributes most to that bin to a random position.
Occasional block‑scaling, Gaussian noise and full re‑initialisation moves keep
the search from getting stuck.  The algorithm runs for the allotted time budget
(≈ budget_s seconds) and returns the best sequence discovered.

All helper functions are defined at top level.  No file I/O is performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _score_and_conv(seq: np.ndarray):
    """
    Compute the evaluation score and the full convolution.
    Mirrors the logic of ``evaluate_sequence`` (including clipping and guards)
    but is much faster because it works directly on NumPy arrays.
    Returns (score, conv) where ``conv`` is the full convolution (np.ndarray)
    or None if the sequence is invalid.
    """
    # clip to the allowed interval
    a = np.clip(seq, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01 or not np.isfinite(sum_a):
        return np.inf, None
    # full linear convolution (length 2*n-1)
    conv = np.convolve(a, a)
    max_b = conv.max()
    n = a.size
    # the evaluation formula
    score = float(2.0 * n * max_b / (sum_a * sum_a))
    return score, conv


def _initial_sequence(rng: np.random.Generator, length: int):
    """
    Produce an initial candidate.
    Prefer a previously discovered sequence if the global variable
    ``height_sequence_1`` exists; otherwise fall back to a uniform vector.
    """
    global height_sequence_1
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            base = np.clip(base, 0.0, 1000.0)
            # optionally resize to the requested length (keep as many entries as possible)
            if base.size != length:
                # repeat / truncate to the wanted size
                repeats = max(1, (length + base.size - 1) // base.size)
                base = np.tile(base, repeats)[:length]
            return base.astype(np.float64)
    except Exception:
        pass
    # fallback: uniform values (all 1.0) – this gives a starting score of exactly 2.0
    return np.ones(length, dtype=np.float64)


def _block_perturb(seq: np.ndarray, rng: np.random.Generator):
    """
    Multiplicative perturbation of a randomly chosen contiguous block.
    Mirrors the move used in the reference implementation.
    """
    n = seq.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    new_seq = seq.copy()
    new_seq[start:stop] *= factor
    # occasional symmetrisation
    if rng.random() < 0.35:
        new_seq = 0.5 * (new_seq + new_seq[::-1])
    return new_seq


def _noise_perturb(seq: np.ndarray, rng: np.random.Generator):
    """
    Add a small Gaussian perturbation to a single randomly chosen entry.
    """
    new_seq = seq.copy()
    idx = int(rng.integers(0, seq.size))
    # relative magnitude of the noise is 10 % of the current value (or 0.01)
    scale = new_seq[idx] * 0.1 + 0.01
    noise = rng.normal(0.0, scale)
    new_seq[idx] = max(0.0, new_seq[idx] + noise)
    return new_seq


def _random_reinit(seq_len: int, rng: np.random.Generator):
    """
    Produce a brand‑new random positive vector of the given length.
    The distribution is a Gamma(1,1) (i.e. exponential) – this gives a
    spread of magnitudes while staying far from the boundary.
    """
    cand = rng.gamma(shape=1.0, scale=1.0, size=seq_len)
    return np.clip(cand, 0.0, 1000.0)


def _targeted_move(seq: np.ndarray, conv: np.ndarray, max_idx: int,
                   rng: np.random.Generator, alpha: float):
    """
    Reduce the contribution to the currently maximal convolution bin.
    Identify the index pair (i, j) that maximises a_i * a_j for i + j = max_idx,
    take a fraction alpha of the smaller of the two values and move that
    mass to a random location.
    """
    n = seq.size
    # indices that contribute to the bin ``max_idx``
    i_vals = np.arange(n, dtype=np.int64)
    j_vals = max_idx - i_vals
    mask = (j_vals >= 0) & (j_vals < n)

    if not mask.any():
        # fallback to a harmless perturbation
        return _block_perturb(seq, rng)

    # contribution of each i (note that the pair (i,j) and (j,i) will be examined
    # twice when i != j – that is fine because the product is the same).
    contrib = np.zeros_like(seq)
    contrib[mask] = seq[i_vals[mask]] * seq[j_vals[mask]]

    i_max = int(contrib.argmax())
    j_max = int(max_idx - i_max)

    # guard against pathological degenerate cases
    if seq[i_max] <= 0.0 or (i_max != j_max and seq[j_max] <= 0.0):
        # fallback to a block perturbation
        return _block_perturb(seq, rng)

    new_seq = seq.copy()

    if i_max == j_max:
        # self‑pair (contributes a_i^2)
        delta = alpha * new_seq[i_max]
        new_seq[i_max] = max(0.0, new_seq[i_max] - delta)
        # redistribute the removed mass uniformly to a random position
        k = int(rng.integers(0, n))
        new_seq[k] = min(1000.0, new_seq[k] + delta)
    else:
        # off‑diagonal pair (contributes 2*a_i*a_j to the bin)
        delta = alpha * min(new_seq[i_max], new_seq[j_max])
        if delta <= 0.0:
            return _block_perturb(seq, rng)
        new_seq[i_max] -= delta
        new_seq[j_max] -= delta
        # spread the removed mass to two random locations
        k1 = int(rng.integers(0, n))
        k2 = int(rng.integers(0, n))
        # divide equally – the division by two is safe because delta>0
        new_seq[k1] = min(1000.0, new_seq[k1] + delta * 0.5)
        new_seq[k2] = min(1000.0, new_seq[k2] + delta * 0.5)

    return new_seq

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a height sequence with a low evaluation score.

    The routine runs for approximately ``budget_s`` seconds, performing a
    simulated‑annealing / hill‑climbing loop that mixes several mutation
    operators (block scaling, noise, targeted max‑bin reduction, and full
    reinitialisation).  The best sequence observed is returned as a plain
    Python list of floats.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget in seconds.
    **kwargs
        Ignored – present for compatibility with the surrounding framework.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # a tiny safety margin

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    target_len = 5582                      # length used by the reference run
    current = _initial_sequence(rng, target_len)
    current_score, current_conv = _score_and_conv(current)
    if not np.isfinite(current_score):
        # fall back to a trivial uniform vector
        current = np.ones(target_len, dtype=np.float64)
        current_score, current_conv = _score_and_conv(current)

    best = current.copy()
    best_score = current_score

    # temperature schedule (simulated annealing)
    # start temperature is a small fraction of the initial score;
    # it linearly decays to zero.
    init_temp = max(0.01, current_score * 0.2)
    # parameters governing the mix of mutation types
    prob_targeted = 0.6
    prob_block = 0.2
    prob_noise = 0.15
    prob_restart = 0.05

    iteration = 0
    last_report = start_time

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        # fraction of budget already spent (for cooling and reporting)
        elapsed = time.monotonic() - start_time
        frac = elapsed / budget_s
        # linear cooling
        temperature = init_temp * (1.0 - frac)

        # decide which operator to use
        move_choice = rng.random()
        if move_choice < prob_targeted:
            # targeted reduction of the maximal bin
            # we already have ``current_conv`` from the previous iteration
            max_idx = int(current_conv.argmax())
            alpha = 0.15 * (1.0 - frac) + 0.02   # slowly shrink the step size
            candidate = _targeted_move(current, current_conv, max_idx, rng, alpha)
        elif move_choice < prob_targeted + prob_block:
            candidate = _block_perturb(current, rng)
        elif move_choice < prob_targeted + prob_block + prob_noise:
            candidate = _noise_perturb(current, rng)
        else:
            # full random restart – keep the best so far as a seed occasionally
            if rng.random() < 0.3 and best_score < np.inf:
                candidate = best.copy()
                # add a slight random scaling so we are not stuck on the exact same point
                candidate = _block_perturb(candidate, rng)
            else:
                candidate = _random_reinit(target_len, rng)

        # clip before scoring (required by the official evaluator)
        candidate = np.clip(candidate, 0.0, 1000.0)

        cand_score, cand_conv = _score_and_conv(candidate)
        if not np.isfinite(cand_score):
            continue

        # Simulated‑annealing acceptance
        if cand_score < current_score or rng.random() < math.exp(-(cand_score - current_score) / max(temperature, 1e-12)):
            current = candidate
            current_score = cand_score
            current_conv = cand_conv

        # Keep the global best
        if cand_score < best_score:
            best = candidate.copy()
            best_score = cand_score
            now = time.monotonic()
            if now - last_report > 5.0:   # periodic status line
                print(f"[{now - start_time:6.1f}s] iteration {iteration:,} – new best score: {best_score:.6f}")
                last_report = now

    # ------------------------------------------------------------------
    # return the best solution found
    # ------------------------------------------------------------------
    print(f"Search finished after {time.monotonic() - start_time:.1f}s, best score = {best_score:.6f}")
    return [float(x) for x in best]
