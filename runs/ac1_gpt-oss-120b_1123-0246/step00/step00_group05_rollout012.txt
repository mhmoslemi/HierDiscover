def fft_convolve(x, y):
    n = len(x) + len(y) - 1
    L = 1 << (n-1).bit_length()

    Y = np.fft.rfft(y, L)

    return conv

"""
Propose a sequence of non‑negative heights that minimizes the
evaluation function.

The algorithm works in three stages:

1. **Parameter sweep** – generate a collection of analytically
   motivated families (exponential, power‑law, linear, reversed) and
   keep the best one.

2. **Soft‑max gradient descent** – minimise a smooth approximation of
   the maximal convolution value.
   For a vector `a` we compute `b = a * a` (linear convolution) and
   construct weights `w_k ∝ exp(α·b_k)`.  The gradient of the smooth max
   w.r.t. `a` is `g = 2·conv(w, a[::-1])`.  A normalised step in the
   negative gradient direction is taken, the result is clipped to
   non‑negative values and renormalised to keep the L1‑norm equal to 1.

   The temperature‑like parameter `α` is increased gradually so that the
   optimisation gradually focuses on the true peak.

3. **Simulated‑annealing block perturbations** – every few iterations a
   cheap random block perturbation (the original “make_block_perturbation”
   routine) is applied and accepted with a Metropolis probability that
   slowly cools down.  This helps to escape shallow local minima.

The best sequence seen so far is always returned.  All operations are
O(n log n) thanks to FFT based convolutions, so the search comfortably
fits into the given time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: FFT based linear convolution (real inputs, full length)
def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.size + y.size - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    conv = np.fft.irfft(X * Y, size)[:n]
    return conv


# ----------------------------------------------------------------------
# Helper: block perturbation (the original baseline routine)
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return one O(n) memory non‑negative perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Helper: generate a handful of analytically motivated candidates
def parametric_candidates(n: int) -> list[np.ndarray]:
    """Return a list of candidate vectors based on simple formulas."""
    cand = []

    # exponential decay (and its reverse)
    for r in (0.99, 0.98, 0.95, 0.90):
        a = np.array([r ** i for i in range(n)], dtype=np.float64)
        a /= a.sum()
        cand.append(a)
        cand.append(a[::-1])

    # power‑law tails
    for p in (0.5, 1.0, 1.5, 2.0):
        a = np.array([1.0 / (i + 1) ** p for i in range(n)], dtype=np.float64)
        a /= a.sum()
        cand.append(a)
        cand.append(a[::-1])

    # simple linear ramps
    a = np.arange(1, n + 1, dtype=np.float64)   # 1,2,…,n
    a /= a.sum()
    cand.append(a)
    cand.append(a[::-1])

    # constant (uniform) – the baseline
    a = np.ones(n, dtype=np.float64)
    a /= a.sum()
    cand.append(a)

    return cand

    The function respects the time budget (in seconds) and returns the best
    sequence discovered as a list of floats.
    """

    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # Initialise a starting vector -------------------------------------------------

        base = np.clip(base, 0.0, 1000.0)

    else:
        base = np.ones(256, dtype=np.float64)

    n = base.size
    parent = np.maximum(base, 1e-12)                # avoid exact zeros
    parent = parent / parent.sum()                  # normalise to L1 = 1

    # ------------------------------------------------------------------

    best = parent.copy()

        cand = np.clip(cand + jitter, 0.0, None)
        cand /= cand.sum()
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            print(f"[param] improved to {best_score:.6f}")

    parent = best.copy()                     # start the local search from the best found

    # ------------------------------------------------------------------
    # Hyper‑parameters for the gradient part
    base_lr = 0.5
    alpha = 1.0               # soft‑max sharpness
    max_alpha = 60.0
    min_alpha = 0.5
    alpha_inc = 1.02          # geometric increase per iteration
    it = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    while time.monotonic() < deadline:
        # ---- soft‑max gradient descent step ------------------------------------
        # compute convolution a*a (full length 2n‑1)
        D = fft_convolve(parent, parent)
        D = np.clip(D, 0.0, None)                    # guard against tiny negatives

        # soft‑max weights focusing on the largest entries
        maxD = D.max()
        w = np.exp(alpha * (D - maxD))               # shift by max to avoid overflow
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(D) / D.size
        else:
            w /= w_sum                               # now w sums to 1

        # gradient of the smooth max (2 * conv(w, parent[::-1]))
        grad_full = fft_convolve(w, parent[::-1])
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]

        # normalise step size
        norm = np.linalg.norm(grad) + 1e-12
        step = base_lr / norm

        # propose a new vector
        cand = parent - step * grad
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            # if everything vanished, fall back to a small uniform perturbation
            cand = np.ones_like(parent) / n

        # occasional random block perturbation (simulated‑annealing style)
        if rng.random() < 0.15:
            cand = make_block_perturbation(cand, rng)

        # ---- evaluation & acceptance -----------------------------------------
        if it % 5 == 0:                               # evaluate every few steps
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand.copy()
                best_score = score
                parent = cand.copy()
                print(f"[{it:05d}] new best {best_score:.6f}  alpha={alpha:.3f}")
                no_improve = 0
                # after a genuine improvement we slightly increase the learning
                # rate to move faster
                base_lr = min(2.0, base_lr * 1.05)
            else:
                # Metropolis acceptance for worse candidates (simulated annealing)
                # temperature proportional to remaining time
                progress = (time.monotonic() - (deadline - budget_s)) / budget_s
                temperature = max(1e-4, (1.0 - progress) * best_score * 0.1)
                delta = score - best_score
                if delta < 0 or rng.random() < np.exp(-delta / temperature):
                    parent = cand.copy()
                else:
                    # keep the old parent
                    pass
                no_improve += 1
        else:
            # on non‑evaluation steps we simply move the parent
            parent = cand.copy()

        # increase the soft‑max sharpness slowly (focus more on the real max)
        alpha = min(max_alpha, max(min_alpha, alpha * alpha_inc))

        # occasional random restart if stuck for long
        if no_improve > 300 and rng.random() < 0.3:
            # perturb the current best a little and continue
            noise = rng.normal(0.0, 0.02, size=n)
            parent = np.clip(best + noise, 0.0, None)
            parent /= parent.sum()
            print(f"[{it:05d}] restart – adding jitter")
            no_improve = 0

        it += 1

    # Final evaluation of the best sequence found
    final_score = float(evaluate_sequence(best.tolist()))
    print(f"[finished] best score = {final_score:.6f} after {it} iterations")
    return [float(x) for x in best]
