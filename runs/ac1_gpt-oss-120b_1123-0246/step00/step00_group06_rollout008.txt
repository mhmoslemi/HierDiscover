"""
Propose a better step‑function sequence using a Frank‑Wolfe style iteration.

The algorithm works as follows:

1. Start from the global ``height_sequence_1`` (or a uniform fallback) and
   normalise it to an L1‑norm of ``1`` – the evaluation function is
   homogeneous, so the absolute scale does not matter.

2. Repeatedly:
   a) Compute the self‑convolution ``c = f * f`` and its current maximum
      ``M = max(c)``.
   b) Build a *small* set of linear constraints that enforce
      ``f * g ≤ M`` only for the most critical convolution indices
      (those where ``c`` is close to ``M``).  For a fixed ``f`` the
      expression ``f * g`` is linear in the unknown vector ``g``.
   c) Solve a linear programme that **maximises** ``sum(g)`` subject to
      the constraints from (b) and ``g ≥ 0`` (bounds are also limited to
      ``1000`` because of the evaluator’s clipping).  This yields a
      direction ``g`` that can potentially increase the L1‑mass while
      keeping the worst convolution value ≤ ``M``.
   d) Scale ``g`` back to the original L1‑norm of ``f`` and take a convex
      combination ``f ← (1‑t)·f + t·g`` (``t`` is a small step, here 0.2).
   e) Keep the new ``f`` only if the evaluation improves.

3. If the LP fails or does not improve the score, fall back to a cheap
   random block perturbation (the method used in the reference baseline).

The loop stops when the allotted time budget is exhausted.  The best
sequence found is returned as a plain list of ``float`` values ready to be
passed to ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

def make_block_perturbation(base, rng):
    """Return a cheap O(n) memory perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # Width of the block is a random fraction of the length.
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # Random multiplicative perturbation on the selected block.
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # Occasionally enforce a mild symmetry – this helped the baseline.
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def _build_constraint_matrix(f, idxs):
    """
    Build a sparse matrix A where each row corresponds to one index ``k``
    from ``idxs`` and contains the coefficients of the linear expression
    (f * g)[k] ≤ M. The matrix therefore has shape (len(idxs), len(f)).
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for row, k in enumerate(idxs):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        # columns j that appear in this row
        js = np.arange(j_start, j_end + 1, dtype=np.int64)
        coeffs = f[k - js]          # f[i] where i = k - j
        rows.extend([row] * js.size)
        cols.extend(js.tolist())
        data.extend(coeffs.tolist())
    return sp.csr_matrix((data, (rows, cols)), shape=(len(idxs), n))


def propose_candidate(seed=42, budget_s=500, **kwargs):
    """
    Search for a sequence with a low evaluation score.

    The function respects the given time ``budget_s`` (seconds).  It
    returns a plain ``list`` of ``float`` that can be fed to the supplied
    ``evaluate_sequence`` routine.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- initialise -------------------------------------------------
    # Use the global construction if it exists, otherwise fall back.
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        parent = np.ones(256, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    if parent.sum() == 0.0:
        parent = np.ones_like(parent)

    # Normalise to L1 = 1 (scale does not affect the evaluation)
    f = parent / parent.sum()
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial score: {best_score:.6f}")

    iteration = 0
    # Parameters controlling the LP
    max_constraints = 200          # maximum number of active convolution rows
    step_size = 0.2                # mixing factor (1‑t)

    while time.monotonic() < deadline:
        # --- current convolution and bound --------------------------------
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # Pick indices where the convolution is close to its maximum.
        # Using a relative threshold keeps the problem small.
        active = np.where(conv_f >= 0.90 * M)[0]
        if active.size > max_constraints:
            # keep the highest values only
            order = np.argsort(-conv_f[active])[:max_constraints]
            active = active[order]

        # Build the linear constraints (f * g <= M for the selected rows)
        A = _build_constraint_matrix(f, active)
        b = np.full(A.shape[0], M, dtype=np.float64)

        # Linear programme: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(f.shape[0], dtype=np.float64)
        bounds = [(0.0, 1000.0) for _ in range(f.shape[0])]

        # Solve the LP (SciPy’s HiGHS works well with sparse matrices)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})

        if res.success:
            g = res.x
            sum_f = f.sum()
            sum_g = g.sum()
            if sum_g > 0.0:
                # Rescale g so that it has the same L1‑norm as f.
                g = g * (sum_f / sum_g)
                # Convex combination – this is the Frank‑Wolfe update.
                new_f = (1.0 - step_size) * f + step_size * g
                new_f = np.clip(new_f, 0.0, 1000.0)
                # Optional: re‑normalise to keep the L1 exactly unchanged.
                # new_f = new_f * (sum_f / new_f.sum())
                score = float(evaluate_sequence(new_f.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = new_f
                    best_score = score
                    f = best.copy()
                    print(f"Iter {iteration:04d}: LP improvement → {best_score:.6f}")
        else:
            # In the rare case the LP fails, fall back to a cheap perturbation.
            candidate = make_block_perturbation(best, rng)
            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best = candidate
                best_score = score
                f = best.copy()
                print(f"Iter {iteration:04d}: random perturb → {best_score:.6f}")

        iteration += 1

    # Finally return the best sequence found (as plain floats).
    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    return [float(x) for x in best]
