def simplex_projection(v):
    # Project onto simplex sum(v)=1, v>=0
    # Using algorithm from:
    # sort v descending, compute cumulative sum, find rho, lambda.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

    rng = np.random.default_rng(seed)
    # Determine n

    if 'height_sequence_1' in globals():

        a = np.clip(a, 0, None)

        if a.sum() == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / a.sum()
    else:
        n = 6121
        a = np.full(n, 1.0/n)
    # Ensure total sum ~1
    a = a / a.sum()
    # Evaluate initial score using internal evaluation
    def compute_score(vec):
        # using FFT for speed
        fa = np.fft.rfft(vec)
        conv = np.fft.irfft(fa * fa, n=2*n-1)
        max_b = conv.max()
        return float(2 * n * max_b)  # sum=1 so denominator 1
    cur_score = compute_score(a)
    best_seq = a.copy()

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    start_time = time.monotonic()

    # temperature schedule: exponential cooling from T0 to Tmin
    T0 = 0.2
    Tmin = 1e-6

    while time.monotonic() < deadline:

        elapsed = time.monotonic() - start_time
        fraction = elapsed / (deadline - start_time)
        T = T0 * (Tmin / T0) ** fraction

        cand = a.copy()
        # Choose move type
        if rng.random() < 0.75:

            i, j = rng.integers(n, size=2)

            cand[j] += delta

"""
Propose a step‐function (list of non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where n = len(a) and conv is the linear (full) convolution.
The algorithm uses a fast projected simulated‑annealing / hill‑climbing
search:

* The sequence is kept on the probability simplex (sum = 1, all entries ≥ 0).
* Two types of elementary moves are applied:
    - Transfer a small amount of mass between two positions.
    - Scale a random contiguous block (multiplicative perturbation) and
      renormalise.
* After each move the new score (computed with an FFT‑based convolution)
  is accepted if it improves the objective, otherwise with a Metropolis
  probability that slowly cools down.
* The current best sequence is remembered and printed whenever it improves.
* The whole search stops after the allocated time budget (default 500 s)
  and the best sequence found so far is returned.

All heavy numeric work is O(n log n) per iteration (FFT), which easily
fits into the 1000 s global limit for n≈6000.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    # Implements the algorithm from
    #   Wang & Carreira‑Perpiñán, “Projection onto the probability simplex:
    #   An efficient algorithm with a simple proof, and an application”.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, len(v) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_fft(a: np.ndarray) -> float:
    """
    Fast evaluation of 2 * n * max(conv(a, a)) when sum(a) == 1.
    Uses an FFT of length 2*n-1 (zero‑padded) to obtain the full linear
    convolution.
    """
    n = a.shape[0]
    # rfft with explicit length 2*n-1 pads with zeros automatically
    fa = np.fft.rfft(a, n=2 * n - 1)
    conv = np.fft.irfft(fa * fa, n=2 * n - 1)
    max_b = conv.max()
    return float(2.0 * n * max_b)   # denominator sum(a)**2 = 1

        T = T_start * (T_end / T_start) ** frac

        # ---- propose a neighbour ---------------------------------------------------
        cand = a.copy()
        if rng.random() < 0.72:                     # 72 % : pairwise mass transfer
            i, j = rng.integers(n, size=2)
            if i != j and cand[i] > 0.0:
                # move at most 30 % of the source mass
                delta = rng.random() * 0.30 * cand[i]
                cand[i] -= delta
                cand[j] += delta
        else:                                        # 28 % : block scaling + renorm
            # random block width (similar to the reference implementation)
            lo = max(1, n // 200)
            hi = max(2, n // 12)
            width = int(rng.integers(lo, hi + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.11))
            cand[start:start + width] *= factor
            # Small chance to symmetrise (helps flatten the convolution)
            if rng.random() < 0.15:
                cand = 0.5 * (cand + cand[::-1])
            # Re‑normalise to keep sum == 1
            s = cand.sum()
            if s > 0.0:
                cand = cand / s
            else:
                cand = np.full(n, 1.0 / n, dtype=np.float64)

        # ---- evaluate --------------------------------------------------------------
        cand_score = _score_fft(cand)

        # ---- Metropolis acceptance -------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability of accepting a worse move
            if T > 0.0:
                prob = math.exp(-(cand_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                # occasional progress report
                now = time.monotonic() - start_time
                print(f"[{now:6.1f}s] iter {iter_count:7d} "
                      f"new best score = {best_score:.6f}")

        iter_count += 1

    # Final check (ensures the external evaluator sees the same numbers)
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Search finished – best internal score {best_score:.6f}, "
          f"external score {final_score:.6f}, iterations {iter_count}")

    # Return plain Python list as required by the framework
    return best_seq.tolist()
