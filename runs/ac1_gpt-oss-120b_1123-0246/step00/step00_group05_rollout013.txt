def propose_candidate(seed=42, budget_s=500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0, float(budget_s) - 0.1)  # small buffer.
    n = len(height_sequence_1) if defined else default maybe 256 but should be length 4498 maybe from original? We'll attempt to read global variable.

    # if invalid, fallback to uniform 1s.

    # optionally normalize sum to 1 (scaling? not needed).
    best = parent
    best_score = float(evaluate_sequence(best.tolist()))
    # Track iteration count.
    iteration = 0

    while time.monotonic() < deadline:
        # Choose random move type with probabilities
        r = rng.random()
        if r < 0.4:
            cand = make_block_perturbation(best, rng)
        elif r < 0.6:
            # swap two values
            cand = best.copy()
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif r < 0.8:
            # add noise to random subset
            cand = best.copy()
            subset_size = int(rng.integers(1, max(2, n//50)))
            indices = rng.choice(n, size=subset_size, replace=False)
            cand[indices] += np.exp(rng.normal(0.0, 0.05, size=subset_size))
        else:
            # random way: generate new shape (decreasing power law)
            expo = rng.uniform(0.0, 3.0)  # shape exponent
            cand = np.linspace(1.0, 0.0, n) ** expo
            # scale heights random factor
            cand = cand * rng.uniform(0.5, 2.0)
            cand = np.clip(cand, 0.0, 1000.0)
        # optional symmetrize with probability
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])
        # ensure clipping
        cand = np.clip(cand, 0.0, 1000.0)
        # compute score
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"Iter {iteration}: new best score {best_score:.6f}")
        # optionally accept with SA (makes no sense but maybe helps)
        # else keep old best; we could also accept worse to move away
        iteration += 1

    return [float(x) for x in best]

def reduce_peak_move(seq, rng):
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)
    n = len(seq)
    # gather paired indices that contribute
    contrib_indices = [(i, k - i) for i in range(max(0, k - n + 1), min(n - 1, k) + 1) if seq[i] > 0 and seq[k - i] > 0]
    # Choose one pair weighted by product
    probs = np.array([seq[i] * seq[j] for i,j in contrib_indices])
    probs_sum = probs.sum()
    if probs_sum == 0:
        return seq.copy()
    probs = probs / probs_sum
    idx = rng.choice(len(contrib_indices), p=probs)
    i, j = contrib_indices[idx]
    # Choose direction: reduce either i or j
    if rng.random() < 0.5:
        # reduce a_i
        delta = min(seq[i], seq[i] * rng.uniform(0.01, 0.2))
        new_seq = seq.copy()
        new_seq[i] -= delta
        # add delta to random index not equal to i or j
        target = rng.integers(0, n)
        # ensure target not equal to i to avoid immediate product increase with its counterpart
        # we can randomly ensure that target not lead to new high product.
        new_seq[target] += delta
    else:
        # reduce a_j analogously
        delta = min(seq[j], seq[j] * rng.uniform(0.01, 0.2))
        new_seq = seq.copy()
        new_seq[j] -= delta
        target = rng.integers(0, n)
        new_seq[target] += delta
    # Clip
    new_seq = np.clip(new_seq, 0, 1000)
    return new_seq

def reduce_peak(seq, rng):
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))
    n = len(seq)
    # compute contributions list as pairs (i, j) where i+j=k and i≤j
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    # contributions: for i in range(i_start, i_end+1), j = k - i
    contributions = []
    weights = []
    for i in range(i_start, i_end + 1):
        j = k - i
        if i > j: 
            continue # avoid double counting if we treat unordered pairs
        prod = seq[i] * seq[j]
        if prod > 0:
            contributions.append((i, j, prod))
            weights.append(prod)
    if not contributions:
        # fallback random block perturb
        return make_block_perturbation(seq, rng)
    # pick a pair with probability proportional to prod
    probs = np.array(weights) / np.sum(weights)
    idx = rng.choice(len(contributions), p=probs)
    i, j, prod = contributions[idx]
    # define delta as fraction of smaller of seq[i], seq[j]
    # We will move mass from larger of them to elsewhere.
    if i == j:
        # self-pair, reduce this value a_i
        delta = seq[i] * rng.uniform(0.01, 0.15)  # reduce by 1-15%
        if delta > seq[i]:
            delta = seq[i]
        cand = seq.copy()
        cand[i] -= delta
        # add delta to random index not i
        tgt = rng.integers(0, n)
        while tgt == i:
            tgt = rng.integers(0, n)
        cand[tgt] += delta
    else:
        # reduce both a_i and a_j a little, but add mass to other index
        # Determine smaller of the two for delta minimal to keep nonnegative
        smaller = min(seq[i], seq[j])
        delta = smaller * rng.uniform(0.01, 0.15)
        if delta <= 0:
            delta = 0
        cand = seq.copy()
        # subtract delta from both
        cand[i] -= delta
        cand[j] -= delta
        # add total delta*2 to random index
        tgt = rng.integers(0, n)
        while tgt == i or tgt == j:
            tgt = rng.integers(0, n)
        cand[tgt] += 2 * delta
    cand = np.clip(cand, 0.0, 1000.0)
    return cand
