"""
Simple power‑law generator.

The evaluation function is scale‑invariant.  A monotone power‑law
sequence a_i = (i/(n‑1))**p has small “self‑convolution” peak because
the overlap at any shift multiplies a small value with a large one.
The maximum of the convolution occurs at the centre and equals
∑ a_i a_{n‑1‑i} ≈ n·B(p+1,p+1).  The resulting score
2·n·max_b/(∑a)^2 = 2(p+1)^2·B(p+1,p+1) decreases rapidly with p,
so even modest p (>1) beats the target 1.503.

The proposer evaluates a handful of exponents, keeps the best, and
then spends the remaining time randomly sampling new exponents and
tiny multiplicative perturbations to possibly improve further.
"""

import time
import numpy as np

def powerlaw_sequence(n: int, p: float) -> np.ndarray:
    """Return a monotone power‑law sequence of length n (0 ≤ x ≤ 1)."""
    if n <= 1:
        # Degenerate case – return a constant non‑zero vector.
        return np.ones(max(n, 1), dtype=np.float64)
    # Normalised index in [0,1]
    idx = np.arange(n, dtype=np.float64)
    seq = (idx / (n - 1)) ** p
    # Safety clipping – evaluate_sequence also clips, but we do it here.
    return np.clip(seq, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a sequence of non‑negative heights that minimises evaluate_sequence.

    The routine builds a monotone power‑law shape a_i = (i/(n‑1))**p.
    Several exponents are evaluated, the best is kept, and the remaining
    time budget is spent searching random exponents + tiny multiplicative
    noise.  The final best sequence (as a plain Python list) is returned.
