def project_to_simplex(v):
    # ensure sum(v)=1, v>=0
    # from https://doi.org/10.1007/978-3-540-68694-9_13
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

```python
import numpy as np
import time

def project_to_simplex(v):
    """Projection of vector v onto the probability simplex {w >= 0, sum w = 1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # Find the rho
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed=42, budget_s=500, **kwargs):
    """
    Projected gradient descent on a smooth approximation of the
    max‑convolution objective.
    The algorithm works on a probability distribution w (sum = 1) of length n.
    It minimizes 2*n*max(conv(w,w)) which is exactly the value returned by
    evaluate_sequence after scaling the result to the range [0,1000].
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Length of the step function – a larger n gives a lower theoretical bound.
    n = 2000

    # Initialise with the uniform distribution (the optimum of the sum‑squares term).
    w = np.full(n, 1.0 / n, dtype=np.float64)

    # Pre‑compute FFT sizes.
    conv_len = 2 * n - 1
    fft_size_conv = 1 << (conv_len - 1).bit_length()  # next power of two
    grad_len = 3 * n - 2
    fft_size_grad = 1 << (grad_len - 1).bit_length()

    best = w.copy()
    best_score = 2.0 * n * np.max(np.convolve(w, w))

    # Parameters for the smoothed max and the step schedule.
    alpha = 5.0                 # initial smoothing factor
    alpha_inc = 0.02            # linear increase per iteration
    step0 = 1.0                 # base step size

    it = 0
    while time.monotonic() < deadline:
        # ----- convolution w * w (full) -----
        fw = np.fft.fft(w, fft_size_conv)
        conv_full = np.fft.ifft(fw * fw).real[:conv_len]

        # ----- smooth approximation of max -----
        # Compute a soft‑max distribution over the convolution entries.
        # Subtract the current maximum for numerical stability.
        mx = conv_full.max()
        exp_arg = alpha * (conv_full - mx)
        p = np.exp(exp_arg)
        p_sum = p.sum()
        if p_sum == 0:
            # fallback to uniform probability
            p = np.full_like(p, 1.0 / conv_len)
        else:
            p = p / p_sum

        # ----- gradient of the smoothed objective -----
        # grad_i = 2 * sum_j w_j * p_{i+j}
        rev_w = w[::-1]
        fp = np.fft.fft(p, fft_size_grad)
        f_rev = np.fft.fft(rev_w, fft_size_grad)
        grad_full = np.fft.ifft(f_rev * fp).real
        # Extract the n entries that correspond to indices i = 0 … n‑1.
        grad = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # ----- projected gradient step -----
        step = step0 / np.sqrt(it + 1)
        w_new = w - step * grad
        w = project_to_simplex(w_new)

        # ----- keep the best vector seen so far -----
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max   # since sum(w)=1
        if cur_score < best_score:
            best = w.copy()
            best_score = cur_score
            print(f"Iter {it:5d}: new best score {best_score:.6f}")

        # ----- update smoothing parameter -----
        alpha += alpha_inc

        # occasional random jitter to escape possible plateaus
        if it % 500 == 0 and it > 0:
            jitter = rng.normal(scale=0.005, size=n)
            w = project_to_simplex(w + jitter)

        it += 1

    # Scale the probability vector to the range required by evaluate_sequence.
    result = (best * 1000.0).tolist()
    result = [float(max(0.0, min(1000.0, x))) for x in result]
    print(f"Finished after {it} iterations – best score {best_score:.6f}")
    return result

import numpy as np
import time

# ------------------------------------------------------------
# Helper: Projection of a vector onto the probability simplex.
# ------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {w >= 0, sum(w) = 1}.

    Implements the algorithm of Duchi et al. (2008).  The result is a
    probability distribution that is used as the optimisation variable.
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Main entry point required by the challenge.
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Projected gradient descent on a smooth approximation of the
    ``max(convolution)`` objective.

    The algorithm works with a probability vector ``w`` (sum(w)=1) of length
    ``n``.  It minimises

        f(w) = 2 * n * max_k (w * w)_k,

    which is exactly the value returned by ``evaluate_sequence`` after the
    final scaling to the interval [0, 1000].

    A small random perturbation breaks the symmetry of the initial uniform
    distribution; afterwards a smoothed max (soft‑max) guides the descent.
    The iterate is projected onto the simplex after every step.
    The best vector observed during the run is returned (scaled to the
    required range).
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the interrupt
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------
    # Problem size – a larger n gives a lower theoretical bound.
    # --------------------------------------------------------
    n = 2000                     # can be increased if time permits

    # --------------------------------------------------------
    # Initialise with a slightly perturbed uniform distribution.
    # --------------------------------------------------------
    w = np.full(n, 1.0 / n, dtype=np.float64)
    w += rng.normal(scale=1e-4, size=n)          # break perfect symmetry
    w = _project_to_simplex(w)

    # --------------------------------------------------------
    # Pre‑compute FFT lengths (next power of two for speed).
    # --------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len_conv = 1 << (conv_len - 1).bit_length()          # for w*w
    grad_len = 3 * n - 2
    fft_len_grad = 1 << (grad_len - 1).bit_length()          # for grad

    # --------------------------------------------------------
    # Helpers for scoring – with sum(w)=1 the objective reduces to
    #   score = 2 * n * max(conv)
    # --------------------------------------------------------
    def _score_from_conv(conv_full):
        return 2.0 * n * conv_full.max()

    # initial score (uniform distribution) is exactly 2.0
    conv_full = np.real(np.fft.ifft(np.fft.fft(w, fft_len_conv) *
                                   np.fft.fft(w, fft_len_conv)))[:conv_len]
    best_w = w.copy()
    best_score = _score_from_conv(conv_full)

    # --------------------------------------------------------
    # Parameters for the smooth max and the step schedule.
    # --------------------------------------------------------
    alpha = 10.0                 # smoothing factor for soft‑max
    alpha_inc = 0.02             # slowly increase to sharpen focus
    step0 = 0.6                  # base step size

    it = 0
    while time.monotonic() < deadline:
        # ---------- convolution w*w (full) ----------
        fw = np.fft.fft(w, fft_len_conv)
        conv_full = np.fft.ifft(fw * fw).real[:conv_len]

        # ---------- soft‑max over the convolution ----------
        mx = conv_full.max()
        # shift to avoid overflow/underflow
        exp_arg = alpha * (conv_full - mx)
        p = np.exp(exp_arg)
        p_sum = p.sum()
        if p_sum == 0.0:
            p = np.full_like(p, 1.0 / conv_len)
        else:
            p = p / p_sum

        # ---------- gradient of the smoothed objective ----------
        # grad_i = 2 * sum_j w_j * p_{i+j}
        rev_w = w[::-1]
        fp = np.fft.fft(p, fft_len_grad)
        fr = np.fft.fft(rev_w, fft_len_grad)
        grad_full = np.fft.ifft(fr * fp).real
        grad = 2.0 * grad_full[(n - 1):(n - 1 + n)]   # length n

        # ---------- projected gradient step ----------
        step = step0 / np.sqrt(it + 1.0)
        w_candidate = w - step * grad
        w = _project_to_simplex(w_candidate)

        # ---------- keep the best vector ----------
        cur_score = _score_from_conv(conv_full)
        if cur_score < best_score:
            best_score = cur_score
            best_w = w.copy()
            print(f"Iter {it:6d}: new best score {best_score:.6f}")

        # ---------- adapt smoothing ----------
        alpha += alpha_inc

        # ---------- occasional random jitter to escape plateaus ----------
        if it % 500 == 0 and it > 0:
            jitter = rng.normal(scale=0.002, size=n)
            w = _project_to_simplex(w + jitter)

        it += 1

    # --------------------------------------------------------
    # Scale the probability vector to the range required by
    # ``evaluate_sequence`` (clip to the allowed interval).
    # --------------------------------------------------------
    result = (best_w * 1000.0).tolist()
    result = [float(max(0.0, min(1000.0, x))) for x in result]

    print(f"Finished after {it} iterations – best score {best_score:.6f}")
    return result
