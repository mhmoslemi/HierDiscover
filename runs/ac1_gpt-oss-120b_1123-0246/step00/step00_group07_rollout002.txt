"""
Gradient‐based flattening of the autocorrelation.
The algorithm starts from a random feasible vector, iteratively
approximates the max of its self‑convolution with a softmax,
takes a projected gradient step to reduce this max, normalises
the vector and repeats.  When the time budget is almost exhausted
a short block‑perturbation local search is run to polish the
solution.  The best sequence found is returned.
"""
import time
import numpy as np
import scipy.signal

def compute_score(sequence: np.ndarray) -> float:
    """Return the exact evaluate_sequence score for a numpy vector."""
    n = sequence.size
    if n == 0:
        return np.inf
    # Linear convolution via FFT
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(sequence, size)
    conv = np.fft.irfft(fa * fa, size)[:m]
    max_c = conv.max()
    sum_a = sequence.sum()
    if sum_a < 0.01:  # same guard as evaluate_sequence
        return np.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a simple O(n) non‑negative perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

"""
Gradient‑based optimisation of a step function.

The algorithm tries to flatten the self‑convolution of the sequence.
It starts from a (randomised) feasible vector, repeatedly

    1. computes the full convolution a*a,
    2. builds a soft‑max approximation of the ∞‑norm of the convolution,
    3. evaluates the gradient of this smooth objective (a correlation),
    4. takes a projected gradient step, re‑normalising to unit sum.

The temperature of the soft‑max is slowly increased, so the optimisation
focuses more on the true maximum as it progresses.

When the time budget is almost exhausted a short block‑perturbation
local search (the simple heuristic used previously) is run to polish
the best sequence found.  The final sequence is rescaled to a convenient
magnitude (the score is scale‑invariant) and returned as a plain list.
"""

import time
import numpy as np
import scipy.signal


def compute_score(seq: np.ndarray) -> float:
    """Exact score used by ``evaluate_sequence`` (sum‑invariant)."""
    n = seq.size
    if n == 0:
        return np.inf
    # linear convolution via FFT (O(n log n))
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())
    fa = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fa * fa, size)[:m]
    max_c = conv.max()
    total = seq.sum()
    if total < 0.01:                    # same guard as the reference evaluator
        return np.inf
    return float(2.0 * n * max_c / (total ** 2))
