def compute_objective_for_r(r: float, n: int):
    if r <= 0 or r >= 1:
        return float('inf')
    # compute with numpy
    ks = np.arange(n)
    log_r = np.log(r)
    b_vals = (ks + 1) * np.exp(log_r * ks)
    M = b_vals.max()
    # compute sum S = (1 - r**n) / (1 - r)
    r_pow_n = np.exp(log_r * n)
    S = (1 - r_pow_n) / (1 - r)
    f_val = 2.0 * n * M / (S * S)
    return f_val
```

Now golden-section:

```
def golden_section_minimize(func, a, b, tol=1e-12, max_iter=100):
    phi = (1 + np.sqrt(5)) / 2
    resphi = 2 - phi
    c = b - resphi * (b - a)
    d = a + resphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - resphi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + resphi * (b - a)
            fd = func(d)
    # return midpoint
    return (a + b) / 2.0
```

"""
Simple geometric + local refinement search for a low‑evaluation step function.

The evaluation objective is
    2 * n * max(conv(a, a)) / (sum(a)**2)
which is scale‑invariant.  For a fixed length `n` the objective depends only on the
relative shape of the sequence.  A pure exponential shape a_i = r**i admits a
closed‑form approximation: the optimal decay satisfies  e^x = 1 + 2x with
    x = -n·log(r) .
The analytical solution (x≈1.2583) gives a good starting point (score ≈1.8 for
n≈6000).  Starting from that shape we run a cheap block‑wise stochastic
perturbation (the same perturbation used in the baseline) for the remaining
time budget.  Any improvement is kept and printed.

The function `propose_candidate` conforms to the required signature and returns
the best sequence found before the time limit expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: evaluate the objective for a given decay ratio r (0 < r < 1)
# ----------------------------------------------------------------------
def _objective_for_r(r: float, n: int) -> float:
    """Return the evaluation value for the pure exponential a_i = r**i."""
    if not (0.0 < r < 1.0):
        return float('inf')
    # use logs for numerical stability
    log_r = math.log(r)
    ks = np.arange(n, dtype=np.float64)
    # b_k = (k+1) * r**k  for k = 0 .. n-1
    b_vals = (ks + 1.0) * np.exp(log_r * ks)
    M = float(b_vals.max())

    # sum of geometric series
    r_n = math.exp(log_r * n)
    S = (1.0 - r_n) / (1.0 - r)

    # evaluation (scale‑invariant)
    return 2.0 * n * M / (S * S)


# ----------------------------------------------------------------------
# Golden‑section search for the optimal decay r
# ----------------------------------------------------------------------
def _golden_section_minimize(func, a: float, b: float, tol: float = 1e-12,
                             max_iter: int = 80):
    """Minimise `func` on [a, b] using golden‑section (returns the argmin)."""
    phi = (1.0 + math.sqrt(5.0)) / 2.0            # 1.618...
    resphi = 2.0 - phi                           # 0.381966...
    c = b - resphi * (b - a)
    d = a + resphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - resphi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + resphi * (b - a)
            fd = func(d)
    return (a + b) / 2.0


# ----------------------------------------------------------------------
# Compute the analytically‑motivated optimal r (fallback to numeric search)
# ----------------------------------------------------------------------
def _optimal_exponential_ratio(n: int) -> float:
    """Return a decay r that (approximately) minimises the objective for length n."""
    # solve e^x = 1 + 2x  (x = -n*log(r))
    x = 1.5  # initial guess
    for _ in range(10):
        fx = math.exp(x) - 2.0 * x - 1.0
        dfx = math.exp(x) - 2.0
        x -= fx / dfx
    # x now ≈ 1.2583
    r = math.exp(-x / n)
    # ensure we stay inside (0,1)
    r = min(max(r, 1e-8), 0.9999999999)
    # a quick numeric refinement around the analytical value
    lo = max(r * 0.5, 1e-8)
    hi = min(r * 1.5, 0.9999999999)
    best_r = _golden_section_minimize(
        lambda rr: _objective_for_r(rr, n), lo, hi, tol=1e-14, max_iter=60)
    return best_r


# ----------------------------------------------------------------------
# Perturbation used by the baseline (kept unchanged for speed)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return one O(n)-memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width: between 0.5% and ~8% of n
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal noise inside the block
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation – helpful for many shapes
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # clip to the limits enforced by evaluate_sequence
    return np.clip(candidate, 0.0, 1000.0)
