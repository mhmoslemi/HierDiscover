"""Propose a candidate step function using an arcsine‐shape distribution.
The arcsine distribution f(x) ∝ 1/√(x(1‑x)) has the property that its
self‑convolution is (up to a constant factor) uniform on the interval.
Consequently the evaluation metric 2·n·max(conv)/(sum^2) approaches the
theoretical lower bound 2·n/(2n‑1)≈1, which is well below the current
upper bound 2.0.
The function builds a discrete version of that distribution on the same
length as the previously found construction, clips the values to the
allowed range, and optionally performs a few cheap hill‑climbing steps
to improve the score further.
"""

import numpy as np
import time

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    # Determine problem size
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 256

    # Build the discrete arcsine (Beta(0.5,0.5)) shape.
    # x in (0,1) – we offset by 0.5/n to avoid division by zero.
    idx = np.arange(n, dtype=np.float64)
    x = (idx + 0.5) / n
    # clamp just in case
    eps = np.finfo(np.float64).tiny
    x = np.clip(x, eps, 1.0 - eps)
    a = 1.0 / np.sqrt(x * (1.0 - x))

    # Clip to the allowed maximum (1000).  For the chosen n the values are
    # well below 1000, but the clipping keeps us safe if the length changes.
    a = np.clip(a, 0.0, 1000.0)

    # Evaluate the initial candidate (using the same formula as evaluate_sequence)
    conv = np.convolve(a, a)
    S = a.sum()
    best = a
    best_score = float(2 * n * np.max(conv) / (S * S))

    # Time budget for simple hill‑climbing refinements.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    rng = np.random.default_rng(seed)

    while time.monotonic() < deadline:
        # Randomly pick a block and scale it slightly.
        width = int(rng.integers(1, max(2, n // 40)))

            best_score = score

    # Return plain Python list as required.

"""Propose a step‑function sequence based on the arcsine (Beta(0.5,0.5)) density.

The continuous arcsine density  f(x)=1/√(x(1‑x))  has the remarkable property that its
self‑convolution is (up to a constant factor) the uniform distribution on [0,2].
For a discrete version this means the aperiodic autocorrelation (the convolution
used in `evaluate_sequence`) is almost flat, so the quantity

    2·n·max(conv)/(sum(a)²)

approaches the theoretical lower bound 2·n/(2n‑1)≈1.
The routine builds such a shape on the same length as the previously found
construction (`height_sequence_1` if it exists, otherwise 256), clips the
values to the allowed range, and then performs a tiny hill‑climbing loop
inside the given time budget to polish the result.

The returned list is ready for the external `evaluate_sequence` call.
"""

import numpy as np
import time


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Generate a candidate height sequence.

    Parameters
    ----------
    seed : int, optional
        Random seed for the optional local‑search perturbations.
    budget_s : float, optional
        Approximate time budget (seconds).  The function tries to stay a few
        seconds below this limit to give the caller some slack.
    **kwargs
        Ignored – kept for API compatibility.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    # ------------------------------------------------------------------
    # 1. Determine the problem size (number of steps)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)          # typically 4498 in the test harness
    else:
        n = 256                              # fallback size

    # ------------------------------------------------------------------
    # 2. Build a discrete arcsine (Beta(0.5,0.5)) shape.
    #    a[i] ∝ 1 / sqrt( x_i * (1-x_i) ),   x_i = (i+0.5)/n
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    x = (idx + 0.5) / n                     # centre of each bin, 0 < x < 1
    eps = np.finfo(np.float64).tiny
    x = np.clip(x, eps, 1.0 - eps)          # safeguard against 0/1

    # arcsine density (up to a constant factor)
    a = 1.0 / np.sqrt(x * (1.0 - x))

    # Clip to the allowed range (the values are far below 1000 for any n,
    # but clipping guarantees correctness if the length changes dramatically)
    a = np.clip(a, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # 3. Evaluate the initial candidate (using the same formula as
    #    `evaluate_sequence`).  This avoids the overhead of repeatedly calling
    #    the external function.
    # ------------------------------------------------------------------
    conv = np.convolve(a, a)
    total = a.sum()
    best = a
    best_score = float(2.0 * n * np.max(conv) / (total * total))

    # ------------------------------------------------------------------
    # 4. Tiny local‑search refinement (optional).  We stay well within the
    #    allocated budget; the loop terminates automatically when time runs
    #    out.  The perturbation is a small multiplicative change of a random
    #    contiguous block.  Because the score is homogeneous of degree 0,
    #    overall scaling does not matter – we only need to avoid clipping.
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    # Reserve a couple of seconds for safety; the rest is used for refinement.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    while time.monotonic() < deadline:
        # Choose a random block size (between 1 and n/40) and position.
        width = int(rng.integers(1, max(2, n // 40)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width

        # Small multiplicative perturbation (log‑normal step).
        factor = np.exp(rng.normal(0.0, 0.06))

        cand = best.copy()
        cand[start:stop] *= factor

        # Re‑clip to the admissible interval.
        cand = np.clip(cand, 0.0, 1000.0)

        # Fast score computation.
        s = cand.sum()
        if s < 0.01:               # protection against the degenerate case
            continue
        c = np.convolve(cand, cand)
        score = float(2.0 * n * np.max(c) / (s * s))

        # Accept strictly improving moves.
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            # occasional progress report – helpful when debugging.
            print(f"Improved score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found as a plain Python list.
    # ------------------------------------------------------------------
    return [float(v) for v in best]
