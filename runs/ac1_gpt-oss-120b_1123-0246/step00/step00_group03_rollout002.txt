def simplex_projection(v):
    # from paper: Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑score step‑function using three complementary ideas:

1. **Simplex‑projected gradient descent** – 
   Starting from a (possibly sparse) probability vector `p`
   (sum = 1, non‑negative) we minimise the peak
   of its self‑convolution `p∗p`.  The gradient of the
   current maximal entry is cheap to compute (just a reversed
   copy of `p` for the offset(s) where the maximum occurs);
   after a gradient step we project back onto the probability
   simplex.

2. **Random sparse restarts** –  many different sparse
   initialisations are optimised with the same gradient loop.
   The best result seen so far is kept.

3. **LP refinement (optional)** –  given a current sequence `f`,
   we solve a tiny linear programme that maximises the total
   mass of a new sequence `g` under the constraints
   `conv(f,g) ≤ max(conv(f,f))`.  Adding `g` can increase the
   denominator of the evaluation function without raising the
   peak, thus lowering the score.

The algorithm runs inside the supplied time budget, repeatedly
updating the best sequence and finally returning it (scaled by
a constant factor – the objective is homogeneous, so any scale
is fine).  All numbers are forced into the allowed range
[0, 1000] before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: project an arbitrary vector onto the probability simplex
# (sum = 1, each entry ≥ 0).  Uses the O(n log n) algorithm of
# Duchi et al.  Returns a new numpy array.
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x ≥ 0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative -> return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: a cheap block‑perturbation (kept from the reference code)
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return one O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # width of the perturbed block – a fraction of n
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Gradient‑descent routine on the simplex.
def smooth_max_descent(p: np.ndarray,
                       steps: int = 200,
                       init_step: float = 0.2,
                       decay: float = 0.998) -> np.ndarray:
    """
    Minimise the peak of the self‑convolution of p while keeping p on the simplex.
    Parameters
    ----------
    p : np.ndarray
        Probability vector (sum = 1, ≥0).
    steps, init_step, decay : control the gradient descent.
    Returns
    -------
    np.ndarray
        Optimised probability vector (still on the simplex).
    """
    n = p.shape[0]
    step = init_step
    for _ in range(steps):
        conv = np.convolve(p, p)
        max_val = conv.max()
        # indices where the maximum is attained (often just one)
        max_idx = np.flatnonzero(conv >= max_val - 1e-15)
        grad = np.zeros_like(p)
        for k in max_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            grad[i] += 2.0 * p[j]          # derivative of conv[k] w.r.t. p[i]
        p = p - step * grad
        p = simplex_projection(p)
        step *= decay
    return p


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the supplied `evaluate_sequence` objective.

    The search consists of:
      * a quick gradient‑descent refinement of the supplied `height_sequence_1`,
      * many random sparse restarts that are also refined,
      * occasional block‑perturbations of the best candidate,
      * very rare linear‑programming refinements (if cvxpy is available).

    The function respects the time budget (approximately `budget_s` seconds)
    and returns the best sequence found (scaled to a convenient magnitude).
    """
    rng = np.random.default_rng(seed)
    # Reserve a few seconds for finalisation / printing
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Determine a workable problem size.  Use at most 800 entries –
    # large enough to get a good bound, small enough for fast LPs.
    try:
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
        base_arr = np.clip(base_arr, 0.0, 1000.0)
        n = min(base_arr.size, 800)
        base_arr = base_arr[:n]
    except Exception:
        # fallback if the global variable does not exist
        n = 600
        base_arr = np.ones(n, dtype=np.float64)

    if base_arr.sum() == 0.0:
        base_arr = np.ones_like(base_arr)

    # Normalise to a probability vector (sum = 1)
    p_parent = base_arr / base_arr.sum()

    # ------------------------------------------------------------------
    # Keep the best sequence seen so far (as a probability vector)
    best_p = p_parent.copy()
    best_seq = (best_p * 100.0).tolist()          # any scaling factor works
    try:
        best_score = float(evaluate_sequence(best_seq))
    except Exception:
        best_score = float('inf')
    print(f"initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # First refine the parent with the gradient descent routine
    p_refined = smooth_max_descent(p_parent)
    seq = (p_refined * 100.0).tolist()
    sc = float(evaluate_sequence(seq))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p_refined.copy()
        print(f"after gradient descent: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    lp_counter = 0
    while time.monotonic() < deadline:
        # ---- random sparse restart ------------------------------------
        sparsity = int(rng.integers(1, max(2, n // 5)))   # 1 … n/5 non‑zeros
        idx = rng.choice(n, size=sparsity, replace=False)
        p_rand = np.zeros(n, dtype=np.float64)
        p_rand[idx] = rng.random(sparsity) + 0.1          # avoid exact zeros
        p_rand /= p_rand.sum()

        p_opt = smooth_max_descent(p_rand)
        seq = (p_opt * 100.0).tolist()
        sc = float(evaluate_sequence(seq))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p_opt.copy()
            print(f"restart improvement: {best_score:.6f}")

        # ---- occasional block perturbation of the current best -------
        if rng.random() < 0.05:
            cand = make_block_perturbation(best_p * 100.0, rng) / 100.0
            cand = simplex_projection(cand)               # keep it on the simplex
            seq = (cand * 100.0).tolist()
            sc = float(evaluate_sequence(seq))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = cand.copy()
                print(f"block perturb improvement: {best_score:.6f}")

        # ---- very rare LP refinement (only a couple of times) -------
        if rp(lp_counter, rng) and time.monotonic() < deadline:
            # attempt a linear‑programming step
            try:
                import cvxpy as cp
                f = best_p.copy()
                conv_f = np.convolve(f, f)
                M = conv_f.max()

                g = cp.Variable(n)
                constraints = [g >= 0, g <= 1000]

                # build convolution constraints: (f * g)[k] ≤ M  for each k
                for k in range(2 * n - 1):
                    lo = max(0, k - (n - 1))
                    hi = min(k, n - 1)
                    i = np.arange(lo, hi + 1, dtype=int)
                    j = k - i
                    expr = cp.sum(cp.multiply(f[i], g[j]))
                    constraints.append(expr <= M)

                prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
                prob.solve(solver=cp.ECOS, max_iters=2000, verbose=False)

                if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                    g_opt = np.array(g.value).ravel()
                    new_f = np.clip(f + g_opt, 0.0, 1000.0)
                    new_f = simplex_projection(new_f)
                    seq = (new_f * 100.0).tolist()
                    sc = float(evaluate_sequence(seq))
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_p = new_f.copy()
                        print(f"LP refinement improvement: {best_score:.6f}")
                lp_counter += 1
            except Exception as exc:
                # If cvxpy is unavailable or the solve fails, just skip.
                lp_counter += 1
                continue

    # ------------------------------------------------------------------
    # Return the best sequence found (scaled back to a convenient magnitude)
    result = (best_p * 100.0).tolist()
    return result


def rp(counter: int, rng: np.random.Generator) -> bool:
    """
    Decide whether to attempt an LP refinement.
    We allow at most 3 LP solves in total and otherwise run with probability 0.02.
    """
    if counter >= 3:
        return False
    return rng.random() < 0.02
