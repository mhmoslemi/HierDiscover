"""Simple LP‑based refinement for the step‑function attack.

The core idea follows the description in the statement:
for a fixed sequence `f` we solve a linear programme that maximises
the total mass of a new sequence `g` under the *linear* constraints
`(f * g)[k] ≤ M` for every convolution index `k`, where `M` is the
current maximum of `f * f`.  The optimiser therefore finds a direction
that increases the sum of the coefficients without increasing the
worst convolution value.  By moving a small step towards this `g`
the maximum `M` never grows, while the total sum grows, which drives
the evaluation function `2·n·max / sum²` downward.

If `cvxpy` is unavailable the routine falls back to the original
block‑perturbation local search (the “baseline” method).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fallback local search (identical to the reference implementation)
# ----------------------------------------------------------------------
def _fallback_propose(seed: int = 42, budget_s: float = 500.0, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # start from the global variable if it looks sane, otherwise from ones
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:                     # pragma: no cover
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    while time.monotonic() < deadline:
        candidate = _make_block_perturbation(best, rng)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
    return [float(x) for x in best]

def _make_block_perturbation(base, rng):
    """Create an O(n) memory perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main candidate generator – LP‑based refinement
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500.0, **kwargs):
    """Return a sequence of non‑negative heights with a low evaluation score.

    The algorithm works on a modest sized vector (default ``n = 200``).
    It iteratively:

    1. computes the current maximum convolution value ``M`` of the best
       sequence ``f``;
    2. builds a linear programme that maximises ``sum(g)`` subject to
       ``(f * g)[k] ≤ M`` for every index ``k`` (the convolution is linear
       in ``g``) and ``g ≥ 0``;
    3. rescales the solution ``g`` to keep the total sum unchanged;
    4. moves a small step towards ``g`` (convex combination) and, if the
       new sequence improves the score, accepts it.

    If the optional ``cvxpy`` package is unavailable the function falls
    back to a pure‑random block‑perturbation search.
    """
    # Try to import cvxpy – the heavy lifting is done there.
    try:
        import cvxpy as cp   # pylint: disable=import-error
    except Exception:          # pragma: no cover
        return _fallback_propose(seed=seed, budget_s=budget_s, **kwargs)

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n = 200                     # length of the vector we optimise
    step_size = 0.20            # convex‑combination weight (0 < t ≤ 1)
    max_iters = 10_000          # safety cap (real stop is time budget)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # initialise a random positive vector (any scale works – eval is scale‑invariant)
    f = rng.random(n) + 0.1        # ensure we are not too close to zero
    f = np.clip(f, 0.0, 1000.0)

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] score = {best_score:.6f}")

    it = 0
    while time.monotonic() < deadline and it < max_iters:
        # --------------------------------------------------------------
        # 1) current maximum convolution M
        # --------------------------------------------------------------
        conv_f = np.convolve(best, best)
        M = np.max(conv_f)

        # --------------------------------------------------------------
        # 2) build the linear programme  max sum(g)  s.t.  f * g ≤ M
        # --------------------------------------------------------------
        # Coefficient matrix A_{k,i} = f_{k‑i}  (if 0 ≤ k‑i < n, else 0)
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for k in range(rows):
            # indices i where the product f_i * g_{k-i} is defined
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            # corresponding column index of g is j = k - i
            A[k, idx_i] = best[k - idx_i]

        # Variables
        g = cp.Variable(n, nonneg=True)

        # Constraints: A @ g ≤ M   (element‑wise)
        constraints = [A @ g <= M]

        # Objective: maximise sum(g)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception as exc:            # pragma: no cover
            print("[lp] solver failed:", exc)
            break

        if g.value is None:
            print("[lp] no solution returned – terminating.")
            break

        g_vec = np.asarray(g.value).reshape(-1)

        # --------------------------------------------------------------
        # 3) rescale g to keep the total sum identical to the current one
        # --------------------------------------------------------------
        sum_best = best.sum()
        sum_g = g_vec.sum()
        if sum_g == 0.0:                     # very unlikely, but guard against division by zero
            break
        g_vec = g_vec * (sum_best / sum_g)

        # --------------------------------------------------------------
        # 4) convex step towards g and evaluate
        # --------------------------------------------------------------
        candidate = (1.0 - step_size) * best + step_size * g_vec
        candidate = np.clip(candidate, 0.0, 1000.0)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"[iter {it:04d}] improved score = {best_score:.6f}")
        else:
            # occasional random shuffle to escape tiny plateaus
            if rng.random() < 0.02:
                shuffled = best.copy()
                rng.shuffle(shuffled)
                shuffled = np.clip(shuffled, 0.0, 1000.0)
                sc2 = float(evaluate_sequence(shuffled.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best = shuffled
                    best_score = sc2
                    print(f"[iter {it:04d}] improved by shuffle = {best_score:.6f}")

        it += 1

    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    return [float(x) for x in best]
