import numpy as np
import time

def project_simplex(v):
    """Projects a vector onto the probability simplex (non-negative, sum=1)."""
    # v is a 1-D numpy array.
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find the index rho where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero, return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def compute_score(seq):
    """Directly compute the objective (no clipping)."""
    n = seq.shape[0]
    b = np.convolve(seq, seq)
    max_b = np.max(b)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def full_convolution(a):
    """Linear full convolution a*a using FFT (real)."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:m]

def gradient_flatten(a):
    """
    Gradient of sum_{s} (b_s - avg)^2 with respect to a,
    where b = a * a (full convolution) and avg = sum(a)^2 /(2n-1).
    Returns a vector of same size as a.
    """
    n = a.shape[0]
    # full convolution
    b = full_convolution(a)
    sum_a = np.sum(a)
    avg = sum_a * sum_a / (2 * n - 1)

    # Use FFT as well

    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex:
    { w | w_i >= 0 , sum_i w_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0          # cumulative sum minus 1
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear full convolution a * a using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:m]


def gradient_flatten(a: np.ndarray, avg: float, b: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_s (b_s - avg)^2  w.r.t. a.
    a : current sequence (length n)
    avg: target flat value = sum(a)^2/(2n-1)
    b  : full convolution a*a (length 2n-1)
    Returns a vector of length n.
    """
    n = a.shape[0]
    # diff = b - avg (broadcasted)
    diff = b - avg

    # convolution of diff with reversed a (full)
    a_rev = a[::-1]
    m = diff.shape[0]                    # 2n-1
    size = 1 << ((m + n - 2).bit_length())    # enough for linear conv (2n-1)+(n)-1
    D = np.fft.rfft(diff, n=size)
    Ar = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(D * Ar, n=size)

    # indices n-1 … n-1+n-1 give the required n components
    start = n - 1
    grad = 2.0 * conv[start:start + n]   # factor 2 from the derivative
    return grad


def compute_score(a: np.ndarray) -> float:
    """
    Direct evaluation of the objective: 2*n*max(a*a)/(sum(a)^2).
    No clipping is performed – the optimizer works with the raw values.
    """
    n = a.shape[0]
    b = np.convolve(a, a)               # cheap for the modest n we use
    max_b = np.max(b)
    sum_a = np.sum(a)
    if sum_a <= 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a non‑negative step sequence that minimises
        2 * n * max(a*a) / (sum(a)^2)

    The method is a projected gradient descent that tries to flatten the
    self‑convolution (making it as close as possible to a constant).
    Occasional random perturbations and full restarts are used to avoid
    getting trapped in local minima.  The current best sequence is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Initialise a starting vector.
    # Prefer the supplied construction but keep the size modest (fast) .
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

    else:

        a = rng.random(n)

    # Non‑negative and normalised to sum = 1 (scale does not affect the ratio)
    a = np.maximum(a, 0.0)
    if np.sum(a) == 0.0:

    a = project_simplex(a)

    best = a.copy()
    best_score = compute_score(best)
