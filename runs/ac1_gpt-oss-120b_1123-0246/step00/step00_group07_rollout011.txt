def propose_candidate(seed=42, budget_s=500, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # fallback to previous best
    if 'height_sequence_1' in globals():
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)

    # Normalize parent to sum 1 for consistency
    parent = parent / max(np.sum(parent), 1e-12)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f'Init score {best_score:.6f} length {len(best)}')

    # store best per length maybe
    # start local search parameters
    while time.monotonic() < deadline:
        # random choose dimension
        n = int(rng.integers(3, 30))  # up to 30
        # generate random dirichlet
        seq = rng.gamma(1.0, 1.0, size=n)
        seq = seq / seq.sum()
        # evaluate
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best = seq
            best_score = score
            print(f'New best {best_score:.6f} length {n}')
        # occasional local refinement of current best
        if rng.random() < 0.1:
            cand = best.copy()
            # pick two indices i!=j
            i, j = rng.choice(len(cand), size=2, replace=False)
            # transfer fraction
            delta = cand[i] * rng.uniform(0.01, 0.3)
            cand[i] -= delta
            cand[j] += delta
            # optionally random scaling factor to whole vector
            # Not necessary as sum unchanged
            score2 = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best = cand
                best_score = score2
                print(f'Local improve {best_score:.6f} length {len(best)}')
        # also try monotonic decreasing shapes
        if rng.random() < 0.05:
            n = int(rng.integers(3, 30))
            # generate decreasing exponential by sampling random r in (0,1)
            r = rng.uniform(0.5, 0.99)
            seq = np.array([r**i for i in range(n)], dtype=np.float64)
            seq = seq / seq.sum()
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best = seq
                best_score = score
                print(f'Exp shape improve {best_score:.6f} n={n} r={r:.3f}')
        # mixing of two sequences (crossover)
        if rng.random() < 0.03:
            n = len(best)
            # create copy and random perturb
            cand = make_block_perturbation(best, rng)
            # re-normalize? Not needed.
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                print(f'Block perturb improve {best_score:.6f} length {len(best)}')
    # Return best as list of floats
    return [float(x) for x in best]

"""
Propose a low‑evaluation step‑function sequence.

The evaluator is scale‑invariant, i.e. multiplying a sequence by a constant
does not change its score.  Therefore we work directly with probability
vectors (sum = 1) and only need to minimise the maximal coefficient of the
linear convolution a*a.  The search uses three complementary stochastic
operators:

  * Random Dirichlet draws of various lengths (3‑30) – gives a wide
    exploration of the simplex.
  * Exponential families a[i] ∝ r**i with a random decay 0.5‑0.99 – a
    simple parametrisation that often produces low max‑convolution.
  * Local “mass‑transfer” moves that keep the sum = 1:
        choose i ≠ j, move a fraction of a[i] to a[j].

The best candidate is kept globally.  The loop runs until the allotted
budget (≈ budget_s seconds) expires, printing any improvement that is
found.  The final best sequence (as a plain list of floats) is returned.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures) as required.
# ----------------------------------------------------------------------


def _evaluate(seq):
    """Thin wrapper around the provided evaluate_sequence."""
    # The evaluator already performs all security checks.
    return float(evaluate_sequence(seq))


def _random_dirichlet(rng, n):
    """Draw a random point on the n‑simplex and return a normalised ndarray."""
    # Gamma(1,1) == Exp(1) yields a flat Dirichlet.
    vec = rng.gamma(1.0, 1.0, size=n)
    total = vec.sum()
    if total == 0.0:          # should never happen, but be defensive
        vec[:] = 1.0 / n
    else:
        vec /= total
    return vec


def _exponential_family(rng, n):
    """Create a decreasing exponential shape a[i] ∝ r**i, r∈(0.5,0.99)."""
    r = rng.uniform(0.5, 0.99)
    vec = np.array([r ** i for i in range(n)], dtype=np.float64)
    vec /= vec.sum()
    return vec, r


def _local_move(rng, seq):
    """
    Transfer a random fraction of mass from one index to another.
    The operation preserves the total sum (which stays =1).
    """
    cand = seq.copy()
    if len(cand) < 2:
        return cand
    i, j = rng.choice(len(cand), size=2, replace=False)
    # move at most 30 % of the source entry (but never make it negative)
    if cand[i] == 0.0:
        return cand
    delta = cand[i] * rng.uniform(0.01, 0.30)
    cand[i] -= delta
    cand[j] += delta
    return cand


def _make_block_perturbation(base, rng):
    """
    Slightly modify a contiguous block – kept for compatibility with the
    original baseline method.
    """
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point – the function that will be called by the harness.
# ----------------------------------------------------------------------


def propose_candidate(seed=42, budget_s=500, **kwargs):
    """
    Search for a sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random generator seed for reproducibility.
    budget_s : float, optional
        Approximate time budget (seconds) for the search.
    **kwargs : dict
        Ignored – present for compatibility.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum > 0.01).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Initialise the best candidate from the global fallback if it exists.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        # Clip to the allowed range, flatten to 1‑D and normalise.
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0).ravel()
        if parent.ndim == 0 or parent.size == 0 or not np.isfinite(parent).all():
            parent = np.ones(256, dtype=np.float64)
    else:
        parent = np.ones(256, dtype=np.float64)

    # Normalise to a probability vector (the evaluator is scale‑invariant).
    if parent.sum() == 0.0:
        parent[:] = 1.0 / len(parent)
    else:
        parent = parent / parent.sum()

    best_seq = parent.copy()
    best_score = _evaluate(best_seq.tolist())
    print(f"[init] score={best_score:.6f}  length={len(best_seq)}")

    # ------------------------------------------------------------------
    # Main stochastic search loop.
    # ------------------------------------------------------------------
    # We will explore lengths from 3 up to 30 – larger lengths increase
    # the linear factor 2·n, so for a given max‑convolution value the
    # smallest n is generally favourable.
    MIN_N, MAX_N = 3, 30

    # Track the last time we printed a status line (to avoid flooding).
    last_print = time.monotonic()

    while time.monotonic() < deadline:
        # --- Choose a generation strategy at random --------------------
        choice = rng.random()
        if choice < 0.45:  # Dirichlet draw
            n = int(rng.integers(MIN_N, MAX_N + 1))
            cand = _random_dirichlet(rng, n)

        elif choice < 0.70:  # Exponential family
            n = int(rng.integers(MIN_N, MAX_N + 1))
            cand, r = _exponential_family(rng, n)

        elif choice < 0.85:  # Local move on the current best
            cand = _local_move(rng, best_seq)

        else:  # Block perturbation of the current best (kept for diversity)
            cand = _make_block_perturbation(best_seq, rng)

        # ---------------------------------------------------------------
        # Evaluate candidate; the evaluator clamps values to [0,1000] and
        # rejects pathological inputs, returning np.inf.
        # ---------------------------------------------------------------
        try:
            score = _evaluate(cand.tolist())
        except Exception:          # Just in case the evaluator misbehaves
            continue

        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            now = time.monotonic()
            # Emit progress information at most once per second.
            if now - last_print > 1.0:
                print(f"[improve] score={best_score:.6f}  length={len(best_seq)}")
                last_print = now

        # ----------------------------------------------------------------
        # Occasionally try a simple hill‑climb from the *current* best.
        # This is cheap and often yields a modest improvement.
        # ----------------------------------------------------------------
        if rng.random() < 0.12:
            cand2 = _local_move(rng, best_seq)
            try:
                score2 = _evaluate(cand2.tolist())
            except Exception:
                continue
            if np.isfinite(score2) and score2 < best_score:
                best_seq = cand2
                best_score = score2
                now = time.monotonic()
                if now - last_print > 1.0:
                    print(f"[hill‑climb] score={best_score:.6f}  length={len(best_seq)}")
                    last_print = now

        # ---------------------------------------------------------------
        # Early exit if we have already beaten the target (1.503…)
        # ---------------------------------------------------------------
        if best_score < 1.503:
            print(f"[target reached] score={best_score:.6f}")
            break

    # ---------------------------------------------------------------
    # Final result – convert to plain Python floats, as required.
    # ---------------------------------------------------------------
    final = [float(x) for x in best_seq]
    print(f"[final] best_score={best_score:.6f}  length={len(final)}")
    return final
