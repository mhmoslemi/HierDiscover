def gradient_step(arr, rng, step_frac, high_percentile=80, low_percentile=20):
    # arr is a copy we modify, returns new array
    n = arr.shape[0]
    max_val, max_idx = max_convolution_with_index(arr)
    # compute gradient contributions
    idx2 = max_idx - np.arange(n)
    valid = (idx2 >= 0) & (idx2 < n)
    grad = arr.copy()
    # add mirrored values where valid
    grad[valid] += arr[idx2[valid]]
    # Determine high/low indices

def block_perturbation(arr, rng):

    if rng.random() < 0.35:

    minimizes 2*n*max(conv(a,a))/sum(a)^2.
    Uses a simple mass-transfer heuristic based on the contributions
    to the current worst convolution index, together with occasional
    random block perturbations.
    """
    import numpy as np, time, math
    rng = np.random.default_rng(seed)

    # initialise base sequence
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()

    if sum_base < 1e-12:
        base = np.ones_like(base)
        sum_base = base.sum()
    # normalise to unit L1 norm to keep numbers O(1)
    a = base / sum_base

    n = a.shape[0]
    best = a.copy()
    best_score = compute_score(best)

import numpy as np, time

def max_convolution(arr):
    """Return the maximum value of the linear convolution of arr with itself."""
    n = arr.shape[0]
    m_len = 2 * n - 1
    N = 1 << ((m_len).bit_length())  # next power of two
    fft_arr = np.fft.rfft(arr, n=N)
    conv = np.fft.irfft(fft_arr * fft_arr, n=N)
    conv = conv[:m_len]
    return float(conv.max())

def max_convolution_with_index(arr):
    """Return (max_value, index_of_max) of the linear convolution arr*arr."""
    n = arr.shape[0]
    m_len = 2 * n - 1
    N = 1 << ((m_len).bit_length())
    fft_arr = np.fft.rfft(arr, n=N)
    conv = np.fft.irfft(fft_arr * fft_arr, n=N)
    conv = conv[:m_len]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx

def compute_score(arr):
    """Return the evaluation score defined in evaluate_sequence."""
    n = arr.shape[0]
    max_b = max_convolution(arr)
    s = arr.sum()
    if s <= 0.0:
        return float('inf')
    return 2.0 * n * max_b / (s * s)

def gradient_step(arr, rng, step_frac):
    """Perform one mass‑transfer step that reduces the peak convolution."""
    # find the index where the convolution is maximal
    max_val, max_idx = max_convolution_with_index(arr)
    n = arr.shape[0]

    # contribution of each coordinate to the current peak
    idx2 = max_idx - np.arange(n)
    valid = (idx2 >= 0) & (idx2 < n)
    grad = arr.copy()
    grad[valid] += arr[idx2[valid]]

    med = np.median(grad)

"""
Search for a non‑negative step function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

using a simple gradient‑based mass‑transfer heuristic combined with occasional
random block perturbations.  The algorithm works entirely with the L1‑norm
fixed (sum(a) = 1); scaling does not affect the score, so the search space is
bounded and cheap to explore.  It runs for the allotted time budget and
returns the best sequence found.
"""

import time
import numpy as np


def max_convolution(arr: np.ndarray) -> float:
    """Maximum value of the linear convolution arr * arr."""
    n = arr.shape[0]
    m_len = 2 * n - 1
    # next power of two for an efficient FFT
    N = 1 << ((m_len).bit_length())
    fft_arr = np.fft.rfft(arr, n=N)
    conv = np.fft.irfft(fft_arr * fft_arr, n=N)
    conv = conv[:m_len]
    return float(conv.max())


def max_convolution_with_index(arr: np.ndarray):
    """(max value, index of max) of arr * arr."""
    n = arr.shape[0]
    m_len = 2 * n - 1
    N = 1 << ((m_len).bit_length())
    fft_arr = np.fft.rfft(arr, n=N)
    conv = np.fft.irfft(fft_arr * fft_arr, n=N)
    conv = conv[:m_len]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def compute_score(arr: np.ndarray) -> float:
    """Exact score used by evaluate_sequence (without the internal clipping)."""
    n = arr.shape[0]
    max_b = max_convolution(arr)
    s = arr.sum()
    if s <= 0.0:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def gradient_step(arr: np.ndarray, rng: np.random.Generator, step_frac: float):
    """
    Transfer a small amount of mass from indices that contribute most to the
    current worst convolution entry to indices that contribute the least.
    This tends to flatten the convolution peak.
    """
    # locate the current peak of the convolution
    _, max_idx = max_convolution_with_index(arr)
    n = arr.shape[0]

    # gradient of the peak w.r.t. each coordinate:
    idx2 = max_idx - np.arange(n)
    valid = (idx2 >= 0) & (idx2 < n)
    grad = arr.copy()
    grad[valid] += arr[idx2[valid]]

    # split indices into a high‑gradient and a low‑gradient set
    median = np.median(grad)
    high = np.where(grad > median)[0]
    low = np.where(grad < median)[0]

    if high.size == 0 or low.size == 0:
        # degenerate case – just move a random amount from a random i to j
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        while j == i:
            j = int(rng.integers(0, n))
        delta = step_frac * arr[i]
        arr[i] -= delta
        arr[j] += delta
        return arr

    i = int(rng.choice(high))
    j = int(rng.choice(low))

    # move a fraction of the source entry (never negative)
    delta = step_frac * arr[i]
    delta = min(delta, arr[i])
    arr[i] -= delta
    arr[j] += delta
    return arr

            # occasional progress report (every few hundred iterations)
            if it % 500 == 0:
                print(f"Iter {it}: new best score = {best_score:.6f}")

        # early exit if we have already beaten the target gap
        if best_score < 1.51:
            break

    # ------------------------------------------------------------------ #
    # final normalisation (any scaling is equivalent, keep sum = 1)
    final = best.copy()
    s = final.sum()
    if s > 0:
        final *= 1.0 / s

    # return as a plain Python list of floats
    return [float(x) for x in final]
```
