    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s

    # Determine length
    if n is None:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            n = 256
        else:
            n = base.size

    a = np.ones(n, dtype=np.float64) / n   # sum = 1
    # may also randomize a bit to escape
    a = a * np.exp(rng.normal(scale=0.2, size=n))
    a = np.clip(a, 0.0, 1000.0)
    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    print(f"Initial score: {best_score}")

    while time.monotonic() < deadline:
        # compute convolution
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))
        # candidates indices that influence this max
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        # choose source index among this range weighted by a[i]
        idx_range = np.arange(i_start, i_end+1)
        weights = a[idx_range] + 1e-12
        src = rng.choice(idx_range, p=weights/weights.sum())
        # choose a target elsewhere (not src)
        tgt_candidates = np.delete(np.arange(n), src)
        tgt = rng.choice(tgt_candidates)
        # transfer fraction
        epsilon = 0.02  # 2% transfer
        delta = epsilon * a[src]
        if delta > 0:
            a_new = a.copy()
            a_new[src] -= delta
            a_new[tgt] += delta
            # ensure non-negative, clip
            a_new = np.clip(a_new, 0.0, 1000.0)
            score = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(score) and score < best_score:
                a = a_new
                best_a = a_new.copy()
                best_score = score
                print(f"Improved score: {best_score:.6f} at time {time.monotonic():.2f}")
                continue  # continue with improved a
        # if not improved, occasionally random perturb
        if rng.random() < 0.01:
            # apply random block perturb similar to original
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))

"""
Propose a candidate sequence for the step‑function optimisation.

The algorithm starts from a random Dirichlet point (all entries are non‑negative
and sum to 1).  At each iteration it looks at the convolution entry that
achieves the current maximum, selects a pair of indices (i, j) that contribute
to this entry (j = k_max‑i) and moves a small amount of mass from both of
these indices to a random third position.  This reduces the product a_i·a_j
that dominates the maximum while keeping the total sum (and therefore the
scale‑invariant objective) unchanged.  Improvements are kept, otherwise the
step is rejected.  Occasionally a random block perturbation (as in the
reference code) is tried to escape plateaus.  The loop runs until the given
time budget expires and returns the best sequence found.

The evaluation uses the provided `evaluate_sequence` function, but the
objective is also recomputed locally for speed.
"""

import time
import numpy as np

def _evaluate(arr: np.ndarray) -> float:
    """Fast local copy of evaluate_sequence (scale‑invariant)."""
    # Apply the same clipping that evaluate_sequence does.
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(arr, arr)
    return float(2.0 * arr.size * conv.max() / (s * s))


def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score step‑function.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (the caller reserves ~1000 s in total).
    **kwargs
        Ignored – kept for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence found (as plain Python floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Determine the length to work with.  Use the length of the previously
    # found construction if it looks sane, otherwise fall back to 256.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        n = 256

    # ------------------------------------------------------------------
    # Initial sequence – a random Dirichlet point (non‑negative, sum = 1).
    # A tiny random perturbation makes the start non‑uniform.
    # ------------------------------------------------------------------
    a = rng.dirichlet(np.ones(n))
    a *= np.exp(rng.normal(scale=0.1, size=n))   # break perfect symmetry
    a = np.clip(a, 0.0, 1000.0)

    best = a.copy()
    best_score = _evaluate(best)
    print(f"[init] length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----- compute the current maximum convolution entry ----------
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))

        # Indices that participate in this diagonal
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)

        # ----- propose a targeted mass‑transfer move -----------------
        # pick a source index from the active set, weighted by its size
        src_candidates = np.arange(i_low, i_high + 1)
        src_weights = a[src_candidates] + 1e-12
        src = rng.choice(src_candidates, p=src_weights / src_weights.sum())
        tgt = rng.integers(0, n)          # any other index (might coincide)

        # partner index that together with src creates the peak
        partner = k_max - src
        if 0 <= partner < n and partner != src:
            # move a small fraction from both src and partner to tgt
            eps = 0.02                     # 2 % of the smaller entry
            delta = eps * min(a[src], a[partner])
            a_new = a.copy()
            a_new[src] -= delta
            a_new[partner] -= delta
            a_new[tgt] += 2.0 * delta
        else:
            # fallback – move only from src (happens when src==partner)
            eps = 0.02
            delta = eps * a[src]
            a_new = a.copy()
            a_new[src] -= delta
            a_new[tgt] += delta

        # enforce bounds (same as evaluate_sequence)
        a_new = np.clip(a_new, 0.0, 1000.0)

        # ----- evaluate the proposal ---------------------------------
        score = _evaluate(a_new)
        if np.isfinite(score) and score < best_score:
            a = a_new
            best = a_new.copy()
            best_score = score
            print(f"[{it}] improved to {best_score:.6f}")
            continue

        # ----- occasional random block perturbation (exploration) -----
        if rng.random() < 0.01:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            a_block = a.copy()
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            a_block[start:stop] *= factor
            a_block = np.clip(a_block, 0.0, 1000.0)
            score_block = _evaluate(a_block)
            if np.isfinite(score_block) and score_block < best_score:
                a = a_block
                best = a_block.copy()
                best_score = score_block

        if rng.random() < 0.001:
            a = rng.dirichlet(np.ones(n))
            a = np.clip(a, 0.0, 1000.0)
            # keep the best we have so far

    print(f"[final] iterations={it}, best score={best_score:.6f}")
    return [float(v) for v in best]
