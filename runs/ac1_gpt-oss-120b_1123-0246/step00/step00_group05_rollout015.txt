import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

def solve_lp(f, M):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # indices i from i_start to i_end inclusive
        length = i_end - i_start + 1
        row_idx = np.full(length, k)
        col_idx = np.arange(i_start, i_end+1)
        # coefficient is f[i] (since g at index k-i)
        coeffs = f[col_idx]  # but g index is k-i = k - col_idx
        # We'll fill column with g index = k - i
        # So column index = k - col_idx
        col_indices = k - col_idx
        rows.extend(row_idx.tolist())
        cols.extend(col_indices.tolist())
        data.extend(coeffs.tolist())
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    # bounds: (0, None) for each variable.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
    if res.success:
        g = res.x
        return g
    else:
        return None
```

        rows = []
        cols = []

"""
search: Gradient‑descent / LP hybrid for the min‑max convolution problem.

The core idea follows the paper sketch that alternates:
    * evaluate the current maximal self‑convolution M = max(f*f)
    * solve a linear program max Σ g_j   s.t.   (f * g)_k ≤ M   and   g ≥ 0
      (the convolution here is linear, therefore the constraints are linear)
    * rescale g to the same ℓ₁‑norm as f and mix:  f ← (1‑t)·f + t·g

Starting from a mildly decreasing exponential gives a non‑trivial
initial point, the LP then pushes mass towards indices where f is small.
Repeated mixing flattens the convolution and drives the evaluation
function  2·n·max(f*f)/Σf²  below the uniform bound 2.0.
The routine works with a modest problem size (≤ 400 entries) so that the
LP can be solved many times within the time budget.

The function `propose_candidate` returns the best sequence found.
"""

import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """Return a step‑function (list of non‑negative heights) with a low score.

    The algorithm:
        1. Choose a modest length n (taken from the provided `height_sequence_1`
           when possible, otherwise default to 300–400).
        2. Initialise `a` as a gentle exponential decay (non‑uniform so the LP
           can make progress).
        3. Repeatedly:
            * compute M = max(convolution(a, a))
            * build the sparse LP matrix that encodes (a * g)_k ≤ M
            * solve the LP (maximize Σ g) with HiGHS
            * rescale g to have the same ℓ₁‑norm as a
            * update a ← (1‑t)·a + t·g   (t slowly decays)
            * keep the best candidate seen so far.
        4. Stop when the time budget is exhausted or a score below 1.503 is
           obtained.

    The returned list can be fed directly to `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1. Pick a problem size that is small enough for a fast LP.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(min(base.size, 400))          # cap at 400 to keep the LP cheap
    except Exception:
        n = 300                                # fallback size

    # ------------------------------------------------------------------
    # 2. Initial sequence – a gentle exponential decay, ℓ₁‑normalised.
    # ------------------------------------------------------------------
    alpha0 = 2.0 / n                           # small decay (≈2 / n)
    a = np.exp(-alpha0 * np.arange(n, dtype=np.float64))
    a = a / a.sum()                            # normalise; scaling does not matter

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # mixing weight (starts relatively large, then shrinks)
    t = 0.2

    # ---------------------------------------------------------------
    # Main optimisation loop (outer iterations)
    # ---------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- 2a. maximal self‑convolution of the current vector ----------
        conv_aa = np.convolve(a, a)
        M = conv_aa.max()

        # ---- 2b. build the LP matrix for (a * g)_k ≤ M -----------------
        # For each shift k we have Σ_i a_i * g_{k‑i} ≤ M,
        # i runs over indices where both a_i and g_{k‑i} are defined.
        rows, cols, data = [], [], []
        two_n_minus_one = 2 * n - 1
        for k in range(two_n_minus_one):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idx = np.arange(i_start, i_end + 1, dtype=np.int32)
            coeff = a[idx]                     # a_i
            rows.extend([k] * len(idx))
            cols.extend((k - idx).tolist())    # column = g_{k-i}
            data.extend(coeff.tolist())
        A = sp.csr_matrix((data, (rows, cols)), shape=(two_n_minus_one, n))
        b = np.full(two_n_minus_one, M, dtype=np.float64)

        # ---- 2c. solve   max Σ g_j   ⇔   min -Σ g_j --------------------
        c = -np.ones(n, dtype=np.float64)      # objective: -sum(g)
        bounds = [(0, None)] * n                # g_j ≥ 0

        # hi‑gHS (default in SciPy ≥1.6) is fast for this size.
        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True,
                     "time_limit": max(0.0, deadline - time.monotonic())},
        )

        if not res.success:
            # If the solver fails we abort the outer loop – the current best
            # candidate is still valid.
            print("[lp] solver failed, exiting")
            break

        g0 = res.x
        sum_a = a.sum()
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break

        # ---- 2d. rescale g0 to the same ℓ₁‑norm as a
        g = g0 * (sum_a / sum_g0)

        # ---- 2e. convex combination – the core of the alternating scheme
        a = (1.0 - t) * a + t * g
        a = np.clip(a, 0.0, 1000.0)            # respect the evaluator’s clipping

        # ---- 2f. evaluate the new candidate ----------------------------
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            print(f"[improve] new best score={best_score:.6f}")
            if best_score < 1.503:            # early stop if target is reached
                break

        # schedule: slowly reduce the mixing weight so the process settles
        t = max(0.05, t * 0.97)

        # safety: bail out if we are out of time
        if time.monotonic() >= deadline:
            break

    print(f"[final] best score={best_score:.6f}")
    return best.tolist()
