"""
Propose a high‑quality step‑function (non‑negative height sequence).

The objective is to minimise
        2 * n * max(conv(a, a)) / (sum(a)**2)

which is invariant to a global scaling of the sequence.  The algorithm works in
the probability‑simplex (the vector is kept normalised to sum 1) and combines
three ideas:

1. **Projected sub‑gradient descent** – the maximal convolution entry is a
   convex‑piecewise function; its sub‑gradient is known analytically:
   g_i = 2·a_{s‑i} for the index `s` where the maximum occurs.
   A small step opposite to this gradient is taken and the vector is
   projected back onto the simplex.

2. **Random block perturbations** – to escape the symmetric stationary point
   (the uniform distribution), the current vector is occasionally mutated by
   scaling a random contiguous block with a log‑normal factor.  After the
   mutation the vector is renormalised.

3. **Optional symmetrisation** – the optimal sequences are usually (close to)
   palindromic.  With a modest probability the vector is replaced by its
   mirror average, i.e.  (a + a[::-1]) / 2.

The search runs until the allotted time expires (the `budget_s` argument)
and returns the best sequence seen so far.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities (all at top‑level – no closures or lambdas)
# ------------------------------------------------------------

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the probability simplex
    {x | x_i >= 0, sum_i x_i = 1}.  Implements the algorithm of Duchi et al.
    """
    if v.size == 0:
        return v
    # shift by the smallest value to avoid negative artefacts after optimisation
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


def _block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a cheap O(n) perturbation to ``base``:
    - pick a random contiguous block,
    - multiply it by a log‑normal factor,
    - optionally symmetrise,
    - renormalise to sum 1.
    """
    n = base.size
    # choose block width: between n/200 and n/12 (as in the reference code)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    candidate = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))       # log‑normal multiplier
    candidate[start:stop] *= factor
    # occasional mirror averaging (same probability as the reference)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # clip to the limits imposed by ``evaluate_sequence``
    candidate = np.clip(candidate, 0.0, 1000.0)
    # renormalise to the simplex – a global scaling does not change the score
    return _project_onto_simplex(candidate)


def _subgradient_step(
    a: np.ndarray,
    max_idx: int,
    step: float,
) -> np.ndarray:
    """
    Perform one projected sub‑gradient step aimed at reducing the
    convolution entry at ``max_idx``.
    """
    n = a.size
    # Build the sub‑gradient: g_i = 2 * a_{max_idx - i} when the partner index is in range
    lo = max(0, max_idx - (n - 1))
    hi = min(n - 1, max_idx)
    i_vals = np.arange(lo, hi + 1, dtype=int)
    j_vals = max_idx - i_vals
    g = np.zeros_like(a)
    # each i gets contribution 2*a[j]
    g[i_vals] = 2.0 * a[j_vals]

    # take a step opposite to the gradient and re‑project onto the simplex
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    return _project_onto_simplex(a_new)

        return float(2.0 * n * max_b / (vec.sum() ** 2))

    # Keep the best solution seen so far
    best = a.copy()
    best_score = _score(best)
    print(f"Initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    iter_no = 0
    # step‑size schedule for the sub‑gradient updates
    eta_start = 0.5
    eta_min   = 5e-5
    decay_rate = 0.9995

    # after this many iterations without improvement we perform a random restart
    stagnation_counter = 0
    RESTART_AFTER = 20000

    while time.monotonic() < deadline:
        iter_no += 1

        # --------------------------------------------------------------
        # (1) occasional random block mutation – essential to leave the
        #     uniform stationary point.
        # --------------------------------------------------------------
        if rng.random() < 0.05:                 # about 5 % of the iterations
            a = _block_perturbation(a, rng)

        # --------------------------------------------------------------
        # (2) compute convolution and identify the active index
        # --------------------------------------------------------------
        conv = np.convolve(a, a)
        max_val = conv.max()
        max_idx = int(np.flatnonzero(conv == max_val)[0])   # pick one max index

        # --------------------------------------------------------------
        # (3) sub‑gradient descent step aimed at lowering that entry
        # --------------------------------------------------------------
        eta = max(eta_start * (decay_rate ** iter_no), eta_min)
        a = _subgradient_step(a, max_idx, eta)

        # --------------------------------------------------------------
        # (4) optional cheap symmetrisation (helps the shape)
        # --------------------------------------------------------------
        if rng.random() < 0.30:
            a = 0.5 * (a + a[::-1])

        # --------------------------------------------------------------
        # (5) evaluate and possibly record a new best solution
        # --------------------------------------------------------------
        current_score = _score(a)
        if current_score < best_score:
            best = a.copy()
            best_score = current_score
            stagnation_counter = 0
            print(f"Iter {iter_no:6d}: new best score = {best_score:.6f}")
        else:
            stagnation_counter += 1

        # --------------------------------------------------------------
        # (6) random restart if we have been stuck for a while
        # --------------------------------------------------------------
        if stagnation_counter >= RESTART_AFTER:
            # create a fresh random Dirichlet start and continue
            a = rng.dirichlet(np.ones(n))
            a = _project_onto_simplex(a)
            stagnation_counter = 0
            print(f"Iter {iter_no:6d}: restarting from a random point")

    # ------------------------------------------------------------------
    # Post‑processing – the evaluator is scale‑invariant, but we return a
    # reasonably sized vector (sum ≈ sqrt(2·n) as used in the original paper).
    # ------------------------------------------------------------------
    scale = np.sqrt(2.0 * n) / best.sum()
    best_scaled = best * scale
    best_scaled = np.clip(best_scaled, 0.0, 1000.0)

    print(f"Finished. Best score = {best_score:.6f} (iterations: {iter_no})")
    return best_scaled.tolist()
