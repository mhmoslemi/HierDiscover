    import numpy as np, time, cvxpy as cp
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)  # leave 1 sec for final
    best = None
    best_score = np.inf

    # Determine length n: try moderate length like 200
    n = 200  # can adjust based on remaining time, maybe 300 if enough time

    # Build convex optimization
    try:
        x = cp.Variable(n)
        t = cp.Variable()
        constraints = [cp.sum(cp.exp(x)) == 1]
        for k in range(2*n - 1):
            low = max(0, k-(n-1))
            high = min(k, n-1)
            idx = np.arange(low, high+1, dtype=int)
            expr = cp.log_sum_exp(x[idx] + x[k - idx])
            constraints.append(expr <= t)
        prob = cp.Problem(cp.Minimize(t), constraints)
        # Solve with a tolerance and time limit
        solve_kwargs = {"eps": 1e-4, "max_iters": 20000, "verbose": False}
        # Use SCS solver with a time limit if possible: SCS allows 'max_iters' and 'eps'.
        prob.solve(solver=cp.SCS, **solve_kwargs)
        if x.value is not None:
            seq = np.exp(x.value)
            # Clip to constraints
            seq = np.clip(seq, 0.0, 1000.0)
            # Normalize sum (in case of slight deviation)
            s = seq.sum()
            if s > 0:
                seq = seq / s
            score = float(evaluate_sequence(seq.tolist()))
            best = seq
            best_score = score
            print(f"Convex optimization finished: score={score:.6f}, n={n}")
    except Exception as e:
        print("Convex optimization failed:", e)

    # If still time left, perform local search improvements
    # Starting from best if available else from uniform distribution
    if best is None:
        best = np.ones(n) / n
        best_score = float(evaluate_sequence(best.tolist()))
        print(f"Fallback uniform init: score={best_score:.6f}")

    # Local search loop
    while time.monotonic() < deadline:
        # generate candidate by block perturbation similar to previous but with dynamic parameters
        candidate = best.copy()
        # randomly choose block width depending on n
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start_idx = int(rng.integers(0, max(1, n - width + 1)))
        stop_idx = start_idx + width
        # Apply multiplicative noise within block
        factor = np.exp(rng.normal(0.0, 0.05))
        candidate[start_idx:stop_idx] *= factor
        # occasional smoothing: average with neighbor
        if rng.random() < 0.2:
            candidate = (candidate + np.roll(candidate, 1) + np.roll(candidate, -1)) / 3.0
        # Clip and renormalize
        candidate = np.clip(candidate, 0.0, 1000.0)

        if s > 0:

        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score

"""
propose_candidate
=================

A two‑stage optimizer for the step‑function evaluation task.

Stage 1 – Convex “flatten‑the‑autocorrelation’’ problem
-------------------------------------------------------
We minimise the peak of the self‑convolution under a unit‑sum
constraint.  Using the change of variables a_i = exp(x_i) makes the
constraints

    sum_i a_i = 1
    sum_i a_i a_{k-i} ≤ M   for all k

equivalent to the convex constraints

    log_sum_exp( x_i + x_{k-i} ) ≤ log(M).

The optimisation variables are x ∈ ℝⁿ and a scalar t = log(M).
The problem is solved with CVXPY (SCS) for a modest length (n≈200)
which is enough to bring the objective close to its theoretical lower
bound ≈ 1.  The resulting sequence is normalised to sum 1.

Stage 2 – Fast local refinements
--------------------------------
With the remaining time budget we perform cheap block‑perturbations
similar to the original baseline, always re‑normalising the candidate
and keeping any improvement.  This step can further reduce the
evaluation score.

The function respects the time budget, falls back to a uniform
distribution if the convex solve fails and finally returns the best
sequence found as a plain list of floats.
"""

import time
import numpy as np

# CVXPY is optional – the algorithm works without it (fallback to random /
# uniform search).  Import inside the function to avoid import‑time overhead
# in environments where it is unavailable.
    
def propose_candidate(seed=42, budget_s=500, **kwargs):
    """
    Generate a step‑function (list of non‑negative heights) that minimises
    evaluate_sequence().  The routine works inside a time budget (seconds).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # reserve a small safety margin for the final return
    deadline = start_time + max(0.0, float(budget_s) - 0.8)

    # --------------------------------------------------------------------
    # Stage 0 : try to start from the previously discovered construction
    # --------------------------------------------------------------------
    # height_sequence_1 may be present in the global namespace; use it if it
    # looks valid, otherwise fall back to a uniform vector.
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
            raise Exception
    except Exception:
        parent = None

    # --------------------------------------------------------------------
    # Stage 1 : convex optimisation (n moderate, e.g. 200)
    # --------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # try to solve the convex problem; if it fails we simply continue
    try:
        import cvxpy as cp

        n = 200                     # length of the candidate vector
        x = cp.Variable(n)          # log‑amplitudes
        t = cp.Variable()           # log of the maximal convolution entry

        # unit‑sum constraint (in the original amplitude domain)
        constraints = [cp.sum(cp.exp(x)) == 1]

        # peak‑of‑convolution constraints:  log_sum_exp(x_i + x_{k-i}) ≤ t
        # for all k = 0 … 2n‑2
        for k in range(2 * n - 1):
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            idx = np.arange(low, high + 1, dtype=int)
            # the term x[idx] + x[k-idx] is a vector; log_sum_exp works on it
            constraints.append(cp.log_sum_exp(x[idx] + x[k - idx]) <= t)

        # minimise the log‑peak
        problem = cp.Problem(cp.Minimize(t), constraints)

        # Solver parameters – we use SCS (exponential‑cone capable) with a
        # moderate tolerance and iteration limit.  These settings are chosen
        # to finish well within the time budget.
        solve_kwargs = {
            "eps": 5e-5,          # target tolerance on the primal/dual residuals
            "max_iters": 20000,   # generous iteration limit
            "verbose": False,
        }
        problem.solve(solver=cp.SCS, **solve_kwargs)

        if x.value is not None and t.value is not None:
            # Recover the amplitudes: a_i = exp(x_i)
            seq = np.exp(x.value)
            # Clip according to the evaluator’s rules and re‑normalise.
            seq = np.clip(seq, 0.0, 1000.0)
            s = seq.sum()
            if s > 0:
                seq = seq / s
            # Evaluate the candidate.
            score = float(evaluate_sequence(seq.tolist()))
            best_seq = seq
            best_score = score
            print(f"[Convex] n={n}, log‑peak={t.value:.6f}, score={score:.6f}")
    except Exception as exc:   # pragma: no cover – CVXPY missing or solver failure
        print("[Convex] optimisation failed:", exc)

    # --------------------------------------------------------------------
    # Stage 1b : if convex optimisation failed, initialise from a known
    #            sequence (height_sequence_1) or uniform.
    # --------------------------------------------------------------------
    if best_seq is None:

            n = 256

        score = float(evaluate_sequence(cand.tolist()))
        best_seq = cand
        best_score = score

    # --------------------------------------------------------------------

    def block_perturbation(seq, rng):

        w_min = max(1, n // 200)

        factor = float(np.exp(rng.normal(0.0, 0.12)))
        new_seq = seq.copy()
        new_seq[start:stop] *= factor

        return np.clip(new_seq, 0.0, 1000.0)

    while time.monotonic() < deadline:
        cand = block_perturbation(best_seq, rng)
        # Re‑normalise to keep the scale (evaluate_sequence is scale‑invariant,
        # but normalising avoids numerical issues).
        total = cand.sum()
        if total > 0:
            cand = cand / total
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_seq = cand
            best_score = sc
            print(f"[Local] improved score = {sc:.6f}")

    # Final sanity – ensure the sequence complies with the evaluator’s
    # clipping and normalisation rules.
    final = np.clip(best_seq, 0.0, 1000.0)
    total = final.sum()
    if total > 0:
        final = final / total

    return [float(v) for v in final]
