def gradient_descent(a, step):
    conv = np.convolve(a, a)
    k_max = int(np.argmax(conv))
    # Determine overlapping indices
    n = a.shape[0]
    i_start = max(0, k_max - (n - 1))
    i_end = min(k_max, n - 1)
    # For i in [i_start, i_end], gradient = 2*a[k_max - i]
    g = np.zeros_like(a)
    i_range = np.arange(i_start, i_end + 1)
    rev_indices = k_max - i_range
    g[i_range] = 2 * a[rev_indices]
    # Update using exponential decay
    a_new = a * np.exp(-step * g)
    # Renormalize to sum 1
    sum_a = a_new.sum()
    if sum_a <= 0:
        # avoid zero
        a_new = np.ones_like(a) / len(a)
    else:
        a_new = a_new / sum_a
    return a_new

"""
Propose a step sequence using a 1/√i power‑law shape (theoretical minimizer of the
self‑convolution sup‑norm) and refine it with cheap block perturbations.
The initial shape yields an evaluation close to π/2 ≈ 1.57, well below the
previous best of 2.0. The local search tries to improve the score until the
time budget expires.
"""

import numpy as np
import time

def _evaluate_fast(seq: np.ndarray) -> float:
    """Fast computation of evaluate_sequence for non‑negative sequences."""
    n = seq.size
    # convolution of seq with itself (full)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    total = seq.sum()
    # guard against zero division – not needed for our construction
    if total < 1e-12:
        return float('inf')
    return float(2.0 * n * max_b / (total * total))

def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    candidate = base.copy()
    n = candidate.size
    # pick a block width between n/200 and n/12 (at least 1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    # multiplicative log‑normal perturbation
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # occasional symmetrisation (helps against a single large peak)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # clip to the domain accepted by evaluate_sequence
    return np.clip(candidate, 0.0, 1000.0)
