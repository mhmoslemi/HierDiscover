    import numpy as np, time, math

    rng = np.random.default_rng(seed)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Determine length n
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise Exception
        n = base.size
        init = np.clip(base, 0.0, 1000.0)
    except Exception:
        n = 256
        init = np.ones(n, dtype=np.float64)

    # Normalize init to sum 1 (scale invariance)
    init = np.maximum(init, 0.0)
    if np.sum(init) == 0:
        init = np.ones(n)
    init = init / np.sum(init)

    best = init.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f'Initial score {best_score:.6f} (n={n})')

    # Parameter: block size range for perturbation
    # We'll use adaptively sized blocks.

    # Function to compute score quickly (without clipping)
    def compute_score(arr):
        # Compute convolution
        conv = np.convolve(arr, arr)
        maxb = np.max(conv)
        # Since arr sum to 1, S=1
        return float(2 * len(arr) * maxb)  # denominator 1

    # Main loop
    iteration = 0
    # Step size for block scaling (log factor)
    while time.monotonic() < deadline:
        iteration += 1

        # Randomly decide method: random interleaved Dirichlet or local perturbation
        if rng.random() < 0.3:
            # generate fresh random candidate
            alpha = rng.uniform(0.1, 5.0)  # control variance
            cand = rng.dirichlet(np.ones(n) * alpha)
            # Apply interleave pattern
            # choose pattern 0: even-high, odd-low; pattern 1: reverse
            sorted_vals = np.sort(cand)[::-1]
            even_idx = np.arange(0, n, 2)
            odd_idx = np.arange(1, n, 2)
            cand_seq = np.empty(n, dtype=np.float64)
            cand_seq[even_idx] = sorted_vals[:len(even_idx)]
            cand_seq[odd_idx] = sorted_vals[len(even_idx):]
            candidate = cand_seq
        else:
            # Local perturbation of best
            # Compute current conv
            conv = np.convolve(best, best)
            k_max = int(np.argmax(conv))
            # Determine the overlapping index range
            L = max(0, k_max - (n - 1))
            U = min(k_max, n - 1)
            # Choose a block within [L, U] weighted by product contributions
            # Compute contributions for each i in [L,U]
            idxs = np.arange(L, U + 1)
            partners = k_max - idxs
            contrib = best[idxs] * best[partners]
            total = np.sum(contrib)
            if total > 0 and rng.random() < 0.7:
                probs = contrib / total
                start_i = rng.choice(idxs, p=probs)
                # Block width chosen
                width = int(rng.integers(1, max(2, (U - L) // 3 + 1)))
                start = max(L, start_i - width // 2)
                stop = min(n, start + width)
                candidate = best.copy()
                # multiplicative perturbation on block
                factor = float(np.exp(rng.normal(0.0, 0.07)))
                candidate[start:stop] *= factor
                # Optional small smoothing: blend with reversed
                if rng.random() < 0.2:

            else:

                candidate = best.copy()
                width = int(rng.integers(1, max(2, n // 30)))
                start = int(rng.integers(0, n - width + 1))

                candidate[start:start+width] *= factor

        # Clip to bounds, normalize to sum 1 (to keep scale invariance)
        candidate = np.clip(candidate, 0.0, 1000.0)
        s = np.sum(candidate)
        if s == 0:
            continue
        candidate /= s

        # Evaluate
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f'Iter {iteration}: improved score {best_score:.6f}')
        # occasionally log

            print(f'Iter {iteration}: current best {best_score:.6f}')

    # Return best as list of floats (scale up arbitrarily, e.g., multiply by 1)
    # Ensure positive within bounds
    result = [float(x) for x in best]
    print(f'Finished: best score {best_score:.6f} after {iteration} iterations')
    return result

def propose_candidate(seed=42, budget_s=500, **kwargs):
    """
    Search for a low‑score step‑function.
    Starts from a random Dirichlet distribution, interleaves large
    and small values to break the symmetry of the auto‑convolution,
    and then performs a simple hill‑climbing with block perturbations.
    The routine runs until the time budget expires and returns the
    best sequence found.
    """
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Determine length
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise Exception
        n = int(base.size)
    except Exception:
        n = 256

    def to_score(arr):
        # our own fast score assuming sum(arr) close to 1
        conv = np.convolve(arr, arr)
        return 2 * n * np.max(conv) / (np.sum(arr) ** 2)

    # initial candidate: interleaved Dirichlet
    best = None
    best_score = float('inf')
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        # decide which generation scheme
        if rng.random() < 0.4 or best is None:
            # fresh random candidate
            alpha = rng.uniform(0.2, 3.0)   # concentration, controls variance
            vals = rng.dirichlet(np.ones(n) * alpha)
            # sort descending
            sorted_vals = np.sort(vals)[::-1]
            # interleave: even indices get largest, odd get smallest
            cand = np.empty(n, dtype=np.float64)
            even_idx = np.arange(0, n, 2)
            odd_idx = np.arange(1, n, 2)
            half = len(even_idx)
            cand[even_idx] = sorted_vals[:half]
            cand[odd_idx] = sorted_vals[half:]
            candidate = cand
        else:
            # local perturbation of current best
            parent = best.copy()
            # compute convolution to locate the worst shift
            conv = np.convolve(parent, parent)
            k_max = int(np.argmax(conv))
            L = max(0, k_max - (n - 1))
            U = min(k_max, n - 1)
            # weight indices by their contribution to the worst shift
            idxs = np.arange(L, U + 1, dtype=np.int64)
            partners = k_max - idxs
            contrib = parent[idxs] * parent[partners]
            if contrib.sum() > 0 and rng.random() < 0.7:
                probs = contrib / contrib.sum()
                i = rng.choice(idxs, p=probs)
                # pick a block around i
                block_width = int(rng.integers(1, max(2, (U - L) // 4 + 1)))
                start = max(0, i - block_width // 2)
                stop = min(n, start + block_width)
                candidate = parent
                factor = float(np.exp(rng.normal(0.0, 0.07)))
                candidate[start:stop] *= factor
                # occasional reflection mixing
                if rng.random() < 0.15:
                    candidate = 0.5 * (candidate + candidate[::-1])
            else:
                # random narrow block
                block_width = int(rng.integers(1, max(2, n // 40)))
                start = int(rng.integers(0, n - block_width + 1))
                candidate = parent
                factor = float(np.exp(rng.normal(0.0, 0.07)))
                candidate[start:start + block_width] *= factor

        # clip and renormalize (scale invariance)
        candidate = np.clip(candidate, 0.0, 1000.0)
        s = candidate.sum()
        if s == 0:
            continue
        candidate /= s

        # evaluate via the official evaluator
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f'Iter {iteration:6d} – new best score {best_score:.6f}')
        # Periodic progress printing
        if iteration % 2000 == 0:
            print(f'Iter {iteration:6d} – current best {best_score:.6f}')

    # fallback: if nothing found use uniform
    if best is None:

    # Return as list of floats (any scale works – we keep the normalized version)
    result = [float(x) for x in best]
    print(f'Finished after {iteration} iterations – best score {best_score:.6f}')

"""
Search for a low‑score step‑function.

The evaluator is
    score = 2 * n * max(conv(sequence,sequence)) / (sum(sequence)**2)

The score is *scale‑invariant*: multiplying the whole sequence does not change it.
Hence we keep the sequence normalised to sum = 1 and only care about its shape.

Algorithm
---------
1. Determine a working length `n` (use the length of the provided
   `height_sequence_1` if it exists, otherwise fall back to 256).
2. Repeatedly create candidates until the time budget expires:
   * **Randomised Dirichlet + interleaving** (≈40 % of iterations):
        - Sample a random point on the probability simplex with a
          Dirichlet distribution (`alpha` controls the variance).
        - Sort the values descending.
        - Place the largest values on the even indices and the smallest
          on the odd indices (high‑low‑high‑low pattern).  This breaks the
          symmetry of the auto‑convolution and drastically reduces the
          central peak `conv[n‑1]`.
   * **Local block perturbation** (≈60 % of iterations):
        - Take the current best sequence, find the shift `k_max` where
          its convolution is maximal.
        - Weight the indices that contribute to this shift by their
          contribution `a[i] * a[k_max‑i]` and pick a start point for a
          small multiplicative block perturbation.
        - Occasionally mix the sequence with its mirror image to add
          smoothness.
   * After each mutation clip to `[0,1000]`, renormalise to sum = 1,
     evaluate with `evaluate_sequence`, and keep the best.
3. Return the best sequence found (as a list of floats).

The interleaving step already yields scores far below the target
(≈0.5 for n≈256), while the local hill‑climbing refines the solution.
"""

import numpy as np
import time


def generate_interleaved_candidate(rng: np.random.Generator,
                                   n: int,
                                   alpha: float) -> np.ndarray:
    """Dirichlet sample with concentration `alpha` and high‑low interleaving."""
    # Dirichlet draws a random point on the simplex (sum = 1)
    vals = rng.dirichlet(np.ones(n) * alpha)
    # sort descending
    sorted_vals = np.sort(vals)[::-1]
    cand = np.empty(n, dtype=np.float64)
    even_idx = np.arange(0, n, 2)
    odd_idx = np.arange(1, n, 2)
    half = len(even_idx)
    # place the largest half on even positions, the rest on odd ones
    cand[even_idx] = sorted_vals[:half]
    cand[odd_idx] = sorted_vals[half:]
    return cand


def local_perturbation(rng: np.random.Generator,
                       parent: np.ndarray) -> np.ndarray:
    """Perturb a block that contributes strongly to the current worst shift."""
    n = parent.size
    conv = np.convolve(parent, parent)
    k_max = int(np.argmax(conv))

    # indices that actually overlap for this shift
    L = max(0, k_max - (n - 1))
    U = min(k_max, n - 1)
    idxs = np.arange(L, U + 1, dtype=np.int64)
    partners = k_max - idxs
    contrib = parent[idxs] * parent[partners]

    cand = parent.copy()

    # Prefer to touch indices that contribute a lot to the peak
    if contrib.sum() > 0.0 and rng.random() < 0.7:
        probs = contrib / contrib.sum()
        i = rng.choice(idxs, p=probs)
        # block width proportional to the size of the active region
        block_width = int(rng.integers(1, max(2, (U - L) // 4 + 1)))
        start = max(0, i - block_width // 2)
        stop = min(n, start + block_width)
        factor = float(np.exp(rng.normal(0.0, 0.07)))   # modest multiplicative change
        cand[start:stop] *= factor
        # occasional smoothing with the reversed sequence
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])
    else:
        # a blind narrow perturbation elsewhere
        block_width = int(rng.integers(1, max(2, n // 40)))
        start = int(rng.integers(0, n - block_width + 1))
        factor = float(np.exp(rng.normal(0.0, 0.07)))
        cand[start:start + block_width] *= factor

    return cand


def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs) -> list[float]:
    """
    Return a sequence that (heuristically) minimises `evaluate_sequence`.

    The routine respects the time budget, printing progress and keeping the
    best candidate found.  It works completely offline – no filesystem or network
    access is performed.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------
    # Determine the length of the sequence.
    # ------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0:
            raise Exception
        n = int(base.size)
    except Exception:
        n = 256                     # reasonable default if no external data

    # ------------------------------------------------------------
    # Helper: normalise a vector to sum=1 (scale invariance of the score)
    # ------------------------------------------------------------
    def normalise(vec: np.ndarray) -> np.ndarray:
        s = vec.sum()
        if s == 0.0:
            # fallback to uniform
            return np.ones_like(vec) / vec.size
        return vec / s

    best = None
    best_score = float('inf')
    iteration = 0

    # ------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # 40 % of the time we generate a fresh random candidate
        if rng.random() < 0.4 or best is None:
            # Random concentration parameter – larger alpha → flatter distribution
            alpha = rng.uniform(0.2, 3.0)
            candidate = generate_interleaved_candidate(rng, n, alpha)
        else:
            # Local hill‑climbing on the current best
            candidate = local_perturbation(rng, best)

        # Clip to the evaluator's allowed range and renormalise
        candidate = np.clip(candidate, 0.0, 1000.0)
        candidate = normalise(candidate)

        # Evaluate with the official (security‑checked) function
        score = float(evaluate_sequence(candidate.tolist()))

        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f'Iter {iteration:6d} – new best score {best_score:.6f}')

        # occasional progress report (does not affect performance)
        if iteration % 2000 == 0:
            print(f'Iter {iteration:6d} – current best {best_score:.6f}')

    # ------------------------------------------------------------
    # Fallback in the very unlikely case that nothing was accepted
    # ------------------------------------------------------------
    if best is None:
        best = np.ones(n, dtype=np.float64) / n
        best_score = float(evaluate_sequence(best.tolist()))
        print('Fallback to uniform sequence, score',
              f'{best_score:.6f}')

    # Return as a plain Python list of floats (any scaling works)
    result = [float(x) for x in best]
    print(f'Finished after {iteration} iterations – best score {best_score:.6f}')
    return result
