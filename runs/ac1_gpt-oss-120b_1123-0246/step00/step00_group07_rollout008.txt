"""
Search for a low‑score step function.

The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where *conv* is the discrete (non‑reversed) convolution.
A uniform sequence gives a score of exactly 2.  To obtain a lower
score we need a distribution whose self‑convolution is flatter than
the triangular shape coming from the uniform case.

The algorithm works in two phases:

1. **Parameterized initialisation** –  we generate families of
   exponentially decaying sequences
        a_i = exp(-α i / n) * (i+1)^{-p}
   for a small grid of the parameters α (which controls the decay)
   and p (a mild power‑law correction).  Because the score is
   scale‑invariant we do not need to renormalise the vectors.
   Every candidate is evaluated and the best one is kept.

2. **Local stochastic refinement** – starting from the best candidate
   we perform a fast memory‑bounded local search that repeatedly
   perturbs a random contiguous block (multiplicative log‑normal
   noise) and optionally symmetrises the result.  Whenever a
   perturbation improves the score it becomes the new current best.

Both phases are time‑limited by the *budget_s* argument (default 500 s).
The function returns the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """
    Return one O(n)‑memory non‑negative perturbation of ``base``.
    A random block is multiplied by exp(N(0, 0.12)).
    With probability 0.35 the candidate is symmetrised.
    """
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width between n/200 and n/12  (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetrisation (helps to keep the convolution shape smooth)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=500, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    # reserve a few seconds for finalisation/printing
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 1) Choose a working length n.
    #    Use the supplied construction if it exists, otherwise fall back.
    try:
        height_sequence_1  # noqa: F821 – may be defined globally
        base_seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = base_seq.size
        if n < 500:                # ensure we have enough degrees of freedom
            n = 2000
    except Exception:
        n = 2000
    # We work on a fresh array of length n (the exact length does not
    # have to match the original one – the evaluator only cares about the
    # values, not about any external identifier).
    n = int(n)

    # ------------------------------------------------------------------
    # 2) Grid‑search over simple parametric families.
    best = np.ones(n, dtype=np.float64)          # uniform baseline (score = 2)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f'baseline (uniform) score = {best_score:.6f}')

    # parameter grids
    alphas = np.linspace(0.5, 1.5, 11)   # α controls exponential decay (α≈1 is good)
    powers = [0.0, 0.5, 1.0, 1.5]        # p controls an extra power‑law tail

    idx = np.arange(n, dtype=np.float64)

    for alpha in alphas:
        beta = alpha / n                         # exponential factor per index
        exp_part = np.exp(-beta * idx)           # common exponential component
        for p in powers:
            if p == 0.0:
                seq = exp_part
            else:
                seq = exp_part * np.power(idx + 1.0, -p)
            # clipping is cheap and guarantees compliance with the evaluator
            seq = np.clip(seq, 0.0, 1000.0)
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best = seq
                best_score = score
                print(f'grid improvement: score={best_score:.6f}  α={alpha:.3f}  p={p}')

    # ------------------------------------------------------------------
    # 3) Stochastic local refinement (block perturbations).
    #    This is essentially the same scheme used in the reference code
    #    but starts from a much better initial point.
    while time.monotonic() < deadline:
        cand = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f'local improvement: score={best_score:.6f}')
        # occasional aggressive restart: re‑initialise from the best grid point
        if rng.random() < 0.02:
            # tiny random re‑scale of the whole vector – keeps scale invariance
            factor = float(np.exp(rng.normal(0.0, 0.05)))
            best = np.clip(best * factor, 0.0, 1000.0)

    print(f'final best score = {best_score:.6f}')
    # Return a plain python list as required by the judge.
    return [float(x) for x in best]
