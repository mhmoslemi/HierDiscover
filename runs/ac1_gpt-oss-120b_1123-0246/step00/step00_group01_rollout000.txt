"""
Simple stochastic‑annealing search for a step‑function with a small
evaluate_sequence() score.

The algorithm works with a fixed length `n` (taken from the global
`height_sequence_1` if it exists, otherwise a default of 256).  A
starting distribution is built from the (discrete) arcsine density,
which already makes the self‑convolution flatter than the uniform
distribution.  The search runs a simulated‑annealing loop that
performs several cheap random moves:

*   multiplicative scaling of a random contiguous block,
*   transfer of a random amount of mass between two points,
*   a targeted move that reduces the contribution to the current
    maximal convolution entry,
*   a small smoothing step (moving‑average).

All moves keep the vector non‑negative and are re‑normalised to unit
sum (the score is scale‑invariant).  The Metropolis acceptance rule
with a linear temperature schedule allows occasional uphill moves.
Whenever a new best score is found it is printed.  The function returns
the best sequence seen before the time budget expires.
"""

import time
import numpy as np

# Helper utilities ------------------------------------------------------

def _initial_arcsine(n: int) -> np.ndarray:
    """
    Return a discretised arcsine density on n points, normalised to unit sum.
    The continuous arcsine PDF on [0,1] is 1/(π√(x(1‑x))) – it puts a lot of
    weight near the ends which helps flatten the self‑convolution.
    """
    i = np.arange(n, dtype=np.float64)
    # centre points of bins
    x = (i + 0.5) / n
    # avoid exact 0 or 1 to keep values finite
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, np.inf)          # all finite already
    a /= a.sum()
    return a


def _clipped_normalise(v: np.ndarray) -> np.ndarray:
    """Clip to allowed range [0,1000] and renormalise to unit sum."""
    v = np.clip(v, 0.0, 1000.0)
    s = v.sum()
    if s < 1e-12:          # should not happen, but guard against div‑by‑0
        v = np.ones_like(v) * (1.0 / v.size)
    else:
        v = v / s
    return v


def _move_block_scale(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    n = seq.size
    # width between n/200 and n/12 (inclusive, at least 1)
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.08)))   # modest perturbation
    cand = seq.copy()
    cand[start:stop] *= factor
    return cand


def _move_pair_transfer(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from i to j."""
    n = seq.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        return seq
    amount = seq[i] * rng.uniform(0.0, 0.15)   # up‑to 15 % of source
    cand = seq.copy()
    cand[i] -= amount
    cand[j] += amount
    return cand


def _move_targeted_reduce(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Reduce the contribution to the currently maximal convolution entry.
    We locate the index k with maximal (seq*seq)[k] and pick the pair
    (i, j) with i+j=k that has the biggest product a_i·a_j.
    A small amount of mass is taken from both i and j and redistributed
    uniformly to all other positions.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    idx = np.arange(i_min, i_max + 1, dtype=np.int64)
    prod = seq[idx] * seq[k - idx]
    # Choose the pair with the largest product (most responsible for the peak)
    best = int(idx[np.argmax(prod)])
    partner = k - best
    if best == partner:
        # Zeros in the diagonal are less harmful – fall back to a simple pair move
        return _move_pair_transfer(seq, rng)

    # Amount to take: a fraction of the smaller of the two entries
    fraction = 0.12
    extract = min(seq[best], seq[partner]) * fraction
    if extract <= 0.0:
        return seq
    cand = seq.copy()
    cand[best] -= extract
    cand[partner] -= extract

    # Redistribute the removed mass uniformly among the remaining entries
    # (excluding the two we just lowered).  This keeps the sum constant.
    mask = np.ones(n, dtype=bool)
    mask[best] = False
    mask[partner] = False
    redistribute = extract * 2.0
    cand[mask] += redistribute / mask.sum()
    return cand


def _move_smoothing(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a tiny moving‑average smoothing (kernel length 5)."""
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1], dtype=np.float64)
    smooth = np.convolve(seq, kernel, mode='same')
    return smooth


def _propose_new(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Randomly select one of the move types and apply it.
    """
    move_type = rng.random()
    if move_type < 0.30:
        cand = _move_block_scale(seq, rng)
    elif move_type < 0.60:
        cand = _move_pair_transfer(seq, rng)
    elif move_type < 0.85:
        cand = _move_targeted_reduce(seq, rng)
    else:
        cand = _move_smoothing(seq, rng)
    return cand

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a sequence of non‑negative heights that (hopefully) gives a
    low value of evaluate_sequence().  The search uses simulated
    annealing with several inexpensive perturbations and runs until
    the allocated time budget expires.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function will try to stop a few
        seconds before the limit to be safe).

    Returns
    -------
    list[float]
        The best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length and initialise a starting point
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        n = 256

    # a few diverse start candidates – the best among them becomes the initial state
    starts = []

    # 1) arcsine‑shaped start (generally good)
    starts.append(_initial_arcsine(n))

    # 2) uniform start (baseline)
    starts.append(np.ones(n, dtype=np.float64) / n)

    # 3) random Dirichlet start (adds randomness)
    random_dir = rng.random(n)
    starts.append(random_dir / random_dir.sum())

    # 4) optional reuse of the previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size == n and np.isfinite(prev).all():
            prev = _clipped_normalise(prev)
            starts.append(prev)
    except Exception:
        pass

    # Evaluate the starters and keep the best
    best_seq = None
    best_score = np.inf
    for cand in starts:
        cand = _clipped_normalise(cand)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = cand.copy()
    if best_seq is None:          # should not happen, but defensive
        best_seq = np.ones(n, dtype=np.float64) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # Simulated annealing loop
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # leave a safety margin
    total_budget = deadline - start_time

    # SA schedule – linear cooling from temp0 to temp_end
    temp0 = 1.0
    temp_end = 0.002

    current = best_seq.copy()
    current_score = best_score

    iteration = 0
    # Main optimisation loop
    while time.monotonic() < deadline:
        iteration += 1
        # linear temperature according to elapsed fraction
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_budget
        temperature = max(temp_end, temp0 * (1.0 - frac))

        # propose a new candidate
        cand = _propose_new(current, rng)
        cand = _clipped_normalise(cand)

        # Evaluate
        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
            # Discard invalid candidates (should be rare)
            continue

        # Record global best
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{elapsed:6.2f}s] New best score: {best_score:.6f}")

        # Metropolis acceptance
        delta = cand_score - current_score
        if delta < 0.0 or rng.random() < np.exp(-delta / max(temperature, 1e-12)):
            current = cand
            current_score = cand_score

        # Occasionally inject a fresh random start to escape deep local minima
        if rng.random() < 0.001:
            fresh = rng.random(n)
            fresh = _clipped_normalise(fresh)
            fresh_score = float(evaluate_sequence(fresh.tolist()))
            if fresh_score < current_score:
                current = fresh
                current_score = fresh_score
                if fresh_score < best_score:
                    best_score = fresh_score
                    best_seq = fresh.copy()
                    print(f"[{elapsed:6.2f}s] New best via restart: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"Search finished after {iteration} iterations, best score = {best_score:.6f}")
    # Convert to plain Python list of floats as required
    return [float(x) for x in best_seq]
