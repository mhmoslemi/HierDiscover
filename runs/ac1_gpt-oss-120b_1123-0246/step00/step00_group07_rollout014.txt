def gradient_step(seq, eta):
    n = len(seq)
    # compute convolution via FFT
    nfft = 1 << (2*n -1).bit_length()
    fa = np.fft.rfft(seq, nfft)
    conv = np.fft.irfft(fa * fa, nfft)
    conv = conv[:2*n-1]
    k_max = np.argmax(conv)
    max_b = conv[k_max]
    sum_a = seq.sum()
    factor = max_b / sum_a
    # compute rev = zeros length n
    rev = np.zeros_like(seq)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n-1, k_max)
    # vectorized
    idx = np.arange(i_start, i_end+1)
    rev[idx] = seq[k_max - idx]
    # gradient step
    seq_new = seq - eta * (rev - factor * seq)
    # ensure nonnegative and clip
    np.clip(seq_new, 0.0, 1000.0, out=seq_new)
    return seq_new, float(2 * n * max_b / (sum_a**2))

    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            n = base.size
        else:
            raise Exception
    except Exception:
        n = 5000   # fallback

    # initial sequence: geometric
    target_x = 1.0   # maybe adjust
    r = 1 - target_x / n
    if r <= 0:
        r = 0.5
    seq = np.exp(np.log(r) * np.arange(n, dtype=np.float64))
    seq *= 1000.0   # scale to full range
    seq = np.clip(seq, 0.0, 1000.0)

    # Precompute FFT size
    nfft = 1 << ((2*n - 1).bit_length())

    # function to compute score and max conv
    def compute_score(arr):
        s = arr.sum()
        if s < 0.01:
            return np.inf, None
        fa = np.fft.rfft(arr, nfft)
        conv = np.fft.irfft(fa * fa, nfft)
        maxb = float(conv[:2*n-1].max())
        score = 2.0 * n * maxb / (s * s)
        return score, maxb

    best_seq = seq.copy()
    best_score, best_maxb = compute_score(best_seq)
    # also we may compute conv for best to reuse
    # main loop
    eta = 0.1
    iter_count = 0

    while time.monotonic() < deadline:
        # choose operation
        op = rng.integers(0, 3)
        cand = best_seq.copy()
        if op == 0:
            # gradient descent step
            # compute conv once
            fa = np.fft.rfft(cand, nfft)
            conv = np.fft.irfft(fa * fa, nfft)[:2*n-1]
            k_max = int(np.argmax(conv))
            maxb = float(conv[k_max])
            s = cand.sum()
            factor = maxb / s if s != 0 else 0.0
            # compute rev vector
            rev = np.zeros_like(cand)
            i_start = max(0, k_max - (n - 1))
            i_end = min(n-1, k_max)
            idx = np.arange(i_start, i_end+1)
            rev[idx] = cand[k_max - idx]
            # gradient step
            cand = cand - eta * (rev - factor * cand)
            np.clip(cand, 0.0, 1000.0, out=cand)
        elif op == 1:
            # block perturbation
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = start + width
            factor = np.exp(rng.normal(0.0, 0.12))
            cand[start:stop] *= factor
            if rng.random() < 0.35:
                cand = 0.5 * (cand + cand[::-1])
            np.clip(cand, 0.0, 1000.0, out=cand)
        else:
            # random shuffle of a subsegment
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]  # swap
        # evaluate
        score, maxb = compute_score(cand)
        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            best_maxb = maxb
            print(f"[{time.monotonic():.2f}] New best {best_score:.6f}")
            # maybe increase step size if improvement
        iter_count += 1

    # return best sequence as list of floats
    return [float(x) for x in best_seq]

def propose_candidate(seed=42, budget_s=500, **kwargs):
    import numpy as np, time, math, sys
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Choose length n
    base_seq = globals().get('height_sequence_1', None)
    if isinstance(base_seq, (list, np.ndarray)):
        arr0 = np.asarray(base_seq, dtype=np.float64)
        if arr0.ndim == 1 and arr0.size > 0 and np.isfinite(arr0).all():
            n = min(int(arr0.size), 8000)   # keep it moderate
        else:
            n = 6000
    else:
        n = 6000

    # geometric seed
    # Target x ≈ 1.2 (optimal for geometric) => 1-r = x/n
    x_opt = 1.2
    r = max(0.0, 1.0 - x_opt / n)
    # generate decreasing geometric values, max at 1.0 then scale to 1000
    seq = np.exp(np.log(r) * np.arange(n, dtype=np.float64))
    seq = seq * 1000.0   # bring maximum to 1000
    seq = np.clip(seq, 0.0, 1000.0)

    # random shuffle of the multiset to break structure
    rng.shuffle(seq)

    # FFT size (next power of two for speed)
    nfft = 1 << ((2 * n - 1).bit_length())

    # helper to compute score using FFT
    def score_of(a):
        s = a.sum()
        if s < 0.01:
            return np.inf, None
        fa = np.fft.rfft(a, nfft)
        conv = np.fft.irfft(fa * fa, nfft)[:2 * n - 1]
        maxb = float(conv.max())
        sc = 2.0 * n * maxb / (s * s)
        return sc, maxb

    best_seq = seq.copy()
    best_score, best_maxb = score_of(best_seq)

    # Simple hill‑climbing: swaps and block perturbations
    # Adjust step sizes adaptively
    temperature = 1.0
    decay = 0.9995

    while time.monotonic() < deadline:
        cand = best_seq.copy()
        op = rng.integers(0, 3)
        if op == 0:
            # swap two random positions (keeps distribution the same)
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif op == 1:
            # perturb a contiguous block multiplicatively
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = np.exp(rng.normal(0.0, 0.07))
            cand[start:start + width] *= factor
        else:
            # small random redistribution: take amount from a random index and give to another
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and cand[i] > 0.0:
                delta = cand[i] * rng.uniform(0.05, 0.15)
                cand[i] -= delta
                cand[j] = min(1000.0, cand[j] + delta)

        np.clip(cand, 0.0, 1000.0, out=cand)

        sc, _ = score_of(cand)

        # accept if improvement or with a small probability (simulated annealing)
        accept = False
        if sc < best_score:
            accept = True
        else:
            # occasional uphill moves to escape local minima
            prob = math.exp(-(sc - best_score) / temperature) if temperature > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            best_seq = cand
            best_score = sc
            # simple cooling schedule
            temperature *= decay
            # optional progress output
            print(f"{time.monotonic():.2f}  best score {best_score:.6f}")

    return [float(v) for v in best_seq]

"""
Propose a low‑score step function.

The algorithm builds a geometric (exponentially decaying) seed, shuffles it to
break the regular pattern and then performs a fast hill‑climbing search.
At each iteration a small random modification (swap, block scaling or a
tiny redistribution) is tried; the convolution is evaluated with an FFT
so the cost per iteration is O(n log n).  Improvements are kept and the best
sequence found before the time budget expires is returned.
"""

import math
import time
import numpy as np
import scipy  # used only to silence linters; not needed at runtime


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """Search for a sequence with a low evaluation score.

    The function runs until ``budget_s`` seconds (minus a small safety margin)
    have elapsed and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    # ----------------------------------------------------------------------
    # 1. Choose a working length n
    # ----------------------------------------------------------------------
    # Prefer the length of the supplied construction if it exists.
    base_seq = globals().get("height_sequence_1", None)
    if isinstance(base_seq, (list, np.ndarray)):
        arr0 = np.asarray(base_seq, dtype=np.float64)
        if arr0.ndim == 1 and arr0.size > 0 and np.isfinite(arr0).all():
            n = min(int(arr0.size), 8000)          # keep it moderate
        else:
            n = 6000
    else:
        n = 6000

    # ----------------------------------------------------------------------
    # 2. Initialise a geometric seed (a_i = r^i) and shuffle it
    # ----------------------------------------------------------------------
    # For a pure geometric shape the optimal asymptotic value of
    # x = n·(1‑r) is about 1.2 – 1.3.  We use x≈1.2.
    x_opt = 1.2
    r = max(0.0, 1.0 - x_opt / n)                  # r ≈ 0.9998 for n≈6000
    seq = np.exp(np.log(r) * np.arange(n, dtype=np.float64))
    seq *= 1000.0                                  # scale – max element hits the limit
    seq = np.clip(seq, 0.0, 1000.0)
    rng.shuffle(seq)                               # break the regular order

    # ----------------------------------------------------------------------
    # 3. Helper that returns the evaluation score (using FFT)
    # ----------------------------------------------------------------------
    nfft = 1 << ((2 * n - 1).bit_length())        # next power‑of‑two for speed

    def evaluate_fast(a):
        """Return (score, max_conv) for a non‑negative array a."""
        total = a.sum()
        if total < 0.01:
            return np.inf, None
        fa = np.fft.rfft(a, nfft)
        conv = np.fft.irfft(fa * fa, nfft)[: 2 * n - 1]
        maxb = float(conv.max())
        score = 2.0 * n * maxb / (total * total)
        return score, maxb

    best_seq = seq.copy()
    best_score, best_maxb = evaluate_fast(best_seq)

    # ----------------------------------------------------------------------
    # 4. Simple hill‑climbing / simulated‑annealing loop
    # ----------------------------------------------------------------------
    temperature = 1.0                # controls occasional uphill moves
    decay = 0.9995                    # cooling schedule

    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    while time.monotonic() < deadline:
        cand = best_seq.copy()
        # three cheap neighbourhood moves
        op = rng.integers(0, 3)
        if op == 0:
            # swap two random entries (preserves the multiset)
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        elif op == 1:
            # multiplicatively perturb a random contiguous block
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = np.exp(rng.normal(0.0, 0.07))
            cand[start:start + width] *= factor
        else:
            # move a small fraction from one index to another
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and cand[i] > 0.0:
                delta = cand[i] * rng.uniform(0.05, 0.15)
                cand[i] -= delta
                cand[j] = min(1000.0, cand[j] + delta)

        # enforce the hard limits required by `evaluate_sequence`
        np.clip(cand, 0.0, 1000.0, out=cand)

        sc, _ = evaluate_fast(cand)

        # Accept if strictly better, otherwise maybe accept uphill move
        accept = False
        if sc < best_score:
            accept = True
        else:
            # Metropolis criterion – allows occasional worsening moves
            prob = math.exp(-(sc - best_score) / temperature) if temperature > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            best_seq = cand
            best_score = sc
            temperature *= decay          # cool down slowly
            # occasional progress report (stdout)
            print(f"{time.monotonic():.2f}  best score {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 5. Return the best sequence as a plain Python list of floats
    # ----------------------------------------------------------------------
    return [float(v) for v in best_seq]
