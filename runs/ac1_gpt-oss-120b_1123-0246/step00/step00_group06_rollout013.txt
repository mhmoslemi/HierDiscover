def propose_candidate(seed=42, budget_s=500, **kwargs):
    """...
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Determine length using existing global if present
    try:
        base_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if base_arr.ndim != 1 or base_arr.size == 0:

        n = int(base_arr.size)

        n = 2000   # fallback length
    # baseline constant sequence
    const_val = np.sqrt(2.0 * n) / n
    best_seq = np.full(n, const_val, dtype=np.float64)
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f'Initial constant sequence score: {best_score:.6f}')
    # Parameters
    max_u_options = [2, 3]  # allowed unordered pair multiplicity
    # main loop
    while time.monotonic() < deadline:
        max_u = rng.choice(max_u_options)
        # Build a set S respecting unordered pair bound max_u
        sum_counts = np.zeros(2 * n - 1, dtype=np.int16)
        S = []
        # Random order of indices each attempt
        order = rng.permutation(n)
        for idx in order:
            # fast early reject: self sum
            s_self = idx + idx
            if sum_counts[s_self] + 1 > max_u:
                continue
            # check all existing sums
            ok = True
            for j in S:
                if sum_counts[idx + j] + 1 > max_u:
                    ok = False
                    break
            if not ok:
                continue
            # accept idx
            S.append(idx)
            sum_counts[s_self] += 1
            for j in S[:-1]:
                sum_counts[idx + j] += 1
        m = len(S)
        if m < 2:
            continue
        # Build sequence: 1 on positions in S, else 0
        seq = np.zeros(n, dtype=np.float64)
        seq[S] = 1.0
        # Scale to target sum sqrt(2n)
        target_sum = np.sqrt(2.0 * n)
        scale = target_sum / m
        seq *= scale
        # Evaluate
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f'New best score {best_score:.6f} with m={m}, max_u={max_u}')
            # If already below target break early
            if best_score <= 1.503:
                break
    # Return best as list of floats
    return [float(x) for x in best_seq]

"""
Title:  Fast construction of low‑max‑convolution step functions.

Summary:
The evaluation score is 2 n·max(b)/sum(a)² where
b = a * a (linear convolution).  For a non‑negative sequence the
quantity max(b) is exactly the largest value among the sums
∑ a_i·a_j with i+j = k.  We therefore seek a support set of indices
in which each possible sum occurs only a few times.  If each unordered
pair (i,j) appears at most *u* times then max(b) ≤ (2·u+1)·c² for a
constant height c on the support, and the score becomes

    score = 2·n·(2·u+1) / m²           (m = |support|).

Consequently a large support with a small bound *u* gives a low score.
The program repeatedly builds random supports obeying a bound
on the number of unordered pairs per sum (a “restricted Sidon” set)
using a simple greedy algorithm, rescales the heights so that the
sum equals √(2 n) (the scaling does not change the score), evaluates
the sequence and keeps the best one found before the time budget
expires.

The implementation is pure NumPy, runs in O(n·m) per attempt,
and can explore many billions of candidates within the allowed time.
"""

import time
import numpy as np


def build_bounded_set(n: int, max_u: int, rng: np.random.Generator) -> list[int]:
    """
    Greedy construction of a set S ⊂ {0,…,n‑1} such that each unordered
    pair {i,j} (including i=j) contributes at most `max_u` times to the
    same sum i+j.  The function returns the list of chosen indices.
    """
    # count of unordered pairs that currently realise each sum
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)
    S = []

    # random order gives many different candidate sets
    order = rng.permutation(n)
    for idx in order:
        # self‑pair i+i
        s_self = idx + idx
        if sum_counts[s_self] + 1 > max_u:
            continue

        ok = True
        # test all sums with already chosen elements
        for j in S:
            if sum_counts[idx + j] + 1 > max_u:
                ok = False
                break
        if not ok:
            continue

        # accept idx: update counters
        S.append(idx)
        sum_counts[s_self] += 1
        for j in S[:-1]:          # all previous elements
            sum_counts[idx + j] += 1

    return S


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimizes the
    provided `evaluate_sequence` function.
    The algorithm builds many low‑collision supports (restricted Sidon
    sets) and keeps the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    # reserve a few seconds for finalisation and printing
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # -----------------------------------------------------------------
    # Determine the length of the searched sequence.
    # Prefer the length of the supplied global `height_sequence_1`.
    # If it is unavailable fall back to a modest default.
    # -----------------------------------------------------------------
    try:

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    const_val = np.sqrt(2.0 * n) / n

    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] constant sequence score = {best_score:.6f}")

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    # We will try several values of the pair‑multiplicity bound.
    max_u_options = [2, 3, 4]   # larger `u` permits larger supports

    while time.monotonic() < deadline:
        max_u = int(rng.choice(max_u_options))

        # Build a random support set obeying the bound `max_u`
        support = build_bounded_set(n, max_u, rng)
        m = len(support)
        if m < 2:                 # degenerate set – ignore
            continue

        # Create the candidate sequence: equal heights on the support
        seq = np.zeros(n, dtype=np.float64)
        seq[support] = 1.0

        # Rescale so that the sum equals √(2 n); the score is scale‑invariant
        target_sum = np.sqrt(2.0 * n)
        scale = target_sum / m
        seq *= scale

        # Safety clipping (required by the evaluator)
        np.clip(seq, 0.0, 1000.0, out=seq)

        # Evaluate the candidate
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:

            best_seq = seq.copy()

            # early exit if we have already reached the target
            if best_score <= 1.503:
                break

    # -----------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # -----------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} (length = {n})")
    return [float(x) for x in best_seq]
