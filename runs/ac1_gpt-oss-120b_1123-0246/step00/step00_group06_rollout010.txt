```python
def fast_score(seq):
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    n = len(seq)
    m = 1 << ((2*n-1).bit_length())  # next power of 2
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft*fft, n=m)[:2*n-1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s*s))
```

def propose_candidate(seed=42, budget_s=500, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    # Determine length
    n = 1024
    # init candidate from height_sequence_1 if present and length matches
    if 'height_sequence_1' in globals():
        h1 = np.asarray(height_sequence_1, dtype=np.float64)
        if h1.size >= n:
            init = h1[:n].copy()
            init = np.clip(init, 0.0, 1000.0)
        else:
            init = np.concatenate([h1, np.zeros(n - h1.size)])
    else:
        init = rng.random(n)
    # normalize sum to 1
    init = np.clip(init, 0.0, 1000.0)
    if init.sum() == 0:
        init = np.ones(n)
    init = init / init.sum()
    # Also ensure min value maybe small epsilon to allow moves
    curr = init.copy()
    best = curr.copy()
    best_score = fast_score(best)
    curr_score = best_score
    start = time.monotonic()
    iteration = 0
    # precompute target avg maybe
    while time.monotonic() < deadline:
        # Temperature decreasing linearly
        t_frac = max(0.0, (deadline - time.monotonic()) / (deadline - start))
        T = max(1e-6, t_frac)  # scaling factor linear
        # Propose move
        cand = curr.copy()
        if rng.random() < 0.6:
            # transfer move
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            if cand[i] > 0:
                delta = cand[i] * rng.random() * 0.5
                cand[i] -= delta
                cand[j] += delta
        else:
            # block scaling
            width = int(rng.integers(1, max(2, n // 10)))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.2))
            block = cand[start_idx:start_idx+width] * factor
            block = np.clip(block, 0.0, 1000.0)
            cand[start_idx:start_idx+width] = block
        # renormalize to keep sum = 1 (optional)
        s = cand.sum()
        if s == 0:
            continue
        cand = cand / s
        cand_score = fast_score(cand)
        # accept
        if cand_score < curr_score or rng.random() < math.exp(-(cand_score - curr_score) / T):
            curr = cand
            curr_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"Iter {iteration}, new best score {best_score:.6f}")
        iteration += 1
        # occasional random restart to avoid stuck
        if iteration % 5000 == 0 and rng.random() < 0.1:
            # new random start but keep best so far
            cand = rng.random(n)
            cand = np.clip(cand, 0.0, 1000.0)
            cand = cand / cand.sum()
            cand_score = fast_score(cand)
            curr = cand
            curr_score = cand_score
    # final scaling to original magnitude (not needed)
    final_seq = best.tolist()
    # optionally compute exact score for logging
    try:
        exact = evaluate_sequence(final_seq)
        print(f"Final exact score: {exact:.6f}")
    except Exception:
        pass
    return final_seq

def fast_score(seq):
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    n = len(seq)
    m = 1 << ((2*n-1).bit_length())
    fft = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft*fft, m)[:2*n-1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s*s))

"""
Propose a candidate step‐function using simulated annealing.

The algorithm works on a fixed length sequence (default 2000).  It
initialises the sequence from the supplied global ``height_sequence_1``
(if present) or from a random simplex, and then performs many fast
iterations (using FFT‐based convolution) to minimise

    2 * n * max(conv(a, a)) / sum(a)^2

Only non‑negative values are allowed; all candidates are normalised to
unit sum because the evaluation metric is scale‑invariant.  Three kinds
of moves are employed:

* mass transfer between two random positions,
* multiplicative scaling of a random block (followed by normalisation),
* a *targeted* move that reduces the entries that contribute most to
  the current maximal convolution entry.

Moves are accepted if they improve the objective, otherwise with a
probability that decreases linearly with the remaining time (simulated
annealing).  The best sequence found before the time budget expires is
returned as a plain list of floats.
"""

import numpy as np
import time
import math

def _fft_convolve(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of ``seq`` with itself using FFT."""
    n = seq.size
    # length of the linear convolution is 2*n-1; next power of two for FFT
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, m)
    conv = np.fft.irfft(f * f, m)[: 2 * n - 1]
    return conv


def _fast_score(seq: np.ndarray) -> float:
    """Fast evaluation of the objective (same as evaluate_sequence)."""
    total = seq.sum()
    if total < 0.01:
        return float("inf")
    n = seq.size
    conv = _fft_convolve(seq)
    max_b = conv.max()
    return float(2.0 * n * max_b / (total * total))


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a good coefficient sequence within the given time budget.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds; the function returns before this time.
    **kwargs : dict
        Ignored – present for compatibility with the driver.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, arbitrary length).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # 1. Choose a working length.  2000 gives a good trade‑off between
    #    speed (FFT size ≈ 8192) and theoretical lower bound 2*n/(2*n-1)
    #    ≈ 1.0005, well below the target 1.5.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # 2. Initialise the sequence.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.size >= n:
            seq = base[:n].copy()
        else:
            seq = np.concatenate([base, np.zeros(n - base.size)])
    else:
        seq = rng.random(n)

    seq = np.clip(seq, 0.0, 1000.0)
    if seq.sum() == 0.0:
        seq = np.ones(n, dtype=np.float64)
    seq = seq / seq.sum()                     # normalise – scale does not matter

    # current state and best state
    current = seq.copy()
    best = current.copy()
    best_score = _fast_score(best)
    current_score = best_score

    start_time = time.monotonic()
    iteration = 0
    # exponential moving average of iteration durations (for diagnostics)
    avg_iter_time = 0.0

    # ------------------------------------------------------------------
    # 3. Main simulated‑annealing loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # linear temperature schedule: high at start, near zero at the end
        remaining = max(0.0, deadline - time.monotonic())
        frac = remaining / (deadline - start_time)
        temperature = max(1e-8, frac)          # small positive temperature

        # ------------------------------------------------------------------
        # 4. Choose a move type
        # ------------------------------------------------------------------
        move_type = rng.choice(["transfer", "block", "targeted"], p=[0.45, 0.15, 0.40])

        cand = current.copy()

        if move_type == "transfer":
            # move a random fraction of mass from i to j
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if cand[i] > 0.0:
                delta = cand[i] * rng.random() * 0.5
                cand[i] -= delta
                cand[j] += delta

        elif move_type == "block":
            # multiply a random contiguous block by a log‑normal factor
            width = int(rng.integers(2, max(2, n // 12)))
            start = int(rng.integers(0, n - width + 1))
            factor = float(np.exp(rng.normal(0.0, 0.18)))
            cand[start : start + width] *= factor
            cand = np.clip(cand, 0.0, 1000.0)

        else:  # targeted move – attacks the current maximal convolution entry
            # compute the full convolution once (still cheap with FFT)
            conv = _fft_convolve(cand)
            s_max = int(conv.argmax())
            # indices that contribute to this maximal entry
            lo = max(0, s_max - (n - 1))
            hi = min(n - 1, s_max)
            idx_range = np.arange(lo, hi + 1, dtype=np.int64)
            partner = s_max - idx_range
            contrib = cand[idx_range] * cand[partner]
            # pick the pair with the largest contribution
            best_pair = idx_range[contrib.argmax()]
            # try to reduce the larger of the two partners
            i = best_pair
            if cand[i] > 0.0:
                delta = cand[i] * rng.random() * 0.3
                # choose a receiver far from i (to avoid creating a new large peak)
                j = (i + rng.integers(1, n)) % n
                cand[i] -= delta
                cand[j] += delta

        # ------------------------------------------------------------------
        # 5. Normalise and evaluate the candidate
        # ------------------------------------------------------------------
        total = cand.sum()
        if total == 0.0:
            continue  # avoid degenerate zero vector
        cand /= total
        cand_score = _fast_score(cand)

        # ------------------------------------------------------------------
        # 6. Accept / reject (Metropolis criterion)
        # ------------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # probability to accept a worsening move
            prob = math.exp(-(cand_score - current_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(
                    f"Iter {iteration:07d} – new best score {best_score:.6f} "
                    f" (temp={temperature:.4g})"
                )

        # occasional random restart to avoid long stagnation
        if iteration % 7000 == 0 and rng.random() < 0.07:
            cand = rng.random(n)
            cand = np.clip(cand, 0.0, 1000.0)
            cand /= cand.sum()
            current = cand
            current_score = _fast_score(current)

        iteration += 1

    # ------------------------------------------------------------------
    # 7. Return the best sequence (as list of Python floats)
    # ------------------------------------------------------------------
    result = best.tolist()
    # final exact evaluation for logging (if evaluate_sequence is available)
    try:
        exact = evaluate_sequence(result)
        print(f"Final exact evaluation score: {exact:.6f}")
    except Exception:
        pass
    return result
