def convolve_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=size)
    C = A * A
    conv = np.fft.irfft(C, n=size)
    return conv[:2*n-1]
```

def simplex_projection(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, n+1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def gradient_descent(a, rng, deadline):
    best_local = a
    best_score = eval(a)
    step = 0.1
    smoothing = 8.0
    while time.monotonic() < deadline:
        # compute b and gradient
        b = conv_fft(a, a)
        # compute approximate weights
        b_max = b.max()
        exps = np.exp(smoothing * (b - b_max))
        w = exps / exps.sum()
        # gradient
        conv = conv_fft(a, w[::-1])
        idx_slice = len(w) - 1 - np.arange(len(a))
        grad = 2.0 * conv[idx_slice]
        # update
        a_candidate = a - step * grad
        # projection onto simplex
        a_candidate = simplex_projection(a_candidate)
        # evaluate
        score = evaluate_sequence(a_candidate.tolist())
        if np.isfinite(score) and score < best_score:
            a = a_candidate
            best_score = score
            best_local = a_candidate
            # adjust step positive
            step = min(step * 1.1, 1.0)
            # optionally reduce smoothing to focus on max later
            smoothing = max(2.0, smoothing * 0.99)
        else:
            # reject
            step = max(step * 0.5, 1e-6)
            # maybe random perturb

                a = a_tmp

def convolve_fft(x, y):
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    Z = X * Y
    conv = np.fft.irfft(Z, n=size)
    return conv[:n]
```

def simplex_projection(v):
    """Project vector v onto the probability simplex (nonnegative, sum=1)."""
    n = v.shape[0]
    # shift values
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / np.arange(1, n+1) > 0)[0]
    # if rho empty, return uniform?
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step sequence with a lower evaluation score.

The search uses a projected (simplex) gradient descent on a smooth
approximation of the objective
    J(a) = 2 * n * max_k (a * a)[k]   (the sum of a is normalised to 1)
where “*” denotes the discrete convolution.
The max is replaced by a soft‑max (controlled by `smoothing`) so that a
gradient can be computed.  The gradient of the soft‑max is
    ∇J ≈ 2 * conv(a, w_rev)[len(w)-1 - i]  (i = 0,…,n‑1)
with w being the soft‑max weights.  After a gradient step the vector is
projected onto the probability simplex (non‑negative, sum = 1).  Several
random restarts, occasional block perturbations and an adaptive step
size are employed to escape local minima.  The whole procedure runs
until the allotted time expires and returns the best sequence encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns a vector of length len(a)+len(b)-1.
    """
    n = len(a) + len(b) - 1
    size = _next_pow2(n)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)
    return conv[:n]


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects an arbitrary vector onto the probability simplex:
        {x | x_i ≥ 0, sum_i x_i = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u + (1.0 - cssv) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _evaluation_from_max(n: int, max_b: float) -> float:
    """
    The external `evaluate_sequence` is scale‑invariant.
    With our internal normalisation sum(a)=1 we have
        eval = 2 * n * max_b .
    """
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a non‑negative step sequence that minimises the provided
    `evaluate_sequence`.  The algorithm runs for up to `budget_s` seconds
    (minus a small safety margin) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0:
            n = base.size
        else:
            n = 2048
    except Exception:
        n = 2048

    # ------------------------------------------------------------------
    # Global best across all restarts
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    restart_counter = 0

    # ------------------------------------------------------------------
    # Main loop: many independent stochastic restarts
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # allocate a modest time slice for this restart
        remaining = deadline - time.monotonic()
        # at most 12 s per restart, or half the remaining time, whichever is smaller
        sub_deadline = time.monotonic() + min(12.0, remaining / 2.0)

        # ---- random initial point (exponential distribution) -----------------
        a = rng.exponential(scale=1.0, size=n)
        a = _simplex_projection(a)                     # sum == 1, non‑negative

        # ---- evaluate the start point (fast internal estimate) ---------------
        b = _convolve_fft(a, a)
        cur_score = _evaluation_from_max(n, b.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[restart {restart_counter}] initial score {best_score:.6f}")

        # ---- parameters for the inner gradient descent -----------------------
        step = 0.5                     # initial step length
        smoothing = 8.0                # initial soft‑max temperature

        # ---- inner optimisation loop ----------------------------------------
        while time.monotonic() < sub_deadline:
            # compute convolution a*a and soft‑max weights
            b = _convolve_fft(a, a)
            b_max = b.max()
            # soft‑max (stable)
            exp_vals = np.exp(smoothing * (b - b_max))
            w = exp_vals / exp_vals.sum()               # shape (2*n-1,)

            # gradient of the smoothed max
            conv = _convolve_fft(a, w[::-1])            # length 3*n-2
            idx = len(w) - 1 - np.arange(n)            # indexes needed for each component
            grad = 2.0 * conv[idx]

            # gradient step + projection onto the simplex
            a_candidate = a - step * grad
            a_candidate = _simplex_projection(a_candidate)

            # fast internal evaluation (no call to the external scorer)
            b_cand = _convolve_fft(a_candidate, a_candidate)
            cand_score = _evaluation_from_max(n, b_cand.max())

            # accept if improvement
            if cand_score < cur_score:
                a = a_candidate
                cur_score = cand_score
                # update global best if necessary
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[restart {restart_counter}] new global best {best_score:.6f}")

                # adapt the step size / smoothing (be more aggressive)
                step = min(step * 1.08, 2.0)
                smoothing = max(2.0, smoothing * 0.995)
                continue

            # ---- rejection path ------------------------------------------------
            # shrink step to become more cautious
            step = max(step * 0.55, 1e-6)

            # occasional random block perturbation to escape plateaus
            if rng.random() < 0.01:
                width = rng.integers(max(1, n // 200), max(2, n // 12))
                start = rng.integers(0, max(1, n - width + 1))
                factor = float(np.exp(rng.normal(0.0, 0.12)))
                a_pert = a.copy()
                a_pert[start:start + width] *= factor
                a_pert = np.clip(a_pert, 0.0, 1000.0)
                a_pert = _simplex_projection(a_pert)
                a = a_pert
                # recompute its score (cheap)
                b = _convolve_fft(a, a)
                cur_score = _evaluation_from_max(n, b.max())

        restart_counter += 1

    # ------------------------------------------------------------------
    # Fallback – should never happen but keeps the contract safe
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _evaluation_from_max(n, (1.0 / n) ** 2 * n)  # = 2
        print("Fallback to uniform sequence.", best_score)

    print("Final best score:", best_score)
    # Return a plain Python list of floats (the external grader expects this)
    return [float(x) for x in best_seq]
