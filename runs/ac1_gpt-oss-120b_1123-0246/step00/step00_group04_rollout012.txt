def exponential_score(n, alpha):
    r = np.exp(-alpha / n)
    # sum of geometric series
    if abs(r - 1) < 1e-12:
        sum_seq = n
    else:
        sum_seq = (1 - r**n) / (1 - r)
    # compute max product L_k * r**k; we find max over k=0..2n-2
    ks = np.arange(2*n-1)
    L = np.minimum(ks+1, 2*n-1 - ks)
    # compute values: L * r**ks
    # For numerical stability compute r**ks as exp(ks * log(r)), but r<1.
    vals = L * np.exp(ks * np.log(r))
    max_val = np.max(vals)
    ratio = 2 * n * max_val / (sum_seq**2)
    return ratio

"""
search_exponential.py

A fast search for low‑score step functions.

The evaluation function is
    score = 2 * n * max(conv(a,a)) / (sum(a))**2   (a_i >= 0)

For a monotone exponential profile
    a_i = exp(-γ i) ,   i = 0 … n‑1
the convolution at shift k simplifies to

    (a*a)_k = L_k * r**k    with r = exp(-γ)  and
    L_k = min(k+1, 2n-1-k)  (the number of overlapping terms).

Both the sum of the sequence and the maximal convolution value can be
computed analytically, so the score can be evaluated in O(n) without a
full convolution.  For each length n we perform a tiny golden‑section
search on the single parameter α = γ·n (i.e. r = exp(-α/n)).  This finds
the exponential that makes the product L_k·r**k as flat as possible and
typically drives the score well below the uniform‑sequence bound of 2.0.

After the cheap parametric sweep a short stochastic hill‑climbing phase
perturbs the best sequence (multiplicative Gaussian noise) to see if a
slightly non‑exponential shape can improve the score further.

The function ``propose_candidate`` respects the time budget, prints a
few progress messages and finally returns the best sequence found as a
list of floats.  If nothing succeeds it falls back to a uniform vector.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: analytic score for a pure exponential vector
# ----------------------------------------------------------------------
def exponential_score(n: int, alpha: float) -> float:
    """
    Return the value 2*n*max(b)/sum(a)**2 for the vector
        a_i = exp(-alpha*i/n)   (i = 0 … n-1).

    The result is computed analytically, avoiding the O(n log n) convolution.
    """
    # r = exp(-alpha/n)  < 1
    r = np.exp(-alpha / n)

    # geometric series sum  S = sum_i r**i
    if abs(r - 1.0) < 1e-12:
        S = float(n)
    else:
        S = (1.0 - r**n) / (1.0 - r)

    # vector of L_k (overlap counts) for k = 0 … 2n-2
    ks = np.arange(2 * n - 1, dtype=np.float64)
    L = np.minimum(ks + 1.0, 2.0 * n - 1.0 - ks)

    # values of L_k * r**k  (max of the convolution up to a constant factor)
    # use log to avoid under‑flow for large k
    log_vals = np.log(L) + ks * np.log(r)
    max_val = np.exp(np.max(log_vals))

    # final score (scale‑invariant)
    return 2.0 * n * max_val / (S * S)


# ----------------------------------------------------------------------
# Helper: build the exponential vector (no scaling needed – scale‑invariant)
# ----------------------------------------------------------------------
def make_exponential_vector(n: int, alpha: float) -> np.ndarray:
    """
    Return a length‑n non‑negative vector a_i = exp(-alpha*i/n).
    The vector is already clipped to the [0,1000] range required by the
    evaluator (values are <= 1 here).
    """
    gamma = alpha / n
    return np.exp(-gamma * np.arange(n, dtype=np.float64))


# ----------------------------------------------------------------------
# The required entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a low‑score step function within the allotted time.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget in seconds (the function will stop a few
        seconds before this limit to guarantee a return).

    Returns
    -------
    list[float]
        The best sequence discovered (as plain Python floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1) Parametric sweep over exponential families
    # ------------------------------------------------------------------
    # A modest set of lengths – larger lengths give a finer “continuous”
    # approximation but cost more time.  3809 (the current length) is kept.
    length_candidates = [256, 512, 1024, 2048, 3809, 4096, 5000]

    for n in length_candidates:
        # Golden‑section (well‑known 3‑point) search on α ∈ (0.01, 10)
        lo, hi = 0.01, 10.0
        for _ in range(30):            # ~30 iterations give ~1e‑9 relative precision
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            s1 = exponential_score(n, m1)
            s2 = exponential_score(n, m2)
            if s1 < s2:
                hi = m2
            else:
                lo = m1

        alpha_opt = (lo + hi) / 2.0
        candidate = make_exponential_vector(n, alpha_opt)

        # evaluate with the official routine (scale‑invariant, but we request the real score)
        try:
            score = float(evaluate_sequence(candidate.tolist()))
        except Exception:                     # defensive – should never happen
            score = float('inf')

        print(f"len={n:5d}  α≈{alpha_opt:7.4f}  score={score:.6f}")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = candidate.copy()

        if time.monotonic() > deadline:
            break

    # ------------------------------------------------------------------
    # 2) Local stochastic refinement (hill‑climbing)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.size
        while time.monotonic() < deadline:
            # multiplicative Gaussian perturbation (keeps shape non‑negative)
            noise = rng.normal(loc=0.0, scale=0.015, size=n)   # about 1‑2 % jitter
            cand = best_seq * np.exp(noise)                   # ensures positivity
            # optional re‑normalisation – not required for the score, but keeps numbers stable
            cand = cand / np.max(cand) * np.max(best_seq)

            try:
                cand_score = float(evaluate_sequence(cand.tolist()))
            except Exception:
                cand_score = float('inf')

            if np.isfinite(cand_score) and cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"    improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Fallback to a simple uniform vector if everything failed
    # ------------------------------------------------------------------
    if best_seq is None or best_score == float('inf'):
        print("Fallback: using uniform vector")
        best_seq = np.ones(256, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"final best score = {best_score:.6f}  (length {best_seq.size})")
    return [float(x) for x in best_seq.tolist()]
