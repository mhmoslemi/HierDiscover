"""
Propose a better step‑function using fast Fourier based evaluation,
randomised local moves and a simple gradient‑descent style step.

The algorithm:
1. Load the existing construction (``height_sequence_1``) and normalise it
   to unit sum (the objective is scale invariant).
2. Initialise a few random Dirichlet‑like candidates and keep the best.
3. Repeatedly modify the current candidate until the time budget expires.
   Move types include:
   • scaling a random contiguous block,
   • swapping two entries,
   • shuffling a short segment,
   • averaging with its reversal (symmetrisation),
   • a “gradient step’’ that reduces the contribution to the currently
     dominant convolution entry by moving mass from a high‑product index
     to a low‑product one.
4. All candidates are re‑normalised to keep the sum ≈ 1, clipped to the
   allowed range and evaluated with an FFT‑based implementation of the
   original ``evaluate_sequence`` (much faster than ``np.convolve``).
5. Whenever a new best score is found it is printed.  After the loop the
   best candidate is checked with the official ``evaluate_sequence`` and
   returned as a plain list of floats.

The method is cheap (FFT cost ≈ O(n log n) per evaluation) and therefore
can perform many thousands of iterations within the typical 500 s budget,
giving a good chance of beating the current bound.
"""

import numpy as np
import time

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a low‑value step‑function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget in seconds. The function stops a few
        seconds before the limit to avoid external termination.

    Returns
    -------
    list[float]
        The best sequence found (as plain Python floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Load & normalise the starting sequence
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:                     # fallback if the global is missing
        base = np.ones(256, dtype=np.float64)

    if base.ndim != 1 or base.size == 0:
        base = np.ones(256, dtype=np.float64)

    # protect against out‑of‑range values
    base = np.clip(base, 0.0, 1000.0)
    n = base.size

    # normalise to unit sum (the objective is scale invariant)
    if base.sum() < 0.01:
        base = np.ones_like(base) / base.size
    else:
        base = base / base.sum()

    # ------------------------------------------------------------------
    # 2.  Helper: fast evaluation using FFT (identical to evaluate_sequence)
    # ------------------------------------------------------------------
    def fast_evaluate(seq: np.ndarray) -> float:
        """Return the same value as evaluate_sequence but using FFT."""
        total = seq.sum()
        if total < 0.01:
            return np.inf
        m = 2 * seq.size - 1                 # length of the full convolution
        fa = np.fft.rfft(seq, n=m)
        conv = np.fft.irfft(fa * fa, n=m)
        max_b = np.max(conv)
        return float(2 * seq.size * max_b / (total * total))

    # ------------------------------------------------------------------
    # 3.  Initialise the search (keep the best of a few random starts)
    # ------------------------------------------------------------------
    cur = base.copy()
    cur_score = fast_evaluate(cur)

    best_seq = cur.copy()
    best_score = cur_score

    # a handful of random Dirichlet‑like starts
    for _ in range(5):
        if time.monotonic() >= time.time() + budget_s - 2.0:
            break
        rnd = rng.exponential(scale=1.0, size=n)
        rnd = rnd / rnd.sum()
        sc = fast_evaluate(rnd)
        if sc < best_score:
            best_seq = rnd.copy()
            best_score = sc
            print(f"[init] new best score {best_score:.6f}")
        if sc < cur_score:
            cur = rnd.copy()
            cur_score = sc

    # ------------------------------------------------------------------
    # 4.  Main optimisation loop
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.5)

    move_names = [
        "scale_block",
        "swap",
        "reorder",
        "symmetrize",
        "gradient_step"
    ]
    # give a little extra bias to the gradient step and block scaling
    move_weights = np.array([2, 1, 1, 1, 3], dtype=np.float64)

    it = 0
    while time.monotonic() < deadline:
        move = rng.choice(move_names, p=move_weights / move_weights.sum())
        cand = cur.copy()

        # --------------------------------------------------------------
        # 4a) Apply the chosen move
        # --------------------------------------------------------------
        if move == "scale_block":
            # pick a random block and scale it by a log‑normal factor
            low = max(1, n // 300)
            high = max(2, n // 10)
            width = int(rng.integers(low, high + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.13)))
            cand[start:start + width] *= factor

        elif move == "swap":
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]

        elif move == "reorder":
            seg_len = int(rng.integers(2, max(2, n // 5) + 1))
            start = int(rng.integers(0, n - seg_len + 1))
            segment = cand[start:start + seg_len]
            rng.shuffle(segment)
            cand[start:start + seg_len] = segment

        elif move == "symmetrize":
            cand = 0.5 * (cand + cand[::-1])

        elif move == "gradient_step":
            # compute convolution via FFT (cheap)
            m = 2 * n - 1
            fa = np.fft.rfft(cand, n=m)
            conv = np.fft.irfft(fa * fa, n=m)
            k_max = int(np.argmax(conv))

            # contributions a[i] * a[k_max - i]
            idx = np.arange(n)
            partner = k_max - idx
            valid = (partner >= 0) & (partner < n)

            prod = np.zeros_like(cand)
            prod[valid] = cand[idx[valid]] * cand[partner[valid]]

            if prod.max() > 0:
                i = int(np.argmax(prod))
                # pick a partner index with the smallest contribution
                low_idx = np.where(valid)[0]
                if low_idx.size > 0:
                    j = int(low_idx[np.argmin(prod[low_idx])])
                else:
                    j = int(rng.integers(0, n))
                # move a fraction of the mass from i to j
                delta = cand[i] * 0.3
                cand[i] -= delta
                cand[j] += delta

        # --------------------------------------------------------------
        # 4b) Clamp, normalise and evaluate
        # --------------------------------------------------------------
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 0.01:          # discard degenerate candidates
            continue
        cand = cand / s       # keep the sum ≈ 1 (scale‑invariant)

        cand_score = fast_evaluate(cand)

        # accept if strictly better for the current walk
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score

        # keep track of the overall best
        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score
            print(f"[iter {it}] new best fast score {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 5.  Final verification with the official evaluator
    # ------------------------------------------------------------------
    best_list = [float(x) for x in best_seq]
    try:
        official = float(evaluate_sequence(best_list))
        print(f"Search finished after {it} iterations.")
        print(f"Fast best   = {best_score:.6f}")
        print(f"Official    = {official:.6f}")
    except Exception as exc:      # safety – the grader will still use the list
        print(f"Official evaluation failed: {exc}")

    return best_list
