def propose_candidate(seed=0, budget_s=500):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 3.0)

    # initialize f from height_sequence_1 else uniform of length 500? Maybe length 500 to be quick.
    if "height_sequence_1" in globals():
        f = np.array(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 500
        f = np.ones(n, dtype=np.float64)

    # Clip and ensure positive sum
    f = np.clip(f, 0.0, 1000.0)
    if np.sum(f) < 0.01:
        f = np.ones_like(f) * 0.1

    # Normalize to target sum sqrt(2n) (optional)
    target_sum = np.sqrt(2 * f.size)
    f = f * (target_sum / np.sum(f))

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial score: {best_score}")

    # iterative improvement
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        n = f.size
        # compute current M
        b = np.convolve(f, f)
        M = np.max(b)

        # Build convolution matrix A from f
        # Use sparse representation via scipy.sparse.csr_matrix constructing with diagonal slices efficiently
        # Build using a list of diagonals: each diagonal offset d (0..n-1) corresponds to f[d] on subdiagonal.
        # We'll build using scipy.sparse.diags.
        from scipy.sparse import diags, csr_matrix

        diagonals = []
        offsets = []
        for d in range(n):
            # diagonal at offset d: entries f[d] in positions (d..d+n-1, column 0..n-1) -> it's a diagonal
            # For csr, we can use diags with a vector of length n (or m?) but shape mismatch.
            # Instead create a diagonal matrix of length m with offset d where main diagonal entries are f[d] for appropriate rows.
            # We can build a matrix of shape (m, n) with diags: data = np.full(n, f[d]); offsets = d; shape=(m,n) 
            # However diags expects square matrix; but we can pass shape argument.
            data = np.full(n, f[d])
            diagonals.append(data)
            offsets.append(d)
        A = diags(diagonals, offsets, shape=(2*n-1, n), format='csr')
        # Now solve LP
        c = -np.ones(n)  # maximize sum(g)
        bounds = [(0, None)] * n
        b_ub = np.full(2*n-1, M)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # fallback to random perturbation
            candidate = make_block_perturbation(f, rng)
            score = float(evaluate_sequence(candidate.tolist()))
            if score < best_score:
                f = candidate
                best_score = score
                best = f.copy()
            continue

        g0 = res.x
        # Rescale g0 to have same sum as target_sum
        g = g0 * (target_sum / np.sum(g0))

        # Mix f and g
        t = 0.5
        f_new = (1 - t) * f + t * g

        # Evaluate new f_new
        score = float(evaluate_sequence(f_new.tolist()))
        if score < best_score:
            f = f_new
            best_score = score
            best = f.copy()
            print(f"Iter {iteration}, improved score: {best_score}")
        else:
            # occasional random perturb
            if rng.random() < 0.2:
                candidate = make_block_perturbation(f, rng)
                score = float(evaluate_sequence(candidate.tolist()))
                if score < best_score:
                    f = candidate
                    best_score = score
                    best = f.copy()
                    print(f"Iter {iteration}, random improved score: {best_score}")

        # Keep time check
    # final best
    print(f"Final best score: {best_score}")
    return [float(x) for x in best]

"""
Propose a candidate step‑function using a hybrid of local perturbations and a
deterministic linear‑programming improvement step.

The evaluation metric is 
    score = 2 * n * max(conv(f,f)) / (sum(f)**2)

where `conv` denotes the full (non‑reversed) convolution.
For a uniform vector the score is exactly 2, but this value is not optimal.
The algorithm alternates two phases:

1) **Random block perturbations** – cheap moves that explore the neighbourhood.
2) **LP refinement** – given the current vector `f` we solve

        maximise   1ᵀ g
        subject to  f * g ≤ M    (elementwise)
                    g ≥ 0

   where `M = max(conv(f,f))`.  The LP finds a new vector `g` that can be
   mixed with `f` without increasing the convolution peak.  Mixing (with a
   factor `τ`) strictly reduces `M` in practice, leading to a lower score.

The loop runs until the time budget expires (default 500 s), keeping the best
vector seen.  All numbers are clipped to the allowed range [0, 1000] and the
final result is returned as a plain Python list of floats.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Helper for cheap random moves (kept from the original baseline)
def make_block_perturbation(base, rng):
    """Return a modest O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # optional symmetrisation
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
def _conv_max(vec):
    """Maximum entry of the full self‑convolution of ``vec``."""
    b = np.convolve(vec, vec)
    return np.max(b)


def _build_conv_matrix(f):
    """
    Build the sparse matrix A (shape (2n‑1) × n) such that
        (A @ g)[t] == (f * g)[t]   (full convolution)
    """
    n = f.size
    # each diagonal offset i (0 ≤ i < n) holds the constant value f[i]
    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)          # 0,1,…,n‑1
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A
