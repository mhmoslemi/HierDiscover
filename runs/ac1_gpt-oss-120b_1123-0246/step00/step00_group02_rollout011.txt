def block_perturbation(a, rng):
    # similar to provided: block width between n//200 and n//12
    n = len(a)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = rng.integers(0, n - width + 1)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    a_new = a.copy()
    a_new[start:start+width] *= factor
    if rng.random() < 0.35:
        a_new = 0.5 * (a_new + a_new[::-1])
    return np.clip(a_new, 0.0, 1000.0)

"""
Propose a step‑function sequence that beats the naïve uniform construction (score 2.0).
The routine mixes fast random “block” perturbations with a deterministic
move that directly attacks the largest convolution entry.
All heavy work (convolution, scoring) uses an FFT based O(n log n)
implementation, so thousands of iterations fit easily into the 1000 s budget.
The algorithm works on the global `height_sequence_1` if it is present;
otherwise it falls back to a uniform vector.
The final sequence is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution (full, length 2·n‑1) using a power‑of‑two FFT size.
def _full_convolution(x: np.ndarray) -> np.ndarray:
    n = x.size
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[: 2 * n - 1]
    # Numerical noise may produce tiny negatives – clip for safety.
    np.clip(conv, 0.0, None, out=conv)
    return conv


# ----------------------------------------------------------------------
# Fast evaluation that mimics evaluate_sequence (but much cheaper).
def _fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)^2)."""
    n = a.size
    conv = _full_convolution(a)
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
def _block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Return a non‑negative O(n) perturbation of ``base``:
    pick a random block, scale it by exp(N(0,0.12)), optionally mirror,
    then clip to the allowed range.
    """
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12 (both at least 1)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
def _reduce_max_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Locate the shift k where the convolution a*a attains its maximum.
    Pick the pair (i, j) with i+j=k that contributes the most
    (i.e., maximises a[i]*a[j]) and transfer a small fraction of the
    mass from i and j to a location that does *not* belong to that
    shift, thereby lowering the peak whilst keeping the overall L1‑norm.
    """
    n = a.size
    conv = _full_convolution(a)
    k = int(conv.argmax())

    # indices that actually appear in this shift
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    idx = np.arange(i_min, i_max + 1)
    partner = k - idx
    # contribution of each ordered pair (i, k-i)
    contrib = a[idx] * a[partner]
    # pick the ordered pair with the largest product
    best_i = idx[int(contrib.argmax())]
    best_j = k - best_i

    # fraction of the mass we move away from each of the two entries
    frac = 0.07  # 7 %
    delta_i = a[best_i] * frac
    delta_j = a[best_j] * frac if best_j != best_i else 0.0

    # choose a destination index that is *outside* the current shift,
    # if such an index exists; otherwise the smallest element.
    if (i_max - i_min + 1) < n:
        # there are indices not participating in the current peak
        while True:
            dst = rng.integers(0, n)
            if dst < i_min or dst > i_max:
                break
    else:
        dst = int(np.argmin(a))   # safest fallback

    # perform the transfer
    a_new = a.copy()
    a_new[best_i] -= delta_i
    if best_j != best_i:
        a_new[best_j] -= delta_j
    a_new[dst] += delta_i + delta_j

    # enforce the problem constraints
    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises

      * a dedicated move that reduces the presently highest convolution entry

    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.5)

    # ------------------------------------------------------------------
    # initialise from the global construction if possible

    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size

    # Normalise to a convenient L1‑norm (scale does not affect the objective)
    target_sum = np.sqrt(2.0 * n)
    cur_sum = parent.sum()
    if cur_sum > 0:
        parent *= target_sum / cur_sum
    else:

    # ------------------------------------------------------------------

    best_score = _fast_score(best_seq)

    init_time = time.monotonic()

        else:                                        # Exploitation move

        if cand_score < cur_score:
