"""
search.py – a fast local‑search for the convolution bound problem.

The core idea is to start from the analytically optimal “arcsine” sequence
a[i] = C·binom(2i,i)/4**i, i=0..n‑1.  This sequence satisfies
      (a * a)[k] = C²  for every k < n,
so the maximum of the convolution equals C² while the sum of the
coefficients grows only like 2·√(n/π).  The resulting score is about
π/2 ≈ 1.57 – a large improvement over the constant‑height baseline (score = 2).

From this strong starting point the routine performs cheap stochastic
updates (block multiplicative perturbations and small mass transfers) and
keeps the best sequence seen while the time budget lasts.
All operations stay O(n) in memory and use only NumPy and the provided
evaluate_sequence() function.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: build the “arcsine” (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i  (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    # recurrence: a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    # all values are ≤ 1, well inside the allowed 0‑1000 range
    return a


# ----------------------------------------------------------------------
# Small random moves
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0, σ))."""
    n = base.size
    # block width roughly between n/200 and n/12 (as in the reference code)
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrisation – it often keeps the sequence well‑behaved
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another (sum stays unchanged)."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1                     # ensure i != j
    # choose a fraction of the source value to move (≤ 30 %)
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    # Clip only to respect the hard 0‑1000 bound from evaluate_sequence
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of floats – the best step‑function heights found within
    ``budget_s`` seconds.  The algorithm:
        1. Build the arcsine sequence (optimal analytically).
        2. Repeatedly apply cheap stochastic moves (block scaling or mass transfer).
        3. Keep the best sequence as measured by evaluate_sequence().
    """
    rng = np.random.default_rng(seed)

    # Length of the problem – fall back to 256 if the global reference is missing.
    n = int(len(globals().get("height_sequence_1", []))) if "height_sequence_1" in globals() else 256
    if n == 0:
        n = 256

    # ------------------------------------------------------------------
    # 1.  Starting point – the analytically derived shape.
    # ------------------------------------------------------------------
    base_seq = build_arcsine_sequence(n)
    best = np.clip(base_seq, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Stochastic improvement loop
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)   # leave a few seconds slack
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # 60 % block scaling, 40 % mass transfer – a simple schedule that works well.
        if rng.random() < 0.6:
            candidate = block_perturbation(best, rng)
        else:
            candidate = transfer_move(best, rng)

        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"[{it:>6}] new best = {best_score:.6f}")

            # early termination if we already beat the target by a comfortable margin
            if best_score < 1.51:
                break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return as plain Python list – the harness expects a list of floats.
    return [float(x) for x in best]
