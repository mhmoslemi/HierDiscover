def compute_max_b(seq):
    n = len(seq)
    L = 1 << ((2*n - 1).bit_length())
    fft_a = np.fft.rfft(seq, n=L)
    b = np.fft.irfft(fft_a * fft_a, n=L)
    # keep first 2*n-1 elements
    max_b = b[:2*n-1].max()
    return max_b

def flatten_move(seq, max_k, rng):
    n = len(seq)
    # overlapping indices i range start = max(0, max_k - n + 1), end = min(max_k, n-1)
    i_start = max(0, max_k - n + 1)
    i_end = min(max_k, n - 1)
    # Choose i weighted by product a_i * a_{max_k - i}
    # Compute products for range
    # but to avoid O(n) each move, we can approximate by random i.
    # Simple: choose i uniformly in range.
    i = rng.integers(i_start, i_end+1)
    j = max_k - i
    # Determine delta: small fraction of min(a_i, a_j)
    min_val = seq[i] if i != j else seq[i]/2
    # If i==j, we reduce a_i only
    if i == j:
        amt = min_val * rng.random() * 0.3
        seq[i] -= amt
        # add to random other index
        k = rng.integers(0, n)
        seq[k] += amt
    else:
        # reduce both
        amt_i = seq[i] * rng.random() * 0.15
        amt_j = seq[j] * rng.random() * 0.15
        seq[i] -= amt_i
        seq[j] -= amt_j
        # add to random other indices
        k1 = rng.integers(0, n)
        k2 = rng.integers(0, n)
        seq[k1] += amt_i
        seq[k2] += amt_j
    # renormalize
    s = seq.sum()
    if s > 0:
        seq /= s
    return seq
```

def compute_max_b_and_k(seq):
    n = len(seq)
    L = 1 << ((2*n - 1).bit_length())
    fft = np.fft.rfft(seq, n=L)
    b = np.fft.irfft(fft * fft, n=L)
    b = b[:2*n-1]
    max_idx = int(np.argmax(b))
    max_val = float(b[max_idx])
    return max_val, max_idx

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm works in three phases:

1. **Initialisation**
   * Load the previous construction `height_sequence_1` (if present) and
     also build an “arcsine” shaped sequence (a_i ∝ 1/√(i·(N‑i))).
   * Normalise both candidates to sum‑to‑1 (the evaluation is scale‑invariant)
     and keep the better one as the current sequence.

2. **Iterative flattening**
   * Repeatedly compute the full linear convolution using an FFT (O(N log N)).
   * Identify the shift `k_max` where the convolution attains its maximum.
   * Perform a *flattening move* that transfers a small amount of mass from
     the two indices that contribute to `k_max` toward random other indices.
     This directly reduces the dominant convolution value.
   * Occasionally apply other cheap moves (block scaling, multiplicative noise,
     local smoothing, symmetrisation) to keep the search diverse.
   * Accept a move if it improves the score, otherwise accept it
     with a low temperature‑dependent probability (simulated annealing).

3. **Return**
   * Scale the best‑found vector into the allowed range `[0,1000]`
     (scaling does not affect the score) and return it as a plain Python
     list of floats.

The whole routine respects the 1000 s overall time limit by stopping
after `budget_s` seconds (minus a small safety margin) and prints regular
progress reports.
"""

def _convolution_max_and_index(seq: np.ndarray, pad_len: int):
    """
    Compute the maximum value of the linear convolution `seq*seq`
    and the index where it occurs.
    Uses an FFT of length `pad_len` (must be >= 2*len(seq)-1).
    Returns (max_value, argmax_index).
    """
    fft = np.fft.rfft(seq, n=pad_len)
    conv = np.fft.irfft(fft * fft, n=pad_len)
    conv = conv[: 2 * len(seq) - 1]      # keep the linear part only
    max_idx = int(np.argmax(conv))
    return float(conv[max_idx]), max_idx


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator):
    """Multiplicative perturbation of a random contiguous block."""
    n = base.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand = base.copy()
    cand[start : start + width] *= factor
    return cand


def _flatten_move(seq: np.ndarray, k_max: int, rng: np.random.Generator):
    """
    Reduce the contribution to the maximal convolution shift `k_max`.
    A small amount of mass is taken from the pair (i, j=k_max-i) and
    redistributed to two random positions.
    """
    n = seq.size
    i_low = max(0, k_max - n + 1)
    i_high = min(k_max, n - 1)
    if i_low > i_high:
        return seq  # should not happen
    i = rng.integers(i_low, i_high + 1)
    j = k_max - i
    cand = seq.copy()

    if i == j:
        # reduce a single entry
        amt = cand[i] * rng.random() * 0.20
        cand[i] = max(0.0, cand[i] - amt)
        # give it to a random location outside the pair
        dst = rng.integers(0, n)
        cand[dst] += amt
    else:
        # reduce both ends of the pair
        amt_i = cand[i] * rng.random() * 0.10
        amt_j = cand[j] * rng.random() * 0.10
        cand[i] = max(0.0, cand[i] - amt_i)
        cand[j] = max(0.0, cand[j] - amt_j)
        dst1 = rng.integers(0, n)
        dst2 = rng.integers(0, n)
        cand[dst1] += amt_i
        cand[dst2] += amt_j

    # renormalise to keep the sum = 1 (scale‑invariant)
    total = cand.sum()
    if total > 0:
        cand /= total
    else:
        # fallback to a tiny uniform vector
        cand = np.full_like(cand, 1.0 / n)
    return cand


def _arcsine_initial(n: int) -> np.ndarray:
    """Return a smooth arcsine‑shaped non‑negative vector of length n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    # 1/√(i·(n‑i)) – the discrete analogue of the continuous arcsine density
    seq = 1.0 / np.sqrt(i * (n - i + 0.5))
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total > 0:
        seq /= total
    else:
        seq = np.full(n, 1.0 / n)
    return seq

            # Local smoothing of a random sub‑segment
            width = int(rng.integers(3, max(5, n // 120)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            seg = cand[start : start + width]
            seg = 0.5 * (seg + np.pad(seg, (1, 1), mode="edge")[1:-1])
            cand[start : start + width] = seg
        else:
            # Simple pairwise mass exchange
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = min(cand[i], cand[j]) * rng.random() * 0.10
                cand[i] -= delta
                cand[j] += delta

        # Occasionally enforce approximate symmetry (helps flattenness)
        if rng.random() < 0.25:
            cand = 0.5 * (cand + cand[::-1])

        # Clip to admissible range and re‑normalise (scale‑invariant)
        cand = np.clip(cand, 0.0, 1000.0)
        total = cand.sum()
        if total > 0:
            cand /= total
        else:
            cand = np.full_like(cand, 1.0 / n)

        # Evaluate the candidate
        cand_max, _ = _convolution_max_and_index(cand, pad_len)
        cand_score = 2.0 * n * cand_max

        # Temperature for Metropolis acceptance
        elapsed = time.monotonic() - start_time
        total_time = deadline - start_time
        if total_time > 0:
            T = T_start * (T_end / T_start) ** (elapsed / total_time)
        else:
            T = T_end

        # Acceptance rule
        if cand_score < cur_score or rng.random() < math.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            seq = cand

        iter_cnt += 1
        # Periodic progress report
        if iter_cnt % 3000 == 0:
            now = time.monotonic()
            print(
                f"[{iter_cnt:06d}] best={best_score:.6f} cur={cur_score:.6f} "
                f"elapsed={now - start_time:.1f}s"
            )

    # ------------------------------------------------------------------
    # 3.  Return the best sequence, scaled to the allowed numeric range.
    # ------------------------------------------------------------------
    final = best_seq * 1000.0                # any positive scale is fine
    final = np.clip(final, 0.0, 1000.0)

    # Guard against the (unlikely) case of a zero‑sum vector
    if final.sum() < 0.01:
        final = np.full_like(final, 0.01 / final.size)

    return final.tolist()
