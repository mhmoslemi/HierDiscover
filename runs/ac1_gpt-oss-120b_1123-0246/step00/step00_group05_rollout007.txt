"""
Arc‑sine construction for a step function.

The continuous arcsine density
    f(x) = 1/(π·√(x·(1-x))) , 0<x<1
has the remarkable property that its self‑convolution is exactly the uniform
distribution on [0,2].  Discretising this density (using the interval width as
the mass of each point) gives a non‑negative sequence whose convolution is
nearly flat, i.e.  max(conv) ≈ average(conv).  For the evaluation function

    score = 2·n·max(convolution) / (sum(sequence)²)

this yields a score close to the theoretical lower bound 1, far better than
the previous uniform construction (score 2).

`propose_candidate` builds such a sequence of the same length as
`height_sequence_1` (or a default length) and then performs a very cheap
local search for the remaining time budget, keeping any improvement that
lowers the score.
"""
import time
import numpy as np
import math

def _arcsine_sequence(length: int) -> np.ndarray:
    """
    Return a length‑`length` array approximating the arcsine density.
    The array contains the probability mass of each point, i.e. the density
    multiplied by the interval width 1/length.  This guarantees that the
    total sum is close to 1, but the exact scale is irrelevant for the score.
    """
    # Mid‑point of each bin to avoid the singularities at 0 and 1.
    i = np.arange(length, dtype=np.float64)
    x = (i + 0.5) / length                # 0 < x < 1
    # arcsine pdf: 1/(π·√(x·(1‑x)))
    # Multiply by the bin width 1/length to obtain a mass.
    a = 1.0 / (math.pi * np.sqrt(x * (1.0 - x))) / length
    # Clamp to the allowed range (they are already well below 1000).
    return np.clip(a, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs) -> list[float]:
    """
    Generate a step‑function sequence based on the discretised arcsine density.
    The routine respects the time budget (`budget_s`) and performs a tiny
    stochastic refinement loop for any remaining time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.0)

    # Length of the sequence – use the previous best length if available.
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 2000                      # a reasonable fallback

    # Initial candidate: discretised arcsine distribution.
    best = _arcsine_sequence(n)

    # Initial score (for logging purposes).
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # Very cheap local refinement while we have time left.
    while time.monotonic() < deadline:
        # Propose a simple multiplicative block perturbation.
        candidate = best.copy()
        # block width: between 1 and n/200 (at least 1)
        max_width = max(2, n // 200)
        width = int(rng.integers(1, max_width + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        factor = float(np.exp(rng.normal(0.0, 0.07)))  # small log‑normal tweak
        candidate[start:start + width] *= factor

        # Occasionally enforce a slight mirror symmetry – this often smooths the
        # convolution without hurting the score.
        if rng.random() < 0.25:
            candidate = 0.5 * (candidate + candidate[::-1])

        # Clamp to the evaluator's admissible range.
        candidate = np.clip(candidate, 0.0, 1000.0)

        # Evaluate and accept if we improve.
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"[improved] new score={best_score:.6f}")

    print(f"[final] best score achieved={best_score:.6f}")
    return [float(x) for x in best]
