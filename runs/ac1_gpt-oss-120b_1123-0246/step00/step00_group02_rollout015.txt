def propose_candidate(seed=42, budget_s=500, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 3.0)

    # initialize a from height_sequence_1 if present else random positive
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        a = np.ones(256, dtype=np.float64)

    # ensure finite and positive
    a = np.clip(a, 0.0, 1000.0)
    if np.sum(a) < 1e-8:
        a = np.ones_like(a)

    # normalize sum to 1
    a = a / np.sum(a)

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # subgradient descent parameters
    eta = 0.01  # step size
    iteration = 0

    while time.monotonic() < deadline:
        # compute convolution
        b = np.convolve(a, a)
        max_idx = np.argmax(b)
        max_val = b[max_idx]
        # subgradient
        n = a.size
        grad = np.zeros_like(a)
        # for i range where i and j valid
        i_min = max(0, max_idx - (n-1))
        i_max = min(n-1, max_idx)
        for i in range(i_min, i_max+1):
            j = max_idx - i
            # i and j within bounds
            if i == j:
                grad[i] += 2*a[i]
            else:
                grad[i] += a[j]
                grad[j] += a[i]
        # update
        a = a - eta * grad
        # enforce nonnegativity
        a = np.clip(a, 0.0, None)
        # renormalize sum
        s = np.sum(a)
        if s <= 1e-12:
            a = np.ones_like(a) / a.size
            s = 1.0
        a = a / s

        # evaluate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            print(f"Iter {iteration}, new best score {best_score:.6f}, max_b {max_val:.6e}")

        iteration += 1
        # adapt step size maybe
        if iteration % 100 == 0:
            eta *= 0.9  # decay

    # after loop, maybe postprocess: ensure no huge integer rounding
    return [float(x) for x in best]

def _max_convolution(seq: np.ndarray) -> float:
    """
    Return the maximum value of the full convolution seq * seq.
    Uses FFT for speed (O(n log n)).  The result is a real scalar.
    """
    n = seq.size
    # next power of two for zero‑padding (speedy FFT)
    size = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    return float(np.max(conv.real))


def _fast_score(seq: np.ndarray) -> float:
    """
    Compute the same quantity that `evaluate_sequence` returns,
    but without the security checks and using the fast convolution.
    The value is scale‑invariant, so we can use the raw sum.

    n = seq.size
    s = float(seq.sum())
    if s < 1e-12:
        return float("inf")
    max_conv = _max_convolution(seq)
    return 2.0 * n * max_conv / (s * s)


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative block perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # block width – a few percent of the length
    w_low = max(1, n // 200)
    w_high = max(2, n // 12)
    width = int(rng.integers(w_low, w_high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    scale = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= scale
    # occasional symmetrisation
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # clipping as done in evaluate_sequence
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _power_law_seq(n: int, exponent: float = 0.5, shift: float = 0.0) -> np.ndarray:
    """Generate a normalised power‑law sequence a_i ∝ (i+shift)^{-exponent}."""
    idx = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.power(idx + shift + 1.0, exponent)  # +1 avoids division by zero
    vals /= vals.sum()
    return vals


def _mixed_seq(n: int, exponent: float, shift: float, alpha: float) -> np.ndarray:
    """Mixture of uniform and a power‑law sequence."""
    uniform = np.full(n, 1.0 / n)
    power = _power_law_seq(n, exponent=exponent, shift=shift)
    mixed = alpha * uniform + (1.0 - alpha) * power
    mixed /= mixed.sum()
    return mixed

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a sequence that minimises `evaluate_sequence`.
    The function respects the provided time budget and returns the
    best sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Choose length and initial best candidate
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base_seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if base_seq.ndim != 1 or base_seq.size == 0 or not np.isfinite(base_seq).all():
            n = 2000
        else:
            n = base_seq.size
    else:
        n = 2000

    # Normalise the starting vector (scale does not matter for the score)
    base_seq = np.clip(base_seq, 0.0, 1000.0)
    if base_seq.sum() == 0.0:
        base_seq = np.ones(n, dtype=np.float64)
    else:
        base_seq = base_seq[:n]               # truncate / pad if necessary
        if base_seq.size < n:
            base_seq = np.pad(base_seq, (0, n - base_seq.size), constant_values=0.0)

    # Ensure we have exactly n entries
    base_seq = base_seq[:n]
    if base_seq.size < n:
        base_seq = np.pad(base_seq, (0, n - base_seq.size), constant_values=0.0)

    # Initial candidate – start from a nice analytic shape
    current = _power_law_seq(n, exponent=0.5, shift=0.0)
    best = current.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Also keep the original (might already be good)
    orig_score = float(evaluate_sequence(base_seq.tolist()))
    if orig_score < best_score:
        best = base_seq.copy()
        best_score = orig_score

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main search loop
    # ------------------------------------------------------------------
    iteration = 0
    # step size for the mass‑swap neighbourhood (fraction of the source mass)
    swap_step = 0.1

    while time.monotonic() < deadline:
        iteration += 1
        # Choose a move type with probabilities
        move_type = rng.choice(["param", "block", "swap"], p=[0.25, 0.45, 0.30])

        if move_type == "param":
            # Randomised mixture of uniform + power‑law
            exponent = float(rng.uniform(0.35, 0.65))   # around the analytic 0.5
            shift = float(rng.exponential(5.0))        # small positive shift
            alpha = float(rng.uniform(0.0, 0.4))        # amount of uniform mixing
            cand = _mixed_seq(n, exponent=exponent, shift=shift, alpha=alpha)

        elif move_type == "block":
            # Block perturbation of the current best
            cand = _make_block_perturbation(best, rng)

        else:  # "swap"
            # Small mass‑swap between two positions
            cand = best.copy()
            i, j = rng.choice(n, size=2, replace=False)
            if cand[i] > 0.0:
                epsilon = swap_step * cand[i] * rng.uniform(0.2, 0.8)
                cand[i] = max(cand[i] - epsilon, 0.0)
                cand[j] = cand[j] + epsilon
            # small normalisation to avoid drift
            s = cand.sum()
            if s > 0:
                cand /= s

        # Fast evaluation first
        fast_sc = _fast_score(cand)
        if fast_sc >= best_score:
            continue  # no chance to improve

        # Authoritative evaluation
        try:
            true_sc = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue  # safety – treat as invalid

        if np.isfinite(true_sc) and true_sc < best_score:
            best = cand.copy()
            best_score = true_sc
            print(f"[{iteration:5d}] new best score {best_score:.6f} (type={move_type})")

        # gradually shrink the swap step to fine‑tune later
        if iteration % 200 == 0:
            swap_step *= 0.9
            swap_step = max(swap_step, 0.01)

    print(f"[final] best score = {best_score:.6f} after {iteration} iterations")
    # Return a plain Python list of floats
    return [float(v) for v in best]
