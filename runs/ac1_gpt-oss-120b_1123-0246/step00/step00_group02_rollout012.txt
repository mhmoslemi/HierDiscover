def propose_candidate(seed=42, budget_s=500, **kwargs):
    import numpy as np, time
    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # choose length
    n = 256
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        # maybe truncate or pad to n
        if base.size >= n:
            a = base[:n].copy()
        else:
            a = np.concatenate([base, np.full(n - base.size, 1e-6)])
    else:
        a = rng.random(n) + 1e-6

    # ensure positive
    a = np.clip(a, 1e-6, 1000.0)

    # function to compute score
    def score_seq(arr):
        return float(evaluate_sequence(arr.tolist()))

    best = a.copy()
    best_score = score_seq(best)

    t = 0.2
    t_decay = 0.9995

    # Pre-allocate arrays for performance
    while time.monotonic() < deadline:
        # compute convolution and max
        b = np.convolve(a, a)
        M = float(b.max())
        # Build constraint matrix A_ub
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = a[i]
        # solve LP
        c = -np.ones(n, dtype=np.float64)  # maximize sum
        ub = np.full(rows, M, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method='highs', options={'presolve': True})
        if not res.success:
            # fallback to random perturbation
            candidate = make_block_perturbation(a, rng)
            a = candidate
        else:
            g = res.x
            # optionally ensure positivity epsilon
            g = np.maximum(g, 1e-12)

            # mix
            a = (1 - t) * a + t * g

        # Clip and enforce min epsilon
        a = np.clip(a, 1e-12, 1000.0)

        # Evaluate occasionally
        if time.monotonic() < deadline:
            cur_score = score_seq(a)
            if cur_score < best_score:
                best = a.copy()
                best_score = cur_score
                print(f"New best score {best_score:.6f} at time {time.monotonic():.2f}")

        # decay parameters
        t *= t_decay
        # maybe small random perturb after some iterations
        if rng.random() < 0.01:
            a = make_block_perturbation(a, rng)

    # Return best
    print(f"Finished with best score {best_score}")
    return [float(x) for x in best]

"""
Hybrid alternating linear‑programming / stochastic local search.

We keep a non‑negative step‑function `a` (the “height sequence”).  For a given
`a` we compute the maximal entry `M` of its self‑convolution `a*a`.  While
keeping this bound fixed we solve the linear program

    maximise   sum(g)
    subject to (a * g)[k] ≤ M   for all k
               g ≥ 0,   g ≤ 1000

which puts as much mass as possible into positions where `a` is small.
The solution `g` is then mixed back into `a` :

    a ← (1‑t)·a + t·g        (t is a small step‑size)

Repeating the procedure flattens the autocorrelation and reduces the
evaluation score `2·n·max(a*a)/sum(a)^2`.  Random block‑perturbations and
re‑starts are interleaved to escape local minima.  The whole search runs
until the allotted time budget expires and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: small block perturbation (same idea as the reference code)
def make_block_perturbation(base, rng):
    """Generate one O(n) non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # choose a block width between n/200 and n/12 (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative noise on the whole block
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetry to keep the shape balanced
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Helper: builds the dense matrix A for the constraints (a * g)[k] ≤ M
def build_conv_matrix(a):
    """Return matrix A (shape (2n‑1, n)) such that A @ g = a * g."""
    n = a.size
    rows = np.arange(2 * n - 1)[:, None]      # shape (2n‑1, 1)
    cols = np.arange(n)                       # shape (n, )
    diff = rows - cols                         # shape (2n‑1, n)
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    # fill only the entries that are inside the valid index range
    A[mask] = a[diff[mask]]
    return A


# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=500, **kwargs):
    """Search for a sequence of non‑negative heights that minimises the
    given evaluation function.  The function runs for up to ``budget_s`` seconds
    and returns the best sequence found (as a list of floats)."""
    import numpy as np
    import time

    # try to import a linear‑programming solver; fall back to pure local search
    try:
        from scipy.optimize import linprog
        have_lp = True
    except Exception:
        have_lp = False

    rng = np.random.default_rng(seed)
    # ---- configuration -------------------------------------------------
    # we work with a moderate size n (large enough to have flexibility,
    # small enough for fast LP solves).  256 works well on most machines.
    n = 256
    target_sum = np.sqrt(2.0 * n)            # normalisation used in the paper
    eps = 1e-8                               # minimal positive value (avoids unbounded LP)

    # initial sequence: random positive values, optionally seeded from the
    # historic best if it exists.
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.size >= n:
            a = base[:n].copy()
        else:
            pad = np.full(n - base.size, eps)
            a = np.concatenate([base, pad])
    else:
        a = rng.random(n) + eps

    a = np.clip(a, eps, 1000.0)

    # ------------------------------------------------------------------
    def evaluate(arr):
        """wrapper around the provided evaluate_sequence."""
        return float(evaluate_sequence(arr.tolist()))

    best_seq = a.copy()
    best_score = evaluate(best_seq)

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # algorithmic parameters
    t = 0.3               # initial mixing weight
    t_min = 0.02          # lower bound for the step size
    t_decay = 0.9995      # geometric decay per iteration
    no_improve_limit = 300   # restart after this many non‑improving steps
    iter_without_improve = 0

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    iteration = 0
    while time.monotonic() < deadline:
        # -------------------------------------------------------------
        # 1) current max of the self‑convolution
        conv_self = np.convolve(a, a)
        M = float(conv_self.max())

        # -------------------------------------------------------------
        # 2) try to obtain a better companion sequence by LP
        if have_lp:
            A = build_conv_matrix(a)          # (2n‑1, n)
            ub = np.full(A.shape[0], M)       # right‑hand side of each inequality
            c = -np.ones(n, dtype=np.float64) # maximise sum(g) => minimise -sum(g)
            # variable bounds: 0 ≤ g_i ≤ 1000 (the same clipping used later)
            bounds = [(0.0, 1000.0) for _ in range(n)]

            # solve the LP (highs is fast for dense problems of this size)
            res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                          method='highs', options={'presolve': True})
            if res.success:
                g = res.x
            else:
                # fallback to a random local move if the LP failed
                g = make_block_perturbation(a, rng)
        else:
            # No LP solver – just use a random move
            g = make_block_perturbation(a, rng)

        # -------------------------------------------------------------
        # 3) scale g to the target ℓ₁‑norm (as described in the paper)
        sum_g = float(g.sum())
        if sum_g > 0:
            scale = target_sum / sum_g
            g = g * scale
        else:
            # degenerate case – keep the old sequence
            g = a.copy()

        # -------------------------------------------------------------
        # 4) convex combination of the old and the new sequence
        a = (1.0 - t) * a + t * g
        a = np.clip(a, eps, 1000.0)          # protect against clipping limits

        # -------------------------------------------------------------
        # 5) occasionally evaluate the true objective
        if iteration % 5 == 0:
            cur_score = evaluate(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{iteration:05d}] new best = {best_score:.6f} (t={t:.4f})")
                iter_without_improve = 0
            else:
                iter_without_improve += 1

        # -------------------------------------------------------------
        # 6) adapt parameters & random shake‑up
        t = max(t_min, t * t_decay)

        if iter_without_improve >= no_improve_limit:
            # restart from the current best with a small random shock
            a = best_seq.copy()
            a = make_block_perturbation(a, rng)
            a = np.clip(a, eps, 1000.0)
            iter_without_improve = 0
            print(f"[{iteration:05d}] restart after {no_improve_limit} stagnant steps")

        iteration += 1

    # ------------------------------------------------------------------
    print(f"[final] best score found: {best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best_seq]
