import numpy as np
import time

def subgradient_max(a, max_inds):
    """Return subgradient of max_k (a*a)[k] w.r.t a for active indices."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in max_inds:
        lo = k - (n - 1)
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        # range of i where both i and k-i are valid
        # i from lo..hi
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        g[idx] += 2.0 * a[k - idx]
    return g

def block_perturb(a, rng):
    """Return a copy of a after a multiplicative block perturbation."""
    n = a.shape[0]
    # choose width between n/200 and n/12
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    out = a.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    out[start:stop] *= factor
    if rng.random() < 0.35:
        out = 0.5 * (out + out[::-1])
    # clamp and renorm to keep sum = 1
    out = np.clip(out, 0.0, None)
    total = out.sum()
    if total <= 0:
        out = rng.random(n)
        total = out.sum()
    out /= total
    return out

def propose_candidate(seed=42, budget_s=500, **kwargs):
    """Search for a sequence minimizing the evaluation function.

    The method mixes a sub‑gradient descent on the maximal convolution coefficient
    with occasional random block perturbations (simulated‑annealing style).  The
    sequence is kept on the probability simplex (unit L1 norm) – the objective
    is scale‑invariant, so this does not affect the score.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 3.0)

    # ----- initial sequence -------------------------------------------------
    # use the supplied construction if it looks valid, otherwise random
    if 'height_sequence_1' in globals():
        try:
            init = np.asarray(height_sequence_1, dtype=np.float64)
            if init.ndim == 1 and init.size > 0 and np.isfinite(init).all():
                a = np.clip(init, 0.0, None)
            else:
                a = None
        except Exception:
            a = None
    else:
        a = None

    if a is None or a.size == 0:
        a = rng.random(256, dtype=np.float64)

    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum <= 0:
        a = rng.random(a.shape[0], dtype=np.float64)
        a_sum = a.sum()
    a = a / a_sum                     # unit L1 norm

    n = a.shape[0]

    # ----- bookkeeping ------------------------------------------------------
    best = a.copy()
    # compute initial score directly (avoid evaluate_sequence overhead)
    cur_b = np.convolve(a, a)
    cur_max = cur_b.max()
    cur_score = 2.0 * n * cur_max      # since sum(a)=1
    best_score = cur_score

    last_report = start_time
    it = 0
    last_improve = start_time

    # parameters for simulated annealing
    T_initial = 0.2
    grad_prob = 0.7         # probability to take a gradient step
    step_initial = 0.5

    while time.monotonic() < deadline:
        it += 1
        # ----- propose a new candidate -------------------------------------
        if rng.random() < grad_prob:
            # sub‑gradient descent step
            # recompute convolution and active set
            b = np.convolve(a, a)
            max_val = b.max()
            max_inds = np.where(b == max_val)[0]

            g = subgradient_max(a, max_inds)

            cur_score = 2.0 * n * np.convolve(a, a).max()
            cur_max = np.convolve(a, a).max()
            last_improve = now
            print(f"Restarted after {now - start_time:.1f}s without improvement.")

    # final best sequence
    # ensure it's a plain Python list of floats (evaluate_sequence will do its own clipping)
    return [float(x) for x in best]

def _subgradient_max(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a*a)[k]  w.r.t. ``a`` for the active indices ``idxs``.
    For a single active index k the gradient is 2*a shifted by k.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in idxs:
        lo = k - (n - 1)
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        i = np.arange(lo, hi + 1, dtype=np.int64)
        g[i] += 2.0 * a[k - i]
    return g


def _block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Small multiplicative perturbation of a contiguous block of ``a``.
    Afterwards the vector is clipped to be non‑negative and renormalised
    (unit L1 norm) – the objective is scale invariant.
    """
    n = a.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))

    s = out.sum()
    if s <= 0.0:
        out = rng.random(n, dtype=np.float64)
        s = out.sum()
    out /= s
    return out
