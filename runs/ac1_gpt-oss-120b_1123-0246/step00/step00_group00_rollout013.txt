def linear_ramp_sequence(length: int) -> np.ndarray:
    """
    Return a linearly increasing sequence of the given length that starts at 0
    and ends exactly at 1000.  The sequence satisfies the evaluator's
    non‑negative and max‑value (≤1000) constraints.
    """
    if length <= 1:
        # A single element must be 1000 to keep the scale optimal.
        return np.full(length, 1000.0, dtype=np.float64)
    # np.linspace already guarantees values within the required range.
    return np.linspace(0.0, 1000.0, length, dtype=np.float64)


def safe_evaluate(sequence: np.ndarray) -> float:
    """
    Wrapper around the provided ``evaluate_sequence`` function.
    Returns ``inf`` on unexpected errors and prints a short diagnostic.
    """

        score = float(evaluate_sequence(sequence.tolist()))
        return score
    except Exception as exc:                     # pragma: no cover
        print("evaluate_sequence raised an exception:", exc)
        return float("inf")
