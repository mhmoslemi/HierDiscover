    start_time = time.monotonic()

    rng = np.random.default_rng(seed)

    n = base.size

    while time.monotonic() < deadline:
        # decide move type
        move_type = rng.integers(0, 4)  # 0: pairwise transfer, 1: max reduction, 2: block scaling, 3: random gaussian perturb
        if move_type == 0:
            # pairwise transfer
            i, j = rng.choice(n, size=2, replace=False)
            # delta limited to small fraction of min(a_i, a_j)
            max_delta = min(current[i], 1000.0 - current[j])  # can't exceed bound for j
            if max_delta <= 0:

            # choose delta relative
            delta = max_delta * rng.random() * 0.1  # up to 10% of possible
            candidate = current.copy()
            candidate[i] -= delta
            candidate[j] += delta
        elif move_type == 1:

            conv = np.convolve(current, current)
            max_idx = int(np.argmax(conv))

            # ensure not same index
            other = rng.integers(0, n)

            max_delta = min(current[i], 1000.0 - current[other])
            if max_delta <= 0:
                continue
            delta = max_delta * rng.random() * 0.05  # small up to 5%
            candidate = current.copy()
            candidate[i] -= delta
            candidate[other] += delta
        elif move_type == 2:

            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            candidate = current.copy()
            candidate[start:stop] *= factor
            # Ensure bounds
            candidate = np.clip(candidate, 0.0, 1000.0)
            # Renormalize sum to original sum (or current sum)
            sum_orig = np.sum(current)
            sum_cand = np.sum(candidate)
            if sum_cand > 0:
                candidate *= (sum_orig / sum_cand)
                candidate = np.clip(candidate, 0.0, 1000.0)
        else:

            candidate = np.clip(candidate, 0.0, 1000.0)
            # No sum renorm needed

        # Evaluate candidate
        score = float(evaluate_sequence(candidate.tolist()))
        if not np.isfinite(score):
            continue

        # acceptance
        if score < current_score:
            accept = True
        else:
            # temperature
            elapsed = time.monotonic() - start_time
            remaining = deadline - time.monotonic()
            # linear temperature schedule

            # if T very small, rarely accept uphill
            prob = math.exp(-(score - current_score) / (T + 1e-12))
            accept = rng.random() < prob

        if accept:
            current = candidate
            current_score = score
            # update best
            if score < best_score:
                best_score = score
                best_seq = candidate.copy()
                print(f"[{time.monotonic():.2f}] New best: {best_score:.6f}")

        # optional occasional re-randomization if stagnant
        # maybe after certain iterations, random restart
        # we can track number of iterations without improvement

        # Check for termination: maybe if best_score close to lower bound (2*n/(2*n-1))
        # but we cannot guarantee.

    # final result
    # Ensure result is list of floats
    return [float(x) for x in best_seq]

    # define helper evaluate

        return float(evaluate_sequence(seq.tolist()))

    # Baseline evaluation
    best_seq = base.copy()
    best_score = eval_seq(best_seq)

    # Uniform baseline: all ones (if not already)
    uniform = np.ones(n, dtype=np.float64)
    uniform_score = eval_seq(uniform)
    if uniform_score < best_score:
        best_seq = uniform.copy()
        best_score = uniform_score

    print(f"[{time.monotonic():.2f}] start score {best_score:.6f} length {n}")

    # 1) Grid search over two‑block step functions
    # sample p from 0.05 to 0.5 (or up to 0.9) with step 0.05
    ps = np.linspace(0.05, 0.5, 10)  # up to half size
    for p in ps:
        m = max(1, int(round(p * n)))
        max_L = 1.0 / p  # from sum constraint (average =1)
        # sample L values logarithmically
        L_vals = np.linspace(0.5, max_L, 30)
        for L in L_vals:
            s = (1 - p * L) / (1 - p)   # because we keep sum = n (average 1)
            if s < 0:   # infeasible
                continue
            candidate = np.empty(n, dtype=np.float64)
            candidate[:m] = L
            candidate[m:] = s
            # scale to keep sum exactly n (may be small floating error)
            candidate = candidate * (n / np.sum(candidate))
            score = eval_seq(candidate)
            if score < best_score:
                best_score = score
                best_seq = candidate.copy()

    # 2) Simulated annealing starting from best_seq
    current = best_seq.copy()
    current_score = best_score

    # parameters for annealing
    T0 = 0.4
    # iterate until time limit
    iter_count = 0
    while time.monotonic() < deadline:
        iter_count += 1
        # pick move type
        move = rng.integers(0, 5)
        if move == 0:
            # simple pairwise transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            while j == i:
                j = rng.integers(0, n)
            # determine max amount we can move without breaking bounds
            max_delta = min(current[i], 1000.0 - current[j])
            if max_delta <= 0.0:

"""
Propose a sequence of non‑negative heights that (hopefully) improves the
evaluation score.

The algorithm works in three stages:

1. **Baseline & grid search** – start from the supplied `height_sequence_1`
   (or a uniform sequence).  Scan a simple two‑block family
   `a = [L]*m + [s]*(n-m)` where `m` and `L` are varied, compute the exact
   score and keep the best member.

2. **Simulated annealing** – use the best candidate as a seed and run a
   time‑bounded stochastic optimisation.  Moves include
   *pairwise mass transfer*, *block scaling with renormalisation*,
   *small Gaussian jitter*, *targeted reduction of the current maximal
   convolution index* and occasional random resets.  A linear temperature
   schedule governs uphill acceptance.

3. **Return the best sequence** – the algorithm prints intermediate progress
   (timestamp, score, iteration) and finally returns the best sequence found
   before the time budget expires.

The objective is invariant under global scaling, therefore the sum of the
sequence is kept close to the initial value (≈ n) for numerical stability,
but any scaling respects the problem constraints.
"""

import math
import time
import numpy as np

def make_block_perturbation(base, rng):
    """Return one O(n) memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

            s = (1.0 - p * L) / (1.0 - p)                     # enforce Σa = n
            if s < 0.0 or s > 1000.0:
                continue
            seq = np.empty(n, dtype=np.float64)
            seq[:m] = L
            seq[m:] = s
            # small renormalisation to correct rounding errors
            seq *= (n / np.sum(seq))
            score = eval_seq(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.monotonic():.2f}] block p={p:.2f} L={L:.3f} → {score:.6f}")

    # ------------------------------------------------------------------ #
    # 2) SIMULATED ANNEALING FROM THE BEST FOUND POINT
    # ------------------------------------------------------------------ #
    current = best_seq.copy()
    current_score = best_score

    T0 = 0.4                         # initial temperature
    it = 0
    while time.monotonic() < deadline:
        it += 1
        move_type = rng.integers(0, 5)

        if move_type == 0:                                            # pairwise transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n)
            max_delta = min(current[i], 1000.0 - current[j])
            if max_delta <= 0.0:
                continue
            delta = max_delta * rng.random() * 0.2
            cand = current.copy()
            cand[i] -= delta
            cand[j] += delta

        elif move_type == 1:                                          # block scaling + renorm
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand = current.copy()
            cand[start:stop] *= factor
            cand = np.clip(cand, 0.0, 1000.0)
            # renormalise to keep the total sum (sum cancels out of the objective)
            cand = cand * (np.sum(current) / np.sum(cand))

        elif move_type == 2:                                          # tiny Gaussian jitter
            cand = current.copy()
            mask = rng.random(n) < 0.02
            if not np.any(mask):
                continue
            jitter = rng.normal(0.0, 0.2, size=n)
            cand[mask] += jitter[mask]
            cand = np.clip(cand, 0.0, 1000.0)

        elif move_type == 3:                                          # targeted max‑conv reduction
            conv = np.convolve(current, current)
            k = int(np.argmax(conv))                                 # index with highest value
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            i = rng.integers(i_min, i_max + 1)
            j = k - i
            # choose a receiver far from the pair
            rec = rng.integers(0, n)
            while rec == i or rec == j:
                rec = rng.integers(0, n)
            donor = i if current[i] > current[j] else j
            max_delta = min(current[donor], 1000.0 - current[rec])
            if max_delta <= 0.0:
                continue
            delta = max_delta * rng.random() * 0.1
            cand = current.copy()
            cand[donor] -= delta
            cand[rec] += delta

        else:                                                          # small random reset of a few entries
            cand = current.copy()
            idx = rng.choice(n, size=3, replace=False)
            cand[idx] = rng.random(3) * 5.0
            cand = np.clip(cand, 0.0, 1000.0)

        # ------------------------------------------------------------------
        # evaluate candidate and decide acceptance
        score = eval_seq(cand)
        if not np.isfinite(score):
            continue

        if score < current_score:
            accept = True
        else:
            elapsed = time.monotonic() - start_time
            fraction = elapsed / max(1e-9, (budget_s - 1.0))
            T = T0 * max(0.0, 1.0 - fraction)          # linear cooling
            prob = math.exp(-(score - current_score) / (T + 1e-12))
            accept = rng.random() < prob

        if accept:
            current = cand
            current_score = score
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[{time.monotonic():.2f}] new best {best_score:.6f} (iter {it})")

        # occasional resets to avoid stagnation
        if it % 50000 == 0:
            tmp = make_block_perturbation(best_seq, rng)
            tmp = np.clip(tmp, 0.0, 1000.0)
            tmp = tmp * (np.sum(best_seq) / np.sum(tmp))
            tmp_score = eval_seq(tmp)
            if tmp_score < best_score:
                best_score = tmp_score
                best_seq = tmp.copy()
                current = tmp.copy()
                current_score = tmp_score
                print(f"[{time.monotonic():.2f}] reset improved → {best_score:.6f}")

        # early exit if we already beat the target
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------ #
    print(f"[{time.monotonic():.2f}] finished – best score {best_score:.6f}")
    return [float(x) for x in best_seq]
