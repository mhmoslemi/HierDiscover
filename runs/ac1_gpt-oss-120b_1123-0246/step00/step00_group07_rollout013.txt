---------

    sum_a = seq.sum()
    # sum_a is never zero in our usage
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves
    the current best score.  The routine works inside the allotted time
    budget (default 500 s) and uses a smooth‑max gradient descent with
    occasional random block perturbations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length of the sequence we will optimise.
    # Prefer the length of the previous construction; otherwise use 256.
    try:
        n = int(len(height_sequence_1))
        if n < 2:
            raise Exception
    except Exception:
        n = 256

    # Initialise: uniform distribution with zero endpoints.
    a = np.full(n, 1.0 / n, dtype=np.float64)
    a[0] = 0.0
    a[-1] = 0.0
    a /= a.sum()

    best = a.copy()
    best_score = _fast_score(best)

    # Hyper‑parameters
    gamma = 200.0          # sharpness of the soft‑max (larger → closer to true max)
    lr = 0.02              # initial learning rate
    decay = 0.998          # learning‑rate decay per iteration
    min_lr = 5e-4          # lower bound for the step size
    max_iter = 10_000      # safety cap – the loop also respects the deadline

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    iteration = 0

    while time.monotonic() < deadline and iteration < max_iter:
        # --------------------------------------------------------------
        # 1) Full convolution a*a  (linear convolution, length 2n‑1)
        b = np.convolve(a, a)                     # shape (2n‑1,)
        max_b = b.max()

        # 2) Soft‑max weights over the convolution entries
        #    w_i = exp(gamma * (b_i - max_b)) / Σ exp(...)
        #    This concentrates the weight on the currently largest entry.
        w_raw = np.exp(gamma * (b - max_b))
        w_sum = w_raw.sum()
        if w_sum == 0.0:          # numerical safeguard
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # 3) Gradient of the smooth‑max objective.
        #    grad_i = a_i + Σ_j a_j * w_{i+j}
        #    The second term is a linear convolution of a with the reversed w.
        w_rev = w[::-1]                           # reverse for the correct shift
        conv_full = np.convolve(a, w_rev)         # length 3n‑2
        # extract the slice that corresponds to Σ_j a_j w_{i+j}
        # For i = 0 … n‑1 the needed indices are n‑1 … 2n‑2
        G = conv_full[n - 1: n - 1 + n]           # shape (n,)

        grad = a + G                               # shape (n,)

        # --------------------------------------------------------------
        # 4) Gradient step (simple SGD with decaying learning rate)
        a = a - lr * grad
        a = np.clip(a, 0.0, None)                  # keep coefficients non‑negative

        # Force the endpoints to stay zero (they cannot help the max‑convolution)
        a[0] = 0.0
        a[-1] = 0.0

        # Re‑normalise to keep the scale invariant (sum = 1)
        total = a.sum()
        if total > 0.0:
            a /= total
        else:
            # Very unlikely – fall back to uniform if everything vanished
            a = np.full(n, 1.0 / n, dtype=np.float64)
            a[0] = a[-1] = 0.0
            a /= a.sum()

        # Occasionally inject a tiny random block perturbation
        # (helps to jump out of tiny plateaus)
        if iteration % 64 == 0 and rng.random() < 0.15:
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.09)))
            a[start:start + width] *= factor
            a = np.clip(a, 0.0, None)
            a[0] = a[-1] = 0.0
            a /= a.sum()

        # --------------------------------------------------------------
        # 5) Keep the best seen sequence (using the exact same formula as
        #    evaluate_sequence – this guarantees that the returned value is
        #    correctly assessed by the external grader).
        cur_score = _fast_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(
                f"Iter {iteration:5d}: improved score = {best_score:.6f}",
                flush=True,
            )

        # Decay learning rate but never let it become too small
        lr = max(min_lr, lr * decay)

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list of floats)
    return [float(x) for x in best]
