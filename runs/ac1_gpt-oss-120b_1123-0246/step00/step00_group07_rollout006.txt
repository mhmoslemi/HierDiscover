import numpy as np
import time

def fast_objective(seq):
    """
    Fast approximation of evaluate_sequence using FFT.
    Returns np.inf for invalid sequences.
    """
    seq = np.asarray(seq, dtype=np.float64)
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = seq.sum()
    if sum_a < 0.01 or not np.isfinite(seq).all():
        return np.inf
    n = seq.shape[0]
    # next power of two for convolution
    m = 1 << ((2 * n - 1).bit_length())
    # compute convolution via FFT
    fft_seq = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft_seq * fft_seq, m)[:2 * n - 1]
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (sum_a * sum_a))

def normalize_to_target(seq, target_sum):
    """Scale seq to have sum == target_sum, clipping to [0,1000]."""
    seq = np.asarray(seq, dtype=np.float64)
    seq = np.clip(seq, 0.0, 1000.0)
    current = seq.sum()
    if current == 0:
        return seq
    factor = target_sum / current
    seq = seq * factor
    # after scaling, clip again (just in case)
    seq = np.clip(seq, 0.0, 1000.0)
    return seq

def generate_parametric_shapes(n, target_sum):
    """Yield a collection of deterministic parametric shapes for a given length."""
    # Uniform
    yield normalize_to_target(np.ones(n, dtype=np.float64), target_sum)

    # Linear decreasing from n to 1
    lin = np.arange(n, 0, -1, dtype=np.float64)
    yield normalize_to_target(lin, target_sum)

    # Half-sine shape
    sine = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    yield normalize_to_target(sine, target_sum)

    # Exponential families
    for p in np.linspace(0.0, 5.0, 51):
        exp_seq = np.exp(-p * np.arange(n, dtype=np.float64))
        yield normalize_to_target(exp_seq, target_sum)

    # Power‑law families a_i = 1/(i+1)^p
    for p in np.linspace(0.1, 3.0, 30):
        power = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        yield normalize_to_target(power, target_sum)

    # Combination: sin * exp
    for p in np.linspace(0.0, 5.0, 26):
        combo = np.sin(np.pi * (np.arange(n) + 0.5) / n) * np.exp(-p * np.arange(n, dtype=np.float64))
        yield normalize_to_target(combo, target_sum)

def block_perturbation(base, rng):
    """Scale a random contiguous block of the sequence."""
    cand = base.copy()
    n = cand.size
    # width between n/200 and n/12 roughly
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))

    cand[start:start + width] *= scale
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

def transfer_mutation(base, rng):
    """Transfer a random amount from one position to another."""
    cand = base.copy()
    n = cand.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j:
        return cand
    amount = rng.random() * cand[i]  # up to whole value at i
    cand[i] -= amount
    cand[j] += amount
    return np.clip(cand, 0.0, 1000.0)

def smooth_mutation(base, rng):
    """Apply a short moving‑average smoothing while preserving the total sum."""
    cand = np.convolve(base, np.ones(3) / 3.0, mode='same')
    total = cand.sum()
    if total > 0:
        cand = cand * base.sum() / total
    return np.clip(cand, 0.0, 1000.0)

def noise_mutation(base, rng):

    avg = np.mean(base) if base.mean() > 0 else 1.0
    sigma = avg * 0.05
    cand = base + rng.normal(0.0, sigma, size=base.shape)

        smooth_mutation,
        noise_mutation,
    ]

        # Here we keep the original sum to avoid artificial scaling effects.
        candidate = normalize_to_target(candidate, target_sum)

        score = fast_objective(candidate)
        if score < best_score:

    # -----------------------------------------------------------------

    print(f"Finished search: best score = {best_score:.6f} (length={len(result)})")

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: fast version of evaluate_sequence (uses FFT for speed)
# ----------------------------------------------------------------------
def fast_objective(seq):
    """
    Compute the same value as evaluate_sequence(seq) but in O(n log n).

    Returns np.inf if the sequence would be rejected by the original
    evaluator (negative entries, NaN/inf, or sum < 0.01).
    """
    seq = np.asarray(seq, dtype=np.float64)
    seq = np.clip(seq, 0.0, 1000.0)

    sum_a = seq.sum()
    if sum_a < 0.01 or not np.isfinite(seq).all():
        return np.inf

    n = seq.size
    # length for zero‑padded FFT (next power of two ≥ 2n‑1)
    m = 1 << ((2 * n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft_seq * fft_seq, m)[: 2 * n - 1]
    max_b = np.max(conv)

    return float(2.0 * n * max_b / (sum_a * sum_a))

    """
    Scale seq so that its L1‑norm equals target_sum,
    then clip to the allowed interval [0,1000].
    """

    cur = seq.sum()
    if cur == 0:
        return seq
    factor = target_sum / cur
    seq = seq * factor
    return np.clip(seq, 0.0, 1000.0)
