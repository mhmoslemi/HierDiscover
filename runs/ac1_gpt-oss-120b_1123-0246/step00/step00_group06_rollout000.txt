```
def fast_score(seq):
    arr = np.clip(seq, 0.0, 1000.0)
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    n = arr.size
    m = 1 << ((2*n-1).bit_length())
    fft_seq = np.fft.rfft(arr, n=m)
    conv = np.fft.irfft(fft_seq*fft_seq, n=m)[:2*n-1]
    max_b = conv.max()
    return 2 * n * max_b / (sum_a**2)
```

import numpy as np
import time

def arcsine_sequence(length):
    """Return a non‑negative sequence of length ``length`` whose
    self‑convolution is (up to discretisation) flat.
    The entries are proportional to 1/√(x(1‑x)) with x = (i+½)/length,
    i.e. the discrete arcsine law.  Values are clipped to the allowed
    interval [0,1000].
    """
    i = np.arange(length, dtype=np.float64) + 0.5
    seq = length / np.sqrt(i * (length - i))
    return np.clip(seq, 0.0, 1000.0)

def make_block_perturbation(base, rng):
    """Return one O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

def fast_score(seq):
    """Fast approximation of ``evaluate_sequence`` using FFT.
    Returns the same functional value (2 n · max(conv) / sum(seq)²)
    but is ≈ O(n log n) instead of the quadratic cost of
    ``np.convolve``.
    """
    a = np.clip(seq, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    n = a.size
    # next power‑of‑two size that can contain the full linear convolution
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)
