"""
Proposed solution
=================

A fast projected gradient descent is used to minimise the evaluation
function

    2 * n * max(a * a) / (sum(a) ** 2)

The sum of the coefficients does not influence the score, therefore we
normalise the sequence to have unit L1‑norm (∑a = 1).  In this
normalisation the objective reduces to

    F(a) = 2·n·max_k (a * a)[k] ,

where “*” denotes the full (linear) convolution.

The non‑smooth max is approximated by a smooth log‑sum‑exp:
    W(b) = exp(α·(b / max(b) – 1))   (0 ≤ W ≤ 1)

For a fixed smoothing factor α the auxiliary weighted sum
    Φ(a) = Σ_k W(b)_k · b_k
has a simple gradient
    ∂Φ/∂a_i = 2·Σ_j W_{i+j}·a_j .
All convolutions are carried out with FFTs (O(n log n)) so each
iteration is cheap even for n≈6000.

The update is a multiplicative (exponential) step which automatically
preserves non‑negativity and after each step the vector is projected
onto the simplex (renormalised to unit sum).  The step size η decays
slowly during the run.

The routine runs for the allotted time, keeps the best sequence seen,
and finally rescales it so that every entry is ≤ 1000 (the evaluator
clips anyway).  The global ``height_sequence_1`` is used as a sensible
starting point; if it is unavailable a uniform sequence of length 5000
is used.

The algorithm frequently prints the current best metric so the progress
can be monitored.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: full convolution via FFT (real inputs)
def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return full linear convolution of x and y using a power‑of‑2 FFT."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[:n]
    return conv


# ----------------------------------------------------------------------
# Helper: random block perturbation (fallback / extra exploration)
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Simple O(n) perturbation used as a fallback and for extra exploration."""
    candidate = base.copy()
    n = candidate.size
    # width of the block varies between n/200 and n/12
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal perturbation on the whole block
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation helps a bit
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------

    The function runs for ``budget_s`` seconds (minus a tiny safety margin).

    of the max‑convolution objective, keeping the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Initialise the sequence ------------------------------------------------
    try:
        # ``height_sequence_1`` is provided by the surrounding harness
        base = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:                     # pragma: no cover
        base = None

    if base is None or base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        # fallback: uniform sequence of moderate length
        n = 5000
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = np.clip(base, 0.0, 1000.0)
        total = a.sum()
        if total < 1e-12:                # degenerate – replace by uniform

        else:
            a = a / total                  # normalise to unit L1‑norm

    n = a.shape[0]

    # direct metric (no clipping) – because sum(a) = 1, metric = 2·n·max(conv)

    best_seq = a.copy()

    # ------------------------------------------------------------------
    # Gradient‑descent parameters
    alpha = 10.0                 # smoothing strength for the soft‑max
    eta0 = 0.15                  # initial step size (multiplicative)
    iteration = 0

    while time.monotonic() < deadline:
        # ----- full convolution a * a ------------------------------------
        conv = _conv_full_fft(a, a)                # length 2n‑1
        max_conv = conv.max()
        metric = 2.0 * n * max_conv                # same metric used by the evaluator

        # keep the best
        if metric < best_metric:
            best_metric = metric
            best_seq = a.copy()
            print(f"Iter {iteration:6d} – new best metric: {best_metric:.6f}")

        # ----- smoothed weights (soft‑max) --------------------------------
        # w_k = exp(alpha * (b_k / max_b - 1))  ∈ (0,1]

            a *= jitter

        iteration += 1

        # ----------------------------------------------------------------
        # Small safety: if we have already reached the target we can stop
        if best_metric < 1.55:      # a little margin below the target 1.503

    # ------------------------------------------------------------------
    # Final polishing with the legacy block‑perturbation search (optional)
    # This runs only a tiny amount of time if anything is left.
    remaining = deadline - time.monotonic()
    if remaining > 2.0:
        parent = best_seq.copy()
        best_score = best_metric
        while time.monotonic() < deadline:

                best_score = score
                best_seq = cand.copy()

    # ------------------------------------------------------------------
    # Scale so that every entry is ≤ 1000 (the evaluator does the same)
    scale = min(1000.0 / best_seq.max(), 1.0)
    best_seq = best_seq * scale

    # Return as a plain Python list of floats
    return [float(x) for x in best_seq.tolist()]
