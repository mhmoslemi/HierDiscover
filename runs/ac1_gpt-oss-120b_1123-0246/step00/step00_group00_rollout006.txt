def propose_candidate(seed=42, budget_s=500, **kwargs):
    """
    Generate a step‐function (list of non‑negative heights) that seeks to minimize
    ``evaluate_sequence``. The algorithm starts from a symmetric power‑law “arcsine”
    family a_i ∝ ((i+δ)*(n‑1‑i+δ))^−α, explores a grid of (α,δ) parameters,
    refines the best candidate with random block perturbations and returns the
    best sequence found before the time budget expires.
    """
    import numpy as np, time
    # Setup RNG and deadline
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Choose length: start with the length of the known construction, possibly enlarge.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n0 = base.size
    except Exception:
        n0 = 256
    # Use at least 4800 items (the current construction size) but cap at 10000 for speed.
    n = max(4800, n0)
    n = min(10000, n)   # avoid too big to keep evaluation fast

    # Helper to produce symmetric power‑law shape
    def make_shape(alpha, delta):
        i = np.arange(n, dtype=np.float64)
        # shift to avoid zero denominator
        denom = (i + delta) * (n - 1 - i + delta)
        a = np.power(denom, -alpha)
        # Normalise L1 to 1 (scale invariant for the objective)
        s = a.sum()
        if s == 0:
            return a
        return a / s

    # Initialise with a simple uniform sequence (guaranteed finite score)
    best_seq = np.ones(n, dtype=np.float64) / n
    best_score = evaluate_sequence(best_seq.tolist())
    print(f'Initial uniform score: {best_score:.6f}')

    # ----- Grid search over the arcsine‑family -----
    alphas = np.linspace(0.10, 0.95, 18)       # 0.05 step roughly
    deltas = np.linspace(0.10, 2.00, 20)       # step 0.095
    for a in alphas:
        for d in deltas:
            cand = make_shape(a, d)
            sc = evaluate_sequence(cand.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f'Grid improvement: alpha={a:.3f} delta={d:.3f} score={sc:.6f}')
    # ----- Local random perturbations (simulated annealing style) -----
    while time.monotonic() < deadline:
        # Randomly pick a parameter set close to the current best
        a = best_seq.copy()
        # Choose a block width as a fraction of n
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        # Multiplicative perturbation (log‑normal)
        sigma = rng.random() * 0.3 + 0.05            # 0.05 … 0.35
        factor = np.exp(rng.normal(0.0, sigma, size=width))
        a[start:stop] *= factor
        # occasional symmetrisation
        if rng.random() < 0.3:
            a = 0.5 * (a + a[::-1])
        # re‑normalise to keep the scale invariant property
        tot = a.sum()
        if tot > 0:

        sc = evaluate_sequence(a.tolist())
        # accept only improvements (simple greedy hill‑climb)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a
            print(f'Local improvement: score={sc:.6f}')
        # Occasionally try a completely new random shape from the family
        if rng.random() < 0.02:

import time
import numpy as np

def propose_candidate(seed=42, budget_s=500, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.

    The main idea is to explore a two‑parameter family of symmetric
    power‑law shapes that are known to flatten the self‑convolution:

        a[i] ∝ ((i+δ)·(n‑1‑i+δ))⁻ᵅ ,

    where *α* controls how strongly the mass concentrates near the ends
    and *δ* regularises the singularity at the borders.  For each
    (α,δ) pair the sequence is normalised to an L¹‑norm of 1 (the
    objective is scale‑invariant).  A coarse grid over (α,δ) is scanned,
    then the best candidate is refined with random block perturbations
    (multiplicative log‑normal noise) and occasional symmetrisation.
    The search stops when the time budget expires and the best sequence
    found is returned as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # Setup RNG and time limit
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Determine the length of the construction
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n0 = base.size
    except Exception:
        n0 = 256                     # fallback length
    # Use at least the current size (≈4800) but cap to keep eval fast
    n = max(4800, n0)
    n = min(12000, n)                # safe upper bound for speed

    # ------------------------------------------------------------------
    # Helper: generate a symmetric power‑law shape
    # ------------------------------------------------------------------
    def make_shape(alpha, delta):
        """
        Return a length‑n non‑negative array:
            a[i] = ((i+δ)*(n‑1‑i+δ))⁻ᵅ   (unnormalised)
        The result is normalised to sum to 1.
        """
        i = np.arange(n, dtype=np.float64)
        denom = (i + delta) * (n - 1 - i + delta)
        # avoid zero division – delta is always >0
        a = np.power(denom, -alpha)
        total = a.sum()
        if total == 0:
            return a
        return a / total

    # ------------------------------------------------------------------
    # Initialise with a uniform sequence (guaranteed finite score)
    # ------------------------------------------------------------------
    best_seq = np.ones(n, dtype=np.float64) / n
    best_score = evaluate_sequence(best_seq.tolist())
    print(f'Initial uniform score: {best_score:.6f}')

    # ------------------------------------------------------------------
    # 1️⃣  Grid search over the (α,δ) family
    # ------------------------------------------------------------------
    alphas = np.linspace(0.10, 0.95, 18)      # 0.05‑ish steps
    deltas = np.linspace(0.10, 2.00, 20)      # roughly 0.1 steps

    for a in alphas:
        for d in deltas:
            cand = make_shape(a, d)
            sc = evaluate_sequence(cand.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f'Grid improve: α={a:.3f} δ={d:.3f} → {sc:.6f}')

    # ------------------------------------------------------------------
    # 2️⃣  Local random refinements (simple greedy hill‑climb)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- block multiplicative perturbation ----
        cand = best_seq.copy()
        # Choose a random block width (O(n) memory, as required)
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        # Log‑normal factor (σ tuned on the fly)
        sigma = rng.random() * 0.30 + 0.05          # 0.05 … 0.35
        factor = np.exp(rng.normal(0.0, sigma, size=width))
        cand[start:stop] *= factor

        # Occasionally enforce (soft) symmetry – helps keep the shape balanced
        if rng.random() < 0.30:
            cand = 0.5 * (cand + cand[::-1])

        # Re‑normalise to keep scale invariance (sum == 1)
        s = cand.sum()
        if s > 0:
            cand = cand / s

        sc = evaluate_sequence(cand.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand
            print(f'Local improve: {sc:.6f}')

        # ---- occasional full random draw from the family (exploration) ----
        if rng.random() < 0.02:
            a_rand = rng.uniform(0.10, 1.00)
            d_rand = rng.uniform(0.10, 2.00)
            cand2 = make_shape(a_rand, d_rand)
            sc2 = evaluate_sequence(cand2.tolist())
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = cand2
                print(f'Random family improve: α={a_rand:.3f} δ={d_rand:.3f} → {sc2:.6f}')

    # ------------------------------------------------------------------
    # Finish
    # ------------------------------------------------------------------
    print(f'Best score after search: {best_score:.6f}')
    # Convert to plain Python list of floats for the caller
    return [float(v) for v in best_seq]
