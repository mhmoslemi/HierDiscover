    start_lr = 0.2

    while time.monotonic() < deadline:

        L = 1 << ((2*n -1).bit_length())
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[:2*n-1]

        max_idx = np.argmax(conv)

        # gradient_i = 2*a[max_idx - i] if within range
        grad = np.zeros_like(a)
        # compute reversed index mapping vectorized:

        idx = max_idx - np.arange(n)

        grad[valid] = 2.0 * a[idx[valid]]
        # gradient of objective: dE/da = 2*n * grad - 4*n * max_val * 1 (since sum=1)

        # step size
        lr = start_lr * (decay**t)
        t += 1
        a = a - lr * grad_obj
        # clip and renormalize
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s == 0:
            a = np.ones(n) / n
        else:
            a = a / s
        # evaluate occasionally
        if t % 10 == 0:
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:

    rng = np.random.default_rng(seed)

    # Determine length
    try:

        if base.ndim!=1 or base.size==0 or not np.isfinite(base).all():

        n = base.size

    except Exception:
        n = 512
        init_seq = np.ones(n, dtype=np.float64)
    # Ensure sum=1
    def normalize(x):
        s = x.sum()
        if s <= 0:
            return np.ones_like(x)/x.size
        return x/s
    best = normalize(init_seq)
    best_score = evaluate_sequence(best.tolist())
    # Simple function to evaluate and maybe update best
    def maybe_update(candidate):
        nonlocal best, best_score
        score = evaluate_sequence(candidate.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = candidate.copy()
            print(f"New best score {best_score:.6f} at time {time.monotonic():.2f}")
    # Main loop
    iteration = 0
    while time.monotonic() < deadline:
        # random restart
        a = rng.random(n)
        a = normalize(a)
        # gradient descent for limited steps
        lr = 0.5
        beta = 5.0
        # maybe schedule for each iteration
        for _ in range(200):
            if time.monotonic() >= deadline:
                break
            # convolution via FFT
            L = 1 << ((2*n - 1).bit_length())
            A = np.fft.rfft(a, n=L)
            conv = np.fft.irfft(A * A, n=L)[:2*n-1]
            # smooth max
            max_val = conv.max()
            # compute softmax
            shifted = conv - max_val
            expw = np.exp(beta * shifted)
            w = expw / expw.sum()
            # grad_approx_max = 2 * convolve(a, w_rev)
            w_rev = w[::-1]
            grad_approx = 2.0 * np.convolve(a, w_rev)[(2*n-2) - np.arange(n)]
            # gradient of objective
            grad = 2.0 * n * grad_approx - 4.0 * n * max_val * np.ones_like(a)
            # update
            a = a - lr * grad
            # enforce bounds
            a = np.clip(a, 0.0, 1000.0)
            a = normalize(a)
            # occasional evaluation
            if iteration % 10 == 0:
                maybe_update(a)
            # anneal lr, beta
            lr = lr * 0.99
            beta = beta * 1.01

        maybe_update(a)

"""
propose_candidate
~~~~~~~~~~~~~~~~~

A two‑phase stochastic optimiser for the convolution‑based evaluation.

Phase 1 – Random restarts: initialise a probability vector (non‑negative,
sum = 1) from a uniform random draw and perform a few gradient‑descent
steps on a smooth approximation of the objective.
The smooth objective uses a soft‑max (log‑sum‑exp) of the self‑convolution,
which yields a well‑behaved gradient even though the true objective
contains a non‑smooth max.

Phase 2 – Local refinement: after the gradient steps a cheap block‑perturbation
(a multiplicative tweak of a random sub‑interval) is occasionally applied
to escape possible plateaus.  The exact ``evaluate_sequence`` function is
called periodically; the best sequence seen so far is stored and finally
returned.

All updates respect the non‑negativity constraint and are re‑normalised
to keep the ℓ₁‑norm equal to 1 (the evaluation is scale‑invariant).  The
implementation uses FFTs to obtain the convolution and the convolution with
the soft‑max weights in O(n log n) time, allowing thousands of iterations
within the 1000 s budget even for a few‑thousand‑element sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: block perturbation (same idea as the baseline but kept simple)
def _block_perturb(a, rng):
    """Multiplicative perturbation of a random contiguous block of `a`."""
    n = a.size
    # block width between 1% and 25% of the sequence length
    width = int(rng.integers(max(1, n // 100), max(2, n // 4)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.15)))      # modest log‑normal factor
    a_new = a.copy()
    a_new[start:stop] *= factor
    # occasional symmetry enforcement (helps reduce the convolution peak)
    if rng.random() < 0.25:
        a_new = 0.5 * (a_new + a_new[::-1])
    return np.clip(a_new, 0.0, 1000.0)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500.0, **kwargs):
    """Return a sequence of non‑negative heights that (hopefully) beats the
    current upper bound.  The function respects the time budget and keeps the
    best candidate seen during the search.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget in seconds (the caller subtracts a small
        constant before the deadline).
    """
    rng = np.random.default_rng(seed)
    # --------------------------------------------------------------
    # Determine problem size and starting vector
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
        a0 = np.clip(base, 0.0, 1000.0)
    except Exception:
        # fallback: a modest length if the global is unavailable
        n = 512
        a0 = np.ones(n, dtype=np.float64)

    # Normalise to ℓ₁‑norm = 1 (the evaluator is scale‑invariant)
    def _normalise(v):
        s = v.sum()
        if s <= 0.0:
            return np.ones_like(v) / v.size
        return v / s

    a0 = _normalise(a0)

    # --------------------------------------------------------------
    # Time handling
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------------
    # Book‑keeping
    best = a0.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Core optimisation loop
    # We perform several random restarts; each restart runs a short
    # gradient descent on a smoothed max‑convolution objective.
    # ------------------------------------------------------------------
    # Pre‑compute the FFT length needed for both the self‑convolution
    # (length 2n‑1) and the convolution with the soft‑max weights
    # (length n + (2n‑1) - 1 = 3n‑2).  Choose a power of two for speed.
    L = 1 << ((3 * n).bit_length())  # smallest power of two >= 3n

    # indices needed to extract the “shift‑by‑i+j” convolution
    base_idx = 2 * n - 2                     # centre index of a*w_rev convolution
    idx_range = base_idx - np.arange(n)      # shape (n,)

    # hyper‑parameters (tuned empirically)
    max_restarts = 30          # how many random starts we allow
    grad_iters   = 400         # gradient steps per restart
    lr_start     = 0.2         # initial learning rate
    beta_start   = 5.0         # initial soft‑max temperature
    lr_decay     = 0.997       # learning‑rate decay per inner iteration
    beta_inc     = 1.003       # beta increase per inner iteration

    restart = 0
    while time.monotonic() < deadline and restart < max_restarts:
        # -------------------------- random start ----------------------
        a = _normalise(rng.random(n).astype(np.float64) + 1e-12)
        lr   = lr_start
        beta = beta_start

        # ----------- inner gradient descent on smoothed max ----------
        for it in range(grad_iters):
            if time.monotonic() >= deadline:
                break

            # ----- self‑convolution via FFT (linear, not circular) -----
            Af = np.fft.rfft(a, n=L)
            conv_full = np.fft.irfft(Af * Af, n=L)[:2 * n - 1]   # size = 2n‑1
            max_val = conv_full.max()

            # ----- soft‑max (log‑sum‑exp) weights --------------------
            # shift by the maximum for numerical stability
            shifted = conv_full - max_val
            exp_w = np.exp(beta * shifted)
            w = exp_w / exp_w.sum()          # shape (2n‑1,)

            # ----- gradient of the smoothed maximum ------------------
            # grad_approx = 2 * (a convolved with w_rev) evaluated at needed indices
            w_rev = w[::-1]
            Wf = np.fft.rfft(w_rev, n=L)
            grad_conv = np.fft.irfft(Af * Wf, n=L)          # linear convolution a * w_rev
            grad_approx = 2.0 * grad_conv[idx_range]       # shape (n,)

            # ----- gradient of the original objective -----------------
            # Objective = 2*n * max_b / (sum_a^2); we keep sum_a = 1
            grad_obj = 2.0 * n * grad_approx - 4.0 * n * max_val * np.ones_like(a)

            # ----- update, projection & renormalisation ---------------
            a = a - lr * grad_obj
            a = np.clip(a, 0.0, 1000.0)
            a = _normalise(a)

            # ----- occasional exact evaluation ------------------------
            if (it & 31) == 0:          # every ~32 inner steps
                score = float(evaluate_sequence(a.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best = a.copy()
                    print(f"[improve] restart {restart}, it {it}, score {best_score:.6f}")

            # ----- schedule ------------------------------------------------
            lr   *= lr_decay
            beta *= beta_inc

        # ----- after gradient phase, try a small block perturbation -----
        # This gives a cheap way to hop out of shallow minima.
        a_pert = _block_perturb(best, rng)
        a_pert = _normalise(a_pert)
        score = float(evaluate_sequence(a_pert.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a_pert.copy()
            print(f"[perturb] restart {restart}, new best {best_score:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # Final sanity check – if nothing better than the start was found,
    # fall back to the (clipped) initial vector.
    # ------------------------------------------------------------------
    if not np.isfinite(best_score):
        best = _normalise(np.clip(a0, 0.0, 1000.0))
        print("[fallback] using initial sequence")

    print(f"[final] best score {best_score:.6f}, length {best.size}")
    return [float(x) for x in best]
