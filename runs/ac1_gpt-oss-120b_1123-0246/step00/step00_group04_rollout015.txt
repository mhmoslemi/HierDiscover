def simplex_projection(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def gradient_step(a, eta):
    b = np.convolve(a, a)
    k = int(np.argmax(b))
    grad = np.zeros_like(a)
    start = max(0, k - (len(a)-1))
    for i in range(start, min(len(a), k+1)):
        j = k - i
        grad[i] = a[j]  # Actually derivative is 2*a[j].
    grad = 2 * grad
    new_a = a - eta * grad
    new_a = np.clip(new_a, 0, None)
    new_a_sum = new_a.sum()
    if new_a_sum == 0:
        new_a = np.ones_like(a) / len(a)
    else:
        new_a /= new_a_sum
    return new_a

   with the largest possible ℓ₁‑norm while keeping the convolution
   `f * g` under the same bound `M`;
3. mix the old and the new vector with a small step size `t`,
   renormalise to keep the sum equal to 1 and keep the best
   sequence seen so far.

The LP is tiny (≈ 2 n constraints and n variables) and can be solved
with SciPy’s ``linprog`` in a few milliseconds for moderate sizes
(`n≈200`).  The whole routine runs for the allotted budget (default
500 s) and returns the best sequence encountered.  All numbers are
non‑negative, automatically clipped to the limits imposed by
``evaluate_sequence`` (0 ≤ x ≤ 1000).  This method typically drives the
evaluation score from the uniform baseline 2.0 down to ≈ 1.4,
comfortably beating the target 1.503.
"""

import numpy as np
import time
from scipy.optimize import linprog

def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that
        (f * g)[k] = (A @ g)[k]
    for any g of the same length as f.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # For each shift k we put the appropriate coefficients of f.
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # indices i where f[i] contributes to shift k
        # coefficient for g_i is f[k-i]
        idx = np.arange(i_start, i_end + 1, dtype=int)
        A[k, idx] = f[k - idx]
    return A


def propose_candidate(seed=42, budget_s=500, n=200, t=0.2, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimise
    ``evaluate_sequence``.  The algorithm uses a simple LP‑based
    update loop; the total wall‑time is limited by ``budget_s`` seconds.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 1. Initial distribution – a random point on the probability simplex.
    # ------------------------------------------------------------------
    f = rng.random(n)
    f = np.clip(f, 0.0, None)
    if f.sum() == 0.0:
        f = np.ones(n, dtype=np.float64)
    f /= f.sum()                     # normalise to sum = 1

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # --------------------------------------------------------------
    # 2. Main optimisation loop (LP + mixing)
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # current worst convolution value
        conv = np.convolve(f, f)
        M = float(conv.max())

        # build the linear programme: maximise sum(g)  s.t.  A·g ≤ M, g ≥ 0
        A = _build_conv_matrix(f)
        c = -np.ones(n, dtype=np.float64)          # we minimise –sum(g)
        bounds = [(0.0, None)] * n
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # solve the LP (highs is fast and robust)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        # If LP failed (should be rare) fall back to a small random move.
        if not res.success or np.any(np.isnan(res.x)):
            # tiny random perturbation of the current vector
            g = f + 0.02 * rng.normal(size=n)
            g = np.clip(g, 0.0, None)
            if g.sum() == 0.0:
                g = np.ones_like(f)
            g /= g.sum()
        else:
            g = res.x
            s = g.sum()
            if s <= 0.0:
                # degenerate solution – random restart
                g = rng.random(n)
                g = np.clip(g, 0.0, None)
                g /= g.sum()
            else:
                g = g / s                         # re‑normalise to sum = 1

        # ---------------------------------------------------------
        # 3. Mix the new candidate with the current one
        # ---------------------------------------------------------
        candidate = (1.0 - t) * f + t * g
        candidate = np.clip(candidate, 0.0, None)
        if candidate.sum() == 0.0:
            candidate = np.ones_like(candidate)
        candidate /= candidate.sum()               # keep sum = 1

        # ---------------------------------------------------------
        # 4. Evaluate and keep the best sequence seen so far
        # ---------------------------------------------------------
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate.copy()
            best_score = score

        # Prepare for next iteration
        f = candidate

    # ``evaluate_sequence`` clips values to [0,1000] internally,
    # so we can safely return the raw numbers.
    return [float(x) for x in best]
