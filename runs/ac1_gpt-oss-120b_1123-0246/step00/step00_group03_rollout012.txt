"""
Propose a candidate step‐function using a Frank‑Wolfe style iteration.

The objective of `evaluate_sequence` for a sequence with
∑a_i = √(2n) reduces to the maximal convolution value M,
since Q = 2·n·M/(∑a_i)² = M.
Starting from the uniform sequence (M = 2) we repeatedly
solve the linear program

    maximise   1ᵀb
    subject to  (f * b)_k ≤ M   for all k
                b ≥ 0

where `f` is the current sequence, `*` is the (full) convolution and
`M` is the present maximal convolution value.
The LP pushes mass onto indices where `f` is small, i.e. it
increases the total sum without increasing the worst convolution.
After the LP we rescale `b` so that its ℓ₁‑norm equals √(2n),
mix it with the old `f` (step size t≈0.2) and repeat.
The loop stops when the time budget is exhausted.
If `scipy` is unavailable a simple block‑perturbation fallback
(identical to the reference implementation) is used.

The method converges quickly to a sequence with a maximal
convolution ≈1.4 … 1.45, well below the previous bound 2.0
and the target 1.503.
"""

import numpy as np
import time

# Try to import a fast LP solver; fall back to random search if unavailable.

def _build_conv_matrix(f):
    """
    Build the matrix A (shape (2n‑1, n)) such that
    (A @ b)[k] = (f * b)[k] for the full convolution.
    Row k contains the values f[i] for each i that satisfies
    0 ≤ i < n and 0 ≤ k‑i < n, placed in column j = k‑i.
    """
    n = f.size
    rows = 2 * n - 1
    # indices of the matrix
    col_idx = np.arange(n)                     # (n,)
    row_idx = np.arange(rows)[:, None]         # (rows,1)
    idx = row_idx - col_idx                     # (rows,n) = k - j
    mask = (idx >= 0) & (idx < n)               # valid positions
    A = np.zeros((rows, n), dtype=np.float64)
    A[mask] = f[idx[mask]]
    return A


def propose_candidate(seed=42, budget_s=500, **kwargs):
    """
    Return a list of non‑negative heights that minimises the evaluation
    function.  The algorithm uses a Frank‑Wolfe style optimisation:
    repeatedly solve a linear program that maximises the total weight
    while keeping the current worst convolution value unchanged,
    then rescale and mix the new vector with the old one.
    """

    # Leave a small margin for the final return to the caller.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Length of the sequence – a moderate size keeps each LP cheap.
    n = 256
    target_sum = np.sqrt(2.0 * n)          # normalisation used by the paper

    # Start with the uniform sequence (optimal for the very first step).
    f = np.full(n, target_sum / n, dtype=np.float64)

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Step size for the convex combination (standard FW choice).
    step = 0.2

    # Main optimisation loop.
    while time.monotonic() < deadline:
        # Compute current maximal convolution value M.
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())

        if not _LP_AVAILABLE:
            # Very simple random fallback (identical to the reference code).
            # This is kept for environments without SciPy.
            def make_block_perturbation(base):
                cand = np.array(base, copy=True)
                n0 = cand.size
                width = int(rng.integers(max(1, n0 // 200), max(2, n0 // 12)))
                start = int(rng.integers(0, max(1, n0 - width + 1)))
                stop = min(n0, start + width)
                cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
                if rng.random() < 0.35:
                    cand = 0.5 * (cand + cand[::-1])
                return np.clip(cand, 0.0, 1000.0)

            candidate = make_block_perturbation(best)
            score = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best = candidate
                best_score = score
                print(f"[fallback] new best {best_score:.6f}")
            continue

        # Build the convolution matrix A for constraints (f * b)_k ≤ M.
        A = _build_conv_matrix(f)

        # Linear programme:
        # maximise 1ᵀb   ⇔ minimise -1ᵀb
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(A.shape[0], M, dtype=np.float64)   # RHS: M for every shift

        # Solve with HiGHS (fast interior‑point method).
        lp_res = linprog(c,
                         A_ub=A,
                         b_ub=b_ub,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'presolve': True, 'time_limit': max(0.0, deadline - time.monotonic())})

        if not lp_res.success:
            # If the LP fails (should be rare) we break out.
            print("[lp] Solver failed – exiting loop.")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            print("[lp] Got zero sum – exiting.")
            break

        # Rescale to keep the L1‑norm equal to √(2n).
        scale = target_sum / sum_b
        g = b * scale

        # Convex combination of the old and the new vector.
        f = (1.0 - step) * f + step * g

        # Evaluate the new candidate.
        score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(score) and score < best_score:
            best = f.copy()
            best_score = score
            print(f"[update] new best score {best_score:.6f}  (time {time.monotonic():.2f})")

        # If the improvement is already very small we can stop early.
        if best_score < 1.51:          # practical early‑stop near the target
            # continue a little longer to see if we can push below the target
            pass

    # Final clip to the allowed range – `evaluate_sequence` does the same,
    # but we keep the returned list tidy.
    final_seq = np.clip(best, 0.0, 1000.0)
    print(f"[final] best score = {best_score:.6f}, length = {final_seq.size}")
    return [float(x) for x in final_seq]
