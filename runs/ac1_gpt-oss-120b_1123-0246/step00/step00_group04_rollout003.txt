def gradient_step(seq, rng, step_factor):
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    max_idxs = np.where(conv == max_val)[0]
    grad = np.zeros_like(seq)
    for k in max_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)  # inclusive
        # accelerate using slice:
        # for i from i_start to i_end inclusive:
        # j = k - i
        # grad[i] += seq[j]
        # use vectorized:
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        grad[i_range] += seq[j_range]
    # Normalize gradient magnitude
    # candidate = seq - step_factor * grad
    cand = seq - step_factor * grad
    # Ensure nonnegative
    cand = np.clip(cand, 0.0, None)
    # Renorm to sum 1
    total = cand.sum()
    if total <= 1e-12:
        # reinitialize random
        cand = rng.dirichlet(np.ones(n))
    else:
        cand = cand / total

"""
Hybrid local‑search for the step‑function minimisation problem.

The algorithm maintains the best sequence found so far and repeatedly
generates new candidates by three different moves:

1. **Gradient‑descent step** – a sub‑gradient of the current maximal
   convolution value is computed and the sequence is moved away from it.
   The step size is annealed during the run.

2. **Random block perturbation** – the same perturbation used in the
   reference solution, but applied to the current best sequence.

3. **Random restart** – draws a fresh Dirichlet‑distributed sequence.

Every candidate is projected onto the simplex (non‑negative, sum = 1) and
occasionally symmetrised to keep the shape balanced.  The best
candidate (lowest `evaluate_sequence` value) is kept and periodically
reported.  The search stops after the allocated time budget (the caller
provides `budget_s`; 3 s are kept as a safety margin).

No external files are accessed; the only required global is
`height_sequence_1` (the current best construction) and the function
`evaluate_sequence` supplied by the runner.
"""

import time
import numpy as np


def make_block_perturbation(base, rng):
    """Return an O(n) memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # Choose a block width between n/200 and n/12 (inclusive)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # Multiply the block by a log‑normal factor
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # Optional mirroring to keep the sequence roughly symmetric
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # Clip to the range accepted by ``evaluate_sequence``
    return np.clip(candidate, 0.0, 1000.0)


def gradient_step(seq, rng, step_factor):
    """
    Perform a single sub‑gradient step that tries to reduce the
    current maximal autocorrelation value.

    The sub‑gradient of max_k (seq*seq)_k is the sum of the
    neighbouring entries that contribute to the active maxima.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    # Indices where the convolution attains its maximum
    max_idxs = np.where(conv == max_val)[0]

    # Build the sub‑gradient (one entry per position in ``seq``)
    grad = np.zeros_like(seq)
    for k in max_idxs:
        # i ranges where both i and k-i are valid indices
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)       # inclusive
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range
        grad[i_range] += seq[j_range]

    # Take a step opposite to the gradient
    candidate = seq - step_factor * grad
    candidate = np.clip(candidate, 0.0, None)

    # Re‑normalise to the simplex (sum = 1) – scaling does not affect
    # the objective because ``evaluate_sequence`` is homogeneous.
    total = candidate.sum()
    if total < 1e-12:
        # If we collapsed to zero, replace by a fresh random point
        candidate = rng.dirichlet(np.ones(n))
    else:
        candidate = candidate / total
    return candidate
