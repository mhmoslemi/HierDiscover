    while time.monotonic() < deadline:
        # compute convolution and max index

"""
A two‑pronged local search for the step‑function evaluation problem.

The objective is
    f(a) = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

with n = len(a) and a_i ≥ 0.  The function is scale‑invariant,
so the goal is to **reduce the maximum convolution entry** while
the ℓ₁‑norm grows.

Two complementary heuristics are used in each iteration:

1. **Spectral flattening** – the FFT of the sequence is forced to
   have a constant magnitude (the RMS value of the current spectrum)
   while preserving the phases.  In the frequency domain this
   corresponds to a constant power spectrum, which tends to make the
   time‑domain autocorrelation (the convolution) flatter.
   The resulting sequence is clipped to the allowed range.

2. **Gradient‑like ℓ₁‑increase** – the current maximal convolution
   index kₘₐₓ is located.  Increasing a coefficient a_i by δ adds
   a_i·a_{kₘₐₓ‑i} to that entry; decreasing it subtracts the same.
   By moving a little mass from indices that heavily contribute to the
   peak towards indices with zero contribution we can increase the
   total sum without raising the peak.  The step is derived from the
   sub‑gradient of the peak entry.

A cheap random block perturbation (the original method) is injected
occasionally to escape plateaus.

The algorithm runs until the supplied time budget expires and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def make_block_perturbation(base, rng):
    """Return one O(n) memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width: between n/200 and n/12 (inclusive)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring – helps to keep the convolution shape balanced
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def _spectral_flatten(seq, L, rng):
    """
    One spectral‑flattening step.
    The FFT magnitude is replaced by its RMS value, phases are kept,
    inverse transformed and the result is clipped to the admissible range.
    """
    # real FFT (rfft) works on real input and returns the positive‑frequency part
    A = np.fft.rfft(seq, n=L)
    rms = np.sqrt(np.mean(np.abs(A) ** 2))
    # keep the phase, enforce constant magnitude
    A_new = rms * np.exp(1j * np.angle(A))
    # back to time domain
    full = np.fft.irfft(A_new, n=L)
    # take the first n entries (the rest are zero‑padding artefacts)
    cand = full[: seq.size]
    # keep numbers inside the allowed range
    return np.clip(cand, 0.0, 1000.0)


def _gradient_l1_step(a, max_idx):
    """
    One small ℓ₁‑increasing step that does not increase the current
    maximum convolution entry (indexed by ``max_idx``).
    """
    n = a.size
    # Build the vector g_i = a_{max_idx - i} when the index is valid,
    # otherwise 0.  This is the sub‑gradient of the entry b[max_idx].
    g = np.zeros_like(a)
    start = max(0, max_idx - (n - 1))
    end = min(n - 1, max_idx)
    if start <= end:
        idx = np.arange(start, end + 1, dtype=int)
        g[idx] = a[max_idx - idx]

    sum_g = g.sum()
    sum_g2 = np.dot(g, g)

    # a tiny step size proportional to the typical magnitude of a
    mean_a = a.mean()
    alpha = 0.01 * max(mean_a, 1e-3)   # never zero

    if sum_g2 > 0.0:
        mu = alpha * sum_g / sum_g2     # chosen so that the peak does not grow
    else:
        mu = 0.0

    delta = alpha - mu * g
    cand = a + delta
    # enforce non‑negativity and the problem's upper bound
    return np.clip(cand, 0.0, 1000.0)

    list[float]
        The best sequence discovered within the time limit.
    """

    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Initialise the current candidate – try to use the provided global.
    # ------------------------------------------------------------------
