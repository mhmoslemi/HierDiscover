"""
Propose candidate sequence via a *projected gradient descent* on the
ℓ₂‑norm of the self‑convolution, mixed with occasional block‑wise random
perturbations.

The evaluation function is
    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

so lowering the maximal convolution value while keeping the total sum fixed
reduces the score.  For a constant sequence the score is exactly 2.  The
gradient of   L(a) = Σ_k (conv(a,a)[k])²   is
    ∇L = 4 * ( (a*a) * a )
where “*” denotes the discrete linear convolution.
This gradient is largest for entries near the centre of the sequence; a
projected gradient step therefore moves mass from the centre to the edges,
flattening the autocorrelation and typically lowering the score below 2.

The algorithm repeatedly:
    • computes the gradient step,
    • projects onto the feasible set (non‑negative, fixed ℓ₁‑norm),
    • evaluates the new sequence,
    • keeps the best one seen,
    • with a small probability performs a large random block perturbation
      (the heuristic used in the reference code) to escape possible local
      minima.

The loop runs until the allotted time budget is exhausted and finally returns
the best sequence discovered.
"""

import numpy as np
import time
from typing import List

# ----------------------------------------------------------------------
# Helper: a simple block perturbation (same idea as the reference code)
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return one O(n) memory non‑negative perturbation of ``base``."""
    candidate = np.array(base, copy=True, dtype=np.float64)
    n = candidate.size
    # block width is a fraction of the length, but at least 1 element
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative random factor (log‑normal perturbation)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # occasional symmetrisation – helps keep the autocorrelation balanced
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # enforce the problem bounds immediately
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500.0, **kwargs) -> List[float]:
    """Return a sequence of non‑negative heights that (hopefully) improves
    on the current best bound.

    Parameters
    ----------
    seed : int, optional
        Base seed for the internal RNG (default 42).
    budget_s : float, optional
        Time budget in seconds (the caller subtracts a few safety seconds).
    kwargs : dict
        Ignored – present only for compatibility with the caller.

    Returns
    -------
    List[float]
        The best sequence found before the time limit expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # initialise the working sequence -----------------------------------
    if 'height_sequence_1' in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if init.ndim != 1 or init.size == 0 or not np.isfinite(init).all():
            init = np.ones(256, dtype=np.float64)
    else:
        init = np.ones(256, dtype=np.float64)

    # keep the numbers inside the allowed range for the evaluator
    init = np.clip(init, 0.0, 1000.0)

    # Normalise to unit ℓ₁‑norm – scaling does not affect the score but
    # improves numerical stability of the gradient step.
    s = init.sum()
    if s == 0.0:
        init[:] = 1.0
        s = init.sum()
    cur = init / s
    cur_score = float(evaluate_sequence(cur.tolist()))

    best = cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # algorithmic parameters
    lr = 0.04                 # learning rate for the gradient step
    lr_decay = 0.9995         # slow exponential decay of the step size
    min_lr = 5e-5
    step_counter = 0
    perturb_interval = 30    # how often a large random jump is tried
    print_interval = 100

    # ------------------------------------------------------------------
    # main optimisation loop
    while time.monotonic() < deadline:
        step_counter += 1

        # --------------------------------------------------------------
        # choose between a gradient step and a random block jump
        if step_counter % perturb_interval == 0:
            # large stochastic move – helps escape flat regions
            cand = make_block_perturbation(cur, rng)
        else:
            # ---- projected gradient descent on  Σ conv² ----
            # 1) convolution a*a
            conv = np.convolve(cur, cur)                     # length 2n‑1
            # 2) convolution (a*a) * a  → triple convolution
            triple = np.convolve(conv, cur)                  # length 3n‑2
            # 3) extract the part that aligns with the original indices
            n = cur.size
            start = n - 1
            grad = 4.0 * triple[start:start + n]             # ∇L = 4*(a*a)*a
            # 4) projected gradient step
            cand = cur - lr * grad
            cand = np.maximum(cand, 0.0)                    # non‑negativity

            # 5) re‑normalise to unit ℓ₁‑norm (scale invariance of the score)
            tot = cand.sum()
            if tot > 0.0:
                cand /= tot
            else:
                # if everything vanished, revert to the previous point
                cand = cur.copy()

        # --------------------------------------------------------------
        # evaluate the candidate (the external security checks are inside)
        cand_score = float(evaluate_sequence(cand.tolist()))

        # keep the best ever seen
        if np.isfinite(cand_score) and cand_score < best_score:
            best = cand.copy()
            best_score = cand_score

        # accept the candidate only if it improves the current position
        # (simple hill‑climbing; the random jump occasionally creates a
        #  better basin which will then be climbed from)
        if np.isfinite(cand_score) and cand_score < cur_score:
            cur = cand
            cur_score = cand_score

        # adapt the learning rate – small increase on a successful step,
        # otherwise slowly decay.
        if cand_score < cur_score:
            lr = min(lr * 1.02, 0.20)
        else:
            lr = max(lr * lr_decay, min_lr)

        # --------------------------------------------------------------
        # optional progress output
        if step_counter % print_interval == 0:
            print(
                f"Step {step_counter:<6d} | best_score={best_score: .6f} "
                f"| cur_score={cur_score: .6f} | lr={lr:.6f}"
            )

    # ------------------------------------------------------------------
    # final post‑processing: ensure the output respects the hard limits
    best = np.clip(best, 0.0, 1000.0)

    # Return as a plain Python list of floats (as required by the evaluator)
    return [float(v) for v in best]
