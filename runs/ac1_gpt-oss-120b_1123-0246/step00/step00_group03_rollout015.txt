"""
propose_candidate
~~~~~~~~~~~~~~~~~

Search for a step‑function (non‑negative vector) that gives a low value of

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm starts from the analytically optimal “constant‑convolution”
sequence a_i = C(2i,i) / 4**i (the discrete version of (1‑z)^‑1/2).  For this
sequence the convolution is constant (=1) for the first *n* shifts and the
score is ≈ π/2 ≈ 1.57.  Small local perturbations (random scaling of short
blocks, additive Gaussian noise on individual entries) are then explored with a
simple simulated‑annealing / hill‑climbing loop.  Only modifications that
strictly improve the score (or, at a very low temperature, occasionally
accept a worse one) are kept, so the search quickly stagnates near a local
minimum that is usually well below the previous bound of 2.0.

The routine respects the 1000‑second budget, keeps all values within the
required bounds and returns the best sequence found as a plain Python list.
"""

import numpy as np
import time

def _constant_convolution_seq(n: int) -> np.ndarray:
    """
    Return the length‑*n* sequence a_i = binom(2i,i)/4**i.
    This sequence satisfies conv(a,a)[k] = 1 for k = 0 … n‑1,
    giving a perfectly flat autocorrelation on the first half
    of the convolution.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence: a_{i} = a_{i-1} * (2i-1) / (2i)
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a score
    below the target 1.503.  The function runs for roughly ``budget_s`` seconds
    (default 500 s) and uses a deterministic seed for reproducibility.
    """
    rng = np.random.default_rng(seed)
    # time budget (leave a small safety margin)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------

    #     constant‑convolution sequence.  If a previous construction is

    try:

    # build the deterministic start vector
    a = _constant_convolution_seq(base_len)

    # we keep the best sequence seen so far
    best = a.copy()

    # parameters for the perturbation schedule
    start_temp = 0.25               # initial acceptance temperature
