"""
A hybrid optimizer for the step‑function bound.

Phase 1 – Convex‑combination update:
    * Treat the current sequence `f` as a fixed coefficient vector.
    * Solve a linear program that finds a non‑negative vector `g`
      with the largest possible ℓ¹‑norm while keeping every entry of
      the convolution `f * g` ≤ the current maximum `M = max(f*f)`.
    * Rescale `g` to the same total sum as `f` and replace `f` by a
      convex combination `(1‑t) f + t g`.  This mimics the scheme
      described in the statement but is implemented directly with
      CVXPY and works very fast for a modest length (≈200).

Phase 2 – Random block perturbations:
    * Starting from the best vector found in Phase 1, perform cheap
      O(n) random “block” mutations (the same perturbation used in the
      reference code) to polish the solution.

The whole routine adapts its running time to the provided budget,
prints intermediate improvements and finally returns the best
sequence found.
"""

import time
import numpy as np
import cvxpy as cp
from scipy import sparse


def _conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build a (2n‑1) × n sparse matrix A such that A @ g  ==  np.convolve(f, g)
    (full convolution).  Each row i contains the coefficients f[j] for which
    the product f[j]·g[i‑j] participates in the i‑th entry of the convolution.
    """
    n = f.size
    rows, cols, data = [], [], []
    # i runs over convolution indices 0 … 2n‑2
    for i in range(2 * n - 1):
        j_start = max(0, i - (n - 1))
        j_end = min(i, n - 1)
        # the slice f[j_start : j_end+1] contributes to row i
        for j in range(j_start, j_end + 1):
            rows.append(i)
            cols.append(j)
            data.append(f[j])
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a sequence of non‑negative heights that (hopefully) beats the
    current bound of 2.0.  The implementation uses a short LP‑based
    optimisation phase followed by a cheap stochastic local‑search phase.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise a reasonably sized parent sequence.
    # ------------------------------------------------------------------
    INIT_N = 200                               # small enough for fast LP solves
    if "height_sequence_1" in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64)[:INIT_N]
    else:
        parent = np.ones(INIT_N, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(INIT_N, dtype=np.float64)

    f = parent.copy()
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 : LP‑driven convex updates.
    # ------------------------------------------------------------------
    lp_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.5
    iteration = 0
    while time.monotonic() < lp_deadline:
        iteration += 1
        n = f.size
        # current max of the full self‑convolution
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())

        # Build the convolution matrix for the constraints  A @ g ≤ M
        A = _conv_matrix(f)                     # (2n‑1) × n sparse matrix

        # Linear program: maximise sum(g) subject to A g ≤ M, g ≥ 0
        g = cp.Variable(n)
        constraints = [A @ g <= M, g >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                       verbose=False)
        except Exception as e:
            print("LP solve error:", e)
            break
        if g.value is None:
            print("LP failed to produce a solution.")
            break

        g_vec = np.maximum(0.0, np.asarray(g.value, dtype=np.float64))

        # Rescale g so that it has the same total mass as f.
        sum_f = float(f.sum())
        sum_g = float(g_vec.sum())
        if sum_g == 0.0:
            break
        g_vec *= (sum_f / sum_g)

        # Convex combination (t controls step size)
        t = 0.3
        new_f = (1.0 - t) * f + t * g_vec
        new_f = np.clip(new_f, 0.0, 1000.0)

        new_score = float(evaluate_sequence(new_f.tolist()))
        if np.isfinite(new_score) and new_score < best_score:
            best = new_f.copy()
            best_score = new_score
            f = new_f.copy()
            print(f"LP iter {iteration:3d}: improved score {best_score:.6f}")
        else:
            # occasional acceptance to escape flat regions
            if rng.random() < 0.08:
                f = new_f.copy()

    # ------------------------------------------------------------------
    # Phase 2 : cheap random block perturbations (the reference heuristic).
    # ------------------------------------------------------------------
    def _block_perturb(base: np.ndarray) -> np.ndarray:
        """Return a slight O(n) non‑negative perturbation of ``base``."""
        cand = np.asarray(base, dtype=np.float64).copy()
        n = cand.size
        # choose a random block length
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        # multiplicative log‑normal noise on the block
        cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
        # occasional mirror averaging (helps symmetry)
        if rng.random() < 0.35:
            cand = 0.5 * (cand + cand[::-1])
        return np.clip(cand, 0.0, 1000.0)

    while time.monotonic() < deadline:
        cand = _block_perturb(best)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"Local tweak: improved score {best_score:.6f}")

    print("Final best score:", best_score)
    return [float(x) for x in best]
