"""
Propose a better step‑function sequence.

The evaluation metric is   2·n·max(conv(a,a)) / (sum(a))**2 .
Since the metric is homogeneous of degree 0 we can work with a *probability*
vector p (p_i ≥ 0, Σp_i = 1).  Then the score is simply  

        score(p) = 2·n·max(p * p)                     (1)

where “*” denotes discrete (linear) convolution.
We minimise (1) by a projected exponentiated‑gradient descent that
pushes probability mass away from the indices that currently give the
largest convolution entries.  The gradient of the soft‑max of the
convolution is a weighted cross‑correlation; we approximate the soft‑max
by a hard ‑top‑k mask (k=3) which is cheap to compute.

Algorithm
---------

1. Initialise p from the given ``height_sequence_1`` (normalised to sum 1)
   or, if that is unavailable, from the uniform distribution.
2. Repeatedly (until the time budget expires)

   * compute c = p * p (full convolution);
   * select the K largest entries of c (K = 3) and build a mask w
     that is 1 at those positions and 0 elsewhere;
   * compute the gradient  g = 2· (w cross‑correlated with p) .
     The cross‑correlation is obtained as
       g = 2·np.convolve(w, p[::-1], mode='full')[n‑1 : n‑1+n]
   * update p with a multiplicative step
       p ← p * exp(‑η·g)            (η = learning‑rate)
   * add a small amount of isotropic noise to avoid symmetry traps,
     renormalise p to the simplex.
   * slowly decay η.

   The update is always on the probability simplex, hence the sum of
   the heights stays 1.  Finally we rescale p so that the largest entry
   is ≤ 1000 (the evaluator clips to the range [0,1000]).

3. Keep the best sequence seen (according to the exact ``evaluate_sequence``
   function) and return it.

The whole optimisation runs in O(n log n) per iteration (two 1‑D
convolutions) and comfortably fits into the 500 s budget for the typical
length (≈ 6000).  The routine prints periodic progress reports.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex Σv = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0 or not np.isfinite(s):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _score_from_prob(p: np.ndarray, n: int) -> float:
    """Exact score for a probability vector (sum(p)=1)."""
    # convolution p * p, then pick the maximum.
    c = np.convolve(p, p)          # length 2n‑1
    return 2.0 * n * c.max()


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """Search for a low‑score height sequence.

    Returns a list of floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # initialise from the globally supplied sequence if possible
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
    except Exception:
        base = None

    if base is None or base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
        n = 6000
        p = np.full(n, 1.0 / n, dtype=np.float64)          # uniform start
    else:
        n = base.size
        base = np.clip(base, 0.0, 1000.0)
        total = base.sum()
        if total <= 0.0:
            p = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            p = base / total                                 # normalise to a probability vector

    # tiny random perturbation to break symmetry
    p += 0.01 * rng.normal(size=n) * p
    p = _project_simplex(p)

    # bookkeeping of the best sequence seen
    best_seq = np.clip(p * 1000.0, 0.0, 1000.0)
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # hyper‑parameters
    eta = 0.7                 # initial learning rate
    decay = 0.9993            # multiplicative decay per iteration
    min_eta = 0.005
    top_k = 3                 # number of convolution entries that define the active set
    noise_scale = 0.004       # relative Gaussian noise added each iteration
    report_every = 200        # iterations between status prints

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    it = 0

    while time.monotonic() < deadline:
        # 1) convolution of the current distribution
        conv = np.convolve(p, p)                     # length 2n‑1

        # 2) build a mask that marks the K largest entries of conv
        if top_k >= conv.size:
            mask = np.ones_like(conv, dtype=np.float64)
        else:
            # partially sort to obtain the K largest values efficiently
            idx = np.argpartition(conv, -top_k)[-top_k:]
            mask = np.zeros_like(conv, dtype=np.float64)
            mask[idx] = 1.0

        # 3) gradient = 2 * cross‑correlation of mask with p
        #    (cross‑correlation can be obtained via a convolution with reversed p)
        grad = 2.0 * np.convolve(mask, p[::-1], mode='full')[n - 1 : n - 1 + n]

        # 4) multiplicative update on the simplex (exponentiated gradient)
        #    shift the gradient to be non‑negative to avoid overflow in exp()
        g_shift = grad - grad.min()
        p = p * np.exp(-eta * g_shift)

        # 5) inject a small amount of isotropic noise and re‑normalise
        if (it + 1) % 12 == 0:  # occasional stronger perturbation
            noise = noise_scale * rng.normal(size=n) * p
            p = p + noise
        p = _project_simplex(p)

        # 6) decay learning rate
        eta = max(eta * decay, min_eta)

        # 7) evaluate occasionally (costly) and keep the best sequence
        if (it + 1) % report_every == 0:
            # we can compute the exact score cheaply from the convolution we already have
            cur_score = 2.0 * n * conv.max()
            # but we still call the official evaluator to respect clipping etc.
            seq = np.clip(p * 1000.0, 0.0, 1000.0)
            eval_score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(eval_score) and eval_score < best_score:
                best_score = eval_score
                best_seq = seq.copy()
                print(f"[{it+1:6d}] improved score: {best_score:.6f}")
            else:
                # keep track of the analytical score just for information
                print(f"[{it+1:6d}] current score≈{cur_score:.6f} (official {eval_score:.6f})")
        it += 1

    # final check – evaluate the best candidate one last time
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[done] iterations={it}, best score={final_score:.6f}")

    # return as a plain Python list of floats
    return [float(x) for x in final_seq]
