def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    # compute convolution f * g with zero padding. We can treat it manually:
    for k in range(2*n - 1):
        # sum_{i} f_i * g_{k-i} where 0 <= i < n and 0 <= k-i < n
        # Determine i range:
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        # Build linear combination:
        # sum_{i=i_start}^{i_end} f[i] * g[k-i]
        # So we can make expression using cp.sum(cp.multiply(...)).
        # We'll construct list of f[i] * g[ index ]
        # Use cp.multiply(f[i], g[k-i]) This is scalar.
        expr = cp.sum(cp.multiply(f[i_start:i_end+1], g[k - np.arange(i_start, i_end+1)]))
        constraints.append(expr <= M)
    # objective
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)

"""
Propose a candidate step‑function sequence that minimizes the supplied
`evaluate_sequence` metric.

The algorithm combines three ideas:

1. **LP‑based improvement** – given a current sequence `f`,
   solve the linear program  

        maximise   Σ g_j
        subject to  g_j ≥ 0
                    (f * g)_k ≤ M   for all k

   where `M = max_k (f * f)_k`.  The LP pushes mass into positions that are
   *under‑utilised* by the convolution, thereby lowering the worst convolution
   value.  The solution `g` is renormalised to `∑ g = √(2n)` and mixed with the
   current vector (`f ← (1‑t) f + t·g`).  Repeating this step converges to a
   lower bound.

2. **Local random perturbations** – occasional block‑wise log‑normal
   scaling (as in the reference code) to escape local minima.

3. **Simple reshaping** – clipping to the allowed range,
   trimming leading/trailing zeros (shortening `n` improves the objective)
   and falling back to a pure random search if `cvxpy` is unavailable.

The routine respects the time budget (`budget_s`) and returns the best
sequence found before the deadline.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: block perturbation (identical to the reference implementation)
def make_block_perturbation(base, rng):
    """Return one O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # width between n/200 and n/12  (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal factor
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetrisation (helps a lot on many runs)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

# ----------------------------------------------------------------------
# Helper: a very light‑weight random shuffle of the whole vector
def random_shuffle(base, rng):
    """Return a copy of ``base`` with its entries randomly permuted."""
    perm = rng.permutation(base.size)
    return base[perm]

# ----------------------------------------------------------------------
# Helper: solve the LP described in the docstring.
# Falls back to ``None`` if the solve fails (e.g. cvxpy missing).
def solve_lp(f, M):
    """
    Solve  maximise Σ g_j   s.t.   g≥0,  (f*g)_k ≤ M  for all k.
    Returns a NumPy array `g` of the same length as `f` or ``None``.
    """
    try:
        import cvxpy as cp
    except Exception:      # cvxpy not available – give up on the LP
        return None

    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = []

    # build the (2n‑1) convolution constraints  Σ_i f_i g_{k-i} ≤ M
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # indices of g that pair with f[i_start:i_end]
        idx = k - np.arange(i_start, i_end + 1)
        # linear expression: Σ f[i] * g[idx]
        expr = cp.sum(cp.multiply(f[i_start:i_end + 1], g[idx]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is a fast interior‑point solver suitable for moderate LPs.
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-6, reltol=1e-6, feastol=1e-6, verbose=False)
    except Exception:
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    return np.array(g.value).flatten()

def propose_candidate(seed=42, budget_s=500, **kwargs):
    """Search for a low‑score step‑function and return it as a Python list.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget (seconds).  The function stops a few seconds before the
        deadline to avoid overrunning the allotted time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.5)

    # ------------------------------------------------------------------
    # Initialise a reasonable starting vector.
    # Prefer the globally supplied `height_sequence_1`; if it is too long,
    # take a random contiguous slice, otherwise pad with zeros.
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
                raise ValueError
        except Exception:
            base = None
    else:
        base = None

    # choose a modest working length (200‑500).  Larger lengths are slower.
    TARGET_N = 256   # a safe middle ground between speed and expressiveness
    if base is not None:
        if base.size > TARGET_N:
            start = rng.integers(0, base.size - TARGET_N + 1)
            f = base[start:start + TARGET_N].copy()
        else:
            f = np.pad(base, (0, TARGET_N - base.size), constant_values=0.0)
    else:
        # random exponential start – ensures positivity
        f = rng.exponential(scale=1.0, size=TARGET_N)

    # keep values in the permitted interval and normalise the L1‑norm to √(2n)
    f = np.clip(f, 0.0, 1000.0)
    target_sum = np.sqrt(2.0 * f.size)
    s = f.sum()
    if s > 0.0:
        f = f * (target_sum / s)
    else:
        f[0] = target_sum

    # ------------------------------------------------------------------
    # Record the current best solution.
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Very small helper to trim zero padding – reduces n and improves objective.
    def trim_zeros(arr):
        nz = np.where(arr > 1e-12)[0]
        if nz.size == 0:
            return arr
        return arr[nz[0]:nz[-1] + 1]

    iter_no = 0
    while time.monotonic() < deadline:
        iter_no += 1

        # ------------------------------------------------------------------
        # Occasionally perform a cheap random move to increase diversity.
        if rng.random() < 0.12:          # 12 % of the iterations
            cand = make_block_perturbation(f, rng)
        else:
            # --------------------------------------------------------------
            # Main LP‑based improvement step.
            #  1) compute current worst convolution value M
            conv_f = np.convolve(f, f)
            M = conv_f.max()

            #  2) solve the LP for a better partner g
            g = solve_lp(f, M)

            if g is None or not np.all(np.isfinite(g)):
                # Fallback to a cheap perturbation when the LP fails
                cand = make_block_perturbation(f, rng)
            else:
                # Renormalise g to have the “canonical” L1‑norm √(2n)
                target_sum = np.sqrt(2.0 * g.size)
                sum_g = g.sum()
                if sum_g > 0.0:
                    g = g * (target_sum / sum_g)
                else:
                    g[0] = target_sum

                # Convex combination with the current vector
                t = 0.18                     # mixing coefficient
                cand = (1.0 - t) * f + t * g

        # ------------------------------------------------------------------
        # Clean up the candidate (clip, trim, ensure non‑empty)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = trim_zeros(cand)
        if cand.size == 0:
            # Guard against accidental total zero vector
            cand = np.array([target_sum], dtype=np.float64)

        # ------------------------------------------------------------------
        # Evaluate
        score = float(evaluate_sequence(cand.tolist()))
        # keep the best we have ever seen
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            f = cand.copy()               # continue search from the new elite
            # re‑normalise f to keep the L1‑norm roughly constant (helps LP)
            target_sum = np.sqrt(2.0 * f.size)
            s = f.sum()
            if s > 0.0:
                f = f * (target_sum / s)

            print(
                f"[Iter {iter_no:4d}] New best = {best_score:.6f} "
                f"(len={len(best)})"
            )
        else:
            # Occasionally replace the working vector with a random shuffle
            # of the current best to shake things up.
            if rng.random() < 0.03:
                f = random_shuffle(best, rng)

    # ----------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python floats).
    return [float(x) for x in best]
