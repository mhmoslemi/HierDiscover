def lp_optimize(f, t=0.5, max_iter=20):
    n = len(f)
    sum_f = f.sum()
    M = np.max(np.convolve(f, f))
    for _ in range(max_iter):

        # for each k, A[k,i] = f[k-i] if 0 <= k-i < n else 0.
        # Since the matrix is Toeplitz with f flipped
        # We can use scipy.linalg.toeplitz for full? but maybe simpler: build using np.array.
        # Use cvxpy.
        g = cp.Variable(n, nonneg=True)
        constraints = []
        # compute conv_f_g using Toeplitz matrix product: A @ g <= M
        # Use for each k:
        for k in range(2*n-1):
            # compute row as f_rev shifted

            # sum_i f[i] * g[k-i] <= M.
            # In terms of g, coefficient for g[j] is f[k-j] if 0 <= k-j < n.
            # So row = f[k - np.arange(start, stop+1)] at appropriate positions
        # Actually easier: use convolution expression: cp.conv(...) not available.
        # Use linear expression building:
        for k in range(2*n-1):

            coeff_f = f[k - idx]   # same length as idx
            constraints.append(cp.sum(cp.multiply(coeff_f, g[idx])) <= M)
        # objective
        obj = cp.Maximize(cp.sum(g))

        prob.solve(solver=cp.ECOS, warm_start=True)

"""
Propose a candidate step‑function sequence.

The algorithm mixes two complementary strategies:

1. **LP‑based diffusion** – Starting from a random non‑negative
   vector `f` (length n≈128–256) we repeatedly solve the linear program

        max   sum(g)
        s.t.  f * g  ≤  M   (pointwise)   ,   g ≥ 0

   where `M = max(f*f)` and `*` denotes the ordinary (non‑circular)
   convolution.  The optimal `g` has a larger ℓ₁‑norm while respecting the
   same convolution bound, so a convex combination
   `f ← (1‑t)·f + t·ĝ` (with `ĝ` scaled to the same total mass as `f`)
   strictly reduces the maximum convolution value.
   Repeating this step quickly lowers the evaluation score.

2. **Stochastic refinement** – Between LP steps we perform cheap
   “block‑perturbations’’ (multiplicative scaling of a random contiguous
   block) and occasional symmetrisation.  These moves help escape
   occasional stalls of the deterministic LP updates.

The whole search runs for the allotted wall‑time (default 990 s) and
keeps the best sequence seen.  If `cvxpy` (or a suitable solver) is not
available the algorithm falls back to pure stochastic search, which still
improves upon the naïve uniform construction.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: cheap block‑perturbation (the same as in the reference code)
# ----------------------------------------------------------------------
def make_block_perturbation(base, rng):
    """Return a single O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width: between 0.5% and ~8% of n
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal scaling of the block
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional symmetric averaging (helps the convolution shape)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Optional: linear‑programming step (requires cvxpy)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _HAS_CVXPY = True
except Exception:      # pragma: no cover
    _HAS_CVXPY = False


def _lp_optimize(f_vec, M, t=0.5):
    """
    Perform a single LP diffusion step.

    Parameters
    ----------
    f_vec : np.ndarray, shape (n,)
        Current (non‑negative) sequence.
    M : float
        Current maximum of the self‑convolution of ``f_vec``.
    t : float, optional
        Mixing coefficient for the update (0 < t ≤ 1).

    Returns
    -------
    new_f : np.ndarray
        Updated sequence after mixing with the LP solution.
    success : bool
        Whether the LP solved to optimality.
    """
    n = f_vec.shape[0]
    g = cp.Variable(n, nonneg=True)

    # Build the convolution constraints: (f * g)[k] ≤ M  for all k
    constraints = []
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        coeff = f_vec[k - idx]            # f[k‑i] for the admissible i
        # linear expression coeff·g[idx] ≤ M
        constraints.append(cp.sum(cp.multiply(coeff, g[idx])) <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS is usually the fastest built‑in solver for dense LPs
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   verbose=False)
    except Exception:      # pragma: no cover
        return f_vec, False

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return f_vec, False

    g_val = np.asarray(g.value).flatten()
    sum_f = f_vec.sum()
    sum_g = g_val.sum()
    if sum_g == 0.0:
        return f_vec, False

    # Scale g to have the same ℓ₁‑mass as f before mixing
    g_scaled = g_val * (sum_f / sum_g)
    new_f = (1.0 - t) * f_vec + t * g_scaled
    # Enforce the evaluator's hard limits
    new_f = np.clip(new_f, 0.0, 1000.0)
    return new_f, True

# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=990, **kwargs):
    """
    Search for a low‑score sequence.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Wall‑time budget in seconds (the function will stop ~1 s
        before this deadline to avoid time‑outs).

    Returns
    -------
    list of float
        The best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # Choose a manageable problem size.  The exact length is free;
    # short vectors converge faster while still allowing scores < 2.
    # ------------------------------------------------------------------
    n = 256
    # If the global starting point is long we truncate it (still safe).
    if 'height_sequence_1' in globals():
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.ndim == 1 and hs.size >= n:
            parent = np.clip(hs[:n], 0.0, 1000.0)
        else:
            parent = np.ones(n, dtype=np.float64)
    else:
        parent = np.ones(n, dtype=np.float64)

    # Randomise the initial vector a little (pure uniform is a fixed point)
    parent = parent * rng.random(n) + 0.1
    parent = np.clip(parent, 0.0, 1000.0)
    # Normalise to a convenient ℓ₁‑mass (the exact value does not
    # affect the evaluation because of scaling invariance).
    parent = parent / max(parent.sum(), 1e-12)

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    start_time = time.monotonic()
    it = 0

    # ------------------------------------------------------------------
    # Main search loop.  We mix cheap stochastic moves with the LP
    # diffusion step whenever enough time is left.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        # Compute the current self‑convolution and its maximum.
        conv_self = np.convolve(best, best)
        M = conv_self.max()

        # --------------------------------------------------------------
        # Try the LP diffusion step when the solver is available.
        # --------------------------------------------------------------
        if _HAS_CVXPY and (deadline - time.monotonic()) > 2.0:
            # The LP may be a bit expensive – cap its use to the early phase.
            new_f, ok = _lp_optimize(best, M, t=0.5)
            if ok:
                candidate = new_f
            else:
                # fallback to a cheap perturbation if LP failed
                candidate = make_block_perturbation(best, rng)
        else:
            # Pure stochastic move
            candidate = make_block_perturbation(best, rng)

        # Occasionally enforce symmetry – this often lowers the central
        # convolution peak.
        if rng.random() < 0.15:
            candidate = 0.5 * (candidate + candidate[::-1])

        # Evaluate the candidate
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            print(f"[{time.monotonic() - start_time:.1f}s] "
                  f"Iter {it:04d} – new best = {best_score:.6f}")

        # Occasionally re‑seed from a fresh random vector to avoid getting
        # stuck in a flat region.
        if rng.random() < 0.02:
            fresh = rng.random(n)
            fresh = fresh / fresh.sum()
            fresh_score = float(evaluate_sequence(fresh.tolist()))
            if fresh_score < best_score:
                best = fresh
                best_score = fresh_score
                print(f"[{time.monotonic() - start_time:.1f}s] "
                      f"Iter {it:04d} – random restart = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence seen (as a plain Python list of floats).
    # ------------------------------------------------------------------
    return [float(x) for x in best]
