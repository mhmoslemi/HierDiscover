"""
Propose a good step‐function using the optimal Catalan‑type start and a
simple convex‑programming refinement.

The sequence a_i = C_i = binom(2i,i)/4**i (i≥0) satisfies
    (a * a)_k = 1   for k < len(a)
so the score 2·n·max(b)/sum(a)^2 is ≈ π/2 ≈ 1.57, already better than the
uniform “all‑ones’’ construction (score = 2).  

To push the score lower we repeatedly solve a linear program

    max   Σ g_i
    s.t.  (a * g)_k ≤ M   for all k
          g_i ≥ 0

where M = max(a*a).  The LP finds a non‑negative vector g that can be added to
a without increasing the current worst convolution value M.  Mixing
a←(1‑t)·a + t·g (with a small step size t) strictly reduces the maximal
convolution and therefore the evaluation score.  The loop runs until the
time budget expires, keeping the best sequence seen.

The implementation uses CVXPY (ECOS) for the LP and falls back to a cheap
random perturbation when the LP does not improve the score.
"""

import time
import numpy as np
import cvxpy as cp

def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan‑type coefficients C_i = binom(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence a[i+1] = a[i] * (2*i+1)/(2*i+2)
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return np.clip(a, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence`.  The routine starts from the Catalan sequence,
    improves it with a linear‑programming based mixing step and finally
    returns the best sequence discovered before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)

    # ------------------------------------------------------------------
    # 1. initialise the starting sequence
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)  # provided globally
        n = base.size
    except Exception:
        # fallback length if the global is missing
        n = 256
        base = None

    # Catalan start (good deterministic baseline)
    a = _catalan_sequence(n)

    # possibly keep the old best if it is already better
    if base is not None and base.size == n:
        try:
            score_base = float(evaluate_sequence(base.tolist()))
            score_catalan = float(evaluate_sequence(a.tolist()))
            if score_base < score_catalan:
                a = base.copy()
                best_score = score_base
                print(f"Using provided sequence as start, score={best_score:.6f}")
            else:
                best_score = score_catalan
        except Exception:
            best_score = float(evaluate_sequence(a.tolist()))
    else:
        best_score = float(evaluate_sequence(a.tolist()))

    best = a.copy()
    print(f"Initial score: {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 2. iterative improvement: LP → mixing → evaluation
    # ------------------------------------------------------------------
    iteration = 0
    # we keep a tiny epsilon to avoid numerical tightness in the LP
    eps = 1e-9
    # maximal number of LP solves (speed / safety guard)
    max_lp_iters = 12

    while time.monotonic() < deadline and iteration < max_lp_iters:
        # current convolution and its worst value
        conv_self = np.convolve(best, best)
        M = float(conv_self.max())

        # --------------------------------------------------------------
        # 2a. Solve the LP: maximise Σ g_i   s.t.   (best * g)_k ≤ M
        # --------------------------------------------------------------
        g = cp.Variable(n, nonneg=True)
        # cvxpy's `conv` atom implements linear convolution
        conv_expr = cp.conv(best, g)          # length 2n‑1
        constraints = [conv_expr <= M + eps]

        if g.value is None:

            break

        g_val = np.maximum(0.0, np.asarray(g.value, dtype=np.float64))

        improved = False

    # ------------------------------------------------------------------

    return [float(v) for v in best]
