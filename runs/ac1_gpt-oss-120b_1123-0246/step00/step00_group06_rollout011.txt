def propose_candidate(seed=42, budget_s=500, **kwargs):

    rng = np.random.default_rng(seed)
    # Determine length n
    # Use length such that n moderate; we can set n=200 or 500 depending.
    # Starting sequence: uniform on n -> np.ones(n)/n (or scaled to sum = sqrt(2n) maybe not needed)
    # Actually evaluate_sequence normalizes using raw values, not probabilities. But scaling doesn't matter.

    f = np.ones(n, dtype=np.float64)  # uniform
    # Apply scaling to have sum sqrt(2n) for convenience but not required.
    # We'll keep raw values.

    # Or better, use a random start
    # Use height_sequence_1 if exists with length > n; else fallback.

    # We'll try multiple restarts with different n.

    start_time = time.monotonic()
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    # Loop while time remains
    while time.monotonic() - start_time < budget_s - some_margin:
        # Compute max conv
        c = np.convolve(best_f, best_f)
        M = np.max(c)
        # Build matrix A
        n = best_f.size
        rows = 2*n-1
        # Build A as dense for simplicity (maybe memory heavy for n up to ~300)
        A = np.zeros((rows, n), dtype=np.float64)
        for k in range(rows):
            # Construct w_k: f_{k-i}
            # Determine valid i range
            i_min = max(0, k-(n-1))
            i_max = min(k, n-1)
            # For i from i_min to i_max: A[k, i] = f[k-i]
            # f[k-i] = f_j where j = k-i
            # So j = k - i, valid.
            # We'll fill slice:
            # say idx_i = slice(i_min, i_max+1)
            # idx_j = k - idx_i
            A[k, i_min:i_max+1] = best_f[k - np.arange(i_min, i_max+1)]
        # Solve LP:
        import cvxpy as cp
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=1000)
        if g.value is None:
            break
        g_vec = np.maximum(g.value, 0)  # ensure nonneg
        sum_f = best_f.sum()
        sum_g = g_vec.sum()
        if sum_g <= 0:
            break
        scale = sum_f / sum_g
        g_scaled = g_vec * scale
        # Update via convex combination with t
        t = 0.5 # maybe 0.5 for faster converge
        new_f = (1 - t) * best_f + t * g_scaled
        new_score = evaluate_sequence(new_f.tolist())
        if new_score < best_score:
            best_f = new_f
            best_score = new_score
        else:
            # if no improvement, maybe reduce t or break
            # fallback to small random perturb
            new_f = best_f + 0.01 * (rng.normal(size=n))
            new_f = np.clip(new_f, 0, None)
            new_score = evaluate_sequence(new_f.tolist())
            if new_score < best_score:
                best_f = new_f
                best_score = new_score

    return best_f.tolist()

def solve_lp_max_g(f, M):
    n = f.size
    rows = 2 * n - 1
    # Build A
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        # indices i = i_min..i_max
        # j = k - i
        A[k, i_min:i_max+1] = f[k - np.arange(i_min, i_max+1)]
    # Solve LP:
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

def build_constraint_matrix(f):
    n = f.shape[0]
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # indices i where f[k-i] exists
        idx_i = np.arange(i_min, i_max + 1)
        A[k, idx_i] = f[k - idx_i]
    return A

def solve_max_sum_under_constraint(f, M):
    # return g vector
    n = f.shape[0]
    A = build_constraint_matrix(f)
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    else:

"""
Propose a step‐function sequence with a low evaluation score.

The objective to minimise is

    R(a) = 2·n·max_k (a * a)_k / (sum_i a_i)^2 ,

where “*” is the ordinary discrete convolution.
Writing p_i = a_i / sum_j a_j (a probability vector) we have

    R = 2·n·max_k (p * p)_k .

Thus the problem is to choose a probability vector p on {0,…,n‑1} that makes its
self‑convolution as flat as possible.  The algorithm below implements the
iterative linear‑programming scheme described in the statement:

1.  Start from the uniform distribution (which gives R = 2).
2.  Let M = max_k (f * f)_k for the current vector f.
3.  Solve the linear programme

        maximise   Σ_j g_j
        subject to A·g ≤ M ,   g ≥ 0

   where A is the (2n‑1)×n convolution matrix of f
   (row k contains the coefficients of (f * g)_k).
   The LP finds a non‑negative vector g with the largest possible ℓ₁‑norm
   that still respects the current peak M.
4.  Scale g so that Σ g = Σ f and replace f by a small convex combination

        f ← (1‑t)·f + t·g_scaled

   with a modest step size t (adapted during the run).  This strictly reduces
   the peak M unless the optimum has already been reached.
5.  Repeat until the time budget expires or the target score is achieved.

The LP is solved with SciPy’s high‑performance “highs” solver.
All numbers are kept non‑negative and clipped to the limits imposed by the
evaluation function (0 ≤ x ≤ 1000).  The routine prints occasional progress
messages and finally returns the best sequence found.
"""

import time
import math
import numpy as np

# SciPy is used for the linear programme; fall back to CVXPY if unavailable.
try:
    from scipy.optimize import linprog
    _SOLVER = "scipy"
except Exception:  # pragma: no cover
    import cvxpy as cp
    _SOLVER = "cvxpy"


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the (2·n‑1)×n matrix A such that (f * g) = A·g."""
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        idx = np.arange(i_min, i_max + 1, dtype=np.int64)
        # f[k‑i] is defined for i in idx
        A[k, idx] = f[k - idx]
    return A


def _lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_j   s.t.   A·g ≤ M , g ≥ 0,
    where A is the convolution matrix of f.
    Returns the optimal g (as a NumPy array) or None on failure.
    """
    n = f.size
    A = _build_conv_matrix(f)
    b = np.full(A.shape[0], M, dtype=np.float64)

    if _SOLVER == "scipy":
        c = -np.ones(n, dtype=np.float64)          # minimise –Σ g_j
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if res.success:
            g = np.maximum(res.x, 0.0)
            return g
        return None
    else:  # pragma: no cover
        g = cp.Variable(n, nonneg=True)
        prob = cp.Problem(cp.Maximize(cp.sum(g)),
                          [A @ g <= b])
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        if g.value is not None:
            return np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        return None

    """
