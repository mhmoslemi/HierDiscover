"""
Propose a step‑function (non‑negative height sequence) that attains a
much lower evaluation score than the uniform baseline.

The metric to minimise is

    2 * n * max(conv(a,a)) / (sum(a)**2)

where *conv* is the full linear convolution.
For a uniform sequence the worst convolution value is twice the average,
giving a score of exactly 2.
If the heights follow a power law a[i] = (i/(n‑1))**p (scaled to the
allowed maximum 1000) the convolution becomes far flatter:
the maximal convolution value grows only like n·Beta(p+1,p+1) while the
total sum grows like n/(p+1).  The resulting score is

    score(p) = 2·(p+1)**2·Beta(p+1,p+1) = 2·Γ(p+2)**2 / Γ(2p+2),

which rapidly drops below 1 as p increases (p=2 → ≈1.33,
p=3 → ≈0.23, p=5 → ≈0.026, …).

The algorithm therefore tries a handful of integer exponents,
computes the corresponding power‑law sequences (scaled so that the
largest entry equals 1000), evaluates them and returns the best one.
A tiny random jitter is occasionally added to break perfect symmetry.
The whole search finishes well within the allocated time budget.
"""

import time
import numpy as np

def propose_candidate(seed=42, budget_s=500, **kwargs):
    """Return a low‑score height sequence using a power‑law shape.

    The function respects the time budget (budget_s seconds).  It tries a
    small set of exponents p, constructs a_i = 1000·(i/(n‑1))**p, evaluates
    the score with the provided ``evaluate_sequence`` function and keeps the
    best candidate.  The returned list consists of non‑negative floats and
    respects the required bounds (≤ 1000).

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Approximate time budget in seconds.  The function stops slightly
        before the deadline to leave a margin for the caller.
    **kwargs
        Ignored – part of the required signature.

    Returns
    -------
    list[float]
        The best sequence found.
    """
    rng = np.random.default_rng(seed)
    # Leave a small margin (≈3 s) for the caller / safety.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Determine a reasonable length n.
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            n = int(base.size)
        else:
            n = 256
    except Exception:          # height_sequence_1 may be missing
        n = 256

    best_seq = None
    best_score = float('inf')

    # Candidate exponents – higher values give flatter convolutions.
    exponents = list(range(0, 11))          # 0…10
    rng.shuffle(exponents)                 # random order for variety

    for p in exponents:
        if time.monotonic() > deadline:
            break

        # Normalised indices in [0,1].
        idx = np.arange(n, dtype=np.float64)
        if n > 1:
            norm = idx / (n - 1)
        else:
            norm = np.ones_like(idx)

        # Power‑law shape, scaled so the maximum entry equals 1000.
        seq = (norm ** p) * 1000.0

        # Occasionally add a tiny random jitter to avoid exact symmetry.
        if p > 0 and rng.random() < 0.3:
            jitter = rng.normal(0.0, 1e-2, size=n)   # ~0.01 absolute noise
            seq = np.clip(seq + jitter, 0.0, 1000.0)

        # Evaluate the candidate.
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[{time.monotonic():.2f}] exponent={p}, score={score:.6f}")

    # If, for any reason, nothing was evaluated, fall back to a uniform vector.
    if best_seq is None:
        best_seq = [1.0] * n
        best_score = float(evaluate_sequence(best_seq))
        print("Fallback to uniform sequence, score:", best_score)

    # Optional tiny local refinement for any remaining time.
    while time.monotonic() < deadline:
        # Simple weight‑transfer move: move a small amount from a large
        # entry to a small entry while keeping the total sum constant.
        a = np.array(best_seq, dtype=np.float64)
        i_large = int(rng.integers(0, n))
        i_small = int(rng.integers(0, n))
        if i_large == i_small:
            continue
        delta = rng.uniform(0.0, a[i_large] * 0.05)  # at most 5 % of source
        if a[i_large] - delta < 0.0:
            continue
        a[i_large] -= delta
        a[i_small] += delta
        candidate = np.clip(a, 0.0, 1000.0).tolist()
        cand_score = float(evaluate_sequence(candidate))
        if np.isfinite(cand_score) and cand_score < best_score:
            best_score = cand_score
            best_seq = candidate
            print(f"[{time.monotonic():.2f}] refined score={cand_score:.6f}")

    return [float(x) for x in best_seq]
