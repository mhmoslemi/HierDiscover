**Implementation of projection onto simplex**: Use standard algorithm:

def simplex_projection(v):
    # Input: any real vector
    # Output: projection onto simplex {x >=0, sum x=1}
    # Implementation O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    max_b = conv.max()

def simplex_projection(v):
    # Projects a vector onto the probability simplex.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # Determine the rho index
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def pairwise_move(p, rng, max_transfer=0.01):
    n = p.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return p

    max_amount = min(p[i], max_transfer)

    epsilon = rng.random() * max_amount
    new_p = p.copy()
    new_p[i] -= epsilon
    new_p[j] += epsilon
    return new_p

"""
Propose a candidate step‑function sequence.

The algorithm works on the probability simplex (the coefficients are
treated as a probability distribution).  For a distribution `p`
the evaluation score is  

    score(p) = 2·n·max(p∗p) / (sum(p)²) = 2·n·max(p∗p)      (sum = 1)

Thus we only need to minimise the ℓ∞‑norm of the auto‑convolution.
The procedure starts from a feasible point (the known `height_sequence_1`,
or a uniform distribution) and performs a mixed local search:

* **Projected gradient step** – moves opposite to the gradient of the
  currently active convolution entry while staying on the simplex.
* **Pairwise exchange** – transfers a small amount of mass from one index
  to another.
* **Block perturbation** – multiplies a random sub‑array by a random factor
  and re‑projects onto the simplex (a cheap analogue of the original code).

The best sequence found within the time budget is returned (scaled to stay
≤ 1000, which does not affect the score because the evaluation function is
scale‑invariant).  Progress is printed every 500 iterations.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection on the probability simplex (sum = 1, all entries ≥ 0)
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of vector v onto the simplex {x ≥ 0, sum(x)=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]          # the last index that satisfies the condition
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise to exactly 1
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Helper: exact evaluation of a candidate (same as the provided evaluate_sequence)
def compute_score(seq: np.ndarray) -> float:
    """Return the same value as evaluate_sequence for a numpy array."""
    n = seq.size
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Helper: a cheap block perturbation (keeps the vector on the simplex)
def block_perturb(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative, simplex‑projected perturbation of ``base``."""
    n = base.size
    candidate = base.copy()
    # choose a block width proportional to n
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))          # log‑normal scaling
    candidate[start:stop] *= factor
    if rng.random() < 0.35:                              # occasional mirroring
        candidate = 0.5 * (candidate + candidate[::-1])
    # keep non‑negative and re‑project onto the simplex
    candidate = np.maximum(candidate, 0.0)
    candidate = simplex_projection(candidate)
    return candidate

    seed : int

    budget_s : float

    Returns
    -------
    list[float]
        The best sequence found (values are clipped to [0, 1000] only for safety).
    """

    # ------------------------------------------------------------------
    # initialise starting point -------------------------------------------------
    try:
        # Use the global variable if it exists and looks sane

        base = np.clip(base, 0.0, 1000.0)

        p = base / base.sum()                     # normalise to a probability vector
        n = base.size
    except Exception:
        # fall back to a modest size uniform vector (the same size used by the
        # reference code when the supplied sequence is invalid)
        n = 256
        p = np.ones(n, dtype=np.float64) / n

    # safety: guarantee a proper probability vector
    p = np.maximum(p, 0.0)
    p = simplex_projection(p)

    best = p.copy()
    best_score = compute_score(best)

    # ------------------------------------------------------------------
    # main optimisation loop ---------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # a little margin for the return
    iteration = 0
    while time.monotonic() < deadline:
        r = rng.random()
        # --------------------------------------------------------------
        # 1) projected gradient step on the current active convolution entry
        if r < 0.40:
            conv = np.convolve(p, p)
            k_max = int(conv.argmax())

            idx = np.arange(n)

    return [float(x) for x in result]
