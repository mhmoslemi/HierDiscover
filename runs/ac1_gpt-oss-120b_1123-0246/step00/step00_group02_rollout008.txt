"""
Iterative LP‑inspired optimisation of a non‑negative step function.

The algorithm maintains a normalised sequence `a` (∑a = 1).  For the
current sequence we compute the maximal convolution peak

    M = max_k (a * a)[k]

and then solve (in closed form) the linear programme

    maximise   Σ b_i
    subject to  a * b ≤ M   (pointwise),   b_i ≥ 0 .

The LP has a very simple structure: each variable `b_i` consumes the
same amount `a_i` of capacity on a contiguous block of length `n`.
Thus a greedy “cheapest‑first’’ allocation gives the optimal `b`.  The
solution satisfies Σb > Σa unless `a` is already uniform, so the
scaled vector `g = b / Σb` (still summing to one) can be mixed with `a`

    a ← (1‑t)·a + t·g ,   0 < t ≤ 0.2

which strictly decreases the worst autocorrelation value `M`.  Repeating
the step drives the objective `2·n·M` towards its theoretical lower
bound `2·n/(2·n‑1) ≈ 1`.  Random block perturbations are injected from
time to time to avoid stagnation.

The function `propose_candidate` runs this loop for the allotted time
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: a cheap random block perturbation (the same idea used in the
# baseline code).  It keeps the candidate non‑negative and respects the
# clamping limits of `evaluate_sequence`.
def make_block_perturbation(base, rng):
    """Return a one‑block O(n) non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # block width between n/200 and n/12 (at least 1)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal factor
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation (helps to keep the autocorrelation flat)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # clip to the limits enforced inside `evaluate_sequence`
    return np.clip(candidate, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Greedy solution of   max Σ b_i   s.t.   a * b ≤ M ,  b ≥ 0 .
def compute_optimal_b(a, M, eps=1e-12):
    """
    Solve the LP described above by a cheapest‑first greedy algorithm.
    Parameters
    ----------
    a : 1‑d ndarray, length n, non‑negative, sum(a)=1 (the current sequence)
    M : float, the current maximal convolution value max(a*a)
    eps : small number to avoid division by zero

    Returns
    -------
    b : ndarray of length n (may be very sparse).  By construction
        a * b ≤ M and Σ b is maximal.
    """
    n = a.shape[0]
    # make sure every coefficient is at least eps to keep the division safe
    a_adj = np.maximum(a, eps)

    # convolution result of a with the (still unknown) b; we build it incrementally.
    conv = np.zeros(2 * n - 1, dtype=np.float64)
    b = np.zeros_like(a_adj)

    # indices sorted by increasing a_i – the “cheapest’’ variables first.
    order = np.argsort(a_adj)

    for i in order:
        ai = a_adj[i]
        # the window of length n on which b_i contributes to the convolution
        start = i
        end = i + n                # slice end is exclusive
        # remaining capacity on this window
        remaining = M - conv[start:end]
        # the smallest remaining capacity determines how much we can add
        min_cap = np.min(remaining)
        if min_cap <= 0.0:
            continue
        delta = min_cap / ai         # amount we can increase b_i
        if delta <= 0.0:
            continue
        b[i] = delta
        conv[start:end] += ai * delta

    return b


# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=500, **kwargs):

    The function runs for at most ``budget_s`` seconds and returns the
    best sequence encountered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Normalise to unit sum (the evaluation is scale‑invariant).
    a = a / a.sum()
    # Enforce a tiny lower bound to keep the greedy solver well‑behaved.
    MIN_EPS = 1e-8
    a = np.maximum(a, MIN_EPS)
    a = a / a.sum()

    n = a.shape[0]

    # ------------------------------------------------------------------
    # Keep the best sequence seen so far.
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n}  score={best_score:.6f}")

    # Algorithmic hyper‑parameters
    t_mix = 0.15          # convex‑combination weight (0 < t ≤ 0.2 works well)

        conv_self = np.convolve(a, a)          # length 2n‑1

        b = compute_optimal_b(a, M, eps=MIN_EPS)
        sum_b = float(b.sum())
        if sum_b <= 0.0 or not np.isfinite(sum_b):

            a = make_block_perturbation(a, rng)

        #    sequence changes permanently if the new candidate looks promising.
        cand = (1.0 - t_mix) * a + t_mix * g
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()          # re‑normalise to unit sum

        # 4) evaluate
        try:

        except Exception:
            # any unexpected error – skip this candidate
            continue

        if np.isfinite(score) and score < best_score:
            best_seq = cand.copy()
            best_score = score
            a = cand                      # continue from the improving point
            patience = 0
            last_improvement = time.monotonic()
            print(f"[+] new best {best_score:.6f}  (t={t_mix:.3f})")
        else:
            patience += 1
            # occasional random perturbation to escape a plateau

                a = make_block_perturbation(a, rng)
                a /= a.sum()
                patience = 0

        # 6) slowly reduce the mixing factor if we have long stagnation
        if patience > max_patience:

            t_mix = max(t_min, t_mix * 0.7)
            patience = 0
            print(f"[info] reducing mixing t from {old_t:.3f} to {t_mix:.3f}")

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}  length = {best_seq.size}")
    return [float(v) for v in best_seq]
