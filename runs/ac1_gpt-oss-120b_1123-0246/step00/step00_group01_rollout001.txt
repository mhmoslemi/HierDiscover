    n = base.size

    # Initialize a: random perturbation around base
    a = np.copy(base)
    # ensure non-negative and clip
    a = np.clip(a, 0.0, 1000.0)

    # Evaluate
    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # parameters
    beta = 5.0
    step = 0.05   # learning rate
    decay = 0.9995   # decay factor per iteration (maybe)
    last_print = time.monotonic()
    iter_count = 0

    while time.monotonic() < deadline:
        # Decide to do gradient step or random block perturb
        if rng.random() < 0.7:  # mostly gradient
            # compute convolution via FFT for speed
            # But we need also b vector for max and gradient weight.
            b = np.convolve(a, a)  # compute full convolution; okay
            # compute softmax weights
            # To avoid overflow, shift by max
            max_b = np.max(b)
            # Choose beta adaptively? keep fixed
            exps = np.exp(beta * (b - max_b))
            w = exps / np.sum(exps)
            # compute gradient via convolution
            # Use convolution of a and w reversed
            grad_full = np.convolve(a, w[::-1])
            # length = n + (2n-1) -1 = 3n-2
            # take slice for gradient indices 0..n-1 as derived.
            start = (len(w) - 1)  # 2n-2
            grad = 2.0 * grad_full[start - np.arange(n)]
            # Update a (gradient descent)
            a = a - step * grad
            # enforce non-negativity
            a = np.clip(a, 0.0, 1000.0)
            # optionally normalize sum to 1 to keep scale consistent
            sum_a = np.sum(a)
            if sum_a > 0:
                a = a * (np.sqrt(2 * n) / sum_a)  # maybe same scaling as baseline but not needed
        else:
            # random block perturb as before
            a = make_block_perturbation(a, rng)

        # Evaluate candidate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_seq = a.copy()
            best_score = score
            print(f"Iter {iter_count}, new best {best_score:.6f}")
        # Adapt step size
        step *= decay
        iter_count += 1

        # occasional print progress
        if time.monotonic() - last_print > 30.0:
            print(f"Iter {iter_count}, best {best_score:.6f}, step {step:.6f}")
            last_print = time.monotonic()

    # final return
    result = [float(x) for x in best_seq]
    print(f"Finished, best score {best_score:.6f}, iterations {iter_count}")

"""
Propose a high‑quality step‑function sequence.

The algorithm starts from the provided ``height_sequence_1`` (or falls back to
a uniform vector) and then alternates two complementary moves inside the
time budget:

1. **Smooth‑max gradient descent** – we minimise a smoothed version of the
   evaluation objective 2·n·max(b)/(sum(a)²) where b = a∗a.
   A soft‑max with temperature β approximates the max, yielding a differentiable
   surrogate f(a)=log‑sum‑exp(β·b).  Its gradient can be computed cheaply by a
   single convolution:
       grad_i = 2·∑_j a_j·w_{i+j},
   where w is the soft‑max weight vector.  After the update we clip to the
   allowed range [0,1000] and renormalise the sum (the objective is scale‑invariant).

2. **Block perturbations** – the original “make_block_perturbation”
   routine is kept as a stochastic explorer that occasionally jumps out of
   local minima.

Both moves are performed many times (the inner loop is O(n log n) via
np.convolve, which is fast for n≈6000).  The temperature β is increased
linearly during the run, turning the smooth surrogate into the true max
as the search progresses.  A simple exponential decay on the learning rate
keeps the steps stable.  The best sequence found so far is stored and
returned at the end.

The implementation respects all competition constraints:
* No closures or lambdas.
* All helper functions are top‑level.
* No file or network I/O.
* Uses only ``numpy`` and the already‑imported ``evaluate_sequence``.
"""

import time
import numpy as np

def make_block_perturbation(base, rng):
    """Return one O(n)-memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

def _smooth_max_weights(b, beta):
    """Soft‑max weights for a vector ``b`` with temperature ``beta``."""
    # shift by the maximum for numerical stability
    b_max = np.max(b)
    exp_shift = np.exp(beta * (b - b_max))
    w = exp_shift / np.sum(exp_shift)
    return w

def _gradient_step(a, beta):
    """
    Perform a single gradient descent step on ``a`` for the smoothed max
    objective with temperature ``beta``.
    Returns the updated array (still inside the allowed range).
    """
    n = a.size
    # convolution a*a (full length = 2n‑1)
    b = np.convolve(a, a)
    # soft‑max weights over convolution entries
    w = _smooth_max_weights(b, beta)

    # gradient = 2 * (a ⋆ w)   where ⋆ is cross‑correlation.
    # Using convolution with the reversed weight vector:
    conv_full = np.convolve(a, w[::-1])          # length = 3n‑2
    start = len(w) - 1                           # = 2n‑2
    grad = 2.0 * conv_full[start - np.arange(n)] # shape (n,)

    # simple fixed learning rate (chosen outside)
    return grad

def propose_candidate(seed=42, budget_s=500, **kwargs):
    """
    Search for a low‑value step‑function sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget in seconds (the outer harness gives us ~1 000 s).

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # 1. Initialise the base vector
    # ------------------------------------------------------------------
    try:

    except Exception:

    if base is None or base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():

        n_default = 3000
        base = np.ones(n_default, dtype=np.float64)
    n = base.size
    a = np.clip(base, 0.0, 1000.0)

    # Normalise the sum to sqrt(2n) (scale does not affect the objective,
    # but keeps numbers in a comfortable range).
    sum_a = np.sum(a)
    if sum_a > 0:
        a *= np.sqrt(2.0 * n) / sum_a
    else:
        a[:] = 1.0
    a = np.clip(a, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # 2. Evaluate the starting point
    # ------------------------------------------------------------------
    b_cur = np.convolve(a, a)
    best_score = 2.0 * n * np.max(b_cur) / (np.sum(a) ** 2)
    best_seq = a.copy()
    print(f"[init] n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Hyper‑parameters
    # ------------------------------------------------------------------
    step = 0.12          # initial learning rate for gradient steps
    step_decay = 0.9995  # exponential decay per iteration
    beta_start = 1.0
    beta_end = 15.0

    # probability of performing a gradient step vs block perturbation
    prob_grad = 0.6

    iteration = 0
    last_report = time.monotonic()

    # ------------------------------------------------------------------
    # 4. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # progress reporting every ~30 s
        now = time.monotonic()
        if now - last_report > 30.0:
            print(f"[report] iter={iteration}, best={best_score:.6f}, step={step:.6f}")
            last_report = now

        elapsed = now - (deadline - float(budget_s))
        # linearly increase beta from start to end
        beta = beta_start + (beta_end - beta_start) * (elapsed / float(budget_s))

        if rng.random() < prob_grad:
            # ----- gradient descent move -----
            grad = _gradient_step(a, beta)
            a = a - step * grad
            # clip to allowed range
            a = np.clip(a, 0.0, 1000.0)
        else:
            # ----- stochastic block perturbation -----
            a = make_block_perturbation(a, rng)

        # occasional symmetrisation (helps because the objective is symmetric)
        if rng.random() < 0.07:
            a = 0.5 * (a + a[::-1])

        # renormalise sum to keep numbers well‑scaled
        sum_a = np.sum(a)
        if sum_a < 1e-8:
            # avoid collapse – inject a tiny uniform mass
            a[:] = 1.0
            sum_a = np.sum(a)
        a *= np.sqrt(2.0 * n) / sum_a
        a = np.clip(a, 0.0, 1000.0)

        # ----- evaluate candidate -----
        b = np.convolve(a, a)
        score = 2.0 * n * np.max(b) / (np.sum(a) ** 2)

        if np.isfinite(score) and score < best_score:
            best_score = float(score)
            best_seq = a.copy()
            print(f"[improve] iter={iteration}, new best={best_score:.6f}")

        # decay learning rate
        step *= step_decay
        iteration += 1

    # ------------------------------------------------------------------
    # 5. Return the best found sequence
    # ------------------------------------------------------------------
    print(f"[finished] iterations={iteration}, best_score={best_score:.6f}")
    return [float(v) for v in best_seq]
