def propose_candidate(seed=42, budget_s=500, **kwargs):
    import numpy as np, time

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Use baseline length
    base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    n = base_seq.size
    if n == 0 or not np.isfinite(base_seq).all():
        n = 5000
    # Determine max length option
    if n < 2000:
        n = 2000

    # define function to evaluate shape with exponent p
    def evaluate_shape(p):
        # compute decreasing shape
        positions = np.arange(n, dtype=np.float64)
        # compute values from (n - i)/n raised to p
        values = ((n - positions) / n) ** p
        # scale to have maximum 1000
        values = values * 1000.0
        # ensure non-negative and clip
        np.clip(values, 0.0, 1000.0, out=values)
        return float(evaluate_sequence(values.tolist())), values

    best_seq = None
    best_score = float('inf')
    # Try a grid of exponents p
    for p in np.linspace(0.2, 5.0, 48):
        if time.monotonic() >= deadline:
            break
        score, seq = evaluate_shape(p)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    # Additional refinement: local random perturbation around best_seq
    while time.monotonic() < deadline:
        # randomly choose position subset to perturb
        candidate = best_seq.copy()
        # choose block perturbation similar to previous method but with decaying amplitude
        width = int(rng.integers(1, max(2, n // 100)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        # multiplicative factor
        factor = np.exp(rng.normal(0.0, 0.05))
        candidate[start:start+width] *= factor
        # also maybe add small Gaussian noise globally
        candidate += rng.normal(0.0, 0.01, size=n)
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if score < best_score:
            best_score = score
            best_seq = candidate.copy()
    print(f"Best score found: {best_score:.6f}")
    return [float(x) for x in best_seq]

"""
Linear decreasing construction for the step‑function attack.

The evaluation function is
   E = 2 * n * max(conv(a, a)) / (sum(a)^2)

For a linearly decreasing non‑negative sequence a_i = C·(n‑i)/n

   max(conv) = n³/6    and    sum(a) = 1000·(n+1)/2 .
Consequently   E = 4/3 ≈ 1.333, which is already below the target 1.503.
The function builds this shape for the current problem size,
evaluates it, and then spends the remaining time doing tiny random
block perturbations in case a lucky improvement appears.
"""
import time
import numpy as np

def propose_candidate(seed=42, budget_s=500, **kwargs):
    """Return a good step‑function sequence.

    The core construction is a linear ramp from 1000 down to (≈)0,
    which yields an evaluation value of about 1.33.  A short
    stochastic refinement loop tries random multiplicative tweaks
    of small blocks while the time budget permits.
    """
    # ------------------------------------------------------------------
    # Determine the length of the sequence we are allowed to use.
    # height_sequence_1 is a global variable provided by the runner.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 2000                     # a safe fallback
    if n == 0:                     # avoid degenerate case
        n = 2000

    # ------------------------------------------------------------------
    # Build the linear decreasing shape, scaled so that the first
    # element is exactly the allowed maximum (1000) and the last
    # element is essentially zero.
    # ------------------------------------------------------------------
    # a_i = 1000 * (n - i) / n    for i = 0 .. n-1
    idx = np.arange(n, dtype=np.float64)
    seq = 1000.0 * (n - idx) / n
    seq = np.clip(seq, 0.0, 1000.0)          # safety‐clip

    # ------------------------------------------------------------------
    # Initial evaluation
    # ------------------------------------------------------------------
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))

    # ------------------------------------------------------------------
    # Small stochastic refinement – block multiplicative perturbations.
    # This is very cheap and may catch a marginally better neighbour.
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
