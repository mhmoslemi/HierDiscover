"""
Propose candidate function using symmetric exponential (U-shaped) sequences.
Searches over length n and decay parameter d (where r = 1 - d/n) to minimize
evaluate_sequence. After a grid search, a short local search with block
perturbations further refines the best sequence.
"""
import time
import numpy as np

def generate_symmetric_exponential(n, d):
    """Return a symmetric exponential sequence of length n.

    The sequence has maximum value 1000 at both ends and decays
    geometrically towards the centre.  Let r = 1 - d/n; then
    a[i] = 1000 * r**min(i, n-1-i).
    """
    if n <= 0:
        raise ValueError
    r = 1.0 - d / float(n)
    if r <= 0:
        r = 0.0
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    seq = 1000.0 * np.power(r, dist)
    return seq

def make_block_perturbation(base, rng):
    """Return a copy of `base` with a random block multiplied by a factor."""
    candidate = np.array(base, copy=True)
    n = candidate.size
    # width between 1% and 15% of n
    low = max(1, n // 200)
    high = max(2, n // 12)
    width = int(rng.integers(low, high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # optional symmetrisation
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    # clip to allowed range
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate

def propose_candidate(seed=42, budget_s=500, **kwargs):
    """Search for a low‑value step function.

    The search consists of:
    1. a deterministic grid over length `n` and decay `d`;
    2. a short stochastic refinement using block perturbations.

    Returns the best sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # start from a fallback uniform sequence in case everything fails
    best = np.ones(256, dtype=np.float64) * 1.0
    best_score = float('inf')
    try:
        # Optional fallback to the global variable if present
        global height_sequence_1
        fallback = np.asarray(height_sequence_1, dtype=np.float64)
        if fallback.ndim == 1 and fallback.size > 0 and np.isfinite(fallback).all():
            best = np.clip(fallback, 0.0, 1000.0).copy()
    except Exception:
        pass

    # --- Grid search -------------------------------------------------
    # Lengths to try (powers of two up to ~8000)
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    # d values (roughly (1‑r)*n) – we concentrate on the region that gives
    # ratios < 2.  The step size adapts to the length.
    d_grid = np.linspace(0.2, 2.0, 19)  # 0.2,0.3,...,2.0

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        for d in d_grid:
            # candidate generation
            seq = generate_symmetric_exponential(n, d)
            # Evaluate
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best = seq
                best_score = score
                print(f"New best: n={n}, d={d:.3f}, score={score:.6f}")

        # also try a plain monotone exponential (non‑symmetric) for comparison
        for d in d_grid:
            if time.monotonic() >= deadline:
                break
            r = 1.0 - d / float(n)
            if r <= 0:
                continue
            seq = 1000.0 * np.power(r, np.arange(n, dtype=np.float64))
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best = seq
                best_score = score
                print(f"New best (mono): n={n}, d={d:.3f}, score={score:.6f}")

    # --- Local refinement (block perturbations) ----------------------
    parent = best.copy()
    while time.monotonic() < deadline:
        cand = make_block_perturbation(parent, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            parent = cand
            print(f"Improved by perturbation: n={best.size}, score={best_score:.6f}")

    # Return plain python list as required
    return [float(x) for x in best]

"""
propose_candidate – search for a low‑value step function.

The core idea is to use a *U‑shaped* (symmetric exponential) sequence
a[i] = 1000·r**min(i, n‑1‑i) where r = 1 – d/n.  For a given length n the
parameter d ≈ (1‑r)·n controls how quickly the heights decay toward the
centre.  The convolution peak of such a shape is roughly (n/2)·r^{n/2},
while the L1 norm grows like n·(1‑e^{‑d})/d, giving the ratio

    2·n·max_b / (sum_a**2) ≈ d²·e^{‑d/2} / (1‑e^{‑d})².

Choosing d in the range 0.2…1.0 yields values well below the target
1.503 (the optimum is ≈1 for d→0).  The function first performs a
lightweight grid search over several lengths and d‑values, then refines
the best candidate with a few block‑perturbation steps (the same perturb
operator used in the reference code).

All numbers are kept in the allowed range [0, 1000] and the final
sequence is returned as a plain Python list.
"""

import time
import numpy as np

def generate_symmetric_exponential(n: int, d: float) -> np.ndarray:
    """Return a symmetric exponential sequence of length `n`.

    The sequence has its maximum value 1000 at both ends and decays
    geometrically towards the centre.
    """
    if n <= 0:
        raise ValueError("n must be positive")
    # r = 1 - d/n  (0 < r <= 1)
    r = 1.0 - d / float(n)
    if r <= 0.0:
        r = 0.0
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)   # distance from the nearer edge
    seq = 1000.0 * np.power(r, dist)
    # clipping is a safety net; with the construction above values stay ≤1000
    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a copy of `base` where a random block is multiplicatively perturbed."""
    candidate = np.array(base, copy=True)
    n = candidate.size
    # block width: between ~0.5 % and ~8 % of the length
    low = max(1, n // 200)
    high = max(2, n // 12)
    width = int(rng.integers(low, high + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    candidate[start:stop] *= factor
    # occasional symmetrisation (helps for symmetric shapes)
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    np.clip(candidate, 0.0, 1000.0, out=candidate)
    return candidate


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs) -> list[float]:
    """Search for a step function with a low evaluation score.

    The function first scans a grid of (length, decay) parameters for the
    symmetric exponential shape described above.  Afterwards it spends the
    remaining time performing cheap block‑perturbations starting from the
    best candidate found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # start from a (tiny) safe fallback in case everything else fails
    best = np.ones(256, dtype=np.float64) * 1.0
    best_score = float('inf')
    try:

        # additional fallback (the reference code does this).
        global height_sequence_1
        fallback = np.asarray(height_sequence_1, dtype=np.float64)
        if fallback.ndim == 1 and fallback.size > 0 and np.isfinite(fallback).all():
            best = np.clip(fallback, 0.0, 1000.0).copy()
    except Exception:

    # ------------------------------------------------------------------
    # 1) deterministic grid search over lengths and decay parameters d
    lengths = [256, 512, 1024, 2048, 4096, 8192]          # powers of two
    d_grid = np.linspace(0.2, 2.0, 19)                    # 0.2,0.3,…,2.0

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        for d in d_grid:

                best_score = score

        # also test the *monotone* exponential (no mirroring) for completeness
        for d in d_grid:
            if time.monotonic() >= deadline:
                break
            r = 1.0 - d / float(n)
            if r <= 0.0:

            seq = 1000.0 * np.power(r, np.arange(n, dtype=np.float64))

    return [float(x) for x in best]
