""" 
Heuristic optimizer for the convolution‐ratio problem.

The algorithm starts from the supplied ``height_sequence_1`` (or a default
uniform vector) and runs two phases inside the allotted budget:

1. A cheap random block perturbation phase – identical to the reference
   implementation – which quickly finds a reasonable starting point.
2. A linear‑programming phase that implements the “LP improvement” step
   described in the problem statement.  For the current sequence `f`
   we solve
        max 1ᵀ g
        s.t. f * g ≤ M,   g ≥ 0,
   where `M = max(f*f)`.  The optimal `g` has a larger total mass while
   respecting the same convolution bound, and we move a small step towards
   `g`.  Repeating this monotonically reduces the maximal convolution value
   and consequently the evaluation score.

Both phases keep track of the best sequence seen so far and return it
when the time budget expires.
"""

import numpy as np
import time
import scipy.optimize

def make_block_perturbation(base, rng):
    """Return one O(n)-memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

def _build_conv_constraint_matrix(f):
    """Build matrix A such that (A @ g)[k] = (f * g)[k] for all k."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        start = max(0, k - (n - 1))
        end   = min(n - 1, k)
        i = np.arange(start, end + 1)
        j = k - i                     # 0 ≤ j < n
        A[k, j] = f[i]                # coefficient for g_j
    return A

def propose_candidate(seed=42, budget_s=500, **kwargs):
    """Return a sequence of non‑negative heights that (hopefully) beats 2.0.

    The function respects the supplied time budget (``budget_s`` seconds)
    and attempts to minimise ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    # Give a small safety margin so that we definitely return before the
    # external watchdog kills the process.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # ------------------------------------------------------------------
    # Initialise the current solution.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        parent = np.ones(256, dtype=np.float64)

    parent = np.clip(parent, 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)

    a = parent.astype(np.float64)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[Init] n={len(a)}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – cheap random block perturbations.
    # ------------------------------------------------------------------
    # Keep a little time reserve for the LP phase.
    while time.monotonic() < deadline - 20.0:
        cand = make_block_perturbation(a, rng)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            a = cand
            best = cand.copy()
            best_score = sc
            print(f"[Phase‑1] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – linear‑programming driven improvement (the “LP‑step”).
    # ------------------------------------------------------------------
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)      # the normalisation used in the paper

    while time.monotonic() < deadline:
        # current maximal convolution value
        conv = np.convolve(a, a)
        M = np.max(conv)

        # Build the (f * g) ≤ M constraints.
        A = _build_conv_constraint_matrix(a)
        rows = A.shape[0]

        # Linear programme:
        #   maximise   1ᵀ g   (i.e. minimise -1ᵀ g)
        #   subject to A @ g ≤ M  and 0 ≤ g ≤ 1000.
        c = -np.ones(n, dtype=np.float64)               # minimise -sum(g)
        bounds = [(0.0, 1000.0) for _ in range(n)]
        b_ub = np.full(rows, M, dtype=np.float64)

        # Solve with HiGHS – fast and robust for our modest problem size.
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=bounds, method='highs')

        if not res.success:
            # Fallback: a tiny random move, this rarely happens.
            cand = make_block_perturbation(a, rng)
        else:
            g = res.x
            # Rescale g so that it has the standard normalisation.
            s = np.sum(g)
            if s > 0:
                g = g * (target_sum / s)

            # Move a short step towards g.
            t = 0.3                                    # mixing factor
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, 1000.0)

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc
            a = cand.copy()          # continue optimisation from the new point
            print(f"[Phase‑2] improved to {best_score:.6f}")
        else:
            # Still accept the candidate as the new centre to keep the
            # optimisation moving – this mimics a very cheap simulated‑annealing.
            a = cand.copy()

        # Early exit if we already passed the target.
        if best_score < 1.55:           # a comfortable safety margin
            break

    print(f"[Done] best score {best_score:.6f}")
    return [float(x) for x in best]

"""
Heuristic optimiser for the “max‑convolution / (sum²)” objective.

The routine works inside the allotted time budget (``budget_s`` seconds) and
returns a list of non‑negative heights that (hopefully) beats the naïve
value 2.0.

Two phases are executed:

1. **Fast random block perturbations** – identical to the reference code.
   This quickly finds a decent starting point.

2. **LP‑driven refinement** – Implements the iteration described in the
   statement.  For the current sequence ``f`` we compute ``M = max(f*f)`` and
   solve the linear programme

        maximise   1ᵀ g
        subject to  f * g ≤ M      (convolution bound)
                    g ≥ 0 , g ≤ 1000

   (the convolution with a fixed ``f`` is linear in ``g``).  The optimiser
   returns a vector ``g`` with a larger total mass while respecting the same
   convolution bound.  After rescaling ``g`` to the normalisation
   ``sqrt(2·n)`` we move a short step towards it:

        f ← (1‑t)·f + t·g   (t≈0.3)

   Repeating this monotone step reduces the maximal convolution value and
   thus the evaluation score.

Both phases keep the best sequence seen so far and return it when the
budget expires.
"""

import numpy as np
import time
import scipy.optimize


def make_block_perturbation(base, rng):
    """Return one O(n)‑memory non‑negative perturbation of ``base``."""
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    # width of the perturbed block (between n/200 and n/12)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # optional mirroring to keep the shape roughly symmetric
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)


def _build_conv_constraint_matrix(f):
    """
    Build the matrix A such that (A @ g)[k] = (f * g)[k] for all k.
    ``f`` is fixed, the rows correspond to the different convolution shifts.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        i = np.arange(start, end + 1)          # indices of f that participate
        j = k - i                               # matching indices of g
        A[k, j] = f[i]                          # coefficient for g_j
    return A

    rng = np.random.default_rng(seed)

    best = a.copy()

    # 2️⃣  Phase 1 – cheap random block perturbations.
    # ------------------------------------------------------------------
    # Reserve a few seconds for the more expensive LP phase.
    while time.monotonic() < time.monotonic() + budget_s - 20.0:
        cand = make_block_perturbation(a, rng)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            a = cand
            best = cand.copy()
            best_score = sc
            print(f"[Phase‑1] new best {best_score:.6f}")

        # stop early if we have run out of the budget
        if time.monotonic() >= time.monotonic() + budget_s - 20.0:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Phase 2 – LP‑driven refinement (the “LP‑step” from the paper).
    # ------------------------------------------------------------------
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)       # normalisation used in the reference algorithm
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    while time.monotonic() < deadline:

        conv = np.convolve(a, a)
        M = np.max(conv)

        A = _build_conv_constraint_matrix(a)          # shape (2n‑1, n)
        rows = A.shape[0]
        c = -np.ones(n, dtype=np.float64)             # maximise sum(g)  ⇔ minimise –sum(g)
        bounds = [(0.0, 1000.0) for _ in range(n)]

            a = cand.copy()

        if best_score < 1.55:

    print(f"[Done] best score {best_score:.6f}")
    return [float(x) for x in best]
