def propose_candidate(seed=42, budget_s=500, **kwargs):
    """
    Generate a step function sequence that minimizes the evaluation metric.
    The algorithm explores simple power-law shapes a_i = (i+1)^alpha,
    scales them to respect the 0..1000 bound and picks the best one.
    """
    import numpy as np
    # hyperparameters
    n = 2000  # length of the sequence; keep power <=1000 after scaling.
    alphas = [0.5, 0.8, 1.0, 1.2, 1.3, 1.35, 1.38, 1.4, 1.42, 1.45, 1.5, 1.6, 2.0]
    best_seq = None
    best_score = float('inf')
    for alpha in alphas:
        # generate raw values
        i = np.arange(1, n + 1, dtype=np.float64)
        seq = np.power(i, alpha)
        max_val = seq[-1]
        if max_val > 1000.0:
            seq = seq * (1000.0 / max_val)
        # evaluate
        score = float(evaluate_sequence(seq.tolist()))
        # track best
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"alpha={alpha:.3f}, score={score:.6f}")
    # optional small local refinement (optional)
    # if best_seq is still None (should not happen) fallback to uniform.
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"fallback uniform, score={best_score}")
    print(f"Best alpha={alpha_of_best:.3f}, best score={best_score:.6f}")
    return [float(x) for x in best_seq]
