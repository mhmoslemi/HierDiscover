"""
Propose a step‑function sequence that minimizes
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages:

1. **U‑shaped seed generation** – a family of simple candidates
   (high values at the two ends, low constant value in the middle)
   is explored analytically.  For a given block length `L` the interior
   weight `w = sqrt(L/(n-2L))` (capped at 1) makes the self‑convolution
   peak essentially equal to `L`.  The theoretical score is
   `2*n*L / (2*L + (n-2*L)*w)^2`.  By evaluating several `L` values we
   obtain a solid starting point with a score well below the previous
   bound of 2.0.

2. **Local random refinement** – from the best seed we run a fast
   time‑bounded stochastic search.  Two perturbation operators are used:
   *block perturbations* (multiplicative noise on a random contiguous
   sub‑array, as in the reference code) and *pairwise mass transfers*
   (move a small amount of probability mass from one index to another).
   Occasionally the candidate is symmetrised to keep the convolution
   shape balanced.

3. **Best‑ever return** – the best sequence seen before the time budget
   expires is returned.  All numbers stay inside the allowed range
   `[0, 1000]`; scaling does not affect the objective, so we keep the
   maximal element ≤ 1.0 to avoid the clamping step in `evaluate_sequence`.

The implementation is deterministic given the `seed` argument and works
within the 1000 s limit.
"""
import time
import numpy as np

def make_block_perturbation(base, rng):
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # width is a sizeable random block (between 0.5 % and ~15 % of n)
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    # multiplicative log‑normal noise on the block
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring – keeps the shape roughly symmetric
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=500, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    provided ``evaluate_sequence`` function.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (default 500 s; the caller allows up to 1000 s).

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # -----------------------------------------------------------------
    # Determine problem size (length of the step function)
    # -----------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        # fallback size if the global variable is unavailable
        n = 256
        base = np.ones(n, dtype=np.float64)

    best = None
    best_score = np.inf

    # -----------------------------------------------------------------
    # 1)  U‑shaped seed exploration (different block lengths L)
    # -----------------------------------------------------------------
    max_L = n // 3          # beyond this the interior weight would exceed 1
    if max_L < 2:
        max_L = 2
    # try ~30 different L values, spaced roughly logarithmically
    L_candidates = np.unique(
        np.concatenate((
            np.arange(2, max_L + 1, max(1, max_L // 30)),
            np.geomspace(2, max_L, num=30, dtype=int)
        ))
    )
    for L in L_candidates:
        L = int(L)
        if 2 * L >= n:               # need a non‑empty interior region
            continue
        # interior weight that makes interior‑interior convolution just
        # touch the edge‑edge peak; capped at 1.0
        w = np.sqrt(L / (n - 2 * L))
        if w > 1.0:
            w = 1.0
        cand = np.full(n, w, dtype=np.float64)
        cand[:L] = 1.0
        cand[-L:] = 1.0
        cand = np.clip(cand, 0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = cand.copy()
            print(f"[Init] L={L:4d}, w={w:.4f} → score={best_score:.6f}")

    # -----------------------------------------------------------------
    # If for some reason no seed succeeded, fall back to the provided base.
    # -----------------------------------------------------------------
    if best is None:
        best = np.clip(base, 0.0, 1000.0)
        best_score = float(evaluate_sequence(best.tolist()))
        print("[Fallback] using raw height_sequence_1, score =", best_score)

    # -----------------------------------------------------------------
    # 2)  Time‑bounded stochastic local search
    # -----------------------------------------------------------------
    stagnation = 0            # counter for iterations without improvement
    while time.monotonic() < deadline:
        # Choose one of the two cheap perturbations
        if rng.random() < 0.55:
            # block‑style multiplicative noise
            cand = make_block_perturbation(best, rng)
        else:
            # pairwise mass transfer
            cand = best.copy()
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            # move up to 30 % of the source mass
            max_delta = cand[i] * 0.3
            if max_delta > 0:
                delta = rng.random() * max_delta
                cand[i] -= delta
                cand[j] += delta

        # occasional symmetrisation (helps the convolution shape)
        if rng.random() < 0.15:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = cand.copy()
            stagnation = 0
            print(f"[Update] new best score={best_score:.6f} at t={time.monotonic():.2f}")
        else:
            stagnation += 1

        # If we have been stuck for a while, inject a fresh random candidate
        if stagnation >= 2000:
            # generate a random Dirichlet vector (keep max ≤ 1)
            random_vec = rng.random(n)
            random_vec /= random_vec.max()
            random_vec = np.clip(random_vec, 0.0, 1000.0)
            # optionally blend a little with the current best
            cand = 0.7 * best + 0.3 * random_vec
            cand = np.clip(cand, 0.0, 1000.0)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = cand.copy()
                print(f"[Refresh] improved after stagnation → score={best_score:.6f}")
            stagnation = 0

    # -----------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # -----------------------------------------------------------------
    if best is None:
        best = np.ones(n, dtype=np.float64)
    return [float(x) for x in best]
