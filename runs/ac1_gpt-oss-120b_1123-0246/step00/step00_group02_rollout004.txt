"""
Propose a high‑quality step‑function sequence.

The evaluation function is scale‑invariant:
    score = 2 * n * max(conv(f,f)) / (sum(f))**2 .
For a monotone power‑law shape

    f_i = C * (i/(n‑1))**p ,   i = 0 … n‑1

the maximal convolution occurs at lag n‑1 and the score can be
computed analytically as

    score(p) = 2·Beta(p+1,p+1)·(p+1)² .

Beta decreases super‑exponentially, so even moderate exponents (p≥3)
give scores far below the target 1.503.  The code below searches a few
integer exponents within the allowed time budget, keeps the best
candidate and returns it.  All values are capped at 1000 (the maximum
allowed by ``evaluate_sequence``) by scaling the whole vector – this
does not change the score because the objective is homogeneous of
degree 0.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Generate a power‑law monotone sequence and keep the exponent giving
    the lowest evaluation score.

    Parameters
    ----------
    seed : int
        Random seed (unused – kept for API compatibility).
    budget_s : float
        Time budget in seconds.  The routine stops early if the budget
        is exhausted.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    # ------------------------------------------------------------------
    # Determine the length of the sequence we have to output.
    # Prefer the length of the provided construction (height_sequence_1);
    # fall back to a modest default if it is unavailable.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 256

    # Normalised positions in [0,1]; using float64 for accuracy.
    positions = np.linspace(0.0, 1.0, n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Search over a few integer exponents.  The objective is monotone
    # decreasing with p, but we evaluate a handful to be safe against
    # numeric artefacts (clipping, round‑off).
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    start_time = time.monotonic()
    # We never need more than a few exponent trials; the loop is tiny.
    for p in range(1, 11):                     # try p = 1 … 10
        # Stop immediately if we exceed the allocated budget.
        if time.monotonic() - start_time > max(0.0, budget_s - 1.0):
            break

        # Build the raw (unclipped) power‑law vector.
        raw = (positions ** p) * 1000.0          # max value exactly 1000

        # Convert to python list – the evaluator expects a list of floats.
        candidate = raw.tolist()

        # Evaluate the candidate; any failure is treated as infinite score.
        try:
            score = float(evaluate_sequence(candidate))
        except Exception:
            score = float("inf")

        # Keep the best one.
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = candidate

        # Log progress – useful for debugging.
        print(f"[p={p}] score = {score:.6f}")

    # ------------------------------------------------------------------
    # Fallback: if something went wrong and we have no candidate, use a
    # plain uniform sequence (the safe default that yields score 2.0).
    # ------------------------------------------------------------------
    if best_seq is None:
        print("Fallback to uniform sequence.")
        best_seq = [1000.0] * n
        best_score = float(evaluate_sequence(best_seq))

    print(f"Best exponent p gives score {best_score:.6f} (length {n})")
    return [float(x) for x in best_seq]
