def transfer_from_max_pair(seq, conv, k_max, rng):
    n = len(seq)
    # Determine valid indices
    valid_i = np.arange(max(0, k_max - (n-1)), min(n-1, k_max)+1)
    partners = k_max - valid_i
    products = seq[valid_i] * seq[partners]
    # source index with max product
    src_idx = valid_i[np.argmax(products)]
    src_partner = k_max - src_idx
    # destination index with min product (excluding source indices)
    # compute product array for all i (including those where partner out of range => product=0)
    all_i = np.arange(n)
    all_partners = k_max - all_i
    mask = (all_partners >= 0) & (all_partners < n)
    all_products = np.zeros(n, dtype=np.float64)
    all_products[mask] = seq[all_i[mask]] * seq[all_partners[mask]]
    # exclude src_idx and src_partner
    exclude = {src_idx, src_partner}
    # candidates for dests: set difference
    candidates = [i for i in range(n) if i not in exclude]
    # Choose dest1 with minimal product among candidates
    if candidates:
        dest_idx = min(candidates, key=lambda i: all_products[i])
    else:
        dest_idx = src_idx  # fallback
    # Choose dest2 similarly maybe next minimal distinct
    candidates2 = [i for i in candidates if i != dest_idx]
    if candidates2:
        dest2_idx = min(candidates2, key=lambda i: all_products[i])
    else:
        dest2_idx = dest_idx
    # Determine amount to move
    max_move = min(seq[src_idx], seq[src_partner])
    if max_move <= 0:
        return seq.copy()
    delta = max_move * rng.random() * 0.5  # up to 50% of smaller
    cand = seq.copy()
    if src_idx == src_partner:
        cand[src_idx] -= 2.0 * delta
    else:
        cand[src_idx] -= delta
        cand[src_partner] -= delta
    cand[dest_idx] += delta
    cand[dest2_idx] += delta
    np.clip(cand, 0, 1000, out=cand)
    return cand

def move_reduce_max(seq, k_max, rng):
    n = seq.size
    # compute partner array for all indices
    partner_idx = k_max - np.arange(n)
    # mask valid
    valid = (partner_idx >= 0) & (partner_idx < n)
    # product for valid indices else 0
    products = np.zeros(n)
    products[valid] = seq[valid] * seq[partner_idx[valid]]
    # find source indices (max product)
    src_idx = int(np.argmax(products))
    src_partner = int(k_max - src_idx)

"""
Propose a step‑function (list of non‑negative heights) that minimizes
evaluate_sequence(seq).

The core idea
-------------
* The evaluator is scale‑invariant, so only the *shape* of the sequence matters.
* For the arcsine (inverse‑square‑root) start the maximal convolution entry
  equals the square of the first coefficient (a₀² = 1).  By moving a small
  amount of mass from the pair that contributes most to the current maximum
  towards positions that contribute *least* we can strictly decrease the
  maximal convolution value while keeping the total sum unchanged.
* This “max‑reduction” move is applied many times inside a simple simulated‑annealing
  loop.  Occasionally a random block‑perturbation is performed to escape
  plateaus.
* The algorithm works for any length that is provided through the global
  ``height_sequence_1`` variable; if it is missing a default length of 256
  is used and the analytical arcsine sequence is generated.

The routine runs for at most *budget_s* seconds (default 900 s) and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Analytic start – the discrete arcsine (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """
    Return a non‑negative sequence a of the given length with
    a[i] = binom(2i,i)/4**i  (the discrete arcsine distribution).
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# 2.  Fast convolution (real FFT) – used repeatedly.
# ----------------------------------------------------------------------
def fast_convolve(seq: np.ndarray) -> np.ndarray:
    """
    Return the full convolution of ``seq`` with itself using a real FFT.
    """
    n = seq.size
    size = 2 * n - 1
    # rfft works on real‑valued data and is a little faster than fft.
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv


# ----------------------------------------------------------------------
# 3.  Targeted move – reduces the current maximal convolution entry.
# ----------------------------------------------------------------------
def reduce_max_move(base: np.ndarray, k_max: int, rng: np.random.Generator) -> np.ndarray:
    """
    Move a fraction of the mass from the pair (i, j) with i+j = k_max
    that contributes the most to the max entry, to two indices where the
    contribution to that entry is minimal.
    The total sum stays unchanged.
    """
    n = base.size
    idx = np.arange(n)
    partner = k_max - idx

    # valid pairs (both indices inside the array)
    valid = (partner >= 0) & (partner < n)
    products = np.zeros(n, dtype=np.float64)
    products[valid] = base[idx[valid]] * base[partner[valid]]

    # source – maximal product
    src = int(np.argmax(products))
    src_partner = int(k_max - src)

    # destination – two distinct positions with minimal product, avoiding the source pair
    exclude = np.zeros(n, dtype=bool)
    exclude[src] = True
    if 0 <= src_partner < n:
        exclude[src_partner] = True

    # Minimal product among the remaining indices
    tmp = products.copy()
    tmp[exclude] = np.inf
    dst1 = int(np.argmin(tmp))
    exclude[dst1] = True
    tmp[exclude] = np.inf
    dst2 = int(np.argmin(tmp))

    # Amount to move (at most 50 % of the smaller source entry)
    if src == src_partner:
        max_move = base[src] / 2.0
    else:
        max_move = min(base[src], base[src_partner])
    if max_move <= 0.0:
        return base.copy()

    delta = max_move * rng.random() * 0.50   # 0 … 50 % of the allowed amount

    cand = base.copy()
    if src == src_partner:
        cand[src] -= 2.0 * delta
    else:
        cand[src] -= delta
        cand[src_partner] -= delta
    cand[dst1] += delta
    cand[dst2] += delta

    # Clip to the evaluator's bounds.
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# 4.  Simple stochastic perturbation (keeps the search from stagnating).
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor.
    Occasionally the sequence is symmetrised (helps stability).
    """
    n = base.size
    # Width of the block – a few percent of the whole sequence.
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width

    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # log‑normal perturbation
    cand[start:stop] *= factor

    if rng.random() < 0.30:               # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])

    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# 5.  Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 900,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a score
    lower than the baseline arcsine construction.
    The routine runs a simulated‑annealing loop for up to *budget_s* seconds.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size – reuse the provided sequence if possible.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        # The harness may give us a (possibly already‑optimised) sequence.
        try:
            init_seq = np.array(height_sequence_1, dtype=np.float64)
            n = init_seq.shape[0]
        except Exception:
            # fallback to analytic start
            n = 256
            init_seq = build_arcsine_sequence(n)
    else:
        n = 256
        init_seq = build_arcsine_sequence(n)

    # Ensure we respect the evaluator's clipping.
    base_seq = np.clip(init_seq, 0.0, 1000.0)

    # --------------------------------------------------------------
    # Initial evaluation.
    # --------------------------------------------------------------
    best = base_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur = best.copy()
    cur_score = best_score

    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # tiny safety margin
    iteration = 0

    # Linear temperature schedule for simulated annealing.
    T_start = 0.015
    T_end = 0.0005

    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1.  Determine the current maximum convolution entry.
        # --------------------------------------------------------------
        conv = fast_convolve(cur)
        k_max = int(np.argmax(conv))

        # --------------------------------------------------------------
        # 2.  Propose a candidate.
        # --------------------------------------------------------------
        # Mostly use the dedicated max‑reduction move; occasionally
        # apply a random block perturbation to keep the search lively.
        if rng.random() < 0.80:
            cand = reduce_max_move(cur, k_max, rng)
        else:
            cand = block_perturbation(cur, rng)

        # --------------------------------------------------------------
        # 3.  Evaluate the candidate.
        # --------------------------------------------------------------
        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
            continue   # ignore illegal candidates

        # --------------------------------------------------------------
        # 4.  Acceptance test (simulated annealing).
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        frac = elapsed / float(budget_s)
        T = T_start * (1.0 - frac) + T_end * frac   # linear cooling

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                print(f"[{iteration:6d}] new best = {best_score:.6f}")

                # Early termination if we already beat the target.
                if best_score < 1.503:
                    print("[early] target reached")
                    break

        # Occasionally reset to the current best (helps escape deep valleys).
        if iteration % 8000 == 0 and rng.random() < 0.25:
            cur = best.copy()
            cur_score = best_score

    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best]
