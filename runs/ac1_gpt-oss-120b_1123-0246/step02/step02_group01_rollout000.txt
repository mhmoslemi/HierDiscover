def propose_candidate(seed=42, budget_s=500, **kwargs):

    deadline = start + max(0., float(budget_s) - 0.5)

    rng = np.random.default_rng(seed)

    # pick n

    a = _beta_sequence(n, 0.4)

    a_sum = a.sum()
    if a_sum > 0:
        a *= (S / a_sum)
    a = (a + a[::-1]) / 2.0
    # enforce sum again
    a *= (S / a.sum())

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2  # base learning rate, we will scale later
    gamma = 0.5
    gamma_max = 25.0
    gamma_factor = 1.01

    best_score = np.inf
    best_seq = a.copy()
    best_n = n
    iteration = 0

    while time.monotonic() < deadline:
        # compute convolution a*a
        size1 = 1 << ((2*n - 2).bit_length())
        A = np.fft.rfft(a, n=size1)
        conv_full = np.fft.irfft(A * A, n=size1)[:2*n-1]
        b = conv_full
        max_b = b.max()
        # smoothed weights
        # use gamma to weight
        w = np.exp(gamma * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0:
            # reduce gamma maybe
            gamma = max(gamma * 0.5, 0.1)
            continue
        p = w / w_sum

        # compute gradient
        a_rev = a[::-1]
        size2 = 1 << ((3*n - 3).bit_length())
        A_rev = np.fft.rfft(a_rev, n=size2)
        P = np.fft.rfft(p, n=size2)
        conv2_full = np.fft.irfft(A_rev * P, n=size2)[:3*n-2]

        # symmetrize gradient
        grad = (grad + grad[::-1]) * 0.5

"""
Propose a low‑score step‑function using a large symmetric sequence and
continuous optimisation of the smoothed max‑convolution objective.
The algorithm:

1. Choose a power‑of‑two length n (default 32768) and initialise the
   sequence from a Beta(α,α) shape (α≈0.4) which already yields a fairly
   flat self‑convolution.
2. Repeatedly:
   - compute the full self‑convolution b = a∗a,
   - turn b into a probability distribution p that concentrates on the
     currently largest entries (smooth‑max with a temperature γ),
   - compute the gradient of the smooth‑max objective via a single FFT‑based
     cross‑correlation,
   - perform an Adam update, clip to non‑negative values and rescale to keep
     the required sum Σa = √(2n),
   - force exact symmetry a_i = a_{n‑1‑i}.
   The temperature γ is increased slowly so that the optimisation first
   smooths the landscape and later focuses on the true maximum.
3. Keep the best sequence (lowest fast objective) seen during the run.
4. If time permits, apply a cheap random‑mass‑transfer fine‑tuning.
5. Return the best sequence as a plain Python list; the final score is
   obtained with the provided ``evaluate_sequence``.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Helper: fast evaluation of the optimisation objective (max of self‑convolution)
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray) -> float:
    """Return 2·n·max(b)/sum(a)^2 where b = a∗a (computed via FFT)."""
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    size = 1 << ((2 * n - 2).bit_length())          # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Helper: generate a symmetric Beta(α,α)‑shaped sequence of length n
# ----------------------------------------------------------------------
def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """Create a length‑n sequence proportional to p^{α‑1}(1‑p)^{α‑1},
    p = (i+0.5)/n, and scale it so that its sum equals √(2·n)."""
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, alpha - 1.0)
    target = sqrt(2.0 * n)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    a = w * (target / w_sum)
    a = np.clip(a, 0.0, 1000.0)
    # enforce exact symmetry (helps the optimiser)
    a = (a + a[::-1]) * 0.5
    a *= target / a.sum()
    return a

# Main entry point required by the runner

    Uses a large symmetric sequence and Adam optimisation on a
    smoothed max‑convolution objective.
    """

    # ------------------------------------------------------------------

    a = _beta_sequence(n, 0.38)

    a += rng.normal(scale=1e-4 * S / n, size=n)

    a = (a + a[::-1]) * 0.5

    beta2 = 0.999

    lr = 0.20                     # base learning rate (will be scaled by Adam)

    gamma_max = 30.0
    gamma_factor = 1.015

    iteration = 0
    beta1_pow = 1.0
    beta2_pow = 1.0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        size1 = 1 << ((2 * n - 2).bit_length())

        conv_full = np.fft.irfft(A * A, n=size1)[: 2 * n - 1]

        beta2_pow *= beta2

            cand[j] += delta

                best_seq = cand.copy()

    # 4. Final accurate evaluation and return the candidate
    # ------------------------------------------------------------------
