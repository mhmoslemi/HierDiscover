def solve_lp_max_sum(a, M):
    n = a.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        # compute which b_j coefficients are nonzero: j from max(0, k-(n-1)) to min(n-1, k)
        # For each j, coefficient = a[k-j]
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        # vectorized assign
        # indexes j: j_start..j_end; a_idx = k - j (range)
        idxs = np.arange(j_start, j_end + 1)
        a_idxs = k - idxs
        A[k, idxs] = a[a_idxs]
    # constraints
    # Negative c for maximize sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None) for _ in range(n)]
    # Use scipy.optimize.linprog
    res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # maybe fallback to zeros or return small solution
        return np.zeros(n)
    b = res.x
    return b

"""
Hybrid Frank‑Wolfe + local search for the self‑convolution bound.

The algorithm works on a probability vector `p` (∑p_i = 1).  For a
sequence `a = p * √(2·n)` the evaluation function reduces to

    2·n·max(p * p),

so only the *shape* of `p` matters.  We minimise the maximal entry of the
self‑convolution by a Frank‑Wolfe style iteration:

1.  Start from the analytically optimal arcsine distribution.
2.  For the current vector `a` compute `M = max(a*a)`.
3.  Solve a linear program

        maximise   Σ b_i
        subject to  a * b ≤ M   (convolution, one inequality per index)
                     b_i ≥ 0

   The LP pushes mass to positions where `a` contributes little to the
   convolution.
4.  Rescale the LP solution `b` to the correct ℓ₁‑norm
    (`√(2·n)`) and mix it with the current iterate:

        a ← (1‑t)·a + t·b̂ ,

   with a slowly decreasing step‑size `t`.
5.  Occasionally apply cheap stochastic moves (block scaling,
   random transfers) to escape possible stagnation.
6.  Keep the best sequence seen and stop when the time budget expires or
   a target score (here 1.503) is reached.

All steps run in O(n²) time (n≈256 gives ≈2·10⁵ elementary operations per
iteration) and therefore fit comfortably into the 1000 s budget.
The final list of heights is returned as plain Python floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: arcsine (inverse‑square‑root) sequence
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the discrete arcsine (inverse‑square‑root) coefficients a_i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Evaluation of the exact objective (identical to evaluate_sequence
# except that we already respect the safety limits).
# ----------------------------------------------------------------------
def sequence_score(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = float(seq.sum())
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    max_conv = float(conv.max())
    n = seq.shape[0]
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Simple stochastic moves (used as occasional perturbations)
# ----------------------------------------------------------------------
def renormalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale *seq* so that its ℓ₁‑norm equals *target_sum* (after clipping)."""
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur == 0.0:
        return seq
    return seq * (target_sum / cur)


def block_perturbation(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # optional occasional symmetrisation
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return renormalise(cand, target_sum)


def transfer_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """Move a random fraction (≤30 %) of mass from one index to another."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


# ----------------------------------------------------------------------
# Linear‑program sub‑problem (Frank‑Wolfe direction)
# ----------------------------------------------------------------------
def solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ b_i
    s.t.    (a * b)_k ≤ M   for every convolution index k,
            b_i ≥ 0.

    Returns the optimal vector `b`.  The problem is a small LP (O(n²)
    constraints) and is solved with SciPy's HiGHS interior‑point solver.
    """
    n = a.size
    rows = 2 * n - 1

    # Build the constraint matrix A such that A @ b = a * b
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        # non‑zero entries appear for j where 0 ≤ k‑j < n  ⇔  max(0,k‑(n‑1)) ≤ j ≤ min(k,n‑1)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        j_idxs = np.arange(j_start, j_end + 1, dtype=int)
        i_idxs = k - j_idxs                     # i = k‑j
        A[k, j_idxs] = a[i_idxs]               # coefficient a_i

    # Linear programming – maximise Σ b_i  ⇔ minimise –Σ b_i
    c = -np.ones(n, dtype=np.float64)          # because scipy does minimise
    bounds = [(0, None) for _ in range(n)]
    b_ub = np.full(rows, M, dtype=np.float64)

    # HiGHS (the default for method='highs') is fast for these sizes
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 1.0})
    if not res.success:
        # If the LP fails (unlikely) fall back to a zero vector.
        return np.zeros(n, dtype=np.float64)
    return res.x

    p = a_raw / a_raw.sum()                         # probability vector (ℓ₁ = 1)
    a = p * target_sum                               # actual heights

    cur_score = sequence_score(a)
    best_a = a.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Frank‑Wolfe loop
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    iteration = 0
    # step‑size schedule – start relatively aggressive, then decay
    t = 0.4
    while time.monotonic() < deadline:
        iteration += 1

        # current maximal self‑convolution
        conv_self = np.convolve(a, a)
        M = float(conv_self.max())

        # ---- LP direction -------------------------------------------------
        # we work with the *probability* version of a (sum = 1) only to keep
        # the coefficients numerically small – the bound M scales quadratically,
        # so we divide it by target_sum**2.
        p_cur = a / target_sum
        M_prob = M / (target_sum ** 2)

        sum_b = b_prob.sum()

        a_new = np.clip(a_new, 0.0, 1000.0)

            cur_score = new_score

                best_score = new_score

            if best_score < 1.503:

        # Every few iterations throw in a random perturbation to escape
        # possible plateaus.

            if move_type < 0.5:

            else:

            cand_score = sequence_score(cand)

            t = max(0.05, t * 0.92)

    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best_a]
