"""
Simple‐gradient descent with heavy softmax smoothing and momentum.

The algorithm starts from a reasonably good sequence (the previous best if
available, otherwise a uniform vector).  It iteratively reduces the
soft‑max approximation of the self‑convolution peak while keeping the total
mass fixed (∑a_i = √(2n)).  A linearly increasing “temperature”
parameter β drives the soft‑max from a smooth to a hard max, and a
momentum‑based step‑size schedule performs a projected gradient update.
Occasional random multiplicative perturbations and a final short simulated‑
annealing phase help escape plateaus.  The routine stops when the supplied
time budget expires and returns the best sequence found (converted to a
plain Python list of floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT length)."""
    return 1 << (x - 1).bit_length()


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using real FFTs.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)
    return conv_full[: 2 * a.size - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full correlation (w ⋆ a) = convolution(w[::-1], a) via FFTs.
    Returns array of length len(w)+len(a)-1 ( = 2*len(a)-1 when w, a same size).
    """
    # reverse w for correlation
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, fft_len)
    fa = np.fft.rfft(a, fft_len)
    corr_full = np.fft.irfft(fw * fa, fft_len)
    return corr_full[: len(w) + len(a) - 1]


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑score non‑negative step‑function within the given time
    budget (seconds).  Returns a list of floats suitable for
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(int(time.time() * 1000) & 0xFFFFFFFF)

    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # 1. Choose working length
    # ------------------------------------------------------------------
    try:
        # Use length of previously discovered construction if present
        n = int(np.asarray(height_sequence_1).size)
    except Exception:
        n = 5000  # a reasonable default

    # Clamp n to a range that keeps the runtime manageable
    n = int(np.clip(n, 2000, 8000))

    target_sum = np.sqrt(2.0 * n)          # normalisation used in the original paper
    fft_len = _next_pow2(3 * n - 2)        # enough for all required FFTs

    # ------------------------------------------------------------------
    # 2. Build a small collection of starting points
    # ------------------------------------------------------------------
    starters = []

    # (a) uniform vector – simple baseline
    starters.append(np.full(n, target_sum / n))

    # (b) random positive vector (Dirichlet‑like)
    starters.append(rng.random(n) * target_sum)

    # (c) sparse random – many zeros, few random entries
    k_sparse = max(10, n // 20)            # about 5 % non‑zero entries
    a_sparse = np.zeros(n)
    idx = rng.choice(n, size=k_sparse, replace=False)
    a_sparse[idx] = rng.random(k_sparse)
    if a_sparse.sum() > 0:
        a_sparse = a_sparse * (target_sum / a_sparse.sum())
        starters.append(a_sparse)

            else:
