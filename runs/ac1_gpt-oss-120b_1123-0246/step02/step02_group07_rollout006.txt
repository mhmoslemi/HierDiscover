fast_score(seq):
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf, None, None
    n = len(seq)
    size = 1 << ((2*n-1).bit_length())
    # Could precompute size constant for each n.
    # compute FFT
    X = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(X*X, n=size)[:2*n-1]
    max_conv = conv.max()
    score = 2.0 * n * max_conv / (s*s)
    # compute max index (argmax among first n elements)
    max_idx = int(np.argmax(conv[:n]))
    return score, conv, max_idx

"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

by starting from the analytically optimal arcsine distribution,
exploring several low‑dimensional parametric families,
optimising a tiny spline representation with a CMA‑ES style evolution,
and finally polishing the result with fast stochastic local moves.
All operations respect the security checks of ``evaluate_sequence``.
The routine respects the given time budget (default 500 s, up to 1000 s)
and returns the best sequence found.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# 0.  Fast objective (same semantics as evaluate_sequence, but much quicker)
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray):
    """Return (score, full convolution, argmax index in the first half).

    The computation mirrors ``evaluate_sequence``:
    * clip to [0,1000]
    * reject sums <0.01 (return np.inf)
    * compute full convolution via FFT (real‑valued)
    * compute 2·n·max(conv) / sum^2
    """
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s < 0.01:
        return np.inf, None, None
    n = seq.shape[0]
    # length for FFT – next power of two >= 2*n-1
    size = 1 << ((2 * n - 1).bit_length())
    X = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(X * X, n=size)[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)        # tiny negative noise → 0
    max_conv = conv.max()
    score = 2.0 * n * max_conv / (s * s)
    # the maximal coefficient is always attained somewhere in the first n entries
    max_idx = int(np.argmax(conv[:n]))
    return float(score), conv, max_idx


# ----------------------------------------------------------------------
# 1.  Helper – build the “arcsine’’ (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    # recurrence: a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# 2.  Simple stochastic moves (kept from the reference code)
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# 3.  CMA‑ES‑style optimiser on a tiny spline (K knots)
# ----------------------------------------------------------------------
def evolve_spline(base_seq: np.ndarray,
                  init_knots: np.ndarray,
                  rng: np.random.Generator,
                  deadline: float,
                  best_overall_score: float,
                  best_overall_seq: np.ndarray):
    """Evolve a piecewise‑linear spline with K control points.
    Returns the best sequence seen inside the time window."""
    n = base_seq.shape[0]
    K = init_knots.shape[0]
    # positions of the knots (0‑based, inclusive)
    knot_pos = np.linspace(0, n - 1, K)

    # CMA‑ES parameters (very small population, isotropic)
    sigma = max(init_knots.max() * 0.2, 0.05)  # step size
    lambda_ = max(4, K * 4)                  # offspring per generation
    mu = lambda_ // 2
    # recombination weights (log‑scale)
    w = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    w /= w.sum()

    # current mean
    mean = init_knots.copy()
    best_seq = best_overall_seq.copy()
    best_score = best_overall_score

    it = 0
    while time.monotonic() < deadline:
        it += 1
        offspring_knots = []
        offspring_scores = []
        # generate offspring
        for _ in range(lambda_):
            cand_knots = mean + sigma * rng.normal(size=K)
            cand_knots = np.clip(cand_knots, 0.0, None)   # non‑negative
            # build full sequence by linear interpolation of the knots
            cand_seq = np.interp(np.arange(n), knot_pos, cand_knots)
            cand_seq = np.clip(cand_seq, 0.0, 1000.0)
            sc, _, _ = fast_score(cand_seq)
            offspring_knots.append(cand_knots)
            offspring_scores.append(sc)
            if sc < best_score:
                best_score = sc
                best_seq = cand_seq
                print(f"[cma] iter {it:4d} new best {best_score:.6f}")

        # select the μ best offspring
        order = np.argsort(offspring_scores)[:mu]
        selected = [offspring_knots[i] for i in order]

        # recompute the mean (weighted)
        new_mean = np.zeros_like(mean)
        for weight, knot_vec in zip(w, selected):
            new_mean += weight * knot_vec
        mean = new_mean

        # simple 1/5‑success rule for sigma adaptation
        success_rate = sum(sc < best_score for sc in offspring_scores) / lambda_
        if success_rate > 0.2:
            sigma *= 1.2
        elif success_rate < 0.2:
            sigma *= 0.85

        # early stop if we already beat the target
        if best_score < 1.503:
            print("[cma] target reached")
            break

        # occasional trace
        if it % 10 == 0:
            print(f"[cma] iter {it:4d} best {best_score:.6f} sigma {sigma:.4f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# 4.  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a step‑function that minimises the evaluation function.
    The algorithm consists of several phases:
    1.  Initialise with the analytical arcsine distribution (and with any
        pre‑computed sequence found in ``height_sequence_1``).
    2.  Exhaustive sweep over a few low‑dimensional families
        (tilt, power, exponential weighting).
    3.  Optimise a tiny spline (K≈10 points) using a lightweight CMA‑ES
        optimiser.
    4.  Finish with a directed stochastic local search that moves
        mass from high‑impact to low‑impact indices and occasional block
        perturbations.
    The routine respects the given time budget (default 500 s, as
    supplied by the harness up to 1000 s) and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size (length of the known sequence)
    # --------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256                     # fallback if nothing else is known

    # ------------------------------------------------------------------
    # Phase 1 – analytical starting point
    # ------------------------------------------------------------------
    arcsine_seq = build_arcsine_sequence(n)
    best_seq = arcsine_seq.copy()
    best_score, _, _ = fast_score(best_seq)
    print(f"[init] length={n}, arcsine score={best_score:.6f}")

    # also try the previously supplied construction if any
    if "height_sequence_1" in globals():
        supplied = np.array(globals()["height_sequence_1"], dtype=float)
        sc, _, _ = fast_score(supplied)
        if sc < best_score:
            best_seq, best_score = supplied, sc
            print(f"[init] supplied sequence score={best_score:.6f}")
        else:
            print(f"[init] supplied sequence not better (score={sc:.6f})")

    # ------------------------------------------------------------------
    # Phase 2 – cheap parametric sweeps
    # ------------------------------------------------------------------
    rev = arcsine_seq[::-1]

    # tilt with reversed arcsine (α ∈ [0,0.9])
    for alpha in np.linspace(0.0, 0.9, 19):
        cand = (1.0 - alpha) * arcsine_seq + alpha * rev
        sc, _, _ = fast_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"[sweep] tilt α={alpha:.3f} → {best_score:.6f}")

    # power‑law deformation (p ∈ [0.3,1.2])
    for p in np.linspace(0.3, 1.2, 19):
        cand = np.power(arcsine_seq, p)
        sc, _, _ = fast_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"[sweep] power p={p:.3f} → {best_score:.6f}")

    # combined tilt + power
    for alpha in np.linspace(0.0, 0.6, 13):
        mixed = (1.0 - alpha) * arcsine_seq + alpha * rev
        for p in np.linspace(0.5, 0.9, 9):
            cand = np.power(mixed, p)
            sc, _, _ = fast_score(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc
                print(f"[sweep] α={alpha:.2f}, p={p:.2f} → {best_score:.6f}")

    # exponential weighting a[i] *= λ**i  (λ ∈ [0.5,1.5])
    idx = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.5, 1.5, 11):
        cand = arcsine_seq * (lam ** idx)
        sc, _, _ = fast_score(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"[sweep] λ={lam:.2f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – spline optimisation with a tiny CMA‑ES
    # ------------------------------------------------------------------
    K = 10                                 # number of spline knots
    knot_positions = np.linspace(0, n - 1, K)
    init_knots = np.interp(knot_positions, np.arange(n), arcsine_seq)

    # allocate a small time slice for the CMA‑ES run (≈ 40 % of budget)
    now = time.monotonic()
    cma_deadline = now + max(0.0, float(budget_s) * 0.4 - 5.0)  # leave slack for phase 4
    best_seq, best_score = evolve_spline(
        arcsine_seq,
        init_knots,
        rng,
        cma_deadline,
        best_score,
        best_seq,
    )

    # ------------------------------------------------------------------
    # Phase 4 – directed stochastic local search (remainder of the budget)
    # ------------------------------------------------------------------
    deadline = now + float(budget_s) - 2.0   # a tiny safety margin
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # decide which move to apply
        if rng.random() < 0.6:
            # ---- directed transfer move (using the current max‑conv index) ----
            _, conv, max_idx = fast_score(best_seq)
            # contributions to the active convolution entry
            idx_range = np.arange(max_idx + 1)
            contrib = best_seq[idx_range] * best_seq[max_idx - idx_range]
            src = int(np.argmax(contrib))
            # pick a low‑value target (20‑th percentile heuristic)
            low_thr = np.percentile(best_seq, 20)
            low_cands = np.where(best_seq <= low_thr)[0]
            if low_cands.size == 0:
                tgt = int(rng.integers(0, n))
                if tgt == src:
                    tgt = (src + 1) % n
            else:
                tgt = int(rng.choice(low_cands))
                if tgt == src:
                    # try another low candidate
                    alt = low_cands[low_cands != src]
                    tgt = int(rng.choice(alt)) if alt.size > 0 else (src + 1) % n
            # move a random fraction (1‑% … 21‑%)
            frac = float(rng.random() * 0.20 + 0.01)
            delta = frac * best_seq[src]
            if delta > 0.0 and src != tgt:
                cand = best_seq.copy()
                cand[src] -= delta
                cand[tgt] += delta
                cand = np.clip(cand, 0.0, 1000.0)
                sc, _, _ = fast_score(cand)
                if sc < best_score:
                    best_seq, best_score = cand, sc
                    print(f"[{it:>6}] directed move improved to {best_score:.6f}")
                    if best_score < 1.503:
                        print("[stop] target reached")
                        break
        else:
            # ---- generic block perturbation ----
            cand = block_perturbation(best_seq, rng)
            sc, _, _ = fast_score(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc
                print(f"[{it:>6}] block perturb improved to {best_score:.6f}")
                if best_score < 1.503:
                    print("[stop] target reached")
                    break

    print(f"[final] best score {best_score:.6f} after {it} local‑search iterations")
    # Return as plain Python list of floats
    return [float(x) for x in best_seq]
