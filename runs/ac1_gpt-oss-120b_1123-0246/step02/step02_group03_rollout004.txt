import cvxpy as cp

def solve_lp_max_sum(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)
```

from scipy.optimize import linprog

def solve_lp_scipy(f, M, target_sum):
    n = f.size
    rows = 2*n - 1
    # Build constraints matrix A
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]
    # Constraint: A @ g <= M
    b = np.full(rows, M, dtype=np.float64)
    # Objective
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    bounds = [(0, target_sum) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        g = np.maximum(res.x, 0.0)
        return g
    else:
        return None
```

def solve_min_M_lp(f, target_sum):
    n = f.size
    rows = 2*n - 1
    # Build A similar for convolution constraints
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]
    # Build A_ub: each row has A[i] for g, and -1 for M variable
    A_ub = np.hstack([A, -np.ones((rows, 1))])
    b_ub = np.zeros(rows)
    # Equality: sum(g) = target_sum
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([target_sum])
    # Bounds: g >=0, M>=0
    bounds = [(0, None)] * n + [(0, None)]
    # Solve
    res = linprog(c=np.r_[np.zeros(n), 1.0], A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.success:
        g = res.x[:n]
        M_opt = res.x[-1]
        return g, M_opt
    else:
        return None, None
```

"""
Propose a step‑function sequence that minimizes
    2·n·max_k (a∗a)_k / (sum a)^2
using a two‑phase strategy:

1.  Small‑scale linear programming (LP) on a down‑sampled vector.
    For a fixed length n₀ we minimise the bound M such that
        (f * g)_k ≤ M for all lags k,
    with the additional constraints  g ≥ 0  and  sum(g) = √(2 n₀).
    The LP yields a new feasible vector g whose autocorrelation peak
    is provably no larger than the previous one.  We mix the old and
    the new vector and repeat a few times.

2.  Upscale the resulting pattern to the target length (≈ the length of
    the previously known construction) and refine it with an
    Adam‑optimised soft‑max approximation of the max‑autocorrelation.
    The optimisation respects the non‑negativity and sum constraints,
    performs periodic log‑normal perturbations and restarts on stagnation.

The algorithm keeps the best sequence seen throughout the run and
returns it before the allotted time expires.
"""

import time
import math
import numpy as np

# Optional import: a linear programme solver from SciPy.
# If SciPy is not available, the LP phase is skipped.
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a power‑of‑two FFT."""
    n = x.size + y.size - 1
    N = 1 << (n - 1).bit_length()  # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv


def _grad_and_maxb(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the soft‑max smoothed objective
        Φ_β(a) = Σ_k w_k (a∗a)_k,
    where w_k ∝ exp(β·((a∗a)_k − max_b)).
    Returns (gradient, exact maximal autocorrelation).
    """
    n = a.size
    # Exact autocorrelation (needed for the real score)
    b = _fft_convolve(a, a)
    maxb = b.max()

    # Soft‑max weights – subtraction of maxb for numerical stability
    if beta <= 0:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        exp_term = np.exp(beta * (b - maxb))
        w = exp_term / exp_term.sum()

    # Gradient of Σ_k w_k (a∗a)_k :
    #   ∂/∂a_i = 2 Σ_k w_k a_{k‑i}
    w_rev = w[::-1]
    conv_rev = _fft_convolve(a, w_rev)  # length = 3n‑2
    idx = (2 * n - 2) - np.arange(n)     # indices picking the correct slice
    grad = 2.0 * conv_rev[idx]

    # Project onto the hyperplane Σ a_i = const (i.e. zero‑sum gradient)
    grad -= grad.mean()
    return grad, maxb


def _evaluate(seq: np.ndarray) -> float:
    """
    Fast evaluation of the official scoring function.
    It reproduces evaluate_sequence(seq) without the safety checks.
    """
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    b = _fft_convolve(seq, seq)
    maxb = b.max()
    return float(2.0 * seq.size * maxb / (s * s))


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the (2·n‑1) × n matrix A such that (A @ g)_k = (f∗g)_k.
    The matrix is banded: each column i contains f[i] on rows i … i+n‑1.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def _solve_min_M_lp(f: np.ndarray, target_sum: float):
    """
    Solve the linear programme:
        minimise   M
        subject to A @ g - M ≤ 0   (i.e. (f∗g)_k ≤ M for all k)
                   sum(g) = target_sum
                   0 ≤ g_i ≤ 1000
        variables   g ∈ ℝⁿ , M ∈ ℝ.
    Returns (g, M_opt) if a solution is found, otherwise (None, None).
    """
    if linprog is None:
        return None, None

    n = f.size
    rows = 2 * n - 1
    A = _build_conv_matrix(f)                 # shape (rows, n)

    # Inequality constraints A @ g - M ≤ 0
    A_ub = np.hstack([A, -np.ones((rows, 1))])   # last column = -M
    b_ub = np.zeros(rows)

    # Equality constraint: sum(g) = target_sum
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([target_sum])

    # Objective: minimise M  →  c = [0,…,0, 1]
    c = np.concatenate([np.zeros(n), np.array([1.0])])

    # Variable bounds
    bounds = [(0.0, 1000.0) for _ in range(n)] + [(0.0, None)]  # M non‑negative

    # Solve with HiGHS (fast for moderate sizes)
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method='highs',
            options={'presolve': True},
        )
    except Exception:  # pragma: no cover
        return None, None

    if not res.success or res.x is None:
        return None, None

    g = res.x[:n]
    M_opt = res.x[-1]
    return g, M_opt
