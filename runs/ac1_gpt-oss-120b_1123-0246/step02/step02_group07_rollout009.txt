"""
Gradient‑descent optimiser for the self‑convolution bound.

The optimiser starts from the analytically optimal arcsine shape,
normalises the vector to unit ℓ¹‑norm (the objective is invariant
under scaling) and then runs a deterministic projected gradient
descent on a smooth approximation of the max‑convolution
objective.  The approximation is a soft‑max (log‑sum‑exp) with
a temperature that is gradually increased, turning the loss
into the true ``max``.  After the annealing phase a short
stochastic local‑search (directed mass transfer and block
perturbations) is applied to escape possible shallow minima.

The routine respects a time budget (default 1000 s) and returns
the best sequence found.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# 1.  Build the “arcsine’’ (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i  (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence: a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a   # values already lie in (0, 1] and decay like i⁻¹ᐟ²


# ----------------------------------------------------------------------
# 2.  Simple stochastic perturbations (kept for the final polishing).
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0, σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def directed_transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a fraction of mass from the index that contributes most
    to the current maximal convolution coefficient to a low‑impact index."""
    n = base.size
    conv = np.convolve(base, base)[:n]          # only first n values matter for the max
    max_idx = int(np.argmax(conv))

    # contributions to the active convolution entry
    idx = np.arange(max_idx + 1)
    contrib = base[idx] * base[max_idx - idx]

    src = int(np.argmax(contrib))               # most harmful index
    low_thr = np.percentile(base, 20)
    low_candidates = np.where(base <= low_thr)[0]
    if low_candidates.size == 0:
        tgt = int(rng.integers(0, n - 1))
        if tgt >= src:
            tgt += 1
    else:
        tgt = int(rng.choice(low_candidates))
    if tgt == src:
        return base.copy()

    frac = rng.random() * 0.20 + 0.01          # 1 %–21 %
    delta = frac * base[src]
    if delta <= 0.0:
        return base.copy()

    cand = base.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# 3.  Utility: evaluate the objective (fast version, does not call
#     evaluate_sequence) – it is scale‑invariant, we keep ‖a‖₁ = 1.
# ----------------------------------------------------------------------
def objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Return 2·n·max(a∗a) for a vector with ℓ¹‑norm 1."""
    # convolution via FFT (real‑to‑real)
    fft_a = np.fft.rfft(a, n=L_fft)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L_fft)[:2 * n - 1]
    max_conv = np.max(conv_full)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# 4.  Main entry point required by the harness
# ----------------------------------------------------------------------

    """
    Return a list of floats – the best step‑function heights found
    within the given time budget.  The optimisation consists of
    three stages:
      1. Initialise from the arcsine distribution (ℓ¹‑norm 1).
      2. Annealed projected gradient descent on a smooth max‑conv
         surrogate (soft‑max with an increasing temperature).
      3. A short stochastic polishing phase (directed transfer /
         block perturbations).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size (length of the known construction)
    # --------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256                     # fallback – still a decent size

    print(f"[setup] problem size n = {n}")

    # --------------------------------------------------------------
    # Helper: next power of two (for FFT padding)
    # --------------------------------------------------------------
    def next_pow2(x):
        return 1 << (x - 1).bit_length()

    L_fft = next_pow2(2 * n)        # length for all real FFTs
    print(f"[setup] FFT length L = {L_fft}")

    # --------------------------------------------------------------
    # Stage 1 – initialise from arcsine, normalized to ℓ¹ = 1
    # --------------------------------------------------------------
    a0 = build_arcsine_sequence(n)
    a0 = a0 / np.sum(a0)            # ℓ¹‑norm = 1 (scale‑invariant)
    best = a0.copy()
    best_score = objective_score(best, n, L_fft)
    print(f"[init] arcsine objective = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – annealed projected gradient descent
    # --------------------------------------------------------------
    # Parameter schedule: (beta, steps, learning_rate)
    schedule = [
        (0.5, 2000, 0.12),
        (1.0, 2000, 0.06),
        (2.0, 2000, 0.03),
        (5.0, 2000, 0.012),
        (10.0, 2000, 0.006),
        (20.0, 2000, 0.003),
        (50.0, 2000, 0.0015),
        (100.0, 2000, 0.001),
    ]

    start_time = time.monotonic()
    total_iters = 0

    # We will reuse some temporary arrays to avoid reallocations
    for beta, steps, lr in schedule:
        # Adam state (optional – we use simple momentum here)
        m = np.zeros_like(best)
        v = np.zeros_like(best)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8

        for it in range(steps):
            total_iters += 1
            # ------------------------------------------------------
            # 2.1  compute convolution a∗a (full) via FFT
            # ------------------------------------------------------
            fft_a = np.fft.rfft(best, n=L_fft)
            conv_full = np.fft.irfft(fft_a * fft_a, n=L_fft)[:2 * n - 1]

            # ------------------------------------------------------
            # 2.2  soft‑max weights for the current temperature β
            # ------------------------------------------------------
            # stability: shift by max value
            max_conv = np.max(conv_full)
            # use shifted exponentials to avoid overflow
            exp_shift = np.exp(beta * (conv_full - max_conv))
            sum_exp = np.sum(exp_shift)
            # probability distribution over convolution indices
            p = exp_shift / sum_exp
            # soft‑max value (expectation of conv under p)
            M = np.dot(p, conv_full)
            # gradient‑related weights (see analysis)
            w = p * (1.0 + beta * (conv_full - M))

            # ------------------------------------------------------
            # 2.3  gradient of the smoothed max w.r.t. a
            #       g = 2 * conv(w, a_rev)_{slice}
            # ------------------------------------------------------
            rev_a = best[::-1]                     # reversed a
            fft_w = np.fft.rfft(w, n=L_fft)
            fft_rev_a = np.fft.rfft(rev_a, n=L_fft)
            grad_full = np.fft.irfft(fft_w * fft_rev_a, n=L_fft)[:3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]   # shape (n,)

            # ------------------------------------------------------
            # 2.4  Adam‑style update, followed by projection onto
            #      the ℓ¹‑simplex (non‑negative, sum = 1)
            # ------------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** total_iters)
            v_hat = v / (1.0 - beta2 ** total_iters)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = best - step
            cand = np.maximum(cand, 0.0)          # enforce non‑negativity
            sum_cand = np.sum(cand)
            if sum_cand == 0.0:
                cand = np.full_like(cand, 1.0 / n)
            else:
                cand /= sum_cand                  # project to ℓ¹ = 1

            # ------------------------------------------------------
            # 2.5  evaluate (fast) and keep the best seen so far
            # ------------------------------------------------------
            # reuse FFT of cand for the true objective (max conv)
            fft_cand = np.fft.rfft(cand, n=L_fft)
            conv_cand_full = np.fft.irfft(fft_cand * fft_cand, n=L_fft)[:2 * n - 1]
            cand_score = 2.0 * n * np.max(conv_cand_full)

            if cand_score < best_score:
                best = cand
                best_score = cand_score
                print(f"[{total_iters:>6}] new best = {best_score:.6f} (β={beta})")

            else:
                # even if not an improvement we still move to the new point
                best = cand

            # ------------------------------------------------------
            # 2.6  time check – leave early if budget exhausted
            # ------------------------------------------------------
            if time.monotonic() - start_time > budget_s - 5.0:
                print("[time] budget about to expire – leaving GD loop")
                break
        # break outer schedule if out of time
        if time.monotonic() - start_time > budget_s - 5.0:
            break

    # --------------------------------------------------------------
    # Stage 3 – short stochastic polishing (directed transfer /
    #            block perturbations).  This mimics the reference code,
    #            but starts from the gradient‑descent solution.
    # --------------------------------------------------------------
    deadline = start_time + max(0.0, float(budget_s) - 2.0)
    it = 0
    while time.monotonic() < deadline:
        it += 1
        if rng.random() < 0.7:
            cand = directed_transfer_move(best, rng)
        else:
            cand = block_perturbation(best, rng)

        # evaluate the true objective (fast via FFT)
        fft_cand = np.fft.rfft(cand, n=L_fft)
        conv_cand_full = np.fft.irfft(fft_cand * fft_cand, n=L_fft)[:2 * n - 1]
        cand_score = 2.0 * n * np.max(conv_cand_full)

        if cand_score < best_score:
            best = cand
            best_score = cand_score
            print(f"[polish {it:>6}] improved = {best_score:.6f}")

            if best_score < 1.503:
                print("[stop] target reached during polishing")
                break

    print(f"[final] best score {best_score:.6f} after {total_iters} GD + {it} polish iters")
    # Clip to the allowed range (0 … 1000) – values are already ≤1
    best = np.clip(best, 0.0, 1000.0)

    # Optionally double‑check using the provided evaluate_sequence
    # (the call is cheap because the sequence is already normalised)
    try:
        from __main__ import evaluate_sequence   # the harness injects it
        real_score = evaluate_sequence(best.tolist())
        print(f"[verify] evaluate_sequence reports {real_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best]

"""
Gradient‑descent optimiser for the self‑convolution bound.

The optimiser starts from the analytically optimal arcsine shape
(normalised to ℓ¹‑norm 1, which makes the objective invariant to
scaling).  It then runs a deterministic projected gradient descent
on a smooth surrogate of the max‑convolution (a soft‑max with an
annealing temperature).  After the annealing phase a short
stochastic polishing (directed transfer and block perturbations)
is performed.  The routine respects a time budget (default 1000 s)
and returns the best sequence found.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper – next power of two (used for FFT padding)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# 1.  Build the “arcsine’’ (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i  (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence: a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a   # values already lie in (0, 1] and decay like i⁻¹ᐟ²


# ----------------------------------------------------------------------
# 2.  Simple stochastic perturbations (kept for the final polishing).
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0, σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def directed_transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a fraction of mass from the index that contributes most
    to the current maximal convolution coefficient to a low‑impact index."""
    n = base.size
    conv = np.convolve(base, base)[:n]          # only first n values matter for the max
    max_idx = int(np.argmax(conv))

    # contributions to the active convolution entry
    idx = np.arange(max_idx + 1)
    contrib = base[idx] * base[max_idx - idx]

    src = int(np.argmax(contrib))               # most harmful index
    low_thr = np.percentile(base, 20)
    low_candidates = np.where(base <= low_thr)[0]
    if low_candidates.size == 0:
        tgt = int(rng.integers(0, n - 1))
        if tgt >= src:
            tgt += 1
    else:
        tgt = int(rng.choice(low_candidates))
    if tgt == src:
        return base.copy()

    frac = rng.random() * 0.20 + 0.01          # 1 %–21 %
    delta = frac * base[src]
    if delta <= 0.0:
        return base.copy()

    cand = base.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# 3.  Fast objective (does not call evaluate_sequence)
# ----------------------------------------------------------------------
def objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Return 2·n·max(a∗a) for a vector with ℓ¹‑norm 1."""
    fft_a = np.fft.rfft(a, n=L_fft)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L_fft)[:2 * n - 1]
    max_conv = np.max(conv_full)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# 4.  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of floats – the best step‑function heights found
    within the given time budget.  The optimisation consists of three
    stages: (1) initialise from the arcsine distribution, (2) annealed
    projected gradient descent on a smooth max‑convolution surrogate,
    and (3) a short stochastic polishing phase.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (length of the known construction)
    # ------------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256                     # a harmless fallback
    print(f"[setup] problem size n = {n}")

    # ------------------------------------------------------------------
    # FFT length (power of two) – used for every convolution
    # ------------------------------------------------------------------
    L_fft = next_pow2(2 * n)
    print(f"[setup] FFT length L = {L_fft}")

    # ------------------------------------------------------------------
    # Stage 1 – initialise from arcsine, normalise to ℓ¹ = 1
    # ------------------------------------------------------------------
    a0 = build_arcsine_sequence(n)
    a0 = a0 / np.sum(a0)           # scale‑invariant normalisation
    a_cur = a0.copy()
    best_seq = a0.copy()
    best_score = objective_score(best_seq, n, L_fft)
    print(f"[init] arcsine objective = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – annealed projected gradient descent
    # ------------------------------------------------------------------
    # schedule: (temperature β, iterations, learning rate)
    schedule = [
        (0.5, 2000, 0.12),
        (1.0, 2000, 0.06),
        (2.0, 2000, 0.03),
        (5.0, 2000, 0.012),
        (10.0, 2000, 0.006),
        (20.0, 2000, 0.003),
        (50.0, 2000, 0.0015),
        (100.0, 2000, 0.001),
    ]

    start_time = time.monotonic()
    total_iters = 0

    for beta, steps, lr in schedule:
        # Adam–style moments (reset at the beginning of each temperature)
        m = np.zeros_like(a_cur)
        v = np.zeros_like(a_cur)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8

        for it in range(steps):
            total_iters += 1

            # ----------------------------------------------------------
            # 2.1 compute convolution a_cur ∗ a_cur (full) via FFT
            # ----------------------------------------------------------
            fft_a = np.fft.rfft(a_cur, n=L_fft)
            conv_full = np.fft.irfft(fft_a * fft_a, n=L_fft)[:2 * n - 1]

            # ----------------------------------------------------------
            # 2.2 soft‑max weights for the current β
            # ----------------------------------------------------------
            max_conv = np.max(conv_full)
            exp_shift = np.exp(beta * (conv_full - max_conv))   # stability shift
            sum_exp = np.sum(exp_shift)
            p = exp_shift / sum_exp                               # probability vector
            M = np.dot(p, conv_full)                              # soft‑max value
            w = p * (1.0 + beta * (conv_full - M))

            # ----------------------------------------------------------
            # 2.3 gradient of the smoothed max w.r.t. a_cur
            # ----------------------------------------------------------
            rev_a = a_cur[::-1]
            fft_w = np.fft.rfft(w, n=L_fft)
            fft_rev_a = np.fft.rfft(rev_a, n=L_fft)
            grad_full = np.fft.irfft(fft_w * fft_rev_a, n=L_fft)[:3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]   # shape (n,)

            # ----------------------------------------------------------
            # 2.4 Adam update + projection onto the ℓ¹‑simplex (non‑negative,
            #     sum = 1).  The update is additive; after the step we clip
            #     negatives and renormalise.
            # ----------------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** total_iters)
            v_hat = v / (1.0 - beta2 ** total_iters)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a_cur - step
            cand = np.maximum(cand, 0.0)
            sum_cand = np.sum(cand)
            if sum_cand == 0.0:
                cand = np.full_like(cand, 1.0 / n)
            else:
                cand /= sum_cand  # project to ℓ¹ = 1

            # ----------------------------------------------------------
            # 2.5 evaluate true objective (fast via FFT) and keep best.
            # ----------------------------------------------------------
            fft_cand = np.fft.rfft(cand, n=L_fft)
            conv_cand_full = np.fft.irfft(fft_cand * fft_cand, n=L_fft)[:2 * n - 1]
            cand_score = 2.0 * n * np.max(conv_cand_full)

            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[{total_iters:>6}] new best = {best_score:.6f} (β={beta})")

            # always move to the new point
            a_cur = cand

            # ----------------------------------------------------------
            # 2.6 time check – break if budget is almost exhausted
            # ----------------------------------------------------------
            if time.monotonic() - start_time > budget_s - 5.0:
                print("[time] budget close – exiting gradient descent")
                break
        if time.monotonic() - start_time > budget_s - 5.0:
            break

    # ------------------------------------------------------------------
    # Stage 3 – stochastic polishing (directed transfer / block perturb.)
    # ------------------------------------------------------------------
    deadline = start_time + max(0.0, float(budget_s) - 2.0)
    polish_iters = 0
    a_polish = best_seq.copy()
    while time.monotonic() < deadline:
        polish_iters += 1
        if rng.random() < 0.7:
            cand = directed_transfer_move(a_polish, rng)
        else:
            cand = block_perturbation(a_polish, rng)

        # evaluate objective fast via FFT
        fft_cand = np.fft.rfft(cand, n=L_fft)
        conv_cand_full = np.fft.irfft(fft_cand * fft_cand, n=L_fft)[:2 * n - 1]
        cand_score = 2.0 * n * np.max(conv_cand_full)

        if cand_score < best_score:
            a_polish = cand
            best_seq = cand.copy()
            best_score = cand_score
            print(f"[polish {polish_iters:>6}] improved = {best_score:.6f}")

            if best_score < 1.503:
                print("[stop] target reached during polishing")
                break
        else:
            a_polish = cand   # continue polishing from the new point

    print(f"[final] best score {best_score:.6f} after "
          f"{total_iters} GD steps + {polish_iters} polish iters")
    # Clamp to the allowed range (0 … 1000). Values are already ≤1.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Optional verification using the provided evaluate_sequence
    try:
        real_score = evaluate_sequence(best_seq.tolist())
        print(f"[verify] evaluate_sequence reports {real_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]

"""
Gradient‑descent optimiser for the self‑convolution bound.

The optimiser starts from the analytically optimal arcsine shape
(normalised to ℓ¹‑norm 1, which makes the objective invariant to
scaling).  It then runs a deterministic projected gradient descent
on a smooth surrogate of the max‑convolution (a soft‑max with an
annealing temperature).  After the annealing phase a short
stochastic polishing (directed transfer and block perturbations)
is performed.  The routine respects a time budget (default 1000 s)
and returns the best sequence found.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper – next power of two (used for FFT padding)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# 1.  Build the “arcsine’’ (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i  (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence: a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a   # values already lie in (0, 1] and decay like i⁻¹ᐟ²


# ----------------------------------------------------------------------
# 2.  Simple stochastic perturbations (kept for the final polishing).
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0, σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def directed_transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a fraction of mass from the index that contributes most
    to the current maximal convolution coefficient to a low‑impact index."""
    n = base.size
    conv = np.convolve(base, base)[:n]          # only first n values matter for the max
    max_idx = int(np.argmax(conv))

    # contributions to the active convolution entry
    idx = np.arange(max_idx + 1)
    contrib = base[idx] * base[max_idx - idx]

    src = int(np.argmax(contrib))               # most harmful index
    low_thr = np.percentile(base, 20)
    low_candidates = np.where(base <= low_thr)[0]
    if low_candidates.size == 0:
        tgt = int(rng.integers(0, n - 1))
        if tgt >= src:
            tgt += 1
    else:
        tgt = int(rng.choice(low_candidates))
    if tgt == src:
        return base.copy()

    frac = rng.random() * 0.20 + 0.01          # 1 %–21 %
    delta = frac * base[src]
    if delta <= 0.0:
        return base.copy()

    cand = base.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# 3.  Fast objective (does not call evaluate_sequence)
# ----------------------------------------------------------------------
def objective_score(a: np.ndarray, n: int, L_fft: int) -> float:
    """Return 2·n·max(a∗a) for a vector with ℓ¹‑norm 1."""
    fft_a = np.fft.rfft(a, n=L_fft)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L_fft)[:2 * n - 1]
    max_conv = np.max(conv_full)
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
# 4.  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of floats – the best step‑function heights found
    within the given time budget.  The optimisation consists of three
    stages: (1) initialise from the arcsine distribution, (2) annealed
    projected gradient descent on a smooth max‑convolution surrogate,
    and (3) a short stochastic polishing phase.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (length of the known construction)
    # ------------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256                     # a harmless fallback
    print(f"[setup] problem size n = {n}")

    # ------------------------------------------------------------------
    # FFT length (power of two) – used for every convolution
    # ------------------------------------------------------------------
    L_fft = next_pow2(2 * n)
    print(f"[setup] FFT length L = {L_fft}")

    # ------------------------------------------------------------------
    # Stage 1 – initialise from arcsine, normalise to ℓ¹ = 1
    # ------------------------------------------------------------------
    a0 = build_arcsine_sequence(n)
    a0 = a0 / np.sum(a0)           # scale‑invariant normalisation
    a_cur = a0.copy()
    best_seq = a0.copy()
    best_score = objective_score(best_seq, n, L_fft)
    print(f"[init] arcsine objective = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – annealed projected gradient descent
    # ------------------------------------------------------------------
    # schedule: (temperature β, iterations, learning rate)
    schedule = [
        (0.5, 2000, 0.12),
        (1.0, 2000, 0.06),
        (2.0, 2000, 0.03),
        (5.0, 2000, 0.012),
        (10.0, 2000, 0.006),
        (20.0, 2000, 0.003),
        (50.0, 2000, 0.0015),
        (100.0, 2000, 0.001),
    ]

    start_time = time.monotonic()
    total_iters = 0

    for beta, steps, lr in schedule:
        # Adam moments (reset for each temperature)
        m = np.zeros_like(a_cur)
        v = np.zeros_like(a_cur)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8

        for it in range(steps):
            total_iters += 1

            # ------------------------------------------------------
            # 2.1 compute convolution a_cur ∗ a_cur (full) via FFT
            # ------------------------------------------------------
            fft_a = np.fft.rfft(a_cur, n=L_fft)
            conv_full = np.fft.irfft(fft_a * fft_a, n=L_fft)[:2 * n - 1]

            # ------------------------------------------------------
            # 2.2 soft‑max weights for the current β
            # ------------------------------------------------------
            max_conv = np.max(conv_full)
            exp_shift = np.exp(beta * (conv_full - max_conv))   # stability shift
            sum_exp = np.sum(exp_shift)
            p = exp_shift / sum_exp
            M = np.dot(p, conv_full)
            w = p * (1.0 + beta * (conv_full - M))

            # ------------------------------------------------------
            # 2.3 gradient of the smoothed max w.r.t. a_cur
            # ------------------------------------------------------
            rev_a = a_cur[::-1]
            fft_w = np.fft.rfft(w, n=L_fft)
            fft_rev_a = np.fft.rfft(rev_a, n=L_fft)
            grad_full = np.fft.irfft(fft_w * fft_rev_a, n=L_fft)[:3 * n - 2]
            grad = 2.0 * grad_full[n - 1:n - 1 + n]   # shape (n,)

            # ------------------------------------------------------
            # 2.4 Adam update + projection onto the ℓ¹‑simplex
            # ------------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** total_iters)
            v_hat = v / (1.0 - beta2 ** total_iters)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            cand = a_cur - step
            cand = np.maximum(cand, 0.0)
            sum_cand = np.sum(cand)
            if sum_cand == 0.0:
                cand = np.full_like(cand, 1.0 / n)
            else:
                cand /= sum_cand

            # ------------------------------------------------------
            # 2.5 evaluate true objective (fast via FFT) and keep best.
            # ------------------------------------------------------
            fft_cand = np.fft.rfft(cand, n=L_fft)
            conv_cand_full = np.fft.irfft(fft_cand * fft_cand, n=L_fft)[:2 * n - 1]
            cand_score = 2.0 * n * np.max(conv_cand_full)

            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[{total_iters:>6}] new best = {best_score:.6f} (β={beta})")

            a_cur = cand  # always move to the new point

            # ------------------------------------------------------
            # 2.6 time check – break if budget is almost exhausted
            # ------------------------------------------------------
            if time.monotonic() - start_time > budget_s - 5.0:
                print("[time] budget close – exiting gradient descent")
                break
        if time.monotonic() - start_time > budget_s - 5.0:
            break

    # ------------------------------------------------------------------
    # Stage 3 – stochastic polishing (directed transfer / block perturb.)
    # ------------------------------------------------------------------
    deadline = start_time + max(0.0, float(budget_s) - 2.0)
    polish_iters = 0
    a_polish = best_seq.copy()
    while time.monotonic() < deadline:
        polish_iters += 1
        if rng.random() < 0.7:
            cand = directed_transfer_move(a_polish, rng)
        else:
            cand = block_perturbation(a_polish, rng)

        # evaluate objective fast via FFT
        fft_cand = np.fft.rfft(cand, n=L_fft)
        conv_cand_full = np.fft.irfft(fft_cand * fft_cand, n=L_fft)[:2 * n - 1]
        cand_score = 2.0 * n * np.max(conv_cand_full)

        if cand_score < best_score:
            a_polish = cand
            best_seq = cand.copy()
            best_score = cand_score
            print(f"[polish {polish_iters:>6}] improved = {best_score:.6f}")

            if best_score < 1.503:
                print("[stop] target reached during polishing")
                break
        else:
            a_polish = cand   # continue polishing from the new point

    print(f"[final] best score {best_score:.6f} after "
          f"{total_iters} GD steps + {polish_iters} polish iters")

    # Clamp to the allowed range (0 … 1000). Values are already ≤1.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Optional verification using the provided evaluate_sequence
    try:
        real_score = evaluate_sequence(best_seq.tolist())
        print(f"[verify] evaluate_sequence reports {real_score:.6f}")
    except Exception:
        pass

    return [float(x) for x in best_seq]
