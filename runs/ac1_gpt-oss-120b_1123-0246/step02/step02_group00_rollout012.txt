def evaluate_sequence(sequence: list[float]) -> float:
    # Security checks...
    # Clamp negative to 0

def simplex_projection(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate – a fast deterministic+stochastic optimiser.

The algorithm works in three phases:

1. **Initialisation** – pick a good start vector from a fine grid of
   power‑law shapes a_i ∝ (i+β)^(-α).  The best candidate (lowest value of
   the evaluation function) becomes the starting point.

2. **Exact sub‑gradient descent** – at each iteration the convolution
   c = a*a is computed (FFT for speed).  All indices whose value is
   within a very small tolerance of the current maximum constitute the
   *active set*.  For each active index k the vector
   d_k[i] = a_{k‑i} (zero where out‑of‑range) is the gradient of the
   corresponding convolution entry.  The sum of these gradients is a
   sub‑gradient of the max‑convolution functional.  Moving a small step
   in the opposite direction, projecting back onto the probability simplex,
   strictly reduces the objective (or is accepted probabilistically while
   a temperature cools).

3. **Escapes** – with a modest probability we apply a random multiplicative
   block perturbation (log‑normal noise on a random contiguous block) and
   occasionally symmetrise the vector.  These moves let the search jump out
   of local minima.

All operations respect the non‑negativity, ≤ 1000 and normalisation
required by ``evaluate_sequence``.  The routine runs for the supplied time
budget (default 500 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level (no closures)
# ----------------------------------------------------------------------


def _max_convolution(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)[k] using an FFT (real → real)."""
    n = seq.size
    size = 2 * n - 1
    # real FFT is faster than the full complex FFT for real input
    freq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(freq * freq, n=size)
    return float(conv.max())


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (O(n log n)).
    """
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _shift_sum(seq: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """
    Compute Σ_{k∈ks} shift(seq, k) where shift(seq, k)[i] = a_{k‑i}
    (zero outside range).  Returned vector has the same length as seq.
    """
    n = seq.size
    out = np.zeros_like(seq)
    for k in ks:
        # valid i satisfy 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        if start > end:
            continue
        idx = np.arange(start, end + 1, dtype=np.int64)
        out[idx] += seq[k - idx]
    return out


def _gradient_direction(seq: np.ndarray, conv: np.ndarray,
                        rel_tol: float = 1e-7) -> np.ndarray:
    """
    Sub‑gradient direction that reduces the maximal convolution entry.
    All ks with conv[k] ≥ max(conv)·(1‑rel_tol) are taken into account.
    The resulting direction has zero sum (so the total mass stays unchanged)
    before the simplex projection.
    """
    max_val = conv.max()
    # absolute tolerance to guard against round‑off when max is zero
    eps = max(rel_tol * max_val, 1e-12)
    active = np.where(conv >= max_val - eps)[0]
    # sum of the shift vectors for the active set
    shift = _shift_sum(seq, active)
    # move *away* from the active shifts
    direction = -shift
    # enforce zero‑sum (projection onto the affine hyperplane Σx = 0)
    direction -= direction.mean()
    return direction


def _block_perturbation(base: np.ndarray,
                        rng: np.random.Generator,
                        sigma: float = 0.12,
                        sym_prob: float = 0.35) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    With probability ``sym_prob`` the result is symmetrised to keep the
    shape roughly balanced.
    """
    cand = base.copy()
    n = cand.size
    # block length between 0.5 % and ~8 % of the whole vector
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, sigma)))
    if rng.random() < sym_prob:
        cand = 0.5 * (cand + cand[::-1])
    # enforce the absolute bounds imposed by evaluate_sequence
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _evaluate_fast(seq: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for already clamped, normalised vectors.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = seq.size
    maxb = _max_convolution(seq)
    return float(2.0 * n * maxb)


# ----------------------------------------------------------------------
# Main entry point expected by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The function respects the 1000 s overall limit of the competition.
    Returns a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine length (use the supplied reference construction if present)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        # fallback: a moderately large size that keeps a single FFT cheap
        n = 2500

    # ------------------------------------------------------------------
    # 1) Build a power‑law grid and pick the best seed.
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)   # 1‑based indices
    best_seq = None
    best_score = float('inf')
    # finer grid around the theoretical exponent 0.5
    alphas = np.linspace(0.42, 0.60, 37)          # 0.42 … 0.60
    betas = [0.0, 0.25, 0.5, 1.0, 2.0]           # small shifts
    for alpha in alphas:
        for beta in betas:
            seq = 1.0 / ((idx + beta) ** alpha)
            seq = np.clip(seq, 0.0, 1000.0)
            total = seq.sum()
            if total == 0.0:
                continue
            seq /= total                     # normalise to sum = 1
            score = _evaluate_fast(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()

    # sanity fallback (should never happen)
    if best_seq is None:
        seq = 1.0 / np.sqrt(idx)
        seq /= seq.sum()
        best_seq = seq
        best_score = _evaluate_fast(seq)

    # ------------------------------------------------------------------
    # 2) Local optimisation – exact sub‑gradient + stochastic escapes
    # ------------------------------------------------------------------
    current_seq = best_seq.copy()
    current_score = best_score
    best_overall_seq = best_seq.copy()
    best_overall_score = best_score

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin

    # annealing / step‑size schedules
    eta_start = 0.30          # initial learning rate
    eta_end   = 5e-5          # final learning rate
    temp_start = 4e-2
    temp_end   = 5e-7
    block_prob = 0.22        # probability of a random block move

    it = 0
    last_improve_iter = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        prog = min(1.0, elapsed / (deadline - start_time))

        # exponential schedules for step size & temperature
        eta = eta_start * (eta_end / eta_start) ** prog
        temp = temp_start * (temp_end / temp_start) ** prog

        # --------------------------------------------------------------
        # Propose a candidate
        # --------------------------------------------------------------
        if rng.random() < block_prob:
            cand = _block_perturbation(current_seq, rng)
        else:
            conv = np.fft.irfft(
                np.fft.rfft(current_seq, n=2 * n - 1) *
                np.fft.rfft(current_seq, n=2 * n - 1), n=2 * n - 1
            )
            direction = _gradient_direction(current_seq, conv, rel_tol=1e-7)
            cand = current_seq + eta * direction
            cand = _simplex_projection(cand)

        # occasional symmetrisation (helps flatten the convolution)
        if rng.random() < 0.10:
            cand = 0.5 * (cand + cand[::-1])

        cand_score = _evaluate_fast(cand)

        # --------------------------------------------------------------
        # Acceptance rule (Metropolis style)
        # --------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # probability to accept a worsening move
            prob = np.exp(-(cand_score - current_score) / max(temp, 1e-16))
            if rng.random() < prob:
                accept = True

        if accept:
            current_seq = cand
            current_score = cand_score
            if cand_score < best_overall_score:
                best_overall_score = cand_score
                best_overall_seq = cand.copy()
                last_improve_iter = it
                print(f"iter {it:06d} – new best: {best_overall_score:.6f}")

        # occasional restart if stuck for a long time
        if it - last_improve_iter > 150000:
            # re‑sample a fresh power‑law seed (helps escaping plateaus)
            alpha = rng.uniform(0.4, 0.6)
            beta = rng.choice([0.0, 0.25, 0.5, 1.0])
            seq = 1.0 / ((idx + beta) ** alpha)
            seq /= seq.sum()
            seq = _simplex_projection(seq)
            current_seq = seq
            current_score = _evaluate_fast(seq)
            last_improve_iter = it
            print(f"iter {it:06d} – restart with fresh seed")

        it += 1

    # ----------------------------------------------------------------------
    # Final sanitisation (the external evaluator will clip again)
    # ----------------------------------------------------------------------
    best_overall_seq = np.clip(best_overall_seq, 0.0, 1000.0)
    best_overall_seq = _simplex_projection(best_overall_seq)

    print(f"finished after {it} iterations, best score = {best_overall_score:.6f}")
    return best_overall_seq.tolist()
