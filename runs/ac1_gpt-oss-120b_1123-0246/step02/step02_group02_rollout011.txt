def project_onto_simplex(v, s):

    # Duchi et al.

    n = len(v)

    u = np.sort(v)[::-1]

    cssv = np.cumsum(u) - s

    ind = np.arange(1, n+1)

    cond = u - cssv / ind > 0

    rho = ind[cond].max()

    theta = cssv[cond][-1] / rho

    w = np.maximum(v - theta, 0)

    return w

def project_onto_simplex(v, z):
    """Project the vector v onto the simplex {x >=0, sum x = z}."""
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all zero?
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function sequence that minimises the evaluation metric.

The algorithm works on a fixed length `n` (taken from the previous best if
available, otherwise a default).  The sequence is always projected onto the
simplex {a_i ≥ 0, Σ a_i = √(2 n)} so that `evaluate_sequence` returns exactly
the maximal self‑convolution value (the quantity we want to minimise).

Two complementary gradient‑based phases are used:

1. **L2‑flattening** – minimise the squared deviation of the convolution from
   the ideal constant value C = S²/(2 n‑1).  This spreads the mass away from
   the centre and reduces the peak of the convolution.

2. **Max‑focused sub‑gradient** – after a warm‑up the update direction is a
   blend of the L2‑gradient and the sub‑gradient of the current maximum
   convolution entry.  The blend weight grows linearly so that the final
   iterations concentrate on the true infinity norm.

Both gradients are computed efficiently with real FFTs.  After every update
the vector is projected onto the simplex (Duchi et al.) to keep it feasible.
Occasional random perturbations help escaping plateaus.  The best sequence
found (according to the exact `evaluate_sequence`) is returned.

The whole search stops before the provided time budget expires, guaranteeing
a valid result.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)
    return conv_full[: 2 * a.size - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full correlation (w ⋆ a) = convolution(w[::-1], a) using real FFTs.
    Returns an array of length len(w) + len(a) - 1.
    """
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, fft_len)
    fa = np.fft.rfft(a, fft_len)
    corr_full = np.fft.irfft(fw * fa, fft_len)
    return corr_full[: w.size + a.size - 1]


def _project_onto_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = z}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if z < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = 0.0
    else:
        rho = ind[cond][-1]          # last True index (0‑based)
        theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Main search routine
# ----------------------------------------------------------------------

        a = (idx + 1) ** p + (n - idx) ** p
        a = np.clip(a, 0, None)
        a = a - a.min() + 1e-12              # ensure positivity
        a = a / a.sum() * target_sum
        starts.append(a)

    # (e) reuse the previous best if it exists (interpolated to current n)
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            if prev.size != n:
                old_x = np.linspace(0.0, 1.0, prev.size)
                new_x = np.linspace(0.0, 1.0, n)
                prev = np.interp(new_x, old_x, prev)
            prev = np.clip(prev, 0.0, 1000.0)
            if prev.sum() > 0:
                prev = prev * (target_sum / prev.sum())
                starts.append(prev)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Core optimisation loop (multiple restarts)
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf           # we track max(b) because of the chosen scaling
    total_iters = 0

    # parameters controlling the optimisation dynamics
    max_iters_per_start = 12000          # generous but will be cut by the deadline
    step0 = 0.5                           # base step size (in the direction unit norm)
    grad_max_freq = 200                   # how often we compute the max‑subgradient

    for start_id, a0 in enumerate(starts):
        if time.monotonic() >= deadline:
            break

        # initialise this restart
        a = np.copy(a0)
        a = _project_onto_simplex(a, target_sum)

        for it in range(max_iters_per_start):
            now = time.monotonic()
            if now >= deadline:
                break

            # ------------------------------------------------------------------
            # (a) current self‑convolution and its maximum
            # ------------------------------------------------------------------
            b = _self_convolution_fft(a, fft_len)
            cur_max = b.max()

            # update the global best if we improved
            if cur_max < best_score:
                best_score = cur_max
                best_seq = a.copy()
                print(f"[global it {total_iters:6d}] new best max = {best_score:.6f}")

            # ------------------------------------------------------------------
            # (b) gradient of the L2 deviation from the ideal constant convolution
            # ------------------------------------------------------------------
            residual = b - ideal_const
            corr_l2 = _correlation_fft(residual, a, fft_len)
            L = residual.size                     # = 2*n - 1
            grad_l2 = 2.0 * corr_l2[L - 1 - np.arange(n)]

            # ------------------------------------------------------------------
            # (c) occasional max‑subgradient (binary mask of the active indices)
            # ------------------------------------------------------------------
            if it % grad_max_freq == 0:
                mask = (b == cur_max).astype(float)   # shape (2*n-1)
                corr_max = _correlation_fft(mask, a, fft_len)
                grad_max = 2.0 * corr_max[L - 1 - np.arange(n)]
                # blending factor – start with pure L2, finish with pure max
                alpha = min(1.0, it / max_iters_per_start)
                grad = (1.0 - alpha) * grad_l2 + alpha * grad_max
            else:
                grad = grad_l2

            # ------------------------------------------------------------------
            # (d) normalise the descent direction
            # ------------------------------------------------------------------
            grad_norm = np.linalg.norm(grad)
            if grad_norm == 0.0:
                direction = np.zeros_like(grad)
            else:
                direction = grad / grad_norm

            # ------------------------------------------------------------------
            # (e) step size schedule (decreasing like 1/√t)
            # ------------------------------------------------------------------
            eta = step0 / np.sqrt(it + 1)

            # ------------------------------------------------------------------
            # (f) candidate update + projection onto the simplex
            # ------------------------------------------------------------------
            cand = a - eta * direction
            cand = _project_onto_simplex(cand, target_sum)

            # ------------------------------------------------------------------
            # (g) accept if the peak actually decreased (monotone acceptance)
            # ------------------------------------------------------------------
            b_cand = _self_convolution_fft(cand, fft_len)
            cand_max = b_cand.max()
            if cand_max < cur_max:
                a = cand
                cur_max = cand_max
            else:
                # occasional stochastic acceptance to escape tiny plateaus
                if rng.random() < 0.001:
                    a = cand
                    cur_max = cand_max

            # ------------------------------------------------------------------
            # (h) periodic random jitter to diversify the search
            # ------------------------------------------------------------------
            if it % 1000 == 0 and it > 0:
                mult = np.exp(rng.normal(0.0, 0.02, size=n))
                a = _project_onto_simplex(a * mult, target_sum)

            total_iters += 1

    # ----------------------------------------------------------------------
    # 4. Final polishing on the best candidate (max–focused only)
    # ----------------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        for it in range(3000):
            if time.monotonic() >= deadline:
                break
            b = _self_convolution_fft(a, fft_len)
            cur_max = b.max()
            mask = (b == cur_max).astype(float)
            corr = _correlation_fft(mask, a, fft_len)
            L = mask.size
            grad = 2.0 * corr[L - 1 - np.arange(n)]
            grad_norm = np.linalg.norm(grad)
            if grad_norm == 0.0:
                break
            direction = grad / grad_norm
            eta = step0 * 0.5 / np.sqrt(it + 1)
            cand = a - eta * direction
            cand = _project_onto_simplex(cand, target_sum)
            b_cand = _self_convolution_fft(cand, fft_len)
            cand_max = b_cand.max()
            if cand_max < cur_max:
                a = cand
                cur_max = cand_max
        best_seq = a.copy()

    # ----------------------------------------------------------------------
    # 5. Final exact evaluation and return
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] best exact score = {final_score:.6f}")
    return [float(v) for v in best_seq]
