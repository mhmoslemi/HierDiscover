def _isotonic_decreasing(seq: np.ndarray) -> np.ndarray:
    # PAVA for decreasing sequence.
    n = seq.size
    # We'll work on negative values for increasing isotonic regression.
    # Use simple loop:
    blocks = []  # each block is (sum, count)
    for x in seq:
        blocks.append([x, 1])
        # while last two blocks have average of previous < average of current (since we want decreasing, i.e., average(prev) < average(curr) violates decreasing)
        while len(blocks) >= 2 and (blocks[-2][0]/blocks[-2][1]) < (blocks[-1][0]/blocks[-1][1]):
            # merge
            sum_new = blocks[-2][0] + blocks[-1][0]
            cnt_new = blocks[-2][1] + blocks[-1][1]
            blocks[-2] = [sum_new, cnt_new]
            blocks.pop()
    # produce output by expanding averages
    out = np.empty(n, dtype=seq.dtype)
    idx = 0
    for s, cnt in blocks:
        avg = s / cnt
        out[idx:idx+cnt] = avg
        idx += cnt
    # Ensure non-negativity: clip negative.
    out = np.maximum(out, 0.0)
    return out

"""
Propose a low‑value step‑function for the convolution minimisation problem.

Algorithm
---------

1️⃣ **Initialisation** – Build a pool of candidate seeds:
   * Power‑law shapes  a_i ∝ (i+β)^‑α  with a fine grid of (α,β).
   * Random Dirichlet draws.
   * The historic best sequence ``height_sequence_1`` if available.
   The best seed (according to the fast evaluation) becomes the start.

2️⃣ **Projected sub‑gradient descent** – At each iteration we compute the
   full convolution ``c = a*a`` and identify the *exact* set of indices
   where the maximum is attained.  The sub‑gradient of
   ``max_k (a*a)[k]`` is the average of the shift vectors
   ``_shift(a, k)`` over this set.  A zero‑sum direction
   ``d = -g + mean(g)`` is taken, the update ``a ← a + η·d`` is applied,
   the result is clipped to [0,1000] and renormalised to unit sum
   (projection onto the simplex).

3️⃣ **Escaping local minima** – With probability `p_block` a stochastic
   *block perturbation* multiplies a random contiguous block by a
   log‑normal factor.  With probability `p_iso` the current vector is
   projected onto the cone of non‑increasing sequences (isotonic regression
   via the PAVA algorithm).  An occasional symmetrisation
   ``a ← (a + a[::-1]) / 2`` helps to flatten the autocorrelation.

4️⃣ **Simulated‑annealing acceptance** – The temperature `T` follows an
   exponential cooling schedule.  Upshifts are accepted with probability
   ``exp(-(new‑old)/T)``; downhill moves are always accepted.

5️⃣ **Best‑track** – The globally best sequence encountered is stored.
   The loop runs until the supplied time budget expires (a 2 s safety
   margin is kept).  The final vector is clipped and normalised before
   returning; the external ``evaluate_sequence`` will re‑apply the same
   safety checks.

The method differs from the previous implementation by using the *exact*
active set of the max‑convolution, adding a monotone projection step,
and employing a simple simulated‑annealing acceptance rule.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _evaluate_fast(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` for already normalised vectors."""
    total = seq.sum()
    if total < 1e-12:                     # should never happen for our calls
        return float("inf")
    conv = np.convolve(seq, seq)          # full convolution (length 2n‑1)
    max_b = conv.max()
    n = seq.size
    return float(2.0 * n * max_b / (total * total))


def _shift(seq: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient contribution for convolution entry ``k``:
    (a*a)[k] = Σ_i a_i a_{k‑i}.  The derivative w.r.t. a_i is a_{k‑i}.
    """
    n = seq.size
    out = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1)
        out[idx] = seq[k - idx]
    return out


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    Occasionally the candidate is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)                # minimal block width
    hi = max(2, n // 12)                 # maximal block width
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:               # mild symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _isotonic_decreasing(seq: np.ndarray) -> np.ndarray:
    """
    Project ``seq`` onto the cone of non‑increasing sequences
    using the Pool Adjacent Violators Algorithm (PAVA).
    """
    n = seq.size
    # block sums and sizes
    sums = []
    sizes = []
    for val in seq:
        sums.append(float(val))
        sizes.append(1)
        # Merge while the average of the previous block is < average of the current
        while len(sums) >= 2 and (sums[-2] / sizes[-2]) < (sums[-1] / sizes[-1]):
            # merge last two blocks
            sums[-2] += sums[-1]
            sizes[-2] += sizes[-1]
            sums.pop()
            sizes.pop()
    # Expand the averaged blocks back to a full vector
    out = np.empty(n, dtype=seq.dtype)
    pos = 0
    for s, sz in zip(sums, sizes):
        avg = s / sz
        out[pos:pos + sz] = avg
        pos += sz
    # enforce non‑negativity (numerical safety)
    out[out < 0.0] = 0.0
    return out
