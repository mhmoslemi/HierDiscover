"""
Hybrid step‑function optimiser.

The optimiser combines three complementary moves:

1. **Power‑law seeding** – a fine grid over the exponent α and offset β
   is searched and the best seed (normalised to sum = 1) becomes the start.

2. **Projected stochastic gradient descent** – at each iteration the
   sub‑gradients of the largest convolution entries are averaged,
   a zero‑sum direction is taken (preserving the total mass) and the
   candidate is projected onto the simplex.

3. **Targeted LP‑boost** – periodically a tiny linear programme is solved.
   Using only the currently largest convolution indices (the “active set”)
   we maximise the total mass under the constraint that the convolution
   with the current sequence does not exceed its present maximum M.
   After scaling to the original ℓ₁‑norm this yields a strong jump
   towards a flatter convolution.

A simulated‑annealing style Metropolis rule accepts uphill moves early
on; the step‑size and temperature decay exponentially.  The search runs
until the supplied time budget expires and returns the best sequence found.
All entries are clipped to [0, 1000] and normalised to ℓ₁‑norm = 1, which
matches the scale‑invariance of ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _evaluate_fast(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` for non‑negative vectors (ℓ₁‑norm = 1)."""
    total = seq.sum()
    if total < 1e-12:                     # should never happen for normalised vectors
        return float("inf")
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    n = seq.size
    return float(2.0 * n * max_b / (total * total))


def _shift(seq: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of (seq*seq)[k] w.r.t. ``seq``.
    Returns a vector d with d[i] = seq[k-i] when the index is valid, else 0.
    """
    n = seq.size
    out = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1)
        out[idx] = seq[k - idx]
    return out


def _gradient_step(seq: np.ndarray, eta: float, top_k: int = 7) -> np.ndarray:
    """
    One projected gradient move.
    The gradient is the (negative) average of the shift‑vectors of the
    ``top_k`` largest convolution values.  The direction is centred to keep the
    ℓ₁‑mass unchanged, then the step is taken and projected back onto the simplex.
    """
    conv = np.convolve(seq, seq)
    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(-conv, top_k)[:top_k]

    shift_sum = np.zeros_like(seq)
    for k in ks:
        shift_sum += _shift(seq, int(k))
    d = shift_sum / len(ks)               # average sub‑gradient

    # zero‑sum direction (move mass away from entries with large d)
    direction = -d + np.mean(d)
    new_seq = seq + eta * direction
    new_seq = np.clip(new_seq, 0.0, 1000.0)

    s = new_seq.sum()
    if s > 0:
        new_seq /= s
    else:
        new_seq = seq.copy()
    return new_seq


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    With probability 0.35 the block‑perturbed vector is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _lp_step(seq: np.ndarray, top_c: int = 40) -> np.ndarray | None:
    """
    Solve a small linear programme that maximises the ℓ₁‑mass while keeping
    the convolution with ``seq`` below the current peak M.
    Only the ``top_c`` largest convolution indices are used as constraints.
    Returns a new candidate (scaled to the same ℓ₁‑norm as ``seq``) or ``None`` on failure.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    M = conv.max()

    # indices of the largest convolution entries
    if top_c >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(-conv, top_c)[:top_c]

    m = ks.size
    A = np.zeros((m, n), dtype=np.float64)

    for row, k in enumerate(ks):
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        idx = np.arange(start, end + 1)
        A[row, idx] = seq[k - idx]

    # maximise sum(b)  ==>  minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]
    b_ub = np.full(m, M, dtype=np.float64)

    # solve with HiGHS (fast, robust)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return None

    b = res.x
    b_sum = b.sum()
    if b_sum <= 0:
        return None

    # Rescale to preserve the original ℓ₁‑norm (keeps the objective scale‑invariant)
    scale = seq.sum() / b_sum
    cand = b * scale
    cand = np.clip(cand, 0.0, 1000.0)
    # Normalise again to avoid drift due to clipping
    s = cand.sum()
    if s > 0:
        cand /= s
    return cand

        if r < block_prob:
            cand = _make_block_perturbation(current_seq, rng)
        elif r < block_prob + lp_prob:
            cand = _lp_step(current_seq, top_c=60)
            if cand is None:
                # fallback to a cheap gradient step
                cand = _gradient_step(current_seq, eta, top_k=9)
        else:
            cand = _gradient_step(current_seq, eta, top_k=9)

        # occasional symmetrisation (helps flatten the convolution)
        if rng.random() < 0.10:
            cand = 0.5 * (cand + cand[::-1])

        # re‑normalise to ℓ₁‑norm = 1 (keeps the objective scale‑invariant)
        s = cand.sum()
        if s > 0:
            cand /= s
        else:
            cand = current_seq.copy()

        cand_score = _evaluate_fast(cand)

        # ----------------------------------------------------------
        # Metropolis‑style acceptance
        # ----------------------------------------------------------
        accept = False
        if cand_score < current_score - 1e-12:
            accept = True
        else:
            if temp > 1e-12:
                prob = np.exp(-(cand_score - current_score) / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            current_seq = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"iter {it:07d} – new best: {best_score:.6f}")

        it += 1

    # ----------------------------------------------------------------------
    # Final sanity checks – ensure the returned list satisfies the wrapper.
    # ----------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        # very unlikely, fall back to a uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"finished after {it} iterations, best score ≈ {best_score:.6f}")
    return best_seq.tolist()
