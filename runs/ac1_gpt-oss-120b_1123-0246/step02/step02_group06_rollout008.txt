"""
Propose a step‑function that minimizes `evaluate_sequence`.

The search combines three simple stochastic moves (transfer, block‑scale,
rebalance) with a custom gradient move that explicitly reduces the
convolution peak.  A linear cooling schedule yields a simulated‑annealing
acceptance rule.  The algorithm starts from several analytic candidates
(arcsine and a family of power‑law shapes), picks the best, and then
iterates until the time budget expires (default 500 s, the harness may
allow up to 1000 s).  The best sequence seen is returned as a plain
list of floats.
"""

import numpy as np
import time
from math import exp

# ----------------------------------------------------------------------
# Helper: build the analytically known arcsine sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative arcsine‑like sequence of given length."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Simple stochastic move primitives.
# ----------------------------------------------------------------------
def transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from an early index to a later one."""
    n = base.size
    i = int(rng.integers(0, max(1, n // 4)))
    j = int(rng.integers(max(1, 3 * n // 4), n))
    if base[i] == 0.0:
        return base.copy()
    frac = rng.random() * 0.9
    delta = frac * base[i]
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.13)))
    cand[start:stop] *= factor
    if rng.random() < 0.30:                      # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def rebalance_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Shift mass from an early prefix to a later suffix while roughly
    preserving the total sum."""
    n = base.size
    cut = int(rng.integers(n // 10, 9 * n // 10))
    prefix = base[:cut]
    suffix = base[cut:]
    sum_prefix = prefix.sum()
    sum_suffix = suffix.sum()
    if sum_prefix == 0.0:
        return base.copy()
    amount = rng.random() * 0.5 * sum_prefix
    factor_pre = max(0.0, (sum_prefix - amount) / sum_prefix)
    new_prefix = prefix * factor_pre
    if sum_suffix > 0.0:
        factor_suf = (sum_suffix + amount) / sum_suffix
        new_suffix = suffix * factor_suf
    else:
        new_suffix = np.full_like(suffix, amount / suffix.size)
    cand = np.concatenate([new_prefix, new_suffix])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Gradient move – explicitly reduces the current convolution maximum.
# ----------------------------------------------------------------------
def _gradient_for_k(base: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the convolution entry b_k = (base*base)[k]."""
    n = base.size
    grad = np.zeros_like(base)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i
    mask = i != j
    # i != j  -> derivative = 2 * base[j]
    grad[i[mask]] = 2.0 * base[j[mask]]
    # i == j  -> derivative = base[i]
    grad[i[~mask]] = base[i[~mask]]
    return grad


def gradient_move(base: np.ndarray,
                  rng: np.random.Generator,
                  temperature: float) -> np.ndarray:
    """Take a step in the direction that most reduces the current max
    convolution value.  The step size is proportional to the temperature."""
    n = base.size
    conv = np.convolve(base, base)
    max_b = conv.max()
    # indices where the maximum is attained (or within a tiny tolerance)
    ks = np.where(conv >= max_b * (1.0 - 1e-12))[0]
    if ks.size == 0:
        ks = np.array([np.argmax(conv)], dtype=np.int64)
    # if many, pick a few at random to keep work cheap
    if ks.size > 3:
        ks = rng.choice(ks, size=3, replace=False)

    # average gradient over the selected critical indices
    grad = np.zeros_like(base)
    for k in ks:
        grad += _gradient_for_k(base, int(k))
    grad /= float(ks.size)

    # direction that lowers the peak; project to zero‑sum hyperplane
    direction = -grad
    direction -= np.mean(direction)               # keep total mass roughly constant

    # scale the step – the magnitude is tied to the temperature
    max_abs = np.max(np.abs(direction))
    if max_abs == 0.0:
        return base.copy()
    # a reasonable scale factor: temperature controls how aggressive the move is
    step_scale = temperature * np.max(base) * 0.5 + 1e-12
    step = direction * (step_scale / max_abs)
    cand = base + step
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a step‑function with a low `evaluate_sequence` score.
    Uses a mixture of stochastic moves and a gradient‑based move,
    governed by a simple simulated‑annealing schedule.
    Returns the best sequence (list of floats) found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length to work with.
    # ------------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256

    # ------------------------------------------------------------------
    # Build a small portfolio of analytic starting points.
    # ------------------------------------------------------------------
    candidates = [build_arcsine_sequence(n)]

    # power‑law family a_i = (i+1)^(-p)   (p in [0.4, 0.8])
    for p in np.linspace(0.4, 0.8, 9):
        a = np.array([1.0 / ((i + 1) ** p) for i in range(n)], dtype=np.float64)
        candidates.append(a)

    # pick the best among the analytic seeds
    cur = min(candidates, key=lambda a: evaluate_sequence(a.tolist()))
    cur_score = float(evaluate_sequence(cur.tolist()))
    best = cur.copy()
    best_score = cur_score

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing / gradient descent loop.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # 1 s safety margin
    it = 0

    # Linear temperature schedule
    T_start = 0.04
    T_end = 1e-5

    while time.monotonic() < deadline:
        it += 1
        elapsed = (time.monotonic() - start_time) / budget_s
        T = T_start * (1.0 - elapsed) + T_end * elapsed

        # choose a move type
        r = rng.random()
        if r < 0.30:
            cand = transfer_move(cur, rng)
        elif r < 0.55:
            cand = block_perturbation(cur, rng)
        elif r < 0.80:
            cand = rebalance_move(cur, rng)
        else:
            cand = gradient_move(cur, rng, T)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
            continue

        # acceptance rule (Metropolis)
        if cand_score < cur_score:
            accept = True
        else:
            prob = exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[{it:>6}] new best = {best_score:.6f}")

        # occasional reset to the best seen (helps maintain diversity)
        if it % 30000 == 0:
            cur = best.copy()
            cur_score = best_score

        # early exit if we already beat a comfortable target
        if best_score < 1.45:
            break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return a plain Python list as required by the harness.
    return [float(x) for x in best]
