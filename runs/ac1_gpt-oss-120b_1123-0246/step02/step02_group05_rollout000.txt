"""
Hybrid FFT‑based gradient descent for the convolution‑bound problem.

The objective to minimise is

    f(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

which is invariant under a global scale, so we fix the ℓ₁‑norm to 1.0.
The algorithm proceeds as follows:

1. Initialise the coefficients with the analytically near‑optimal “arcsine”
   (central‑binomial) sequence, optionally symmetrised, and renormalise to
   unit sum.

2. In each iteration:
   * Compute the self‑convolution a*a using a single FFT (real‑valued).
   * Approximate the non‑smooth “max” by a soft‑max with an increasing
     temperature α.  The soft‑max weights w_k form a probability distribution
     over the convolution entries.
   * Derive the gradient of the smooth approximation:
         ∂/∂a_i  (∑_k w_k·(a*a)_k) = 2·∑_k w_k·a_{k-i}
     which is efficiently obtained by another FFT (correlation of w and a).
   * Perform a momentum‑guided projected gradient step, clip negatives to
     zero and renormalise the vector back to unit sum.

3. Every few thousand iterations a random block‑scaling perturbation
   (the “block_perturbation” move from the reference implementation) is
   applied to help escape shallow local minima.

4. Every 100 iterations the exact objective is evaluated via the supplied
   ``evaluate_sequence`` function; the best sequence observed is remembered
   and printed.

5. The loop terminates when the allotted time budget expires (or the target
   bound ≈ 1.503 is already achieved).

The use of FFT gives O(n log n) per iteration, allowing tens of thousands of
iterations for n≈5 000 inside the 1000 s budget, which is far more thorough
than the pure hill‑climber in the baseline code.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: arcsine (inverse‑square‑root) sequence
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i) / 4**i (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence: a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Simple normalisation (project onto the simplex with a given ℓ₁‑norm)
# ----------------------------------------------------------------------
def renormalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to the allowed range, then scale so that the sum equals target_sum."""
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur == 0.0:
        return seq
    seq *= target_sum / cur
    return seq


# ----------------------------------------------------------------------
# Random block perturbation (same as in the baseline)
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ)) and re‑normalise."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                 # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return renormalise(cand, target_sum)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of non‑negative floats – the best step‑function heights
    found within ``budget_s`` seconds.

    The implementation uses FFT‑based smooth‑max gradient descent with
    occasional stochastic block perturbations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    # Use the length of the provided reference sequence if present,
    # otherwise fall back to a moderate default.
    ref = globals().get("height_sequence_1", [])
    n = int(len(ref)) if ref else 256

    # ------------------------------------------------------------------
    # 1.  Starting point – symmetrised arcsine shape.
    # ------------------------------------------------------------------
    a0 = build_arcsine_sequence(n)
    # Symmetrise to make the convolution more uniform
    a0 = 0.5 * (a0 + a0[::-1])
    target_sum = 1.0                 # the absolute scale is irrelevant
    cur_seq = renormalise(a0, target_sum)

    # ------------------------------------------------------------------
    # Helper: exact evaluation (the provided black‑box)
    # ------------------------------------------------------------------
    def exact_score(arr):
        # ``evaluate_sequence`` is assumed to be present in the global namespace.
        return evaluate_sequence(arr.tolist())

    # ------------------------------------------------------------------
    # Initial exact score
    # ------------------------------------------------------------------
    best_seq = cur_seq.copy()
    best_score = exact_score(best_seq)
    print(f"[init] n={n}, initial exact score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  FFT set‑up
    # ------------------------------------------------------------------
    # Length for convolutions: at least 3·n to cover conv(w, rev(a)).
    L = 1 << ((3 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 3.  Hyper‑parameters
    # ------------------------------------------------------------------
    alpha = 5.0          # start temperature for the soft‑max
    alpha_max = 200.0    # final temperature
    lr = 0.25            # base learning rate
    momentum = 0.9
    velocity = np.zeros_like(cur_seq)

    # ------------------------------------------------------------------
    # 4.  Main optimisation loop
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # keep a tiny slack
    it = 0
    last_log = start_time
    # Number of iterations after which we increase the temperature
    alpha_step = 3000

    while time.monotonic() < deadline:
        it += 1

        # ---- self‑convolution of the current sequence -----------------
        X = np.fft.rfft(cur_seq, n=L)
        conv_self = np.fft.irfft(X * X, n=L)[:2 * n - 1]   # length 2n‑1

        # ---- soft‑max weights over the convolution entries ------------
        max_conv = conv_self.max()
        # Stabilise exponentials by subtracting the maximum value
        shifted = conv_self - max_conv
        w_exp = np.exp(alpha * shifted)
        w = w_exp / w_exp.sum()          # probability vector, sum = 1

        # ---- gradient of the smooth approximation --------------------
        # grad_i = 2 * sum_k w_k * a_{k-i}
        # This is a correlation of w with the reversed a.
        rev_seq = cur_seq[::-1]
        W = np.fft.rfft(w, n=L)
        R = np.fft.rfft(rev_seq, n=L)
        corr = np.fft.irfft(W * R, n=L)  # full correlation, length L
        grad_smooth = 2.0 * corr[(n - 1):(n - 1 + n)]

        # Scale to the true objective (the constant 2·n / sum^2)
        const = 2.0 * n / (target_sum ** 2)
        grad = const * grad_smooth

        # ---- momentum step -------------------------------------------
        velocity = momentum * velocity - lr * grad
        cand_seq = cur_seq + velocity

        # ---- projection onto the feasible set (non‑negative, sum = 1) --
        cand_seq = np.maximum(cand_seq, 0.0)
        s = cand_seq.sum()
        if s > 0.0:
            cand_seq = cand_seq * (target_sum / s)
        else:
            cand_seq = cur_seq.copy()   # fallback (should rarely happen)

        # ---- occasional random large move to diversify ----------------
        if it % 2000 == 0:
            cand_seq = block_perturbation(cand_seq, rng, target_sum)

        # ---- decide whether to accept the candidate -------------------
        # Use the cheap smooth‑max score for the acceptance test.
        Xcand = np.fft.rfft(cand_seq, n=L)
        conv_self_cand = np.fft.irfft(Xcand * Xcand, n=L)[:2 * n - 1]
        cand_score_approx = 2.0 * n * conv_self_cand.max() / (target_sum ** 2)
        cur_score_approx = 2.0 * n * max_conv / (target_sum ** 2)

        if cand_score_approx < cur_score_approx - 1e-12:
            cur_seq = cand_seq
        else:
            # Very low probability uphill move to avoid complete stagnation
            if rng.random() < 0.001:
                cur_seq = cand_seq

        # ---- periodic exact evaluation --------------------------------
        if it % 100 == 0:
            exact = exact_score(cur_seq)
            if exact < best_score:
                best_score = exact
                best_seq = cur_seq.copy()
                print(f"[{it:>6}] new best exact = {best_score:.6f}")

                # Early exit if we have already beaten the target bound
                if best_score < 1.503:
                    print("[early stop] target bound reached")
                    break

        # ---- log progress every ~20 s ---------------------------------
        now = time.monotonic()
        if now - last_log > 20.0:
            elapsed = now - start_time
            print(f"[{it:>6}] time {elapsed:.1f}s, cur approx = {cur_score_approx:.6f}, "
                  f"best exact = {best_score:.6f}")
            last_log = now

        # ---- slowly increase the temperature -------------------------
        if it % alpha_step == 0 and alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.07)

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    final_score = exact_score(best_seq)
    total_time = time.monotonic() - start_time
    print(f"[final] best exact score {final_score:.6f} after {it} iterations, "
          f"elapsed {total_time:.1f}s")
    # Return a plain Python list as required by the harness.
    return [float(v) for v in best_seq]
