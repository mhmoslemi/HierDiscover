    # optionally scale up to use larger numbers (e.g., factor 1000)

"""
Gradient‑descent + simulated annealing optimizer for the convolution‑bound
problem.

* The sequence is kept on the (non‑negative) simplex (sum = 1).  Scaling does
  not affect the evaluation score, therefore we can work with tiny numbers
  safely.
* At each iteration the convolution a∗a is computed, the *top_k* largest
  entries are identified and an averaged gradient of those entries is taken.
  The gradient of (a∗a)[k] w.r.t. a[i] is 2·a[k‑i] (or a[i] when i=j), so the
  update pushes mass away from indices that contribute most to the current
  worst convolution value.
* A simple linear cooling schedule (temperature) is used to occasionally
  accept uphill moves (Simulated Annealing) and to escape local minima.
* Every few thousand iterations a global perturbation (mixing with the
  analytic arcsine start) is performed, and a restart is triggered if the
  algorithm stagnates for many steps.
* The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Analytic start – the arcsine (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the classical arcsine (inverse‑square‑root) sequence of
    given length: a[i] = binom(2i,i) / 4**i  (exact up to floating point)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# 2.  Helper utilities.
# ----------------------------------------------------------------------
def compute_score(seq: np.ndarray) -> float:
    """
    Direct implementation of the evaluator (without the security wrappers).
    The sequence must be non‑negative and already clipped to [0,1000].
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def gradient_of_top_convolutions(seq: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Compute the average gradient of the convolution entries whose indices are
    listed in *top_idxs*.  For a fixed k the gradient w.r.t. a[i] is
        2·a[k‑i]  (if i ≠ k‑i)  or  a[i]  (if i = k‑i).
    The function returns the sum of these gradients divided by len(top_idxs).
    """
    n = seq.size
    grad = np.zeros_like(seq)

    for k in top_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        # weight 1 for the diagonal term, 2 otherwise
        w = np.where(i == j, 1.0, 2.0)
        grad[i] += w * seq[j]

    if top_idxs.size > 0:
        grad /= float(top_idxs.size)
    return grad


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x_i ≥ 0, Σ x_i = 1 }.
    """
    # shift by the mean and sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return a uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 3.  Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 950,
                     **kwargs) -> list[float]:
    """
    Gradient‑descent / simulated‑annealing optimizer.
    Returns a list of non‑negative heights that (hopefully) improves on the
    arcsine baseline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and an initial sequence.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        n = init_seq.size
        # safety: clip to the evaluator's limits
        init_seq = np.clip(init_seq, 0.0, 1000.0)
        if n == 0:
            n = 256
            init_seq = build_arcsine_sequence(n)
    else:
        n = 256
        init_seq = build_arcsine_sequence(n)

    # Normalise to the probability simplex (sum = 1)
    if init_seq.sum() == 0.0:
        cur = np.ones(n, dtype=np.float64) / n
    else:
        cur = init_seq / init_seq.sum()

    best = cur.copy()
    best_score = compute_score(best)

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optimisation parameters.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # 1‑second safety margin
    it = 0

    # Learning‑rate schedule (linear decay)
    lr_start = 0.5
    lr_end = 0.001

    # Temperature schedule for the Metropolis test
    T_start = 0.01
    T_end = 0.001

    top_k = 5                 # number of convolution entries to target
    noise_prob = 0.02         # occasional Gaussian jitter
    noise_std = 0.01          # size of the jitter
    stagnation_counter = 0    # counts iterations without improvement
    stagnation_limit = 20000  # after this we re‑start from the current best

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1.  Compute convolution and locate the most offending entries.
        # --------------------------------------------------------------
        conv = np.convolve(cur, cur)
        if top_k == 1:
            top_idxs = np.array([int(np.argmax(conv))], dtype=int)
        else:
            # argpartition is O(n), we then sort the selected indices
            part = np.argpartition(conv, -top_k)[-top_k:]
            top_idxs = part[np.argsort(-conv[part])]   # descending order

        # --------------------------------------------------------------
        # 2.  Gradient step aimed at reducing those entries.
        # --------------------------------------------------------------
        grad = gradient_of_top_convolutions(cur, top_idxs)

        # Linear cooling of the learning rate
        elapsed = time.time() - start_time
        frac = elapsed / float(budget_s)
        lr = lr_start * (1.0 - frac) + lr_end * frac

        cand = cur - lr * grad

        # occasional isotropic Gaussian noise to escape plateaus
        if rng.random() < noise_prob:
            cand += rng.normal(0.0, noise_std, size=n)

        # enforce the feasible set (non‑negative, sum = 1)
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            # restart from the current best if we collapsed to the zero vector
            cand = best.copy()
        cand = project_onto_simplex(cand)

        # --------------------------------------------------------------
        # 3.  Evaluate candidate (exact objective, no extra clipping needed).
        # --------------------------------------------------------------
        cand_score = compute_score(cand)

        # --------------------------------------------------------------
        # 4.  Acceptance test (Metropolis with linear cooling temperature).
        # --------------------------------------------------------------
        cur_score = compute_score(cur)
        T = T_start * (1.0 - frac) + T_end * frac
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                stagnation_counter = 0
                print(f"[{it:>6}] new best = {best_score:.6f}")
                if best_score < 1.503:
                    print("[early] target reached")
                    break
            else:
                stagnation_counter += 1
        else:
            stagnation_counter += 1

        # --------------------------------------------------------------
        # 5.  Global perturbation every few thousand iterations.
        # --------------------------------------------------------------
        if it % 4000 == 0 and rng.random() < 0.4:
            # blend current state with a fresh arcsine start
            fresh = build_arcsine_sequence(n)
            fresh = fresh / fresh.sum()
            cur = 0.5 * cur + 0.5 * fresh
            cur = project_onto_simplex(cur)

        # --------------------------------------------------------------
        # 6.  Restart if we have been stagnant for too long.
        # --------------------------------------------------------------
        if stagnation_counter > stagnation_limit:
            cur = best.copy()
            stagnation_counter = 0

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # ------------------------------------------------------------------
    # 7.  Scale the result (scaling does not change the score) and return.
    # ------------------------------------------------------------------
    # We push the numbers into the allowed range [0,1000] for aesthetic
    # reasons; the evaluator will clip again anyway.
    scale = 1000.0
    result = [float(min(scale, max(0.0, x * scale))) for x in best.tolist()]
    return result
