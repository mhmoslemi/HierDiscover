def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

"""
Simple simulated‑annealing search for a low‑value step‑function.

The objective is 2·n·max(conv(a,a)) / (sum(a)²).  For a fixed length n we
renormalise every candidate to the standard sum √(2 n); after that the
objective equals max(conv(a,a)).  The algorithm works with several random
initialisations, performs cheap FFT‑based max‑convolution updates and
occasionally calls the exact ``evaluate_sequence`` (which also checks the
security constraints).  A temperature schedule and occasional random
re‑initialisations help to escape local minima.  The search respects the
budget supplied in ``budget_s`` (seconds).  The best sequence found (as a
list of Python floats) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using real FFTs.
    Returns the first (2*len(a)-1) elements.
    """
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)
    return conv_full[: 2 * a.size - 1]


def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Return the (approximate) maximum value of the convolution a*a.
    Uses the FFT routine above; the result is a float.
    """
    return float(_self_convolution_fft(a, fft_len).max())


def _normalise_to_target(a: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale the vector so that its L1‑norm becomes exactly ``target_sum``.
    Values are clipped to [0,1000] before scaling.
    """
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    # After scaling we might have crossed the 1000‑limit: clip again and
    # renormalise a tiny amount (the error after one extra clip is negligible).
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    return a


def _initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """
    Produce a list of diverse starting vectors (all already normalised).
    """
    starts = []

    # (a) uniform
    starts.append(np.full(n, target_sum / n))

    # (b) Dirichlet – random positive with sum target_sum
    gamma = rng.gamma(shape=1.0, scale=1.0, size=n)
    starts.append(_normalise_to_target(gamma, target_sum))

    # (c) exponential decay profiles
    for alpha in rng.uniform(0.0005, 0.02, size=3):
        idx = np.arange(n, dtype=float)
        a = np.exp(-alpha * idx)
        starts.append(_normalise_to_target(a, target_sum))

    # (d) a few “spikes’’ – one large entry plus a low uniform background
    for _ in range(3):
        a = np.full(n, 1e-3)
        spike_idx = rng.integers(0, n)
        a[spike_idx] = target_sum * 0.9  # allocate most mass to a spike
        starts.append(_normalise_to_target(a, target_sum))

    return starts


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    Search for a low‑score height sequence within the given time budget.
    Returns a list of floats (the best sequence found).
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a short safety margin for final printing / possible overhead
    deadline = start_time + max(0.0, float(budget_s) - 1.5)

    # --------------------------------------------------------------
    # 1) Determine a working length n
    # --------------------------------------------------------------
    try:

    except Exception:
        n = 6000  # default works well for the given budget
    # Clamp n to avoid absurd runtimes
    n = int(np.clip(n, 2000, 9000))

    target_sum = np.sqrt(2.0 * n)          # sum that makes the objective = max(conv)
    fft_len = _next_pow2(3 * n - 2)        # enough for all FFT convolutions

    # --------------------------------------------------------------
    # 2) Build a collection of diverse starting points
    # --------------------------------------------------------------
    starts = _initial_vectors(n, target_sum, rng)

    # expose any externally provided best sequence as an extra start
    try:

            starts.append(prev)

    # 3) Simulated annealing on each start
    # --------------------------------------------------------------
    best_seq = None

            accept = False

            else:
                # probabilistic uphill move

                if rng.random() < prob:
                    accept = True

            if accept:

                cur_max = cand_max
                no_improve = 0
                # occasional exact evaluation to keep best_seq up‑to‑date
                if total_evals % 150 == 0:

                    total_evals += 1
                    if exact < best_score:

                        best_seq = a.copy()

                    if cand_max < best_score:
                        best_score = cand_max
                        best_seq = a.copy()
                # end of acceptance block
            else:
                no_improve += 1

            # ------------------------------------------------------
            # (c) temperature update & occasional restart
            # ------------------------------------------------------
            T = max(1e-8, T * cooling)

            if no_improve > 800:

                a = _normalise_to_target(rng.exponential(scale=1.0, size=n), target_sum)

                no_improve = 0
                T = T0

            total_evals += 1

    # --------------------------------------------------------------
    # 4) Final exact evaluation of the best candidate
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform vector (guaranteed to give score 2)
        best_seq = np.full(n, target_sum / n)

    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] exact score = {final_score:.6f}")

    # ensure we return a plain Python list of floats
    return [float(v) for v in best_seq]
