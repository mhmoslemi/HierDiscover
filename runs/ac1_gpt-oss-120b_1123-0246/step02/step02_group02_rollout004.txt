"""
Propose a low‑score step‑function using an Adam‑style optimizer on a
soft‑max approximation of the max‑convolution.  The method differs from the
previous soft‑max gradient descent by:

* employing Adam moments for adaptive step‑sizes,
* using a larger final inverse‑temperature (β) and a linear warm‑up,
* trying a diverse set of initial shapes (uniform, random, U‑shaped,
  sinusoidal, triangular and the previously found best),
* optionally enlarging the sequence length after an initial optimisation
  phase (interpolation + short refinement).

All operations respect the required non‑negativity, clipping to the range
[0,1000] and exact normalisation  Σ a_i = √(2 n) (the “target_sum”).  The
objective that is minimised is max(a * a); because of the normalisation the
return value of ``evaluate_sequence`` equals this maximum, so we can use a
fast FFT‑based convolution for cheap progress checks and call the exact
``evaluate_sequence`` only periodically.

The routine respects a time budget (default 1000 s) and returns the best
sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using real FFTs.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)
    return conv_full[: 2 * a.size - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full correlation (w ⋆ a) = convolution(w[::-1], a) using FFTs.
    Returns an array of length (len(w) + len(a) - 1).
    """
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, fft_len)
    fa = np.fft.rfft(a, fft_len)
    corr_full = np.fft.irfft(fw * fa, fft_len)
    return corr_full[: w.size + a.size - 1]


def _normalize_to_target(a: np.ndarray, target: float) -> np.ndarray:
    """
    Scale a non‑negative vector so that its sum equals ``target``.
    If the sum is effectively zero, replace with a uniform distribution.
    """
    s = a.sum()
    if s > 1e-12:
        a = a * (target / s)
    else:
        a = np.full_like(a, target / a.size)
    return a


def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the soft‑max of ``beta * b`` in a numerically stable way.
    Returns a probability vector that sums to 1.
    """
    max_b = b.max()
    shifted = b - max_b
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        # fallback to a hard max
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
        return w
    return w / w_sum


def _create_start_shapes(n: int, target_sum: float, rng: np.random.Generator):
    """
    Generate a list of diverse starting vectors of length ``n``.
    All vectors are non‑negative and have sum ``target_sum``.
    """
    starts = []

    # (a) uniform
    starts.append(np.full(n, target_sum / n))

    # (b) random positive
    rand = rng.random(n) * target_sum
    starts.append(_normalize_to_target(rand, target_sum))

    # (c) U‑shaped (negative power law)
    for p in rng.uniform(-4.0, -0.2, size=4):
        idx = np.arange(n, dtype=float)
        a = (idx + 1) ** p + (n - idx) ** p
        a = a - a.min() + 1e-8
        starts.append(_normalize_to_target(a, target_sum))

    # (d) sinusoidal (peaked in the middle)
    sin = np.sin(np.pi * (np.arange(n) + 0.5) / n) + 1e-8
    starts.append(_normalize_to_target(sin, target_sum))

    # (e) triangle (linear up‑down)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2, endpoint=True)])
    starts.append(_normalize_to_target(tri, target_sum))

    # (f) reuse the best known sequence if it exists (interpolate if needed)
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            if prev.size != n:
                old_x = np.linspace(0.0, 1.0, prev.size)
                new_x = np.linspace(0.0, 1.0, n)
                prev = np.interp(new_x, old_x, prev)
            prev = np.clip(prev, 0.0, 1000.0)
            starts.append(_normalize_to_target(prev, target_sum))
    except Exception:
        pass

    return starts


def _adam_step(a: np.ndarray,
               grad: np.ndarray,
               m: np.ndarray,
               v: np.ndarray,
               t: int,
               lr: float,
               beta1: float = 0.9,
               beta2: float = 0.999,
               eps: float = 1e-8) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform one Adam update (returns new a, new m, new v).
    The update is *additive* (a ← a − step), but the function returns the
    modified a for convenience.
    """
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)

    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    return a_new, m, v

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    ``evaluate_sequence`` score.  The function respects the supplied time
    budget and returns the best sequence discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # Keep a safety margin for final reporting / printing
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # 1. Choose a working length (n) and normalisation constant
    # ------------------------------------------------------------------
    # Use the length of the previously known sequence if possible,
    # otherwise fall back to a moderate default.
    try:
        n = int(np.asarray(height_sequence_1).size)
    except Exception:
        n = 5000
    # Clamp to a range that keeps the FFT cheap while still giving us a
    # sufficiently expressive search space.
    n = int(np.clip(n, 2000, 9000))

    target_sum = math.sqrt(2.0 * n)          # makes the final score = max conv
    fft_len = _next_pow2(3 * n - 2)           # ≥ len(w)+len(a)‑1 for correlations

    # ------------------------------------------------------------------
    # 2. Build a collection of starting points
    # ------------------------------------------------------------------
    starts = _create_start_shapes(n, target_sum, rng)

    # ------------------------------------------------------------------
    # 3. Core optimisation loop
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    total_iters = 0
    max_total_iters = 200_000   # absolute safety guard

    # Hyper‑parameters for the Adam optimiser
    lr0 = 0.3 * target_sum      # base learning rate (will be decayed)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Soft‑max inverse‑temperature schedule
    beta_start = 0.5
    beta_end = 30.0

    # Number of iterations per start (adapts to remaining time)
    max_iters_per_start = 8000

    for start_id, a0 in enumerate(starts):
        if time.monotonic() >= deadline:
            break

        a = np.copy(a0)
        a = np.clip(a, 0.0, 1000.0)
        a = _normalize_to_target(a, target_sum)

        # Evaluate the initial point exactly (cheap if we already know it)
        init_exact = float(evaluate_sequence(a.tolist()))
        if init_exact < best_score:
            best_score = init_exact
            best_seq = a.copy()
            print(f"[start {start_id}] exact initial score = {best_score:.6f}")

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # Current max‑convolution for monotone acceptance
        cur_max = _self_convolution_fft(a, fft_len).max()

        # Iteration loop for this start
        it = 0
        while it < max_iters_per_start and total_iters < max_total_iters:
            now = time.monotonic()
            if now >= deadline:
                break

            # --------------------------------------------------------------
            # a) compute soft‑max weights from the current self‑convolution
            # --------------------------------------------------------------
            b = _self_convolution_fft(a, fft_len)          # length 2n‑1
            cur_max = b.max()
            beta = beta_start + (beta_end - beta_start) * (it / max_iters_per_start)

            w = _softmax_weights(b, beta)                  # sum(w) == 1

            # --------------------------------------------------------------
            # b) gradient of the smooth objective
            # --------------------------------------------------------------
            # correlation of w with a gives the gradient before scaling
            corr = _correlation_fft(w, a, fft_len)          # length 2n‑1
            L = 2 * n - 1
            grad = 2.0 * beta * corr[L - 1 - np.arange(n)]

            # --------------------------------------------------------------
            # c) Adam update (with a linearly decaying learning rate)
            # --------------------------------------------------------------
            lr = lr0 * max(0.0, 1.0 - it / max_iters_per_start)
            a_cand, m, v = _adam_step(a, grad, m, v, it + 1, lr,
                                      beta1=beta1, beta2=beta2, eps=eps)

            # enforce the admissible range and re‑normalise
            a_cand = np.clip(a_cand, 0.0, 1000.0)
            a_cand = _normalize_to_target(a_cand, target_sum)

            # --------------------------------------------------------------
            # d) cheap acceptance test (max of the self‑convolution)
            # --------------------------------------------------------------
            cand_max = _self_convolution_fft(a_cand, fft_len).max()
            if cand_max < cur_max:
                a = a_cand
                cur_max = cand_max
                # occasional exact evaluation to keep the best sequence up‑to‑date
                if total_iters % 150 == 0:
                    exact = float(evaluate_sequence(a.tolist()))
                    if exact < best_score:
                        best_score = exact
                        best_seq = a.copy()
                        print(f"[{total_iters:6d}] new best exact = {best_score:.6f}")
                else:
                    # use the cheap max for bookkeeping
                    if cand_max < best_score:
                        best_score = cand_max
                        best_seq = a.copy()
                # small stochastic kick every few hundred successful steps
                if it % 1000 == 0 and it > 0:
                    mult = np.exp(rng.normal(0.0, 0.025, size=n))
                    a = np.clip(a * mult, 0.0, 1000.0)
                    a = _normalize_to_target(a, target_sum)
                    cur_max = _self_convolution_fft(a, fft_len).max()
            else:
                # accept with a tiny probability to occasionally escape plateaus
                if rng.random() < 0.0005:
                    a = a_cand
                    cur_max = cand_max

            total_iters += 1
            it += 1

            # --------------------------------------------------------------
            # e) random diversification every few hundred iterations
            # --------------------------------------------------------------
            if it % 500 == 0 and it > 0:
                pert = np.exp(rng.normal(0.0, 0.04, size=n))
                a_tmp = np.clip(a * pert, 0.0, 1000.0)
                a_tmp = _normalize_to_target(a_tmp, target_sum)
                max_tmp = _self_convolution_fft(a_tmp, fft_len).max()
                if max_tmp < cur_max:
                    a = a_tmp
                    cur_max = max_tmp
                    if max_tmp < best_score:
                        best_score = max_tmp
                        best_seq = a.copy()
                        print(f"[{total_iters:6d}] diversification improved = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 4. Optional length‑expansion phase (if time permits)
    # ----------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline - 5.0:
        # try to enlarge the sequence (more degrees of freedom) and refine
        n_big = min(int(n * 1.5), 9000)
        if n_big > n:
            # linear interpolation to the larger size
            old_x = np.linspace(0.0, 1.0, best_seq.size)
            new_x = np.linspace(0.0, 1.0, n_big)
            a_big = np.interp(new_x, old_x, best_seq)
            target_big = math.sqrt(2.0 * n_big)
            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = _normalize_to_target(a_big, target_big)

            # run a short Adam optimisation on the enlarged vector
            fft_len_big = _next_pow2(3 * n_big - 2)
            m_big = np.zeros_like(a_big)
            v_big = np.zeros_like(a_big)
            cur_max_big = _self_convolution_fft(a_big, fft_len_big).max()

            max_iters_big = 3000
            for it_big in range(max_iters_big):
                if time.monotonic() >= deadline:
                    break

                b_big = _self_convolution_fft(a_big, fft_len_big)
                cur_max_big = b_big.max()
                beta_big = beta_start + (beta_end - beta_start) * (it_big / max_iters_big)

                w_big = _softmax_weights(b_big, beta_big)
                corr_big = _correlation_fft(w_big, a_big, fft_len_big)
                L_big = 2 * n_big - 1
                grad_big = 2.0 * beta_big * corr_big[L_big - 1 - np.arange(n_big)]

                lr_big = lr0 * max(0.0, 1.0 - it_big / max_iters_big)
                a_big, m_big, v_big = _adam_step(a_big, grad_big,
                                                m_big, v_big,
                                                it_big + 1, lr_big,
                                                beta1=beta1, beta2=beta2, eps=eps)

                a_big = np.clip(a_big, 0.0, 1000.0)
                a_big = _normalize_to_target(a_big, target_big)

                cand_max_big = _self_convolution_fft(a_big, fft_len_big).max()
                if cand_max_big < cur_max_big:
                    cur_max_big = cand_max_big

                # occasional exact evaluation
                if it_big % 200 == 0:
                    exact_big = float(evaluate_sequence(a_big.tolist()))
                    if exact_big < best_score:
                        best_score = exact_big
                        best_seq = a_big.copy()
                        print(f"[expand] new best exact = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 5. Final exact evaluation and return
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector of the selected length
        best_seq = np.full(n, target_sum / n)

    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] best exact score = {final_score:.6f}")

    # Convert to plain Python floats for the caller
    return [float(v) for v in best_seq]
