def simplex_projection(v):
    """Project onto simplex {x >=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    # ensures nonnegative and sum(v) = 1
    if v.sum() == 0:
        # If all zero, return uniform distribution
        return np.full_like(v, 1.0/v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, v.size+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
```

We'll incorporate small epsilon to avoid division zero.

Now convolve using FFT:

```
def convolve_fft(a):
    n = a.size
    m = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, m)
    B = A * A
    conv = np.fft.irfft(B, m)[:2*n-1]
    return conv
```

Works for real valued.

Now compute softmax weights:

```
def softmax_weights(b, beta):
    # b is 1D numpy array length L = 2n-1
    # scale by subtracting max to avoid overflow
    b_max = b.max()
    exp_vals = np.exp(beta * (b - b_max))
    w = exp_vals / exp_vals.sum()
    return w
```

"""
Hybrid simulated‑annealing / gradient‑descent optimizer.

The algorithm starts from the optimal analytic “arcsine’’ vector (the
coefficients a_i = binom(2i,i)/4**i, normalised to unit ℓ₁‑norm).
From there a stochastic search explores the space of non‑negative
vectors with fixed sum 1.  Moves keep the ℓ₁‑norm invariant:
    – multiplicative scaling of a random block,
    – transfer of a random fraction of mass between two points,
    – transfer from the early part of the vector to the late part,
    – a gradient‑inspired transfer that reduces the current maximal
      convolution entry.
A simple simulated‑annealing acceptance rule (Metropolis) allows
temporary uphill moves, which lets the search escape the large flat
plateau where many convolution entries equal the maximum.  The score
is the exact objective 2·n·max(conv)/(∑a)² – with ∑a = 1 this reduces
to 2·n·max(conv).  The best vector seen is returned.

The implementation uses FFT‑based convolution (O(n log n)) so that
hundreds of thousands of iterations comfortably fit into the 1000 s
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Basic utilities
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the discrete arcsine (a_i = binom(2i,i)/4**i) sequence."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project the vector v onto the probability simplex {x ≥ 0, Σx = 1}.
    Uses the O(n log n) algorithm of
    Duchi et al., “Efficient Projections onto the L1‑Ball”.
    """
    if v.sum() == 0.0:
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety, fall back to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a power‑of‑two FFT.
    Returned length is 2·len(a)‑1.
    """
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    B = A * A
    conv = np.fft.irfft(B, m)[: 2 * n - 1]
    return conv


def softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """
    Smooth approximation of the arg‑max selector.
    Returns a probability vector of the same size as b.
    """
    b_max = b.max()
    # subtract max for numerical stability
    exp_vals = np.exp(beta * (b - b_max))
    w = exp_vals / exp_vals.sum()
    return w


def gradient_of_weighted_conv(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum Σ_k w_k (a*a)_k.
    Implements the formula ∂/∂a_i Σ_k w_k b_k = 2·(w ⋆ a_rev)_i,
    where a_rev is a reversed copy of a and ⋆ is linear convolution.
    """
    n = a.size
    a_rev = a[::-1]
    m = 1 << ((2 * n - 1).bit_length())
    W = np.fft.rfft(w, m)
    A_rev = np.fft.rfft(a_rev, m)
    conv = np.fft.irfft(W * A_rev, m)[: 2 * n - 1]
    grad = 2.0 * conv[n - 1 : n - 1 + n]          # slice of length n
    return grad


# ----------------------------------------------------------------------
# 2.  Move operators (all keep ℓ₁‑norm = 1)
# ----------------------------------------------------------------------
def renorm_to_one(seq: np.ndarray) -> np.ndarray:
    """Clamp to the allowed range and re‑normalise to sum = 1."""
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total == 0.0:
        return np.full_like(seq, 1.0 / seq.size)
    return seq / total


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.15)))   # σ ≈ 0.15
    cand[start:stop] *= factor
    if rng.random() < 0.2:                         # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return renorm_to_one(cand)


def transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random fraction (≤30 %) of mass from i to j."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renorm_to_one(cand)


def early_to_late_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Shift a fraction of mass from the first 10 % to the last 10 %."""
    n = base.size
    early_cut = max(1, n // 10)
    late_start = max(early_cut, n - early_cut)
    i = int(rng.integers(0, early_cut))
    j = int(rng.integers(late_start, n))
    if i == j:
        return base.copy()
    frac = rng.random() * 0.40
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renorm_to_one(cand)


def gradient_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Move mass from the index that most contributes to the current maximum
    convolution entries towards an index with low contribution.
    """
    n = base.size
    b = convolve_fft(base)
    max_b = b.max()
    # indices where the convolution is (almost) maximal
    tol = max_b * (1.0 - 1e-12)
    mask = (b >= tol).astype(np.float64)          # length = 2n‑1
    # contribution of each coefficient to those maximal entries
    contrib_conv = np.convolve(base, mask)        # length = 2n‑1
    contrib = base * contrib_conv[:n]             # one contribution per index
    heavy_idx = int(np.argmax(contrib))

    # pick a light index – preferably a low‑contribution one
    mean_contrib = contrib.mean()
    light_candidates = np.where(contrib <= mean_contrib)[0]
    light_candidates = light_candidates[light_candidates != heavy_idx]
    if light_candidates.size == 0:
        # fall back to a random different index
        light_idx = int(rng.integers(0, n - 1))
        if light_idx >= heavy_idx:
            light_idx += 1
    else:
        light_idx = int(rng.choice(light_candidates))

    # transfer a random fraction (≤30 %) from heavy to light
    frac = rng.random() * 0.30
    delta = frac * base[heavy_idx]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[heavy_idx] -= delta
    cand[light_idx] += delta
    return renorm_to_one(cand)


# ----------------------------------------------------------------------
# 3.  Scoring routine (exact objective, fast because sum = 1)
# ----------------------------------------------------------------------
def exact_score(seq: np.ndarray) -> float:
    """Exact objective value 2·n·max(conv) for a unit‑sum sequence."""
    n = seq.size
    b = convolve_fft(seq)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# 4.  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 900, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    target bound 1.503.  The search runs for ``budget_s`` seconds,
    using simulated annealing with several stochastic move types.
    The best sequence seen (according to the exact objective) is
    returned as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the problem size.
    # If a reference sequence is present we honour its length,
    # otherwise fall back to a modest size (256).
    # --------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256

    # --------------------------------------------------------------
    # 1) Initial vector – the analytic arcsine shape, normalised.
    # --------------------------------------------------------------
    a0 = build_arcsine_sequence(n)
    cur_seq = renorm_to_one(a0)
    cur_score = exact_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # --------------------------------------------------------------
    # 2) Simulated‑annealing loop.
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)   # small safety margin
    it = 0

    # temperature schedule
    T = 0.02 * cur_score                 # start with ~2 % of the score
    T_min = 1e-7
    decay = 0.9998                       # slow exponential cooling

    while time.monotonic() < deadline:
        it += 1

        # ----- choose a move -------------------------------------------------
        r = rng.random()
        if r < 0.30:
            cand = block_perturbation(cur_seq, rng)
        elif r < 0.55:
            cand = transfer_move(cur_seq, rng)
        elif r < 0.80:
            cand = early_to_late_move(cur_seq, rng)
        else:
            cand = gradient_move(cur_seq, rng)

        cand_score = exact_score(cand)

        # ----- Metropolis acceptance -----------------------------------------
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                print(f"[{it:>7}] new best = {best_score:.6f}")

                # early exit if we already beat the target comfortably
                if best_score < 1.503:
                    break

        # ----- temperature decay ---------------------------------------------
        T = max(T_min, T * decay)

        # ----- occasional global perturbation (helps escape very flat plateaus)
        if it % 25000 == 0:
            noise = rng.normal(0.0, 0.04, size=n)
            noisy = cur_seq + noise
            noisy = np.clip(noisy, 0.0, None)
            noisy = renorm_to_one(noisy)
            noisy_score = exact_score(noisy)
            if noisy_score < cur_score or rng.random() < np.exp(-(noisy_score - cur_score) / max(T, 1e-12)):
                cur_seq = noisy
                cur_score = noisy_score
                if noisy_score < best_score:
                    best_seq = noisy
                    best_score = noisy_score
                    print(f"[{it:>7}] new best after noise = {best_score:.6f}")

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return a plain Python list (the harness expects list[float]).
    return [float(x) for x in best_seq]
