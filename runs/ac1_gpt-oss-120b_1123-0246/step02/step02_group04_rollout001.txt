def project_to_simplex(v):
    # from https://doi.org/10.1007/s10107-009-0296-1
    # Input v is 1D array.
    # Returns w s.t sum w=1, w>=0, and it is the Euclidean projection.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    # find rho
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‐function sequence.

The algorithm works directly with the *shape* of the step function,
i.e. with a probability vector p (p_i ≥ 0, Σ p_i = 1).  For such a vector

    evaluate_sequence(p) = 2·n·max_k (p * p)_k

so the scaling of p does not matter.  We therefore minimise the
∞‑norm of the self‑convolution of p.

We use a fast projected‑gradient descent scheme:

* compute the linear convolution by FFT (O(n log n));
* locate the index k_max where the convolution attains its maximum;
* the gradient of that maximum entry is
        g_j = 2·p_{k_max‑j}   (if the index is inside the array,
                               with a factor 1 on the centre if k_max is even);
* take a small step p←p−η·g and project back onto the probability simplex;
* the step size η is linearly cooled; occasional random perturbations
  and a cheap “global‑swap” are inserted to avoid stagnation.

The routine runs for the allotted time budget (default 900 s) and
returns the best sequence found.  All values are non‑negative and
respect the evaluator’s clipping limits.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of a vector onto the probability simplex.
        Returns w with w_i ≥ 0 and Σ w_i = 1.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest i satisfying u_i > cssv_i / i
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # all entries are non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[mask][-1]
    theta = cssv[mask][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny floating‑point drift
    w_sum = w.sum()
    if w_sum <= 0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


def linear_convolution(p: np.ndarray, N_fft: int) -> np.ndarray:
    """ Return the linear (aperiodic) convolution p * p using FFT.
        The result has length 2·len(p) – 1.
    """
    # real‑to‑real FFT is sufficient (inputs are real, output is real)
    fa = np.fft.rfft(p, n=N_fft)
    conv = np.fft.irfft(fa * fa, n=N_fft)[: 2 * p.size - 1]
    return conv


def build_arcsine_sequence(length: int) -> np.ndarray:
    """ Analytic arcsine (inverse‑square‑root) start sequence. """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 900,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step function to minimise the evaluator.
    A simple projected‑gradient descent (with occasional random moves)
    is performed for up to *budget_s* seconds.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initial shape.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = int(len(globals()["height_sequence_1"]))
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        # safety: fall back to arcsine if the provided sequence is malformed
        if init_seq.shape[0] != n or np.any(np.isnan(init_seq)):
            init_seq = build_arcsine_sequence(n)
    else:
        n = 256
        init_seq = build_arcsine_sequence(n)

    # Normalise to a probability vector (shape only matters)
    total = init_seq.sum()
    if total <= 0.0:
        p = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        p = init_seq.astype(np.float64) / total

    # ------------------------------------------------------------
    # Pre‑compute FFT length (next power of two) for speed.
    # ------------------------------------------------------------
    conv_len = 2 * n - 1
    N_fft = 1 << (conv_len - 1).bit_length()   # next power of two

    # ------------------------------------------------------------
    # Evaluation helpers (fast version, exact up to floating point).
    # ------------------------------------------------------------
    def evaluate_fast(vec: np.ndarray) -> float:
        conv = linear_convolution(vec, N_fft)
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # Keep track of the best solution found.
    # ------------------------------------------------------------------
    best_p = p.copy()
    best_score = evaluate_fast(best_p)
    cur_p = p.copy()
    cur_score = best_score

    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # small safety margin
    it = 0

    # ------------------------------------------------------------
    # Annealing / step‑size schedule
    # ------------------------------------------------------------
    eta_start = 0.12          # initial gradient step size
    eta_end = 0.001           # final step size
    temp_start = 0.015        # for occasional Metropolis acceptance
    temp_end = 0.001

    # ------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- 1. compute convolution, locate worst index -----
        conv = linear_convolution(cur_p, N_fft)
        max_c = conv.max()
        k_max = int(conv.argmax())

        # ----- 2. gradient of the maximal convolution entry -----
        d = k_max - np.arange(n)
        mask = (d >= 0) & (d < n)
        grad = np.zeros_like(cur_p)
        grad[mask] = 2.0 * cur_p[d[mask]]
        # central term (i=j) should have factor 1 instead of 2
        if k_max % 2 == 0:
            mid = k_max // 2
            if 0 <= mid < n:
                grad[mid] = cur_p[mid]

        # ----- 3. step‑size according to elapsed time -----
        elapsed = time.time() - start_time
        frac = elapsed / float(budget_s)    # 0 → 1
        eta = eta_start * (1.0 - frac) + eta_end * frac
        temperature = temp_start * (1.0 - frac) + temp_end * frac

        # ----- 4. generate a candidate by gradient step -----
        cand = cur_p - eta * grad

        # Occasionally add a tiny random jitter to avoid exact plateaux
        if rng.random() < 0.02:
            cand += rng.normal(0.0, 1e-4, size=n)

        # ----- 5. project back onto the simplex (preserves sum=1) -----
        cand = project_to_simplex(cand)

        # ----- 6. fast evaluation of the candidate -----
        cand_score = evaluate_fast(cand)

        # ----- 7. Acceptance test (Metropolis for occasional uphill moves) -----
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_p = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_p = cand.copy()
                best_score = cand_score
                print(f"[{it:>6}] new best score = {best_score:.6f}")

                # early exit if we are already below the target bound
                if best_score < 1.503:
                    print("[early] target reached")
                    break

        # ----- 8. Global random perturbation every few thousand iterations -----
        if it % 4000 == 0:
            # re‑seed from the current best with a modest random reshuffle
            pert = best_p * (1.0 + rng.normal(0.0, 0.04, size=n))
            pert = np.clip(pert, 0.0, None)
            pert = project_to_simplex(pert)
            cur_p = pert
            cur_score = evaluate_fast(cur_p)

        # ----- 9. Very occasional "swap‑move" (exchange two entries) -----
        if it % 2500 == 0 and rng.random() < 0.3:
            i, j = rng.choice(n, size=2, replace=False)
            swapped = cur_p.copy()
            swapped[i], swapped[j] = swapped[j], swapped[i]
            # No projection needed – swapping preserves the simplex
            swapped_score = evaluate_fast(swapped)
            if swapped_score < cur_score or rng.random() < 0.05:
                cur_p = swapped
                cur_score = swapped_score
                if swapped_score < best_score:
                    best_p = swapped.copy()
                    best_score = swapped_score
                    print(f"[{it:>6}] swap gave new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final conversion back to a plain list (the evaluator will clip)
    # ------------------------------------------------------------------
    final_seq = best_p.tolist()
    # Verify with the official evaluator (cheap; does not affect timing much)
    try:
        official_score = float(evaluate_sequence(final_seq))
        print(f"[final] best official score = {official_score:.6f} after {it} iterations")
    except Exception as e:
        # fallback – keep the fast‑computed score
        official_score = best_score
        print(f"[final] (fallback) best score = {official_score:.6f} after {it} iterations")

    return [float(x) for x in final_seq]
