def simplex_projection(v):
    # based on sorting.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

This ensure sum exactly 1.

"""
Propose a step‑function sequence that minimises
    2 * n * max(a * a) / (sum a)^2

The algorithm is a fast stochastic projected gradient descent
with a soft‑max approximation of the L∞‑norm of the autocorrelation.
Key ideas
* Use the FFT for O(n log n) convolution and for the gradient
  (the gradient of max_k (a*a)[k] is a weighted sum of the
  “shift” vectors a_{k‑i}).
* A temperature τ controls a soft‑max weight w_k ∝ exp((c_k−c_max)/τ);
  τ is annealed so that early iterations see a smooth landscape and
  later iterations focus on the true maximum.
* The update direction is projected onto the simplex (non‑negative,
  sum‑to‑one) by subtracting its mean and clipping; the sequence is
  renormalised after each step.
* Occasional random block perturbations and symmetrisation give the
  optimiser the ability to jump out of local minima.
* Several power‑law/exponential seeds are tried and the best ever
  sequence is kept.  The whole search runs for the allotted time
  budget (≈1000 s) and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _evaluate_fast(seq: np.ndarray) -> float:
    """Fast evaluation of the objective for a normalised (sum=1) sequence."""
    # sum is assumed ≈1, otherwise the scaling cancels out.
    n = seq.size
    conv = np.convolve(seq, seq)      # full linear convolution
    max_c = conv.max()
    # The function evaluate_sequence returns 2 * n * max_c / (sum^2).
    # With sum≈1 this is simply 2 * n * max_c.
    return float(2.0 * n * max_c)


def _fft_convolution(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution of a and b using real FFT.
    `size` must be a power of two ≥ len(a)+len(b)-1.
    Returns a real array of length `size`; caller should slice the
    relevant part.
    """
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    return np.fft.irfft(fa * fb, n=size)


def _softmax_gradient_step(seq: np.ndarray,
                           eta: float,
                           tau_factor: float,
                           size: int) -> np.ndarray:
    """
    One projected gradient step where the gradient of
    max_k (seq * seq)[k] is approximated by a soft‑max over all
    convolution entries.
    """
    n = seq.size

    # ---- convolution of seq with itself (needed for both score and τ) ----
    conv_full = _fft_convolution(seq, seq, size)
    conv = conv_full[: 2 * n - 1]                     # linear convolution
    max_c = conv.max()

    # ---- soft‑max weights -------------------------------------------------
    # τ is chosen as a fraction of the current maximum value.
    # The fraction `tau_factor` is annealed by the outer loop.
    tau = max_c * max(tau_factor, 1e-12)              # avoid division by zero
    # w_k ∝ exp((c_k - max_c) / τ); the max entry gets weight 1.
    w_raw = np.exp((conv - max_c) / tau)
    w = w_raw / w_raw.sum()                           # normalised, sum=1

    # ---- gradient of the soft‑max (via FFT) ------------------------------
    # grad_i = 2 * Σ_k w_k * seq_{k-i}  = 2 * (w ⋆ seq_rev)[i]
    rev_seq = seq[::-1]
    grad_full = _fft_convolution(w, rev_seq, size)
    # The relevant slice is indices n‑1 … n‑1+n‑1.
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]

    # ---- projected update -------------------------------------------------
    direction = -grad
    direction -= direction.mean()                     # zero‑sum direction
    new_seq = seq + eta * direction
    # clip to the allowed interval (the clip is cheap; values stay << 1000)
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    s = new_seq.sum()
    if s > 0:
        new_seq /= s
    else:                     # numerical safety – should never happen
        new_seq = seq.copy()
    return new_seq


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12 (as in the reference code)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation (helps flatten the autocorrelation)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs):
    """
    Return a list of non‑negative step heights that (hopefully) yields a low
    value of `evaluate_sequence`.  The routine runs for `budget_s` seconds
    (leaving a small safety margin) and uses a stochastic projected gradient
    descent with a soft‑max approximation of the L∞‑norm.
    """
    rng = np.random.default_rng(seed)

    # ----- time management -------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----- choose length ----------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        n = 2500                     # fallback size used in the reference code

    # ----- preprocessing: FFT size -----------------------------------------
    # For the gradient we need a size ≥ len(w)+len(seq_rev)-1 = (2n‑1)+n‑1 = 3n‑2
    fft_size = 1 << ((3 * n - 2).bit_length())   # next power of two

    # ----- generate a portfolio of seeds (power‑law / exponential / random)-
    best_seq = None
    best_score = float("inf")
    idx = np.arange(1, n + 1, dtype=np.float64)   # 1‑based indices

    # systematic grid of power‑law exponents & offsets
    alphas = np.linspace(0.45, 0.55, 21)
    betas = np.linspace(0.0, 5.0, 11)
    for alpha in alphas:
        for beta in betas:
            seq = 1.0 / ((idx + beta) ** alpha)
            seq /= seq.sum()
            sc = _evaluate_fast(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()

    # random power‑law + mild exponential cut‑off
    for _ in range(20):
        alpha = rng.uniform(0.35, 0.65)
        beta = rng.uniform(0.0, 5.0)
        gamma = rng.uniform(0.0, 1e-3)               # exponential tail strength
        seq = np.exp(-gamma * idx) / ((idx + beta) ** alpha)
        seq /= seq.sum()
        sc = _evaluate_fast(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # pure Dirichlet samples (uniformly random simplex points)
    for _ in range(10):
        seq = rng.dirichlet(np.ones(n))
        sc = _evaluate_fast(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # fallback – should never happen
    if best_seq is None:
        seq = 1.0 / np.sqrt(idx)
        seq /= seq.sum()
        best_seq = seq
        best_score = _evaluate_fast(seq)

    # ----- main optimisation loop -----------------------------------------
    current_seq = best_seq.copy()
    current_score = best_score
    it = 0

    # annealing schedules (exponential)
    eta_start = 0.20
    eta_end   = 4e-5
    tau_factor_start = 0.5          # τ = τ_factor * max_c  (start relatively large)
    tau_factor_end   = 0.001        # very sharp at the end
    temp_start = 5e-2
    temp_end   = 1e-6
    block_prob = 0.20               # % of iterations that use a random block move
    sym_prob = 0.07                 # occasional symmetrisation

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - (deadline - float(budget_s) + 2.0)
        total   = deadline - (deadline - float(budget_s) + 2.0)
        prog = min(1.0, elapsed / total)

        # schedules
        eta = eta_start * (eta_end / eta_start) ** prog
        tau_factor = tau_factor_start * (tau_factor_end / tau_factor_start) ** prog
        temp = temp_start * (temp_end / temp_start) ** prog

        # ------------------------------------------------------------------
        # propose a candidate
        # ------------------------------------------------------------------
        if rng.random() < block_prob:
            cand = _make_block_perturbation(current_seq, rng)
        else:
            cand = _softmax_gradient_step(current_seq, eta, tau_factor, fft_size)

        # occasional extra symmetrisation (helps flatten the autocorrelation)
        if rng.random() < sym_prob:
            cand = 0.5 * (cand + cand[::-1])

        # renorm – after block moves the sum may deviate slightly
        s = cand.sum()
        if s > 0:
            cand /= s
        else:
            cand = current_seq.copy()

        cand_score = _evaluate_fast(cand)

        # ------------------------------------------------------------------
        # Acceptance (Metropolis‑like) – allows occasional uphill moves
        # ------------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - current_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            current_seq = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"iter {it:06d} – new best: {best_score:.6f}")

        it += 1

    # ----------------------------------------------------------------------
    # Final sanity checks – the external evaluator will clip again
    # ----------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s

    print(f"finished after {it} iterations, best score = {best_score:.6f}")
    return best_seq.tolist()
