def project_onto_simplex(v, s):
    """Project vector v onto simplex {x>=0, sum x = s}."""
    # We use method from Duchi.
    n = len(v)
    u = np.sort(v)[::-1]  # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    # find rho = max i where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=0, then solution is uniform?
        # But for positive s, there must be some positive.
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent optimiser for the convolution‑bound problem.

The score is 2 n·max(conv(a,a)) / (sum(a)²).  For a fixed total sum the
objective reduces to minimising the maximum entry of the self‑convolution.
Starting from the analytic arcsine (inverse‑square‑root) sequence we repeatedly
compute the index k* where the convolution attains its maximum and move mass
away from the indices that contribute to that peak.  The move is a projected
gradient step:   a ← Π_S(a – η·∇max conv)  where Π_S projects onto the simplex
{a≥0, sum(a)=S}.  An adaptive step‑size and occasional random block perturbations
help escape plateaus.  The routine runs for at most *budget_s* seconds and
returns the best sequence found.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Monotone decreasing arcsine coefficients a[i]=C(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
    {x | x_i >= 0, sum x_i = s}.
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find rho = max { i | u_i > cssv_i / i }
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def convolution_fft(a: np.ndarray, n_fft: int) -> np.ndarray:
    """Self‑convolution using a real FFT (fast for large n)."""
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)
    return conv[: 2 * a.size - 1]


def gradient_of_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution entry with respect to a.
    For the current maximising index k_max, ∂b_k/∂a_i = a_i + a_{k-i}
    (with the factor 2 when i == k-i).  Entries not participating have
    zero gradient.
    """
    n = a.size
    i = np.arange(n)
    j = k_max - i
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)

    if not np.any(mask):
        return g

    i_sel = i[mask]
    j_sel = j[mask]

    eq = i_sel == j_sel
    # diagonal terms (i == j)
    g[i_sel[eq]] = 2.0 * a[i_sel[eq]]
    # off‑diagonal terms
    neq = ~eq
    i_ne = i_sel[neq]
    j_ne = j_sel[neq]
    g[i_ne] = a[i_ne] + a[j_ne]

    return g


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Random multiplicative perturbation of a contiguous block (used for
    occasional diversification)."""
    n = base.size
    # block width between 1 % and 10 % of n
    w_min = max(1, n // 100)
    w_max = max(2, n // 10)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.15)))  # log‑normal factor
    cand[start:stop] *= factor
    return cand

                     budget_s: float = 500,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative height sequence with a low evaluation score.
    The optimiser performs projected gradient descent on the maximal
    convolution entry, with adaptive step‑size and occasional random
    block perturbations.  It runs for at most *budget_s* seconds.
    """
    rng = np.random.default_rng(seed)

    # ---- Determine problem size and initial sequence -----------------
    if "height_sequence_1" in globals():
        init = np.array(height_sequence_1, dtype=np.float64)
        n = init.size

        try:

        except Exception:

    else:

    else:

    # clip into the evaluator's allowed range (just in case)

    # fixed total sum – the optimizer works on the simplex with this sum
    target_sum = float(np.sum(a))
    if target_sum < 0.01:

    # ---- Initial evaluation -----------------------------------------
    conv = convolution_fft(a, n_fft)

    best_score = 2.0 * n * max_conv / (target_sum ** 2)

    # ---- Optimisation parameters ------------------------------------
    step_factor = 0.04                      # initial step size
    min_step = 1e-8

            cur_score = cand_score
            step_factor = min(step_factor * 1.10, max_step)  # be a bit more aggressive
            no_improve_counter = 0
        else:

            step_factor = max(step_factor * 0.50, min_step)
            no_improve_counter += 1

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{iteration:>6}] new best = {best_score:.6f}")

                print("[early] target reached")
                break

        # occasional random diversification to avoid stagnation
        if no_improve_counter >= 2000:

            pert = block_perturbation(a, rng)
            pert = project_onto_simplex(pert, target_sum)
            # accept if it improves

            score_pert = 2.0 * n * float(conv_pert.max()) / (target_sum ** 2)
            if score_pert < cur_score:

                cur_score = score_pert
                no_improve_counter = 0
                step_factor = max(step_factor, 0.02)
            else:

                no_improve_counter = 0

    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
