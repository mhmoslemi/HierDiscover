def upsample(a, new_n):
    old_n = len(a)
    old_p = (np.arange(old_n) + 0.5)/old_n
    new_p = (np.arange(new_n) + 0.5)/new_n
    new_a = np.interp(new_p, old_p, a)
    # Scale to maintain sum sqrt(2*new_n)
    target_sum = np.sqrt(2.0 * new_n)
    new_a *= target_sum / new_a.sum()
    return new_a

def generate_beta(n, alpha, beta):
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, beta - 1.0)

    target = np.sqrt(2.0 * n)
    w = w * target / w.sum()
    w = np.clip(w, 0.0, 1000.0)
    return w

"""
A linear‑programming based optimizer for the step‑function objective.

The idea follows the description in the statement: start from a non‑negative
sequence f (here a symmetric Beta(α,α) – the “arcsine” shape) normalised so
that Σf = sqrt(2·n).  Let M = max(f*f).  Solve the linear program

    maximise   Σ g_j
    subject to g_j ≥ 0
               (f * g)_k ≤ M       for all k = 0 … 2n‑2

The optimum g adds as much mass as possible while keeping the convolution
with the current f under the current worst‑case value M.  After rescaling g
to the same ℓ₁‑norm as f we mix the two sequences

    f ← (1‑t)·f + t·g                     (t small, e.g. 0.2)

which strictly reduces the maximal convolution entry.  Repeating a few
times yields a sequence with a lower evaluation score.

The implementation uses `scipy.optimize.linprog` (HiGHS solver) with a
sparse constraint matrix.  Several problem sizes (n = 128, 256, 512, 1024)
are tried within the time budget and the best result is returned.
If the linear program cannot be solved we fall back to a simple random
hill‑climber (the strategy used previously).
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Helpers (no closures, top‑level as required)
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray) -> float:
    """Quick evaluation of the objective (2·n·max(conv)/(sum²))."""
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    # convolution via FFT (size = next power of two for speed/accuracy)
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)


def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) shape: w_i ∝ p_i^{α‑1}(1‑p_i)^{α‑1},
    p_i = (i+0.5)/n, scaled so that Σw = sqrt(2·n).
    """
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, alpha - 1.0)
    # guard against a completely flat vector (α≈1)
    if w.sum() == 0.0:
        w = np.ones_like(w)
    target = sqrt(2.0 * n)
    w = w * (target / w.sum())
    w = np.clip(w, 0.0, 1000.0)
    return w


def _build_convolution_matrix(f: np.ndarray) -> "scipy.sparse.csr_matrix":
    """
    Build the (2n‑1) × n matrix A such that (A @ g) = f * g.
    The matrix is stored in CSR format.
    """
    import scipy.sparse as sp

    n = f.shape[0]
    rows = []
    cols = []
    data = []
    # for each output index k we add the appropriate coefficients
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs over the indices of f that contribute to (f*g)[k]
        for i in range(i_start, i_end + 1):
            j = k - i               # corresponding index in g
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_step(current_f: np.ndarray, mix_factor: float,
             target_sum: float, deadline: float):
    """
    Perform a single LP‑based improvement step.
    Returns (new_f, new_M, succeeded) where succeeded is False if the LP
    could not be solved or the deadline was exceeded.
    """
    import scipy.optimize as opt
    import scipy.sparse as sp

    if time.monotonic() >= deadline:
        return current_f, None, False

    n = current_f.shape[0]
    M = np.max(np.convolve(current_f, current_f))

    # Build the convolution matrix A such that A @ g = current_f * g
    A = _build_convolution_matrix(current_f)

    # Upper bounds vector (all entries must stay ≤ M)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Objective: maximise Σ g  → minimise -Σ g
    c = -np.ones(n, dtype=np.float64)

    # Solve the LP (HiGHS, accepts sparse matrices)
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b,
                          bounds=(0, None), method='highs',
                          options={'presolve': True, 'time_limit': max(0.0, deadline - time.monotonic())})
    except Exception:      # pragma: no cover
        return current_f, None, False

    if not res.success:
        return current_f, None, False

    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 == 0.0:
        return current_f, None, False

    # Rescale g to have the same ℓ₁‑norm as the target (sqrt(2·n))
    g = g0 * (target_sum / sum_g0)

    # Mix the new direction into the current sequence
    new_f = (1.0 - mix_factor) * current_f + mix_factor * g
    new_f = np.clip(new_f, 0.0, 1000.0)

    new_M = np.max(np.convolve(new_f, new_f))
    return new_f, new_M, True


def _hill_climb(f: np.ndarray, deadline: float) -> np.ndarray:
    """
    Very simple random hill‑climber used as a fallback when the LP
    cannot be solved.  It moves a random mass from one index to another
    and keeps the improvement.
    """
    n = f.shape[0]
    target_sum = sqrt(2.0 * n)
    conv = np.convolve(f, f)
    best_score = conv.max()
    best_f = f.copy()
    rng = np.random.default_rng()
    it = 0
    while time.monotonic() < deadline and it < 20000:
        i, j = rng.integers(0, n, size=2)
        if i == j or f[i] <= 0.0:
            continue
        # move a small fraction of the donor mass
        delta = rng.random() * 0.5 * f[i]
        if delta <= 0.0:
            continue
        # tentative update (copy only the convolution slice)
        conv_new = conv.copy()
        conv_new[i:i + n] -= 2.0 * delta * f
        conv_new[j:j + n] += 2.0 * delta * f
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta
        new_score = conv_new.max()
        if new_score < best_score - 1e-12:
            # accept
            f[i] -= delta
            f[j] += delta
            conv = conv_new
            best_score = new_score
            best_f = f.copy()
            if it % 2000 == 0:
                print(f"[hill] iter={it} score={best_score:.6f}")
        it += 1
    return best_f

            best_seq = f.copy()
            best_n = n
            print(f"[best_update] new best score {best_score:.6f} with n={n}")

    # ------------------------------------------------------------------
    # 2.  If nothing was found (unlikely) fall back to a uniform sequence
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = np.full(n, sqrt(2.0 * n) / n)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        best_n = n
        print(f"[fallback] uniform seq score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final printout
    # ------------------------------------------------------------------
    print(f"[overall] best length = {best_n}, best score = {best_score:.6f}")
    return [float(v) for v in best_seq]
