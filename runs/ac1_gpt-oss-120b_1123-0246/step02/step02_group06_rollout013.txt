def run_iterative_lp(f, max_iters=30, t=0.2, tol=1e-6):
    import cvxpy as cp
    n = len(f)
    M = np.max(np.convolve(f, f))
    sqrt2n = np.sqrt(2*n)
    for it in range(max_iters):
        # Build LP
        b = cp.Variable(n, nonneg=True)
        constraints = []
        # For each shift
        for k in range(2*n-1):
            # compute contributions
            # variable product: sum_i f[i] * b[k-i] or equivalently sum_j b[j] * f[k-j] when indices valid.
            # We'll compute as cp.sum(cp.multiply(f_sub, b_sub)) maybe.
            # Determine valid j range
            j_start = max(0, k - (n-1))
            j_end = min(k, n-1)
            # slice length = j_end - j_start + 1
            if j_start <= j_end:
                f_slice = f[k - np.arange(j_start, j_end+1)]
                b_slice = b[j_start:j_end+1]
                constraints.append(cp.sum(cp.multiply(f_slice, b_slice)) <= M)
        # objective is maximize sum(b)
        objective = cp.Maximize(cp.sum(b))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
        if b.value is None:

            break
        b_val = np.array(b.value, dtype=np.float64)

        sum_b = np.sum(b_val)
        if sum_b < 1e-12:

        M = new_M
    return f

def propose_candidate(seed=123, budget_s=500, **kwargs):
    import time, numpy as np, cvxpy as cp, math

    # Determine length
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 2000
    else:
        n = min(n, 2000)  # keep manageable

    rng = np.random.default_rng(seed)

    # arcsine shape
    def arcsine_seq(length):
        a = np.empty(length, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, length):
            a[i] = a[i-1] * ((2*i - 1) * (2*i) / (4.0 * i * i))
        return a

    # Build initial f normalized
    f = arcsine_seq(n)
    sqrt2n = math.sqrt(2.0 * n)
    f = f * (sqrt2n / np.sum(f))

    # evaluate
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, init_score={best_score:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # 2 secs margin

    # parameters
    t = 0.2   # mixing factor
    iter_idx = 0

    # main LP+SA loop
    while time.monotonic() < deadline:
        iter_idx += 1
        # compute current bound M
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))

        # Build convolution matrix A (shape (2n-1, n))
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        # fill using broadcasting: each column j gets f placed at rows j .. j+n-1
        # this can be done with a loop for clarity
        for j in range(n):
            A[j : j + n, j] = f
        M_vec = np.full(rows, M, dtype=np.float64)

        # Solve LP: maximize sum(b) subject to A @ b <= M, b >= 0
        b = cp.Variable(n, nonneg=True)
        constraints = [A @ b <= M_vec]
        objective = cp.Maximize(cp.sum(b))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)

            break
        if b.value is None:
            print(f"[iter {iter_idx}] LP did not produce solution.")
            break

        b_val = np.array(b.value, dtype=np.float64)

"""
Propose a candidate step function using a convex‐linear‐programming iteration
followed by a short simulated‑annealing refinement.

The algorithm follows the iterative scheme described in the statement:
1. Start from the analytically optimal arcsine shape and scale it so that
   its ℓ₁‑norm equals √(2·n) (the objective is scale‑invariant).
2. Compute M = max(f·f) (full convolution).
3. Solve the linear program
        maximise   Σ b_j
        subject to b_j ≥ 0
                  (f * b)_k ≤ M   for all k (full convolution)
   The LP is a standard linear program because f is fixed – the constraints are
   linear in b.  The constraint matrix is just the convolution matrix of f.
4. Rescale the LP solution b to have the same ℓ₁‑norm √(2·n) and mix it into f:
        f ← (1‑t)·f + t·g   (with t≈0.3).  This mixture keeps the norm invariant.
5. Repeat steps 2‑4 until the time budget expires or the improvement stalls.
6. Finally perform a few simulated‑annealing moves (transfer, block scaling,
   rebalance) to polish the best sequence found.

The LP step reliably reduces the autocorrelation bound M, hence the
evaluation score 2·n·M/(‖f‖₁²).  The whole routine stays well within the
1000 s budget for sequences of length ≈2000.
"""

import time
import math
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the discrete arcsine sequence a_i = C·binom(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i) / (4.0 * i * i))
    return a


def conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the full convolution matrix A such that A @ b == np.convolve(f, b).
    For sequence length n, A has shape (2n‑1, n).
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j receives the whole vector f shifted by j positions
    for j in range(n):
        A[j : j + n, j] = f
    return A


# ----------------------------------------------------------------------
# Move primitives for the final annealing phase
# ----------------------------------------------------------------------
def transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = base.size
    i = int(rng.integers(0, max(1, n // 4)))
    j = int(rng.integers(3 * n // 4, n))
    if base[i] <= 0.0:
        return base.copy()
    frac = rng.random() * 0.9
    delta = frac * base[i]
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.13)))
    cand[start : start + width] *= factor
    if rng.random() < 0.30:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def rebalance_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = base.size
    cut = int(rng.integers(n // 10, 9 * n // 10))
    prefix = base[:cut]
    suffix = base[cut:]
    sum_prefix = prefix.sum()
    sum_suffix = suffix.sum()
    if sum_prefix <= 0.0:
        return base.copy()
    amount = rng.random() * 0.5 * sum_prefix
    factor_pre = max(0.0, (sum_prefix - amount) / sum_prefix)
    new_prefix = prefix * factor_pre
    if sum_suffix > 0.0:
        factor_suf = (sum_suffix + amount) / sum_suffix
        new_suffix = suffix * factor_suf
    else:
        new_suffix = np.full_like(suffix, amount / suffix.size)
    cand = np.concatenate([new_prefix, new_suffix])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a non‑negative step‑function minimizing `evaluate_sequence`.
    Uses a linear‑programming based iteration (the “alternating‑projection”
    method) and finishes with a brief simulated‑annealing polish.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the length to work with.
    # Keep it modest (≤2000) so that the LP stays tractable.
    # --------------------------------------------------------------
    external = globals().get("height_sequence_1", [])
    n = int(len(external))
    if n == 0:
        n = 2000
    else:
        n = min(n, 2000)          # avoid huge matrices

    # --------------------------------------------------------------
    # 1. Start from the arcsine shape and normalise ‖f‖₁ = √(2n).
    # --------------------------------------------------------------
    f = build_arcsine_sequence(n)
    target_norm = math.sqrt(2.0 * n)
    f = f * (target_norm / np.sum(f))

    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # keep a small margin
    it = 0
    mix_factor = 0.30               # t in the description

    # --------------------------------------------------------------
    # 2. Main optimisation loop: LP step → mix → evaluate.
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # a) current autocorrelation bound M
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))

        # b) build the convolution matrix A (dense, but still modest)
        A = conv_matrix(f)               # shape (2n‑1, n)
        M_vec = np.full(A.shape[0], M, dtype=np.float64)

        # c) linear program: maximise Σ b  subject to  A·b ≤ M , b ≥ 0
        b = cp.Variable(n, nonneg=True)
        constraints = [A @ b <= M_vec]
        objective = cp.Maximize(cp.sum(b))
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except Exception as exc:
            print(f"[{it}] LP solve exception: {exc}")
            break

        if b.value is None:
            print(f"[{it}] LP did not return a solution.")
            break

        b_val = np.asarray(b.value).astype(np.float64)
        sum_b = float(np.sum(b_val))
        if sum_b <= 0.0:
            print(f"[{it}] Obtained zero sum from LP, aborting.")
            break

        # d) scale to keep the ℓ₁‑norm = √(2n)
        g = b_val * (target_norm / sum_b)

        # e) mix the new candidate with the current sequence
        f_new = (1.0 - mix_factor) * f + mix_factor * g
        f_new = np.clip(f_new, 0.0, 1000.0)

        # f) evaluate
        cur_score = float(evaluate_sequence(f_new.tolist()))
        if not np.isfinite(cur_score):
            # ignore pathological candidates
            continue

        # keep the best ever seen
        if cur_score < best_score:
            best = f_new.copy()
            best_score = cur_score
            print(f"[{it:>5}] new best score = {best_score:.6f}")

        # replace the working sequence
        f = f_new

        # early stop if we are already comfortably below the target
        if best_score < 1.50:
            break

    # --------------------------------------------------------------
    # 3. Optional short simulated‑annealing refinement (few seconds).
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        # very quick SA to wiggle a little
        cur = best.copy()
        cur_score = best_score
        T_start = 0.02
        T_end = 1e-4
        sa_end = time.monotonic() + remaining
        sa_it = 0
        while time.monotonic() < sa_end:
            sa_it += 1
            elapsed = (time.monotonic() - start_time) / budget_s
            T = T_start * (1.0 - elapsed) + T_end * elapsed
            # pick a move
            r = rng.random()
            if r < 0.45:
                cand = transfer_move(cur, rng)
            elif r < 0.80:
                cand = block_perturbation(cur, rng)
            else:
                cand = rebalance_move(cur, rng)

            cand_score = float(evaluate_sequence(cand.tolist()))
            if not np.isfinite(cand_score):
                continue

            if cand_score < cur_score:
                accept = True
            else:
                prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
                accept = rng.random() < prob

            if accept:
                cur = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                    print(f"[SA {sa_it:>6}] improved best = {best_score:.6f}")

    print(f"[final] best score {best_score:.6f} after {it} LP iterations")
    # Return plain Python list as required by the harness.
    return [float(x) for x in best]
