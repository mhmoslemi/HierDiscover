    1. the current vector a = exp(w) is normalised to the target ℓ₁‑norm

    3. a soft‑max (temperature β) is applied to the autocorrelation to obtain

    4. the gradient of the smooth objective w.r.t. a is computed analytically
       (2·a⋆w_rev) using another FFT.
    5. the gradient is transferred to the log‑variables (∂/∂log a_i = a_i·∂/∂a_i)
       and an Adam update is performed.
    6. occasional log‑normal perturbations and restart on stagnation are used
       to escape plateaux.

    The best sequence (lowest true evaluation value) found before the time
    budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    # 1. Choose length and initialise

    w = np.log(a0 + eps)

    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # hyper‑parameters
    base_lr = 0.5          # learning‑rate for log‑variables
    min_lr = 0.01
    beta_start = 0.5
    beta_end = 40.0        # final temperature (sharp soft‑max)

    b_full = _convolve_fft(best_seq, best_seq, _next_pow2(2 * n - 1))

            best_seq = a.copy()
            best_score = maxb
            last_improve = it
            best_iter = it

        elapsed = time.monotonic() - start

        beta = beta_start + (beta_end - beta_start) * frac

        # soft‑max weights over all lags
        exp_term = np.exp(beta * (b - maxb))
        wts = exp_term / exp_term.sum()

        # Gradient of the smooth objective (w.r.t. a)
        # ------------------------------------------------------------------
        w_rev = wts[::-1]                     # reversed weights, length 2n‑1
        W_rev = np.fft.rfft(w_rev, N_fft)
        conv_aw_full = np.fft.irfft(A * W_rev, N_fft)

        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv_aw[idx]

        grad_log = grad_a * a
        # remove pure scaling direction (does not affect the constrained problem)
        grad_log -= grad_log.mean()

        # ------------------------------------------------------------------
        # Adam update on log‑variables
        # ------------------------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        v_hat = v / (1.0 - beta2 ** it)

        # ------------------------------------------------------------------

            w += noise

            # next loop iteration

        # ------------------------------------------------------------------

            m.fill(0.0)
            v.fill(0.0)

            print(f"[{it:6d}] restart due to stagnation")

    # Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[done] iterations = {it}, best score = {best_score:.6f} (found at iter {best_iter})")
    # Ensure the returned sequence respects the evaluator's clipping rules
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
