"""
Hybrid exponentiated‑gradient / simulated‑annealing optimizer.

The algorithm starts from the analytically optimal “arcsine” sequence.
During the search we repeatedly:

* compute the full convolution a*a,
* construct a soft‑max weight vector that concentrates on the
  largest convolution entries (parameter α),
* obtain the gradient of the weighted sum w.r.t. a via an FFT‑based
  cross‑correlation,
* perform an exponentiated‑gradient step a ← a·exp(−η·∇) and renormalise
  to keep the ℓ₁‑norm fixed,
* occasionally try a random block‑scaling perturbation (helps to escape
  plateaus) and accept only moves that improve the exact objective
  `evaluate_sequence`.

The temperature for the occasional worsening moves and the step size η
are slowly annealed, while α is slowly increased so that the soft‑max
focuses more on the true maximum as the optimisation proceeds.
The best sequence seen is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: arcsine (inverse‑square‑root) sequence
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a where a[i] = binom(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence: a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def renormalise(seq: np.ndarray, target_sum: float = 1.0) -> np.ndarray:
    """Scale seq so that its ℓ₁‑norm equals `target_sum` (clip first)."""
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur == 0.0:
        return seq
    seq = seq * (target_sum / cur)
    return seq


def block_perturbation(base: np.ndarray,
                       rng: np.random.Generator,
                       target_sum: float = 1.0) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:start + width] *= factor
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return renormalise(cand, target_sum)


# ----------------------------------------------------------------------
# FFT‑based convolution / correlation (linear, not circular)
# ----------------------------------------------------------------------
def conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length `size`."""
    Fa = np.fft.rfft(a, size)
    conv_full = np.fft.irfft(Fa * Fa, size)
    return conv_full[: 2 * a.size - 1]


def corr_fft(w: np.ndarray, a: np.ndarray, size: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ⋆ a) and extraction of the
    gradient vector g_j = Σ_i w_i a_{i-j}, j=0..n‑1.
    """
    Fw = np.fft.rfft(w, size)
    Fa = np.fft.rfft(a, size)
    corr_full = np.fft.irfft(Fw * np.conj(Fa), size)
    L = w.size + a.size - 1
    corr_full = corr_full[:L]                # linear correlation length
    n = a.size
    # g_j = correlation at lag = -j → index = (n‑1)‑j
    g = corr_full[(n - 1) :: -1][:n]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 900, **kwargs):
    """
    Return a list of non‑negative heights (step‑function values) that
    minimise `evaluate_sequence`.  The optimisation combines a
    soft‑max guided exponentiated‑gradient descent with occasional
    stochastic block perturbations.  The search runs until the time
    budget expires (or the target ≤1.503 is reached) and the best
    sequence encountered is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size: use the provided construction if available,
    # otherwise fall back to a modest size.
    # ------------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256

    # ------------------------------------------------------------------
    # 1. Initialise with the arcsine shape and normalise to ℓ₁‑norm = 1.
    # ------------------------------------------------------------------
    base_seq = build_arcsine_sequence(n)
    a = renormalise(base_seq, target_sum=1.0)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # Pre‑compute FFT length (big enough for both convolution and correlation)
    fft_len = next_pow2(3 * n)          # ≥ 2n‑1 + n‑1

    # Hyper‑parameters (will be annealed)
    alpha = 10.0                        # soft‑max sharpness
    eta = 0.35                          # exponentiated‑gradient step size
    temp = 0.01                         # temperature for occasional uphill moves
    iteration = 0

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # ----- convolution and current score -----
        conv = conv_fft(a, fft_len)               # a*a, length 2n‑1
        max_c = conv.max()
        cur_score = 2.0 * n * max_c               # sum(a)=1 ⇒ objective = 2n·max_c
        # keep exact best when using the provided `evaluate_sequence`
        # (the function does the same computation but with safety checks)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{iteration:>6}] new best = {best_score:.6f}")

            if best_score < 1.503:               # early exit if target reached
                break

        # ----- soft‑max over convolution entries -----
        # shift by max_c for numerical stability (max term → 1)
        weights = np.exp(alpha * (conv - max_c))
        w_sum = weights.sum()
        if w_sum == 0.0:                         # extremely unlikely, fall back to uniform
            weights = np.ones_like(weights) / weights.size
        else:
            weights /= w_sum

        # ----- gradient of the weighted sum (exponentiated‑gradient) -----
        g = corr_fft(weights, a, fft_len)        # shape (n,)

        # ----- propose an exponentiated‑gradient step -----
        a_candidate = a * np.exp(-eta * g)
        a_candidate = renormalise(a_candidate, target_sum=1.0)

        # evaluate the candidate with the exact objective
        cand_score = evaluate_sequence(a_candidate.tolist())

        # accept improving moves; otherwise allow a small uphill move
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(temp, 1e-9)):
            a = a_candidate
            cur_score = cand_score

        # ----- occasional random block perturbation (helps to escape plateaus) -----
        if iteration % 50 == 0:
            pert = block_perturbation(a, rng, target_sum=1.0)
            pert_score = evaluate_sequence(pert.tolist())
            if pert_score < cur_score or rng.random() < 0.05:
                a = pert
                cur_score = pert_score

        # ----- annealing / schedule -----
        # increase the softness parameter (focus more on true max)
        alpha = min(alpha * 1.008, 500.0)
        # slowly reduce step size
        eta = max(eta * 0.995, 5e-4)
        # cool the temperature for uphill moves
        temp = max(temp * 0.996, 1e-6)

        # occasional reset with a tiny random jitter (prevents complete stagnation)
        if iteration % 1000 == 0:
            jitter = rng.normal(0.0, 0.02, size=n)
            a = renormalise(np.clip(a + jitter, 0.0, None), target_sum=1.0)

    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best_seq]
