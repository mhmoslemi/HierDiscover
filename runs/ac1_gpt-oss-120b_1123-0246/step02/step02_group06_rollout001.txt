"""
A fast, gradient‑guided simulated‑annealing optimizer for the
``evaluate_sequence`` objective.

The algorithm starts from the analytically optimal “arcsine” sequence,
normalises it to unit ℓ1‑norm, and then repeatedly performs
   – a *gradient transfer* move that shifts mass from indices that
     contribute most to the current worst autocorrelation (the active
     lag(s)) towards indices that contribute the least,
   – a *block‑scale* move (random contiguous block multiplied by a
     factor) to explore larger deformations,
   – a cheap random *transfer* move and a tiny Gaussian *noise*
     perturbation for diversity.

At each iteration the temperature is linearly cooled; moves that
increase the objective are occasionally accepted according to the
Metropolis rule.  The best sequence found before the time budget expires
is returned as a plain list of floats.

The code uses FFT‑based convolution (O(n log n)) to evaluate the
objective quickly even for several‑thousand‑element sequences.
"""

import time
import math
import numpy as np
from math import exp

# ----------------------------------------------------------------------
# Helper: build the arcsine (optimal) sequence of length n
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i) / 4**i   (as a float)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    # normalise to unit ℓ1‑norm (scale does not affect the objective)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Convolution via FFT (real‑valued, returns the full length 2n‑1 array)
# ----------------------------------------------------------------------
def fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """Return a*a (full convolution) using the FFT."""
    n = a.shape[0]
    size = 2 * n - 1
    # choose power‑of‑two length for speed
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    B = np.fft.irfft(A * A, n=m)[:size]
    # Numerical noise may create tiny negative values – clip them
    B = np.maximum(B, 0.0)
    return B


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence but much cheaper)
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """2·n·max(a*a)/(sum(a)²)  – the same quantity returned by
    evaluate_sequence for valid inputs."""
    total = a.sum()
    if total < 0.01:
        return float('inf')
    conv = fft_convolve_self(a)
    max_b = conv.max()
    return 2.0 * a.shape[0] * max_b / (total * total)


# ----------------------------------------------------------------------
# Move primitives
# ----------------------------------------------------------------------
def transfer_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from a random source to a random target."""
    n = a.shape[0]
    src = rng.integers(0, n)
    dst = rng.integers(0, n)
    if src == dst or a[src] == 0.0:
        return a.copy()
    frac = rng.random() * 0.9          # up to 90 % of the source mass
    delta = frac * a[src]
    b = a.copy()
    b[src] -= delta
    b[dst] += delta
    return b


def block_perturbation(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    n = a.shape[0]
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.13)))   # ~log‑normal(σ=0.13)
    b = a.copy()
    b[start:stop] *= factor
    # occasional symmetrisation – helps keep the shape regular
    if rng.random() < 0.30:
        b = 0.5 * (b + b[::-1])
    # renormalise to unit sum (preserve scale‑invariance)
    b_sum = b.sum()
    if b_sum > 0:
        b /= b_sum
    return b


def noise_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Add tiny Gaussian noise and re‑normalise."""
    noise = rng.normal(0.0, 1e-4, size=a.shape)
    b = a + noise
    b = np.clip(b, 0.0, None)
    s = b.sum()
    if s > 0:
        b /= s
    return b


def gradient_transfer_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Identify the lag(s) where the autocorrelation is maximal.
    Compute, for each index i, a “responsibility” r_i = sum_{k∈active}
    (a_{i+k}+a_{i‑k}) (out‑of‑range terms are ignored).
    Transfer a small fraction of mass from a high‑responsibility index
    to a low‑responsibility one.  The total ℓ1‑norm stays unchanged.
    """
    n = a.shape[0]
    conv = fft_convolve_self(a)
    M = conv.max()
    # all lags whose value is within a tiny tolerance of the maximum
    eps = 1e-12
    active = np.where(conv >= M - eps)[0]

    # compute responsibilities
    r = np.zeros_like(a)
    for k in active:
        if k == 0:
            # derivative of a_i^2 contributes 2*a_i
            r += 2.0 * a
            continue
        # shift left: a[i+k] contributes to r[i] when i+k < n
        r[: n - k] += a[k:]
        # shift right: a[i‑k] contributes to r[i] when i‑k >= 0
        r[k:] += a[: n - k]

    # safety: if all responsibilities are zero (unlikely) return a copy
    if np.all(r == 0):
        return a.copy()

    # probability of picking a source (high responsibility) and a target (low)
    prob_src = r / r.sum()
    src = rng.choice(n, p=prob_src)

    # low‑responsibility indices – invert the responsibility
    inv = (r.max() - r) + 1e-12
    prob_tgt = inv / inv.sum()
    dst = rng.choice(n, p=prob_tgt)

    if src == dst or a[src] == 0.0:
        return a.copy()

    # move a modest fraction of the source mass toward the target
    frac = rng.random() * 0.4          # up to 40 % of source mass
    delta = frac * a[src]
    b = a.copy()
    b[src] -= delta
    b[dst] += delta
    return b


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 950.0,
                     **kwargs) -> list[float]:
    """
    Gradient‑guided simulated annealing for minimise ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length of the search vector.
    # ------------------------------------------------------------------
    # Prefer the length of a previously supplied construction.
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        # fallback – a size that offers a good trade‑off between quality
        # and speed.  5600 is close to the size used in the reference run.
        n = 5600

    # ------------------------------------------------------------------
    # Initialise from the arcsine distribution (the provably good baseline).
    # ------------------------------------------------------------------
    cur = build_arcsine_sequence(n)          # already ℓ1‑normalised
    cur_score = compute_score(cur)

    best = cur.copy()
    best_score = cur_score

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing loop.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # keep 1 s margin
    it = 0

    # Linear temperature schedule (high → low)
    T_start = 0.04
    T_end = 1e-6

    while time.monotonic() < deadline:
        it += 1
        elapsed = (time.monotonic() - start_time) / budget_s
        T = T_start * (1.0 - elapsed) + T_end * elapsed

        # ----------------------------------------------------------------
        # Pick a move type (probabilities tuned empirically)
        # ----------------------------------------------------------------
        r = rng.random()
        if r < 0.35:
            cand = gradient_transfer_move(cur, rng)
        elif r < 0.60:
            cand = block_perturbation(cur, rng)
        elif r < 0.85:
            cand = transfer_move(cur, rng)
        else:
            cand = noise_move(cur, rng)

        cand_score = compute_score(cand)
        if not np.isfinite(cand_score):
            continue

        # ----------------------------------------------------------------
        # Metropolis acceptance
        # ----------------------------------------------------------------
        if cand_score < cur_score:
            accept = True
        else:
            prob = exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[{it:>7}] new best = {best_score:.6f}")

        # ----------------------------------------------------------------
        # Occasionally re‑anchor the current state to the best seen
        # ----------------------------------------------------------------
        if it % 25000 == 0:
            cur = best.copy()
            cur_score = best_score

        # ----------------------------------------------------------------
        # Early exit if we have already crossed an ambitious target.
        # ----------------------------------------------------------------
        if best_score < 1.50:          # comfortably below the π/2 barrier
            break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return a plain Python list as required by the harness.
    return [float(x) for x in best]
