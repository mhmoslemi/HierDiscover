        if s == 0:

    best_seq = None

    deadline = start + budget_s - 0.5

        # we will compute conv and find max index.
        conv = np.convolve(cur_seq, cur_seq)
        kmax = int(np.argmax(conv))
        # pick i such that 0 <= i <= kmax and i < n, j = kmax - i (both in [0,n))
        # choose a random i in the overlap region
        i0 = rng.integers(max(0, kmax - (n-1)), min(kmax, n-1) + 1)

        # Choose donor = max(i0,j0), recipient = random other index far from both
        donor = max(i0, j0)

        if donor < n//2:

    idx = np.arange(len(base))
    if gamma != 0:
        factor_power = np.power(idx+1, gamma)
    else:

    if t != 1.0:

    else:
        factor_exp = np.ones_like(base)
    a = base * factor_power * factor_exp
    # Protect against zero sum maybe
    total = a.sum()
    if total == 0:
        return a
    a = a / total

import numpy as np
import time
import math

def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return arcsine step heights for given length."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i-1] * ((2i-1)(2i) / (4 i^2))
        a[i] = a[i-1] * ((2*i-1)*(2*i)) / (4.0*i*i)
    return a

def tilted_arcsine(base: np.ndarray, gamma: float, t: float) -> np.ndarray:
    """Return a new sequence obtained from `base` by applying a power tilt
    (i+1)^gamma and an exponential tilt t^i, then renormalised to sum==1."""
    n = base.size
    idx = np.arange(n)
    # power factor
    if gamma == 0.0:
        power = 1.0
    else:
        power = np.power(idx + 1.0, gamma)
    # exponential factor
    if t == 1.0:
        expfac = 1.0
    else:
        # use log to avoid overflow
        logt = math.log(t)
        expfac = np.exp(idx * logt)
    seq = base * power * expfac
    total = seq.sum()
    if total <= 0.0:
        # fallback to uniform
        seq = np.full(n, 1.0 / n)
        total = 1.0
    seq = seq / total
    return seq

    is returned as a plain Python list.
    """

    n = int(len(ref))
    if n == 0:
        n = 256

"""
Search for a low‑score step function.

The algorithm builds on the analytically optimal arcsine shape
(a_i = C·C(2i,i)/4**i).  It then explores a 2‑parameter family

    a_i = base_i · (i+1)^γ · t^i ,

where `base` is the normalised arcsine sequence, `γ` is a power tilt
and `t` is an exponential tilt.  A coarse grid over (γ,t) is scanned;
the best grid point is kept.  Afterwards a lightweight simulated‑annealing
loop refines (γ,t) by small random perturbations, always keeping the
best sequence found.  The evaluation function `evaluate_sequence`
is called directly; the final sequence is returned as a plain list
of floats.  The whole procedure respects the given time budget.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# 1.  Fundamental building block: the arcsine step heights.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the arcsine coefficients a_i = binom(2i,i)/4**i for i=0..length‑1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    return a


# ----------------------------------------------------------------------
# 2.  Tilted version of the arcsine shape.
# ----------------------------------------------------------------------
def tilted_arcsine(base: np.ndarray, gamma: float, t: float) -> np.ndarray:
    """
    Apply a power tilt (i+1)^γ and an exponential tilt t^i to `base`,
    then renormalise so that the sum equals 1 (the objective is scale‑invariant).
    """
    n = base.size
    idx = np.arange(n, dtype=np.float64)

    # power factor (i+1)^γ
    if gamma == 0.0:
        power = 1.0
    else:
        power = np.power(idx + 1.0, gamma)

    # exponential factor t^i  (computed via log to avoid overflow)
    if t == 1.0:
        expfac = 1.0
    else:
        logt = math.log(t)
        expfac = np.exp(idx * logt)

    seq = base * power * expfac
    total = seq.sum()
    if total <= 0.0:               # fallback to a uniform vector (should never happen)
        seq = np.full(n, 1.0 / n)
        total = 1.0
    seq = seq / total
    return seq


# ----------------------------------------------------------------------
# 3.  Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500.0, **kwargs):
    """
    Find a step‑function with a low `evaluate_sequence` score.

    * start from the arcsine shape,
    * perform a modest grid search over (γ, t),
    * refine the best parameters with a simple simulated‑annealing loop,
    * and finally return the best sequence as a list of floats.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the length we have to use.
    # --------------------------------------------------------------
    ref = globals().get("height_sequence_1", [])
    n = int(len(ref))
    if n == 0:
        n = 256

    # --------------------------------------------------------------
    # Base arcsine sequence (normalised to sum = 1 – scale does not matter).
    # --------------------------------------------------------------
    base = build_arcsine_sequence(n)
    base = base / base.sum()

    # Helper to call the evaluator (already imported in the environment).
    def evaluate(arr):
        return float(evaluate_sequence(arr.tolist()))

    # --------------------------------------------------------------
    # 1) Coarse grid search over the two tilt parameters.
    # --------------------------------------------------------------
    best_seq = base.copy()
    best_score = evaluate(best_seq)
    best_gamma = 0.0
    best_t = 1.0

    print(f"[init] length={n}, base score={best_score:.6f}")

    start = time.time()
    # parameter ranges – chosen to stay safely inside floating‑point limits
    gamma_vals = np.linspace(-0.5, 0.5, 21)          # 21 values
    t_vals = np.linspace(0.90, 1.13, 25)            # exponential tilt (≤1.13 avoids overflow)

    # --------------------------------------------------------------

            cur_t = best_t
            cur_seq = best_seq.copy()
            cur_score = best_score

    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    # Return a plain Python list as required by the harness.
    return [float(x) for x in best_seq]
