    • finds a k where the convolution is maximal,
    • moves a little mass away from the indices that most contribute to that peak,
    • projects back onto the simplex {a_i≥0, Σa_i = √(2n)}.
Occasional random block perturbations are added to avoid stagnation.
Fast convolution is performed with an FFT, so thousands of iterations fit easily
into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def fast_convolve(a: np.ndarray) -> np.ndarray:
    """
    Return the linear convolution a*a using a power‑of‑two FFT.
    Length of the result is 2*len(a)-1.
    """
    n = a.size
    size = 2 * n - 1
    # next power of two for efficient FFT
    m = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:size]
    return conv


def simplex_projection(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project the vector v onto the simplex {x≥0, Σx = target_sum}.
    Implements the O(n log n) algorithm of
    “Efficient Projections onto the L1‑Ball”.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    # shift to non‑negative part first
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - target_sum
    rho_array = u - cssv / (np.arange(1, v.size + 1))
    rho = np.nonzero(rho_array > 0)[0]
    if rho.size == 0:
        # all entries are zero – return a uniform distribution
        return np.full_like(v, target_sum / v.size)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def build_arcsine_sequence(length: int) -> np.ndarray:
    """
    arcsine (inverse‑square‑root) coefficients:
    a[i] = binom(2i,i) / 4**i  (exact up to floating‑point noise).
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def block_perturbation(base: np.ndarray,
                       rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """
    Randomly scale a contiguous block (log‑normal factor) and re‑normalise.
    """
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # clip to evaluator's limits, then renormalise
    cand = np.clip(cand, 0.0, 1000.0)
    s = cand.sum()
    if s > 0:
        cand *= target_sum / s
    return cand


def eval_score(a: np.ndarray, target_sum: float) -> float:
    """
    Fast evaluation of the objective.
    After normalising the ℓ1‑norm to √(2n) the denominator is constant,
    so the score is simply max(conv).
    """
    conv = fast_convolve(a)
    max_conv = float(conv.max())
    return max_conv   # equal to evaluate_sequence for the normalised sum


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of floats) with a low
    `evaluate_sequence` score.  The routine runs a fast projected
    sub‑gradient descent for up to *budget_s* seconds, starting from the
    analytically optimal arcsine shape (normalised to Σa = √(2n)).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem dimension (use existing sequence if present)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        n = init.shape[0]
        # sanity‑check length – if the provided sequence is something else,
        # fall back to an arcsine start.
        if n == 0:
            n = 256
            base = build_arcsine_sequence(n)
        else:
            # use the given sequence as a starting point (after normalisation)
            base = init.copy()
    else:
        n = 256
        base = build_arcsine_sequence(n)

    # --------------------------------------------------------------
    # Normalise the ℓ1‑norm to √(2n) – this makes the score equal to max(conv)
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    base = np.clip(base, 0.0, 1000.0)
    if base.sum() == 0:
        base[:] = target_sum / n
    else:
        base *= target_sum / base.sum()

    cur = base.copy()
    cur_score = eval_score(cur, target_sum)
    best = cur.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin
    it = 0

    # --------------------------------------------------------------
    # Parameters for the sub‑gradient descent
    # --------------------------------------------------------------
    step0 = 0.12               # initial step size (fraction of the gradient)
    target = 1.503             # stop early once we beat the target
    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------------
        # 1.  Choose a move – mostly a directed sub‑gradient step,
        #     occasionally a random block perturbation.
        # ----------------------------------------------------------
        if rng.random() < 0.75:
            # ----- directed sub‑gradient move -----
            conv = fast_convolve(cur)
            max_val = conv.max()
            # there are often many ties; pick one uniformly at random
            max_idxs = np.flatnonzero(conv >= max_val - 1e-12)
            k = int(rng.choice(max_idxs))

            # sub‑gradient of max(conv) w.r.t. the entries of cur
            grad = np.zeros_like(cur)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            grad[idx] = cur[k - idx]          # g_i = a_{k-i}

            # step size slowly decays – classic √t schedule
            eta = step0 / np.sqrt(it)

            cand = cur - eta * grad
            cand = np.clip(cand, 0.0, None)   # keep non‑negative
            s = cand.sum()
            if s == 0.0:
                # degenerate – revert to current state
                cand = cur.copy()
                s = s if s != 0 else target_sum
            cand *= target_sum / s            # renormalise to exactly √(2n)
        else:
            # ----- occasional random block perturbation -----
            cand = block_perturbation(cur, rng, target_sum)

        # ----------------------------------------------------------
        # 2.  Evaluate the candidate (fast version)
        # ----------------------------------------------------------
        cand_score = eval_score(cand, target_sum)
        if not np.isfinite(cand_score):
            continue

        # ----------------------------------------------------------
        # 3.  Simple acceptance: keep any improving move.
        #    A tiny probability of accepting a worse move helps escape
        #    shallow local minima.
        # ----------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        elif rng.random() < 0.003:  # rare uphill move
            accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[{it:>6}] new best = {best_score:.6f}")

                if best_score < target:
                    print("[early] reached target")
                    break

        # ----------------------------------------------------------
        # 4.  Periodic global jitter to avoid long stagnation.
        # ----------------------------------------------------------
        if it % 4000 == 0 and rng.random() < 0.3:
            # re‑initialise from the arcsine baseline (still respects the best)
            cur = build_arcsine_sequence(n)
            cur = np.clip(cur, 0.0, 1000.0)
            cur *= target_sum / cur.sum()   # normalise
            cur_score = eval_score(cur, target_sum)

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # final sanity – clip once more (evaluate_sequence also does this)
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
