def random_transfer_move(base, rng):
    n = len(base)
    src = rng.integers(n)
    # ensure src have positive mass
    if base[src] <= 0:
        return base.copy()
    tgt = rng.integers(n-1)
    if tgt >= src:
        tgt += 1
    # fraction to move
    frac = rng.random() * 0.2 + 0.01  # 1% to 21%
    delta = min(frac * base[src], base[src])
    cand = base.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)

"""
Search for a low‑value step‑function sequence.

The search proceeds as follows:

1. Construct the classic “arcsine’’ (inverse‑square‑root) sequence of the
   required length and normalise its ℓ₁‑norm to √(2 n).  This sequence is the
   analytical baseline with score ≈ π/2.

2. Perform an extensive low‑dimensional sweep:
   * linear interpolation with the reversed arcsine (tilt),
   * power‑law deformation a_i←a_i**p,
   * a combination of both.
   The best candidate from this sweep becomes the starting point for the
   stochastic phase.

3. Stochastic local optimisation under a time budget.  Four move operators
   are used, each respecting the security checks of ``evaluate_sequence``:
   * **directed_transfer_move** – move a fraction of mass from the index
     that contributes most to the current maximal convolution entry to a
     low‑impact index;
   * **block_perturbation** – multiply a random contiguous block by a
     log‑normally distributed factor;
   * **gradient_step_move** – a cheap gradient descent step on the
     objective f = 2 n·max_b/∑a², derived analytically; and
   * **dct_perturbation** – add a small Gaussian perturbation to a few low‑frequency
     DCT coefficients of the current sequence.

   Moves are accepted if they improve the current score; otherwise they are
   occasionally accepted according to a simple simulated‑annealing schedule
   to escape local minima.

The routine respects the supplied time budget (default 500 s, the harness
gives 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.fft import dct, idct

# ----------------------------------------------------------------------
# 1.  Helper – build the “arcsine’’ (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the discrete arcsine sequence a_i = C(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# 2.  Move operators
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0, σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                      # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def directed_transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Move a fraction of mass from the index that contributes most
    to the current maximal convolution coefficient to a low‑impact index.
    """
    n = base.size
    conv = np.convolve(base, base)
    # pick the first occurrence of the global maximum (stable)
    max_idx = int(np.argmax(conv[:n]))
    # contributions a[i] * a[max_idx-i] for i ≤ max_idx
    idx = np.arange(max_idx + 1)
    contrib = base[idx] * base[max_idx - idx]
    src = int(idx[np.argmax(contrib)])             # most harmful index
    # choose a target among the 20 % smallest values
    low_thr = np.percentile(base, 20)
    low_candidates = np.where(base <= low_thr)[0]
    if low_candidates.size == 0:
        tgt = int(rng.integers(0, n - 1))
        if tgt >= src:
            tgt += 1
    else:
        tgt = int(rng.choice(low_candidates))
    if tgt == src:
        return base.copy()
    # fraction to move – allow a slightly larger range than before
    frac = rng.random() * 0.40 + 0.05            # 5 % … 45 %
    delta = frac * base[src]
    cand = base.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)


def gradient_step_move(base: np.ndarray, rng: np.random.Generator,
                      step_frac: float = 0.02) -> np.ndarray:
    """
    One inexpensive gradient‑descent step on the objective
    f = 2·n·max(conv)/sum².
    The gradient (ignoring the arg‑max change) is:
        df/da_i ∝ g_i·S – 2·M,
    where g_i = 2·a_{max_idx‑i} if max_idx‑i in range else 0.
    """
    n = base.size
    conv = np.convolve(base, base)
    max_idx = int(np.argmax(conv))
    M = conv[max_idx]
    S = base.sum()
    # build g_i
    g = np.zeros_like(base)
    if max_idx >= 0:
        idx = np.arange(max_idx + 1)
        g[idx] = 2.0 * base[max_idx - idx]
    # direction = 2·M – g·S  (negative gradient)
    direction = 2.0 * M - g * S
    # Apply a modest step; clamp updates to stay non‑negative
    step = step_frac * direction
    cand = base + step
    cand = np.maximum(cand, 0.0)
    cand = np.minimum(cand, 1000.0)
    return cand


def dct_perturbation(base: np.ndarray, rng: np.random.Generator,
                     n_coeff: int = 12, sigma: float = 0.02) -> np.ndarray:
    """
    Perturb the first ``n_coeff'' DCT coefficients with Gaussian noise.
    This produces smooth global changes.
    """
    coeffs = dct(base, type=2, norm='ortho')
    # keep DC and a few low‑frequency components, add small noise
    pert = coeffs.copy()
    noise = sigma * rng.normal(size=n_coeff)
    pert[:n_coeff] += noise
    cand = idct(pert, type=2, norm='ortho')
    return np.clip(cand, 0.0, 1000.0)

    """

    # --------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))

    # ------------------------------
    base_seq = build_arcsine_sequence(n)
    # Normalise ℓ₁‑norm to √(2 n) – scaling does not affect the score
    target_sum = np.sqrt(2.0 * n)
    base_seq = base_seq * (target_sum / base_seq.sum())
    best = base_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, start score={best_score:.6f}")

    # ------------------------------

    # ------------------------------
    rev_seq = best[::-1]
