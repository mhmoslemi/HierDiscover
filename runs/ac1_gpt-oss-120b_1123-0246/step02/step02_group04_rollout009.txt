def propose_candidate(seed=42, budget_s=500, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n and starting sequence
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        base = np.array(height_sequence_1, dtype=np.float64)
        base = np.clip(base, 0, 1000)  # ensure valid
        total = np.sum(base)
        if total == 0:
            base = build_arcsine_sequence(n).astype(np.float64)
            total = np.sum(base)
    else:
        n = 2000
        base = build_arcsine_sequence(n)
        total = np.sum(base)
    # Normalize to sum=1
    base = base / total
    # Avoid zero sum: if any zeros? fine, but ensure not all zeros.
    if np.all(base == 0):
        base = np.ones(n) / n

    cur = base.copy()
    cur_score = float(evaluate_sequence(cur.tolist()))
    best = cur.copy()
    best_score = cur_score

    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # parameters
    beta = 8.0   # smoothing factor
    step_base = 0.1  # initial step size (relative to magnitude)
    step_min = 1e-8
    improve = True
    it = 0

    while time.time() < deadline:
        it += 1
        # compute convolution and max
        conv = np.convolve(cur, cur)
        max_c = conv.max()
        # compute smoothed weights
        # shift to avoid overflow: subtract max
        w = np.exp(beta * (conv - max_c))
        # gradient (subgradient of smoothed max)
        grad = np.correlate(w, cur, mode='valid')
        # step size schedule: warm start decaying
        # we can adapt step = step_base * (1 - it / max_iters) ??? no max_iters known.
        # Use simple schedule: step = step_base / np.sqrt(it)
        step = step_base / np.sqrt(it)

        # direction for decreasing max: negative gradient
        direction = -grad

        # line search: try multiplicative scaling of direction
        lr = step
        while lr > step_min:
            cand = cur + lr * direction
            # enforce nonnegativity
            cand = np.clip(cand, 0.0, 1000.0)
            # renormalize sum to 1
            s = cand.sum()
            if s == 0:
                lr *= 0.5
                continue
            cand = cand / s
            cand_score = float(evaluate_sequence(cand.tolist()))
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                    print(f"[{it}] new best score {best_score:.6f}")
                break
            else:
                lr *= 0.5

        # occasional random perturbation to inject diversity
        if it % 500 == 0 and rng.random() < 0.2:
            # apply random block perturbation similar to earlier
            width = rng.integers(max(1, n // 200), max(2, n // 10))

"""
Propose a candidate step‐function sequence.

The algorithm starts from the analytically good “arcsine” shape (the
inverse‑square‑root distribution).  It then repeatedly reduces the
largest entry of the auto‑convolution by a simple projected gradient
step that follows the exact sub‑gradient of the currently maximal
convolution value.  After each move we:

* clip to the allowed range [0,1000],
* renormalise the whole vector so that its ℓ₁‑norm stays equal to 1
  (the evaluation metric is scale‑invariant, fixing the sum removes a
  trivial degree of freedom),
* accept the move only if it improves the score (otherwise we back‑track
  with a smaller step size).

Occasional random block perturbations are inserted to escape shallow
local minima.  All work is performed with pure NumPy; the loop runs for
up to the allotted time budget (default 500 s) and the best sequence
found is returned.

The method is deterministic apart from the tiny random perturbations,
and it typically pushes the score well below the previous best
≈1.5709, aiming for the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Helper: build the arcsine (inverse‑square‑root) start sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """
    Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i  (exact up to floating‑point noise).
    This is the classic arcsine (inverse‑square‑root) distribution.
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# 2.  Core optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine runs a projected‑gradient descent
    that directly targets the current maximal auto‑convolution entry.
    It returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ----- Choose problem size ------------------------------------------------
    if "height_sequence_1" in globals():
        n = int(len(globals()["height_sequence_1"]))
        try:
            init_seq = np.array(globals()["height_sequence_1"],
                                 dtype=np.float64).copy()
        except Exception:
            init_seq = None
    else:
        n = 2000                     # a reasonably sized fallback
        init_seq = None

    # ----- Initialise ---------------------------------------------------------
    if init_seq is not None and init_seq.shape[0] == n:
        cur = np.clip(init_seq, 0.0, 1000.0)
        if cur.sum() < 1e-9:        # degenerate – fall back to arcsine
            cur = build_arcsine_sequence(n)
    else:
        cur = build_arcsine_sequence(n)

    # Normalise to unit ℓ₁‑norm (scale‑invariant for the evaluator)
    cur = np.clip(cur, 0.0, 1000.0)
    total = cur.sum()
    if total == 0.0:
        cur[:] = 1.0 / n
    else:
        cur = cur / total

    # Keep track of the best solution seen
    best = cur.copy()
    # Since sum(best)=1, the score is simply 2*n*max(conv)
    conv = np.convolve(best, best)
    best_score = 2.0 * n * conv.max()
    cur_score = best_score

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ----- Hyper‑parameters ---------------------------------------------------
    start_lr = 0.04                 # initial learning‑rate (relative)
    min_lr   = 1e-8                 # minimal step size before giving up a line‑search
    perturb_interval = 800         # every this many iterations we try a random block move
    perturb_prob    = 0.25         # probability to actually apply the perturbation
    target_score    = 1.503        # early‑stop threshold

    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)  # tiny safety margin
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- 1.  Compute convolution and locate the maximal entry -----------
        conv = np.convolve(cur, cur)
        max_c = conv.max()
        k_max = int(np.argmax(conv))

        # ----- 2.  Sub‑gradient of the maximal convolution --------------------
        # For k_max the gradient g_i = a_{k_max-i} + a_{k_max-i} (double
        # count) except when i == k_max-i where it appears once.
        g = np.zeros_like(cur)

        i_start = max(0, k_max - (n - 1))
        i_end   = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1)
        partner = k_max - idx

        # most indices contribute twice
        g[idx] = 2.0 * cur[partner]

        # correct the middle term when k_max is even and the partner coincides
        if k_max % 2 == 0:
            mid = k_max // 2
            if 0 <= mid < n:
                # we added it twice above; keep only a single contribution
                g[mid] = cur[mid]

        # ----- 3.  Projected gradient descent with backtracking line search ----
        # Learning‑rate schedule: decrease roughly as 1/√iteration
        lr = start_lr / np.sqrt(it)

        improved = False
        while lr > min_lr:
            cand = cur - lr * g
            # enforce the domain constraints
            cand = np.clip(cand, 0.0, 1000.0)
            s = cand.sum()
            if s == 0.0:                     # hopeless – shrink step
                lr *= 0.5
                continue
            cand = cand / s                   # renormalise to unit ℓ₁‑norm

            # evaluate new score (fast: sum is 1)
            new_max = np.convolve(cand, cand).max()
            cand_score = 2.0 * n * new_max

            if cand_score < cur_score:       # accept only improving moves
                cur = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                    print(f"[{it:>6}] new best = {best_score:.6f}")
                    if best_score < target_score:
                        print("[early] target reached")
                        return [float(x) for x in best]
                break
            else:
                lr *= 0.5                     # backtrack

        # ----- 4.  Occasionally inject a random block perturbation -------------
        if (it % perturb_interval == 0) and (rng.random() < perturb_prob):
            # Random contiguous block scaled by a log‑normal factor
            w_min = max(1, n // 200)
            w_max = max(2, n // 12)
            width = int(rng.integers(w_min, w_max + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.20)))
            pert = cur.copy()
            pert[start:start + width] *= factor
            pert = np.clip(pert, 0.0, 1000.0)
            pert = pert / pert.sum()
            pert_score = 2.0 * n * np.convolve(pert, pert).max()
            if pert_score < cur_score:
                cur = pert
                cur_score = pert_score
                if pert_score < best_score:
                    best = pert.copy()
                    best_score = pert_score
                    print(f"[{it:>6}] pert best = {best_score:.6f}")

        # ----- 5.  If we made no progress for a while, randomise a bit -------
        if (it % 5000 == 0) and not improved:
            # Small random re‑initialisation around the current shape
            noise = rng.normal(0.0, 0.02, size=n)
            cur = cur + noise
            cur = np.clip(cur, 0.0, 1000.0)
            cur = cur / cur.sum()
            cur_score = 2.0 * n * np.convolve(cur, cur).max()
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(f"[{it:>6}] randomised best = {best_score:.6f}")

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    return [float(x) for x in best]
