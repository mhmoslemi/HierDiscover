"""
Improved search for a low‑max‑autocorrelation step function.

The algorithm combines three ideas that complement the previous
soft‑max gradient descent:

1. **Adam optimisation** – an adaptive first‑order method that
   accelerates convergence compared with a plain fixed‑step scheme.
2. **Gradual sharpening** – we start with a low soft‑max temperature
   (β≈0.5) and increase it linearly up to β≈30, after which we switch
   to the exact sub‑gradient of the true maximum (hard‑max).  This
   lets the optimiser first move quickly in a smooth landscape and
   later fine‑tune the worst convolution value.
3. **Periodic perturbations & restarts** – multiplicative log‑normal
   noise is applied every few thousand steps to escape plateaux;
   long stagnation triggers a random Dirichlet‑like restart.

All updates keep the total mass close to the normalisation
√(2 n) required by the scoring function.  The routine respects the
clip‑to‑[0, 1000] rule, works for any starting sequence stored in the
global variable ``height_sequence_1`` and returns the best sequence
found before the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real inputs, power‑of‑two length)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real‑valued FFT."""
    n = x.size + y.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max (smooth) approximation of max_k (a*a)_k
# ----------------------------------------------------------------------
def _softmax_grad(a: np.ndarray, beta: float):
    """
    Returns:
        grad   – projected gradient (zero‑sum) of the soft‑max max.
        max_b  – exact maximal value of the convolution a*a.
    """
    n = a.size
    b = _fft_convolve(a, a)               # full convolution, length 2n‑1
    max_b = b.max()

    # soft‑max weights over all lags (stabilised by subtracting max_b)
    exp_term = np.exp(beta * (b - max_b))
    w = exp_term / exp_term.sum()

    # gradient of Σ_k w_k (a*a)_k = 2 Σ_k w_k a_{k‑i}
    w_rev = w[::-1]
    conv_rev = _fft_convolve(a, w_rev)    # length 2n‑1
    idx = (2 * n - 2) - np.arange(n)      # pick the n relevant entries
    grad = 2.0 * conv_rev[idx]

    # project onto the hyper‑plane Σ a_i = const (i.e. zero‑sum)
    grad -= grad.mean()
    return grad, max_b


# ----------------------------------------------------------------------
# Exact sub‑gradient of the true maximum (hard max)
# ----------------------------------------------------------------------
def _hardmax_grad(a: np.ndarray):
    """
    Returns:
        grad   – projected sub‑gradient of max_k (a*a)_k.
        max_b  – the maximal convolution value.
    """
    n = a.size
    b = _fft_convolve(a, a)
    k_max = int(np.argmax(b))
    max_b = b[k_max]

    # ∂/∂a_i max b = 2 * a_{k_max - i}  if the index is inside the array
    grad = np.zeros_like(a)
    i = np.arange(n)
    j = k_max - i
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * a[j[mask]]

    grad -= grad.mean()
    return grad, max_b


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    provided ``evaluate_sequence`` function within the allotted time.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (default 500 s).
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise the vector
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # keep the vector reasonably sized (≤12000)
        if base.size > 12000:
            lo = (base.size - 12000) // 2
            base = base[lo:lo + 12000]
    else:
        # fallback: uniform vector of moderate length
        base = np.ones(4000, dtype=np.float64)

    n = base.size
    a = np.clip(base, 0.0, 1000.0)

    # normalise the total mass to sqrt(2 n)
    target_sum = math.sqrt(2.0 * n)
    if a.sum() == 0.0:
        a[:] = 1.0
    a *= (target_sum / a.sum())
    a = np.clip(a, 0.0, 1000.0)
    a *= (target_sum / a.sum())   # second normalisation after clipping

    # ------------------------------------------------------------------
    # 2. Book‑keeping
    # ------------------------------------------------------------------
    # initial exact score (evaluate_sequence equals max_b under our normalisation)
    _, init_maxb = _softmax_grad(a, beta=1.0)
    best_seq = a.copy()
    best_score = float(2 * n * init_maxb / (a.sum() ** 2))
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate schedule
    init_lr = target_sum * 0.02           # ≈2 % of total mass
    min_lr = init_lr * 0.02

    # temperature schedule (soft‑max β)
    beta_start = 0.5
    beta_end = 30.0

    iteration = 0
    last_improve = 0

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / budget_s

        # temperature (β) – linearly annealed
        beta = beta_start + (beta_end - beta_start) * frac

        # ----- gradient -------------------------------------------------
        if beta < 10.0:                     # smooth phase
            grad, maxb = _softmax_grad(a, beta)
        else:                               # hard‑max phase
            grad, maxb = _hardmax_grad(a)

        # Adam update (projected onto zero‑sum hyper‑plane)
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        bias_correction1 = 1.0 - beta1 ** iteration
        bias_correction2 = 1.0 - beta2 ** iteration
        m_hat = m / bias_correction1
        v_hat = v / bias_correction2

        lr = max(min_lr, init_lr * (1.0 - frac))
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)

        a += step

        # ----- enforce problem constraints -------------------------------
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)
        a = np.clip(a, 0.0, 1000.0)
        # a second normalisation fixes the tiny loss caused by clipping
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)

        # ----- evaluate true objective -----------------------------------
        conv = _fft_convolve(a, a)
        maxb_true = conv.max()
        score = float(2 * n * maxb_true / (a.sum() ** 2))

        # keep the best ever seen
        if score < best_score:
            best_seq = a.copy()
            best_score = score
            last_improve = iteration
            print(f"[{iteration:6d}] new best  score = {best_score:.6f}")

        # ----- occasional large perturbations ----------------------------
        if iteration % 2000 == 0:
            perturb = np.exp(rng.normal(0.0, 0.08, size=n))
            cand = a * perturb
            cand = np.clip(cand, 0.0, 1000.0)
            cand *= (target_sum / cand.sum())
            cand = np.clip(cand, 0.0, 1000.0)
            cand *= (target_sum / cand.sum())
            cand_score = float(2 * n * _fft_convolve(cand, cand).max() /
                               (cand.sum() ** 2))
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                a = cand
                last_improve = iteration
                print(f"[{iteration:6d}] perturb improved score = {best_score:.6f}")

        # ----- stagnation handling ---------------------------------------
        if iteration - last_improve > 5000:
            a = rng.random(n)
            a = np.clip(a, 0.0, 1000.0)
            a *= (target_sum / a.sum())
            a = np.clip(a, 0.0, 1000.0)
            a *= (target_sum / a.sum())
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"[{iteration:6d}] restart due to stagnation")

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    return [float(v) for v in best_seq]
