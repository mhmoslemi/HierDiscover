    n = a.size

    """
    Gradient-based optimizer for the convolution max/min problem.
    Starts from the arcsine sequence (or a provided one), then iteratively
    updates the step heights by moving mass from indices contributing most
    to the current maximal convolution towards the tail. The update uses the
    exact gradient of the evaluation function (with the max replaced by the
    current argmax) and a simple backtracking line search.
    """
    rng = np.random.default_rng(seed)
    # Determine length n and initial sequence
    if "height_sequence_1" in globals():
        init = np.array(height_sequence_1, dtype=np.float64)
        n = init.shape[0]

        base = build_arcsine_sequence(n)
    # Ensure count is not zero
    base = np.clip(base, 0.0, 1000.0)
    cur = base.copy()
    cur_score = 2 * n * np.max(np.convolve(cur, cur)) / (np.sum(cur) ** 2)
    best = cur.copy()
    best_score = cur_score
    print(f"[init] n={n}, score={cur_score:.6f}")
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 2.0)  # leave margin
    it = 0
    # parameters
    base_eta = 0.05   # base step as fraction of max(a)
    min_eta = 1e-5
    max_backtrack = 20
    # main loop
    while time.time() < deadline:
        it += 1
        # compute convolution via FFT
        conv = convolve_self(cur)
        k = int(np.argmax(conv))
        M = conv[k]
        S = np.sum(cur)
        # compute grad_M (derivative of max convolution at k)
        grad_M = np.zeros_like(cur)
        i0 = max(0, k - n + 1)
        i1 = min(k, n - 1)
        if i0 <= i1:

            jdx = k - idx
            grad_M[idx] = 2.0 * cur[jdx]

"""
Gradient‑based optimizer for the convolution‑max problem.

The evaluator is scale‑invariant:
 score = 2 n·max(conv(a))/sum(a)² .
We therefore minimise the ratio r = 2 n·M/S² directly.
For a current sequence a we locate k* = argmax(conv(a)).
Then
 ∂M/∂a_i = 2·a_{k*‑i}   (if the index is in range, else 0)
and
 ∂r/∂a = (2 n / S³)·(S·∂M/∂a – 2 M·1).

A simple projected gradient descent (with a back‑tracking line search)
moves mass from indices that contribute heavily to the current maximum
towards the tail, thereby increasing the total sum while keeping the
maximum convolution only mildly larger.  The iteration is monotone
decreasing in the true score, and we stop as soon as the target
(≈1.503) is reached or the time budget expires.

The routine works on the original (non‑normalised) arcsine sequence
or any supplied `height_sequence_1`.  It uses an FFT‑based convolution
for speed and clips all entries to the evaluator’s allowed range
[0, 1000].
"""

import time
import numpy as np


def _build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the classic arcsine (inverse‑square‑root) sequence."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a power‑of‑two FFT (real input)."""
    n = a.size
    # length of the full convolution = 2n‑1; round up to next power of two
    L = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv


def propose_candidate(seed: int = 42,
                     budget_s: float = 900,
                     **kwargs) -> list[float]:
    """
    Gradient‑descent search for a step‑function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed (used only for occasional perturbations).
    budget_s : float
        Time budget in seconds.
    kwargs : dict
        Ignored – present for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  Initialise – use the provided sequence if it exists, otherwise
    #     fall back to the analytical arcsine construction.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.array(height_sequence_1, dtype=np.float64)
        n = init.shape[0]
        cur = np.clip(init, 0.0, 1000.0)
    else:
        n = 256
        cur = _build_arcsine_sequence(n)
        cur = np.clip(cur, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # 2.  Helper to compute the true score (identical to evaluate_sequence
    #     but cheap because we already have the convolution).
    # ------------------------------------------------------------------
    def _score(seq: np.ndarray) -> float:
        conv = _conv_self_fft(seq)
        M = conv.max()
        S = seq.sum()
        # safety – the evaluator would return inf for S < 0.01,
        # but our iterates always keep S positive.
        return float(2.0 * n * M / (S * S))

    best = cur.copy()
    best_score = _score(best)
    cur_score = best_score

    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # tiny safety margin
    it = 0

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop (projected gradient descent with line search).
    # ------------------------------------------------------------------
    # parameters for the line search
    base_eta = 0.07          # relative size of a step (as a fraction of max(a))
    min_eta = 1e-6
    max_backtrack = 12

    while time.time() < deadline:
        it += 1

        # ---------- convolution and arg‑max ----------
        conv = _conv_self_fft(cur)
        k_star = int(np.argmax(conv))
        M = conv[k_star]
        S = cur.sum()

        # ---------- gradient of the score ----------
        # ∂M/∂a_i = 2·a_{k*‑i}  (zero when index out of range)
        grad_M = np.zeros_like(cur)
        i0 = max(0, k_star - n + 1)
        i1 = min(k_star, n - 1)
        if i0 <= i1:
            idx = np.arange(i0, i1 + 1)
            jdx = k_star - idx
            grad_M[idx] = 2.0 * cur[jdx]

        coeff = (2.0 * n) / (S ** 3)
        grad = coeff * (S * grad_M - 2.0 * M)       # gradient of the score
        direction = -grad                           # descent direction

        # ---------- step size (scaled to a fraction of the current max entry) ----------
        max_abs_dir = np.max(np.abs(direction))
        if max_abs_dir == 0.0:
            # degenerate – nothing to improve
            break
        step = base_eta * np.max(cur) / max_abs_dir
        if step < min_eta:
            step = min_eta

        # ---------- back‑tracking line search ----------
        improved = False
        for _ in range(max_backtrack):
            cand = cur + step * direction
            cand = np.clip(cand, 0.0, 1000.0)

            cand_score = _score(cand)
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best = cand.copy()
                    best_score = cand_score
                    print(f"[{it:>6}] new best = {best_score:.6f}")
                    if best_score < 1.503:
                        print("[early] target reached")
                        deadline = time.time()    # break outer loop next check
                break
            else:
                # reduce step and try again
                step *= 0.5
                if step < min_eta:
                    break

        # ---------- occasional random perturbation to escape plateaus ----------
        if not improved and it % 800 == 0:
            # Random multiplicative tweak of a small block
            block_w = max(1, n // 250)
            start = rng.integers(0, n - block_w + 1)
            factor = 1.0 + rng.normal(0.0, 0.015)
            cur[start:start + block_w] *= factor
            cur = np.clip(cur, 0.0, 1000.0)
            cur_score = _score(cur)

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Ensure the returned list obeys the evaluator's clipping rules.
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
