def simplex_projection(v, target_sum):
    # v is array, target_sum > 0.
    # Clip to nonnegative after projection
    u = np.sort(v)[::-1]   # descending
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros?
        theta = cssv[-1] / len(v)
    else:
        rho = ind[cond][-1]  # last true
        theta = cssv[rho-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def softmax_gradient_descent(a, n, target_sum, max_iters, init_eta, beta_start, beta_end):
    fft_len = _next_pow2(3*n-2)
    v = np.zeros_like(a)   # momentum
    mu = 0.9
    for it in range(max_iters):
        # compute self conv
        b = _self_convolution_fft(a, fft_len)
        cur_max = b.max()
        beta = beta_start + (beta_end - beta_start) * (it / max_iters)
        # softmax weights
        shifted = b - cur_max
        w = np.exp(beta * shifted)   # shape length 2n-1
        w_sum = w.sum()
        if w_sum == 0:
            w_norm = np.zeros_like(w)
            w_norm[np.argmax(b)] = 1.0

            w_norm = w / w_sum
        # correlation for gradient
        corr = _correlation_fft(w_norm, a, fft_len)

"""
Propose a step‑function sequence that minimizes the evaluation score.

The algorithm works in three stages for each chosen length n:

1.  Build a collection of diverse initial vectors (uniform, random, U‑shaped,
    and an interleaved version of the U‑shape).  Each vector is projected onto
    the simplex {x≥0, Σx = √(2 n)}.

2.  Refine the vector with a *soft‑max gradient descent* on the
    smoothed max‑convolution.  Momentum is used and the soft‑max temperature
    β is slowly increased.  The update is projected back onto the simplex.

3.  Perform a simulated‑annealing style *permutation annealing* that swaps
    entries of the vector while keeping the multiset of heights fixed.
    Swaps that lower the maximal self‑convolution are always accepted;
    occasional uphill moves are allowed to escape plateaus.

All heavy operations (self‑convolution, correlation) are performed with
real FFTs for O(n log n) speed.  The routine respects the supplied time
budget (default 500 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using real FFTs (returns length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)
    return conv_full[: 2 * a.size - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full correlation (w ⋆ a) = convolution(w[::-1], a) via FFT.
    Returns array of length len(w)+len(a)-1.
    """
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, fft_len)
    fa = np.fft.rfft(a, fft_len)
    corr_full = np.fft.irfft(fw * fa, fft_len)
    return corr_full[: len(w) + len(a) - 1]


def simplex_projection(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a vector onto the simplex {x ≥ 0, Σx = target_sum}.
    Implementation of the O(n) algorithm from Duchi et al. (2008).
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        # all entries will be zero
        theta = cssv[-1] / len(v)
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _interleave_ends(arr: np.ndarray) -> np.ndarray:
    """Place the largest values alternately at the left and right ends."""
    idx_desc = np.argsort(-arr)  # descending order
    out = np.empty_like(arr)
    left, right = 0, len(arr) - 1
    toggle = True
    for i in idx_desc:
        if toggle:
            out[left] = arr[i]
            left += 1
        else:
            out[right] = arr[i]
            right -= 1
        toggle = not toggle
    return out


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------


def softmax_gradient_descent(
    a: np.ndarray,
    n: int,
    target_sum: float,
    max_iters: int = 3000,
    init_eta: float = 0.02,
    beta_start: float = 0.5,
    beta_end: float = 12.0,
    momentum: float = 0.9,
    deadline: float = None,
) -> np.ndarray:
    """
    Smooth‑max gradient descent on max(a*a) with momentum.
    The step size decays linearly; β ramps from beta_start to beta_end.
    The iterate is always projected onto the simplex of sum=target_sum.
    """
    rng = np.random.default_rng()
    fft_len = _next_pow2(3 * n - 2)          # enough for correlation
    v = np.zeros_like(a)                    # momentum buffer
    for it in range(max_iters):
        if deadline is not None and time.monotonic() >= deadline:
            break

        # self‑convolution and soft‑max weighting
        b = _self_convolution_fft(a, fft_len)
        cur_max = b.max()
        beta = beta_start + (beta_end - beta_start) * (it / max_iters)

        shifted = b - cur_max                # shift for numerical stability
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.zeros_like(w)
            w_norm[np.argmax(b)] = 1.0
        else:
            w_norm = w / w_sum

        # gradient of the soft‑max (section 2 of the paper)
        corr = _correlation_fft(w_norm, a, fft_len)
        L = 2 * n - 1
        grad = 2.0 * beta * corr[L - 1 - np.arange(n)]

        # direction = -grad / ||grad||
        norm = np.linalg.norm(grad)
        if norm > 0.0:
            direction = -grad / norm
        else:
            direction = np.zeros_like(a)

        # line‑arly decaying step size
        eta = init_eta * (1.0 - it / max_iters)

        # momentum update
        v = momentum * v + (1.0 - momentum) * direction

        cand = a + eta * v
        cand = np.clip(cand, 0.0, 1000.0)
        cand = simplex_projection(cand, target_sum)

        # accept if the (approximate) max improves, otherwise keep current a
        b_cand = _self_convolution_fft(cand, fft_len)
        if b_cand.max() < cur_max:
            a = cand
        else:
            # occasional stochastic acceptance to escape stagnation
            if rng.random() < 0.001:
                a = cand
    return a


def permutation_annealing(
    a_init: np.ndarray,
    n: int,
    max_swaps: int,
    init_temp: float,
    final_temp: float,
    deadline: float,
) -> np.ndarray:
    """
    Simulated‑annealing on the ordering of the entries of a.
    Swaps that lower the maximal self‑convolution are always accepted;
    uphill moves are accepted with probability exp(-Δ/T).
    """
    rng = np.random.default_rng()
    a_cur = a_init.copy()
    fft_len = _next_pow2(2 * n - 1)
    cur_max = _self_convolution_fft(a_cur, fft_len).max()
    best_a = a_cur.copy()
    best_max = cur_max

    for s in range(max_swaps):
        if time.monotonic() >= deadline:
            break

        # linear cooling
        T = init_temp + (final_temp - init_temp) * (s / max_swaps)

        # propose a random swap
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a_cur[i], a_cur[j] = a_cur[j], a_cur[i]

        new_max = _self_convolution_fft(a_cur, fft_len).max()
        delta = new_max - cur_max

        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a_cur.copy()
                print(f"[perm] new best max = {best_max:.6f}")
        else:
            # reject: revert the swap
            a_cur[i], a_cur[j] = a_cur[j], a_cur[i]

    return best_a

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a low‑score step‑function under the given time budget.
    Returns a list of floats (the best sequence found).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin

    # ------------------------------------------------------------------
    # Determine candidate lengths
    # ------------------------------------------------------------------
    try:
        prev_seq = np.asarray(height_sequence_1, dtype=float)
    except Exception:
        prev_seq = None

    lengths = []
    if prev_seq is not None and prev_seq.size > 0:
        n0 = int(prev_seq.size)
        lengths.append(n0)
        # try a larger instance (up to the allowed 9000)
        n_big = min(9000, int(n0 * 1.5))
        if n_big != n0:
            lengths.append(n_big)
        # try a slightly smaller instance for diversity
        n_small = max(2000, int(n0 * 0.8))
        if n_small != n0:
            lengths.append(n_small)
    else:
        lengths = [6000, 8000]

    lengths = sorted(set(lengths))

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Iterate over each chosen length
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        target_sum = np.sqrt(2.0 * n)
        print(f"\n=== Optimising length n={n} (target sum {target_sum:.3f}) ===")

        # ---- Build starting points -------------------------------------------------
        starts = []

        # uniform
        starts.append(np.full(n, target_sum / n))

        # random dense
        starts.append(rng.random(n) * target_sum)

        # a few U‑shaped profiles
        for p in rng.uniform(-4.0, -0.2, size=4):
            idx = np.arange(n, dtype=float)
            a = (idx + 1) ** p + (n - idx) ** p
            a = a - a.min() + 1e-8
            a = a / a.sum() * target_sum
            starts.append(a)

        # interleaved version of a U‑shape (to explore different orderings)
        for p in rng.uniform(-3.5, -0.5, size=2):
            idx = np.arange(n, dtype=float)
            a = (idx + 1) ** p + (n - idx) ** p
            a = a - a.min() + 1e-8
            a = a / a.sum() * target_sum
            starts.append(_interleave_ends(a))

        # interpolate previous best if available
        if prev_seq is not None and prev_seq.size > 0:
            old_n = prev_seq.size
            old_x = np.linspace(0.0, 1.0, old_n)
            new_x = np.linspace(0.0, 1.0, n)
            a_interp = np.interp(new_x, old_x, prev_seq)
            a_interp = np.clip(a_interp, 0.0, 1000.0)
            if a_interp.sum() > 0:
                a_interp = a_interp * (target_sum / a_interp.sum())
                starts.append(a_interp)

        # ------------------------------------------------------------------
        # Process each start vector
        # ------------------------------------------------------------------
        for sid, a0 in enumerate(starts):
            if time.monotonic() >= deadline:
                break

            a = np.clip(a0, 0.0, 1000.0)
            a = simplex_projection(a, target_sum)

            # exact evaluation of the initial point
            init_score = evaluate_sequence(a.tolist())
            if init_score < best_score:
                best_score = init_score
                best_seq = a.copy()
                print(f"[n={n} start={sid}] initial exact score = {best_score:.6f}")

            # -------------------- Soft‑max gradient descent --------------------
            a_opt = softmax_gradient_descent(
                a,
                n,
                target_sum,
                max_iters=2500,
                init_eta=0.018 * target_sum,
                beta_start=0.5,
                beta_end=12.0,
                momentum=0.9,
                deadline=deadline,
            )
            score_opt = evaluate_sequence(a_opt.tolist())
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"[n={n} start={sid}] after gradient score = {best_score:.6f}")

            # -------------------- Permutation annealing -----------------------
            # allocate swaps based on remaining time (rough estimate)
            time_left = deadline - time.monotonic()
            if time_left <= 0:
                break
            # assume ~0.0004 s per swap (conservative)
            max_swaps = int(min(20000, max(500, time_left / 0.0005)))
            a_perm = permutation_annealing(
                a_opt,
                n,
                max_swaps=max_swaps,
                init_temp=0.02 * best_score,
                final_temp=1e-6,
                deadline=deadline,
            )
            perm_score = evaluate_sequence(a_perm.tolist())
            if perm_score < best_score:
                best_score = perm_score
                best_seq = a_perm.copy()
                print(f"[n={n} start={sid}] after perm score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a safe uniform vector
        n_fallback = 6000
        target_sum = np.sqrt(2.0 * n_fallback)
        best_seq = np.full(n_fallback, target_sum / n_fallback)

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best exact score = {final_score:.6f}")
    return [float(v) for v in best_seq]
