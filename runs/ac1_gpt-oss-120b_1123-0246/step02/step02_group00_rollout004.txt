def proj_simplex(v):
    # based on Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _proj_simplex(v):
    # grabs v (array), returns projection onto simplex {x >=0, sum=1}
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not cond.any():
        # fallback: uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _make_block_perturbation(seq,rng):
    n = seq.size
    # choose block width between lo and hi; choose lo = max(1, n//200)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi+1))
    start = int(rng.integers(0, n - width + 1))
    factor = np.exp(rng.normal(0.0, 0.12))
    cand = seq.copy()
    cand[start:start+width] *= factor
    # optionally symmetrise
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    cand = np.clip(cand, 0.0, 1000.0)
    # projection onto simplex after we call later
    return cand

"""
Propose a step‐height sequence that minimises
    2 * n * max(a*a) / (sum a)^2

The algorithm combines three ideas:

1. **Power‑law seeding** – a fine grid over
   a_i ∝ (i+β)^(-α) is scanned; the best seed becomes the start point.

2. **Projected Adam optimisation** – the sub‑gradient of the current
   maximal convolution entry is computed (via a cheap FFT convolution)
   and an Adam‐style momentum update is performed.
   The update is projected onto the probability simplex (non‑negative,
   sum = 1) after each step.

3. **Stochastic block perturbations** – with a modest probability a
   random contiguous block is multiplied by log‑normal noise; the
   resulting vector is again projected onto the simplex.  This helps
   escape shallow local minima.

All heavy work (convolutions) uses an FFT‑based O(n log n) routine,
so thousands of iterations fit comfortably into the time budget.
The best sequence encountered is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – no closures, no lambdas
# ----------------------------------------------------------------------
def _fft_convolve(seq: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself using FFT."""
    n = seq.size
    # next power of two that accommodates 2*n-1 points
    m = 1 << ((2 * n - 1).bit_length())
    # real FFT is enough because the data are real
    fft_a = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)
    return conv[: 2 * n - 1]  # keep the true linear convolution part


def _evaluate_fast(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` for non‑negative vectors."""
    total = seq.sum()
    if total <= 0.0:
        return float("inf")
    conv = _fft_convolve(seq)
    max_b = conv.max()
    n = seq.size
    return float(2.0 * n * max_b / (total * total))


def _shift_one(seq: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of (seq*seq)[k] w.r.t. ``seq``.
    Returns a vector d such that d[i] = seq[k-i] if the index exists,
    otherwise 0.
    """
    n = seq.size
    out = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1)
        out[idx] = seq[k - idx]
    return out


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    Implements the algorithm of Duchi et al. (2008) – O(n log n).
    """
    n = v.size
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    Occasionally the candidate is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:start + width] *= factor
    if rng.random() < 0.3:               # mild symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

# Main entry point expected by the external harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative step heights that (hopefully) yields a low
    value of ``evaluate_sequence``.  The routine runs for at most the
    supplied ``budget_s`` seconds (a small safety margin is kept).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Determine the length of the vector.
    # ------------------------------------------------------------------
    try:

        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = int(base.size)
    except Exception:
        n = 2500                      # a safe default

    # ------------------------------------------------------------------
    # 2) Coarse power‑law grid search for a good seed.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    idx = np.arange(1, n + 1, dtype=np.float64)    # 1‑based indices

    alphas = np.linspace(0.44, 0.60, 33)           # slightly wider search
    betas = np.linspace(0.0, 2.0, 11)

    for alpha in alphas:
        for beta in betas:
            seq = 1.0 / ((idx + beta) ** alpha)
            seq = np.clip(seq, 0.0, 1000.0)
            s = seq.sum()
            if s == 0.0:
                continue
            seq /= s
            sc = _evaluate_fast(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()

    # If the global reference sequence is available, try it as well.
    if 'height_sequence_1' in globals():

                best_score = sc

    # Fallback – uniform vector (should never be used)
    if best_seq is None:

    # ------------------------------------------------------------------
    # 3) Adam‑style projected optimisation.
    # ------------------------------------------------------------------
    a = best_seq.copy()               # current candidate (always on the simplex)
    m = np.zeros_like(a)              # 1st moment
    v = np.zeros_like(a)              # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0

    # learning‑rate and temperature schedules (exponential decay)
    lr_start = 0.25
    lr_end = 0.004
    temp_start = 3e-2
    temp_end = 2e-6
    block_prob = 0.18                 # probability of a random block move

    curr_score = _evaluate_fast(a)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # --------------------------------------------------------------
        # a) compute self‑convolution via FFT and locate maximal entry
        # --------------------------------------------------------------
        conv = _fft_convolve(a)
        max_b = conv.max()
        # treat all entries that are numerically maximal
        tol = max_b * 1e-14 + 1e-16
        max_inds = np.where(conv >= max_b - tol)[0]

        # --------------------------------------------------------------
        # b) sub‑gradient of the maximal convolution entry
        # --------------------------------------------------------------
        shift_sum = np.zeros_like(a)
        for k in max_inds:
            shift_sum += _shift_one(a, int(k))
        shift_avg = shift_sum / max(len(max_inds), 1)

        grad = -shift_avg                     # descent direction
        grad -= grad.mean()                   # keep the sum of a unchanged

        # --------------------------------------------------------------
        # c) Adam update (bias‑corrected)
        # --------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # schedule (linear in log‑space)
        prog = (time.monotonic() - start_time) / (deadline - start_time)
        lr = lr_start * (lr_end / lr_start) ** prog
        temp = temp_start * (temp_end / temp_start) ** prog

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = a + step

        # --------------------------------------------------------------
        # d) occasional stochastic block perturbation
        # --------------------------------------------------------------
        if rng.random() < block_prob:
            cand = _make_block_perturbation(cand, rng)

        # --------------------------------------------------------------
        # e) projection onto the probability simplex
        # --------------------------------------------------------------
        cand = _proj_simplex(cand)

        cand_score = _evaluate_fast(cand)

        # --------------------------------------------------------------
        # f) Metropolis‑style acceptance (allows occasional uphill moves)
        # --------------------------------------------------------------
        accept = False
        if cand_score < curr_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - curr_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            curr_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"iter {it:06d} – new best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final sanitisation – the external evaluator will clip again,
    #    but we keep the vector well‑behaved.
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total > 0.0:
        best_seq /= total

    print(f"finished after {it} iterations, best score = {best_score:.6f}")
    return best_seq.tolist()
