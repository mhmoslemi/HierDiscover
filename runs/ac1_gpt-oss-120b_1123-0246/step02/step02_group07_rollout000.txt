def directed_transfer_move(base, rng):
    n = base.size
    # compute convolution via FFT
    fft = np.fft.rfft(base, n=2*n-1)
    conv = np.fft.irfft(fft*fft, n=2*n-1)
    # we only need central part (0..n-1)
    # find max index
    max_idx = np.argmax(conv[:n])
    idx = np.arange(max_idx+1)
    contrib = base[idx] * base[max_idx-idx]
    src = idx[np.argmax(contrib)]
    # target: low-valued indices
    # choose among bottom quantile
    low_thr = np.percentile(base, 20)
    low_idxs = np.where(base <= low_thr)[0]
    if low_idxs.size == 0:
        tgt = rng.integers(0, n)
        while tgt == src:
            tgt = rng.integers(0, n)
    else:
        tgt = int(rng.choice(low_idxs))
        if tgt == src:
            # pick second or random
            tgt = int(rng.integers(0, n))
    if tgt == src:
        return base
    frac = rng.random() * 0.2 + 0.01   # 0.01-0.21
    delta = frac * base[src]
    cand = base.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)
```

def random_redistribute_move(base, rng):
    n=base.size
    k = int(rng.integers(2, min(20, n)))  # select k indices
    idx = rng.choice(n, size=k, replace=False)
    total = base[idx].sum()
    # generate random dirichlet distribution
    new_vals = rng.random(k)
    new_vals = new_vals / new_vals.sum() * total
    cand = base.copy()
    cand[idx] = new_vals
    return np.clip(cand,0.0,1000.0)

"""
Convolution‑bound optimiser – fast local search.

The optimiser builds a variety of analytical and simple parametric
starting points (arcsine, tilted mixtures, power‑law, exponential decay,
moving‑average smoothing, …).  All candidates are evaluated with a fast
FFT‑based convolution routine (identical to ``evaluate_sequence`` but
much cheaper).  The best candidate is then refined by a stochastic
local search that repeatedly moves a small amount of mass from the
indices that contribute most to the current maximal convolution value
to indices that have very small values.  Additional moves (block
multiplication, smoothing, random redistribution) diversify the
exploration.  An annealing schedule allows occasional uphill moves.
The search respects the 1000 s time budget, prints occasional progress
messages and finally returns the best sequence found (as a list of
floats) – the harness will evaluate it with the official
``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Fast evaluation (FFT based, identical to evaluate_sequence)
# ----------------------------------------------------------------------
def fast_evaluate(seq: np.ndarray) -> float:
    """Return the objective value using an FFT based convolution.
    The sequence is first clipped to the allowed range [0,1000].
    """
    a = np.clip(seq, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return np.inf
    # linear convolution via padded FFT
    length = 2 * n - 1
    fft = np.fft.rfft(a, n=length)
    conv = np.fft.irfft(fft * fft, n=length)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01 or np.isnan(max_b) or np.isinf(max_b):
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# 2.  Sequence generators
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the discrete arcsine (inverse‑square‑root) sequence."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a  # already in [0,1]


# ----------------------------------------------------------------------
# 3.  Perturbation moves
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def directed_transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small fraction of mass from the index most responsible for
    the current maximal convolution coefficient to a low‑value index."""
    n = base.size
    # convolution via FFT (fast)
    length = 2 * n - 1
    fft = np.fft.rfft(base, n=length)
    conv = np.fft.irfft(fft * fft, n=length)
    # focus on the first n entries (the interior part)
    max_idx = int(np.argmax(conv[:n]))

    # contributions for this index
    idx = np.arange(max_idx + 1)
    contrib = base[idx] * base[max_idx - idx]
    src = int(idx[np.argmax(contrib)])

    # pick a target among the lowest 20 % values
    low_thr = np.percentile(base, 20)
    low_cands = np.where(base <= low_thr)[0]
    if low_cands.size == 0:
        tgt = int(rng.integers(0, n))
        while tgt == src:
            tgt = int(rng.integers(0, n))
    else:
        tgt = int(rng.choice(low_cands))
        if tgt == src:
            # fallback to a random different index
            tgt = int(rng.integers(0, n))
            while tgt == src:
                tgt = int(rng.integers(0, n))

    # amount to move – a small, slowly decreasing fraction
    frac = rng.random() * 0.20 + 0.01      # 1 % … 21 %
    delta = frac * base[src]
    if delta <= 0.0:
        return base.copy()

    cand = base.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)


def smoothing_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a short moving‑average filter and renormalise to keep the sum."""
    n = base.size
    # window length – random but at least 3
    w = int(rng.integers(3, max(4, n // 50 + 1)))
    kernel = np.ones(w, dtype=np.float64) / w
    smoothed = np.convolve(base, kernel, mode='same')
    total = base.sum()
    if smoothed.sum() == 0:
        return base.copy()
    cand = smoothed * (total / smoothed.sum())
    return np.clip(cand, 0.0, 1000.0)


def random_redistribute_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Select a small random subset of indices and reshuffle their mass."""
    n = base.size
    k = int(rng.integers(2, min(30, n)))
    idx = rng.choice(n, size=k, replace=False)
    tot = base[idx].sum()
    # generate a random Dirichlet‑like distribution
    raw = rng.random(k)
    if raw.sum() == 0:
        new_vals = np.zeros(k)
    else:
        new_vals = raw / raw.sum() * tot
    cand = base.copy()
    cand[idx] = new_vals
    return np.clip(cand, 0.0, 1000.0)

        # occasional true verification and early stopping
        if it % 3000 == 0:
            true_sc = float(evaluate_sequence(best_seq.tolist()))
            if true_sc < best_score:
                best_score = true_sc
                print(f"[{it}] verified true score = {best_score:.6f}")
            if best_score < 1.503:
                print("[stop] target threshold reached")
                break

    # ------------------------------------------------------------------
    # Final verification using the official evaluator.
    # ------------------------------------------------------------------
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] true score {final_score:.6f} after {it} iterations")
    return [float(x) for x in best_seq]
