def correlation_valid(w, a):
    # compute g_i = sum_t w[t + i] * a[t] for i=0..n-1
    # pad length for FFT: length = next_pow2(len(w) + len(a) - 1)
    L = len(w)
    n = len(a)
    size = 1 << ((L + n - 2).bit_length())
    # FFT of w
    fw = np.fft.rfft(w, size)
    # FFT of reversed a
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa_rev, size)  # length size
    # the full convolution result length = L + n - 1 = 
    conv = conv[:L + n - 1]  # slice needed length
    offset = n - 1
    # take slice offset:offset+n
    return conv[offset: offset + n]
```

"""
Gradient‑descent optimizer for the convolution‑max problem.

The objective 2·n·max(conv) / (sum(a)²) is scale‑invariant, so we fix
∑a_i = 1 and minimise max(conv) where conv = a * a (linear convolution).
A soft‑max (temperature β) approximates the max, providing a smooth gradient:

    f_β(a) = (1/β)·log Σ_k exp(β·conv_k)

∂f_β/∂a = 2·correlation(softmax_w, a)   with   w_k = exp(β·conv_k).

Each iteration:
  * compute conv,
  * form the normalized soft‑max weights w,
  * obtain the gradient via an FFT‑based correlation,
  * take a projected‑gradient step onto the probability simplex,
  * gradually raise β (sharpening the max) and shrink the step size,
  * occasionally apply a random block perturbation to avoid stagnation.

The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Generate the analytic arcsine (inverse‑square‑root) start sequence."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def _next_pow2(x: int) -> int:
    """Small utility – next power of two (at least 1)."""
    return 1 << (x - 1).bit_length()


def conv_full(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def correlation_valid(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_t w[t + i]·a[t] for i = 0 … n‑1 (valid‐mode correlation).
    Implemented via FFT: correlate(w, a) = (w * reversed(a)) slice.
    """
    n = a.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    size = _next_pow2(L + n - 1)       # ≥ 3·n‑2
    fw = np.fft.rfft(w, size)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa_rev, size)[: L + n - 1]
    offset = n - 1
    return conv[offset : offset + n]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    { x | x_i ≥ 0, Σ_i x_i = 1 }.
    """
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array.")
    n = v.shape[0]

    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0

    if not np.any(cond):
        # All entries are ≤ 0 – return the uniform point.
        return np.full_like(v, 1.0 / n)

    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Randomly scale a contiguous block (log‑normal factor) and optionally
    symmetrise.  The result is re‑normalised to a unit sum.
    """
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    cand = np.clip(cand, 0.0, 1000.0)
    # re‑normalise to keep the simplex constraint
    total = cand.sum()
    if total > 0:
        cand /= total
    return cand

    """
    Optimise a non‑negative step‑function sequence under the given evaluation
    function.  Returns the best sequence (list of floats) found before the
    time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the problem size and a sensible initial point.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base_seq = np.array(height_sequence_1, dtype=np.float64)
        n = base_seq.shape[0]
    else:
        n = 256
        base_seq = build_arcsine_sequence(n)

    # Normalise to unit sum (the objective is scale‑invariant).
    a = np.clip(base_seq, 0.0, 1000.0)
    total = a.sum()
    if total < 0.01:                     # safety fallback
        a = np.ones(n) / n
    else:
        a = a / total

    best = a.copy()
    best_score = 2.0 * n * np.max(conv_full(best))
    print(f"[init] n={n}, start score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # small safety margin

    iteration = 0
    # Soft‑max temperature schedule (β): from low (smooth) to high (sharp).
    beta_min, beta_max = 0.5, 30.0
    # Desired step magnitude (relative to the current Euclidean norm of a).
    step_rel_start, step_rel_end = 0.6, 0.05   # L2 step = factor * ||a||

    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start_time
        frac = elapsed / float(budget_s)   # 0 … 1

        # --------------------------------------------------------------
        # 1.  Current convolution and score.
        # --------------------------------------------------------------
        conv = conv_full(a)
        max_conv = np.max(conv)
        cur_score = 2.0 * n * max_conv

        # --------------------------------------------------------------
        # 2.  Soft‑max (smooth max) weights.
        # --------------------------------------------------------------
        beta = beta_min + (beta_max - beta_min) * frac
        # shift for numerical stability (max conv becomes zero)
        conv_shift = conv - np.max(conv)
        w = np.exp(beta * conv_shift)
        w_sum = np.sum(w)
        if w_sum == 0.0:              # should not happen, but guard against it
            w = np.ones_like(w) / w.size
            w_sum = 1.0
        w_norm = w / w_sum

        # --------------------------------------------------------------
        # 3.  Gradient of the smooth objective.
        # --------------------------------------------------------------
        g = 2.0 * correlation_valid(w_norm, a)   # size n
        g_norm = np.linalg.norm(g)
        if g_norm == 0.0:
            # Stagnated – inject a random perturbation and continue.
            a = block_perturbation(a, rng)
            a = project_simplex(a)
            continue

        # --------------------------------------------------------------
        # 4.  Projected gradient step.
        # --------------------------------------------------------------
        a_norm = np.linalg.norm(a)
        step_factor = step_rel_start * (1.0 - frac) + step_rel_end * frac
        step_len = step_factor * a_norm                 # desired L2 move length
        cand = a - (step_len / g_norm) * g
        cand = project_simplex(cand)

        # --------------------------------------------------------------
        # 5.  Evaluate candidate (fast, using the same convolution).
        # --------------------------------------------------------------
        conv_cand = conv_full(cand)
        cand_score = 2.0 * n * np.max(conv_cand)

        # --------------------------------------------------------------
        # 6.  Acceptance.
        # --------------------------------------------------------------
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[{iteration:>6}] new best = {best_score:.6f}")
                if best_score < 1.503:
                    print("[early] target reached")
                    break
        else:
            # Tiny probability to accept a non‑improving move (helps escape flat regions)
            if rng.random() < 0.001:
                a = cand

        # --------------------------------------------------------------
        # 7.  Occasional random block perturbation (exploration).
        # --------------------------------------------------------------
        if iteration % 400 == 0 and rng.random() < 0.3:
            a = block_perturbation(a, rng)
            a = project_simplex(a)

    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    # Return as a plain Python list (the harness will apply its final clipping/evaluation).
    return [float(x) for x in best]
