"""
Hybrid simulated‑annealing + gradient descent for the
max‑convolution minimisation problem.

The algorithm:
* Initialise with the analytically optimal arcsine distribution,
  scaled to the required ℓ₁‑norm √(2 n).
* Repeatedly propose stochastic modifications:
    – block scaling,
    – elementary mass transfers,
    – early‑to‑late transfers,
    – smoothing of a single coordinate,
    – swapping two entries,
    – a gradient‑inspired move that transfers mass from the index
      contributing most to the current maximal convolution entries
      toward the index contributing least.
* Accept the new candidate with a Metropolis rule (temperature
  decreasing exponentially with the elapsed time).  This allows
  occasional uphill steps and helps to escape local minima.
* Keep the best sequence seen (according to the exact objective
  2·n·max(b)/(∑a)²).  The loop runs until the time budget expires
  or the target score 1.503 is reached.

All convolution work uses FFT‑based convolution for speed
(O(N log N) instead of the naïve O(N²)).  The routine never
produces values outside the allowed range [0, 1000] and respects
the required normalisation.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def renormalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to the evaluation limits and rescale to a given ℓ₁‑norm."""
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur == 0.0:
        return seq
    seq = seq * (target_sum / cur)
    return seq


def sequence_score(seq: np.ndarray) -> float:
    """Fast evaluation of   2·n·max(b)/(∑a)²   using FFT convolution."""
    n = seq.shape[0]
    s = float(seq.sum())
    if s < 0.01:
        return np.inf
    # length for a power‑of‑two FFT (at least 2 n‑1)
    size = 1 << ((2 * n - 1).bit_length())
    A = rfft(seq, size)
    conv = irfft(A * A, size)[: 2 * n - 1]
    max_b = float(conv.max())
    return 2.0 * n * max_b / (s * s)


def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the discretised arcsine (inverse‑square‑root) sequence."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    return a


# ----------------------------------------------------------------------
# Elementary moves (all keep the ℓ₁‑norm via renormalise)
# ----------------------------------------------------------------------


def block_perturbation(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Scale a random contiguous block by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrisation (helps for the symmetric optimum)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return renormalise(cand, target_sum)


def transfer_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """Transfer a random fraction (≤30 %) of mass from i to j."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


def early_to_late_move(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Move a fraction of mass from the first 10 % of entries to the last 10 %."""
    n = base.size
    early_cut = max(1, n // 10)
    late_start = max(early_cut, n - early_cut)
    i = int(rng.integers(0, early_cut))
    j = int(rng.integers(late_start, n))
    if i == j:
        return base.copy()
    frac = rng.random() * 0.40
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


def smooth_move(base: np.ndarray, rng: np.random.Generator,
                target_sum: float) -> np.ndarray:
    """Replace a randomly chosen coordinate by the average of its neighbours."""
    n = base.size
    i = int(rng.integers(0, n))
    cand = base.copy()
    if 0 < i < n - 1:
        cand[i] = 0.5 * (cand[i - 1] + cand[i + 1])
    elif i == 0:
        cand[i] = cand[i + 1]
    else:  # i == n‑1
        cand[i] = cand[i - 1]
    return renormalise(cand, target_sum)


def swap_move(base: np.ndarray, rng: np.random.Generator,
              target_sum: float) -> np.ndarray:
    """Swap two distinct entries (sum unchanged)."""
    n = base.size
    i, j = rng.choice(n, size=2, replace=False)
    cand = base.copy()
    cand[i], cand[j] = cand[j], cand[i]
    # no scaling needed, but we clip for safety
    return renormalise(cand, target_sum)


def gradient_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """
    Sub‑gradient move: transfer mass from the index that contributes most
    to the currently maximal convolution entries toward the index that
    contributes the least.
    """
    n = base.size
    # FFT‑based convolution of base with itself
    size = 1 << ((2 * n - 1).bit_length())
    A = rfft(base, size)
    conv = irfft(A * A, size)[: 2 * n - 1]
    max_b = conv.max()

    # mask of (near) maximal positions (tiny tolerance helps with floating noise)
    tol = max_b * (1.0 - 1e-12)
    mask = (conv >= tol).astype(np.float64)  # length = 2n‑1

    # gradient of the (sum of) maximal entries w.r.t. a:
    # G = 2 * (mask ⋆ a), where ⋆ denotes correlation → np.correlate
    G = 2.0 * np.correlate(mask, base, mode="valid")

    heavy_idx = int(np.argmax(G))
    light_idx = int(np.argmin(G))
    if heavy_idx == light_idx:
        # pick a different light index at random
        light_idx = (heavy_idx + rng.integers(1, n)) % n

    # move a modest fraction of the heavy mass
    frac = rng.random() * 0.30
    delta = frac * base[heavy_idx]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[heavy_idx] -= delta
    cand[light_idx] += delta
    return renormalise(cand, target_sum)

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of floats – the best step‑function heights found within
    ``budget_s`` seconds.

    The search combines simulated annealing with several stochastic
    moves (including a gradient‑inspired transfer).  The best candidate
    seen so far (according to the exact objective) is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the problem size.
    # ------------------------------------------------------------------
    height_seq = globals().get("height_sequence_1", [])
    n = int(len(height_seq))
    if n == 0:
        n = 256  # fallback if the global is missing

    # ------------------------------------------------------------------
    # 1.  Starting point – the analytically optimal arcsine shape,
    #     normalised to the required ℓ₁‑norm √(2 n).
    # ------------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    base_seq = renormalise(build_arcsine_sequence(n), target_sum)
    cur_seq = base_seq.copy()
    cur_score = sequence_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Simulated‑annealing loop
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)  # small safety margin
    iteration = 0

    # temperature schedule: exponential cooling from T0 to Tf
    T0 = 1.0
    Tf = 1e-5

    while time.monotonic() < deadline:
        iteration += 1
        # pick a move type (probabilities tuned heuristically)
        r = rng.random()
        if r < 0.18:
            cand = block_perturbation(cur_seq, rng, target_sum)
        elif r < 0.34:
            cand = transfer_move(cur_seq, rng, target_sum)
        elif r < 0.50:
            cand = early_to_late_move(cur_seq, rng, target_sum)
        elif r < 0.66:
            cand = smooth_move(cur_seq, rng, target_sum)
        elif r < 0.82:
            cand = swap_move(cur_seq, rng, target_sum)
        else:
            cand = gradient_move(cur_seq, rng, target_sum)

        cand_score = sequence_score(cand)

        # Metropolis acceptance
        delta = cand_score - cur_score
        elapsed = (time.monotonic() - start) / (budget_s - 0.5)
        T = T0 * (Tf / T0) ** elapsed   # exponential cooling
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[{iteration:>6}] new best = {best_score:.6f}")

                if best_score < 1.503:   # early exit if the target is reached
                    break

    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    # Return as plain Python list – the harness expects a list of floats.
    return [float(x) for x in best_seq]
