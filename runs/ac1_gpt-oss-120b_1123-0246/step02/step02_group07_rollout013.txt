def build_arcsine_sequence(length: int) -> np.ndarray:
    """Generate the discrete arcsine sequence a_i = binom(2i,i)/4**i
    for i=0..length-1. Values lie in [0,1] and satisfy (a*a)[k]=1
    for 0<=k<length."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a

def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor,
    optionally symmetrising the sequence."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

def directed_transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small amount of mass from the index that contributes most
    to the current maximal convolution entry to a low‑impact index."""
    n = base.size

    src = int(np.argmax(contrib))

            tgt += 1

            tgt = 0

"""
Optimiser for the convolution‑bound problem.

The procedure is:
    1. Build the analytically optimal “arcsine’’ sequence.
    2. Solve a single linear programme
           max Σ b_j   s.t.   (a * b)_k ≤ M   (for all k),
       where a is the arcsine sequence and M = max(a*a).  This gives a
       feasible point b with a larger ℓ₁‑norm while respecting the same
       convolution bound.
    3. Mix the LP solution with the arcsine baseline (convex combination)
       and keep the best mixture.
    4. Refine the result with a fast stochastic local search that moves a
       small amount of mass from the most harmful index (the one that
       contributes most to the current maximal convolution coefficient)
       towards low‑impact indices.

        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a
