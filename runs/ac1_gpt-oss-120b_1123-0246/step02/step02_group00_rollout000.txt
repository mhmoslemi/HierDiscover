"""
Simple annealing optimiser for the convolution‐minimisation task.

The algorithm consists of three stages:

1) **Diverse seeding** – generate a collection of analytic shapes
   (power‑law, exponential, Gaussian, triangular, Hamming, uniform)
   for the chosen length and pick the best one according to a fast
   replica of ``evaluate_sequence``.

2) **Hybrid stochastic search** – repeatedly propose a new candidate
   by one of four moves:
        * targeted projected gradient step (using the sub‑gradient of
          the current maximal convolution entries);
        * multiplicative log‑normal block perturbation;
        * small “swap” mass transfer between two random indices;
        * gentle smoothing via a 3‑point moving‑average filter.
   The move probabilities are annealed.  After each move the vector is
   clipped to the allowed box, projected back onto the probability
   simplex and optionally symmetrised.

3) **Metropolis acceptance** – a temperature schedule (exponential
   cooling) controls uphill moves.  The best sequence found so far is
   retained and returned when the budget expires.

All helper functions are top‑level; no closures or lambda expressions
are used.  The code respects the 1000 s wall‑time limit by stopping
shortly before the supplied budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _evaluate_fast(seq: np.ndarray) -> float:
    """
    Fast surrogate for ``evaluate_sequence``.
    Assumes ``seq`` is non‑negative and already normalised to sum = 1.
    Returns 2 * n * max(conv(seq, seq)).
    """
    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    conv = np.convolve(seq, seq)
    return float(2.0 * seq.size * conv.max() / (total * total))


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip to the allowed box, enforce non‑negativity and renormalise."""
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s == 0.0:
        # avoid division by zero – return a uniform vector
        seq = np.full_like(seq, 1.0 / seq.size)
    else:
        seq = seq / s
    return seq


def _shift(seq: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the k‑th convolution entry w.r.t. ``seq``:
    (seq shifted and mirrored at distance k).
    """
    n = seq.size
    out = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1, dtype=int)
        out[idx] = seq[k - idx]
    return out


def _gradient_step(seq: np.ndarray,
                   eta: float,
                   top_k: int,
                   momentum: np.ndarray,
                   beta: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, np.ndarray]:
    """
    One projected gradient step using the average sub‑gradient of the
    ``top_k`` largest convolution entries.
    Returns (new_sequence, new_momentum).
    """
    conv = np.convolve(seq, seq)
    # indices of the largest entries
    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(-conv, top_k)[:top_k]

    # accumulate the shift vectors
    shift_sum = np.zeros_like(seq)
    for k in ks:
        shift_sum += _shift(seq, int(k))
    grad = shift_sum / len(ks)          # average sub‑gradient

    # centre the direction so that sum stays unchanged
    direction = -grad + np.mean(grad)

    # momentum update (exponential moving average)
    momentum = beta * momentum + (1.0 - beta) * direction

    cand = seq + eta * momentum
    cand = _normalize(cand)
    return cand, momentum


def _block_perturbation(seq: np.ndarray,
                        rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    Occasionally the candidate is symmetrised.
    """
    n = seq.size
    # choose a block width between 0.5% and 8% of the length
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand = seq.copy()
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.14)))
    # mild symmetrisation with some probability
    if rng.random() < 0.32:
        cand = 0.5 * (cand + cand[::-1])
    cand = _normalize(cand)
    return cand


def _swap_move(seq: np.ndarray,
               rng: np.random.Generator,
               delta_frac: float = 0.12) -> np.ndarray:
    """
    Transfer a small fraction of mass from a random source index to a
    random target index.
    """
    n = seq.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    if i == j or seq[i] == 0.0:
        return seq.copy()
    # amount moved is a random fraction of the source mass,
    # bounded by ``delta_frac``.
    delta = delta_frac * rng.random() * seq[i]
    cand = seq.copy()
    cand[i] -= delta
    cand[j] += delta
    cand = _normalize(cand)
    return cand


def _smooth_move(seq: np.ndarray) -> np.ndarray:
    """
    Apply a short moving‑average filter (kernel [0.25,0.5,0.25]) and
    renormalise.  This reduces high‑frequency noise while preserving the
    overall mass.
    """
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    cand = np.convolve(seq, kernel, mode='same')
    cand = _normalize(cand)
    return cand


def _seed_collection(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a diverse set of analytic seed vectors of length ``n``.
    Each seed is normalised to sum = 1.
    """
    idx = np.arange(n, dtype=np.float64)

    seeds = []

    # 1) Power‑law family
    for alpha in np.linspace(0.42, 0.58, 9):
        for beta in (0.0, 0.5, 1.0, 2.0):
            seq = 1.0 / ((idx + 1.0 + beta) ** alpha)
            seeds.append(_normalize(seq))

    # 2) Exponential decay
    for lam in (0.0003, 0.0005, 0.001, 0.002, 0.003):
        seq = np.exp(-lam * idx)
        seeds.append(_normalize(seq))

    # 3) Gaussian centred in the middle
    center = (n - 1) / 2.0
    for sigma_frac in (0.12, 0.17, 0.22, 0.27):
        sigma = sigma_frac * n
        seq = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
        seeds.append(_normalize(seq))

    # 4) Triangular (peak in the centre)
    t = np.linspace(0.0, 1.0, n)
    seq = 1.0 - np.abs(2.0 * t - 1.0)
    seeds.append(_normalize(seq))

    # 5) Hamming window
    seq = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
    seeds.append(_normalize(seq))

    # 6) Uniform
    seeds.append(_normalize(np.ones_like(idx)))

    # 7) Randomised log‑normal (used only as a fallback)
    seq = np.exp(rng.normal(0.0, 0.8, size=n))
    seeds.append(_normalize(seq))

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs):
    """
    Return a list of non‑negative step heights that (hopefully) minimises
    the provided ``evaluate_sequence`` function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget in seconds.  The function stops a few
        seconds before the limit to avoid termination penalties.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the length of the step function.
    # Prefer the length of a previously supplied solution if present.
    # --------------------------------------------------------------
    global height_sequence_1
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim == 1 and base.size >= 500 and np.isfinite(base).all():
            n = int(base.size)
        else:
            raise ValueError
    except Exception:
        n = 2500                     # default length used in earlier runs

    # --------------------------------------------------------------
    # Initialise with a collection of analytic seeds and pick the best.
    # --------------------------------------------------------------
    seeds = _seed_collection(n, rng)
    best_seq = None
    best_score = float("inf")
    for seq in seeds:
        sc = _evaluate_fast(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    # sanity check
    if best_seq is None:
        raise RuntimeError("Failed to initialise a candidate sequence.")

    # --------------------------------------------------------------
    # Optimisation loop (hybrid simulated annealing)
    # --------------------------------------------------------------
    current_seq = best_seq.copy()
    current_score = best_score
    momentum = np.zeros_like(current_seq)   # for momentum in gradient steps
    beta_mom = 0.85                         # exponential smoothing factor

    # time management
    safety_margin = 1.5                     # seconds reserved for cleanup
    deadline = time.monotonic() + max(0.0, float(budget_s) - safety_margin)

    # annealing schedules (exponential)
    eta_start = 0.20
    eta_end   = 1e-4
    temp_start = 3e-2
    temp_end   = 2e-7

    # move probabilities (will be slowly shifted towards gradient as time passes)
    prob_grad   = 0.45
    prob_block  = 0.25
    prob_swap   = 0.20
    prob_smooth = 0.10

    it = 0
    last_report = time.monotonic()

    while time.monotonic() < deadline:
        # progress fraction for schedules
        elapsed = time.monotonic() - (deadline - float(budget_s) + safety_margin)
        total   = float(budget_s) - safety_margin
        prog = min(1.0, max(0.0, elapsed / total))

        # schedules
        eta  = eta_start  * (eta_end   / eta_start)  ** prog
        temp = temp_start * (temp_end / temp_start) ** prog

        # adapt move probabilities: favour gradient later on
        prob_grad   = 0.45 + 0.40 * prog     # up to ~0.85
        prob_block  = 0.25 - 0.20 * prog     # down to ~0.05
        prob_swap   = 0.20 - 0.05 * prog
        prob_smooth = 0.10 - 0.05 * prog

        r = rng.random()
        if r < prob_grad:
            # targeted gradient step
            # top_k adapts: start with a few indices, increase slowly
            top_k = max(5, int(0.001 * (2 * n - 1) * (1.0 - prog) + 15 * prog))
            cand, momentum = _gradient_step(current_seq, eta, top_k,
                                            momentum, beta_mom, rng)
        elif r < prob_grad + prob_block:
            cand = _block_perturbation(current_seq, rng)
        elif r < prob_grad + prob_block + prob_swap:
            cand = _swap_move(current_seq, rng, delta_frac=0.12 * (1.0 - prog) + 0.04 * prog)
        else:
            cand = _smooth_move(current_seq)

        # occasional global symmetrisation (helps flatten the convolution)
        if rng.random() < 0.09:
            cand = 0.5 * (cand + cand[::-1])

        cand_score = _evaluate_fast(cand)

        # Metropolis acceptance
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # probability decays with temperature
            prob = np.exp(-(cand_score - current_score) / max(temp, 1e-15))
            if rng.random() < prob:
                accept = True

        if accept:
            current_seq = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                # log improvement
                now = time.monotonic()
                if now - last_report > 5.0:
                    print(f"iter {it:07d} – new best: {best_score:.6f}")
                    last_report = now

        it += 1

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    best_seq = _normalize(best_seq)
    print(f"finished after {it} iterations, best score = {best_score:.6f}")

    # Convert to plain Python list as required by the harness
    return best_seq.tolist()
