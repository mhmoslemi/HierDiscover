def build_convolution_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        # j runs over possible indices.
        # j range max(0, k-(n-1)) to min(n-1, k)
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        # for each j, coefficient is f[k-j]
        # i = k - j
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n), dtype=np.float64)
    return A

"""
Propose a candidate step‑function using a simple convex‑programming
iteration (the “LP‑boost” method).

The idea is classical in the original research:
* Keep the sequence f normalized so that Σ f = √(2 n).  With this
  normalization the evaluation function equals the L∞‑norm of the
  self‑convolution (M = max(f∗f)).
* Given the current f and its current bound M, solve the linear program

        maximise   Σ g_j
        subject to  (f∗g)_k ≤ M   for all k
                    g_j ≥ 0

  which is linear in the unknown g because f is fixed.
  The optimum pushes as much mass as possible into indices where f is
  small, thereby giving a candidate g with a larger total sum while still
  respecting the convolution bound.

* Rescale g so that Σ g = √(2 n) and replace f by a convex combination
  f←(1−t)f + t g with a decreasing step size t.
  This reduces the maximum convolution value M and is repeated until the
  time budget expires.

If SciPy (required for the LP) is unavailable, a simple simulated‑annealing
fallback (the previous best‑known method) is used instead.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: arcsine (Catalan) base sequence
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative array a[i] = binom(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence: a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Helper: build the Toeplitz constraint matrix for a given f
# ----------------------------------------------------------------------
def build_convolution_matrix(f: np.ndarray):
    """
    Build a sparse matrix A (shape (2n‑1, n)) such that
        (A @ g)[k] = (f ∗ g)[k]   for 0 ≤ k < 2n‑1 .
    """
    from scipy.sparse import csr_matrix

    n = f.size
    rows = []
    cols = []
    data = []

    # For each output index k, the valid input indices j satisfy
    #   0 ≤ j ≤ n‑1  and  0 ≤ k‑j ≤ n‑1  →  max(0, k‑(n‑1)) ≤ j ≤ min(n‑1, k)
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            i = k - j                       # coefficient from f
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=np.float64)

# Fallback: simple Simulated Annealing (the previously used routine)
# ----------------------------------------------------------------------
def _fallback_simulated_annealing(seed: int, budget_s: float):

    # use a modest length when we fall back – 256 gives the same baseline as before
    n = 256
    cur = build_arcsine_sequence(n)
    cur = np.clip(cur, 0.0, 1000.0)

    target_sum = np.sqrt(2.0 * n)
    cur = cur * (target_sum / cur.sum())

    cur_score = float(evaluate_sequence(cur.tolist()))
    best = cur.copy()
    best_score = cur_score
    print(f"[fallback‑init] n={n}, score={best_score:.6f}")

    # very small annealing schedule – enough to improve a bit if the LP fails
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # simple random perturbation
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if cur[i] == 0:
            continue
        frac = rng.random() * 0.5
        delta = cur[i] * frac
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (target_sum / cand.sum())
        cand_score = float(evaluate_sequence(cand.tolist()))
        if cand_score < cur_score:

            if cand_score < best_score:

                print(f"[fallback {it:>6}] new best = {best_score:.6f}")

        if it % 5000 == 0 and best_score < 1.55:
            break

    print(f"[fallback‑final] best score {best_score:.6f}")
    return [float(x) for x in best]

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    LP‑boost search for a step‑function that minimises ``evaluate_sequence``.
    The routine runs inside the allotted time and returns the best sequence
    it has seen.
    """
    # ------------------------------------------------------------------
    # 0. Imports – try to get the LP solver, fall back otherwise
    # ------------------------------------------------------------------
    try:

        from scipy.sparse import csr_matrix
        have_lp = True
    except Exception:
        have_lp = False

    if not have_lp:

        return _fallback_simulated_annealing(seed, budget_s)

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a manageable length (the problem is scale‑invariant).
    # ------------------------------------------------------------------
    global_len = len(globals().get("height_sequence_1", []))
    max_n = 1024                     # hard limit for the LP size
    if global_len == 0:
        n = 512
    else:

            n = 256

    # 2. Starting point – normalized arcsine shape.
    # ------------------------------------------------------------------
    f = build_arcsine_sequence(n)
    f = np.clip(f, 0.0, 1000.0)

    target_sum = np.sqrt(2.0 * n)    # the normalising constant
    f = f * (target_sum / f.sum())

    cur_score = float(evaluate_sequence(f.tolist()))
    best_f = f.copy()
    best_score = cur_score

    # 3. Main LP‑boost loop.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # keep a margin
    iteration = 0

    # step‑size schedule – start fairly aggressive, then decay.
    def mixing_factor(elapsed):
        # linear decay from 0.4 to 0.05
        return max(0.05, 0.4 * (1.0 - elapsed / budget_s))

    while time.monotonic() < deadline:
        iteration += 1
        elapsed = time.monotonic() - start_time

        # --------------------------------------------------------------
        # 3.1  Current bound M = max(f∗f)   (since sum(f)=√(2n) this is the score)
        # --------------------------------------------------------------
        M = float(evaluate_sequence(f.tolist()))   # equals max convolution

        # --------------------------------------------------------------
        # 3.2  Build the linear constraints  (A @ g) <= M
        # --------------------------------------------------------------
        A = build_convolution_matrix(f)   # sparse (2n‑1) × n matrix

        # --------------------------------------------------------------
        # 3.3  Linear program – maximise Σ g_i  under A·g ≤ M, g≥0
        # --------------------------------------------------------------
        c = -np.ones(n, dtype=np.float64)          # linprog minimises c·x
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # Give the solver a tiny per‑iteration time budget so we never overrun.
        remaining = max(0.0, deadline - time.monotonic())
        lp_options = {"presolve": True}
        if remaining > 0:
            lp_options["time_limit"] = remaining   # supported by HiGHS

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options=lp_options)

        if not res.success:
            print(f"[{iteration}] LP failed – {res.message}")
            break   # cannot improve further without a valid direction

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            print(f"[{iteration}] LP gave zero mass, aborting.")
            break

        # --------------------------------------------------------------
        # 3.4  Rescale g to the required normalisation Σ g = √(2n)
        # --------------------------------------------------------------
        g = g * (target_sum / sum_g)

        # --------------------------------------------------------------
        # 3.5  Convex combination and re‑normalise.
        # --------------------------------------------------------------
        t = mixing_factor(elapsed)
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0.0, 1000.0)

        # --------------------------------------------------------------
        # 3.6  Keep the best seen candidate.
        # --------------------------------------------------------------

            best_score = new_score
            print(f"[{iteration:>6}] new best max‑conv = {best_score:.6f}")

        # 3.7  Prepare next iteration.
        # --------------------------------------------------------------
        f = f_new

        # early exit if we already beat the target comfortably
        if best_score < 1.45:
            print("[early exit] reached a convincing improvement.")
            break

    print(f"[final] best max‑conv = {best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best_f]
