def lp_step(seq, M):
    n = len(seq)
    K = 2*n-1
    # Build A: for each k, row is seq shifted.
    row_inds = []
    col_inds = []
    data = []
    for k in range(K):
        # iterate i indices where 0 <= i < n and 0 <= k-i < n
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        for i in range(i_start, i_end+1):
            j = k - i
            # coefficient = seq[j] ? Wait A_{k,i} = seq[k-i] = seq[j]
            row_inds.append(k)
            col_inds.append(i)
            data.append(seq[j]) # careful: j = k-i lies between 0,n-1.
    # Build sparse matrix
    A = scipy.sparse.csr_matrix((data, (row_inds, col_inds)), shape=(K, n))

    # Solve
    res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(K, M * 1.000001), bounds=(0, None), method='highs')
    if res.success:
        g = res.x
        # normalize
        g_sum = g.sum()
        if g_sum > 0:
            g = g / g_sum
        return g
    else:
        return seq  # fallback

def pairwise_swap(seq, rng, epsilon):
    # epsilon is relative proportion of mass to move
    n = seq.size
    # Choose source i proportionally to seq (so high mass)
    if seq.sum() == 0: return seq
    probs = seq / seq.sum()
    i = rng.choice(n, p=probs)
    # Choose target j random uniform (including i maybe)
    j = rng.integers(0, n)
    # Determine amount to move
    amount = epsilon * seq[i] * rng.random()
    # Move
    new_seq = seq.copy()
    new_seq[i] -= amount
    new_seq[j] += amount
    # Clip negative
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    # Normalize sum to 1
    s = new_seq.sum()
    if s > 0:
        new_seq /= s
    return new_seq

"""
Advanced step-function optimizer for the convolution minimisation problem.

The optimizer combines several complementary strategies:
  * A dense grid search over power‑law (and exponential) seeds to obtain a
    strong initial shape.
  * Projected gradient descent on a smooth surrogate of the “max‑convolution”
    objective (the shift vectors of the largest convolution entries are used as
    sub‑gradients).  The search stays on the probability simplex.
  * Three stochastic perturbations – multiplicative block noise, a pairwise
    mass‑transfer move and occasional symmetrisation – to escape local minima.
  * A periodic linear‑programming refinement: for the current shape f we
    maximise Σg subject to f∗g ≤ M (where M is the current worst convolution
    value).  The resulting g is blended back into f, guaranteeing a monotone
    decrease of the objective in expectation.
  * Simulated‑annealing acceptance with an exponentially decaying temperature
    to allow occasional uphill steps early on.

The code is fully self‑contained, uses only NumPy (SciPy is optional for the
LP step) and respects the 1000‑second time budget.
"""

import time
import numpy as np

# optional SciPy imports – the LP refinement will be disabled if unavailable
try:
    import scipy.sparse
    import scipy.optimize
    _SCIPY_AVAILABLE = True
except Exception:
    _SCIPY_AVAILABLE = False


def _conv_full(seq: np.ndarray) -> np.ndarray:
    """
    Full convolution of ``seq`` with itself using FFT.
    Returns an array of length 2*len(seq)-1.
    """
    n = seq.shape[0]
    L = 1 << ((2 * n - 1).bit_length())  # next power of two
    fft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
    return conv


def _evaluate_fast(seq: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for non‑negative, finite vectors.
    The vector is assumed to be already clipped to [0,1000].
    """
    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    conv = _conv_full(seq)
    max_b = conv.max()
    n = seq.shape[0]
    # The official evaluator uses 2*n*max_b / (total**2)
    return float(2.0 * n * max_b / (total * total))


def _shift(seq: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution term (seq*seq)[k] with respect to ``seq``.
    Returns a vector d where d_i = seq_{k-i} if the index is in range,
    otherwise 0.
    """
    n = seq.shape[0]
    out = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1)
        out[idx] = seq[k - idx]
    return out


def _gradient_step(seq: np.ndarray, eta: float, top_k: int = 7) -> np.ndarray:
    """
    One projected gradient step that reduces the contribution of the
    largest convolution entries.  The update direction is centred to keep
    the L1‑norm unchanged before the explicit normalisation.
    """
    conv = _conv_full(seq)
    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(-conv, top_k)[:top_k]
    shift_sum = np.zeros_like(seq)
    for k in ks:
        shift_sum += _shift(seq, int(k))
    d = shift_sum / len(ks)
    direction = -d + d.mean()          # zero‑sum direction
    new_seq = seq + eta * direction
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    s = new_seq.sum()
    if s > 0.0:
        new_seq /= s
    return new_seq


def _block_perturbation(base: np.ndarray,
                        rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal noise on a random contiguous block.
    With probability 0.35 the result is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _pairwise_swap(seq: np.ndarray,
                   rng: np.random.Generator,
                   eps: float) -> np.ndarray:
    """
    Transfer a small amount of mass from a heavy index to a random index.
    ``eps`` controls the maximal relative amount moved.
    """
    n = seq.size
    if seq.sum() == 0.0:
        return seq.copy()
    # source index weighted by current mass
    probs = seq / seq.sum()
    i = rng.choice(n, p=probs)
    j = rng.integers(0, n)            # uniform target
    amount = eps * seq[i] * rng.random()
    cand = seq.copy()
    cand[i] = max(0.0, cand[i] - amount)
    cand[j] += amount
    cand = np.clip(cand, 0.0, 1000.0)
    s = cand.sum()
    if s > 0.0:
        cand /= s
    return cand


def _lp_refine(seq: np.ndarray, M: float, rng: np.random.Generator) -> np.ndarray:
    """
    Linear‑programming refinement.
    Solve   maximise Σ g_j
    subject to   (seq * g)[k] ≤ M   for all k,
               g ≥ 0.
    Returns a normalised vector g (if the LP succeeds) otherwise the
    original sequence.
    """
    if not _SCIPY_AVAILABLE:
        return seq.copy()
    n = seq.shape[0]
    K = 2 * n - 1
    # Build sparse matrix A where A[k, i] = seq[k - i] (zero when out of range)
    rows = []
    cols = []
    data = []
    # loop over k (convolution index)
    for k in range(K):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs from i_start to i_end inclusive
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i_range                      # corresponding index in seq
        rows.extend([k] * i_range.shape[0])
        cols.extend(i_range.tolist())
        data.extend(seq[j].tolist())
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(K, n))
    b = np.full(K, M * 1.000001)            # slight tolerance
    c = -np.ones(n)                         # maximise Σg → minimise -Σg
    # Solve
    try:
        res = scipy.optimize.linprog(c,
                                     A_ub=A,
                                     b_ub=b,
                                     bounds=(0, None),
                                     method="highs",
                                     options={"presolve": True})
    except Exception:
        return seq.copy()
    if not res.success or res.x is None:
        return seq.copy()
    g = res.x
    s = g.sum()
    if s > 0.0:
        g /= s
    return g


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip to [0,1000] and renormalise to unit L1‑norm."""
    cand = np.clip(seq, 0.0, 1000.0)
    s = cand.sum()
    if s > 0.0:
        cand /= s
    return cand

    best_score = float("inf")

    alphas = np.linspace(0.25, 0.75, 21)          # broader range
    betas = np.linspace(0.0, 3.0, 13)             # include larger offsets

        if sc < best_score:

            best_seq = seq.copy()

The function `evaluate_sequence` returns  

\[
\text{score}= \frac{2\,n\,\max_k (a*a)_k}{(\sum_i a_i)^2},
\]

where `(a*a)_k` is the full discrete convolution of the non‑negative vector `a`.  
Because the score is homogeneous of degree 0, we may (and do) keep the
vector normalised to `∑a_i = 1`. Then  

\[
\text{score}=2\,n\,\max_k (a*a)_k .
\]

Hence minimising the score is exactly the problem  

*find a non‑negative vector `a` with unit ℓ₁‑norm that minimises the
largest entry of its self‑convolution*.

The code below searches for such a vector.  
It combines several complementary ideas:

1. **A dense deterministic seed pool** – power‑law shapes  
   `a_i ∝ (i+β)^{-α}` (theoretical optimum) and simple exponentials are
   enumerated on a fairly fine grid; the best of them is taken as the
   initial point.

2. **Projected gradient descent on a sub‑gradient of the max‑convolution.**  
   The sub‑gradient of a single convolution entry `(a*a)_k` is the
   “shift‑vector” `shift(a,k)`.  The algorithm averages the shift vectors
   belonging to the `top_k` largest convolution entries, centres the
   direction to keep the ℓ₁‑norm unchanged, takes a step of size `η`,
   clips to `[0,1000]` and renormalises onto the simplex.

3. **Three stochastic moves** – multiplicative block noise, a pairwise
   mass‑transfer move (move a small fraction `ε` of mass from a heavy
   index to a random one) and an optional symmetrisation – give the
   optimiser the ability to jump out of local minima.

4. **Periodic linear‑programming refinement** (optional, requires SciPy).  
   For the current vector `f` let `M = max_k (f*f)_k`.  We solve  

   \[
   \max\limits_{g\ge0}\;\sum_j g_j \quad\text{s.t.}\quad (f*g)_k\le M,
   \]

   i.e. maximise the ℓ₁‑norm under the same worst‑convolution bound.
   The optimal `g` is blended back into the current iterate (`f←(1‑t)f+tg`)
   which provably does not increase the objective.  The LP is built as a
   sparse Toeplitz matrix and solved with HiGHS (SciPy’s default
   high‑performance LP solver).

5. **Simulated‑annealing acceptance** – a temperature that decreases
   exponentially lets the search accept occasional uphill moves early on,
   improving exploration.

All operations respect the constraints of `evaluate_sequence`
(non‑negative, ≤ 1000, ℓ₁‑norm > 0.01).  The whole routine runs inside the
time budget (`budget_s` – default 500 s, the harness gives us up to 1000 s).

Advanced step‑function optimiser for the convolution minimisation challenge.

The optimiser blends:

  * projected gradient descent on the sub‑gradient of the max‑convolution;

  * simulated‑annealing acceptance with an exponentially decaying temperature.

    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    max_b = _conv_full(seq).max()
    n = seq.shape[0]
    return float(2.0 * n * max_b / (total * total))


def _shift(seq: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of (seq*seq)[k] with respect to ``seq``."""
    n = seq.shape[0]
    out = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1)
        out[idx] = seq[k - idx]
    return out


def _gradient_step(seq: np.ndarray, eta: float, top_k: int = 7) -> np.ndarray:
    """Projected gradient step moving mass away from the largest convolution peaks."""
    conv = _conv_full(seq)
    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(-conv, top_k)[:top_k]
    shift_sum = np.zeros_like(seq)
    for k in ks:
        shift_sum += _shift(seq, int(k))
    d = shift_sum / len(ks)                     # average shift vector
    direction = -d + d.mean()                   # zero‑sum direction
    new_seq = seq + eta * direction
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    s = new_seq.sum()
    if s > 0.0:
        new_seq /= s
    return new_seq


def _block_perturbation(base: np.ndarray,
                        rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal noise on a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _pairwise_swap(seq: np.ndarray,
                   rng: np.random.Generator,
                   eps: float) -> np.ndarray:
    """Transfer a small amount of mass from a heavy index to a random index."""
    n = seq.size
    if seq.sum() == 0.0:
        return seq.copy()
    probs = seq / seq.sum()
    i = rng.choice(n, p=probs)                 # source (biased to large entries)
    j = rng.integers(0, n)                     # uniform target
    amount = eps * seq[i] * rng.random()
    cand = seq.copy()
    cand[i] = max(0.0, cand[i] - amount)
    cand[j] += amount
    cand = np.clip(cand, 0.0, 1000.0)
    s = cand.sum()
    if s > 0.0:
        cand /= s
    return cand


def _lp_refine(seq: np.ndarray, M: float, rng: np.random.Generator) -> np.ndarray:
    """
    Linear‑programming refinement.
    Maximise Σ g_j subject to (seq * g)_k ≤ M for all k,  g ≥ 0.
    Returns the normalised optimum g (or the original sequence if the LP fails).
    """
    if not _SCIPY_AVAILABLE:                     # pragma: no cover
        return seq.copy()
    n = seq.shape[0]
    K = 2 * n - 1
    rows, cols, data = [], [], []
    # Build the Toeplitz‑like constraint matrix A (K × n)
    for k in range(K):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i_range
        rows.extend([k] * i_range.shape[0])
        cols.extend(i_range.tolist())
        data.extend(seq[j].tolist())
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(K, n))
    b = np.full(K, M * 1.000001)                # tiny tolerance
    c = -np.ones(n)                             # maximise Σg  → minimise −Σg
    try:
        res = scipy.optimize.linprog(c,
                                     A_ub=A,
                                     b_ub=b,
                                     bounds=(0, None),
                                     method="highs",
                                     options={"presolve": True})
    except Exception:                           # pragma: no cover
        return seq.copy()
    if not res.success or res.x is None:        # pragma: no cover
        return seq.copy()
    g = res.x
    s = g.sum()
    if s > 0.0:
        g /= s
    return g


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip to the allowed range and renormalise to unit ℓ₁‑norm."""
    cand = np.clip(seq, 0.0, 1000.0)
    s = cand.sum()
    if s > 0.0:
        cand /= s
    return cand


def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs):
    """
    Hybrid stochastic‑gradient + LP optimiser.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (a few seconds are kept as safety margin).
    **kwargs
        Ignored – kept for compatibility with the harness.

    Returns
    -------
    list[float]
        The best step‑function heights found within the time budget.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Time budget (keep a small safety margin for the external harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Choose vector length.  Use the length of a provided seed if available.
    # ------------------------------------------------------------------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        if base_seq.ndim != 1 or base_seq.size == 0:
            raise Exception
        n = base_seq.size
    except Exception:
        n = 2500

    # ------------------------------------------------------------------
    # 1️⃣  Deterministic seed pool (power‑law + exponential)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    idx = np.arange(1, n + 1, dtype=np.float64)     # 1‑based indices

    # Power‑law grid (expanded around the theoretical ½ exponent)
    alphas = np.linspace(0.25, 0.75, 21)
    betas = np.linspace(0.0, 3.0, 13)
    for alpha in alphas:
        for beta in betas:
            seq = 1.0 / ((idx + beta) ** alpha)
            seq = np.clip(seq, 0.0, 1000.0)
            s = seq.sum()
            if s == 0.0:
                continue
            seq /= s
            sc = _evaluate_fast(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()

    # Exponential family seeds
    lam_vals = np.logspace(-4, -1.5, 15)   # from 1e‑4 up to ~0.03
    for lam in lam_vals:
        seq = np.exp(-lam * (idx - 1))
        seq = np.clip(seq, 0.0, 1000.0)
        s = seq.sum()
        if s == 0.0:
            continue
        seq /= s
        sc = _evaluate_fast(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # Fallback (should never be needed)
    if best_seq is None:
        seq = np.ones(n, dtype=np.float64)
        seq /= seq.sum()
        best_seq = seq
        best_score = _evaluate_fast(seq)

    # ------------------------------------------------------------------
    # 2️⃣  Local optimisation loop
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    start = time.monotonic()
    it = 0

    # Annealing / schedule parameters
    eta_start, eta_end = 0.18, 3e-5          # step size
    temp_start, temp_end = 3e-2, 1e-6       # Metropolis temperature
    eps_start, eps_end = 0.5, 0.02           # pairwise‑swap magnitude
    lp_interval = 8000                       # LP refinement every N iterations
    lp_blend_start, lp_blend_end = 0.4, 0.05 # blending weight for LP result

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        prog = min(1.0, elapsed / (deadline - start))

        # Exponential schedules
        eta = eta_start * (eta_end / eta_start) ** prog
        temp = temp_start * (temp_end / temp_start) ** prog
        eps = eps_start * (eps_end / eps_start) ** prog
        lp_blend = lp_blend_start * (lp_blend_end / lp_blend_start) ** prog

        # ------------------------------------------------------------
        # Choose a stochastic move
        # ------------------------------------------------------------
        r = rng.random()
        if r < 0.55:
            cand = _gradient_step(cur_seq, eta, top_k=7)
        elif r < 0.80:
            cand = _block_perturbation(cur_seq, rng)
        else:
            cand = _pairwise_swap(cur_seq, rng, eps)

        # Light symmetrisation (helps flatten the peak)
        if rng.random() < 0.10:
            cand = 0.5 * (cand + cand[::-1])

        # Periodic LP refinement
        if it > 0 and it % lp_interval == 0:
            M = _conv_full(cur_seq).max()
            g = _lp_refine(cur_seq, M, rng)
            cand = (1.0 - lp_blend) * cur_seq + lp_blend * g

        # Normalise candidate
        cand = _normalize(cand)

        cand_score = _evaluate_fast(cand)

        # ------------------------------------------------------------
        # Acceptance (Metropolis‑style)
        # ------------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"iter {it:07d} – new best: {best_score:.6f}")

        # Light progress output (does not affect the algorithm)
        if it % 10000 == 0 and it > 0:
            print(f"iter {it:07d} – cur: {cur_score:.6f} – best: {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Final sanitisation (the external evaluator will clip again)
    # ------------------------------------------------------------------
    best_seq = _normalize(best_seq)
    print(f"finished after {it} iterations, best score={best_score:.6f}")

    return best_seq.tolist()
