"""
Propose a low‑score step‑function.

The search works in four stages:

1. **Broad shape sweep** – For several lengths (powers of two up to 65536)
   we generate candidate sequences from a variety of analytic families:
   symmetric Beta(α,α), asymmetric Beta(α,β), sin‑power, cosine‑half‑wave
   and powered triangular shapes.  Each candidate is scaled so that its
   ℓ₁‑norm equals √(2·n) (the scale does not affect the objective) and
   evaluated with a fast FFT‑based routine that mimics `evaluate_sequence`.

2. **Gradient refinement** – Starting from the best candidate we iteratively
   reduce the maximal convolution entry.  At each iteration we locate the
   shift `k_max` attaining the current maximum, compute the sub‑gradient
   `g_i = 2·a_{k_max‑i}` (zero outside the support) and take a projected
   gradient step with an adaptive learning rate, re‑normalising the sum
   after each step.

3. **Stochastic fine‑tuning** – A tiny random “mass‑transfer’’ walk
   (moving a small amount of weight between two positions) explores the
   neighbourhood of the refined solution.  Improvements are kept, otherwise
   the walk continues (simulated‑annealing style) while the remaining time
   permits.

4. **Final evaluation** – The best sequence found is passed to the provided
   `evaluate_sequence` function for the exact score, printed and returned
   as a plain list of floats.

All helpers are defined at module level; the only public entry point is
`propose_candidate`, which respects the given time budget.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Fast evaluation (same clipping as evaluate_sequence)
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray) -> float:
    """Return 2·n·max(b)/sum(a)^2, where b = a*a (self‑convolution)."""
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution of a via FFT (no clipping)."""
    n = a.shape[0]
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    return conv


# ----------------------------------------------------------------------
# Normalisation helper (scale to √(2·n) and clip)
# ----------------------------------------------------------------------
def _scale_and_clip(w: np.ndarray) -> np.ndarray:
    n = w.shape[0]
    target = sqrt(2.0 * n)
    total = w.sum()
    if total == 0.0:
        w = np.ones_like(w)
        total = w.sum()
    a = w * (target / total)
    a = np.clip(a, 0.0, 1000.0)
    return a


# ----------------------------------------------------------------------
# Shape families
# ----------------------------------------------------------------------
def _beta_sym_sequence(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on (0,1) sampled at mid‑points."""
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1.0) * (np.log(p) + np.log(1.0 - p))
    w = np.exp(logw)
    w[~np.isfinite(w)] = 0.0
    return _scale_and_clip(w)


def _beta_asym_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density."""
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1.0) * np.log(p) + (beta - 1.0) * np.log(1.0 - p)
    w = np.exp(logw)
    w[~np.isfinite(w)] = 0.0
    return _scale_and_clip(w)


def _sin_power_sequence(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ sin(π·p_i)^γ."""
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(np.pi * p) ** gamma
    return _scale_and_clip(w)


def _cos_half_sequence(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ max(0,cos(π·(p_i−½)))^γ (half‑wave cosine)."""
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.maximum(0.0, np.cos(np.pi * (p - 0.5))) ** gamma
    return _scale_and_clip(w)


def _triangular_power_sequence(n: int, power: float) -> np.ndarray:
    """a_i ∝ min(p,1−p)^power."""
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.minimum(p, 1.0 - p) ** power
    return _scale_and_clip(w)


# ----------------------------------------------------------------------
# Gradient‑based refinement of a single sequence
# ----------------------------------------------------------------------
def _gradient_refine(a: np.ndarray,
                     target_sum: float,
                     max_iter: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """Iteratively reduce max(self‑convolution) by projected gradient steps."""
    n = a.shape[0]
    best = a.copy()
    best_score = _fast_objective(best)
    cur = a.copy()
    cur_score = best_score
    # initial learning rate (relative to typical entry size)
    lr = 0.5 * target_sum / n
    for _ in range(max_iter):
        if time.monotonic() >= deadline:
            break
        conv = _conv_fft(cur)
        k_max = int(np.argmax(conv))
        # Build gradient g_i = 2·a_{k_max-i} where defined
        g = np.zeros_like(cur)
        start_i = max(0, k_max - (n - 1))
        end_i = min(n - 1, k_max)
        if start_i <= end_i:
            idx = np.arange(start_i, end_i + 1, dtype=np.int64)
            g[idx] = 2.0 * cur[k_max - idx]
        # Take a gradient step
        cand = cur - lr * g
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > 0.0:
            cand = cand * (target_sum / s)
        cand_score = _fast_objective(cand)
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            # keep best ever
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            # no improvement – shrink step size
            lr *= 0.7
            if lr < 1e-12:
                break
    return best, best_score
