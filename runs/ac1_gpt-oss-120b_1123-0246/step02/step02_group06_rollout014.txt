def build_conv_matrix(a):
    n = a.shape[0]
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # For each col j
    for j in range(n):
        A[j: j+n, j] = a
    return A

def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np, math
    try: import scipy.optimize as opt
    except ImportError: fallback to SA.

    rng = np.random.default_rng(seed)

    # Determine n
    if 'height_sequence_1' in globals():
        n0 = len(height_sequence_1)
    else:
        n0 = 0
    n_target = min(max(200, n0), 800)  # ensure at least 200, limit 800
    if n_target == 0:
        n_target = 400

    # start from arcsine
    f = build_arcsine_sequence(n_target)
    f_sum = f.sum()
    # scale
    target_sum = math.sqrt(2 * n_target)
    f = f * (target_sum / f_sum)

    cur = f.copy()
    cur_score = evaluate_sequence(cur.tolist())
    best = cur.copy()
    best_score = cur_score

    start = time.monotonic()
    deadline = start + budget_s - 0.5

    # initial M
    M = np.max(np.convolve(cur, cur))

    iter_count = 0
    while time.monotonic() < deadline:
        iter_count += 1
        # Build A matrix
        A = np.zeros((2*n_target-1, n_target), dtype=np.float64)
        # fill columns
        for j in range(n_target):
            A[j: j+n_target, j] = cur
        # LP: maximize sum(g) subject to A @ g <= M, g >=0, g <= 1000
        c = -np.ones(n_target)
        bounds = [(0.0, 1000.0)] * n_target
        # Use high speed method
        res = opt.linprog(c, A_ub=A, b_ub=np.full(2*n_target-1, M), bounds=bounds, method='highs')
        if not res.success:
            print(f"[{iter_count}] LP failed: {res.message}")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= cur.sum() * (1 + 1e-6):
            print(f"[{iter_count}] No sum improvement: sum_g={sum_g:.6f} <= sum_cur")
            break
        # scale g to same total sum as cur
        scale = cur.sum() / sum_g
        g_scaled = g * scale
        # combination step sizes: try t values decreasing
        improved = False
        for t in [0.5, 0.2, 0.8]:
            cand = (1 - t) * cur + t * g_scaled
            cand = np.clip(cand, 0.0, 1000.0)  # ensure constraints
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score - 1e-9:
                best = cand.copy()
                best_score = cand_score
                cur = cand.copy()
                cur_score = cand_score
                M = np.max(np.convolve(cur, cur))
                improved = True
                print(f"[{iter_count}] new best score={best_score:.6f} with t={t}")
                break
        if not improved:
            # fallback: simple convex combination small step
            t = 0.1
            cand = (1 - t) * cur + t * g_scaled
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                M = np.max(np.convolve(cur, cur))
        # early termination if best_score already below 1.5
        if best_score < 1.5:
            break

    print(f"[final] best_score={best_score:.6f} after {iter_count} iterations")
    return [float(x) for x in best]

"""
Iterative LP‑based improvement of the arcsine step‑function.

The algorithm starts from the analytically optimal arcsine sequence,
scaled so that Σa = √(2·n).  For a fixed upper bound M = max(a∗a) we
solve a linear program

    maximise   Σ g_i
    subject to  (a ∗ g)_k ≤ M   for all k,
                g_i ≥ 0, g_i ≤ 1000.

Because the convolution (a∗g) is linear in g, the LP is cheap for
moderate n (a few hundred).  The optimum has a strictly larger
ℓ₁–norm than the current vector a, i.e. we can pack more mass into
the same convolution bound by moving weight to indices where a is
small.  We then rescale g to keep the total sum unchanged and mix it
with the current sequence:

        a ← (1−t)·a + t·g_scaled   (t≈0.5)

The mixed vector still respects the same bound M up to a term of order
t²·max(g∗g); with a moderate step size the bound strictly decreases.
Repeating the LP → rescaling → mixing step yields a monotone descent
in the evaluation score 2·n·max(a∗a)/Σa².  The process stops when the
time budget expires or no further increase of the ℓ₁‑norm is possible.

If SciPy is unavailable the code falls back to a simple simulated‑annealing
search (the previous implementation).  All numbers are kept in the
allowed range [0,1000] and the best sequence found is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Try to import a fast LP solver from SciPy.  If unavailable we will fall
# back to a simple stochastic search.
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Helper: arcsine (central binomial) coefficients.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative array a[i] = binom(2i,i) / 4**i (i=0..length-1)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Build the convolution matrix A such that (a ∗ x) = A @ x .
# A has shape (2·n−1, n) and row k contains the coefficients a_{k−i}
# (zero outside the original support).
# ----------------------------------------------------------------------
def conv_matrix(seq: np.ndarray) -> np.ndarray:
    n = seq.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j corresponds to seq shifted by j positions
    for j in range(n):
        A[j : j + n, j] = seq
    return A

    def transfer_move(base, rng):
        n = base.size

            return base.copy()
        amount = rng.random() * 0.5 * sum_prefix
        factor_pre = max(0.0, (sum_prefix - amount) / sum_prefix)
        new_prefix = prefix * factor_pre
        if sum_suffix > 0.0:
            factor_suf = (sum_suffix + amount) / sum_suffix
            new_suffix = suffix * factor_suf
        else:
            new_suffix = np.full_like(suffix, amount / suffix.size)
        cand = np.concatenate([new_prefix, new_suffix])
        return np.clip(cand, 0.0, 1000.0)

    cur = build_arcsine_sequence_local(n)
    cur = np.clip(cur, 0.0, 1000.0)
    cur_score = float(evaluate_sequence(cur.tolist()))
    best = cur.copy()
    best_score = cur_score

    print(f"[init‑fallback] length={n}, initial score={best_score:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)
    it = 0
    T_start = 0.03
    T_end = 1e-5
    while time.monotonic() < deadline:
        it += 1
        elapsed = (time.monotonic() - start_time) / budget_s
        T = T_start * (1.0 - elapsed) + T_end * elapsed
        r = rng.random()
        if r < 0.45:
            cand = transfer_move(cur, rng)
        elif r < 0.80:
            cand = block_perturbation(cur, rng)
        else:
            cand = rebalance_move(cur, rng)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
            continue
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = rng.random() < prob
        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                print(f"[{it:>6}] new best = {best_score:.6f}")

        if it % 30000 == 0:
            cur = best.copy()
            cur_score = best_score

        if best_score < 1.50:
            break

    print(f"[final‑fallback] best score {best_score:.6f} after {it} iterations")
    return [float(x) for x in best]


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Return a step‑function (list of non‑negative floats) that minimises
    ``evaluate_sequence`` within the given time budget.  For the usual
    case SciPy is available and an LP‑based refinement of the arcsine
    construction is performed; otherwise a simple simulated‑annealing
    fallback is used.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a moderate length.  If a reference sequence is present we
    # honour its length up to a safe cap; otherwise we default to 400.
    # ------------------------------------------------------------------
    ref_len = int(len(globals().get("height_sequence_1", [])))
    if ref_len > 0:
        n = min(max(200, ref_len), 800)   # keep the LP tractable
    else:
        n = 400

    # ------------------------------------------------------------------
    # 1. Start from the analytically optimal arcsine shape.
    # ------------------------------------------------------------------
    cur = build_arcsine_sequence(n)
    # Scale so that Σa = √(2·n) – this makes the evaluation equal to the
    # maximum convolution value (the objective = M).
    target_sum = math.sqrt(2.0 * n)
    cur = cur * (target_sum / cur.sum())
    cur = np.clip(cur, 0.0, 1000.0)

    cur_score = float(evaluate_sequence(cur.tolist()))
    best = cur.copy()
    best_score = cur_score

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Iteratively enlarge the ℓ₁‑norm under the same convolution bound.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)  # 0.5 s safety margin
    iteration = 0
    max_iters = 30          # safety limit; the loop will also stop by time

    if not _SCIPY_AVAILABLE:
        # SciPy missing – use the fallback SA routine.
        return _fallback_simulated_annealing(seed, budget_s)

    while time.monotonic() < deadline and iteration < max_iters:
        iteration += 1

        # current convolution bound
        conv_cur = np.convolve(cur, cur)
        M = conv_cur.max()

        # Build the convolution matrix A such that (cur ∗ g) = A @ g .
        A = conv_matrix(cur)
        b_vec = np.full(A.shape[0], M, dtype=np.float64)

        # Linear program: maximise Σ g_i   ⇔ minimise -Σ g_i .
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, 1000.0)] * n

        # Solve the LP (highs is very robust for moderate sizes)
        res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
        if not res.success:
            print(f"[{iteration}] LP failure: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        sum_cur = cur.sum()

        # If the LP cannot increase the ℓ₁‑norm we have reached a
        # (discrete) extremal point – stop the refinement.
        if sum_g <= sum_cur * (1.0 + 1e-8):
            print(f"[{iteration}] No ℓ₁‑norm increase (sum_g={sum_g:.6f} ≤ sum_cur={sum_cur:.6f})")
            break

        # Rescale g so that it has the same ℓ₁‑norm as cur.
        g_scaled = g * (sum_cur / sum_g)

        # Try a few mixing coefficients; the best improvement is kept.
        improved = False
        for t in (0.5, 0.3, 0.7):
            cand = (1.0 - t) * cur + t * g_scaled
            cand = np.clip(cand, 0.0, 1000.0)
            cand_score = float(evaluate_sequence(cand.tolist()))
            if cand_score < best_score - 1e-9:
                best = cand.copy()
                best_score = cand_score
                cur = cand.copy()
                cur_score = cand_score
                improved = True
                print(f"[{iteration}] new best = {best_score:.6f} (mix t={t:.2f})")
                break

        if not improved:
            # Small step to retain progress even if the full mix does not
            # improve the objective; accept only if it strictly lowers the
            # score.
            t_small = 0.1
            cand = (1.0 - t_small) * cur + t_small * g_scaled
            cand = np.clip(cand, 0.0, 1000.0)
            cand_score = float(evaluate_sequence(cand.tolist()))
            if cand_score < cur_score:
                cur = cand.copy()
                cur_score = cand_score

        # Early termination if we have already crossed a useful threshold.
        if best_score < 1.503:
            print(f"[{iteration}] target reached (score={best_score:.6f})")
            break

    total_time = time.monotonic() - start_time
    print(f"[final] best score {best_score:.6f} after {iteration} iterations, time {total_time:.1f}s")
    return [float(x) for x in best]
