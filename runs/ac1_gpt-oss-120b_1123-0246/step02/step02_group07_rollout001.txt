"""
Convolution‑bound optimiser – block‑scaled arcsine + annealing

The algorithm works in three stages:

1. **Base shape** – the analytically optimal “arcsine’’ sequence
   a_i = C(2i,i)/4^i is built (or taken from the provided
   `height_sequence_1`).  Its shape is close to optimal.

2. **Block‑wise multiplicative scaling** – the sequence is divided into
   K contiguous blocks (≈30 elements per block).  Each block carries a
   single scaling factor.  Starting from all factors =1 we perform a
   simulated‑annealing / hill‑climbing search that repeatedly proposes
   small multiplicative changes of one or two blocks.  The objective
   (2·n·max(conv)/sum(a)^2) is evaluated by an FFT‑based convolution
   (exactly what `evaluate_sequence` uses, but far faster).

3. **Fine‑grained local search** – after the block‑scale phase the
   best sequence is refined with the “directed transfer’’ move from the
   reference solution, moving a little mass from indices that contribute
   most to the current maximal convolution coefficient towards low‑impact
   indices.

Both phases respect the security checks of `evaluate_sequence`
(non‑negative, ≤1000).  The whole routine honors the time budget
(`budget_s`, default 500 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Base arcsine sequence
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the inverse‑square‑root (arcsine) sequence of given length.
    a[i] = binom(2i,i) / 4**i  (values lie in [0,1])."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# 2.  Fast evaluation of the objective (identical to evaluate_sequence)
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray, n: int, fft_len: int) -> float:
    """Compute 2·n·max(conv(seq,seq)) / (sum(seq)^2) with the same
    clipping rules as `evaluate_sequence`.  Fast via FFT."""
    seq = np.clip(seq, 0.0, 1000.0)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    # convolution by FFT
    f = np.fft.rfft(seq, fft_len)
    conv = np.fft.irfft(f * f, fft_len)[: 2 * n - 1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# 3.  Directed transfer move (local fine‑tuning)
# ----------------------------------------------------------------------
def directed_transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small fraction of mass from the index that contributes most
    to the current maximal convolution entry to an index with a low value."""
    n = base.size
    conv = np.convolve(base, base)
    max_idx = int(np.argmax(conv))

    # range of valid indices that participate in this convolution entry
    lo = max(0, max_idx - (n - 1))
    hi = min(n - 1, max_idx)
    idx = np.arange(lo, hi + 1)
    contrib = base[idx] * base[max_idx - idx]

    src = int(idx[np.argmax(contrib)])  # most harmful index

    # pick a low‑impact target (20‑th percentile of the whole sequence)
    low_thr = np.percentile(base, 20)
    low_candidates = np.where(base <= low_thr)[0]
    if low_candidates.size == 0:
        # fallback: a random index far from src
        tgt = (src + n // 2) % n
    else:
        tgt = int(rng.choice(low_candidates))

    if tgt == src:
        return base.copy()

    # fraction of mass to relocate
    frac = rng.random() * 0.20 + 0.01      # 1 % … 21 %
    delta = frac * base[src]
    if delta <= 0.0:
        return base.copy()

    cand = base.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# 4.  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of floats – the best step‑function heights found
    within the time budget.  The optimisation proceeds in three phases:

    * Phase 1 – start from the (possibly provided) arcsine sequence.
    * Phase 2 – block‑wise multiplicative scaling of the arcsine shape,
      explored by simulated‑annealing / hill‑climbing.
    * Phase 3 – directed‑transfer local search for final polishing.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (use the supplied reference sequence if present)
    # --------------------------------------------------------------
    height_seq = globals().get("height_sequence_1", [])
    n = int(len(height_seq))
    if n == 0:
        n = 256                     # fallback – still a reasonable size
        base_seq = build_arcsine_sequence(n)
    else:
        base_seq = np.array(height_seq, dtype=np.float64)

    # --------------------------------------------------------------
    # Pre‑computations
    # --------------------------------------------------------------
    # FFT length – next power of two ≥ 2·n‑1
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    # Block partition (≈30 elements per block, at least 20 blocks)
    K = max(20, min(200, n // 30))
    block_idx = (np.arange(n) * K // n).astype(np.int32)   # maps each i → block

    # Initial scaling factors (all 1 → pure arcsine)
    scaling = np.ones(K, dtype=np.float64)

    # Evaluate the initial sequence
    cur_seq = base_seq * scaling[block_idx]
    cur_score = fast_score(cur_seq, n, fft_len)
    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"[init] n={n}, K={K}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – block‑wise simulated annealing
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a few seconds slack
    it = 0
    sigma = 0.22                     # std. dev. for multiplicative factor
    T0 = max(best_score - 1.0, 0.1)  # initial temperature (scale of score gaps)

    while time.monotonic() < deadline:
        it += 1

        # linear temperature schedule (cold‑to‑hot = low‑to‑high)
        elapsed = time.monotonic() - start_time
        fraction = max(0.0, 1.0 - elapsed / (deadline - start_time))
        T = T0 * fraction

        # ------------------------------------------------------------------
        # Propose a move on the scaling vector
        # ------------------------------------------------------------------
        cand_scaling = scaling.copy()
        move_type = rng.choice(["single", "pair", "boost"])
        if move_type == "single":
            # tweak a single block multiplicatively
            i = rng.integers(K)
            factor = float(np.exp(rng.normal(0.0, sigma)))
            cand_scaling[i] *= factor
        elif move_type == "pair":
            # transfer a relative amount from one block to another
            i, j = rng.choice(K, size=2, replace=False)
            delta = rng.random() * 0.14 + 0.01      # 1 % … 15 %
            cand_scaling[i] *= (1.0 - delta)
            cand_scaling[j] *= (1.0 + delta)
        else:  # boost – aggressively increase a random block
            i = rng.integers(K)
            cand_scaling[i] *= (1.0 + rng.random() * 0.25)

        # keep scaling within safe bounds (prevent overflow / vanishing)
        cand_scaling = np.clip(cand_scaling, 1e-12, 1000.0)

        # Build candidate sequence and evaluate (fast)
        cand_seq = base_seq * cand_scaling[block_idx]
        cand_score_fast = fast_score(cand_seq, n, fft_len)

        # ------------------------------------------------------------------
        # Acceptance decision
        # ------------------------------------------------------------------
        accept = False
        if cand_score_fast < cur_score:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp(-(cand_score_fast - cur_score) / T)
                accept = rng.random() < prob

        if accept:
            # Re‑evaluate with the official evaluator to avoid FFT artefacts
            cand_score_true = float(evaluate_sequence(cand_seq.tolist()))
            cur_seq = cand_seq
            cur_score = cand_score_true
            scaling = cand_scaling

            if cand_score_true < best_score:
                best_seq = cand_seq.copy()
                best_score = cand_score_true
                print(f"[{it:>6}] new best = {best_score:.6f}")

                if best_score < 1.503:
                    print("[stop] target reached")
                    break

        # slowly reduce the step size
        if it % 5000 == 0:
            sigma *= 0.985
            sigma = max(sigma, 0.04)

    # --------------------------------------------------------------
    # Phase 3 – directed transfer fine‑tuning (uses the original move)
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        cand_seq = directed_transfer_move(best_seq, rng)
        cand_score = float(evaluate_sequence(cand_seq.tolist()))
        if cand_score < best_score:
            best_seq = cand_seq
            best_score = cand_score
            print(f"[fine] improved → {best_score:.6f}")
            if best_score < 1.503:
                break

    print(f"[final] best score {best_score:.6f} after {it} annealing iterations")
    return [float(x) for x in best_seq]
