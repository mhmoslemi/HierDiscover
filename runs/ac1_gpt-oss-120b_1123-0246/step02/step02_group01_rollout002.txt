```
def gradient_at_max(f, k):
    n = len(f)
    g = np.zeros_like(f)
    # indices i such that j = k - i within [0, n-1]
    i_low = max(0, k - (n-1))
    i_high = min(n-1, k)
    # For i in [i_low, i_high], j = k - i
    # g[i] = 2 * f[j]
    # So g[i_low : i_high+1] = 2 * f[k - i_low : k - i_high - 1: -1]
    # ironically we need reversed slice
    segment = f[k - i_high : k - i_low + 1]  # This slice is f in increasing order of j as i goes decreasing
    # But we want reversed
    g[i_low : i_high+1] = 2 * segment[::-1]
    return g
```

```
    `evaluate_sequence` objective.  The algorithm starts from a
    symmetric Beta(α,α) distribution (arcsine shape) and then
    performs a projected sub‑gradient descent on the max‑convolution
    objective, keeping the vector on the probability simplex.
    Random perturbations are applied while time permits.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # Choose sequence length (power of two for FFT efficiency)

    best_seq = None
    best_score = np.inf

        sc = _fast_objective(seq0)
        best_seq = seq0.copy()
        best_score = sc
        print(f"[init] global baseline score = {best_score:.6f}")

        pass

    def init_beta(alpha):
        p = (np.arange(n, dtype=np.float64) + 0.5) / n

        if w_sum == 0.0:

        a = w / w_sum  # normalise to sum 1
```

"""
Projected sub‑gradient optimizer for the self‑convolution max‑norm.

The algorithm proceeds as follows:

1.  Initialise a symmetric Beta(α,α) distribution (the “arcsine” case
    α≈0.5 is a strong starting point).  The distribution is normalised to
    unit sum – the evaluation function is invariant under a global scaling,
    so fixing the sum simplifies the optimisation.

2.  Repeatedly:
    * compute the linear self‑convolution via an FFT,
    * locate the index (or indices) where the convolution attains its
      maximum,
    * form a sub‑gradient of the max‑convolution (for a given index *k*
      the gradient is `g_i = 2·f_{k‑i}` wherever the index is valid),
    * take a diminishing step in the negative direction,
    * project back onto the probability simplex (non‑negative, unit sum).

    The step‑size is `η_t = η₀ / sqrt(t)` with a small adaptive factor.
    Improvements are kept as the current best sequence.

3.  When the time budget permits a cheap random “mass‑swap” walk is
    performed: a tiny amount of probability mass is moved from one
    position to another while maintaining the simplex constraint; any
    improvement is retained.

4.  Finally the best sequence found is fed to the supplied
    `evaluate_sequence` function (which also enforces the [0,1000] clip)
    and the resulting list of heights is returned.

The whole routine respects the time budget supplied via `budget_s`
(typically 500 s) and never exceeds the 1000 s limit.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Helper: fast objective (the same formula as evaluate_sequence)
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray) -> float:
    """Return the evaluation score 2·n·max(conv)/sum(a)² using FFT."""
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Helper: linear self‑convolution via FFT (returns full length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    return conv


# ----------------------------------------------------------------------
# Helper: sub‑gradient of the max‑convolution entry at index k
# ----------------------------------------------------------------------
def _gradient_at_max(f: np.ndarray, k: int) -> np.ndarray:
    """
    Compute ∇_f (f*f)[k] = 2·f_{k‑i} for all i where the term exists.
    """
    n = f.shape[0]
    g = np.zeros_like(f)
    i_low = max(0, k - (n - 1))
    i_high = min(n - 1, k)
    # i runs from i_low .. i_high
    i = np.arange(i_low, i_high + 1, dtype=np.int64)
    j = k - i                                     # valid because of the bounds above
    g[i] = 2.0 * f[j]
    return g


# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (non‑negative, unit sum)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto {x ≥ 0, sum x = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # uniform fallback – should rarely happen
        n = v.shape[0]
        return np.full_like(v, 1.0 / n)
    return v / s


# ----------------------------------------------------------------------
# Helper: generate a symmetric Beta(α,α) shaped sequence, normalised to sum 1
# ----------------------------------------------------------------------
def _beta_normalized(n: int, alpha: float) -> np.ndarray:
    """
    Build a length‑n non‑negative sequence proportional to
    p^{α‑1}(1‑p)^{α‑1} with p = (i+0.5)/n, then normalise to sum 1.
    """

    if w.sum() == 0.0:
        w[:] = 1.0
    a = w / w.sum()
    a = np.clip(a, 0.0, 1000.0)                 # safety – never triggered for our scales
    return a
