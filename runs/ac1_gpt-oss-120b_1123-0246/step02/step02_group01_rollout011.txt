def gradient_descent(a, target_sum, best_score, deadline):
    n = len(a)
    step = target_sum / (5 * n)  # initial step size ~typical a magnitude/5
    while time.monotonic() < deadline and iteration < max_iter:
        conv = compute conv via FFT (fast)
        max_val = conv.max()
        max_idx = int(np.argmax(conv))
        # compute gradient of max conv term
        lo = max(0, max_idx - (n-1))
        hi = min(n-1, max_idx)
        # indices in range [lo, hi]
        idxs = np.arange(lo, hi+1, dtype=np.int64)
        # matched counterpart = max_idx - idxs
        jdxs = max_idx - idxs
        grad = np.zeros_like(a)
        grad[idxs] = 2 * a[jdxs]   # gradient contributions
        # Potential check: If max_idx odd? Works.
        # Update a candidate
        a_new = a - step * grad
        # Clip
        a_new = np.maximum(a_new, 0.0)
        # renormalise
        s = a_new.sum()
        if s > 0:
            a_new = a_new * (target_sum / s)
        # compute score of a_new
        score_new = _fast_objective(a_new)
        # simple acceptance
        if score_new < best_score:
            best_seq = a_new.copy()
            best_score = score_new
            a = a_new
            # possibly increase step slightly
            step *= 1.05
        else:
            # reject; decrease step
            step *= 0.5
        iteration += 1
        # occasional log
        if iteration % 100 == 0:
            print(f"[gd] iter {iteration} step {step:.6g} best_score {best_score:.6f}")
        # also break if step becomes too small (<1e-12)
        if step < 1e-12:
            break
    return best_seq, best_score

"""
Propose a step‐function with a low evaluation score.

Algorithm
---------
1.  Initialise a collection of candidate shapes (constant, Beta(α,α),
    general Beta(α,β), sin^p, semi‑circular) for several lengths n.
    All candidates are scaled to have sum sqrt(2·n) (the scale that
    makes ``evaluate_sequence`` equal to the maximal convolution entry).
2.  Keep the best candidate (lowest fast objective) and, if it is a
    symmetric Beta, refine the α‑parameter with a golden‑section search.
3.  If the best shape is a Beta, also try mixing it with the uniform shape
    (a convex combination) to flatten the self‑convolution further.
4.  Starting from the best sequence run a projected gradient descent that
    directly minimises the maximum entry of the self‑convolution:
    * compute the convolution via FFT,
    * locate the lag where the maximum occurs,
    * move a tiny amount of mass opposite to the gradient of that lag,
    * project back onto the non‑negative orthant and renormalise the sum.
    The step size is adapted (increase on success, decrease on failure).
5.  A final short random‑perturbation sweep (mass swaps) is performed
    while time permits.
6.  The resulting sequence is evaluated with the provided
    ``evaluate_sequence`` and returned as a plain Python list.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Fast objective (same formula as evaluate_sequence but without clipping)
# ----------------------------------------------------------------------
def _fast_objective(seq: np.ndarray) -> float:
    """2·n·max(conv)/sum(seq)^2 computed via FFT."""
    seq = np.clip(seq, 0.0, 1000.0)
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return np.inf
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    return 2.0 * n * np.max(conv) / (s * s)


# ----------------------------------------------------------------------
# Shape generators (all return a numpy array of length n with sum sqrt(2n))
# ----------------------------------------------------------------------
def _target_sum(n: int) -> float:
    return sqrt(2.0 * n)


def _uniform_sequence(n: int) -> np.ndarray:
    a = np.ones(n, dtype=np.float64)
    a *= _target_sum(n) / n
    return a


def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) shape."""
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, alpha - 1.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    a = w * (_target_sum(n) / w_sum)
    return np.clip(a, 0.0, 1000.0)


def _beta_general_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) shape (possibly asymmetric)."""
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, beta - 1.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    a = w * (_target_sum(n) / w_sum)
    return np.clip(a, 0.0, 1000.0)


def _sin_power_sequence(n: int, power: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^power."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(np.pi * x) ** power
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    a = w * (_target_sum(n) / w_sum)
    return np.clip(a, 0.0, 1000.0)


def _semicircle_sequence(n: int, exponent: float) -> np.ndarray:
    """Semi‑circular shape: (1‑t^2)^{exponent/2} with t∈[-1,1]."""
    t = 2.0 * np.arange(n, dtype=np.float64) / (n - 1) - 1.0
    w = np.maximum(0.0, 1.0 - t ** 2) ** (exponent / 2.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    a = w * (_target_sum(n) / w_sum)
    return np.clip(a, 0.0, 1000.0)


def _mixture_sequence(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination of ``base`` with a uniform sequence."""
    n = base.shape[0]
    uniform = _uniform_sequence(n)
    mixed = lam * base + (1.0 - lam) * uniform
    mixed = np.clip(mixed, 0.0, 1000.0)
    s = mixed.sum()
    if s > 0.0:
        mixed *= (_target_sum(n) / s)
    return mixed


# ----------------------------------------------------------------------
# Convolution helper (returns full linear convolution)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    return conv


# ----------------------------------------------------------------------
# Gradient descent on the L∞-norm of the self‑convolution
# ----------------------------------------------------------------------
def _gradient_descent(a: np.ndarray, deadline: float,
                      init_step: float = None,
                      max_iter: int = 20000) -> (np.ndarray, float):
    """Projected GD that tries to reduce max_a*a."""
    n = a.shape[0]
    target = _target_sum(n)
    step = init_step or target / (10.0 * n)  # a modest starting step
    best_seq = a.copy()
    best_score = _fast_objective(best_seq)

    it = 0
    while time.monotonic() < deadline and it < max_iter:
        conv = _full_convolution(a)
        max_val = conv.max()
        max_idx = int(conv.argmax())

        # gradient of the active lag (2*a[j] where j = max_idx - i)
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idxs = np.arange(lo, hi + 1, dtype=np.int64)
        js = max_idx - idxs
        grad = np.zeros_like(a)
        grad[idxs] = 2.0 * a[js]

        # try a step; if it doesn't improve, halve the step
        improved = False
        for _ in range(5):  # up to 5 reductions of step size
            cand = a - step * grad
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0.0:
                cand = cand * (target / s)
            cand = np.clip(cand, 0.0, 1000.0)
            cand_score = _fast_objective(cand)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                step *= 1.05          # slightly increase on success
                improved = True
                break
            else:
                step *= 0.5           # reduce step and retry

        if not improved:
            # no improvement after step reductions – give up a little
            step *= 0.5

        it += 1
        if it % 200 == 0:
            print(f"[gd] iter {it:5d}, step {step:.3e}, best_score {best_score:.6f}")

        if step < 1e-12:
            break

    return best_seq, best_score

                best_score, best_seq, best_n, best_type, best_params = sc, seq.copy(), n, "beta_sym", {"alpha": alpha}
                print(f"[grid] n={n:5d} β(α,α) α={alpha:5.3f} → score={sc:.6f}")

        # ----- General Beta(α,β) ----------------------------------------
        for alpha in ab_vals:
            for beta in ab_vals:
                if time.monotonic() >= deadline:
                    break
                seq = _beta_general_sequence(n, alpha, beta)
                sc = _fast_objective(seq)
                if sc < best_score:
                    best_score, best_seq, best_n, best_type, best_params = sc, seq.copy(), n, "beta_gen", {"alpha": alpha, "beta": beta}
                    print(f"[grid] n={n:5d} β(α,β) α={alpha:.2f} β={beta:.2f} → score={sc:.6f}")

        # ----- sin^p ----------------------------------------------------
        for p in p_grid:
            if time.monotonic() >= deadline:
                break
            seq = _sin_power_sequence(n, p)
            sc = _fast_objective(seq)
            if sc < best_score:
                best_score, best_seq, best_n, best_type, best_params = sc, seq.copy(), n, "sin_pow", {"p": p}
                print(f"[grid] n={n:5d} sin^p p={p:5.3f} → score={sc:.6f}")

        # ----- semi‑circular --------------------------------------------
        for expo in expo_grid:
            if time.monotonic() >= deadline:
                break
            seq = _semicircle_sequence(n, expo)
            sc = _fast_objective(seq)
            if sc < best_score:
                best_score, best_seq, best_n, best_type, best_params = sc, seq.copy(), n, "semi", {"expo": expo}
                print(f"[grid] n={n:5d} semi expo={expo:5.3f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Local refinement for the best Beta(α,α) shape (if applicable)
    # ------------------------------------------------------------------
    if best_type == "beta_sym" and best_params is not None:
        n = best_n
        lo = max(0.05, best_params["alpha"] - 0.3)
        hi = min(1.5, best_params["alpha"] + 0.3)
        phi = (1.0 + 5 ** 0.5) / 2.0
        # initialise golden‑section points
        c = hi - (hi - lo) / phi
        d = lo + (hi - lo) / phi
        ac = _fast_objective(_beta_sequence(n, c))
        ad = _fast_objective(_beta_sequence(n, d))

        while time.monotonic() < deadline and (hi - lo) > 1e-5:
            if ac < ad:
                hi, ad, d = c, ac, c
                c = hi - (hi - lo) / phi
                ac = _fast_objective(_beta_sequence(n, c))
            else:
                lo, ac, c = d, ad, d
                d = lo + (hi - lo) / phi
                ad = _fast_objective(_beta_sequence(n, d))

            # keep the best seen
            if ac < best_score:
                best_score = ac
                best_seq = _beta_sequence(n, c).copy()
                best_params["alpha"] = c
            if ad < best_score:
                best_score = ad
                best_seq = _beta_sequence(n, d).copy()
                best_params["alpha"] = d

        print(f"[refine] n={n} β(α,α) α≈{best_params['alpha']:.6f} → score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Mixture with uniform shape (only sensible for a Beta base)
    # ------------------------------------------------------------------
    if best_type in ("beta_sym", "beta_gen") and time.monotonic() < deadline:
        n = best_n
        base_seq = best_seq.copy()
        for lam in np.linspace(0.0, 1.0, 21):
            if time.monotonic() >= deadline:
                break
            cand = _mixture_sequence(base_seq, lam)
            sc = _fast_objective(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand.copy()
                best_type = "mixture"
                best_params = {"lam": lam, "base_type": best_type}
                print(f"[mix] λ={lam:5.3f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 4.  Gradient‑descent polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        print("[gd] starting gradient descent")
        best_seq, best_score = _gradient_descent(best_seq, deadline)

    # ------------------------------------------------------------------
    # 5.  Tiny random‑swap perturbation sweep (if time remains)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        a = best_seq.copy()
        n = a.size
        target = _target_sum(n)
        swap_trials = 0
        while time.monotonic() < deadline and swap_trials < 50000:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # propose swapping the two entries
            cand = a.copy()
            cand[i], cand[j] = a[j], a[i]
            # renormalise just in case numerical drift has occurred
            s = cand.sum()
            if s > 0.0:
                cand = cand * (target / s)
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _fast_objective(cand)
            if sc < best_score:
                best_score, best_seq, a = sc, cand.copy(), cand
                print(f"[swap] improved to {best_score:.6f}")
            swap_trials += 1

    # ------------------------------------------------------------------
    # 6.  Final evaluation with the official function
    # ------------------------------------------------------------------
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] evaluated with evaluate_sequence → score = {final_score:.6f}")

    # Return as plain Python list
    return [float(v) for v in best_seq]
