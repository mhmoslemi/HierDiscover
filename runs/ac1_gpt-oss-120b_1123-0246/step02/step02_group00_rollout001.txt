def _softmax_gradient(seq, beta):
    conv = np.convolve(seq, seq)
    # shift max to avoid overflow
    maxc = conv.max()
    exp_weights = np.exp(beta * (conv - maxc))
    w = exp_weights / exp_weights.sum()
    grad = np.zeros_like(seq)
    n = seq.size
    for k, wk in enumerate(w):
        if wk == 0:
            continue
        grad += wk * _shift(seq, k)
    return grad

def _project_simplex(v):
    # ensures non-negative and sum to 1
    # using algorithm from Duchi et al.
    if v.sum() == 0:
        return np.full_like(v, 1.0/v.size)
    # shift to make sum 1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _gradient_step_multi(seq, eta, top_k):
    conv = np.convolve(seq, seq)
    # find top_k indices with largest conv
    ks = np.argpartition(-conv, top_k)[:top_k]
    # weigh them by conv values
    weights = conv[ks]
    weights = np.maximum(weights,0)
    if weights.sum()>0:
        weights /= weights.sum()
    else:
        weights = np.full_like(weights, 1.0/top_k)
    direction = np.zeros_like(seq)
    for w,k in zip(weights, ks):
        direction += w * _shift(seq,k)
    # move opposite direction: subtract direction (mass from large contributions)
    direction = -direction + np.mean(direction)
    new_seq = seq + eta * direction
    new_seq = np.clip(new_seq,0,1000)
    s = new_seq.sum()
    if s>0:
        new_seq /= s
    else:
        new_seq = seq.copy()
    return new_seq

def _pairwise_move(seq, rng, eps_scale=0.02):
    n = seq.size
    i = rng.integers(0,n)
    j = rng.integers(0,n-1)
    if j>=i:
        j+=1
    eps = eps_scale * rng.random() * min(seq[i], 0.5) # transfer up to proportion
    if eps==0: return seq
    new_seq = seq.copy()
    new_seq[i] -= eps
    new_seq[j] += eps
    # re-normalize (sum unchanged)
    return new_seq

"""
Step‑function optimiser for the convolution bound.

The algorithm combines several techniques that differ from the previous
implementation:

* **FFT‑based fast evaluation** – the convolution and its maximum are
  computed by a single zero‑padded FFT, which is ~O(n log n) and allows
  us to work with longer sequences (up to several ×10⁴) inside the
  time budget.

* **Power‑law seeding with a wide exponent sweep** – we start from a
  family a_i ∝ (i+β)^{-α} with α∈[0.3,0.8] and a few β offsets,
  keeping the best seed.

* **Simulated‑annealing schedule** – a temperature T(t) and a step‑size η(t)
  decay exponentially.  Moves are accepted with a Metropolis rule,
  which lets the search escape local minima early on.

* **Diverse moves**:
  - *Weighted gradient step* – using a soft‑max weighting of the
    largest convolution entries we obtain a direction that pushes mass
    away from the peaks.
  - *Multiplicative block perturbation* – a random contiguous block is
    multiplied by a log‑normal factor.
  - *Pairwise mass transfer* – a small ε is moved from a random index
    to another.
  - *Occasional symmetrisation* and *monotone projection* to keep the
    shape regular.

All updates keep the vector non‑negative and re‑normalise it to sum to 1,
which is harmless because the evaluation function is homogeneous:
score = 2·n·max(a*a) when Σa=1.

The best sequence seen before the time budget expires is returned as a
plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _full_convolution(seq: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a power‑of‑two FFT."""
    n = seq.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fft_seq * fft_seq, m)[: 2 * n - 1]
    return conv


def _score(seq: np.ndarray) -> float:
    """
    Fast surrogate for evaluate_sequence when the vector is already
    normalised to sum‑one.  Returns 2·n·max(a*a).
    """
    n = seq.shape[0]
    max_conv = _full_convolution(seq).max()
    return float(2.0 * n * max_conv)


def _shift(seq: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of (a*a)[k] w.r.t. a.  Returns a vector d where
    d[i] = a_{k-i} for indices inside the valid range, otherwise 0.
    """
    n = seq.size
    out = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1)
        out[idx] = seq[k - idx]
    return out


def _gradient_step_weighted(seq: np.ndarray,
                            eta: float,
                            top_k: int,
                            beta: float,
                            rng: np.random.Generator) -> np.ndarray:
    """
    One projected gradient step.  The gradient is a convex combination of
    the sub‑gradients of the `top_k` largest convolution entries,
    weighted by a soft‑max with temperature `beta`.
    """
    n = seq.size
    conv = _full_convolution(seq)                     # O(n log n)
    # indices of the largest entries (unstable order is fine)
    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(-conv, top_k)[:top_k]
    # soft‑max weights (more focus on the very largest)
    scores = conv[ks]
    # prevent overflow/underflow
    max_score = scores.max()
    weights = np.exp(beta * (scores - max_score))
    weights_sum = weights.sum()
    if weights_sum > 0:
        weights /= weights_sum
    else:
        weights.fill(1.0 / len(ks))

    # average (weighted) shift vector
    direction = np.zeros_like(seq)
    for w, k in zip(weights, ks):
        direction += w * _shift(seq, int(k))

    # move opposite to the direction that inflates the peaks
    direction = -direction + np.mean(direction)

    new_seq = seq + eta * direction
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    s = new_seq.sum()
    if s > 0:
        new_seq /= s
    return new_seq


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal noise on a random contiguous block.
    Occasionally the candidate is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    # block width roughly between 0.5 % and 8 % of n
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    cand[start : start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:               # mild symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _pairwise_transfer(seq: np.ndarray,
                       rng: np.random.Generator,
                       eps_scale: float = 0.02) -> np.ndarray:
    """
    Transfer a small amount of mass from a random index i to j.
    The amount is a fraction (≤ eps_scale) of a_i.
    """
    n = seq.size
    i = rng.integers(0, n)
    # ensure j != i
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    if seq[i] <= 0:
        return seq.copy()
    eps = eps_scale * rng.random() * seq[i]
    new_seq = seq.copy()
    new_seq[i] -= eps
    new_seq[j] += eps
    return new_seq


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Replace the vector by the average with its mirror image."""
    return 0.5 * (seq + seq[::-1])


def _make_monotone(seq: np.ndarray) -> np.ndarray:
    """Project onto the set of non‑increasing sequences (simple sort)."""
    return np.sort(seq)[::-1]
