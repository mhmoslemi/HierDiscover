"""
Hybrid optimiser for the auto‑convolution supremum problem.

The method combines three ingredients:

1. **Power‑law initialisation** – a 1/√(i+β) seed (and a small grid over the
   exponent) provides a solid starting point.

2. **Projected stochastic gradient descent** – each iteration moves probability
   mass away from the indices that currently contribute most to the maximal
   convolution value.  A simple Metropolis rule with an exponential cooling
   schedule allows occasional uphill moves.

3. **Linear‑programming corrective step** – once during the run a linear
   programme

        maximise   Σ g_i
        subject to conv(f,g) ≤ M_f ,   g_i ≥ 0

   (where `M_f = max(conv(f,f))`) is solved with `scipy.optimize.linprog`.
   The optimiser `g` is mixed with the current distribution, which
   typically lowers the peak of `conv(f,f)` further.

All moves respect the non‑negative, ≤ 1000 constraints required by the
external `evaluate_sequence`.  The best sequence found before the time budget
expires is returned as a plain Python list.
"""

import time
import numpy as np

# optional SciPy import – the LP step is used only if SciPy is available
try:
    from scipy.optimize import linprog
except Exception:                     # pragma: no cover
    linprog = None                     # type: ignore


def _evaluate_fast(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` for non‑negative vectors."""
    total = seq.sum()
    if total < 1e-12:                     # guard – should never happen
        return float("inf")
    conv = np.convolve(seq, seq)          # full convolution
    max_b = conv.max()
    n = seq.size
    return float(2.0 * n * max_b / (total * total))


def _shift(seq: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of (seq*seq)[k] w.r.t. ``seq``.
    Returns a vector d such that d_i = seq_{k-i} (zero outside the range).
    """
    n = seq.size
    out = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1)
        out[idx] = seq[k - idx]
    return out


def _gradient_step(seq: np.ndarray, eta: float, top_k: int = 7) -> np.ndarray:
    """
    One projected gradient step.
    The direction is the (negative) average of the sub‑gradients of the
    ``top_k`` largest convolution entries, centred to keep the total mass
    unchanged.  The result is clipped and re‑normalised.
    """
    conv = np.convolve(seq, seq)
    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(-conv, top_k)[:top_k]

    shift_sum = np.zeros_like(seq)
    for k in ks:
        shift_sum += _shift(seq, int(k))
    d = shift_sum / len(ks)               # average contribution

    # zero‑sum direction: move mass away from large entries of d
    direction = -d + np.mean(d)

    new_seq = seq + eta * direction
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    s = new_seq.sum()
    if s > 0:
        new_seq /= s
    else:
        new_seq = seq.copy()
    return new_seq


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    Occasionally the candidate is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:               # mild symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _build_conv_matrix(seq: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that A @ g = conv(seq, g) (full convolution).
    The matrix is dense; for n≈2500 the size is ~100 MiB which is acceptable for
    a few LP solves.
    """
    n = seq.size
    K = 2 * n - 1
    A = np.zeros((K, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = seq[i]
    return A


def _lp_solve(seq: np.ndarray) -> np.ndarray | None:
    """
    Solve the linear programme
        maximise   Σ g_i
        subject to conv(seq, g) ≤ M   (component‑wise)
                    g_i ≥ 0 .
    Returns a normalised feasible vector g (sum = 1) or ``None`` if the
    optimisation fails or SciPy is unavailable.
    """
    if linprog is None:                     # pragma: no cover
        return None

    n = seq.size
    M = np.convolve(seq, seq).max()
    A = _build_conv_matrix(seq)             # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)       # maximise → minimise -Σ g_i
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:                       # pragma: no cover
        return None

    if not res.success:
        return None

    g = res.x
    s = g.sum()
    if s <= 0:
        return None
    return g / s


def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative step heights that (hopefully) yields a low
    value of ``evaluate_sequence``.  The optimiser runs for up to
    ``budget_s`` seconds (a small safety margin is kept for the harness).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Choose a length and an initial distribution.
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be defined globally by the harness.
        base = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore
        if base.ndim != 1 or base.size == 0:
            raise Exception
        n = base.size
        seq = np.clip(base, 0.0, 1000.0)
    except Exception:
        # fallback: a modestly long vector (≈2500) – larger lengths give
        # a finer approximation of the continuous optimum but increase the
        # cost of the occasional LP step.
        n = 2500
        idx = np.arange(1, n + 1, dtype=np.float64)
        seq = 1.0 / np.sqrt(idx)                 # classic 1/√i seed

    seq = np.clip(seq, 0.0, 1000.0)
    seq = seq / seq.sum()                         # normalise to a probability vector

    best_seq = seq.copy()
    best_score = _evaluate_fast(best_seq)

    current_seq = best_seq.copy()
    current_score = best_score

    # ------------------------------------------------------------------
    # 2) Main optimisation loop.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    it = 0
    lp_done = False                              # we perform the LP step only once

    # schedules (exponential cooling)
    eta_start = 0.18
    eta_end   = 3e-5
    temp_start = 2.5e-2
    temp_end   = 1e-6

    block_prob = 0.25
    lp_trigger_fraction = 0.55                  # attempt LP after ~55 % of the budget

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        prog = min(1.0, elapsed / total)

        # step size and temperature schedules
        eta = eta_start * (eta_end / eta_start) ** prog
        temp = temp_start * (temp_end / temp_start) ** prog

        # --------------------------------------------------------------
        # Decide whether to invoke the LP correction (once only)
        # --------------------------------------------------------------
        if (not lp_done and
                elapsed > total * lp_trigger_fraction and
                (deadline - now) > 12.0 and   # leave a few seconds for the solve
                linprog is not None):
            g = _lp_solve(current_seq)
            lp_done = True
            if g is not None:
                # Evaluate a few mixtures of the LP solution with the current state.
                mixture_weights = (0.30, 0.50, 0.70)
                candidates = [g]
                for w in mixture_weights:
                    cand = (1.0 - w) * current_seq + w * g
                    candidates.append(cand)

                for cand in candidates:
                    cand = np.clip(cand, 0.0, 1000.0)
                    cand = cand / cand.sum()
                    cand_score = _evaluate_fast(cand)

                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = cand.copy()
                        print(f"LP iter {it:07d} – new best: {best_score:.6f}")

                    # simple hill‑climbing acceptance for the *current* state
                    if cand_score < current_score or rng.random() < np.exp(
                            -(cand_score - current_score) / max(temp, 1e-12)):
                        current_seq = cand
                        current_score = cand_score
                        break
                # after the LP step we continue with normal gradient moves
                it += 1
                continue
            # If the LP failed we simply continue with the gradient routine.

        # --------------------------------------------------------------
        # 3) Stochastic move: either a block perturbation or a gradient step
        # --------------------------------------------------------------
        if rng.random() < block_prob:
            cand = _make_block_perturbation(current_seq, rng)
        else:
            cand = _gradient_step(current_seq, eta, top_k=10)

        # occasional symmetrisation (helps to keep the shape balanced)
        if rng.random() < 0.12:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand / cand.sum()
        cand_score = _evaluate_fast(cand)

        # --------------------------------------------------------------
        # 4) Acceptance (Metropolis style) and bookkeeping
        # --------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - current_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            current_seq = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"iter {it:07d} – new best: {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 5) Final sanity checks and return
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s

    print(f"finished after {it} iterations, best score={best_score:.6f}")
    return best_seq.tolist()
