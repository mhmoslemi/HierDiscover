def solve_lp_for_sum(f, M):
    n = len(f)
    rows = np.arange(n).repeat(n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    data = np.repeat(f, n)
    A = coo_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Convert to csr for better performance
    A_csr = A.tocsr()
    # constraints
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(b)
    res = linprog(c, A_ub=A_csr, b_ub=b_ub, bounds=(0, None), method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    b = res.x
    sum_b = -res.fun
    return b, sum_b

"""
Search for a low‑score non‑negative step function.

The algorithm works in three phases:

1. **Initialisation** – a sequence of length ≤ 3000 is built.  If the
   global variable ``height_sequence_1`` exists it is trimmed, otherwise an
   exponential shape a_i = c·α^i is searched (grid over α) to obtain a
   reasonable starting point.

2. **Linear‑programming mass re‑allocation** – for the current vector
   ``a`` we compute M = max (a∗a).  Solving the LP
        maximise  Σ b_i
        subject  (a∗b)_k ≤ M  (for all k)
        b_i ≥ 0
   yields a vector ``b`` that can carry *more* total mass without breaking
   the bound M.  After normalising ``b`` to the required ℓ₁‑norm we
   interpolate between ``a`` and the normalised ``b`` (several mixing
   factors are tried).  Any improvement is kept and the process is repeated
   a few times (the LP is the heaviest step, so we limit it to a handful of
   iterations).

3. **Soft‑max gradient descent polishing** – starting from the best
   sequence obtained above we run the annealed soft‑max gradient method
   (the same idea as the previous baseline) for the remaining time.
   Random perturbations and occasional restarts prevent stagnation.

The total running time is limited by the ``budget_s`` argument (default
950 s); the best sequence discovered before the deadline is returned.
"""

# ----------------------------------------------------------------------


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a power‑of‑two FFT (real‑valued)."""
    n = x.size + y.size - 1
    N = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv


def _grad_and_maxb(a: np.ndarray, beta: float):
    """Gradient of the soft‑max approximation of max_k (a∗a)_k and the exact max."""
    n = a.size
    b = _fft_convolve(a, a)
    maxb = b.max()

    # soft‑max weights over all lags – improve numerical stability
    exp_term = np.exp(beta * (b - maxb))
    w = exp_term / exp_term.sum()

    # gradient of Σ_k w_k (a∗a)_k = 2 Σ_k w_k a_{k‑i}
    w_rev = w[::-1]
    conv_rev = _fft_convolve(a, w_rev)                # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)                 # indices n‑1 … 2n‑2
    grad = 2.0 * conv_rev[idx]

    # project onto the hyper‑plane Σ a_i = const (zero‑sum gradient)
    grad -= grad.mean()
    return grad, maxb


def _exponential_initial(n: int, target_sum: float):
    """Search a simple exponential (geometric) shape for a good start."""
    best_a = None
    best_score = np.inf

    # coarse grid in (0,1); we avoid α extremely close to 1 because of numerical issues
    alphas = np.concatenate(
        [np.linspace(0.01, 0.99, 200), np.logspace(-3, -0.01, 100)]
    )
    indices = np.arange(n, dtype=np.float64)

    for alpha in alphas:
        if not (0.0 < alpha < 1.0):
            continue
        log_alpha = np.log(alpha)
        # denominator 1-α^n may under‑flow for tiny α; use exp(log) formulation
        denom = 1.0 - np.exp(n * log_alpha)
        if denom <= 0.0:
            continue
        c = target_sum * (1.0 - alpha) / denom
        a = c * np.exp(indices * log_alpha)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)
        maxb = _fft_convolve(a, a).max()
        if maxb < best_score:
            best_score = maxb
            best_a = a.copy()
    return best_a, best_score

            print(f"[GD {gd_iter:6d}] restart due to stagnation")

        gd_iter += 1

    # ------------------------------------------------------------------
    # 4. Final clean‑up and return
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    best_seq = best_seq * (target_sum / best_seq.sum())
    final_score = float(_fft_convolve(best_seq, best_seq).max())
    print(f"[done] final score={final_score:.6f}, length={n}")
    return [float(v) for v in best_seq.tolist()]
