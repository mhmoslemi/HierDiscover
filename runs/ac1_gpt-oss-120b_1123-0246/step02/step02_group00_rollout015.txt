Propose a step‑function that minimizes ``evaluate_sequence``.

The algorithm proceeds in three stages:

1. **Diverse seeding** – a quick grid search over power‑law and
   exponential families creates a handful of candidates; the best of
   them (according to the exact evaluator) is kept.

2. **Linear‑programming refinement** – following the idea from the
   statement, we repeatedly solve the LP
        maximise   Σ g_j
        subject to g_j ≥ 0  and  (f * g)_k ≤ M   ∀k
   where *M* is the current maximal convolution value of the
   candidate *f*.  The solution *g* carries extra mass that does not
   increase the peak.  After rescaling *g* to the same total mass as *f*
   we mix the two vectors (convex combination with a decreasing weight)
   and keep the improvement if the objective drops.

3. **Gradient fallback & restart** – if the LP solver is unavailable or
   fails, a cheap projected‑gradient step (identical to the previous
   baseline) is used.  Periodic random perturbations avoid stagnation.

All vectors are kept non‑negative and ≤ 1000, and finally normalised to
sum = 1 (the evaluation function is homogeneous, so any scaling is

def _evaluate_fast(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` for non‑negative vectors."""
    total = seq.sum()
    if total <= 0.0:
        return float("inf")
    conv = np.convolve(seq, seq)          # full convolution
    max_b = conv.max()
    n = seq.size
    return float(2.0 * n * max_b / (total * total))


def _build_conv_indices(n: int):
    """
    Pre‑compute the row/col index arrays needed to build the sparse
    convolution matrix A such that A @ g = f * g (full convolution).
    """
    # row index = i + j  (i iterates over f, j over g)
    row = (np.arange(n, dtype=np.int32)[:, None] +
           np.arange(n, dtype=np.int32)[None, :]).ravel()
    # column index = j
    col = np.repeat(np.arange(n, dtype=np.int32), n)
    return row, col


def _conv_matrix(f: np.ndarray, row_idx, col_idx):
    """
    Construct the sparse matrix A (shape (2n‑1, n)) representing the
    linear map g ↦ f * g (full convolution).  ``row_idx`` and
    ``col_idx`` are the constant index arrays produced by
    ``_build_conv_indices``.
    """
    data = np.repeat(f, f.size)               # each coefficient of f repeats n times
    n = f.size
    return sp.csr_matrix((data, (row_idx, col_idx)),
                         shape=(2 * n - 1, n))


def _gradient_step(seq: np.ndarray, eta: float, top_k: int = 7) -> np.ndarray:
    """
    Cheap projected gradient step used as a fallback when the LP is not
    available.  It pushes mass away from the indices that currently take
    part in the largest convolution entries.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    ks = (np.argpartition(-conv, top_k)[:top_k]
          if top_k < conv.size else np.arange(conv.size))

    shift_sum = np.zeros_like(seq)
    for k in ks:
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        if start <= end:
            idxs = np.arange(start, end + 1)
            shift_sum[idxs] += seq[k - idxs]

    d = shift_sum / max(1, len(ks))
    direction = -d + np.mean(d)            # zero‑sum direction
    new_seq = seq + eta * direction
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    s = new_seq.sum()
    if s > 0:
        new_seq /= s
    return new_seq

    best_seq_overall = best_seq.copy()
    best_score_overall = best_score

    # ------------------------------------------------------------------
    # Pre‑compute the index arrays for the sparse convolution matrix.
    # ------------------------------------------------------------------
    if SCIPY_AVAILABLE:
        row_idx_const, col_idx_const = _build_conv_indices(n)

    # mixing weight schedule (large at the beginning, shrinking)
    t_start = 0.6
    t_end = 0.05

    # simple stagnation detector
    it = 0
    no_improve = 0
    MAX_NO_IMPROVE = 30

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        prog = min(1.0, elapsed / (deadline - start_time))

        # current maximal convolution value
        conv = np.convolve(current_seq, current_seq)
        M = conv.max()

        cand = None

        # --------------------------------------------------------------
        # 2) LP improvement step (if SciPy is available)
        # --------------------------------------------------------------
        if SCIPY_AVAILABLE:
            try:
                A = _conv_matrix(current_seq, row_idx_const, col_idx_const)
                b = np.full(2 * n - 1, M, dtype=np.float64)
                c = -np.ones(n, dtype=np.float64)          # maximise Σ g_j
                bounds = [(0, None)] * n

                res = linprog(c, A_ub=A, b_ub=b,
                              bounds=bounds,
                              method='highs',
                              options={'presolve': True})

                if res.success:
                    g0 = res.x
                    sum_g0 = g0.sum()
                    if sum_g0 > 0.0:
                        # rescale g to keep the total mass equal to sum_target
                        g = g0 * (sum_target / sum_g0)
                        # convex mixing – weight shrinks as optimisation proceeds
                        t = t_start * (t_end / t_start) ** prog
                        cand = (1.0 - t) * current_seq + t * g
            except Exception:                 # pragma: no cover
                cand = None

        # --------------------------------------------------------------
        # 3) Gradient fallback when the LP is not usable.
        # --------------------------------------------------------------
        if cand is None:
            cand = _gradient_step(current_seq,
                                  eta=0.02 * (0.5 ** (it // 4000)),
                                  top_k=7)
            cand = cand * (sum_target / cand.sum())

        # --------------------------------------------------------------
        # 4) Evaluate and possibly accept the new candidate.
        # --------------------------------------------------------------
        cand_score = _evaluate_fast(cand)

        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # occasional uphill move to increase diversity
            if rng.random() < 0.005:
                accept = True

        if accept:
            current_seq = cand
            current_score = cand_score
            if cand_score < best_score_overall:
                best_score_overall = cand_score
                best_seq_overall = cand.copy()
                print(f"iter {it:06d} – new best: {best_score_overall:.6f}")
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 5) Simple restart when stagnation is detected.
        # --------------------------------------------------------------
        if no_improve >= MAX_NO_IMPROVE:
            perturb = 0.9 + 0.2 * rng.random(n)
            current_seq = best_seq_overall * perturb
            current_seq = np.clip(current_seq, 0.0, 1000.0)
            current_seq = current_seq * (sum_target / current_seq.sum())
            current_score = _evaluate_fast(current_seq)
            no_improve = 0
            print(f"restart at iter {it}")

        it += 1

    # ----------------------------------------------------------------------
    # Final sanity checks – the external evaluator will clip again.
    # ----------------------------------------------------------------------
    final_seq = np.clip(best_seq_overall, 0.0, 1000.0)
    final_seq = final_seq / final_seq.sum()          # normalise to sum = 1
    print(f"finished after {it} iterations, best score={best_score_overall:.6f}")
    return final_seq.tolist()
