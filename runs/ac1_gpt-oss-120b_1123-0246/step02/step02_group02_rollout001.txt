"""
Propose a low‑score step‑function using multi‑restart
soft‑max gradient descent with momentum, simplex projection,
and a final simulated‑annealing refinement.

The algorithm:
1. Choose a length n (default = length of the previously found sequence,
   or 6000) and fix the total mass S = sqrt(2*n).  After each update the
   vector is projected onto the simplex {x>=0, sum(x)=S}.
2. Build several diverse starting points (uniform, random, U‑shaped,
   Gaussian, and the previous best if available).
3. For each start run a gradient descent on the smooth approximation
   of max(conv(a,a)).  The gradient is obtained via FFT‑based correlation.
   Momentum and a linearly decaying step size are used.  The temperature
   β of the soft‑max is increased from a small value to a large one
   (up to 30) so that the optimisation first moves smoothly then
   concentrates on the true maximum.
4. Keep the best sequence seen (by the cheap FFT max) and occasionally
   evaluate the exact objective with evaluate_sequence().
5. After all gradient runs finish, run a short simulated‑annealing
   local search around the current best vector to escape possible
   plateaus.
6. Return the best vector (as a plain Python list).  All values are
   non‑negative and respect the 0‑1000 per‑element bound enforced by
   evaluate_sequence().

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helper functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a * a using real FFTs."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.size - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Correlation (w ⋆ a) = convolution(w[::-1], a) via FFT.
    Returns an array of length len(w)+len(a)-1.
    """
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, fft_len)
    fa = np.fft.rfft(a, fft_len)
    corr = np.fft.irfft(fw * fa, fft_len)
    return corr[: w.size + a.size - 1]


# ----------------------------------------------------------------------
# Simplex projection (onto {x >= 0, sum(x)=z})
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x >= 0, sum(x)=z}.
    """
    if z < 0:
        raise ValueError("Simplex target must be non‑negative")
    n = v.shape[0]
    # shift to ensure positivity
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

    fft_len: int,
    max_iters: int,
    beta_start: float,
    beta_end: float,
    step0: float,
    momentum: float,
    rng: np.random.Generator,
    deadline: float,
    global_best_score: float,
    global_best_seq: np.ndarray,
):
    """
    Perform one descent run.  Returns (best_seq, best_score, total_iters).
    """
    a = a0.copy()
    a = _project_simplex(a, target_sum)

    # velocity for momentum
    v = np.zeros_like(a)

    best_local_seq = a.copy()
    best_local_max = _self_convolution_fft(a, fft_len).max()
    # we keep a separate counter for total iterations across all runs
    iters = 0

    # pre‑compute some constants
    L = 2 * n - 1  # length of self‑convolution

    while iters < max_iters and time.monotonic() < deadline:
        # ----- schedule -----
        beta = beta_start + (beta_end - beta_start) * (iters / max_iters)
        step = step0 * max(0.0, 1.0 - iters / max_iters)

        # ----- self‑convolution and soft‑max weights -----
        b = _self_convolution_fft(a, fft_len)
        cur_max = b.max()
        # stable soft‑max
        shifted = b - cur_max
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.zeros_like(w)
            w_norm[np.argmax(b)] = 1.0
        else:
            w_norm = w / w_sum

        # ----- gradient of smoothed max -----
        corr = _correlation_fft(w_norm, a, fft_len)
        grad = 2.0 * beta * corr[L - 1 - np.arange(n)]

        # normalise gradient direction
        norm = np.linalg.norm(grad)
        if norm > 0:
            direction = grad / norm
        else:
            direction = np.zeros_like(grad)

        # ----- momentum update -----
        v = momentum * v - step * direction
        a_next = a + v
        a_next = _project_simplex(a_next, target_sum)

        # ----- acceptance (always keep the better max) -----
        b_next = _self_convolution_fft(a_next, fft_len)
        next_max = b_next.max()
        if next_max <= cur_max:
            a = a_next
            cur_max = next_max
            # occasional exact evaluation for logging
            if iters % 200 == 0:
                exact = float(evaluate_sequence(a.tolist()))
                if exact < global_best_score:
                    global_best_score = exact
                    global_best_seq = a.copy()
                    print(f"[{iters:6d}] new global exact score = {global_best_score:.6f}")
            # update local best
            if next_max < best_local_max:
                best_local_max = next_max
                best_local_seq = a.copy()
        else:
            # rare stochastic acceptance to escape plateaus
            if rng.random() < 0.001:
                a = a_next
                cur_max = next_max

        # ----- diversification: small multiplicative perturbation -----
        if iters % 1000 == 0 and iters > 0:
            mult = np.exp(rng.normal(0.0, 0.025, size=n))
            a = _project_simplex(a * mult, target_sum)

        iters += 1

    return best_local_seq, best_local_max, iters, global_best_score, global_best_seq


# ----------------------------------------------------------------------
# Simulated‑annealing style local refinement
# ----------------------------------------------------------------------


def _local_refinement(
    a0: np.ndarray,
    n: int,
    target_sum: float,
    fft_len: int,
    steps: int,
    temp_start: float,
    temp_end: float,
    rng: np.random.Generator,
    deadline: float,
    global_best_score: float,
    global_best_seq: np.ndarray,
):
    """
    Simple pair‑wise mass‑transfer perturbation with a temperature schedule.
    Returns (best_seq, best_score).
    """
    a = a0.copy()
    cur_max = _self_convolution_fft(a, fft_len).max()
    best_seq = a.copy()
    best_score = cur_max

    for t in range(steps):
        if time.monotonic() >= deadline:
            break
        # linear temperature schedule
        temp = temp_start + (temp_end - temp_start) * (t / steps)

        # propose a transfer from i -> j
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] == 0.0:
            continue
        # take a random fraction (up to 5%) of the source mass
        delta = a[i] * rng.random() * 0.05
        a_prop = a.copy()
        a_prop[i] -= delta
        a_prop[j] += delta

        # projection not needed because sum unchanged and we ensure non‑negative
        prop_max = _self_convolution_fft(a_prop, fft_len).max()

        if prop_max < cur_max or rng.random() < np.exp(-(prop_max - cur_max) / max(temp, 1e-12)):
            a = a_prop
            cur_max = prop_max
            if cur_max < best_score:
                best_score = cur_max
                best_seq = a.copy()
                # occasional exact check
                if t % 200 == 0:
                    exact = float(evaluate_sequence(a.tolist()))
                    if exact < global_best_score:
                        global_best_score = exact
                        global_best_seq = a.copy()
                        print(f"[refine {t:6d}] new global exact score = {global_best_score:.6f}")

    return best_seq, best_score, global_best_score, global_best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 950, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    Returns a list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # safety margin for final printing
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # 1. Choose length n and normalization target
    # --------------------------------------------------------------
    try:
        n = int(np.asarray(height_sequence_1).size)
    except Exception:
        n = 6000
    n = int(np.clip(n, 2000, 9000))

    target_sum = np.sqrt(2.0 * n)          # sum we keep constant during optimisation
    fft_len = _next_pow2(3 * n - 2)        # enough for correlation (w + a)

    # --------------------------------------------------------------
    # 2. Build a collection of starting points
    # --------------------------------------------------------------
    starts = []

    # uniform
    starts.append(np.full(n, target_sum / n))

    # random positive vectors
    for _ in range(3):
        starts.append(rng.random(n) * target_sum)

    # U‑shaped (powers of index)
    for p in rng.uniform(-4.0, -0.2, size=4):
        idx = np.arange(n, dtype=float)
        a = (idx + 1) ** p + (n - idx) ** p
        a = a - a.min() + 1e-8
        a = a / a.sum() * target_sum
        starts.append(a)

    # Gaussian‑like bump (center heavy, could help exploring other shapes)
    for sigma in rng.uniform(0.05, 0.25, size=3):
        x = np.linspace(-0.5, 0.5, n)
        a = np.exp(-0.5 * (x / sigma) ** 2)
        a = a / a.sum() * target_sum
        starts.append(a)

    # reuse previous best if available (interpolated to current n)
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            if prev.size != n:
                old_x = np.linspace(0.0, 1.0, prev.size)
                new_x = np.linspace(0.0, 1.0, n)
                prev = np.interp(new_x, old_x, prev)
            prev = np.clip(prev, 0.0, 1000.0)
            if prev.sum() > 0:
                prev = _project_simplex(prev, target_sum)
                starts.append(prev)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Run gradient‑descent runs on the various starts
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    total_iters = 0

    # hyper‑parameter ranges (will be sampled per start)
    beta_end_vals = [8, 15, 25, 30]
    step0_vals = [0.02, 0.04, 0.06]  # will be multiplied by target_sum inside loop
    momentum_vals = [0.85, 0.9, 0.95]

    for start_id, a_start in enumerate(starts):
        if time.monotonic() >= deadline:
            break

        # random choice of hyper‑parameters for this run
        beta_end = float(rng.choice(beta_end_vals))
        step0 = float(rng.choice(step0_vals)) * target_sum
        momentum = float(rng.choice(momentum_vals))
        max_iters = int(4000 + rng.integers(0, 2000))

        # evaluate the raw starting point (exact)
        init_score = float(evaluate_sequence(a_start.tolist()))
        if init_score < best_score:
            best_score = init_score
            best_seq = a_start.copy()
            print(f"[start {start_id}] exact initial score = {best_score:.6f}")

        # run descent
        seq, local_max, it_used, best_score, best_seq = _run_descent(
            a_start,
            n,
            target_sum,
            fft_len,
            max_iters,
            beta_start=0.5,
            beta_end=beta_end,
            step0=step0,
            momentum=momentum,
            rng=rng,
            deadline=deadline,
            global_best_score=best_score,
            global_best_seq=best_seq,
        )
        total_iters += it_used

        # try a cheap update of the global best based on fast max
        if local_max < best_score:
            best_score = local_max
            best_seq = seq.copy()
            print(f"[start {start_id}] improved fast max = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Final local refinement (simulated annealing style)
    # --------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        refine_steps = 8000
        temp_start = 1.0
        temp_end = 0.001
        seq_ref, max_ref, best_score, best_seq = _local_refinement(
            best_seq,
            n,
            target_sum,
            fft_len,
            steps=refine_steps,
            temp_start=temp_start,
            temp_end=temp_end,
            rng=rng,
            deadline=deadline,
            global_best_score=best_score,
            global_best_seq=best_seq,
        )
        if max_ref < best_score:
            best_score = max_ref
            best_seq = seq_ref.copy()
            print(f"[refine] final refined fast max = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Final exact evaluation before returning
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] exact score = {final_score:.6f}")
    return [float(v) for v in best_seq]
