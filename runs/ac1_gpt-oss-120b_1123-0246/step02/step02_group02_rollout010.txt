def self_convolve_fft(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[:2*a.size-1]

def correlation_fft(w, a, fft_len):
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, fft_len)
    fa = np.fft.rfft(a, fft_len)
    corr = np.fft.irfft(fw * fa, fft_len)
    return corr[:w.size + a.size - 1]

def gradient_softmax(a, beta, fft_len):
    b = self_convolve_fft(a, fft_len)
    max_b = b.max()
    shifted = b - max_b
    # numeric stability:
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    w_norm = w / w_sum
    # correlation:
    corr = correlation_fft(w_norm, a, fft_len)
    L = 2 * len(a) - 1
    grad = 2.0 * beta * corr[L - 1 - np.arange(len(a))]
    return grad, max_b, b

"""
Propose a low‑score step‑function.

The algorithm consists of three stages:

1. **Initialisation** – build a small collection of diverse seed vectors
   (uniform, random, interpolated previous best, power‑law, exponential).

2. **Smooth‑max Adam optimisation** – minimise a smooth approximation of the
   ∞‑norm of the self‑convolution using an Adam optimiser.
   The soft‑max temperature β is increased from a modest value up to a
   large value (≈30) so that the gradient focuses on the true maximum.
   The sum of the coefficients is kept constant (√(2 n)) so that the
   evaluation score equals the maximal convolution value.

3. **Simulated‑annealing fine‑tuning** – starting from the Adam result,
   perform many cheap Metropolis moves that transfer a small amount of mass
   between two positions while preserving the total sum.  The max‑convolution
   is evaluated with FFTs; moves that improve the max are always kept, and
   occasional uphill moves are accepted according to a decreasing temperature.

The best sequence found across all seeds is finally evaluated with the exact
`evaluate_sequence` function and returned.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using real FFTs."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.size - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full correlation (w ⋆ a) using FFTs."""
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, fft_len)
    fa = np.fft.rfft(a, fft_len)
    corr = np.fft.irfft(fw * fa, fft_len)
    return corr[: w.size + a.size - 1]


def _max_conv_fft(a: np.ndarray, fft_len: int) -> float:
    """Return max_{k} (a*a)[k] using FFTs."""
    return float(_self_conv_fft(a, fft_len).max())


def _grad_softmax(a: np.ndarray, beta: float, fft_len: int):
    """
    Gradient of the smooth‑max (log‑sum‑exp) of the self‑convolution.
    Returns (grad, max_value).
    """
    # self‑convolution
    b = _self_conv_fft(a, fft_len)
    max_b = b.max()
    # numerically stable soft‑max weights
    shifted = b - max_b
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        w_norm = np.zeros_like(w)
        w_norm[np.argmax(b)] = 1.0
    else:
        w_norm = w / w_sum
    # correlation w_norm ⋆ a
    corr = _correlation_fft(w_norm, a, fft_len)
    L = b.size                      # = 2·n‑1
    grad = 2.0 * beta * corr[L - 1 - np.arange(a.size)]
    return grad, max_b


def _adam_optimize(
    a0: np.ndarray,
    target_sum: float,
    max_iters: int,
    lr0: float,
    beta_start: float,
    beta_end: float,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
):
    """
    Adam optimiser on the smooth‑max objective.
    Keeps the L1‑sum exactly = target_sum after each step.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_max = _max_conv_fft(a, fft_len)

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        # exponential schedule for β
        beta = beta_start * (beta_end / beta_start) ** (it / max_iters)

        grad, cur_max = _grad_softmax(a, beta, fft_len)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # enforce non‑negativity and re‑normalise sum
        np.maximum(a, 0.0, out=a)
        s = a.sum()
        if s > 0.0:
            a *= target_sum / s

        # cheap max‑convolution check
        cur_max = _max_conv_fft(a, fft_len)
        if cur_max < best_max:
            best_max = cur_max
            best_seq = a.copy()

        # occasional exact evaluation (just for logging)
        if it % 300 == 0:
            exact = float(evaluate_sequence(a.tolist()))
            if exact < best_max:
                best_max = cur_max
                best_seq = a.copy()
                print(f"[Adam it {it:5d}] exact score = {exact:.6f}")

    return best_seq, best_max


def _simulated_annealing(

    step_scale = target_sum * 0.01

    it = 0
    while time.monotonic() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue

        delta = rng.normal(scale=step_scale)
        # keep a[i] non‑negative
        if delta > a[i]:
            delta = a[i] - 1e-12

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_max = _max_conv_fft(cand, fft_len)

        # Metropolis acceptance
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-12)):
            a = cand
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_seq = a.copy()
                if it % 500 == 0:
                    exact = float(evaluate_sequence(a.tolist()))
                    print(f"[Anneal it {it:5d}] new best exact = {exact:.6f}")

        # temperature decay
        temp *= cooling
        if it % 2000 == 0:
            step_scale *= 0.9  # gradually shrink proposal size

    return best_seq, best_max


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs) -> List[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The routine respects the time budget and returns the
    best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a couple of seconds as safety margin
    deadline = start_time + max(0.0, float(budget_s) - 4.0)

    # ------------------------------------------------------------------
    # Choose a working length
    # ------------------------------------------------------------------
    prev_global = globals().get("height_sequence_1")
    if isinstance(prev_global, (list, np.ndarray)):
        prev_global = np.asarray(prev_global, dtype=float)
    else:
        prev_global = None

    if prev_global is not None:
        n0 = prev_global.size
        n = int(np.clip(n0, 2000, 8000))
    else:
        n = 5000                     # default length

    target_sum = np.sqrt(2.0 * n)    # normalisation used in the original paper
    fft_len = _next_pow2(4 * n)      # ample space for all FFT‑based ops

    # ------------------------------------------------------------------
    # Build a collection of start vectors
    # ------------------------------------------------------------------
    starts = []

    # (a) uniform
    starts.append(np.full(n, target_sum / n))

    # (b) random positive
    starts.append(rng.random(n) * target_sum)

    # (c) previous best (interpolated if necessary)
    if prev_global is not None:
        if prev_global.size != n:
            old_x = np.linspace(0.0, 1.0, prev_global.size)
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, prev_global)
        else:
            interp = prev_global.copy()
        s = interp.sum()
        if s > 0.0:
            interp *= target_sum / s
        starts.append(interp)

    # (d) power‑law decreasing shapes
    for _ in range(3):
        p = rng.uniform(-3.5, -0.3)
        a = (np.arange(n, dtype=float) + 1) ** p
        a = a - a.min() + 1e-8
        a = a / a.sum() * target_sum
        starts.append(a)

    # (e) exponential decreasing shapes
    for _ in range(3):
        lam = rng.uniform(0.001, 0.02)
        a = np.exp(-lam * np.arange(n, dtype=float))
        a = a / a.sum() * target_sum
        starts.append(a)

    # ------------------------------------------------------------------
    # Optimisation loop over the different starts
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    for sid, seed_vec in enumerate(starts):
        if time.monotonic() >= deadline:
            break

        a0 = seed_vec.copy()
        np.maximum(a0, 0.0, out=a0)
        s0 = a0.sum()
        if s0 > 0.0:
            a0 *= target_sum / s0

        # evaluate the raw start (exact)
        init_score = float(evaluate_sequence(a0.tolist()))
        if init_score < best_score:
            best_score = init_score
            best_seq = a0.copy()
            print(f"[seed {sid}] initial score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 1. Adam optimisation on the smooth‑max objective
        # --------------------------------------------------------------
        max_iters_adam = 2500
        lr0 = 0.02 * target_sum
        beta_start = 0.5
        beta_end = 30.0
        seq_opt, max_opt = _adam_optimize(
            a0,
            target_sum,
            max_iters_adam,
            lr0,
            beta_start,
            beta_end,
            fft_len,
            deadline,
            rng,
        )

        exact_opt = float(evaluate_sequence(seq_opt.tolist()))
        if exact_opt < best_score:
            best_score = exact_opt
            best_seq = seq_opt.copy()
            print(f"[seed {sid}] after Adam best = {best_score:.6f}")

        # --------------------------------------------------------------
        # 2. Simulated‑annealing fine‑tuning (if time permits)
        # --------------------------------------------------------------
        if time.monotonic() >= deadline:
            continue
        # allocate roughly 60 % of the remaining budget to annealing
        remaining = deadline - time.monotonic()
        anneal_deadline = time.monotonic() + remaining * 0.6

        seq_fine, max_fine = _simulated_annealing(
            seq_opt,
            target_sum,
            max_opt,
            fft_len,
            anneal_deadline,
            rng,
        )
        exact_fine = float(evaluate_sequence(seq_fine.tolist()))
        if exact_fine < best_score:
            best_score = exact_fine
            best_seq = seq_fine.copy()
            print(f"[seed {sid}] after annealing best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation (fallback if nothing better found)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] best exact score = {final_score:.6f}")

    # Return as plain list of Python floats
    return [float(v) for v in best_seq]
