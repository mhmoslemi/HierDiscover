def _beta_sequence_general(n, al, ar):
    p = (np.arange(n) + 0.5) / n
    w = np.power(p, al - 1) * np.power(1 - p, ar - 1)
    target_sum = sqrt(2 * n)
    w_sum = w.sum()
    a = w * (target_sum / w_sum)
    a = np.clip(a, 0.0, 1000.0)
    return a

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm proceeds in several stages:

1. **Initialisation** – take the global `height_sequence_1` (if it exists)
   as a starting candidate.

2. **Coarse grid search** – evaluate symmetric Beta(α,α) shapes for a
   range of lengths `n` and shape parameters `α`.  This quickly finds a
   promising length and shape.

3. **Golden‑section refinement** – fine‑tune the best α for the chosen
   length.

4. **Asymmetric Beta exploration** – random search over two‑parameter
   Beta(αₗ,αᵣ) shapes to break symmetry and possibly lower the peak.

5. **Sub‑gradient descent** – minimise the convex objective
   `max(a*a)` directly.  At each step we compute the active convolution
   indices (those attaining the maximum) and move the sequence opposite
   to the sub‑gradient while keeping non‑negativity and the L¹ norm
   (≈√(2n)) fixed.

6. **Random mass‑transfer hill‑climbing** – tiny stochastic changes of
   the sequence (with a simple simulated‑annealing acceptance rule) to
   polish the result.

7. **Final evaluation** – the best sequence found is passed through the
   provided `evaluate_sequence` (which applies the official clipping) and
   returned as a plain Python list.

The whole procedure respects the supplied time budget (default 500 s,
leaving a safety margin for the final call).  All helper functions are
top‑level to avoid closure capture.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast surrogate objective (identical to evaluate_sequence's internal logic)
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray) -> float:
    """Return 2·n·max(b)/sum(a)^2 where b = a*a (linear convolution)."""
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    size = 2 * n - 1
    # next power of two for speed / numerical stability
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Beta‑shaped sequences (symmetric and general)
# ----------------------------------------------------------------------
def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) shape, scaled so that Σa = √(2n)."""
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, alpha - 1.0)
    target = math.sqrt(2.0 * n)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    a = w * (target / w_sum)
    a = np.clip(a, 0.0, 1000.0)
    return a


def _beta_sequence_general(n: int, alpha_l: float, alpha_r: float) -> np.ndarray:
    """Beta(αₗ,αᵣ) shape (asymmetric), scaled to Σa = √(2n)."""
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(p, alpha_l - 1.0) * np.power(1.0 - p, alpha_r - 1.0)
    target = math.sqrt(2.0 * n)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    a = w * (target / w_sum)
    a = np.clip(a, 0.0, 1000.0)
    return a


# ----------------------------------------------------------------------
# Sub‑gradient optimisation of max(a*a)
# ----------------------------------------------------------------------
def _subgradient_optimize(
    start_seq: np.ndarray,
    deadline: float,
    best_score: float,
    best_seq: np.ndarray,
) -> tuple[float, np.ndarray]:
    """Perform a simple sub‑gradient descent on max‑convolution."""
    a = start_seq.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)
    eps_active = 1e-12
    step0 = 0.5  # initial step magnitude
    it = 0

    while time.time() < deadline and it < 3000:
        # linear convolution via FFT
        size = 2 * n - 1
        m = 1 << ((size - 1).bit_length())
        A = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(A * A, n=m)[:size]
        max_b = conv.max()
        # active indices (where max is attained, within tolerance)
        active = np.where(conv >= max_b - eps_active)[0]

        # compute sub‑gradient: g_j = 2 * Σ_{k∈active} a_{k-j}
        grad = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]

        # diminishing step size
        step = step0 / math.sqrt(it + 1.0)

        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)

        # renormalise to the prescribed L1 norm
        s = a_new.sum()
        if s > 0.0:
            a_new *= (target_sum / s)

        sc = _fast_objective(a_new)
        if sc < best_score:
            best_score = sc
            best_seq = a_new.copy()

        a = a_new
        it += 1

    return best_score, best_seq


# ----------------------------------------------------------------------
# Tiny random perturbations (hill‑climbing / simulated annealing)
# ----------------------------------------------------------------------
def _random_perturbation(
    seq: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    best_score: float,
    best_seq: np.ndarray,
) -> tuple[float, np.ndarray]:
    """Move tiny amounts of mass between random positions."""
    a = seq.copy()
    n = a.size
    target_sum = math.sqrt(2.0 * n)
    # typical step magnitude (adjustable)
    step_scale = target_sum / (20.0 * n)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.random() * step_scale
        if a[i] < delta:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑normalise (keeps total mass essentially unchanged)
        s = cand.sum()
        if s > 0.0:
            cand *= (target_sum / s)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = _fast_objective(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            a = cand.copy()
        else:
            # occasional uphill move (simulated annealing style)
            if rng.random() < 0.001:
                a = cand.copy()
    return best_score, best_seq


# ----------------------------------------------------------------------
# Main entry point required by the runner
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    The procedure follows the multi‑stage strategy described in the
    module docstring.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + max(0.0, float(budget_s) - 1.0)

    best_seq = None
    best_score = np.inf
    best_n = None
    best_alpha = None  # for symmetric case

    # ------------------------------------------------------------------
    # 0.  Try the existing global sequence if it exists (fallback)
    # ------------------------------------------------------------------
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        seq0 = np.clip(seq0, 0.0, 1000.0)
        if seq0.ndim == 1 and seq0.size > 0:
            sc0 = _fast_objective(seq0)
            if sc0 < best_score:
                best_score = sc0
                best_seq = seq0.copy()
                best_n = seq0.size
                best_alpha = None
                print(f"[init] global height_sequence_1 score = {sc0:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Coarse grid over lengths and symmetric Beta shape parameters
    # ------------------------------------------------------------------
    n_candidates = [
        2000, 3000, 4000, 5000, 6000, 8000, 10000,
        12000, 16000, 20000, 25000, 32768, 50000, 65536
    ]
    # α values – coarse sweep covering the most promising region
    alpha_grid = np.array([0.20, 0.30, 0.40, 0.45, 0.50, 0.55, 0.60, 0.70, 0.80])

    for n in n_candidates:
        if time.time() >= deadline:
            break
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            a = _beta_sequence(n, alpha)
            sc = _fast_objective(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a.copy()
                best_n = n
                best_alpha = alpha
                print(f"[grid] n={n:6d} α={alpha:5.3f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Refine the best α (symmetric) for the selected n by golden‑section
    # ------------------------------------------------------------------
    if best_seq is not None and best_alpha is not None and best_n is not None:
        lo = max(0.05, best_alpha - 0.30)
        hi = min(1.5, best_alpha + 0.30)
        phi = (1.0 + 5 ** 0.5) / 2.0
        c = hi - (hi - lo) / phi
        d = lo + (hi - lo) / phi
        a_c = _beta_sequence(best_n, c)
        f_c = _fast_objective(a_c)
        a_d = _beta_sequence(best_n, d)
        f_d = _fast_objective(a_d)

        while time.time() < deadline and (hi - lo) > 1e-5:
            if f_c < f_d:
                hi, a_d, f_d = d, a_c, f_c
                d = c
                c = hi - (hi - lo) / phi
                a_c = _beta_sequence(best_n, c)
                f_c = _fast_objective(a_c)
            else:
                lo, a_c, f_c = c, a_d, f_d
                c = d
                d = lo + (hi - lo) / phi
                a_d = _beta_sequence(best_n, d)
                f_d = _fast_objective(a_d)

            # keep best
            if f_c < best_score:
                best_score = f_c
                best_seq = a_c.copy()
                best_alpha = c
            if f_d < best_score:
                best_score = f_d
                best_seq = a_d.copy()
                best_alpha = d

        print(f"[refine] n={best_n} α≈{best_alpha:.6f} → score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Random search over asymmetric Beta(αₗ,αᵣ) shapes (same n)
    # ------------------------------------------------------------------
    if best_seq is not None and best_n is not None and time.time() < deadline:
        n = best_n
        trials = 300  # adjustable – low enough to stay within budget
        for _ in range(trials):
            if time.time() >= deadline:
                break
            al = rng.uniform(0.10, 1.20)
            ar = rng.uniform(0.10, 1.20)
            a = _beta_sequence_general(n, al, ar)
            sc = _fast_objective(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                best_alpha = (al, ar)
                print(f"[asym] αₗ={al:.3f}, αᵣ={ar:.3f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 4.  Sub‑gradient descent on the current best sequence
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_score, best_seq = _subgradient_optimize(
            best_seq, deadline, best_score, best_seq
        )
        print(f"[subgrad] improved score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Tiny random perturbation hill‑climbing (optional)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_score, best_seq = _random_perturbation(
            best_seq, rng, deadline, best_score, best_seq
        )
        print(f"[perturb] final perturb score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final accurate evaluation (and final printout)
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist() if best_seq is not None else [0.0]
    final_score = float(evaluate_sequence(final_seq))
    print(f"[final] evaluated with evaluate_sequence → score = {final_score:.6f}")

    # Return as a plain Python list of floats
    return final_seq
