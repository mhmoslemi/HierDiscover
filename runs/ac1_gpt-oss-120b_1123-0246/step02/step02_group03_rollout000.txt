def propose_candidate(seed=0, budget_s=1000):
    import numpy as np
    import time
    import math
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2

    # Choose length and base sequence
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        # possibly truncate to <=12000
        if base.size > 12000:
            # centre
            lo = (base.size - 12000)//2
            base = base[lo:lo+12000]
    else:
        base = None

    n = None
    if base is not None:
        n = base.size
        a = np.clip(base, 0, 1000)
    else:
        n = 5000
        a = np.ones(n, dtype=np.float64)

    # Ensure even length for symmetry simplifying
    if n % 2 == 1:
        # drop one element
        a = a[:-1]
        n = a.size

    target_sum = math.sqrt(2.0 * n)
    # Normalise sum
    if a.sum() == 0:
        a[:] = 1.0
    a = a * (target_sum / a.sum())
    # Clip
    a = np.clip(a, 0, 1000)
    a = a * (target_sum / a.sum())

    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule parameters
    beta_start = 0.3
    beta_end = 15.0
    init_lr = target_sum * 0.02
    min_lr = init_lr * 0.02

    # stats
    iteration = 0
    last_improve = 0
    best_a = a.copy()
    best_score = float('inf')
    # Compute initial exact score
    b = _fft_convolve(a, a)
    best_score = b.max()
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    while time.time() < deadline:
        # time fraction
        elapsed = time.time() - start
        frac = elapsed / budget_s

        # update temperature
        beta = beta_start + (beta_end - beta_start) * frac

        # Grad
        grad, maxb = _grad_and_maxb(a, beta)

        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        # bias correction factors
        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))

        # learning rate schedule (linear decay)
        lr = max(min_lr, init_lr * (1.0 - frac))

        # Adam step
        step = - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Apply update
        cand = a + step

        # Enforce symmetry
        cand = 0.5 * (cand + cand[::-1])

        # Clip and renormalise
        cand = np.clip(cand, 0.0, 1000.0)

"""
Propose a step‑function sequence by minimizing the maximal autocorrelation
(the quantity returned by ``evaluate_sequence``).  

The algorithm combines:

* a smooth soft‑max approximation of the max‑convolution,
* Adam‑style adaptive moment updates,
* enforced symmetry (a[i] = a[n‑1‑i]) which is known to improve the bound,
* linear learning‑rate decay,
* periodic random log‑normal perturbations,
* a stagnation‑restart that re‑initialises from a fresh random symmetric vector.

The search runs for the supplied time budget (default 1000 s) and returns the
best sequence (as a list of floats) discovered before the deadline.
"""

import time
import numpy as np
import math

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a power‑of‑two FFT (real inputs)."""
    n = x.size + y.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv

def _grad_and_maxb(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the soft‑max approximation to max_k (a∗a)_k and the exact
    maximal autocorrelation value.
    """
    n = a.size
    # exact autocorrelation (needed for the true score)
    b = _fft_convolve(a, a)
    maxb = b.max()

    # soft‑max weights over all lags – subtraction of maxb improves stability
    exp_term = np.exp(beta * (b - maxb))
    w = exp_term / exp_term.sum()

    # gradient of Σ_k w_k (a∗a)_k:
    #   ∂/∂a_i = 2 Σ_k w_k a_{k‑i}
    w_rev = w[::-1]
    conv_rev = _fft_convolve(a, w_rev)                # length = 3n‑2
    idx = (2 * n - 2) - np.arange(n)                  # indices n‑1 … 2n‑2
    grad = 2.0 * conv_rev[idx]

    # project onto the hyper‑plane Σ a_i = const (zero‑sum gradient)
    grad -= grad.mean()
    return grad, maxb
