    n = f.size
    # Build sparse matrix A
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # i runs from i_start..i_end, j = k-i
        # each term contributes f[i] * g[j]
        # So coefficient for variable g[j] = f[i], where i = k - j.
        # We'll iterate over j to fill rows/cols.
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M, dtype=np.float64)
    # c = -np.ones(n) for maximizing sum g
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    return res.x

"""
Hybrid optimizer for the autoconvolution ‑max problem.

The algorithm works in three stages:

1️⃣  **Analytic start** – build the classic arcsine (sqrt‑law) sequence and
    normalise it to unit ℓ¹‑norm.  This gives a solid baseline (≈π/2≈1.57).

2️⃣  **Parametric exploration** – generate whole families of candidate shapes
    (monotone power‑law and symmetric beta‑law) with a single exponent
    parameter.  Each candidate is normalised, its exact objective
    (2·n·max(conv)/ℓ¹²) is evaluated and the best one is kept.  The search
    sweeps a fine grid of exponents (≈100 evaluations) and therefore can
    discover modest improvements (≈4 % below the arcsine bound).

3️⃣  **Local improvement** – starting from the best candidate, a cheap
    stochastic hill‑climber (block scaling, mass transfers, early→late moves,
    a gradient‑inspired transfer) is run until the time budget expires.
    Only strictly improving candidates are accepted, and the overall best
    sequence seen is returned.

All operations respect the problem’s safety checks (non‑negative,
≤1000, ℓ¹‑norm fixed), and the function `propose_candidate` conforms to the
required signature.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# --------------------------------------------------------------
# 1.  Analytic building blocks
# --------------------------------------------------------------

def build_arcsine_sequence(length: int) -> np.ndarray:
    """Discrete arcsine sequence a[i] = C(2i,i)/4**i (i≥0)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


def build_power_law_sequence(length: int, p: float) -> np.ndarray:
    """Monotone power‑law a[i] = (i+1)^(-p)."""
    idx = np.arange(length, dtype=np.float64)
    return (idx + 1.0) ** (-p)


def build_symmetric_power_sequence(length: int, p: float) -> np.ndarray:
    """Symmetric beta‑law a[i] = ((i+0.5)*(n‑i‑0.5))^(-p)."""
    i = np.arange(length, dtype=np.float64)
    # shift by 0.5 to avoid division by zero at the boundaries
    left = i + 0.5
    right = length - i - 0.5
    return (left * right) ** (-p)


def renormalise(seq: np.ndarray, target_sum: float = 1.0) -> np.ndarray:
    """Scale a non‑negative sequence so that its ℓ¹‑norm equals `target_sum`."""
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur == 0.0:
        return seq
    return seq * (target_sum / cur)


def sequence_score(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq))/sum(seq)²."""
    s = float(seq.sum())
    if s < 0.01:
        return np.inf
    b = np.convolve(seq, seq)
    return 2.0 * seq.shape[0] * float(b.max()) / (s * s)


# --------------------------------------------------------------
# 2.  Stochastic move primitives (preserve ℓ¹‑norm)
# --------------------------------------------------------------

def block_perturbation(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                 # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return renormalise(cand, target_sum)


def transfer_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """Move a random fraction (≤30 %) of mass from one index to another."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


def early_to_late_move(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Shift mass from the first 10 % of entries to the last 10 %."""
    n = base.size
    early_cut = max(1, n // 10)
    late_start = max(early_cut, n - early_cut)
    i = int(rng.integers(0, early_cut))
    j = int(rng.integers(late_start, n))
    if i == j:
        return base.copy()
    frac = rng.random() * 0.40
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


def gradient_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """
    Transfer mass from the coefficient that contributes most to the current
    maximal convolution entries toward a coefficient that contributes little.
    """
    n = base.size
    b = np.convolve(base, base)
    max_b = b.max()
    tol = max_b * (1.0 - 1e-12)
    mask = (b >= tol).astype(np.float64)          # positions attaining max
    contrib_conv = np.convolve(base, mask)        # length 2n‑1
    contrib = base * contrib_conv[:n]             # per‑index contribution
    heavy = int(np.argmax(contrib))
    # Choose a “light” index (low contribution, not the heavy one)
    light_candidates = np.where(contrib < contrib.mean())[0]
    if light_candidates.size == 0:
        light = (heavy + n // 2) % n
    else:
        light = int(rng.choice(light_candidates[light_candidates != heavy]))
    frac = rng.random() * 0.30
    delta = frac * base[heavy]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[heavy] -= delta
    cand[light] += delta
    return renormalise(cand, target_sum)


# --------------------------------------------------------------
# 3.  Linear‑programming sub‑routine (optional, expensive)
# --------------------------------------------------------------

def build_conv_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (size (2n‑1)×n) such that (A @ g)[k] = (f*g)[k].
    The matrix contains the shifted copies of ``f`` on the –d diagonals.
    """
    n = f.size
    # Each diagonal –d (d≥0) is constant = f[d]
    diagonals = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = [-d for d in range(n)]
    A = sp.diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def solve_lp_max_sum(f: np.ndarray, bound: float) -> np.ndarray | None:
    """
    Linear program:
        maximise   Σ g_j
        subject to   f * g ≤ bound   (element‑wise)
                     g ≥ 0
    Returns the optimal g (un‑scaled) or ``None`` on failure.
    """
    n = f.size
    A = build_conv_constraint_matrix(f)
    b = np.full(2 * n - 1, bound, dtype=np.float64)
    # maximise Σ g → minimise -Σ g
    c = -np.ones(n, dtype=np.float64)
    try:
        res = linprog(c=c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
    except Exception:  # pragma: no cover – safety net
        return None
    if not res.success:
        return None
    return res.x


# --------------------------------------------------------------
# 4.  Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs) -> list[float]:
    """
    Return the best step‑function heights found within ``budget_s`` seconds.
    The routine combines an analytic start, a small parametric sweep over
    power‑law families, and a final stochastic hill‑climber.
    """
    rng = np.random.default_rng(seed)

    # ---- problem size -------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256                     # fall‑back size

    # ---- initialise with an analytic sequence -------------------------
    base = build_arcsine_sequence(n)
    base = renormalise(base, target_sum=1.0)   # ℓ¹‑norm = 1
    cur_seq = base.copy()
    best_seq = cur_seq.copy()
    best_score = sequence_score(best_seq)
    print(f"[init] n={n}, score={best_score:.9f}")

    # ---- 2. parametric exploration ------------------------------------
    # power‑law (decreasing) family
    for p in np.linspace(0.45, 0.85, 41):
        cand = build_power_law_sequence(n, p)
        cand = renormalise(cand, target_sum=1.0)
        sc = sequence_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[grid‑p] p={p:.3f} score={sc:.9f}")

    # symmetric beta‑law family (singular at both ends)
    for p in np.linspace(0.45, 0.85, 41):
        cand = build_symmetric_power_sequence(n, p)
        cand = renormalise(cand, target_sum=1.0)
        sc = sequence_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[grid‑sym] p={p:.3f} score={sc:.9f}")

    # early exit if we already beat the target
    TARGET = 1.503
    if best_score < TARGET:
        print(f"[early] achieved target with score={best_score:.9f}")
        return [float(x) for x in best_seq]

    # ---- 3. local stochastic refinement --------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score
    target_sum = 1.0
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # pick a move at random
        r = rng.random()
        if r < 0.35:
            cand = block_perturbation(cur_seq, rng, target_sum)
        elif r < 0.55:
            cand = transfer_move(cur_seq, rng, target_sum)
        elif r < 0.80:
            cand = early_to_late_move(cur_seq, rng, target_sum)
        else:
            cand = gradient_move(cur_seq, rng, target_sum)

        cand_score = sequence_score(cand)
        if np.isfinite(cand_score) and cand_score < cur_score:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[{it:>6}] new best = {best_score:.9f}")
                if best_score < TARGET:
                    print("[target] reached during hill‑climb")
                    break
        else:
            # Occasionally try the LP‑based direction (expensive, limited)
            if it % 2500 == 0:
                # solve LP using current sequence as bound
                bound = np.convolve(cur_seq, cur_seq).max()
                g_raw = solve_lp_max_sum(cur_seq, bound)
                if g_raw is not None:
                    g = renormalise(g_raw, target_sum=target_sum)
                    # try a half‑step mixture
                    cand2 = 0.5 * cur_seq + 0.5 * g
                    cand2 = renormalise(cand2, target_sum)
                    cand2_score = sequence_score(cand2)
                    if cand2_score < cur_score:
                        cur_seq = cand2
                        cur_score = cand2_score
                        if cand2_score < best_score:
                            best_seq = cand2.copy()
                            best_score = cand2_score
                            print(f"[{it:>6}] LP‑mix improved -> {best_score:.9f}")
                            if best_score < TARGET:
                                break

    print(f"[final] best score {best_score:.9f} after {it} iterations")
    # Convert to plain Python list, clip to required range
    result = [float(min(max(v, 0.0), 1000.0)) for v in best_seq]
    return result
