    deadline = start + budget_s - 1

    while time.monotonic() < deadline:
        # compute convolution and identify peak
        conv = np.convolve(cur, cur)
        kmax = int(np.argmax(conv))
        # maybe compute contributions list
        # define a little weight for moving mass
        # Randomly decide which move
        move_type = rng.random()
        cand = cur.copy()
        if move_type < 0.5:
            # transfer from high contributor to random low-index
            # find indices that contribute to kmax
            contrib_indices = []
            for i in range(max(0, kmax-(n-1)), min(n-1, kmax)+1):
                j = kmax - i
                if j < 0 or j >= n:
                    continue
                # each pair (i,j) counted twice except i=j.
                # treat we can pick i or j
                # compute product
                prod = cur[i] * cur[j]
                contrib_indices.append((prod, i))
            # sort descending
            contrib_indices.sort(reverse=True, key=lambda x: x[0])
            # pick top index
            if contrib_indices:
                _, i_src = contrib_indices[0]
                # choose a target index not close to i_src to avoid same pair
                # choose j_target randomly from middle region
                tgt_region_start = n//4
                tgt_region_end = 3*n//4
                j_target = int(rng.integers(tgt_region_start, tgt_region_end))

                amount = cur[i_src] * rng.random() * 0.5
                cand[i_src] -= amount
                cand[j_target] += amount
        elif move_type < 0.8:

            width = int(rng.integers(n//200, n//12))
            start = int(rng.integers(n//4, 3*n//4 - width))
            factor = float(np.exp(rng.normal(0.0, 0.12)))

            cut = int(rng.integers(n//10, 9*n//10))

def guided_transfer_move(base, rng):
    # compute convolution
    conv = np.convolve(base, base)
    kmax = int(np.argmax(conv))
    # Compute contributions at kmax
    n = base.size
    # Compute i range
    i_start = max(0, kmax - (n-1))
    i_end = min(n-1, kmax)
    # Among these indices, find i where base[i] is largest and also a_{kmax-i} large.
    best_i = i_start
    best_contrib = 0.0
    for i in range(i_start, i_end+1):
        j = kmax - i
        contrib = base[i] * base[j]
        if contrib > best_contrib:
            best_contrib = contrib
            best_i = i
    # Choose a target index far from those that affect kmax:
    # for simplicity choose random index from middle region not overlapping.
    target = int(rng.integers(n//4, 3*n//4))
    # Ensure target not equal to i or j
    if target == best_i or target == kmax-best_i:
        target = (target+1)%n
    # Amount to shift: small fraction of base[best_i]
    frac = rng.random() * 0.5
    amount = frac * base[best_i]
    cand = base.copy()
    cand[best_i] -= amount
    cand[target] += amount
    return np.clip(cand, 0.0, 1000.0)

"""
Propose a step‑function sequence that (hopefully) beats the classical
arcsine construction (score ≈ π/2).  The search consists of three stages:

1. **Parameter sweep** – generate a handful of analytically‑motivated
   families (arcsine, beta‑shapes, simple power laws) and keep the best
   one according to a fast local implementation of the evaluation
   function.

2. **Guided Simulated Annealing** – start from the best sweep‑candidate
   and run a SA loop for the remaining time budget.  Moves are
   deliberately chosen to move mass away from the indices that dominate
   the current maximal convolution value.  Additional generic moves
   (block scaling, rebalance, power‑transform, smoothing) keep the
   search diverse.

3. **Final output** – the best sequence seen during the run is returned
   as a plain Python list.  All constraints of ``evaluate_sequence`` are
   respected (non‑negative, ≤ 1000, sum ≥ 0.01).

A lightweight ``fast_score`` reproduces the exact behaviour of
``evaluate_sequence`` without the Python‑level checks, allowing many
thousands of evaluations inside the time limit.
"""

import math
import time

import numpy as np

# ----------------------------------------------------------------------
# Fast replica of evaluate_sequence – used for the internal optimisation.
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray) -> float:
    """Return the same objective as ``evaluate_sequence`` but without
    the security checks.  Input is a NumPy array.
    """
    a = np.clip(seq, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(a, a)
    m = conv.max()
    n = a.size
    return 2.0 * n * m / (s * s)


# ----------------------------------------------------------------------
# Analytic families of candidate sequences.
# ----------------------------------------------------------------------
def build_arcsine_sequence(n: int) -> np.ndarray:
    """Return the discrete arcsine (central binomial) sequence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return np.clip(a, 0.0, 1000.0)


def beta_shape_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Beta(α,α) density sampled on the mid‑points of n equal sub‑intervals.
    The raw values are NOT normalised – scaling does not affect the
    objective, only the relative shape matters.
    """
    # avoid division by zero for extreme α
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # density proportional to x**(α-1)*(1-x)**(α-1)
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    return np.clip(a, 0.0, 1000.0)


def power_law_sequence(n: int, p: float) -> np.ndarray:
    """
    Simple symmetric power‑law: a_i ∝ ((i+½)(n-i-½))^{-p}.
    For p≈0.5 we recover the arcsine shape.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = n - 0.5 - i
    a = np.power(i * j, -p)
    return np.clip(a, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Move primitives used inside the annealing loop.
# ----------------------------------------------------------------------
def transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from an early index to a far later one."""
    n = base.size
    i = int(rng.integers(0, n // 4))
    j = int(rng.integers(3 * n // 4, n))
    if base[i] == 0.0:
        return base.copy()
    frac = rng.random() * 0.9
    delta = frac * base[i]
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random sub‑array by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.13)))
    cand[start:start + width] *= factor
    if rng.random() < 0.30:
        cand = 0.5 * (cand + cand[::-1])  # occasional symmetrisation
    return np.clip(cand, 0.0, 1000.0)


def rebalance_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Shift mass from an early prefix to a later suffix (approximately preserving total)."""
    n = base.size
    cut = int(rng.integers(n // 10, 9 * n // 10))
    prefix = base[:cut]
    suffix = base[cut:]

    sum_prefix = prefix.sum()
    sum_suffix = suffix.sum()
    if sum_prefix == 0.0:
        return base.copy()

    amount = rng.random() * 0.5 * sum_prefix
    factor_pre = max(0.0, (sum_prefix - amount) / sum_prefix)
    new_prefix = prefix * factor_pre

    if sum_suffix > 0.0:
        factor_suf = (sum_suffix + amount) / sum_suffix
        new_suffix = suffix * factor_suf
    else:
        new_suffix = np.full_like(suffix, amount / suffix.size)

    cand = np.concatenate([new_prefix, new_suffix])
    return np.clip(cand, 0.0, 1000.0)


def power_transform_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply an element‑wise power x→x^γ (γ∈[0.8,1.2]) and renormalise to the original sum."""
    gamma = rng.uniform(0.8, 1.2)
    # avoid 0**γ problems – 0 stays 0
    cand = np.where(base > 0, np.power(base, gamma), 0.0)
    s = cand.sum()
    if s > 0.0:
        cand *= base.sum() / s
    return np.clip(cand, 0.0, 1000.0)


def smooth_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Replace the sequence by a moving‑average smoothed version (renormalised)."""
    n = base.size
    width = int(rng.integers(3, 15))
    kernel = np.ones(width, dtype=np.float64) / width
    smoothed = np.convolve(base, kernel, mode='same')
    s = smoothed.sum()
    if s > 0.0:
        smoothed *= base.sum() / s
    return np.clip(smoothed, 0.0, 1000.0)


def guided_transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Identify the index k where the autocorrelation is maximal.
    Pick the individual element that contributes most to that peak and move
    a random fraction of its mass to a location unlikely to increase the peak.
    """
    n = base.size
    conv = np.convolve(base, base)
    kmax = int(conv.argmax())

    # Determine the pair (i,j) with the largest product a_i*a_j for i+j=kmax
    i_start = max(0, kmax - (n - 1))
    i_end = min(n - 1, kmax)
    best_i = i_start
    best_contrib = -1.0
    for i in range(i_start, i_end + 1):
        j = kmax - i
        contrib = base[i] * base[j]
        if contrib > best_contrib:
            best_contrib = contrib
            best_i = i

    # Choose a target far from both ends used in the current peak
    tgt_start = n // 4
    tgt_end = 3 * n // 4
    target = int(rng.integers(tgt_start, tgt_end))
    # ensure we are not moving mass back to the same location
    if target == best_i:
        target = (target + 1) % n

    # Transfer a random fraction of the selected source element
    if base[best_i] == 0.0:
        return base.copy()
    frac = rng.random() * 0.5   # up to 50 %
    delta = frac * base[best_i]
    cand = base.copy()
    cand[best_i] -= delta
    cand[target] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a low‑score step‑function using a fast score replica,
    an initial analytical sweep and a guided simulated‑annealing phase.
    The best sequence seen before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose the length of the sequence.
    # ------------------------------------------------------------------
    n = 0
    try:
        # height_sequence_1 is provided by the harness if a reference exists
        n = len(globals().get("height_sequence_1", []))
    except Exception:
        n = 0
    if n == 0:
        # no reference – pick a power‑of‑two length that works well with FFT
        n = 8192

    print(f"[info] Using sequence length n = {n}")

    # ------------------------------------------------------------------
    # 2. Build a collection of simple analytic candidates and keep the best.
    # ------------------------------------------------------------------
    candidates = []

    # a) arcsine
    candidates.append(build_arcsine_sequence(n))

    # b) beta‑shapes (α in [0.5, 2.0])
    for a in np.linspace(0.5, 2.0, 12):
        candidates.append(beta_shape_sequence(n, a))

    # c) simple power‑law family (p in [0.3,0.7])
    for p in np.linspace(0.35, 0.65, 7):
        candidates.append(power_law_sequence(n, p))

    # d) a random initialise (helps to escape pathological local minima)
    rand_vec = rng.random(n) * 0.5 + 0.5   # keep values modest
    candidates.append(rand_vec)

    best_seq = None
    best_score = float("inf")
    for idx, seq in enumerate(candidates):
        sc = fast_score(seq)
        if sc < best_score:
            best_seq = seq.copy()
            best_score = sc
            print(f"[sweep] candidate {idx:02d} score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Simulated annealing with several move types.
    # ------------------------------------------------------------------
    cur = best_seq.copy()
    cur_score = best_score
    it = 0

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # keep a 1‑s safety margin

    # temperature schedule (linear cooling)
    T_start = 0.04
    T_end = 1e-6

    move_functions = [
        transfer_move,
        block_perturbation,
        rebalance_move,
        power_transform_move,
        smooth_move,
        guided_transfer_move,
    ]

    while time.monotonic() < deadline:
        it += 1
        # linear temperature interpolation
        elapsed = (time.monotonic() - start_time) / budget_s
        T = T_start * (1.0 - elapsed) + T_end * elapsed

        # pick a move at random
        move = rng.choice(move_functions)
        cand = move(cur, rng)

        cand_score = fast_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[{it:>7}] new best score = {best_score:.6f}")

        # occasional restart to the best seen so far (prevents drift)
        if it % 25000 == 0:
            cur = best_seq.copy()
            cur_score = best_score

        # early exit if we already beat the target comfortably
        if best_score < 1.503:
            print(f"[stop] target beaten after {it} iterations")
            break

    # ------------------------------------------------------------------
    # 4. Return result as plain list of Python floats.
    # ------------------------------------------------------------------
    print(f"[final] best score after {it} iterations: {best_score:.6f}")
    return [float(x) for x in best_seq]
