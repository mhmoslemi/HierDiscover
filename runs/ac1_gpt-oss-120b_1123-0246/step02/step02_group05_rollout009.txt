def full_convolution(seq):
    n = seq.shape[0]
    m = 2*n - 1
    N = 1 << ((m - 1).bit_length())
    fft_a = np.fft.rfft(seq, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)[:m]
    return conv

"""
Simulated‑annealing search for a step‑function minimizing the convolution‑
bound objective used in the competition.  

The algorithm:
    1. Initialise the sequence with the analytically optimal “arcsine”
       (inverse‑square‑root) shape and normalise its ℓ₁‑norm to 1.0.
    2. Repeatedly propose a candidate obtained by one of several cheap
       stochastic moves:
           – block scaling (multiply a random contiguous block);
           – simple mass transfer between two positions;
           – early‑to‑late transfer (push mass from the first part to the last);
           – a gradient‑inspired move that shifts mass from the index
             contributing most to the current maximal convolution value
             toward the index contributing the least;
           – a “tail‑balance” move that deliberately shifts mass from the
             early quarter of the array to the final quarter.
    3. Accept a candidate if it improves the objective or with a probability
       exp(−Δ / T) where T is a linearly decreasing temperature.  This
       simulated‑annealing scheme allows occasional uphill moves and helps
       escape shallow local minima.
    4. Keep the best sequence seen and return it when the time budget expires.

All moves keep the total sum (ℓ₁‑norm) fixed via a renormalisation step.
The objective is scale‑invariant, so we simply target a sum of 1.0.  The
convolution is evaluated with an FFT‑based routine for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def full_convolution(seq: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a (length 2·len(a)‑1) using FFT."""
    n = seq.size
    m = 2 * n - 1
    # next power of two for efficient FFT
    N = 1 << ((m - 1).bit_length())
    fft_a = np.fft.rfft(seq, n=N)
    conv_full = np.fft.irfft(fft_a * fft_a, n=N)[:m]
    # Numerical noise may give tiny negative numbers – clamp to zero
    np.maximum(conv_full, 0.0, out=conv_full)
    return conv_full


def sequence_score(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 0.01:
        return np.inf
    b = full_convolution(seq)
    max_b = b.max()
    n = seq.size
    return 2.0 * n * max_b / (s * s)


def renormalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to the allowed range [0,1000] and scale to the given ℓ₁‑norm."""
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur == 0.0:
        return seq
    seq = seq * (target_sum / cur)
    # a second clipping prevents rare overshoots after scaling
    seq = np.clip(seq, 0.0, 1000.0)
    # re‑scale again to keep the exact target sum (only a tiny correction)
    cur = seq.sum()
    if cur == 0.0:
        return seq
    seq = seq * (target_sum / cur)
    return seq


# ----------------------------------------------------------------------
# Analytic starting point (inverse‑square‑root / arcsine shape)
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative arcsine sequence a_i = binom(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Stochastic move primitives (all preserve the ℓ₁‑norm via renormalise)
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Scale a random contiguous block by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # σ≈0.12
    cand[start:stop] *= factor
    if rng.random() < 0.35:                         # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return renormalise(cand, target_sum)


def transfer_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """Move a random fraction (≤30 %) of mass from one index to another."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


def early_to_late_move(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Move a random fraction (≤40 %) of mass from the first 10 % to the last 10 %."""
    n = base.size
    early_cut = max(1, n // 10)
    late_start = max(early_cut, n - early_cut)
    i = int(rng.integers(0, early_cut))
    j = int(rng.integers(late_start, n))
    if i == j:
        return base.copy()
    frac = rng.random() * 0.40
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


def early_tail_balance_move(base: np.ndarray, rng: np.random.Generator,
                            target_sum: float) -> np.ndarray:
    """Stronger early‑to‑late move that prefers the most/least populated quarters."""
    n = base.size
    q = max(1, n // 4)
    # choose source in the first quarter, target in the last quarter
    i = int(rng.integers(0, q))
    j = int(rng.integers(n - q, n))
    if i == j:
        return base.copy()
    # move a larger fraction if the early part is noticeably heavier
    early_sum = base[:q].sum()
    tail_sum = base[-q:].sum()
    imbalance = max(0.0, early_sum - tail_sum)
    max_frac = 0.6
    # fraction proportional to the relative imbalance, but capped
    frac = min(max_frac, (imbalance / (early_sum + 1e-12)) * 0.6 + 0.05 * rng.random())
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


def random_noise_move(base: np.ndarray, rng: np.random.Generator,
                      target_sum: float, sigma: float = 0.03) -> np.ndarray:
    """Add small Gaussian noise (relative to the current value)."""
    noise = rng.normal(0.0, sigma, size=base.shape)
    cand = base + noise * base
    cand = np.clip(cand, 0.0, None)
    return renormalise(cand, target_sum)


def gradient_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """
    Shift mass from the index with the largest contribution to the maximal
    convolution values toward the index with the smallest contribution.
    The contribution of index i is approximated by the sub‑gradient of
    max(conv(base,base)).
    """
    n = base.size
    b = full_convolution(base)
    max_b = b.max()
    # indices attaining the (near) maximum; a tiny tolerance avoids numeric issues
    tol = max_b * (1.0 - 1e-12)
    max_idxs = np.where(b >= tol)[0]

    # compute a sub‑gradient: for each i, sum_{k in max_idxs} a_{k-i}
    contrib = np.zeros(n, dtype=np.float64)
    for idx in max_idxs:
        start = max(0, idx - (n - 1))
        end = min(idx, n - 1)
        js = np.arange(start, end + 1, dtype=np.int64)
        partners = idx - js
        # derivative contribution: a_{partner} added to grad[j],
        # and a_{j} added to grad[partner]; diagonal gets 2*a_j automatically.
        contrib[js] += base[partners]
        contrib[partners] += base[js]

    heavy_idx = int(np.argmax(contrib))
    light_idx = int(np.argmin(contrib))
    if heavy_idx == light_idx:
        # fallback to a random transfer
        return transfer_move(base, rng, target_sum)

    # the fraction moved is temperature‑aware: larger early on, smaller later
    frac = rng.random() * 0.30               # up to 30 %
    delta = frac * base[heavy_idx]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[heavy_idx] -= delta
    cand[light_idx] += delta
    return renormalise(cand, target_sum)

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of floats – the best step‑function heights found within
    ``budget_s`` seconds.  The search uses simulated annealing with several
    specialised moves to lower the convolution‑bound objective.
    """
    rng = np.random.default_rng(seed)

    # Determine the length of the problem (fallback to 256 if not provided)
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256

    # ------------------------------------------------------------------
    # 1.  Initial sequence – analytically optimal arcsine shape,
    #     normalised to ℓ₁‑norm = 1.0 (the objective is scale invariant).
    # ------------------------------------------------------------------
    target_sum = 1.0
    base_seq = build_arcsine_sequence(n)
    cur_seq = renormalise(base_seq, target_sum)
    cur_score = sequence_score(cur_seq)

    best_seq = cur_seq.copy()
    best_score = cur_score

    print(f"[init] length={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Simulated‑annealing loop
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # a tiny safety margin
    iteration = 0
    # initial temperature – a fraction of the initial score
    init_temp = max(1e-12, cur_score * 0.20)

    while time.monotonic() < deadline:
        iteration += 1
        elapsed = time.monotonic() - start_time
        progress = elapsed / budget_s
        temperature = init_temp * (1.0 - progress)          # linear cooling

        # ---- choose a move -------------------------------------------------
        r = rng.random()
        if r < 0.18:
            cand = block_perturbation(cur_seq, rng, target_sum)
        elif r < 0.36:
            cand = transfer_move(cur_seq, rng, target_sum)
        elif r < 0.54:
            cand = early_to_late_move(cur_seq, rng, target_sum)
        elif r < 0.72:
            cand = early_tail_balance_move(cur_seq, rng, target_sum)
        elif r < 0.90:
            cand = gradient_move(cur_seq, rng, target_sum)
        else:
            cand = random_noise_move(cur_seq, rng, target_sum, sigma=0.02)

        cand_score = sequence_score(cand)

        # ---- acceptance test (simulated annealing) ------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # uphill move – accept with probability exp(-Δ/T)
            if temperature > 0.0:
                accept = rng.random() < np.exp(-(cand_score - cur_score) / temperature)

        if accept:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                print(f"[{elapsed:6.2f}s] new best = {best_score:.6f} (it {iteration})")
                if best_score < 1.503:          # early exit if target reached
                    break

        # occasional random restart to avoid stagnation (very low probability)
        if rng.random() < 0.0005:
            cur_seq = renormalise(build_arcsine_sequence(n), target_sum)
            cur_score = sequence_score(cur_seq)

    total_time = time.monotonic() - start_time
    print(f"[final] best score {best_score:.6f} after {iteration} iterations ({total_time:.2f}s)")
    # Return a plain Python list as required by the harness
    return [float(x) for x in best_seq]
