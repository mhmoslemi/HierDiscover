def project_simplex(v, s=1.0):
    n = v.shape[0]
    if s < 0:
        raise ValueError("s must be non-negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Projected Adam optimisation of a non‑negative step‑function.

The evaluation objective for a sequence a of length n is

    score = 2*n*max(conv(a,a)) / (sum(a)^2)

Scaling cancels if we enforce   sum(a)=√(2n) .
Thus, with p = a / √(2n) (so ∑p = 1) the problem reduces to

    minimise   max( p * p )               (linear convolution)

subject to p_i ≥ 0, ∑p_i = 1.

We perform a projected Adam optimisation on a smooth approximation of the
max‑operator (log‑sum‑exp with an increasing temperature β) and repeatedly
project the iterate onto the probability simplex.  Several random starts,
different initial shapes and occasional multiplicative perturbations are used
to escape local minima.  The best sequence found (converted back to the
required scaling) is returned.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

    """
    Linear convolution a * a using real FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)
    return conv_full[: 2 * a.size - 1]


def _correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full correlation (w ⋆ a) = convolution(w[::-1], a) via FFT.
    Returns an array of length len(w)+len(a)-1.
    """
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, fft_len)
    fa = np.fft.rfft(a, fft_len)
    corr_full = np.fft.irfft(fw * fa, fft_len)
    return corr_full[: w.size + a.size - 1]


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Projects vector v onto the simplex { x | x_i ≥ 0, ∑ x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum s must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """
    Returns a normalised soft‑max weight vector for the entries of b.
    Uses a shift by the maximum for numerical stability.
    """
    max_b = b.max()
    shifted = b - max_b
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        w_norm = np.zeros_like(w)
        w_norm[np.argmax(b)] = 1.0
    else:
        w_norm = w / w_sum
    return w_norm


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 950.0, **kwargs):
    """
    Search for a low‑score height sequence within the allotted budget.
    Returns a list of floats (the best sequence found).
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # leave a small safety margin for the final printing / shutdown
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Choose a working length and derived constants
    # ------------------------------------------------------------------
    # try to reuse the length of a previously found construction if available
    try:
        prev_len = int(np.asarray(height_sequence_1).size)
    except Exception:
        prev_len = None

    # moderate size – enough to get a good lower bound but cheap to optimise
    n = 600
    if prev_len is not None:
        # prefer a similar size but do not exceed a few thousand
        n = int(np.clip(prev_len, 400, 1500))

    target_sum = np.sqrt(2.0 * n)      # we enforce sum(a)=target_sum
    fft_len = _next_pow2(3 * n)        # length for all FFTs (≥ len(w)+len(a)-1)

    # -----------------------------------------------------------------------------
    # Build several starting points (probability vectors that sum to 1)
    # ----------------------------------------------------------------------------- 
    starts = []

    # (a) Uniform distribution
    starts.append(np.full(n, 1.0 / n))

    # (b) Random Dirichlet (Gamma(1) i.i.d.)
    starts.append(rng.random(n))
    starts[-1] /= starts[-1].sum()

    # (c) Slightly “U‑shaped’’ – larger near the centre
    idx = np.arange(n, dtype=float)
    u_shape = np.sqrt((idx + 1) * (n - idx))
    u_shape = np.maximum(u_shape, 1e-12)
    u_shape /= u_shape.sum()
    starts.append(u_shape)

    # (d) Exponentially decaying tail
    exp_decay = np.exp(-0.005 * idx)
    exp_decay /= exp_decay.sum()
    starts.append(exp_decay)

    # (e) Re‑use the global best if it exists (interpolated to length n)
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            if prev.size != n:
                old_x = np.linspace(0.0, 1.0, prev.size)
                new_x = np.linspace(0.0, 1.0, n)
                prev = np.interp(new_x, old_x, prev)
            prev = np.clip(prev, 0.0, 1000.0)
            if prev.sum() > 0:
                p_prev = prev / prev.sum()
                starts.append(p_prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    beta_start = 1.0
    beta_end = 30.0          # final temperature – closer to the true max
    max_iters_per_start = 3000
    lr = 0.02                # base learning rate for Adam (before bias‑correction)
    b1, b2 = 0.9, 0.999      # Adam decay rates
    eps = 1e-8

    best_seq = None          # will hold the best *a* (not p)
    best_score = np.inf
    total_iters = 0

    # ------------------------------------------------------------------
    # Core optimisation loops over the different starts
    # ------------------------------------------------------------------
    for start_id, p0 in enumerate(starts):
        if time.monotonic() >= deadline:
            break

        # initialise Adam state
        p = np.copy(p0)
        p = _project_simplex(p, s=1.0)   # ensure exact sum=1
        m = np.zeros_like(p)
        v = np.zeros_like(p)

        # evaluate the initial point (exact evaluation)
        a0 = p * target_sum
        init_score = float(evaluate_sequence(a0.tolist()))
        if init_score < best_score:
            best_score = init_score
            best_seq = a0.copy()
            print(f"[start {start_id}] initial exact score = {best_score:.6f}")

        # track current max of self‑convolution (on the p‑scale)
        b = _self_convolution_fft(p, fft_len)
        cur_max_c = b.max()

        for it in range(max_iters_per_start):
            now = time.monotonic()
            if now >= deadline:
                break

            # --------- (1) self‑convolution and soft‑max weights ----------
            b = _self_convolution_fft(p, fft_len)            # c = p * p
            cur_max_c = b.max()
            beta = beta_start + (beta_end - beta_start) * (it / max_iters_per_start)

            w_norm = _softmax_weights(b, beta)               # length 2n‑1

            # --------- (2) gradient of the smoothed max ------------
            # correlation(w_norm, p) gives an array of length 3n‑2
            corr = _correlation_fft(w_norm, p, fft_len)
            L = 2 * n - 1                     # centre index of the correlation output
            grad = 2.0 * beta * corr[L - 1 - np.arange(n)]

            # --------- (3) Adam update (projected onto simplex) ----------
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * (grad * grad)

            m_hat = m / (1.0 - b1 ** (it + 1))
            v_hat = v / (1.0 - b2 ** (it + 1))

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            p_candidate = p - step
            p_candidate = _project_simplex(p_candidate, s=1.0)

            # optional monotone acceptance (helpful stability)
            b_candidate = _self_convolution_fft(p_candidate, fft_len)
            cand_max_c = b_candidate.max()
            if cand_max_c < cur_max_c or rng.random() < 0.001:
                p = p_candidate
                cur_max_c = cand_max_c

            # --------- (4) bookkeeping of the best ever seen ------------
            score = cur_max_c * (2.0 * n)      # exactly what evaluate_sequence returns
            if score < best_score:
                best_score = score
                best_seq = (p * target_sum).copy()
                print(f"[{total_iters:6d}] new best score = {best_score:.6f}")

            total_iters += 1

            # --------- (5) occasional random perturbation -------------
            if (it + 1) % 500 == 0:
                # multiplicative Gaussian jitter
                mult = np.exp(rng.normal(0.0, 0.02, size=n))
                p_pert = p * mult
                p_pert = _project_simplex(p_pert, s=1.0)
                b_pert = _self_convolution_fft(p_pert, fft_len)
                pert_max_c = b_pert.max()
                if pert_max_c < cur_max_c:
                    p = p_pert
                    cur_max_c = pert_max_c
                    pert_score = pert_max_c * (2.0 * n)
                    if pert_score < best_score:
                        best_score = pert_score
                        best_seq = (p_pert * target_sum).copy()
                        print(f"[{total_iters:6d}] perturbed improvement = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation of the best candidate before returning
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (should never happen)
        best_seq = np.full(n, target_sum / n)

    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] best exact score = {final_score:.6f}")

    # ensure the output complies with the required format (list of floats)
    return [float(v) for v in best_seq]
