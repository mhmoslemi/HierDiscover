"""
Propose a low‑score step function.

Algorithm
---------
1.  Fast objective:   2·n·max(conv(a,a)) / (sum(a)**2)   (no clipping needed
    because we keep the values inside the allowed range).

2.  Grid search over several families of sequences (Beta‑shaped,
    exponential decay, power‑law, linear, uniform) for a handful of
    lengths.  Each family is normalised to the target sum
    sqrt(2·n) (the scale used by the official judge – any other scale
    gives the same score unless clipping occurs).

3.  Keep the best candidate (lowest fast objective).

4.  Simulated‑annealing refinement:
        * start from the best candidate,
        * repeatedly move a tiny amount of mass from one index to another
          (or swap two values),
        * accept moves that improve the score, otherwise accept with a
          temperature‑dependent probability,
        * temperature and step size decrease linearly with the elapsed
          time.

5.  Return the best sequence found (as a plain Python list).  The final
    score is printed using the provided `evaluate_sequence` function.
"""

import time
import numpy as np
from math import sqrt, log, exp

# ----------------------------------------------------------------------
# Fast objective (the same formula that evaluate_sequence finally uses)
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2) using FFT."""
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    size = 2 * n - 1
    # next power of two for fast FFT
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Normalisation helper – scale to sqrt(2·n) and clip to [0,1000]
# ----------------------------------------------------------------------
def _normalise(v: np.ndarray) -> np.ndarray:
    target = sqrt(2.0 * v.size)
    s = v.sum()
    if s == 0:
        v = np.full_like(v, target / v.size)
    else:
        v = v * (target / s)
    return np.clip(v, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------
def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, alpha - 1.0)
    return _normalise(w)


def _exp_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i / n)   (i = 0 … n‑1)"""
    idx = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * idx / n)
    return _normalise(w)


def _power_sequence(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)   (i = 0 … n‑1)"""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = np.power(idx, -p)
    return _normalise(w)


def _linear_desc_sequence(n: int) -> np.ndarray:
    """Descending linear shape (largest at i=0, zero at the end)."""
    w = np.linspace(n - 1, 0, n, dtype=np.float64)
    return _normalise(w)


def _linear_asc_sequence(n: int) -> np.ndarray:
    """Ascending linear shape (small at i=0, large at the end)."""
    w = np.linspace(0, n - 1, n, dtype=np.float64)
    return _normalise(w)


def _uniform_sequence(n: int) -> np.ndarray:
    w = np.ones(n, dtype=np.float64)
    return _normalise(w)

# ----------------------------------------------------------------------

    of non‑negative heights.
    """

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)  # tiny safety margin

    best_score = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # Helper to possibly update the current best solution
    # ------------------------------------------------------------------

        nonlocal best_seq, best_score, best_n
        if np.isfinite(score) and score < best_score:
            best_seq = seq.copy()
            best_score = score
            best_n = seq.size
            print(f"[update] n={best_n:5d} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 0.  If a global sequence exists, evaluate it first
    # ------------------------------------------------------------------
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        seq0 = np.clip(seq0, 0.0, 1000.0)

            _maybe_update(seq0, sc0)
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Grid search over families, lengths and parameters
    # ------------------------------------------------------------------
    n_candidates = [1024, 2048, 4096, 8192, 12288, 16384]
    # ----- beta -------------------------------------------------------
    alphas = np.concatenate(
