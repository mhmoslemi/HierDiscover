```
    while time.time() < deadline:

        fa = np.fft.rfft(a, size)

        max_conv = conv.max()

        lambda_val = 1.0 + 0.5 * iteration / 1000.0   # maybe adjust
```

```python
"""
Gradient‑based optimizer for the discrete self‑convolution L∞ problem.

The algorithm:
1. Initialise a non‑negative sequence (the analytic “arcsine” shape) and
   normalise it to unit ℓ₁ norm (the evaluator is scale‑invariant).
2. Repeatedly minimise a smooth approximation of the max‑convolution
   ‑f(a)=max_k (a⋆a)[k] using a soft‑max (log‑sum‑exp) with a temperature
   λ that is annealed from a low to a high value.  The gradient of the
   soft‑max can be written as a correlation and is computed efficiently
   with a single FFT per iteration.
3. Apply Adam‑style updates, project onto the non‑negative orthant and
   renormalise the ℓ₁‑norm after each step.
4. Every few hundred iterations the exact evaluator is called; the best
   sequence is remembered.
5. After the main gradient phase a short simulated‑annealing refinement
   (transfer / swap moves) is performed to escape possible shallow local
   minima.

The routine respects the 1000‑second budget, prints progress and finally
returns the best sequence found as a plain list of floats.
"""
import time
import numpy as np

def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the classical “arcsine” (inverse‑square‑root) sequence."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs) -> list[float]:
    """
    Gradient descent + lightweight local search for the convolution L∞ problem.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the harness may ask for up to 1000 s).

    Returns
    -------
    list[float]
        A list of non‑negative heights (the best sequence found).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initialise the sequence.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        n = init_seq.shape[0]
    else:
        n = 256
        init_seq = build_arcsine_sequence(n)

    # Normalise to ℓ₁ = 1 (the evaluator is scale‑invariant)
    a = np.clip(init_seq, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / a_sum

    # ------------------------------------------------------------------
    # Pre‑compute FFT size (must accommodate both convolution (2n‑1) and
    # correlation (3n‑2)).  Use a power of two for speed.
    # ------------------------------------------------------------------
    fft_len = _next_pow2(3 * n - 2)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # Annealing schedules (percent of the total time)
    lr_start = 0.1
    lr_end   = 0.001
    lam_start = 0.5   # soft‑max temperature (small → smoother)
    lam_end   = 30.0  # larger → closer to true max

    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin
    it = 0
    # We will perform the heavy gradient loop for the bulk of the budget.
    # The remaining time (≈0.5 s) is kept for a quick local refinement.
    while time.time() < deadline:
        it += 1
        # --------------------------------------------------------------
        # 1.  Compute convolution a⋆a (full length 2n‑1).
        # --------------------------------------------------------------
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # --------------------------------------------------------------
        # 2.  Soft‑max weights that approximate the arg‑max of conv.
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / float(budget_s))
        lr = lr_start * (1.0 - frac) + lr_end * frac
        lam = lam_start * (1.0 - frac) + lam_end * frac

        # stability: subtract max before exponentiation
        shifted = conv - conv.max()
        # exp may underflow for large negative numbers – clamp to avoid NaNs
        exp_weights = np.exp(lam * shifted)
        weight_sum = exp_weights.sum()
        if weight_sum == 0.0:   # extremely unlikely, fall back to uniform
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_weights / weight_sum

        # --------------------------------------------------------------
        # 3.  Gradient of the soft‑max objective:
        #     ∇_a  f ≈ 2 * (w ⋆ a)   (correlation of w with a)
        # --------------------------------------------------------------
        fw = np.fft.rfft(w, fft_len)
        # correlation = irfft(conj(F(w)) * F(a))
        corr = np.fft.irfft(np.conj(fw) * fa, fft_len)[: n]
        grad = 2.0 * corr

        # --------------------------------------------------------------
        # 4.  Adam update (projected onto the non‑negative orthant,
        #     then renormalise to ℓ₁ = 1).
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project onto non‑negative reals.
        a_new = np.clip(a_new, 0.0, None)

        # Small random jitter to keep the dynamics alive (scale ~1e‑5).
        jitter = rng.normal(scale=1e-5, size=n)
        a_new += jitter
        a_new = np.clip(a_new, 0.0, None)

        # Renormalise to ℓ₁ = 1 (preserves the denominator of the score).
        s = a_new.sum()
        if s == 0.0:
            # extremely unlikely – reinitialise to a tiny uniform mass
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new = a_new / s

        a = a_new

        # --------------------------------------------------------------
        # 5.  Periodic exact evaluation (every ~200 iterations).
        # --------------------------------------------------------------
        if it % 200 == 0 or it == 1:
            cur_score = float(evaluate_sequence(a.tolist()))
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{it:>6}] new best = {best_score:.6f}")
                if best_score < 1.503:
                    print("[early] target reached")
                    break

    # ------------------------------------------------------------------
    # 6.  Short local refinement (simulated‑annealing style) using the
    #     cheap moves defined in the previous baseline code.  This step
    #     works on the best sequence found so far.
    # ------------------------------------------------------------------
    def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
        n = base.size
        w_min = max(1, n // 200)
        w_max = max(2, n // 12)
        width = int(rng.integers(w_min, w_max + 1))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = start + width
        cand = base.copy()
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start:stop] *= factor
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])
        return np.clip(cand, 0.0, None)

    def transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
        n = base.size
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n - 1))
        if j >= i:
            j += 1
        frac = rng.random() * 0.30
        delta = frac * base[i]
        if delta == 0.0:
            return base.copy()
        cand = base.copy()
        cand[i] -= delta
        cand[j] += delta
        return np.clip(cand, 0.0, None)

    def swap_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
        n = base.size
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n - 1))
        if j >= i:
            j += 1
        cand = base.copy()
        cand[i], cand[j] = cand[j], cand[i]
        return cand

    cur = best_seq.copy()
    cur_score = best_score
    # Allocate a small slice of the remaining time for refinement.
    refine_deadline = start_time + float(budget_s) - 0.05
    refine_it = 0
    while time.time() < refine_deadline:
        refine_it += 1
        move_type = rng.random()
        if move_type < 0.3:
            cand = block_perturbation(cur, rng)
        elif move_type < 0.6:
            cand = transfer_move(cur, rng)
        else:
            cand = swap_move(cur, rng)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                print(f"[refine {refine_it}] new best = {best_score:.6f}")
                if best_score < 1.503:

Gradient‑based optimizer for the discrete self‑convolution L∞ problem.

The algorithm:
1. Initialise a non‑negative sequence (the analytic “arcsine” shape) and
   normalise it to unit ℓ₁ norm (the evaluator is scale‑invariant).
2. Repeatedly minimise a smooth approximation of the max‑convolution
   f(a)=max_k (a⋆a)[k] using a soft‑max (log‑sum‑exp) with a temperature
   λ that is annealed from a low to a high value.  The gradient of the
   soft‑max can be written as a correlation and is computed efficiently
   with a single FFT per iteration.
3. Apply Adam‑style updates, project onto the non‑negative orthant and
   renormalise the ℓ₁‑norm after each step.
4. Every few hundred iterations the exact evaluator is called; the best
   sequence is remembered.
5. After the main gradient phase a short simulated‑annealing refinement
   (transfer / swap moves) is performed to escape possible shallow local
   minima.

The routine respects the 1000‑second budget, prints progress and finally
returns the best sequence found as a plain list of floats.
"""
import time
import numpy as np

def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the classical “arcsine” (inverse‑square‑root) sequence."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

    beta2 = 0.999

    it = 0

    # Main gradient‑descent loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # 1. Convolution a⋆a (full length 2n‑1)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        delta = frac * base[i]

        cand = base.copy()
        cand[i] -= delta
```
