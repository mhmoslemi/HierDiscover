def solve_lp(base, M):
    import numpy as np
    from scipy.optimize import linprog
    n = len(base)
    K = 2 * n - 1
    # Build constraint matrix efficiently
    # We'll allocate dense matrix
    A = np.zeros((K, n), dtype=np.float64)
    # For each col i, rows i..i+n-1 get base
    for i in range(n):
        A[i:i+n, i] = base
    b = np.full(K, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum
    bounds = [(0.0, 1000.0) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

def solve_lp(base, M):
    n=len(base)
    K=2*n-1
    # Build matrix
    # Using np.zeros may be large but for n<=512 it's okay.
    A = np.zeros((K, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = base
    b = np.full(K, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, 1000.0) for _ in range(n)]
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

"""
Linear‑programming based iterative refinement.

The algorithm starts from the classic arcsine (inverse‑square‑root) step
function, computes its maximal auto‑convolution M and then repeatedly solves
the linear program

    maximise   Σ g_i
    subject to g_i ≥ 0 , g_i ≤ 1000
               (f * g)_k ≤ M  for all k

where f is the current shape.  The LP finds a companion sequence g that can
increase the total ℓ¹‑mass without worsening the convolution bound.
We scale g to have the same total sum as f and blend the two shapes with a
small mixing factor t.  Because the convolution is quadratic, a sufficiently
small t guarantees that the new maximal convolution does not increase, while
the sum typically grows – this drives the evaluation score
    2·n·max(conv) / (sum²)
downwards.  The process is repeated until the time budget expires (or no
further improvement is observed).  If SciPy is unavailable we fall back to a
simple simulated‑annealing optimiser (the one from the reference solution).

The returned list of non‑negative heights can be of any length; we use a
moderate size (n≈256) so that the LP can be solved quickly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Utility – arcsine (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """
    Return the discrete arcsine sequence a_i = binom(2i,i)/4**i,
    i.e. a[i] = a[i-1] * (2*i-1)/(2*i), with a[0]=1.
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# 2.  Linear‑programming sub‑routine (requires SciPy).
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None  # type: ignore


def solve_lp_for_g(base: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear programme
        maximise   Σ g_i
        subject to g_i ≥ 0 , g_i ≤ 1000
                   (base * g)_k ≤ M   for all k
    Returns the optimal g as a 1‑D NumPy array or ``None`` on failure.
    """
    if linprog is None:                     # SciPy not available
        return None

    n = base.size
    K = 2 * n - 1                           # length of the convolution

    # Build the Toeplitz matrix that encodes (base * g)_k.
    # Row k contains base_{k-j} for column j (zero outside the support).
    A = np.zeros((K, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = base                 # shifted copy of base

    b = np.full(K, M, dtype=np.float64)      # right‑hand side  ≤ M
    c = -np.ones(n, dtype=np.float64)       # maximise sum(g)  → minimise -sum(g)
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # Use the high‑performance HiGHS interior‑point solver.
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method="highs", options={"presolve": True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


# ----------------------------------------------------------------------
# 3.  Very simple simulated‑annealing fallback (identical to the reference).
# ----------------------------------------------------------------------
def fallback_sa(seed: int, budget_s: float) -> list[float]:
    """
    Minimal simulated‑annealing optimiser (the reference implementation).
    Used only when SciPy is unavailable.
    """
    rng = np.random.default_rng(seed)
    n = 256
    base = build_arcsine_sequence(n)
    base = np.clip(base, 0.0, 1000.0)

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur = best.copy()
    cur_score = best_score

    start = time.time()
    deadline = start + max(0.0, budget_s - 2.0)

    # simple move set (block perturbation, transfer, swap, targeted)
    def block_perturbation(arr):
        N = arr.size
        w = max(1, N // 200)
        start = rng.integers(0, max(1, N - w + 1))
        cand = arr.copy()
        factor = float(np.exp(rng.normal(0.0, 0.12)))
        cand[start:start + w] *= factor
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])
        return np.clip(cand, 0.0, 1000.0)

    def transfer_move(arr):
        N = arr.size
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        frac = rng.random() * 0.30
        delta = frac * arr[i]
        cand = arr.copy()
        cand[i] -= delta
        cand[j] += delta
        return np.clip(cand, 0.0, 1000.0)

    def swap_move(arr):
        N = arr.size
        i = rng.integers(N)
        j = rng.integers(N - 1)
        if j >= i:
            j += 1
        cand = arr.copy()
        cand[i], cand[j] = cand[j], cand[i]
        return cand

    def targeted_move(arr, conv, k_max):
        N = arr.size
        i0 = max(0, k_max - (N - 1))
        i1 = min(N - 1, k_max)
        idx = np.arange(i0, i1 + 1)
        partners = k_max - idx
        products = arr[idx] * arr[partners]
        imax = int(np.argmax(products))
        i = int(idx[imax])
        j = int(partners[imax])
        l1, l2 = rng.choice(N, size=2, replace=False)
        max_move = min(arr[i], arr[j])
        if max_move == 0:
            return arr.copy()
        delta = rng.random() * 0.20 * max_move
        cand = arr.copy()
        if i == j:
            cand[i] -= 2.0 * delta
        else:
            cand[i] -= delta
            cand[j] -= delta
        cand[l1] += delta
        cand[l2] += delta
        return np.clip(cand, 0.0, 1000.0)

    it = 0
    T_start, T_end = 0.02, 0.001

    while time.time() < deadline:
        it += 1
        move = rng.random()
        if move < 0.25:
            cand = block_perturbation(cur, rng)
        elif move < 0.5:
            cand = transfer_move(cur, rng)
        elif move < 0.7:
            cand = swap_move(cur, rng)
        else:
            conv = np.convolve(cur, cur)
            k_max = int(np.argmax(conv))
            cand = targeted_move(cur, conv, k_max, rng)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
            continue

        elapsed = time.time() - start
        frac = elapsed / float(budget_s)
        T = T_start * (1.0 - frac) + T_end * frac

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                print(f"[{it:>6}] new best = {best_score:.6f}")

        if it % 5000 == 0 and rng.random() < 0.5:
            cur = np.clip(build_arcsine_sequence(n), 0.0, 1000.0)

    print(f"[fallback final] best score {best_score:.6f} after {it} iters")
    return best.tolist()


# ----------------------------------------------------------------------
# 4.  Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a score below
    the baseline arcsine construction.  The routine runs a linear‑programming
    driven refinement loop for up to *budget_s* seconds.  If SciPy is missing
    it falls back to a simple simulated‑annealing optimiser.
    """
    rng = np.random.default_rng(seed)

    # ----- Problem size ----------------------------------------------------
    # Use a modest length so that the LP stays small.
    # If the harness provides a reference sequence we ignore its length
    # (its size 5582 is far too large for the LP).
    n = 256

    # ----- Initial shape ---------------------------------------------------
    base_seq = build_arcsine_sequence(n)
    base_seq = np.clip(base_seq, 0.0, 1000.0)

    # ----- Baseline evaluation ---------------------------------------------
    best_seq = base_seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    cur_seq = best_seq.copy()
    cur_score = best_score

    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # safety margin

    iteration = 0
    max_iterations = 300  # we will stop earlier if time runs out

    # ----- Main LP‑driven loop ---------------------------------------------
    while time.time() < deadline and iteration < max_iterations:
        iteration += 1

        # 1. current convolution bound
        conv_cur = np.convolve(cur_seq, cur_seq)
        M_cur = float(np.max(conv_cur))

        # 2. solve LP for a companion sequence g
        g = solve_lp_for_g(cur_seq, M_cur)
        if g is None:
            print("[lp] failed – falling back to simple SA")
            # fall back to simulated annealing for the remaining time
            remaining = deadline - time.time()
            if remaining > 0:
                return fallback_sa(seed + iteration, remaining)
            break

        # 3. scale g to have the same total sum as the current sequence
        sum_cur = float(np.sum(cur_seq))
        sum_g = float(np.sum(g))
        if sum_g == 0.0:
            # degenerate solution – abort LP phase
            print("[lp] zero‑mass solution – aborting LP loop")
            break
        scale = sum_cur / sum_g
        g_scaled = g * scale

        # 4. choose a mixing factor t.
        # We pick a conservative t so that the self‑convolution of g does not
        # dominate.  Estimate the maximal convolution of g_scaled.
        conv_g = np.convolve(g_scaled, g_scaled)
        M_g = float(np.max(conv_g))
        # If M_g ≤ M_cur we can use a relatively large t; otherwise shrink t.
        if M_g > 0:
            t_candidate = np.sqrt(M_cur / M_g) * 0.9
            t = min(0.5, max(0.01, t_candidate))
        else:
            t = 0.5

        # 5. form the candidate by convex combination
        cand_seq = (1.0 - t) * cur_seq + t * g_scaled
        cand_seq = np.clip(cand_seq, 0.0, 1000.0)

        cand_score = float(evaluate_sequence(cand_seq.tolist()))
        if not np.isfinite(cand_score):
            # Should never happen because of clipping, but be safe.
            continue

        # 6. accept if we have an improvement
        if cand_score < cur_score:
            cur_seq = cand_seq
            cur_score = cand_score
            if cand_score < best_score:
                best_seq = cand_seq.copy()
                best_score = cand_score
                print(f"[iter {iteration:>4}] new best = {best_score:.6f}")

            # Optional early exit when we already beat the target.
            if best_score < 1.503:
                print("[early stop] target reached")
                break
        else:
            # Occasionally take a small random perturbation to escape plateaus
            if iteration % 20 == 0 and rng.random() < 0.15:
                # tiny log‑normal jitter on a random block
                w = max(1, n // 100)
                start = rng.integers(0, n - w + 1)
                factor = float(np.exp(rng.normal(0.0, 0.07)))
                tmp = cur_seq.copy()
                tmp[start:start + w] *= factor
                tmp = np.clip(tmp, 0.0, 1000.0)
                tmp_score = float(evaluate_sequence(tmp.tolist()))
                if tmp_score < cur_score:
                    cur_seq = tmp
                    cur_score = tmp_score
                    if tmp_score < best_score:
                        best_seq = tmp.copy()
                        best_score = tmp_score
                        print(f"[perturb] new best = {best_score:.6f}")

    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best_seq]
