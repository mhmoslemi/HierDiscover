def solve_lp(f, M):
    n = f.shape[0]
    T = conv_matrix(f)  # shape (2n-1, n)

    constraints = [T @ b <= M]

"""
Iterative linear‑programming refinement of a step‑function.

The evaluation function (after normalising Σa = √(2 n)) equals the
maximum entry of the self‑convolution.  An arcsine‑shaped (Beta(½,½))
distribution already gives an almost flat convolution, but we can
still improve the L∞‑norm by the following two‑phase scheme:

1.  A cheap grid‑search over a few lengths `n` and Beta‑shapes
    (including the arcsine case) using an FFT based objective.
    The best candidate is kept as the initial sequence `f`.

2.  Starting from this `f` we run an iterative LP‑step:
        – compute `M = max(f*f)`.
        – solve the linear programme
              maximise   Σ b_j
              subject to  f * b ≤ M  (full linear convolution)
                         b_j ≥ 0
        – rescale `b` so that Σb = √(2 n) and mix
              f ← (1‑t)·f + t·b   (t≈0.2)
        – keep the best sequence seen.
    The LP is solved with CVXPY (ECOS).  Because each LP reduces the
    weight on the “peak” part of `f`, the maximum convolution value
    drops monotonically (empirically).  After the LP loop a short
    random‑perturbation phase fine‑tunes the result.

The whole routine respects the supplied time budget (default 950 s)
and finally returns the best sequence as a plain Python list.
"""
import time
import numpy as np
from math import sqrt
import cvxpy as cp

# ----------------------------------------------------------------------
# Fast evaluation of the objective (max of self‑convolution)
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray) -> float:
    """Return the evaluation score defined by `evaluate_sequence`
    for a sequence `a`.  The sum is assumed to be >0."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Generate a symmetric Beta(α,α)‑shaped sequence of length n.
# ----------------------------------------------------------------------
def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, alpha - 1.0)
    target_sum = sqrt(2.0 * n)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    a = w * (target_sum / w_sum)
    a = np.clip(a, 0.0, 1000.0)
    return a


# ----------------------------------------------------------------------
# Build the full (linear) convolution matrix for a given kernel f.
# For each shift k (0 … 2n‑2) we have (T @ b)[k] = (f * b)[k].
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    L = 2 * n - 1
    T = np.zeros((L, n), dtype=np.float64)
    # each column i contains f[i] on a band of length n
    for i in range(n):
        T[i:i + n, i] = f[i]
    return T


# ----------------------------------------------------------------------
# Solve the LP: maximise Σb subject to f*b ≤ M and b ≥ 0.
# Returns the optimal b as a NumPy array (or None on failure).
# ----------------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float, warm_start: bool = False):
    n = f.shape[0]
    T = _conv_matrix(f)          # (2n‑1) × n dense matrix
    b = cp.Variable(n, nonneg=True)
    constraints = [T @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=warm_start, max_iters=500)
    except cp.error.SolverError:
        return None
    if b.value is None:
        return None
    # numerical cleanup
    val = np.maximum(0.0, np.asarray(b.value, dtype=np.float64))
    return val


# ----------------------------------------------------------------------
# Main entry point required by the runner
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 950, **kwargs):
    """Return a list of non‑negative heights (the candidate step function)."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # safety margin

    # ------------------------------------------------------------------
    # 0.  Global fallback (if the host supplies a previous sequence)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        seq0 = np.clip(seq0, 0.0, 1000.0)
        if seq0.ndim == 1 and seq0.size > 0:
            sc0 = _fast_objective(seq0)
            best_seq = seq0.copy()
            best_score = sc0
            print(f"[init] global height_sequence_1 score = {sc0:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Grid search over lengths and Beta‑shapes (cheap FFT based)
    # ------------------------------------------------------------------
    # We restrict the grid to moderate lengths – large n does not help
    # the eventual LP step (its size would explode).
    n_grid = [256, 512, 1024, 2048]          # keep LP tractable
    # dense sampling of α around the arcsine value 0.5
    alpha_grid = np.unique(
        np.concatenate((
            np.linspace(0.15, 0.9, 16),
            np.linspace(0.35, 0.65, 13),
        ))
    )
    for n in n_grid:
        if time.monotonic() >= deadline:
            break
        target_sum = sqrt(2.0 * n)          # normalisation used by the objective
        for alpha in alpha_grid:
            if time.monotonic() >= deadline:
                break
            a = _beta_sequence(n, alpha)
            # fast exact objective (since we normalised the sum)
            sc = _fast_objective(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                best_n = n
                best_alpha = alpha
                print(f"[grid] n={n:5d} α={alpha:5.3f} → score={sc:.6f}")

    # If the grid never found a candidate, fall back to a modest arcsine.
    if best_seq is None:
        best_n = 512
        best_seq = _beta_sequence(best_n, 0.5)
        best_score = _fast_objective(best_seq)

    # ------------------------------------------------------------------
    # 2.  LP‑based iterative refinement (run only on a modest n)
    # ------------------------------------------------------------------
    # The LP size grows quadratically with n, so we keep n ≤ 1024.
    refine_n = min(best_n, 1024)
    if refine_n != best_n:
        # Resample the current best sequence onto the smaller grid.
        # Simple nearest‑neighbour down‑sampling preserves the shape roughly.
        factor = best_n // refine_n
        best_seq = best_seq[:factor * refine_n].reshape(refine_n, factor).mean(axis=1)
        best_score = _fast_objective(best_seq)
        best_n = refine_n
        print(f"[downsample] using n={best_n} for LP refinement")

    # Ensure exact normalisation
    target_sum = sqrt(2.0 * best_n)
    best_seq = best_seq * (target_sum / best_seq.sum())
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Parameters for the LP loop
    f = best_seq.copy()
    t_mix = 0.25                     # mixing weight
    max_lp_iters = 200
    lp_iter = 0
    warm = False                    # first solve does not have a warm start
    print(f"[lp] starting refinement on n={best_n}")

    while time.monotonic() < deadline and lp_iter < max_lp_iters:
        lp_iter += 1
        # current maximal convolution entry
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # solve the linear programme
        b_opt = _solve_lp(f, M, warm_start=warm)
        warm = True
        if b_opt is None or b_opt.sum() <= 0.0:
            print("[lp] LP failed – exiting refinement")
            break

        # rescale to keep the L1 norm fixed
        scale = target_sum / b_opt.sum()
        g = b_opt * scale

        # convex combination
        f_new = (1.0 - t_mix) * f + t_mix * g
        # renormalise (numerical drift)
        f_new = f_new * (target_sum / f_new.sum())

        # evaluate new max‑convolution
        new_M = np.convolve(f_new, f_new).max()
        if new_M < best_score:
            best_score = new_M
            best_seq = f_new.copy()
            f = f_new
            print(f"[lp] iter {lp_iter:3d}: improved score = {new_M:.6f}")
            # keep the mixing weight, maybe increase it a little
            t_mix = min(0.5, t_mix * 1.1)
        else:
            # no improvement – shrink the mixing step
            t_mix = max(0.05, t_mix * 0.7)
            # still move towards g to escape possible plateau
            f = f_new
            # If mixing weight becomes tiny we stop
            if t_mix <= 0.05 and lp_iter > 10:
                print("[lp] mixing weight too small, terminating LP loop")
                break

    # ------------------------------------------------------------------
    # 3.  Tiny random perturbations – stochastic fine‑tuning
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        a = best_seq.copy()
        n = a.size
        step_scale = target_sum / (20.0 * n)   # a few percent of a typical entry
        while time.monotonic() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * step_scale
            if a[i] < delta:
                continue
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            # re‑normalise to keep the sum exact
            cand = cand * (target_sum / cand.sum())
            cand = np.clip(cand, 0.0, 1000.0)
            cand_score = np.max(np.convolve(cand, cand))
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                a = cand
                if rng.random() < 0.02:
                    print(f"[perturb] new best = {cand_score:.6f}")
            else:
                a = cand  # keep walking

    # ------------------------------------------------------------------
    # 4.  Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] evaluated with evaluate_sequence → score = {final_score:.6f}")

    # Return plain Python list
    return [float(v) for v in best_seq]
