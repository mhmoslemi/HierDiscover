"""
Hybrid simulated‑annealing + projected‑gradient search for the
self‑convolution bound problem.

The algorithm works with a population of a few candidate sequences.
Each candidate is a non‑negative vector with fixed ℓ₁‑norm (the sum is
set to 1 – the objective is scale‑invariant).  At every iteration a
random parent is mutated by one of several cheap moves:
    * multiplicative block scaling,
    * pairwise mass transfer,
    * early‑to‑late transfer,
    * a gradient‑guided transfer that lowers the current maximal
      convolution entry,
    * a targeted “heavy‑to‑light” move that explicitly reduces the
      products a_i·a_{k_max‑i} that realise the current maximum.
The new vector is projected onto the simplex (non‑negative, prescribed
sum) and evaluated.  Improvements replace the parent; the best
overall sequence is remembered.  A simple temperature schedule allows
occasional uphill moves (simulated annealing) to escape plateaus.
The search runs until the allotted time expires and returns the best
sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1.  Utility functions (projection, convolution, scoring)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1, dtype=np.float64)
    # find the rho index (last where u_i > cssv_i / i)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def convolve_fft(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of seq with itself using an FFT of power‑of‑two length."""
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fft_a = np.fft.rfft(seq, size)
    prod = fft_a * fft_a
    conv = np.fft.irfft(prod, size)[: 2 * n - 1]
    return conv


def sequence_score(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(b)/(sum(a)²) for a given non‑negative sequence."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    b = convolve_fft(seq)
    max_b = b.max()
    n = seq.shape[0]
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# 2.  Construction of analytic starting shapes
# ----------------------------------------------------------------------


def build_arcsine_sequence(length: int) -> np.ndarray:
    """Discrete arcsine (inverse‑square‑root) shape a_i = C·binom(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


def build_powerlaw_sequence(length: int, p: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i + offset)^{-p}.  Returned vector is already ℓ₁‑normalised."""
    idx = np.arange(length, dtype=np.float64) + offset + 1.0
    a = idx ** (-p)
    a = project_onto_simplex(a, 1.0)
    return a

# 3.  Primitive mutation operators (all keep the ℓ₁‑norm via projection)
# ----------------------------------------------------------------------


def block_perturbation(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Scale a random contiguous block by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand = base.copy()
    cand[start:stop] *= factor
    if rng.random() < 0.35:          # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return project_onto_simplex(cand, target_sum)


def transfer_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """Move a random fraction (≤30 %) of mass from one index to another."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return project_onto_simplex(cand, target_sum)


def early_to_late_move(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Shift a fraction of mass from the first 10 % of entries to the last 10 %."""
    n = base.size
    early_cut = max(1, n // 10)
    late_start = max(early_cut, n - early_cut)
    i = int(rng.integers(0, early_cut))
    j = int(rng.integers(late_start, n))
    if i == j:
        return base.copy()
    frac = rng.random() * 0.40
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return project_onto_simplex(cand, target_sum)


def gradient_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float, max_frac: float = 0.30) -> np.ndarray:
    """
    Sub‑gradient step: move a fraction of the coefficient that contributes most
    to the current maximal convolution entry towards a light coefficient.
    """
    n = base.size
    b = convolve_fft(base)
    max_b = b.max()
    # mask of almost‑maximal positions (tolerance = 1e-12 relative)
    tol = max_b * (1.0 - 1e-12)
    mask = (b >= tol).astype(np.float64)           # length = 2n‑1
    # sub‑gradient of max(b) is 2 * (base * (mask convolved with base))
    grad = 2.0 * np.convolve(base, mask)[:n]
    heavy_idx = int(np.argmax(grad))
    # pick a light index: preferably a low‑gradient coordinate
    light_candidates = np.where(grad < grad.mean())[0]
    light_candidates = light_candidates[light_candidates != heavy_idx]
    if light_candidates.size == 0:
        light_idx = (heavy_idx + 1) % n
    else:
        light_idx = int(rng.choice(light_candidates))

    frac = rng.random() * max_frac
    delta = frac * base[heavy_idx]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[heavy_idx] -= delta
    cand[light_idx] += delta
    return project_onto_simplex(cand, target_sum)


def large_transfer_move(base: np.ndarray, rng: np.random.Generator,
                        target_sum: float) -> np.ndarray:
    """
    Move a large fraction (up to 90 %) from the heaviest contributor to a
    light one, based on the current maximal convolution index.
    """
    n = base.size
    b = convolve_fft(base)
    max_b = b.max()
    # choose one of the (possibly multiple) maximal positions
    k_candidates = np.where(b >= max_b * (1 - 1e-12))[0]
    k = int(rng.choice(k_candidates))
    # contribution of each i to this maximal entry (product a_i·a_{k-i})
    start = max(0, k - (n - 1))
    stop = min(n - 1, k)
    idx = np.arange(start, stop + 1, dtype=int)
    contrib = base[idx] * base[k - idx]
    heavy_idx = int(idx[np.argmax(contrib)])
    # try to find a light partner
    light_candidates = np.where(contrib <= contrib.mean())[0]
    if light_candidates.size == 0:
        light_idx = (heavy_idx + 1) % n
    else:
        # map back to original indices
        light_idx = int(idx[light_candidates[rng.integers(0, light_candidates.size)]])
    # transfer up to 90 % of heavy mass
    frac = rng.random() * 0.90
    delta = frac * base[heavy_idx]
    cand = base.copy()
    cand[heavy_idx] -= delta
    cand[light_idx] += delta
    return project_onto_simplex(cand, target_sum)


def targeted_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """
    Exhaustively test a small set of destinations for the heaviest
    contributor to the current maximal convolution value and keep the
    best improvement (if any).  Falls back to ``gradient_move`` when no
    improvement is found.
    """
    n = base.size
    b = convolve_fft(base)
    max_b = b.max()
    k_candidates = np.where(b >= max_b * (1 - 1e-12))[0]
    k = int(rng.choice(k_candidates))

    # contributions for this k
    start = max(0, k - (n - 1))
    stop = min(n - 1, k)
    idx = np.arange(start, stop + 1, dtype=int)
    contrib = base[idx] * base[k - idx]
    heavy_idx = int(idx[np.argmax(contrib)])

    # sample a few potential receivers
    num_samples = min(15, n - 1)
    possible = np.delete(np.arange(n), heavy_idx)
    receivers = rng.choice(possible, size=num_samples, replace=False)

    best_cand = None
    best_score = np.inf

    # test a few transfer fractions
    for recv in receivers:
        for frac in (0.15, 0.30, 0.45):
            delta = frac * base[heavy_idx]
            cand = base.copy()
            cand[heavy_idx] -= delta
            cand[recv] += delta
            cand = project_onto_simplex(cand, target_sum)
            sc = sequence_score(cand)
            if sc < best_score:
                best_score = sc
                best_cand = cand

    if best_cand is not None and best_score < sequence_score(base):
        return best_cand
    # otherwise fall back to a cheaper gradient step
    return gradient_move(base, rng, target_sum)

# 4.  Main entry point – the harness will call this function
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The routine runs for up to ``budget_s`` seconds
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – try to reuse the reference construction if present.
    # ------------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256

    target_sum = 1.0                     # scale does not matter (objective is homogeneous)

    # ------------------------------------------------------------------
    # Initialise a small population of diverse candidates.
    # ------------------------------------------------------------------
    pop_size = 5
    population = []
    scores = []

    # 1) analytic arcsine shape
    arcsine = build_arcsine_sequence(n)
    arcsine = project_onto_simplex(arcsine, target_sum)
    population.append(arcsine)
    scores.append(sequence_score(arcsine))

    # 2) uniform distribution (often a decent baseline)
    uniform = np.full(n, target_sum / n, dtype=np.float64)
    population.append(uniform)
    scores.append(sequence_score(uniform))

    # 3) a few power‑law variations
    for p, off in [(0.48, 0.0), (0.52, 0.0), (0.5, 1.0), (0.5, 2.0)]:
        seq = build_powerlaw_sequence(n, p, offset=off)
        population.append(seq)
        scores.append(sequence_score(seq))

    # 4) a random Dirichlet sample
    rand = rng.exponential(size=n)
    rand = project_onto_simplex(rand, target_sum)
    population.append(rand)
    scores.append(sequence_score(rand))

    # Keep the best seen so far
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_score = scores[best_idx]

    # ------------------------------------------------------------------
    # Simulated annealing schedule
    # ------------------------------------------------------------------
    start_temp = max(0.02 * best_score, 1e-4)
    temperature = start_temp
    # exponential decay that brings temperature close to zero before the deadline
    decay_per_sec = 0.95
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    it = 0
    last_improve = it

    while time.monotonic() < deadline:
        it += 1

        # pick a parent uniformly from the population
        parent_idx = int(rng.integers(0, len(population)))
        parent = population[parent_idx]
        parent_score = scores[parent_idx]

        # ------------------------------------------------------------------
        #  Choose a mutation operator (probabilities tuned empirically)
        # ------------------------------------------------------------------
        r = rng.random()
        if r < 0.20:
            cand = block_perturbation(parent, rng, target_sum)
        elif r < 0.40:
            cand = transfer_move(parent, rng, target_sum)
        elif r < 0.55:
            cand = early_to_late_move(parent, rng, target_sum)
        elif r < 0.70:
            cand = gradient_move(parent, rng, target_sum)
        elif r < 0.85:
            cand = large_transfer_move(parent, rng, target_sum)
        else:
            cand = targeted_move(parent, rng, target_sum)

        cand_score = sequence_score(cand)
        delta = cand_score - parent_score

        # accept if better, otherwise with Metropolis probability
        if delta < 0 or rng.random() < math.exp(-delta / temperature):
            # replace parent in the population
            population[parent_idx] = cand
            scores[parent_idx] = cand_score

            # update global best
            if cand_score < best_score:
                best_seq = cand.copy()
                best_score = cand_score
                best_idx = parent_idx
                last_improve = it
                print(f"[{it:>6}] new global best = {best_score:.6f}")

        # temperature schedule (time‑based)
        elapsed = time.monotonic() + 0.0 - (deadline - budget_s)
        temperature = start_temp * (decay_per_sec ** (elapsed / 1.0))

        # occasional diversification: if we have been stuck for a while,
        # replace a random individual with a jittered copy of the best.
        if it % 20000 == 0:
            if it - last_improve > 60000:
                idx = int(rng.integers(0, len(population)))
                jitter = np.exp(rng.normal(0.0, 0.08, size=n))
                new_seq = best_seq * jitter
                new_seq = project_onto_simplex(new_seq, target_sum)
                population[idx] = new_seq
                scores[idx] = sequence_score(new_seq)
                print(f"[{it}] diversification step (stagnation)")

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Clip to the allowed range just in case (values are far below 1000 anyway).
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # Return as a plain Python list of floats.
    return [float(x) for x in best_seq]
