def _gradient_step(a, lam, step):
    # compute convolution
    n = len(a)
    size = 2*n - 1
    m = 1 << ((size-1).bit_length())
    A = np.fft.rfft(a, n=m)
    b = np.fft.irfft(A*A, n=m)[:size]
    # compute weights
    maxb = np.max(b)
    # Use stable softmax: w_i = exp(lam * (b_i - maxb))  (so max weight=1)
    w = np.exp(lam * (b - maxb))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum
    # compute gradient
    grad = _gradient_weights(w, a)  # returns vector length n
    # step: a - step*grad
    a_new = a - step * grad
    # project
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s > 0:
        a_new = a_new * (np.sqrt(2 * n) / s)
    else:
        # rare; fallback to previous.
        a_new = a
    return a_new, np.max(b)

Fast step‑function search for a low evaluation score.

The algorithm consists of three stages:

1. **Coarse global search** – enumerate a few lengths `n` (powers of two) and
   generate a rich family of Beta‑shaped sequences
   a_i ∝ p_i^{α‑1}(1‑p_i)^{β‑1} (p_i = (i+0.5)/n).  Both α and β are
   scanned on a modest grid, the sequences are rescaled so that
   Σa_i = sqrt(2·n) (the scaling makes the evaluation function equal to the
   maximum of the self‑convolution).  The best candidate so far is kept.

2. **Local smooth optimisation** – starting from the best sequence,
   minimise the max‑convolution using projected gradient descent with a
   soft‑max approximation of the `max` operator.  Each iteration evaluates
   the self‑convolution via FFT, builds weights w_i ∝ exp(λ·(b_i‑max(b))),
   computes the gradient ∂(Σ w·(a*a))/∂a using another FFT‑based convolution,
   takes a small step, projects onto the non‑negative simplex and re‑symmetrises
   the vector (the problem is symmetric, so enforcing symmetry never hurts).

3. **Final evaluation** – the best sequence found is finally evaluated with the
   supplied `evaluate_sequence` (which also enforces clipping to [0,1000]).
   The function returns the sequence as a plain Python list.

All heavy work is performed with vectorised NumPy/FFT operations, so the
search fits comfortably into the time budget (≈ 500 s) while significantly
improving the previous score (target ≈ 1.503)."""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Fast evaluation of the objective (max of self‑convolution)
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray) -> float:
    """Return 2·n·max(b)/sum(a)^2 using FFT (no clipping)."""
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)

# Generate Beta‑shaped (possibly asymmetric) sequence
# ----------------------------------------------------------------------
def _beta_general_sequence(
    n: int, alpha: float, beta_: float
) -> np.ndarray:
    """
    Build a non‑negative sequence of length n with shape
        p^{α‑1} (1‑p)^{β‑1},  p = (i+0.5)/n.
    The sequence is scaled so that its sum equals sqrt(2·n).
    """
    p = (np.arange(n, dtype=np.float64) + 0.5) / n

    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    a = w * (target_sum / w_sum)
    a = np.clip(a, 0.0, 1000.0)
    return a


# ----------------------------------------------------------------------
# Soft‑max weights for a smooth maximum
# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, temperature: float) -> np.ndarray:
    """Stable soft‑max on the convolution vector `b`."""
    b_max = np.max(b)
    # subtract max to avoid overflow
    ex = np.exp(temperature * (b - b_max))
    s = ex.sum()
    if s == 0.0:
        return np.full_like(b, 1.0 / b.size)
    return ex / s


# ----------------------------------------------------------------------
# Gradient of Σ w_i·(a*a)_i with respect to a (using FFT)
# ----------------------------------------------------------------------
def _gradient_from_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_i w_i·(a*a)_i = 2·(w * a_rev)[n‑1 : n‑1+n],
    where * denotes linear convolution and a_rev is a reversed.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv_len = w.size + a_rev.size - 1
    m = 1 << ((conv_len - 1).bit_length())
    W = np.fft.rfft(w, n=m)
    A_rev = np.fft.rfft(a_rev, n=m)
    conv = np.fft.irfft(W * A_rev, n=m)[:conv_len]
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
# One projected gradient step (soft‑max smoothed max)
# ----------------------------------------------------------------------
def _gradient_step(
    a: np.ndarray, temperature: float, step_size: float
) -> tuple[np.ndarray, float]:
    """
    Perform a single projected gradient descent step on `a`.
    Returns the updated sequence and the (unsmoothed) max convolution.
    """
    n = a.shape[0]
    # self‑convolution
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    b = np.fft.irfft(A * A, n=m)[:size]
    max_b = np.max(b)

    # soft‑max weights for a smooth upper‑bound
    w = _softmax_weights(b, temperature)

    # gradient w.r.t. a
    grad = _gradient_from_weights(w, a)

    # gradient step (move opposite to gradient to reduce max)
    a_new = a - step_size * grad

    # positivity & simplex projection (sum = sqrt(2·n))
    a_new = np.clip(a_new, 0.0, None)
    target_sum = sqrt(2.0 * n)
    s = a_new.sum()
    if s > 0.0:
        a_new = a_new * (target_sum / s)
    else:
        # fallback – should never happen
        a_new = np.full_like(a, target_sum / n)

    # enforce the natural symmetry of the problem
    a_new = (a_new + a_new[::-1]) * 0.5

    # final hard clipping (required by `evaluate_sequence`)
    a_new = np.clip(a_new, 0.0, 1000.0)
    return a_new, max_b
