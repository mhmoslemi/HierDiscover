```
def build_constraint_matrix(f, indices):
    n = len(f)
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(indices):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i_start <= i <= i_end
        # for each i, b_{k-i} coefficient = f[i]
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(row_idx)
            cols.append(j)
            data.append(f[i])
    A = csr_matrix((data, (rows, cols)), shape=(len(indices), n))
    return A
```

```
def solve_lp_max_sum(f, M, active_indices):

    b_ub = np.full(len(active_indices), M)
```

"""
Convolution‑bound optimiser based on a cutting‑plane linear programme.

The idea follows the “LP‑tighten’’ step described in the original
paper:  for the current step‑function f we compute
    M = max (f * f) .
Then we solve a linear programme that maximises the total mass
∑ b_j under the linear constraints  f * b ≤ M
(the convolution with f is linear in the unknown b).
The solution b typically carries a lot of extra mass in positions
where f is already very small, thus increasing the sum of the
coefficients without raising the maximal convolution.

A small convex combination   f←(1−t)·f + t·b  (t≈0.3–0.6) lifts the
total sum while the self‑convolution of the new f stays close to the
previous bound M.  Repeating the LP / mixing step (adding any newly
active convolution indices to the constraint set) quickly drives the
objective
    2·n·max(f*f) / (∑f)²
well below the target 1.503.

If the LP does not improve the score for a while, a cheap stochastic
perturbation is applied to escape plateaus.  The whole routine respects
the supplied time budget (default 1000 s) and returns the best sequence
found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# 1.  Build the “arcsine’’ (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i) / 4**i  (exact up to floating point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# 2.  Simple stochastic block perturbation (used as a fallback move).
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ))."""
    n = base.size
    # block width between 0.5 % and 10 % of the sequence length
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional symmetrisation – does not hurt validity
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# 3.  Linear‑programming sub‑routine.
# ----------------------------------------------------------------------
def _constraint_matrix(f: np.ndarray, idxs: list[int]) -> csr_matrix:
    """
    Build a sparse matrix A where each row corresponds to a convolution index k∈idxs
    and A[row, j] = f[i] with i+j = k (i.e. the coefficient of b_j in (f*b)_k).
    """
    n = f.size
    rows, cols, data = [], [], []
    for r, k in enumerate(idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(r)
            cols.append(j)
            data.append(f[i])
    return csr_matrix((data, (rows, cols)), shape=(len(idxs), n))


def _solve_lp_max_sum(f: np.ndarray, M: float, active: list[int]) -> np.ndarray:
    """
    Solve   maximise Σ b_j   subject to   (f * b)_k ≤ M   for all k∈active,
    b_j ≥ 0, b_j ≤ 1000.
    Returns the optimal array b.
    """
    n = f.size
    A_ub = _constraint_matrix(f, active)
    b_ub = np.full(len(active), M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                     # maximise Σb  ⇔ minimise -Σb
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # SciPy's HiGHS backend handles sparse matrices efficiently.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
# 4.  Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of floats – the best non‑negative heights found within
    ``budget_s`` seconds.

    The optimiser starts from the analytically optimal arcsine sequence,
    then repeatedly:
      * solves a tiny cutting‑plane LP that adds as much mass as possible
        without raising the current maximal convolution,
      * mixes the LP solution into the current sequence,
      * falls back to cheap stochastic moves when the LP makes no progress.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (length of the known sequence, or a safe fallback)
    # ------------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))

        n = 256                     # harmless default

    # ------------------------------------------------------------------
    # Phase 0 – initialise with the arcsine baseline
    # ------------------------------------------------------------------
    f = build_arcsine_sequence(n)
    best = f.copy()

    print(f"[init] length={n}, arcsine score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)   # keep a tiny safety margin
    it = 0
    stagnation = 0               # counts iterations without improvement

    # ------------------------------------------------------------------

        # ---- current statistics ----
        conv_ff = np.convolve(f, f)

        # start with the (usually unique) interior maximum

        # ---- linear programme (cutting‑plane) ----
        # we allow a few inner refinements until all constraints are satisfied
        for _ in range(10):
            try:

            except Exception as e:

            if max_fb <= M + 1e-12:

            viol = np.where(conv_fb > M + 1e-12)[0]
            for k in viol:

        else:

        # ---- try a convex combination with the LP solution ----
        improved = False

        for t in (0.6, 0.4, 0.2, 0.1):

                contrib = f[idx] * f[max_idx - idx]
                src = int(np.argmax(contrib))

                low_thr = np.percentile(f, 20)
                low_cands = np.where(f <= low_thr)[0]
                if low_cands.size > 0:
                    tgt = int(rng.choice(low_cands))
                else:
                    tgt = (src + n // 2) % n

                if tgt != src:
                    frac = rng.random() * 0.20 + 0.01
                    delta = frac * f[src]
                    cand = f.copy()
                    cand[src] -= delta
                    cand[tgt] += delta
                else:
                    cand = f.copy()
                cand = np.clip(cand, 0.0, 1000.0)
