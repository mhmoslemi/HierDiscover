def fft_convolve(a,b):
    n = a.shape[0] + b.shape[0] - 1
    m = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    B = np.fft.rfft(b, n=m)
    conv = np.fft.irfft(A * B, n=m)[:n]
    return conv

"""
Low‑score step‑function search using exponentiated mirror descent.

The evaluation function is homogeneous: scaling the sequence does not change the
score, therefore we optimise a probability distribution `p` (sum = 1) that
minimises the maximum entry of its self‑convolution `p * p`.  Afterwards the
distribution is scaled to the required sum `sqrt(2·n)`.

Algorithm
---------
1.  Choose a handful of lengths `n`.
2.  Initialise a distribution for each length:
    * uniform,
    * the Beta(½,½) (arcsine) shape,
    * a couple of random Dirichlet draws.
3.  For each initial distribution run an exponentiated‑gradient (mirror‑descent)
    loop:
       * compute the full convolution `c = p * p` by FFT,
       * compute a smoothed max via a soft‑max weight
         `w_i ∝ exp(γ·(c_i – max(c)))`,
       * gradient of the smoothed max is `g = 2·(w * p)[valid]`,
       * update `p ← p·exp(–η·g)` and renormalise to a simplex.
       * slowly increase `γ` (sharper max) and decay `η`.
    The loop stops after a time limit, a maximal number of iterations,
    or when the score has not improved for many steps.
4.  Keep the best distribution across all runs, scale it to sum
    `sqrt(2·n)`, and evaluate it with the supplied `evaluate_sequence`.
5.  Return the best sequence found.

The whole procedure respects the given time budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: FFT‑based full convolution
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two 1‑D float arrays using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    m = 1 << ((n - 1).bit_length())          # next power‑of‑two
    X = np.fft.rfft(x, n=m)
    Y = np.fft.rfft(y, n=m)
    conv = np.fft.irfft(X * Y, n=m)[:n]
    return conv


# ----------------------------------------------------------------------
# Fast evaluation (same as the provided evaluate_sequence but without I/O)
# ----------------------------------------------------------------------
def _fast_objective(seq: np.ndarray) -> float:
    """Return 2·n·max(b)/sum(a)^2 for a sequence (same as evaluate_sequence)."""
    seq = np.clip(seq, 0.0, 1000.0)
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = _fft_convolve(seq, seq)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Beta‑shaped sequence (arcsine when α=0.5)
# ----------------------------------------------------------------------
def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """Beta(α,α) shape, scaled so that sum = sqrt(2·n).  Returned values are clipped."""
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, alpha - 1.0)
    target = np.sqrt(2.0 * n)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    a = w * (target / w_sum)
    a = np.clip(a, 0.0, 1000.0)
    return a


# ----------------------------------------------------------------------
# Mirror‑descent optimiser on the simplex
# ----------------------------------------------------------------------
def _optimize_distribution(p0: np.ndarray,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex to minimise max(p * p).
    Returns the best distribution (sum = 1) and its corresponding score.
    """
    p = p0.copy()
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s <= 0.0:
        p = np.full_like(p, 1.0 / n)
    else:
        p /= s

    # FFT size – must accommodate the longest convolution (w * p) of length 3n‑2
    conv_len = 3 * n - 2
    m_fft = 1 << ((conv_len - 1).bit_length())

    gamma = 5.0          # start with a mild soft‑max
    gamma_max = 400.0
    eta = 0.5            # initial step size
    eta_min = 0.01

    best_score = np.inf
    best_p = p.copy()
    no_improve = 0
    max_no_improve = 250
    iteration = 0
    max_iter = 5000

    while time.monotonic() < deadline and iteration < max_iter:
        # ----- convolution p * p (full) -----
        P_hat = np.fft.rfft(p, n=m_fft)
        conv_pp = np.fft.irfft(P_hat * P_hat, n=m_fft)[: 2 * n - 1]
        cur_max = conv_pp.max()
        cur_score = 2.0 * n * cur_max

        # keep global best for this `n`
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        # ----- soft‑max weights for the max -----
        # shift by cur_max for numerical stability
        w = np.exp(gamma * (conv_pp - cur_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        W_hat = np.fft.rfft(w, n=m_fft)
        conv_wp = np.fft.irfft(W_hat * P_hat, n=m_fft)[: 3 * n - 2]
        grad = 2.0 * conv_wp[n - 1 : n - 1 + n]   # valid part (length n)

        # ----- mirror‑descent update (exponentiated gradient) -----
        # ensure finite gradient
        if not np.all(np.isfinite(grad)):
            break
        p = p * np.exp(-eta * grad)
        # re‑normalise to the simplex
        s = p.sum()
        if s <= 0.0:
            p = np.full_like(p, 1.0 / n)
        else:
            p /= s

        # ----- schedule -----
        gamma = min(gamma * 1.02, gamma_max)
        eta = max(eta * 0.99, eta_min)

        iteration += 1

    return best_p, best_score
