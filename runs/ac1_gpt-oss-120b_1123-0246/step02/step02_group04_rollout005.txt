def build_conv_matrix(a):
    n = len(a)
    diags_data = []
    offsets = []
    for d, val in enumerate(a):
        if val == 0: continue
        offsets.append(d)
        diags_data.append(np.full(n - d, val))
    # shape = (2*n-1, n)
    A = scipy.sparse.diags(diags_data, offsets, shape=(2*n-1, n), format='csr')
    return A

"""Optimization of step function via iterative linear programming.
We start from the analytic arcsine distribution, then repeat:
- compute the current maximal convolution value M,
- solve a linear program that maximizes the total mass of a new
  non‑negative vector b under the linear constraints (a * b) ≤ M,
- rescale b to preserve the L1 norm,
- replace a by a convex combination with the new vector.
Occasional random block perturbations are applied when progress stalls.
The algorithm terminates when the time budget expires or a target score
below 1.503 is reached.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the discrete arcsine (inverse‑square‑root) sequence of given length."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a

def build_conv_matrix(seq: np.ndarray) -> sp.csr_matrix:
    """
    Construct the matrix A such that (A @ b)[k] = (seq * b)[k]
    (linear convolution of ``seq`` with a variable vector ``b``).
    The matrix has shape (2*len(seq)-1, len(seq)) and is built
    from constant diagonals; it is returned in CSR format.
    """
    n = seq.shape[0]
    # Each offset d (0 ≤ d < n) contributes a diagonal with constant value seq[d]
    offsets = np.arange(n, dtype=int)            # 0,1,…,n‑1
    data = [np.full(n, seq[d], dtype=np.float64) for d in range(n)]
    return sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')

def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Random multiplicative perturbation of a contiguous block."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.3:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

                     budget_s: float = 500,

    Return a non‑negative height sequence that (hopefully) improves
    over the standard arcsine construction.
    The routine runs for up to ``budget_s`` seconds, using an
    iterative linear‑programming scheme described in the problem
    statement.  A simple random block perturbation is added when
    the LP step stops making progress.
    """
    rng = np.random.default_rng(seed)

"""Optimization of a non‑negative step function via an iterative linear‑programming
scheme.

We start from the analytically optimal “arcsine’’ (inverse‑square‑root) vector,
normalise its ℓ¹‑norm to √(2 n) (the evaluator is scale‑invariant), and then
repeatedly perform the following block:

1. Compute the current maximal convolution value M = max(a∗a).
2. Solve the linear programme
       maximise   Σ b_j
       subject to 0 ≤ b and (a∗b)_k ≤ M   for all k,
   which asks how much total mass can be packed under the same bound M.
3. Rescale the LP solution b to the original ℓ¹‑norm (call the result g).
4. Replace a by a convex combination (1‑τ)·a + τ·g (τ≈0.5).  The combination is
   guaranteed not to increase M, so the objective decreases monotonically.
5. If several iterations stall we apply a cheap random block perturbation
   (multiplicative scaling of a contiguous block) and continue.

The loop runs until the time budget expires or the bound drops below the
target 1.503.  The final sequence is returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Discrete arcsine (inverse‑square‑root) sequence of given length."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def build_conv_matrix(seq: np.ndarray) -> sp.csr_matrix:
    """
    Return CSR matrix A such that (A @ b)[k] = (seq * b)[k] (linear convolution).
    The matrix has shape (2*len(seq)-1, len(seq)) and consists of constant
    diagonals: A_{k,i}=seq[k-i] for 0≤i<n, 0≤k-i<n, otherwise 0.
    """
    n = seq.shape[0]
    offsets = np.arange(n, dtype=int)                # 0 … n‑1
    data = [np.full(n, seq[d], dtype=np.float64) for d in range(n)]
    return sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative scaling of a random contiguous block (plus optional symmetrisation)."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # log‑normal factor ≈ 1 ± 12 %
    cand[start:stop] *= factor
    if rng.random() < 0.3:                         # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)
