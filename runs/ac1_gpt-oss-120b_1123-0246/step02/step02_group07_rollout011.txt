"""
Convolution‑bound optimiser.

The optimiser starts from the exact “arcsine’’ (inverse‑square‑root)
sequence, explores a rich family of low‑dimensional deformations
(tilt towards the reversed arcsine, power‑law, mixtures thereof) and
then runs a temperature‑controlled stochastic local search.  The moves
are designed to move mass from indices that dominate the current
maximal convolution coefficient towards low‑impact indices, while also
allowing occasional smoothing and block‑wise scaling.  The routine
honours a time budget (default 500 s, the harness supplies up to 1000 s)
and returns the best sequence found.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# 1.  Build the “arcsine’’ (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative arcsine sequence a_i = C(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence  a_i = a_{i-1} * ((2i-1)(2i))/(4 i^2)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# 2.  Simple perturbations.
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random block by exp(N(0, σ)) and optionally symmetrise."""
    n = base.size
    # block width: between 0.5 % and 20 % of the length
    w_min = max(1, n // 200)
    w_max = max(2, n // 5)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.25)))  # a bit larger than before
    cand[start:stop] *= factor
    # occasional symmetrisation (helps with the convolution shape)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def directed_transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Move a noticeable fraction of mass from a “high‑impact’’ index
    (largest contribution to the current maximal convolution entry)
    to an index that currently holds little mass.
    """
    n = base.size
    conv = np.convolve(base, base)
    # the location of the current maximum
    max_idx = int(np.argmax(conv))

    # contribution of each i to that entry (i runs where both factors exist)
    idx = np.arange(max_idx + 1)
    contrib = base[idx] * base[max_idx - idx]

    src = int(np.argmax(contrib))                     # most harmful index
    # select a low‑value target (20 % quantile of the whole vector)
    low_thr = np.percentile(base, 20)
    low_candidates = np.where(base <= low_thr)[0]
    if low_candidates.size == 0:
        # fallback: any index different from src
        tgt = int(rng.integers(0, n - 1))
        if tgt >= src:
            tgt += 1
    else:
        tgt = int(rng.choice(low_candidates))

    if tgt == src:
        return base.copy()

    # fraction to move – up to 50 % of the source entry
    frac = rng.random() * 0.40 + 0.10          # 10 % … 50 %
    delta = frac * base[src]
    if delta <= 0.0:
        return base.copy()

    cand = base.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)


def smooth_move(base: np.ndarray) -> np.ndarray:
    """Apply a tiny three‑point smoothing kernel (helps to flatten peaks)."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    smooth = np.convolve(base, kernel, mode='same')
    return np.clip(smooth, 0.0, 1000.0)
