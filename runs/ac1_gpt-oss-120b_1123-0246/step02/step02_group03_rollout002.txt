Implement as:

```
def proj_simplex(v, s):
    # v: ndarray
    # s: target sum
    # returns w >=0, sum(w)=s, minimizing L2 distance to v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function sequence that minimises the evaluation function.

The algorithm is a simple exponentiated‑gradient / Adam optimiser on the
soft‑max smoothing of the +∞‑norm of the self‑convolution.
Key ideas
-----------
* Work on the simplex  Σ a_i = √(2 n)  (the evaluation equals the
  maximal convolution value for this normalisation).
* Use a smooth approximation to max by a soft‑max with a temperature β
  that grows from a small to a large value during the run.
* Gradient of the smoothed objective is computed analytically with FFT‑based
  convolution (O(n log n)).
* Updates are performed with Adam‑style adaptive step sizes and the
  resulting vector is projected onto the non‑negative simplex.
* Periodic multiplicative noise (log‑normal perturbations) and a simple
  stagnation‑restart strategy keep the search from getting trapped.
* After the main phase a short local‑search (pairwise mass transfer)
  finishes the solution.

The routine respects the 1000 s time budget, prints occasional diagnostics
and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – required by the platform)
# ----------------------------------------------------------------------


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a power‑of‑two real FFT.
    The result has length len(x)+len(y)-1.
    """
    n = x.size + y.size - 1
    N = 1 << (n - 1).bit_length()         # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv


def _grad_and_maxb(a: np.ndarray, beta: float):
    """
    Compute the gradient of the soft‑max approximation to the maximal
    self‑convolution value and also return the exact maximal convolution.

    Parameters
    ----------
    a : np.ndarray
        Current non‑negative vector (already normalised to the target sum).
    beta : float
        Soft‑max temperature.  Larger beta makes the approximation sharper.

    Returns
    -------
    grad : np.ndarray
        Gradient of the smoothed objective (same shape as ``a``).
    maxb : float
        Exact maximal entry of a*a (the true objective value).
    """
    n = a.size
    # exact self‑convolution and its maximum
    b = _fft_convolve(a, a)
    maxb = b.max()

    # soft‑max weights over all lags
    # shift by maxb for numerical stability
    exp_term = np.exp(beta * (b - maxb))
    w = exp_term / exp_term.sum()                     # shape (2n-1,)

    # gradient of Σ_k w_k (a*a)_k  = 2 Σ_k w_k a_{k‑i}
    # we need convolution of a with reversed w
    w_rev = w[::-1]
    conv_rev = _fft_convolve(a, w_rev)                # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)                  # indices n‑1 … 2n‑2
    grad = 2.0 * conv_rev[idx]

    # Projected gradient (zero‑sum) is not needed – we will keep the sum
    # by a simplex projection later.
    return grad, maxb


def _proj_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    # Work on a copy to avoid side‑effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 950,
                     **kwargs):
    """
    Search for a low‑score height sequence within the allotted time.
    Returns a list of floats (the best sequence found).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (a few seconds are kept as safety margin).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a tiny safety margin for the wrapper that calls us
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # 1. Choose the problem size and initialise the sequence
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # keep the length moderate (≤12000) to stay fast
        if base.size > 12000:
            lo = (base.size - 12000) // 2
            base = base[lo:lo + 12000]
    else:
        # fallback: a uniform vector of moderate size
        base = np.ones(4000, dtype=np.float64)

    n = base.size
    a = np.clip(base, 0.0, 1000.0)

    # Normalise the sum to √(2 n).  Under this normalisation
    # the evaluation function equals the maximal convolution entry.
    target_sum = np.sqrt(2.0 * n)
    if a.sum() == 0.0:
        a[:] = 1.0
    a *= (target_sum / a.sum())
    # enforce positivity and the upper bound (1000) – then renormalise
    a = np.clip(a, 0.0, 1000.0)
    a *= (target_sum / a.sum())

    # ------------------------------------------------------------------
    # 2. Book‑keeping and optimiser state
    # ------------------------------------------------------------------
    best_seq = a.copy()
    _, best_maxb = _grad_and_maxb(a, beta=1.0)   # true maximal convolution
    best_score = float(best_maxb)                # eval = maxb for our normalisation
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    t_iter = 0

    # Learning‑rate schedule (linear decay)
    lr_start = target_sum * 0.0005      # ≈5 % of a typical entry for sparse vectors
    lr_min = lr_start * 0.02

    # Soft‑max temperature schedule
    beta_start = 0.3
    beta_end = 30.0

    # Stagnation handling
    last_improve = 0
    stagnation_limit = 5000   # iterations without improvement => restart

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = elapsed / budget_s      # 0 … 1

        # 3.1 temperature (soft‑max β) and learning‑rate for this iteration
        beta = beta_start + (beta_end - beta_start) * frac
        lr = max(lr_min, lr_start * (1.0 - frac))

        # 3.2 gradient of the smoothed objective
        grad, maxb_exact = _grad_and_maxb(a, beta)

        # 3.3 Adam update (descending direction = -grad)
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_candidate = a - step                               # descent step

        # 3.4 project onto the non‑negative simplex Σ a_i = target_sum
        a_candidate = _proj_simplex(a_candidate, target_sum)

        # 3.5 evaluate the **exact** score (no smoothing)
        conv = _fft_convolve(a_candidate, a_candidate)
        cand_score = float(conv.max())

        # 3.6 keep the best ever seen
        if cand_score < best_score:
            best_seq = a_candidate.copy()
            best_score = cand_score
            last_improve = t_iter
            print(f"[{t_iter:6d}] new best  score = {best_score:.6f}")

        # always keep walking – the Adam state lives on the current a
        a = a_candidate

        # ------------------------------------------------------------------
        # 4. Occasional random perturbations (helps escaping plateaux)
        # ------------------------------------------------------------------
        if t_iter % 400 == 0:
            # a log‑normal multiplicative jitter
            perturb = np.exp(rng.normal(0.0, 0.12, size=n))
            pert = a * perturb
            pert = _proj_simplex(pert, target_sum)
            pert_score = float(_fft_convolve(pert, pert).max())
            if pert_score < best_score:
                best_seq = pert.copy()
                best_score = pert_score
                a = pert
                last_improve = t_iter
                print(f"[{t_iter:6d}] perturb improved score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 5. Stagnation detection → restart from a fresh random vector
        # ------------------------------------------------------------------
        if t_iter - last_improve > stagnation_limit:
            a = rng.random(n)
            a = _proj_simplex(a, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            last_improve = 0
            print(f"[restart] stagnation – re‑initialising random vector")

        # ------------------------------------------------------------------
        # 6. Optional quick local search after a while (optional, cheap)
        # ------------------------------------------------------------------
        if t_iter % 2000 == 0 and time.monotonic() < deadline:
            # Try a handful of pairwise mass‑transfer moves
            for _ in range(200):
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                # transfer a small fraction of mass from i to j
                delta = min(a[i], 0.02 * target_sum)   # keep move modest
                if delta <= 0:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                # Simple projection is not required – the sum is unchanged
                # and positivity is preserved by construction
                tmp_score = float(_fft_convolve(a_try, a_try).max())
                if tmp_score < best_score:
                    a = a_try
                    best_seq = a_try.copy()
                    best_score = tmp_score
                    last_improve = t_iter
                    # break early to keep the loop cheap
                    break

    # ------------------------------------------------------------------
    # 7. Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[done] iterations = {t_iter}, best score = {best_score:.6f}")
    return [float(v) for v in best_seq]
