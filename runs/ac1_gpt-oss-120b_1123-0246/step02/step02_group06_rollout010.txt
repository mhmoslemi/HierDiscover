def project_simplex(v):
    # Project onto simplex sum=1, v >=0 in final.
    # Use algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)

Classic algorithm (Duchi et al.):

```
def proj_simplex(v):
    # return w >=0 and sum(w)=1 closest to v in Euclidean norm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all cond false then return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid stochastic / gradient search for a low‑value step function.

The algorithm proceeds in three stages:

1.  Initialise with the exact discretisation of the arcsine
    distribution (the known solution of the continuous problem) and
    normalise the vector to sum = 1.

2.  Randomly explore a low‑dimensional family of deformations of this
    base shape:
        – modify the singular exponent (i·(n‑i))^{Δ},
        – apply a linear tilt,
        – add a sinusoidal modulation.
    After each random perturbation the vector is projected back onto the
    probability simplex (non‑negative, unit‑sum).

3.  For every candidate perform a few soft‑max gradient steps,
    i.e. minimise a smooth approximation of ``max(conv(a,a))``.
    The gradient of the smooth max can be computed efficiently with a
    convolution, so each refinement costs only O(n log n).

The best sequence seen before the time budget expires is returned.
"""

import time
import numpy as np
from math import pi

# ----------------------------------------------------------------------
# Utility – projection onto the probability simplex (non‑negative, sum=1)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=1}."""
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1          # zero‑based index
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact discretisation of the arcsine density on [0,1]
# ----------------------------------------------------------------------
def _arcsine_integral_shape(n: int) -> np.ndarray:
    """Return a length‑n vector approximating the arcsine pdf integrated over each cell."""
    i = np.arange(n, dtype=np.float64)
    left = i / n
    right = (i + 1) / n
    # CDF of arcsine: 2/π·arcsin(√x)
    a = (2.0 / pi) * (np.arcsin(np.sqrt(right)) - np.arcsin(np.sqrt(left)))
    return a


# ----------------------------------------------------------------------
# Random deformation of a base shape
# ----------------------------------------------------------------------
def _deform_shape(base: np.ndarray, rng: np.random.Generator,
                  delta_exp: float, tilt: float,
                  amp: float, freq: int, phase: float) -> np.ndarray:
    """Apply exponent, tilt and sinusoidal modulation to a base shape."""
    n = base.size
    idx = np.arange(n, dtype=np.float64) + 0.5          # 0.5 … n‑0.5
    # exponent modification (base behaves like (i·(n‑i))^{-0.5})
    dist = idx * (n - idx + 1e-12)                     # avoid zero
    factor = np.exp(delta_exp * np.log(dist + 1e-12))
    a = base * factor

    # linear tilt, centred at the middle of the interval
    a *= 1.0 + tilt * ((np.arange(n) / (n - 1)) - 0.5)

    # sinusoidal modulation
    if amp != 0.0:
        a *= 1.0 + amp * np.sin(2.0 * pi * freq * np.arange(n) / n + phase)

    a = np.maximum(a, 0.0)          # keep non‑negative
    a = _proj_simplex(a)            # re‑normalise to sum = 1
    return a


# ----------------------------------------------------------------------
# One soft‑max gradient descent step (smooth approximation of max)
# ----------------------------------------------------------------------
def _softmax_grad_step(a: np.ndarray, lr: float, beta: float) -> np.ndarray:
    """
    Perform a single gradient step on the smooth max of the autocorrelation.
    a  – current vector (already normalised to sum = 1)
    lr – learning rate (positive)
    beta – inverse temperature; larger => closer to true max.
    """
    n = a.size
    # autocorrelation
    c = np.convolve(a, a)
    # soft‑max weights (stable formulation)
    c_max = np.max(c)
    w = np.exp(beta * (c - c_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        return a
    p = w / w_sum                     # probability distribution over shifts

    # correlation of p with reversed a:   g_i = 2 * Σ_k p_k a_{k-i}
    # This can be obtained by a single convolution.
    corr = np.convolve(p, a[::-1], mode='full')
    start = n - 1
    g = 2.0 * corr[start:start + n]    # gradient of the smooth max

    a_new = a - lr * g
    a_new = np.maximum(a_new, 0.0)
    a_new = _proj_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
# Simple mass‑transfer move (kept for occasional random jumps)
# ----------------------------------------------------------------------
def _transfer_move(x: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    n = x.size
    # early part (first quarter) → late part (last quarter)
    i = rng.integers(0, max(1, n // 4))
    j = rng.integers(max(1, 3 * n // 4), n)
    if x[i] == 0.0:
        return x.copy()
    frac = rng.random() * 0.9
    delta = frac * x[i]
    y = x.copy()
    y[i] -= delta
    y[j] += delta
    return y


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 500, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds and returns the
    best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length of the sequence.
    # ------------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256

    # ------------------------------------------------------------------
    # 1. Starting point – exact arcsine pdf integrated on the grid.
    # ------------------------------------------------------------------
    base = _arcsine_integral_shape(n)
    base = _proj_simplex(base)          # ensure sum == 1
    cur = base.copy()
    cur_score = float(evaluate_sequence(cur.tolist()))
    best = cur.copy()
    best_score = cur_score

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Optimisation loop (mix of stochastic deformation + gradient steps)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # leave a second margin
    it = 0

    # learning‑rate schedule (starts larger, decays)
    lr_start = 0.5
    lr_end = 0.01

    # soft‑max temperature schedule
    beta_start = 50.0
    beta_end = 5000.0

    while time.monotonic() < deadline:
        it += 1
        # linear interpolation for the schedules
        t_frac = it / 20000.0          # arbitrary scaling for schedule
        lr = lr_start * (1.0 - t_frac) + lr_end * t_frac
        beta = beta_start * (1.0 - t_frac) + beta_end * t_frac

        # ----------------------------------------------------------------
        # 2a. Randomly deform the base shape (low‑dimensional parameters)
        # ----------------------------------------------------------------
        delta_exp = rng.uniform(-0.25, 0.25)          # around the -0.5 exponent
        tilt = rng.uniform(-0.5, 0.5)                # linear tilt
        amp = rng.uniform(0.0, 0.3)                  # sinusoid amplitude
        freq = rng.integers(1, 5)                    # few whole cycles
        phase = rng.random() * 2.0 * pi

        cand = _deform_shape(base, rng,
                             delta_exp=delta_exp,
                             tilt=tilt,
                             amp=amp,
                             freq=freq,
                             phase=phase)

        # ----------------------------------------------------------------
        # 2b. A few smooth‑max gradient refinements
        # ----------------------------------------------------------------
        for _ in range(3):
            cand = _softmax_grad_step(cand, lr, beta)

        # Occasionally inject a pure transfer move to escape plateaus
        if rng.random() < 0.07:
            cand = _transfer_move(cand, rng)

        # Ensure the candidate stays on the simplex (the gradient already does)
        cand = _proj_simplex(cand)

        # ----------------------------------------------------------------
        # 2c. Evaluate
        # ----------------------------------------------------------------
        cand_score = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(cand_score):
            continue

        # Update current point (always keep the better of cur / cand)
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score

        # Record the best overall
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            print(f"[{it:>6}] new best = {best_score:.6f}")

        # Every few thousand iterations re‑centre the search around the current best
        if it % 4000 == 0:
            base = best.copy()          # focus the random deformations

        # Early exit if we already beat the target comfortably
        if best_score < 1.45:
            break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return a plain Python list as required by the harness.
    return [float(x) for x in best]
