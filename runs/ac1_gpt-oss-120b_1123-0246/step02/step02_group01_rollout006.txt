"""
Propose a low‑score step‑function using a convex‑programming refinement
of a Beta(½,½) (arcsine) shape.

The algorithm:
1. Initialise a symmetric Beta(α,α) distribution (α≈0.5) for a chosen
   length n and normalise it so that Σa = √(2 n).
2. Repeatedly solve a linear programme:
      maximise    Σ g_i
      subject to   g ≥ 0
                    (f * g)_k ≤ M   for all k,
   where M = max(f*f) for the current f.  For fixed f this constraint
   is linear in g because the convolution is linear in its second argument.
   The LP therefore finds a direction g that can increase the total mass
   without increasing the current worst‑case convolution value.
3. Scale g to have the same ℓ1‑norm as f and replace
      f ← (1‑t)·f + t·g          (t≈0.5, reduced if progress stalls).
   This convex combination keeps the normalisation and strictly reduces M
   (or leaves it unchanged).  Iterate a few times.
4. Run the above refinement for a few different lengths (n≈1 k–4 k) and
   keep the best sequence seen.  Finally evaluate the best candidate with
   the provided `evaluate_sequence` (which also applies the hard clipping).

The whole search respects the time budget (default ≈500 s, up to 1000 s in
the judging harness).  If the optional cvxpy library is unavailable the
fallback is the original Beta‑grid search.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective (max of self‑convolution)
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray) -> float:
    """Return the evaluation score 2·n·max(conv(a,a))/sum(a)^2.
    The sequence is clipped to the bounds enforced by evaluate_sequence.
    """
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    size = 2 * n - 1
    # next power of two for an efficient FFT
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Helper: generate a symmetric Beta(α,α)‑shaped sequence for a given n
# ----------------------------------------------------------------------
def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Build a non‑negative sequence of length n with shape
    p^{α‑1}(1‑p)^{α‑1} where p = (i+0.5)/n.
    The sequence is scaled so that its sum equals sqrt(2·n).
    """
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, alpha - 1.0)
    target = np.sqrt(2.0 * n)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    a = w * (target / w_sum)
    a = np.clip(a, 0.0, 1000.0)
    return a


# ----------------------------------------------------------------------
# Helper: build the convolution matrix A such that (A @ g) == f * g
# ----------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    For a fixed vector f (length n) construct the dense matrix A (size (2n‑1)×n)
    with the property (A @ g) == np.convolve(f, g, mode='full').
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


# ----------------------------------------------------------------------
# Helper: solve the LP  maximise Σ g_i   s.t.   g ≥ 0,  A @ g ≤ M
# ----------------------------------------------------------------------
def _solve_lp(A: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear programme using cvxpy.
    Returns the optimal vector g (numpy array) or None on failure.
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return None

    n = A.shape[1]
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M + 1e-12]   # relax a tiny bit for numerical safety
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception:  # pragma: no cover
        return None
    if g.value is None:
        return None
    return np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)

            A = _conv_matrix(f)

            # solve the LP
            g = _solve_lp(A, M)
            if g is None:
                print("[lp] solver unavailable or failed – aborting LP phase")
                break

            # scale g to have the same ℓ1‑norm as f (maintains Σ a)
            sum_g = g.sum()
            if sum_g <= 0.0:
                break
            g = g * (f.sum() / sum_g)

            # convex combination
            f_new = (1.0 - t) * f + t * g

            # re‑normalise to the target sum to avoid drift caused by rounding
            f_new = f_new * (target_sum / f_new.sum())
            f_new = np.clip(f_new, 0.0, 1000.0)

            # evaluate improvement
            new_score = _fast_objective(f_new)
            if new_score < best_score - 1e-6:
                best_score = new_score
                best_seq = f_new.copy()
                print(f"[n={n} it={it+1}] improved score = {best_score:.6f}")

            # if the max‑convolution (i.e. M) did not decrease, shrink the step size
            conv_new = np.convolve(f_new, f_new)
            M_new = float(conv_new.max())
            if M_new >= M - 1e-9:
                # stagnation – reduce t, but stop if t becomes tiny
                t *= 0.5
                if t < 0.01:
                    break
                # keep the old f for the next trial
                continue
            else:
                # successful descent – keep the new f and reset t
                f = f_new
                t = 0.5

        # after finishing the LP sweep for this n, optionally apply a tiny
        # random‑perturbation fine‑tuning (same idea as the previous code)
        if best_seq is not None and still_time():
            a = best_seq.copy()
            step_scale = target_sum / (30.0 * n)   # tiny moves
            while still_time():
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                delta = rng.random() * step_scale
                if a[i] < delta:
                    continue
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                # renormalise to the target sum (just in case)
                s = cand.sum()
                if s > 0:
                    cand *= (target_sum / s)
                cand = np.clip(cand, 0.0, 1000.0)
                sc = _fast_objective(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    a = cand
                    # occasional progress report
                    if rng.random() < 0.02:
                        print(f"[perturb n={n}] new best score = {best_score:.6f}")
                else:
                    a = cand   # keep walking even without improvement

    # ------------------------------------------------------------------
    # 2.  Final accurate evaluation (and final printout)
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: a single constant step (should never happen)
        best_seq = np.array([np.sqrt(2.0)], dtype=np.float64)

    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] evaluated with evaluate_sequence → score = {final_score:.6f}")

    # Return as a plain Python list of floats
    return [float(v) for v in best_seq]
