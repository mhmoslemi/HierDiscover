def multi_source_transfer(base, conv, k_max, rng, max_src=3, low_pct=20, frac_range=(0.005,0.02)):
    n = base.size
    # Determine source indices for k_max
    i_start = max(0, k_max - (n-1))
    i_end = min(n-1, k_max)
    idx = np.arange(i_start, i_end+1)
    contrib = base[idx] * base[k_max - idx]
    # Get top src indices
    num_src = min(max_src, len(idx))
    src_rank = np.argpartition(contrib, -num_src)[-num_src:]
    src_inds = idx[src_rank]
    # low target candidates
    low_thr = np.percentile(base, low_pct)
    low_candidates = np.where(base <= low_thr)[0]
    # If no low candidates, fallback random
    if low_candidates.size == 0:
        low_candidates = np.arange(n)
    # choose random tgt for each src
    cand = base.copy()
    for src in src_inds:
        tgt = int(rng.choice(low_candidates))
        if tgt == src:
            continue
        frac = rng.uniform(frac_range[0], frac_range[1])
        delta = frac * cand[src]
        if delta <= 0:
            continue
        cand[src] -= delta
        cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)
```

def flatten_move(base, rng):
    n = base.size
    # get sorted indices descending
    idx_desc = np.argsort(-base)  # descending
    K = min(10, n) # maybe 5 or 10
    src_inds = idx_desc[:K]
    # target low
    low_thr = np.percentile(base, 5)
    tgt_candidates = np.where(base <= low_thr)[0]
    if tgt_candidates.size < K:
        tgt_candidates = np.arange(n)
    tgt_inds = rng.choice(tgt_candidates, size=K, replace=False)
    cand = base.copy()
    for src, tgt in zip(src_inds, tgt_inds):
        # move a fraction
        frac = rng.uniform(0.001, 0.02)  # small
        delta = frac * cand[src]
        cand[src] -= delta
        cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)
```

"""
Hybrid heuristic optimizer for the convolution‑bound sequence.
It starts from the analytically optimal “arcsine’’ sequence,
explores many low‑dimensional deformations (tilt, power‑law,
interleaved ordering, random shuffles) and then runs a
simulated‑annealing‑style stochastic local search that
combines:
 - multi‑source flattening (move mass from the largest entries
   to the smallest),
 - directed transfer from the index most responsible for the
   current maximal convolution coefficient,
 - block perturbations.
The temperature gradually cools, allowing occasional uphill
moves to escape plateaus. The routine respects the given time
budget and returns the best sequence found.
"""
import numpy as np
import time

# ----------------------------------------------------------------------
# 1.  Core utilities
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the inverse‑square‑root (“arcsine’’) sequence of given length."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    return a

def convolve_self(a: np.ndarray, n_fft: int) -> np.ndarray:
    """Return a*a (full convolution) using FFT of length n_fft."""
    # real FFT is sufficient because a is real
    fa = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(fa * fa, n=n_fft)
    return conv[: 2 * a.size - 1]

def eval_score(a: np.ndarray, n_fft: int) -> float:
    """Fast evaluation of the objective (same formula as evaluate_sequence)."""
    conv = convolve_self(a, n_fft)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * a.size * max_b / (sum_a * sum_a)

# ----------------------------------------------------------------------
# 2.  Simple perturbations (kept from reference)
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0, σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

def directed_transfer_move(base: np.ndarray, conv: np.ndarray, k_max: int,
                           rng: np.random.Generator) -> np.ndarray:
    """Move a small fraction of mass from the index that contributes most to
    the current maximal convolution coefficient to a low‑impact index."""
    n = base.size
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idx = np.arange(i_start, i_end + 1)
    contrib = base[idx] * base[k_max - idx]
    src = int(idx[np.argmax(contrib)])

    low_thr = np.percentile(base, 20)
    low_candidates = np.where(base <= low_thr)[0]
    if low_candidates.size == 0:
        tgt = int(rng.integers(0, n))
        while tgt == src:
            tgt = int(rng.integers(0, n))
    else:
        tgt = int(rng.choice(low_candidates))
        if tgt == src:
            # fallback to a random different index
            tgt = int(rng.integers(0, n))
            while tgt == src:
                tgt = int(rng.integers(0, n))
    # fraction to move – slowly decreasing with time
    frac = rng.random() * 0.20 + 0.01          # between 1 % and 21 %
    delta = frac * base[src]
    if delta <= 0.0:
        return base.copy()
    cand = base.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)

# ----------------------------------------------------------------------
# 3.  Multi‑source flattening move
# ----------------------------------------------------------------------
def flatten_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a tiny fraction of mass from the few largest entries to the
    smallest ones, trying to flatten the distribution."""
    n = base.size
    # number of sources/targets – a modest fraction of the size
    K = max(2, min(8, n // 700))      # ≈ 0.1 % of entries, at least 2, at most 8
    # sources = K largest values
    src_inds = np.argsort(-base)[:K]
    # targets = K smallest values (or very small values)
    low_thr = np.percentile(base, 5)
    low_candidates = np.where(base <= low_thr)[0]
    if low_candidates.size < K:
        low_candidates = np.arange(n)
    tgt_inds = rng.choice(low_candidates, size=K, replace=False)
    cand = base.copy()
    for src, tgt in zip(src_inds, tgt_inds):
        # move a very small fraction (0.1‑2 %)
        frac = rng.uniform(0.001, 0.02)
        delta = frac * cand[src]
        if delta > 0:
            cand[src] -= delta
            cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)

# ----------------------------------------------------------------------
# 4.  Random permutation move
# ----------------------------------------------------------------------
def random_permutation_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Shuffle the entries – useful when the multiset of values is good
    but the ordering is sub‑optimal."""
    perm = rng.permutation(base.shape[0])
    return np.clip(base[perm], 0.0, 1000.0)

# ----------------------------------------------------------------------
# 5.  Structured interleaving (large‑small alternating)
# ----------------------------------------------------------------------
def interleave_ends(base: np.ndarray) -> np.ndarray:
    """Place the largest values alternately at the beginning and the end
    of the sequence (large‑small‑large‑small pattern)."""
    vals = np.sort(base)[::-1]                    # descending
    n = vals.size
    out = np.empty_like(vals)
    left, right = 0, n - 1
    i = 0
    while left <= right:
        out[i] = vals[left]
        i += 1
        left += 1
        if left <= right:
            out[i] = vals[right]
            i += 1
            right -= 1
    return out

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of floats – the best step‑function heights found
    within the time budget.  The optimisation proceeds in three phases:

    1. Initialise with the analytical arcsine sequence.
    2. Perform a broad sweep over several low‑dimensional families
       (tilt, power, tilt + power, interleaving, random shuffles).
    3. Run a simulated‑annealing‑style stochastic local search that
       repeatedly applies flattening, directed transfers and block
       perturbations.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size (length of the known sequence)
    # --------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256                     # a harmless fallback

    # Pre‑compute FFT length (next power‑of‑two for speed)
    n_fft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Phase 1 – analytical starting point
    # ------------------------------------------------------------------
    base_seq = build_arcsine_sequence(n)
    best_seq = base_seq.copy()
    best_score = eval_score(best_seq, n_fft)
    print(f"[init] length={n}, arcsine score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – extensive parameter sweep
    # ------------------------------------------------------------------
    rev_seq = base_seq[::-1]

    # (a) tilt (mix original with reversed)
    for alpha in np.linspace(0.0, 1.0, 101):        # 0 % … 100 %
        cand = (1.0 - alpha) * base_seq + alpha * rev_seq
        cand = np.clip(cand, 0.0, 1000.0)
        sc = eval_score(cand, n_fft)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"[sweep] tilt α={alpha:.3f} → score={sc:.6f}")

    # (b) power‑law deformation
    for p in np.linspace(0.40, 1.00, 31):           # 0.40 … 1.00
        cand = np.power(base_seq, p)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = eval_score(cand, n_fft)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"[sweep] power p={p:.3f} → score={sc:.6f}")

    # (c) tilt + power (coarse grid)
    for alpha in np.linspace(0.0, 0.9, 10):
        mixed = (1.0 - alpha) * base_seq + alpha * rev_seq
        for p in np.linspace(0.6, 0.9, 7):
            cand = np.power(mixed, p)
            cand = np.clip(cand, 0.0, 1000.0)
            sc = eval_score(cand, n_fft)
            if sc < best_score:
                best_seq, best_score = cand, sc
                print(f"[sweep] α={alpha:.2f}, p={p:.2f} → score={sc:.6f}")

    # (d) interleaved ordering (large‑small‑large‑small)
    cand = interleave_ends(base_seq)
    sc = eval_score(cand, n_fft)
    if sc < best_score:
        best_seq, best_score = cand, sc
        print(f"[sweep] interleaved ends → score={sc:.6f}")

    # (e) a handful of random permutations of the arcsine multiset
    for _ in range(50):
        perm = rng.permutation(n)
        cand = base_seq[perm]
        sc = eval_score(cand, n_fft)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"[sweep] random perm → score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing‑style stochastic local search
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # keep a small slack
    cur_seq = best_seq.copy()
    cur_score = best_score

    # temperature schedule – start a bit above the typical improvement size
    T0 = 0.05

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # current convolution (needed for the directed move)
        conv_cur = convolve_self(cur_seq, n_fft)
        k_max = int(np.argmax(conv_cur))

        # pick a move
        r = rng.random()
        if r < 0.45:
            cand_seq = flatten_move(cur_seq, rng)
        elif r < 0.70:
            cand_seq = directed_transfer_move(cur_seq, conv_cur, k_max, rng)
        elif r < 0.90:
            cand_seq = block_perturbation(cur_seq, rng)
        else:
            cand_seq = random_permutation_move(cur_seq, rng)

        cand_score = eval_score(cand_seq, n_fft)

        # simple simulated‑annealing acceptance
        if cand_score < cur_score:
            accept = True
        else:
            # linear cooling based on elapsed fraction
            elapsed = time.monotonic() - start_time
            T = T0 * max(0.0, 1.0 - elapsed / (deadline - start_time))
            # protect against division by zero
            accept = rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12))

        if accept:
            cur_seq = cand_seq
            cur_score = cand_score

        if cand_score < best_score:
            best_seq = cand_seq
            best_score = cand_score
            print(f"[{it:>6}] new best = {best_score:.6f}")
            if best_score < 1.503:
                print("[stop] target reached")
                break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return plain Python list as required by the harness
    return [float(x) for x in best_seq]

The optimizer works in three stages:

1. **Analytical starter** – the inverse‑square‑root (arcsine) sequence,

2. **Broad sweep** – we explore several low‑dimensional families

# 5. Interleaved large‑small ordering
# ----------------------------------------------------------------------
def interleave_ends(base: np.ndarray) -> np.ndarray:
    """Create a large‑small‑large‑small pattern by alternating the biggest
    remaining value from the front and the back of the sorted list."""
    vals = np.sort(base)[::-1]                     # descending
    n = vals.size
    out = np.empty_like(vals)
    left, right = 0, n - 1
    i = 0
    while left <= right:
        out[i] = vals[left]
        i += 1
        left += 1
        if left <= right:
            out[i] = vals[right]
            i += 1
            right -= 1
    return out

# 6. Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of floats – the best step‑function heights found
    within the time budget.  The optimisation proceeds in three phases:

    1. Initialise with the analytical arcsine sequence.
    2. Perform a broad sweep over low‑dimensional families
       (tilt, power‑law, tilt + power, interleaving, random permutations).
    3. Run a simulated‑annealing‑style stochastic local search that
       repeatedly applies flattening, directed transfers and block
       perturbations.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size (length of the known sequence)
    # --------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256                     # fallback length

    # pre‑compute FFT size (next power of two) for speed
    n_fft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Phase 1 – analytical starting point
    # ------------------------------------------------------------------
    base_seq = build_arcsine_sequence(n)
    best_seq = base_seq.copy()
    best_score = eval_score(best_seq, n_fft)
    print(f"[init] length={n}, arcsine score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – extensive parameter sweep
    # ------------------------------------------------------------------
    rev_seq = base_seq[::-1]

    # (a) tilt: mix with the reversed arcsine
    for alpha in np.linspace(0.0, 1.0, 101):        # 0 % … 100 %
        cand = (1.0 - alpha) * base_seq + alpha * rev_seq
        cand = np.clip(cand, 0.0, 1000.0)
        sc = eval_score(cand, n_fft)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"[sweep] tilt α={alpha:.3f} → score={sc:.6f}")

    # (b) power‑law deformation
    for p in np.linspace(0.40, 1.00, 31):           # 0.40 … 1.00
        cand = np.power(base_seq, p)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = eval_score(cand, n_fft)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"[sweep] power p={p:.3f} → score={sc:.6f}")

    # (c) tilt + power (coarse grid)
    for alpha in np.linspace(0.0, 0.9, 10):
        mixed = (1.0 - alpha) * base_seq + alpha * rev_seq
        for p in np.linspace(0.6, 0.9, 7):
            cand = np.power(mixed, p)
            cand = np.clip(cand, 0.0, 1000.0)
            sc = eval_score(cand, n_fft)
            if sc < best_score:
                best_seq, best_score = cand, sc
                print(f"[sweep] α={alpha:.2f}, p={p:.2f} → score={sc:.6f}")

    # (d) interleaved large‑small ordering
    cand = interleave_ends(base_seq)
    sc = eval_score(cand, n_fft)
    if sc < best_score:
        best_seq, best_score = cand, sc
        print(f"[sweep] interleaved ends → score={sc:.6f}")

    # (e) a handful of random permutations of the arcsine multiset
    for _ in range(50):
        perm = rng.permutation(n)
        cand = base_seq[perm]
        sc = eval_score(cand, n_fft)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"[sweep] random perm → score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic local search with simulated annealing
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # small slack
    cur_seq = best_seq.copy()
    cur_score = best_score

    # temperature parameters
    T0 = 0.05

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # current convolution (needed for the directed move)
        conv_cur = convolve_self(cur_seq, n_fft)
        k_max = int(np.argmax(conv_cur))

        # decide which move to apply
        r = rng.random()
        if r < 0.45:
            cand_seq = flatten_move(cur_seq, rng)
        elif r < 0.70:
            cand_seq = directed_transfer_move(cur_seq, conv_cur, k_max, rng)
        elif r < 0.90:
            cand_seq = block_perturbation(cur_seq, rng)
        else:
            cand_seq = random_permutation_move(cur_seq, rng)

        cand_score = eval_score(cand_seq, n_fft)

        # simulated‑annealing acceptance
        if cand_score < cur_score:
            accept = True
        else:
            elapsed = time.monotonic() - start_time
            T = T0 * max(0.0, 1.0 - elapsed / (deadline - start_time))
            accept = rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12))

        if accept:
            cur_seq, cur_score = cand_seq, cand_score

        if cand_score < best_score:
            best_seq, best_score = cand_seq, cand_score
            print(f"[{it:>6}] new best = {best_score:.6f}")
            if best_score < 1.503:
                print("[stop] target reached")
                break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return a plain list for the harness
    return [float(x) for x in best_seq]
