"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

The algorithm combines three ideas that differ from the previous
implementation:

1. **Symmetric power‑law seed** – a grid search over a few exponents
   and offsets is performed, after which the vector is explicitly
   symmetrised (a_i = a_{n-1-i}) and normalised to unit sum.

2. **Momentum projected gradient descent on the ℓ∞‑norm** – at every
   iteration the index of the current maximal convolution value
   is found, the exact sub‑gradient (the shifted copy of the current
   vector) is computed, zero‑meaned, and added to a velocity vector.
   The updated vector is projected onto the simplex (non‑negative,
   unit sum) and optionally symmetrised again.  An exponential cooling
   schedule controls the step size and a Metropolis rule permits
   occasional uphill moves.

3. **Occasional linear‑programming “boost”** – roughly every few
   thousand iterations (or when the time budget permits) a small LP
   is solved: maximise the sum of a new vector `b` under the linear
   constraints (current f * b) ≤ M, where M is the current maximal
   convolution.  The LP solution is mixed into the current iterate,
   giving a rapid reduction of the bound.

All helper functions are defined at top level; the main entry point
is `propose_candidate`.  The implementation stays well within the
1000‑second budget on a typical CPU.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------
def _evaluate_fast(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` for non‑negative vectors."""
    total = seq.sum()
    if total < 1e-12:               # should never happen
        return float("inf")
    conv = np.convolve(seq, seq)    # full convolution
    max_b = conv.max()
    n = seq.size
    return float(2.0 * n * max_b / (total * total))


def _shift(seq: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of (seq*seq)[k] w.r.t. ``seq``.
    Returns a vector d such that d_i = seq_{k-i} for valid indices,
    otherwise 0.
    """
    n = seq.size
    out = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1, dtype=int)
        out[idx] = seq[k - idx]
    return out


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    With moderate probability the candidate is symmetrised afterwards.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                     # mild symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _lp_step(f: np.ndarray, M: float, target_sum: float,
             time_limit: float) -> np.ndarray | None:
    """
    Solve the linear program:
        maximise   1ᵀ b
        subject to (f * b)[k] ≤ M   for all k,
                   b ≥ 0.
    Returns the optimal b (scaled to `target_sum`) or None if the
    solver fails or time limit is too short.
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        return None

    n = f.size
    L = 2 * n - 1                     # length of the full convolution
    # Build constraint matrix A_ub (shape L × n)
    # Row k contains f shifted so that (f * b)[k] = A_ub[k]·b
    A = np.zeros((L, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    b_ub = np.full(L, M, dtype=np.float64)

    # Objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    # Use the Highs solver with a modest time limit
    # (SciPy's API does not expose a direct time limit, we simply bail
    # out if the remaining budget is too small).
    if time_limit < 0.1:
        return None

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': time_limit})

    if not res.success:
        return None

    b = res.x
    s = b.sum()
    if s > 0:
        b = b * (target_sum / s)      # rescale to keep the same total mass
    else:
        b = f.copy()
    # enforce symmetry (helps the next gradient phase)
    b = 0.5 * (b + b[::-1])
    return np.clip(b, 0.0, 1000.0)


def _gradient_step(seq: np.ndarray, eta: float,
                   momentum: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    """
    Perform one projected gradient step aimed at reducing the
    maximal convolution entry.

    Returns the new sequence and the updated momentum vector.
    """
    # 1. locate the (first) maximal convolution index
    conv = np.convolve(seq, seq)
    k_max = int(conv.argmax())
    # 2. exact sub‑gradient for that peak
    grad = _shift(seq, k_max)          # shape (n,)
    # 3. make the direction zero‑sum (preserve total mass)
    direction = -grad + np.mean(grad)

    # 4. momentum update
    beta = 0.9
    new_mom = beta * momentum + (1 - beta) * direction

    # 5. take a step and project onto simplex
    cand = seq + eta * new_mom
    cand = np.clip(cand, 0.0, 1000.0)
    s = cand.sum()
    if s > 0:
        cand /= s
    else:
        cand = seq.copy()
    # 6. enforce symmetry (mirroring)
    cand = 0.5 * (cand + cand[::-1])
    return cand, new_mom


# ----------------------------------------------------------------------
# Main entry point expected by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative step heights that (hopefully) yields a
    low value of ``evaluate_sequence``.  The routine respects the
    1000‑second time budget, uses a symmetric power‑law seed,
    accelerated projected gradient descent with momentum, occasional
    block perturbations, and a rare linear‑programming boost.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Choose the problem size.
    # Prefer the length of the previously found solution, otherwise 3000.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        n = 3000

    # ------------------------------------------------------------------
    # 1) Build a symmetric power‑law seed via a small grid search.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    idx = np.arange(1, n + 1, dtype=np.float64)        # 1‑based indices
    alphas = np.linspace(0.45, 0.60, 22)               # around 0.5
    betas = [0.0, 0.25, 0.5, 1.0, 2.0]

    for alpha in alphas:
        for beta in betas:
            seq = 1.0 / ((idx + beta) ** alpha)
            seq = np.clip(seq, 0.0, 1000.0)
            # symmetrise
            seq = 0.5 * (seq + seq[::-1])
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
            sc = _evaluate_fast(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()

    # fallback (should never happen)
    if best_seq is None:
        seq = 1.0 / np.sqrt(idx)
        seq = 0.5 * (seq + seq[::-1])
        seq /= seq.sum()
        best_seq = seq
        best_score = _evaluate_fast(seq)

    # ------------------------------------------------------------------
    # 2) Local optimisation: momentum projected gradient descent,
    #    stochastic block moves and occasional LP boost.
    # ------------------------------------------------------------------
    current_seq = best_seq.copy()
    current_score = best_score
    best_seq = current_seq.copy()
    best_score = current_score

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # small safety margin
    it = 0
    momentum = np.zeros_like(current_seq)

    # Annealing schedules
    eta_start = 0.20
    eta_end   = 4e-5
    temp_start = 2e-2
    temp_end   = 5e-7

    block_prob = 0.18                # probability of a random block move
    symm_prob  = 0.12                # extra symmetrisation chance

    # LP parameters
    lp_interval = 5000               # try an LP roughly every this many iters
    last_lp_iter = 0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        total   = deadline - start_time
        prog = min(1.0, elapsed / total)

        # Exponential schedules
        eta  = eta_start * (eta_end / eta_start) ** prog
        temp = temp_start * (temp_end / temp_start) ** prog

        # ------------------------------------------------------------------
        # Propose a candidate
        # ------------------------------------------------------------------
        if rng.random() < block_prob:
            cand = _make_block_perturbation(current_seq, rng)
            # reset momentum if we do a big jump
            momentum = np.zeros_like(current_seq)
        else:
            cand, momentum = _gradient_step(current_seq, eta, momentum)

        # occasional extra symmetrisation (helps flatten the convolution)
        if rng.random() < symm_prob:
            cand = 0.5 * (cand + cand[::-1])

        # Re‑normalise to keep the sum safely away from zero
        s = cand.sum()
        if s > 0:
            cand /= s
        else:
            cand = current_seq.copy()

        cand_score = _evaluate_fast(cand)

        # ------------------------------------------------------------------
        # Acceptance (Metropolis style)
        # ------------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # probability to accept a worse move
            prob = np.exp(-(cand_score - current_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            current_seq = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"iter {it:07d} – new best: {best_score:.6f}")

        # ------------------------------------------------------------------
        # Periodic LP boost
        # ------------------------------------------------------------------
        if (it - last_lp_iter) >= lp_interval:
            remaining = deadline - time.monotonic()
            # compute current bound M
            conv_cur = np.convolve(current_seq, current_seq)
            M_cur = conv_cur.max()
            # try an LP if we still have a few seconds left
            lp_b = _lp_step(current_seq, M_cur, current_seq.sum(), remaining * 0.6)
            if lp_b is not None:
                lp_score = _evaluate_fast(lp_b)
                if lp_score < best_score:
                    best_score = lp_score
                    best_seq = lp_b.copy()
                    print(f"iter {it:07d} – LP improvement: {best_score:.6f}")
                # mix the LP solution with the current iterate
                tau = 0.3
                current_seq = (1 - tau) * current_seq + tau * lp_b
                current_seq = np.clip(current_seq, 0.0, 1000.0)
                s = current_seq.sum()
                if s > 0:
                    current_seq /= s
                else:
                    current_seq = best_seq.copy()
                current_score = _evaluate_fast(current_seq)
                # reset momentum after a big LP jump
                momentum = np.zeros_like(current_seq)
            last_lp_iter = it

        it += 1

    # ----------------------------------------------------------------------
    # Final sanity checks
    # ----------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    print(f"finished after {it} iterations, best score={best_score:.6f}")
    return best_seq.tolist()
