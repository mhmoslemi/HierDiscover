def propose_candidate(seed=0, budget_s=500, n=256):

    rng = np.random.default_rng(seed)

    f = build_arcsine_sequence(n).astype(np.float64)
    f = f / f.sum() # normalize to sum=1.

    best_f = f.copy()
    best_score = sequence_score(f)
    cur_M = np.convolve(f, f).max()
    target_sum = 1.0

    # parameters
    t = 0.3
    min_t = 0.01
    max_iter = large

    deadline = time.monotonic() + budget_s - 0.5

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # Build A matrix:
        # rows = 2n-1, columns n
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        # compute for each k
        for k in range(rows):
            # indices i: max(0,k-(n-1)) <= i <= min(n-1,k)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # for i in range(i_start, i_end+1):
            # A[k,i] = f[k-i]
            # Equivalent to fill row with f indices reversed.
            # Could use vectorization but loops fine for n=256.
            for i in range(i_start, i_end+1):
                A[k, i] = f[k - i]

        # Solve LP
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= cur_M]
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)

        # Solve with appropriate solver and tolerance
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
        except cp.error.SolverError:
            # fallback to SCS
            prob.solve(solver=cp.SCS, verbose=False, max_iters=2000)

        if g.value is None:
            # solve failure, break
            break

        g0 = np.array(g.value).reshape(-1)
        sum_g0 = g0.sum()
        if sum_g0 < 1e-12:
            # nothing to improve
            break

        # Scale g0 to target sum:
        g_scaled = g0 * (target_sum / sum_g0)

        # Mix with f:
        f_new = (1 - t) * f + t * g_scaled

        # Evaluate new max:
        b_new = np.convolve(f_new, f_new)
        M_new = b_new.max()

        # If improvement:
        if M_new < cur_M * (1 - 1e-6):
            f = f_new
            cur_M = M_new
            cur_score = 2.0 * n * cur_M / (target_sum**2)  # since sum= target_sum
            if cur_score < best_score:
                best_f = f.copy()
                best_score = cur_score
                print(f"[{iteration}] new best {best_score:.6f}")
                # Reduce t maybe increase step for faster convergence
            # Optionally increase mixing factor.
        else:
            # No improvement; reduce t
            t *= 0.5
            if t < min_t:
                # maybe random restart?
                # restart with arcsine again
                f = build_arcsine_sequence(n)
                f = f / f.sum()
                cur_M = np.convolve(f, f).max()
                t = 0.3
                continue

        # maybe break when best_score < target

"""
Propose a near‑optimal non‑negative step‑function using a simple
LP‑based exchange heuristic.

The algorithm works on a moderate length (default 256) to keep the
linear program tiny.  Starting from the analytically optimal arcsine
distribution, we repeatedly:

1. Compute the current maximal autocorrelation M = max(a*a).
2. Solve the linear program
        maximise    sum(g)
        subject to   g ≥ 0
                     (a  * g)[k] ≤ M   for every convolution index k
   where the matrix of the convolution with the fixed a is built
   explicitly (size (2n−1)×n).  The solution g is a “direction” that can be
   mixed with a without increasing the maximal convolution.
3. Scale g to have the same ℓ₁‑norm as a, then replace a by a convex
   combination (1−t)·a + t·g (t starts at 0.4 and is reduced if the move
   does not improve the score).  This provably never increases the
   objective and often strictly decreases it.
4. Keep the best sequence seen.

The loop runs until the time budget expires (or the target bound
1.503 is reached).  The final sequence (as a plain list of floats) is
returned.
"""

import time
import numpy as np

# cvxpy is used for the small linear program.
# If unavailable, we fall back to a simple stochastic hill‑climber.
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:               # pragma: no cover
    _HAS_CVX = False


def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the arcsine (inverse‑square‑root) sequence of given length."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


def sequence_score(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = float(seq.sum())
    if s < 0.01:
        return np.inf
    b = np.convolve(seq, seq)
    return 2.0 * seq.size * float(b.max()) / (s * s)


def _convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ g)[k] = (a * g)[k]
    for k = 0 … 2n‑2.
    """
    n = a.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # For each row k we place a reversed slice of a.
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # a[k-i] runs from a[k-i_start] down to a[0] (or a[k] down to a[k-i_end]).
        # Vectorised assignment:
        A[k, i_start:i_end + 1] = a[k - np.arange(i_start, i_end + 1)]
    return A


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of floats – the best step‑function heights found within
    ``budget_s`` seconds.

    The search uses the LP‑based exchange described above; if cvxpy is not
    available it falls back to a simple stochastic local search.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a workable size.
    # The global variable `height_sequence_1` may be huge (several thousand);
    # we keep the problem small to make the LP cheap.
    # ------------------------------------------------------------------
    n = 256
    # If a user provided a different length via kwargs, honour it if reasonable.
    if "length" in kwargs:
        try:
            cand_n = int(kwargs["length"])
            if 64 <= cand_n <= 2048:
                n = cand_n
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Initialise with the analytically optimal arcsine shape,
    #     normalised to ℓ₁‑norm = 1 (the objective is scale‑invariant).
    # ------------------------------------------------------------------
    a = build_arcsine_sequence(n).astype(np.float64)
    a /= a.sum()                     # ℓ₁‑norm = 1
    best_seq = a.copy()
    best_score = sequence_score(best_seq)

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop (LP‑exchange + occasional random jump).
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.1)
    iteration = 0
    # Mixing weight – start fairly aggressive and shrink on failures.
    t = 0.4
    min_t = 0.02
    # How many consecutive LP attempts without improvement before a random jump.
    stagnation = 0
    max_stagnation = 30

    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # Compute current maximal autocorrelation (M) and build A.
        # --------------------------------------------------------------
        conv_aa = np.convolve(a, a)
        M = float(conv_aa.max())

        # If cvxpy is available we run the LP, otherwise we skip to fallback.
        if _HAS_CVX:
            # Build the convolution matrix for the current a.
            A = _convolution_matrix(a)

            # Linear program: maximise sum(g)  s.t.  A@g <= M,  g >= 0
            g = cp.Variable(n, nonneg=True)
            constraints = [A @ g <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
            except Exception:  # pragma: no cover
                # ECOS may fail on some platforms – fall back to SCS.
                prob.solve(solver=cp.SCS, max_iters=2000, verbose=False)

            if g.value is None:
                # Solver failed – abort the LP path and switch to fallback.
                _HAS_CVX = False
                continue

            g0 = np.array(g.value, dtype=np.float64).reshape(-1)
            sum_g0 = g0.sum()
            if sum_g0 < 1e-12:
                # No feasible positive direction – abort LP loop.
                break

            # Scale g0 to have the same ℓ₁‑norm as a (which is 1).
            g_scaled = g0 / sum_g0

            # ----------------------------------------------------------
            # Mixing step – try to improve with the current mixing weight.
            # ----------------------------------------------------------
            improved = False
            trial_t = t
            while trial_t >= min_t:
                a_candidate = (1.0 - trial_t) * a + trial_t * g_scaled
                # Ensure numerical non‑negativity.
                a_candidate = np.clip(a_candidate, 0.0, None)
                a_candidate /= a_candidate.sum()   # renormalise to 1

                conv_candidate = np.convolve(a_candidate, a_candidate)
                M_cand = float(conv_candidate.max())
                score_cand = 2.0 * n * M_cand  # denominator = 1²

                if score_cand < best_score - 1e-12:
                    # Accept update.
                    a = a_candidate
                    best_seq = a.copy()
                    best_score = score_cand
                    print(f"[{iteration:>5}] new best = {best_score:.6f} (t={trial_t:.3f})")
                    improved = True
                    # Slightly increase t for the next iteration to move faster.
                    t = min(0.9, trial_t * 1.1)
                    stagnation = 0
                    break
                else:
                    # Reduce t and retry.
                    trial_t *= 0.5

            if not improved:
                # No improvement: shrink mixing weight for future attempts.
                t = max(min_t, t * 0.7)
                stagnation += 1
        else:
            # --------------------------------------------------------------
            # Simple stochastic hill‑climber fallback (identical to the
            # reference hybrid local‑search).  Used only if cvxpy is missing.
            # --------------------------------------------------------------
            # pick a move type
            r = rng.random()
            if r < 0.35:
                # block scaling move
                width = rng.integers(max(1, n // 200), max(2, n // 12) + 1)
                start = rng.integers(0, max(1, n - width + 1))
                factor = np.exp(rng.normal(0.0, 0.12))
                cand = a.copy()
                cand[start:start + width] *= factor
                cand = cand / cand.sum()
            elif r < 0.55:
                # transfer move
                i = rng.integers(0, n)
                j = rng.integers(0, n - 1)
                if j >= i:
                    j += 1
                frac = rng.random() * 0.30
                delta = frac * a[i]
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = cand / cand.sum()
            elif r < 0.80:
                # early‑to‑late move
                early_cut = max(1, n // 10)
                late_start = max(early_cut, n - early_cut)
                i = rng.integers(0, early_cut)
                j = rng.integers(late_start, n)
                frac = rng.random() * 0.40
                delta = frac * a[i]
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = cand / cand.sum()
            else:
                # gradient‑like move (same idea as in the original code)
                b = np.convolve(a, a)
                max_b = b.max()
                mask = (b >= max_b * (1.0 - 1e-12)).astype(np.float64)
                contrib_conv = np.convolve(a, mask)
                contrib = a * contrib_conv[:n]
                heavy = int(np.argmax(contrib))
                light_candidates = np.where(contrib < contrib.mean())[0]
                if light_candidates.size == 0:
                    light = (heavy + 1) % n
                else:
                    light_candidates = light_candidates[light_candidates != heavy]
                    if light_candidates.size == 0:
                        light = (heavy + 1) % n
                    else:
                        light = int(rng.choice(light_candidates))
                delta = rng.random() * 0.30 * a[heavy]
                cand = a.copy()
                cand[heavy] -= delta
                cand[light] += delta
                cand = cand / cand.sum()

            cand_score = sequence_score(cand)
            cur_score = 2.0 * n * np.max(np.convolve(a, a)) / (a.sum() ** 2)

            if cand_score < cur_score:
                a = cand
                if cand_score < best_score:
                    best_seq = cand.copy()
                    best_score = cand_score
                    print(f"[{iteration:>5}] new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # Periodic random restart if stagnation is too long.
        # --------------------------------------------------------------
        if stagnation >= max_stagnation:
            a = build_arcsine_sequence(n).astype(np.float64)
            a /= a.sum()
            t = 0.4
            stagnation = 0
            print(f"[{iteration:>5}] restart (stagnation)")

        # Early exit if we have already beaten the target.
        if best_score < 1.503:
            break

    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    # Return as plain Python list for the harness.
    return [float(x) for x in best_seq]
