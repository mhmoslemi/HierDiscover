"""
Lightweight simulated‑annealing optimiser for the autocorrelation bound.

The routine builds the classic arcsine step‑function, optionally
augments it with a few cheap low‑dimensional sweeps (mixture with the
reversed arcsine, power‑law deformations and an exponential tilt), and
then runs a temperature‑controlled stochastic local search.
Moves are of two types:

* *directed transfer* – move a small amount of mass from the index that
  contributes most to the current maximal convolution coefficient to a
  low‑value index.  The convolution array is updated in O(n) time by a
  slice update, so millions of iterations fit easily into the time budget.
* *block perturbation* – a random contiguous block is multiplied by a
  log‑normal factor; this move is evaluated by recomputing the full
  convolution (rarely used).

The objective is 2·n·max(b)/sum(a)², which is invariant under scaling,
so the optimisation reduces to minimising the maximal autocorrelation
value `max(b)`.  The algorithm keeps the best sequence ever seen and
stops early if the target score < 1.503 is reached.

The function `propose_candidate` follows the required signature and
returns the best sequence found within the supplied time budget.
"""
import time
import numpy as np


def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the classic arcsine (inverse‑square‑root) sequence of the given length."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a  # values lie in (0, 1] for all i≥0


def compute_score(max_b: float, sum_a: float, n: int) -> float:
    """Objective value from the maximal convolution coefficient."""
    return 2.0 * n * max_b / (sum_a * sum_a)


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise

        2·n·max(conv(a,a)) / (sum(a)²)

    within the time budget `budget_s` (seconds).  The algorithm consists
    of a short parameter sweep followed by a simulated‑annealing style
    local search that uses cheap O(n) convolution updates.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256                      # fallback – never used in the real test

    # ------------------------------------------------------------------
    # 1.  Analytic starting point
    # ------------------------------------------------------------------
    a = build_arcsine_sequence(n)
    sum_a = a.sum()
    b = np.convolve(a, a)           # full convolution, length 2·n‑1
    cur_max = b.max()
    cur_score = compute_score(cur_max, sum_a, n)

    best_a = a.copy()
    best_max = cur_max
    best_score = cur_score

    print(f"[init] n={n}, arcsine score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Cheap low‑dimensional sweeps (mixture, power‑law, exponential tilt)
    # ------------------------------------------------------------------
    rev = a[::-1]

    # mixture with reversed arcsine
    for alpha in np.linspace(0.0, 0.80, 41):            # 0 … 0.8
        cand = (1.0 - alpha) * a + alpha * rev
        cand = np.clip(cand, 0.0, 1000.0)
        b_cand = np.convolve(cand, cand)
        max_cand = b_cand.max()
        sc = compute_score(max_cand, sum_a, n)
        if sc < best_score:
            best_a, best_max, best_score = cand, max_cand, sc
            print(f"[sweep] mix α={alpha:.3f} → score={sc:.6f}")

    # power‑law deformation of the arcsine sequence
    for p in np.linspace(0.70, 1.30, 13):               # 0.70 … 1.30
        cand = np.power(a, p)
        cand = np.clip(cand, 0.0, 1000.0)
        b_cand = np.convolve(cand, cand)
        max_cand = b_cand.max()
        sc = compute_score(max_cand, sum_a, n)
        if sc < best_score:
            best_a, best_max, best_score = cand, max_cand, sc
            print(f"[sweep] power p={p:.3f} → score={sc:.6f}")

    # exponential tilt: a_i ← a_i·λ^i
    for lam in np.linspace(0.60, 1.00, 9):              # λ ∈ [0.6,1]
        factor = lam ** np.arange(n)
        cand = a * factor
        cand = np.clip(cand, 0.0, 1000.0)
        b_cand = np.convolve(cand, cand)
        max_cand = b_cand.max()
        sc = compute_score(max_cand, sum_a, n)

            print(f"[sweep] tilt λ={lam:.3f} → score={sc:.6f}")

    # Start the local optimiser from the best candidate found so far
    a = best_a.copy()
    b = np.convolve(a, a)

    iteration = 0

    # temperature schedule – linear cooling from a fraction of cur_max

    while time.monotonic() < deadline:

            idxs = np.arange(low, high + 1)
            contrib = a[idxs] * a[k_max - idxs]

            src = int(idxs[np.argmax(contrib)])

            thr = np.percentile(a, 15)

            if src_val <= 0.0:

            # respect the global cap of 1000
            if a[tgt] + delta > 1000.0:
                delta = 1000.0 - a[tgt]
            if delta <= 0.0:

            # ---- incremental convolution update ---------------------------------
            # use the *old* a for the slice updates
            old_a = a
            b_new = b.copy()
            # contribution of moving δ away from `src`
            b_new[src:src + n] -= 2.0 * delta * old_a
            # contribution of adding δ to `tgt`
            b_new[tgt:tgt + n] += 2.0 * delta * old_a
            # quadratic correction for the pair (src,tgt)
            if src != tgt:
                b_new[src + tgt] -= 2.0 * delta * delta

            new_max = b_new.max()
            # ------------------------------------------------------------------
            # Acceptance test (simulated annealing)
            # ------------------------------------------------------------------
            accept = False
            if new_max < cur_max:
                accept = True
            elif temperature > 0:
                prob = np.exp(-(new_max - cur_max) / temperature)
                if rng.random() < prob:
                    accept = True

            if accept:

                a = a.copy()
                a[src] -= delta
                a[tgt] += delta
                b = b_new
                cur_max = new_max
                cur_score = compute_score(cur_max, sum_a, n)

                if cur_max < best_max:
                    best_a = a.copy()
                    best_max = cur_max
                    best_score = cur_score
                    print(f"[{iteration:>6}] new best = {best_score:.6f}")

                    if best_score < 1.503:
                        print("[stop] target reached")
                        break

        else:
            # ------------------------------------------------------------------
            # occasional random block perturbation (full recompute)
            # ------------------------------------------------------------------
            w_min = max(1, n // 200)
            w_max = max(2, n // 12)
            width = int(rng.integers(w_min, w_max + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.12)))

            cand = a.copy()
            cand[start:start + width] *= factor
            cand = np.clip(cand, 0.0, 1000.0)

            b_cand = np.convolve(cand, cand)
            cand_max = b_cand.max()
            cand_score = compute_score(cand_max, sum_a, n)

            if cand_score < best_score:
                best_a, best_max, best_score = cand, cand_max, cand_score
                a, b, cur_max = cand, b_cand, cand_max
                cur_score = cand_score
                print(f"[{iteration:>6}] block ⇒ new best = {best_score:.6f}")

                if best_score < 1.503:
                    print("[stop] target reached")
                    break
            elif cand_score < cur_score:
                # accept as an improving move (no SA for block moves)

                cur_score = cand_score
