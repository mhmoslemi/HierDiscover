Implementation of simplex_projection (as per Duchi et al.):

def simplex_projection(v):
    """Project vector v onto the probability simplex {x: x>=0, sum x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    end = min(k, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1)
        out[idx] = seq[k - idx]
    return out

"""
Propose a low‐value step‑function for the convolution minimisation task.

The algorithm works in two phases:

1. **Coarse phase (moderate length).**
   - Build a library of simple shapes (power‑law, exponential, triangular)
     on a moderate size (≈2500).  The best shape (by the fast objective) is kept.
   - Run a projected sub‑gradient descent on the probability simplex, where the
     descent direction is the (negative) sum of the sub‑gradients of all
     convolution entries attaining the current maximum.  A small amount of
     stochastic block perturbation and a Metropolis acceptance rule provide
     exploration.  Symmetry (a_i = a_{n‑1‑i}) is enforced at every step,
     which is optimal for the symmetric objective.

2. **Refinement phase (larger length).**
   - The best sequence from phase 1 is up‑sampled (linear interpolation) to a
     larger size (≈2×).  The same optimiser is applied again, now on the finer
     grid, using the remaining time budget.

Both phases use an exponential‑decay schedule for the step size and a
temperature that controls uphill moves.  The simplex projection (Duchi et al.)
keeps the iterates non‑negative and of unit sum, while clipping to the limits
required by ``evaluate_sequence`` (0 ≤ a_i ≤ 1000).  The best sequence found
before the time budget expires is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top level (no closures)
# ----------------------------------------------------------------------


def _evaluate_fast(seq: np.ndarray) -> float:
    """
    Fast replica of ``evaluate_sequence`` for non‑negative vectors whose sum
    is positive.  The result is identical to the official evaluator because
    scaling does not affect the objective.
    """
    total = seq.sum()
    if total <= 0.0:
        return float("inf")
    conv = np.convolve(seq, seq)          # full convolution (length 2n‑1)
    max_b = conv.max()
    n = seq.size
    return float(2.0 * n * max_b / (total * total))


def _shift(seq: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of (seq*seq)[k] with respect to ``seq``.
    Returns a vector d such that d_i = seq_{k-i} for valid indices,
    otherwise 0.
    """
    n = seq.size
    out = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1)
        out[idx] = seq[k - idx]
    return out


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    The block width is chosen between n//200 and n//12.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    return np.clip(cand, 0.0, 1000.0)


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    (implementation of Duchi‑Shalev‑Shwartz et al. 2008)
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidates(n: int) -> list[np.ndarray]:
    """
    Build a list of simple shape candidates for a given length ``n``.
    All candidates are normalised to sum to one, symmetrised and projected
    onto the simplex.
    """
    idx = np.arange(n, dtype=np.float64)          # 0‑based indices
    cand_list = []

    # ----- Power‑law family -----
    alphas = np.linspace(0.35, 0.65, 31)
    betas = [0.0, 0.25, 0.5, 1.0, 2.0]
    for alpha in alphas:
        for beta in betas:
            # i+1 to avoid division by zero
            seq = 1.0 / ((idx + 1.0 + beta) ** alpha)
            seq = np.clip(seq, 0.0, 1000.0)
            if seq.sum() == 0:
                continue
            seq /= seq.sum()
            seq = 0.5 * (seq + seq[::-1])
            seq = simplex_projection(seq)
            cand_list.append(seq)

    # ----- Exponential family -----
    lambdas = np.linspace(0.001, 0.020, 30)
    for lam in lambdas:
        seq = np.exp(-lam * idx)
        seq = np.clip(seq, 0.0, 1000.0)
        seq /= seq.sum()
        seq = 0.5 * (seq + seq[::-1])
        seq = simplex_projection(seq)
        cand_list.append(seq)

    # ----- Triangular (tent) family -----
    radii = [int(0.20 * n), int(0.30 * n), int(0.40 * n), int(0.45 * n)]
    centre = (n - 1) // 2
    for rad in radii:
        if rad <= 0:
            continue
        seq = np.maximum(0.0, 1.0 - np.abs(idx - centre) / rad)
        seq = np.clip(seq, 0.0, 1000.0)
        seq /= seq.sum()
        seq = 0.5 * (seq + seq[::-1])
        seq = simplex_projection(seq)
        cand_list.append(seq)

    # fallback uniform if nothing else survived
    if not cand_list:
        seq = np.ones(n) / n
        cand_list.append(seq)

    return cand_list


def _local_optimize(initial_seq: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with occasional block perturbations.
    Returns the best sequence found and its fast‑evaluated score.
    """
    n = initial_seq.size
    current_seq = initial_seq.copy()
    current_score = _evaluate_fast(current_seq)
    best_seq = current_seq.copy()
    best_score = current_score

    # Hyper‑parameters (tuned empirically)
    start_time = time.monotonic()
    eta_start = 0.25        # initial step size
    eta_end = 5e-5          # final step size
    temp_start = 5e-2       # initial temperature for Metropolis rule
    temp_end = 1e-6         # final temperature
    block_prob = 0.18       # probability of a random block move per iteration
    max_no_progress = 200000   # after this many stagnant iterations we may restart

    last_improve_it = 0
    it = 0
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        prog = elapsed / total if total > 0 else 1.0

        # exponential schedules
        eta = eta_start * (eta_end / eta_start) ** prog
        temp = temp_start * (temp_end / temp_start) ** prog

        # ----- compute active set (all indices attaining the max convolution) -----
        conv = np.convolve(current_seq, current_seq)
        max_conv = conv.max()
        eps = max(1e-12, max_conv * 1e-9)
        active_idxs = np.where(conv >= max_conv - eps)[0]

        # ----- sub‑gradient (average of shifts of the active constraints) -----
        direction = np.zeros_like(current_seq)
        for k in active_idxs:
            direction += _shift(current_seq, int(k))
        direction = -direction
        # make the direction tangent to the simplex (zero‑sum)
        direction -= direction.mean()

        # ----- propose a new candidate -----
        cand = current_seq + eta * direction

        if rng.random() < block_prob:
            cand = _make_block_perturbation(cand, rng)

        # projection onto the simplex and enforce symmetry
        cand = simplex_projection(cand)
        cand = 0.5 * (cand + cand[::-1])      # symmetry
        cand = simplex_projection(cand)      # re‑project after symmetrisation

        cand_score = _evaluate_fast(cand)

        # ----- Metropolis acceptance -----
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - current_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            current_seq = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve_it = it
                print(f"iter {it:07d} – new best: {best_score:.6f}")

        # ----- occasional random restart if we have been stuck for long -----
        if it - last_improve_it > max_no_progress and rng.random() < 0.001:
            # pick a fresh power‑law seed
            alpha = rng.uniform(0.35, 0.65)
            beta = rng.choice([0.0, 0.25, 0.5, 1.0, 2.0])
            idx = np.arange(n, dtype=np.float64)
            seq = 1.0 / ((idx + 1.0 + beta) ** alpha)
            seq = np.clip(seq, 0.0, 1000.0)
            seq /= seq.sum()
            seq = 0.5 * (seq + seq[::-1])
            seq = simplex_projection(seq)
            current_seq = seq
            current_score = _evaluate_fast(seq)
            print(f"iter {it:07d} – restart, score={current_score:.6f}")

        it += 1

    return best_seq, best_score


def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of ``seq`` to a longer length ``new_n``.
    The result is symmetrised and projected onto the simplex.
    """
    old_n = seq.size
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n, dtype=np.float64)
    up = np.interp(new_x, old_x, seq)
    up = np.clip(up, 0.0, 1000.0)
    up /= up.sum()
    up = 0.5 * (up + up[::-1])
    up = simplex_projection(up)
    return up
