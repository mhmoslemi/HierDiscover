def propose_candidate(seed=123, budget_s=500):
    import numpy as np, time, math
    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)

    # Choose length n: for speed, maybe 300. If height_sequence_1 exists and longer, we might pick min length.
    n_global = len(globals().get("height_sequence_1", []))
    n = min(max(50, n_global // 3), 500)  # picking moderate length.
    if n == 0:
        n = 400

    # Build initial arcsine
    f = build_arcsine_sequence(n)  
    # Clip
    f = np.clip(f, 0.0, 1000.0)
    # optionally scale to target sum sqrt(2n)
    target_sum = math.sqrt(2 * n)
    f = f * (target_sum / f.sum())

    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # parameters
    t = 0.1  # mixing factor
    iter_cnt = 0

    while time.monotonic() < deadline:
        iter_cnt += 1
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # Build constraint matrix A (2n-1 x n)
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        # Fill each row with f indices that contribute
        for k in range(rows):
            # i in [max(0, k-(n-1)), min(k, n-1)]
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # j = k - i: index in g
            # For each i, A[k,i] = f[i]
            A[k, i_start:i_end+1] = f[i_start:i_end+1]

        # Solve LP: max sum(g) => min -sum(g)
        c = -np.ones(n, dtype=np.float64)
        # RHS M repeated
        b = np.full(rows, M, dtype=np.float64)
        # bounds between 0 and 1000
        bounds = [(0, 1000.0) for _ in range(n)]

        # linprog
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # if fails, break
            print(f"[iter {iter_cnt}] LP failed: {res.message}")
            break

        g = res.x
        # combine
        f_new = (1 - t) * f + t * g
        # optional re-normalize sum to target for numerical stability
        # keep scaling to keep sum = target_sum
        f_new = f_new * (target_sum / f_new.sum())

        # evaluate
        score = evaluate_sequence(f_new.tolist())
        if not np.isfinite(score):
            continue

        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            print(f"[{iter_cnt}] new best {best_score:.6f}")

        # update f for next iteration
        f = f_new

        # adapt t: reduce gradually to improve stability
        if iter_cnt % 100 == 0:
            t = max(0.01, t * 0.9)

        # early stop if good enough
        if best_score < 1.45:
            break

    total_iters = iter_cnt
    print(f"[final] best score {best_score:.6f} after {total_iters} iterations")
    return [float(x) for x in best_f]

"""
Iterative LP‑based optimizer for the step‑function score.

The algorithm works with a fixed length n (chosen automatically, but never
larger than a few hundred – this keeps the linear program cheap).
We keep the ℓ¹‑norm fixed to the natural normalisation
    Σ a_i = sqrt(2·n)
so that the evaluation score is exactly the current maximal
convolution value M = max((a∗a)_k).  The goal is therefore to
reduce M.

One iteration:
    * Compute the current bound M = max conv(f,f).
    * Build the linear constraints (f∗g)_k ≤ M, which are *linear*
      in the unknown vector g (the convolution matrix is a Toeplitz
      sub‑matrix).  Together with 0 ≤ g_i ≤ 1000 this is a classic
      linear program.
    * Solve the LP, maximising Σ g_i.  The optimum puts as much extra
      mass as possible on indices where the constraints are slack
      (typically the far tail of the sequence).
    * Mix the old sequence f and the LP solution g, then rescale
      back to the fixed ℓ¹‑norm.  Because we keep the norm fixed,
      the score after the rescaling is exactly the new maximal
      convolution value; a drop in this number means an improvement.

The mix factor t is gradually decreased – large steps give quick
progress, small steps allow fine tuning.  The loop stops when the
time budget is exhausted or when no further improvement is seen.
The best sequence encountered is returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: arcsine sequence (the classical starting point)
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i) / 4**i  (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The method uses an LP that adds as much extra mass as possible without
    increasing the current maximal self‑convolution, then mixes the result
    back into the current sequence.  The loop runs until the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a workable length.
    #    If a reference exists we use a fraction of its size,
    #    otherwise we fall back to n≈300 which is fast for LP solves.
    # ------------------------------------------------------------------
    ref = globals().get("height_sequence_1", [])
    n = int(len(ref) * 0.05) if ref else 0
    if n < 60:
        n = 300                     # default length – cheap enough for many LP solves
    # Guard against absurdly large n
    n = min(n, 800)

    # Fixed ℓ¹‑norm (the normalisation used in the original description)
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 2. Starting sequence – the analytically optimal arcsine shape,
    #    rescaled to the required ℓ¹‑norm.
    # ------------------------------------------------------------------
    f = build_arcsine_sequence(n)
    f = np.clip(f, 0.0, 1000.0)
    f = f * (target_sum / f.sum())

    best_f = f.copy()
    best_score = float(evaluate_sequence(best_f.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Iterative improvement.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # keep a margin
    iteration = 0
    mix_factor = 0.30                # initial mixing weight
    no_improve = 0

    while time.monotonic() < deadline:
        iteration += 1

        # Current maximal self‑convolution (the score for a fixed sum)
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # ------------------------------------------------------------------
        # Build the linear programme:
        #   maximise    Σ g_i
        #   subject to  (f ∗ g)_k ≤ M   for every shift k
        #               0 ≤ g_i ≤ 1000
        # ------------------------------------------------------------------
        rows = 2 * n - 1                         # number of convolution points
        A = np.zeros((rows, n), dtype=np.float64)

        # Fill A: entry A[k, i] = f_i if 0 ≤ k-i < n else 0
        for k in range(rows):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # slice assignment – very cheap
            A[k, i_start:i_end + 1] = f[i_start:i_end + 1]

        c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i  → minimise -Σ g_i
        b = np.full(rows, M, dtype=np.float64)    # right‑hand side (all constraints ≤ M)

        bounds = [(0.0, 1000.0) for _ in range(n)]

        # Solve the LP (SciPy's HiGHS interior‑point method is fast for these sizes)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})

        if not res.success:
            # If the LP fails we simply fall back to a tiny random perturbation.
            print(f"[iter {iteration}] LP failed ({res.message}); applying random tweak.")
            tweak = rng.normal(scale=1e-3, size=n)
            f = np.clip(f + tweak, 0.0, 1000.0)
            f = f * (target_sum / f.sum())
            continue

        g = res.x

        # ------------------------------------------------------------------
        # Mix the old sequence with the LP solution.
        # The mix factor is gradually reduced to allow fine tuning.
        # ------------------------------------------------------------------
        f_mix = (1.0 - mix_factor) * f + mix_factor * g

        # Rescale back to the fixed ℓ¹‑norm (keeps the score invariant but
        # keeps numbers in a comfortable range).
        f_mix = f_mix * (target_sum / f_mix.sum())

        # Evaluate the new candidate.
        score = float(evaluate_sequence(f_mix.tolist()))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = f_mix.copy()
            print(f"[{iteration:>5}] new best score = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # Prepare for next iteration.
        f = f_mix

        # Adaptive schedule for the mixing weight.
        if iteration % 200 == 0:
            mix_factor = max(0.02, mix_factor * 0.85)

        # Stop early if we have already reached a comfortable target.
        if best_score < 1.45:
            break

        # Break if we see no improvement for a long while.
        if no_improve >= 800:
            break

    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best_f]
