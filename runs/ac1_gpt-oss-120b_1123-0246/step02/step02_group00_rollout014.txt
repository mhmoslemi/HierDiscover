def _make_constraint_matrix(f):
    n = f.size
    # shape (2n-1, n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        # shift f for this k
        # start index i where k-i in [0, n-1] => i from max(0, k-(n-1)) to min(k, n-1)
        start = max(0, k-(n-1))
        end = min(k, n-1)
        # for i in start..end: A[k, i] = f[k-i]
        A[k, start:end+1] = f[k - np.arange(start, end+1)]
    return A

def _constraint_matrix(f):
    n = f.size
    rows = 2*n - 1
    A = sp.dok_matrix((rows, n), dtype=np.float64)
    for k in range(rows):
        start = max(0, k - (n-1))
        end = min(k, n-1)
        # indices where i runs from start to end inclusive
        # for each i, set A[k, i] = f[k-i]
        idx = np.arange(start, end+1)
        A[k, idx] = f[k - idx]
    return A.tocsr()
```

"""
Propose a low‑value step‑height sequence for `evaluate_sequence`.

The algorithm works in two phases under the given time budget:

1. **Exploratory phase (cheap stochastic moves).**
   * Starts from the best known sequence if `height_sequence_1` exists,
     otherwise from a power‑law seed.
   * Uses a gradient‑like update that pushes probability mass away from
     the indices that contribute most to the current maximal convolution.
   * Occasionally applies a multiplicative log‑normal block perturbation
     and a mild symmetrisation.  Acceptance follows a Metropolis rule
     with an exponential cooling schedule.

2. **Refinement phase (linear‑programming step).**
   * For the current vector `f` we compute `M = max(f*f)`.
   * We solve the LP  
        maximise   Σ b_i  
        subject to (f * b)_k ≤ M   (for all k) ,  b_i ≥ 0
     where `(f * b)_k` is a linear form in `b` (the Toeplitz convolution
     matrix of `f`).  The LP is solved with SciPy’s HiGHS solver.
   * The optimum `b` is scaled to have unit ℓ₁‑norm, yielding `g`.
   * A convex combination `h = (1‑t)·f + t·g` (t≈0.4) is formed;
     because both `f` and `g` respect the bound `M`, also `h` does.
   * If `h` improves the evaluation score it is kept; otherwise a small
     random block perturbation is tried.

The two phases are interleaved until the time budget expires.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np
import scipy.optimize
import scipy.linalg

# ----------------------------------------------------------------------
# Fast evaluation (identical to evaluate_sequence up to the clipping step)
# ----------------------------------------------------------------------
def _evaluate_fast(seq: np.ndarray) -> float:
    """2 · n · max(conv(seq,seq)) / (sum(seq)²) – same scaling as evaluate_sequence."""
    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    n = seq.size
    return float(2.0 * n * max_b / (total * total))

# Helpers for the stochastic gradient phase
# ----------------------------------------------------------------------
def _shift(seq: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of (seq*seq)[k] w.r.t. seq.
    Returns the vector d_i = seq_{k-i} (zero outside the domain).
    """
    n = seq.size
    out = np.zeros_like(seq)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1)
        out[idx] = seq[k - idx]
    return out


def _gradient_step(seq: np.ndarray, eta: float, top_k: int = 12) -> np.ndarray:
    """One projected gradient step based on the `top_k` largest convolution entries."""
    conv = np.convolve(seq, seq)
    if top_k >= conv.size:
        ks = np.arange(conv.size)
    else:
        ks = np.argpartition(-conv, top_k)[:top_k]
    shift_sum = np.zeros_like(seq)
    for k in ks:
        shift_sum += _shift(seq, int(k))
    d = shift_sum / len(ks)               # average sub‑gradient
    direction = -d + np.mean(d)           # zero‑sum direction (keeps the sum unchanged)
    new_seq = seq + eta * direction
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    s = new_seq.sum()
    if s > 0:
        new_seq /= s
    else:
        new_seq = seq.copy()
    return new_seq


def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = start + width
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional symmetrisation
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Linear‑programming refinement phase
# ----------------------------------------------------------------------
def _make_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense Toeplitz matrix A (shape (2n‑1, n)) such that
    A @ b = np.convolve(f, b) for any vector b of length n.
    """
    n = f.size
    # first column = f followed by n‑1 zeros
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # first row = f[0] followed by n‑1 zeros
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    A = scipy.linalg.toeplitz(c, r)       # dense (2n‑1) × n matrix
    return A


def _lp_step(f: np.ndarray, M: float, rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve   max Σ b_i   subject to   (f * b)_k ≤ M  ∀k,   b_i ≥ 0.
    Returns the optimal b (or None if the solver fails).
    """
    n = f.size
    A = _make_constraint_matrix(f)        # shape (2n‑1, n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)     # minimise –Σb_i  ⇒ maximise Σb_i
    bounds = [(0.0, None)] * n
    # HiGHS is the default method and works well for medium‑sized LPs
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                bounds=bounds, method='highs',
                                options={"presolve": True, "time_limit": 30})
    if res.success:
        return res.x
    # Solver failed – fall back to a random perturbation later
    return None

        else:
            cand = _gradient_step(current, eta, top_k=12)

        # occasional symmetrisation helps flatten the convolution
        if rng.random() < 0.12:
            cand = 0.5 * (cand + cand[::-1])

        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand / cand.sum()
        cand_score = _evaluate_fast(cand)

        # ------------------------------------------------------------------
        # Metropolis acceptance
        # ------------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - current_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"grad‑phase iteration {it:06d} – new best {best_score:.6f}")

        it += 1

        # early exit if the time budget is almost exhausted
        if time.monotonic() >= deadline - 30.0:
            break

    # --------------------------------------------------------------
    # Phase 2 – linear‑programming refinement
    # --------------------------------------------------------------
    # use a fixed mixing factor (t) that worked well in practice
    mix_factor = 0.4

    while time.monotonic() < deadline:
        # compute the current maximal convolution value M
        conv = np.convolve(current, current)
        M = conv.max()

        # solve the LP to obtain a vector b with larger ℓ₁‑mass
        b = _lp_step(current, M, rng)

        if b is not None and b.sum() > 0:
            # scale to unit sum (our evaluation uses a sum of 1)
            scale = 1.0 / b.sum()
            g = b * scale
            # convex combination – stays feasible w.r.t. the bound M
            cand = (1.0 - mix_factor) * current + mix_factor * g
        else:
            # LP failed – fall back to a gentle block perturbation
            cand = _make_block_perturbation(current, rng)

        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand / cand.sum()
        cand_score = _evaluate_fast(cand)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"LP‑phase – new best {best_score:.6f}")

        # accept only if we actually improve the score (keeps the run stable)
        if cand_score < current_score:
            current = cand
            current_score = cand_score

    # ----------------------------------------------------------------------
    # Final sanity checks – the external evaluator will clip again
    # ----------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s

    print(f"finished after {it} gradient iters, best score = {best_score:.6f}")
    return best_seq.tolist()
