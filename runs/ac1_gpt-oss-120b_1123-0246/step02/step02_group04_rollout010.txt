"""
A hybrid optimiser for the convolution‑bound problem.

The algorithm starts from the classic “arcsine” (inverse‑square‑root) shape,
symmetrises it and then repeatedly tries two kinds of updates:

1. **Linear‑programming step** – a small coordinated change Δa is computed
   by solving a tiny LP that minimises a linearised upper bound on the
   maximal convolution entry (only the currently worst few shifts are
   considered).  The LP respects non‑negativity and a limited step size,
   thus producing a sensible descent direction.

2. **Randomised local moves** – block perturbations, mass transfers or a
   targeted reduction of the dominant convolution term (as in the reference
   simulated‑annealing code).  These keep the search stochastic and help
   escape shallow minima.

All candidates are mirrored to keep the sequence symmetric (a_i = a_{n‑1‑i}),
clipped to the evaluator limits, and the best sequence seen so far is
returned.  The routine runs for at most ``budget_s`` seconds (default 500 s)
and never exceeds the time limit.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# 1.  Analytic start – the arcsine (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# 2.  Simple stochastic moves (same ideas as the reference code).
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    return np.clip(cand, 0.0, 1000.0)


def transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    # move at most 30 % of the source
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def swap_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Exchange two entries."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    cand = base.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand


def targeted_move(base: np.ndarray,
                  conv: np.ndarray,
                  k_max: int,
                  rng: np.random.Generator) -> np.ndarray:
    """Reduce the dominant convolution entry by moving a small amount of mass."""
    n = base.size
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idxs = np.arange(i_start, i_end + 1)
    partners = k_max - idxs
    products = base[idxs] * base[partners]
    max_pos = int(np.argmax(products))
    i = int(idxs[max_pos])
    j = int(partners[max_pos])

    l1, l2 = rng.choice(n, size=2, replace=False)

    max_move = min(base[i], base[j])
    if max_move == 0:
        return base.copy()
    delta = rng.random() * 0.20 * max_move

    cand = base.copy()
    if i == j:
        cand[i] -= 2.0 * delta
        cand[l1] += delta
        cand[l2] += delta
    else:
        cand[i] -= delta
        cand[j] -= delta
        cand[l1] += delta
        cand[l2] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# 3.  Linear‑programming step (core of the new optimiser).
# ----------------------------------------------------------------------
def lp_step(a: np.ndarray,
            conv: np.ndarray,
            top_k: int = 5,
            step_frac: float = 0.15):
    """
    Solve a tiny LP that finds a feasible delta with
        a_i + delta_i >= 0
        sum(delta_i) = 0
        conv_k + 2*sum_i a_{k-i}*delta_i <= t    for the selected k's
    and minimises t (a linearised bound on the new maximal convolution).

    Returns (delta, t) if successful, otherwise None.
    """
    n = a.shape[0]
    # pick the worst (largest) convolution entries
    idxs = np.argpartition(conv, -top_k)[-top_k:]

    # Build inequality matrix A_ub * x <= b_ub where x = [delta, t]
    rows = []
    for k in idxs:
        coeff = np.zeros(n + 1, dtype=np.float64)          # delta_0..delta_{n-1}, t
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx                                  # partner index
        coeff[i_idx] = 2.0 * a[j_idx]                      # derivative 2*a_{k-i}
        coeff[n] = -1.0                                     # -t term
        rows.append(coeff)

    A_ub = np.vstack(rows)
    b_ub = -conv[idxs].astype(np.float64)

    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0], dtype=np.float64)

    # Bounds for delta_i: keep non‑negative entries and cap the step size
    lower = -a
    avg = float(a.mean())
    upper = step_frac * (a + avg)          # allow a modest increase everywhere
    bounds = [(float(l), float(u)) for l, u in zip(lower, upper)]
    bounds.append((None, None))            # t is free (will be minimised)

    # Objective: minimise t
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    # Solve with HiGHS (fast interior‑point for large variable counts)
    res = linprog(c,
                  A_ub=A_ub,
                  b_ub=b_ub,
                  A_eq=A_eq,
                  b_eq=b_eq,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True})

    if not res.success:
        return None
    delta = res.x[:n]
    t_val = res.x[-1]
    return delta, t_val

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,

    """
    Search for a non‑negative step function with a small evaluation score.
    The routine runs for at most *budget_s* seconds and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size and initial sequence.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        n = init_seq.size
    else:
        n = 256
        init_seq = build_arcsine_sequence(n)

    # Start from a symmetric version of the arcsine shape.
    cur = 0.5 * (init_seq + init_seq[::-1])
    cur = np.clip(cur, 0.0, 1000.0)

    best = cur.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    cur_score = best_score
    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    it = 0
    step_frac = 0.15          # initial LP step size factor
    min_step = 1e-4
    step_decay = 0.98
    top_k = 6                 # how many worst shifts are guarded in the LP

    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------------
        # 1.  Compute convolution and attempt an LP‑guided move.
        # ----------------------------------------------------------
        conv = np.convolve(cur, cur)
        lp_res = lp_step(cur, conv, top_k=top_k, step_frac=step_frac)

        if lp_res is not None:
            delta, t_est = lp_res
            cand = cur + delta
            # enforce symmetry, bounds and positivity
            cand = 0.5 * (cand + cand[::-1])
            cand = np.clip(cand, 0.0, 1000.0)

            cand_score = float(evaluate_sequence(cand.tolist()))
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best = cand
                    best_score = cand_score
                    print(f"[{it:6d}] LP improvement → {best_score:.6f}")
                # after a success we can be slightly more aggressive
                step_frac = min(step_frac * 1.07, 0.30)
                continue
            else:
                # no improvement – shrink step size
                step_frac = max(step_frac * step_decay, min_step)

        # ----------------------------------------------------------
        # 2.  Fallback to a cheap stochastic move.
        # ----------------------------------------------------------
        move_type = rng.random()
        if move_type < 0.33:
            cand = block_perturbation(cur, rng)
        elif move_type < 0.66:
            cand = transfer_move(cur, rng)
        else:
            # targeted move needs the current worst shift
            k_max = int(np.argmax(conv))
            cand = targeted_move(cur, conv, k_max, rng)

        # keep symmetry (helps the search) and clip
        cand = 0.5 * (cand + cand[::-1])
        cand = np.clip(cand, 0.0, 1000.0)
        cand_score = float(evaluate_sequence(cand.tolist()))

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                print(f"[{it:6d}] random improvement → {best_score:.6f}")

        # occasional global shake‑up
        if it % 4000 == 0 and rng.random() < 0.4:
            cur = build_arcsine_sequence(n)
            cur = 0.5 * (cur + cur[::-1])
            cur = np.clip(cur, 0.0, 1000.0)
            cur_score = float(evaluate_sequence(cur.tolist()))
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
                print(f"[{it:6d}] re‑seeded, new best {best_score:.6f}")

        # gradually tighten LP parameters if we are stuck
        if it % 2000 == 0:
            step_frac = max(step_frac * step_decay, min_step)

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    return [float(x) for x in best]
