# define fast objective
def _fast_objective(sequence: np.ndarray) -> float:
    """Compute 2*n*max(conv(seq,seq))/sum(seq)^2 using FFT.

    Sequence is assumed 1‑D real array; values are clipped to [0,1000] as
    evaluate_sequence does. Returns np.inf for degenerate input.
    """
    seq = np.clip(sequence, 0.0, 1000.0)
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return np.inf
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    # Real FFT for speed
    A = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)

def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Return a symmetric Beta(α,α) shape of length n whose sum equals sqrt(2n).

    The values are not clipped – this function is used only for building
    candidate sequences, the final clipping is performed by _fast_objective.
    """
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    # handle extreme alpha to avoid overflow/underflow at p=0 or 1
    # exponent = alpha-1 which may be negative.
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, alpha - 1.0)
    target = sqrt(2.0 * n)
    total = w.sum()
    if total == 0.0:
        w = np.ones_like(w)
        total = w.sum()
    return w * (target / total)

"""
Optimize a non‑negative step function to minimise the provided
`evaluate_sequence` score.

The algorithm works in three stages:

1. **Coarse grid search** over length `n`, a symmetric Beta(α,α) shape,
   and a mixing weight `λ` between that shape and a flat background.
   Both components are normalised to have the same total sum, so the
   mixing keeps the sum invariant.

2. **Local refinement** of the two continuous parameters (α and λ)
   using a golden‑section search on each coordinate while the other is
   held fixed.

3. **Stochastic hill‑climbing** on the concrete sequence:
   repeatedly transfer a tiny amount of mass from one index to another,
   accepting improvements and occasionally accepting worse moves
   (simulated‑annealing style).  The best sequence found is finally
   evaluated with the official `evaluate_sequence` function and returned.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective (max of self‑convolution)
# ----------------------------------------------------------------------
def _fast_objective(seq: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(conv(seq,seq)) / (sum(seq)^2)
    using an FFT.  Values are clipped to the interval [0,1000] exactly as
    `evaluate_sequence` does.
    """
    a = np.clip(seq, 0.0, 1000.0)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())       # next power of two
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Helper: generate a symmetric Beta(α,α)‑shaped sequence of length n
# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Return a symmetric Beta(α,α) density sampled at the mid‑points
    p_i = (i+0.5)/n, normalised so that its sum equals √(2·n).
    """
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    # p^{α‑1}(1‑p)^{α‑1}  – for α<1 this diverges near the edges
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, alpha - 1.0)
    target = sqrt(2.0 * n)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    return w * (target / w_sum)

# Helper: convex combination of a flat background and a Beta shape
# ----------------------------------------------------------------------
def _mixed_sequence(n: int, uniform: np.ndarray, alpha: float, lam: float) -> np.ndarray:
    """
    Build a candidate of length n as (1‑λ)·uniform + λ·Beta(α,α).
    Both ingredients already sum to √(2·n), so the mixed sequence does too.
    """
    beta_seq = _beta_shape(n, alpha)
    return (1.0 - lam) * uniform + lam * beta_seq

# Main entry point required by the runner
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) achieves a
    low `evaluate_sequence` score within the allotted time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget in seconds (the caller reserves a small
        safety margin automatically).
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)

    best_seq = None
    best_score = np.inf
    best_params = None               # (n, alpha, lam)

    # --------------------------------------------------------------
    # Optional: start from the global sequence if it is available
    # --------------------------------------------------------------
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)
        seq0 = np.clip(seq0, 0.0, 1000.0)
        sc0 = _fast_objective(seq0)
        if np.isfinite(sc0) and sc0 < best_score:
            best_score = sc0
            best_seq = seq0.copy()
            best_params = None
            print(f"[init] height_sequence_1 score = {sc0:.6f}")
    except NameError:
        pass

    # --------------------------------------------------------------
    # Phase 1 – coarse grid over (n, α, λ)
    # --------------------------------------------------------------
    n_candidates = [1024, 2048, 4096, 8192, 16384, 32768, 65536]
    alpha_grid = np.concatenate((
        np.linspace(0.02, 0.8, 15),
        np.linspace(0.3, 1.0, 15)
    ))
    alpha_grid = np.unique(alpha_grid)
    lam_grid = np.linspace(0.0, 1.0, 11)   # 0 = pure uniform, 1 = pure Beta

    for n in n_candidates:
        if time.monotonic() >= deadline:
            break
        uniform = np.full(n, sqrt(2.0 * n) / n, dtype=np.float64)   # sum = √(2n)

        for alpha in alpha_grid:
            if time.monotonic() >= deadline:
                break
            beta_seq = _beta_shape(n, alpha)   # sum = √(2n)
            for lam in lam_grid:
                cand = (1.0 - lam) * uniform + lam * beta_seq
                score = _fast_objective(cand)
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = cand.copy()
                    best_params = (n, alpha, lam)
                    print(f"[grid] n={n:6d} α={alpha:5.3f} λ={lam:4.2f} → score={score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – coordinate golden‑section refinement of α and λ
    # --------------------------------------------------------------
    if best_params is not None:
        n_best, alpha_best, lam_best = best_params
        uniform = np.full(n_best, sqrt(2.0 * n_best) / n_best, dtype=np.float64)

        # ---- refine α (holding λ fixed) ---------------------------------
        for _ in range(3):          # a few alternating refinement rounds
            if time.monotonic() >= deadline:
                break
            lo_a = max(0.01, alpha_best - 0.20)
            hi_a = min(2.0,  alpha_best + 0.20)
            phi = (1.0 + 5 ** 0.5) / 2.0

            c = hi_a - (hi_a - lo_a) / phi
            d = lo_a + (hi_a - lo_a) / phi
            seq_c = _mixed_sequence(n_best, uniform, c, lam_best)
            f_c = _fast_objective(seq_c)
            seq_d = _mixed_sequence(n_best, uniform, d, lam_best)
            f_d = _fast_objective(seq_d)

            while time.monotonic() < deadline and (hi_a - lo_a) > 1e-5:
                if f_c < f_d:
                    hi_a, seq_d, f_d = d, seq_c, f_c
                    d = c
                    c = hi_a - (hi_a - lo_a) / phi
                    seq_c = _mixed_sequence(n_best, uniform, c, lam_best)
                    f_c = _fast_objective(seq_c)
                else:
                    lo_a, seq_c, f_c = c, seq_d, f_d
                    c = d
                    d = lo_a + (hi_a - lo_a) / phi
                    seq_d = _mixed_sequence(n_best, uniform, d, lam_best)
                    f_d = _fast_objective(seq_d)

                if f_c < best_score:
                    best_score = f_c
                    best_seq = seq_c.copy()
                if f_d < best_score:
                    best_score = f_d
                    best_seq = seq_d.copy()

            alpha_best = c if f_c < f_d else d
            # ---- refine λ (holding α fixed) -----------------------------
            lo_l = max(0.0, lam_best - 0.30)
            hi_l = min(1.0, lam_best + 0.30)
            phi = (1.0 + 5 ** 0.5) / 2.0

            c = hi_l - (hi_l - lo_l) / phi
            d = lo_l + (hi_l - lo_l) / phi
            seq_c = _mixed_sequence(n_best, uniform, alpha_best, c)
            f_c = _fast_objective(seq_c)
            seq_d = _mixed_sequence(n_best, uniform, alpha_best, d)
            f_d = _fast_objective(seq_d)

            while time.monotonic() < deadline and (hi_l - lo_l) > 1e-5:
                if f_c < f_d:
                    hi_l, seq_d, f_d = d, seq_c, f_c
                    d = c
                    c = hi_l - (hi_l - lo_l) / phi
                    seq_c = _mixed_sequence(n_best, uniform, alpha_best, c)
                    f_c = _fast_objective(seq_c)
                else:
                    lo_l, seq_c, f_c = c, seq_d, f_d
                    c = d
                    d = lo_l + (hi_l - lo_l) / phi
                    seq_d = _mixed_sequence(n_best, uniform, alpha_best, d)
                    f_d = _fast_objective(seq_d)

                if f_c < best_score:
                    best_score = f_c
                    best_seq = seq_c.copy()
                if f_d < best_score:
                    best_score = f_d
                    best_seq = seq_d.copy()

            lam_best = c if f_c < f_d else d

        print(f"[refine] n={n_best} α={alpha_best:.6f} λ={lam_best:.6f} → score≈{best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – stochastic hill‑climbing on the concrete sequence
    # --------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        n = a.size
        target_sum = sqrt(2.0 * n)
        avg = target_sum / n
        step_scale = max(avg * 0.5, 1e-4)          # typical move size
        temp = best_score * 0.02                  # initial “temperature”
        decay = 0.9999

        it = 0
        while time.monotonic() < deadline:
            it += 1
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = rng.random() * step_scale
            if a[i] < delta:
                continue
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            sc = _fast_objective(cand)

            # deterministic improvement or Metropolis acceptance
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                a = cand
                if it % 2000 == 0:
                    print(f"[hill] iter={it} → score={best_score:.6f}")
            else:
                if temp > 0 and rng.random() < np.exp(-(sc - best_score) / temp):
                    a = cand
            temp *= decay

    # --------------------------------------------------------------
    # Final evaluation with the official routine
    # --------------------------------------------------------------
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] evaluate_sequence score = {final_score:.6f}")

    # Return as ordinary Python list of floats
    return [float(v) for v in best_seq]
