"""
Propose a low‑score height sequence using projected Adam optimisation
with exact sub‑gradient of the max‑convolution.

The algorithm works on the simplex of non‑negative numbers with a fixed
L1‑norm (the standard normalisation √(2 n)).  At each iteration we

  * compute the full self‑convolution a∗a,
  * identify the lag(s) attaining the maximal value,
  * build a weight vector that is uniform on those active lags,
  * obtain the exact sub‑gradient of the max‑convolution by a single
    FFT‑based convolution,
  * perform an Adam update and finally project back onto the simplex
    (non‑negative, prescribed sum).

Occasional log‑normal perturbations and a stagnation‑restart policy
prevent the optimiser from getting trapped in local minima.  The best
ever seen sequence is kept and returned at the end of the allotted
time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT convolution (linear convolution, real inputs)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT of power‑of‑two length."""
    n = x.size + y.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, sum(x)=s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Exact sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def _subgradient_max_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of max_k (a∗a)_k.
    `b` must be the full convolution a∗a (length 2n‑1).
    """
    n = a.size
    maxb = b.max()
    # indices where the maximum is attained (within a tiny tolerance)
    tol = maxb * 1e-12 + 1e-12
    active = np.where(b >= maxb - tol)[0]
    w = np.zeros_like(b, dtype=np.float64)
    if active.size > 0:
        w[active] = 1.0 / active.size
    else:                     # fallback – should never happen
        w[n - 1] = 1.0
    # gradient = 2 * a * w_rev   (convolution)
    w_rev = w[::-1]
    conv_rev = _fft_convolve(a, w_rev)            # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)              # indices that give the gradient
    grad = 2.0 * conv_rev[idx]
    return grad

# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 900.0, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence().  The function returns the best sequence found
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.2)   # safety margin

    # --------------------------------------------------------------
    # 1. Choose a starting length and initialise the vector
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        # keep a reasonable size (max 12 000 entries)
        max_len = 12000
        if base.size > max_len:
            lo = (base.size - max_len) // 2
            base = base[lo:lo + max_len]
        n = base.size
        a = np.clip(base, 0.0, 1000.0).copy()
    else:
        n = 5000                           # a moderate default
        a = np.ones(n, dtype=np.float64)

    # Normalise to the standard L1‑norm √(2 n); this makes the
    # evaluate_sequence() value equal to max_b.
    target_sum = np.sqrt(2.0 * n)
    if a.sum() == 0.0:
        a[:] = 1.0
    a *= (target_sum / a.sum())
    a = np.clip(a, 0.0, 1000.0)
    a *= (target_sum / a.sum())            # second normalisation after clipping

    # --------------------------------------------------------------
    # 2. Book‑keeping
    # --------------------------------------------------------------
    best_seq = a.copy()
    b_full = _fft_convolve(a, a)
    best_score = 2.0 * n * b_full.max() / (a.sum() ** 2)   # equals max_b
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # learning‑rate schedule (linear decay)
    lr0 = 0.2                      # initial step size
    iteration = 0
    last_improve = 0

    # pre‑compute index array needed for the gradient extraction
    grad_idx = (2 * n - 2) - np.arange(n)

    # --------------------------------------------------------------
    # 3. Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        # --- convolution and sub‑gradient ---------------------------------
        b_full = _fft_convolve(a, a)                 # length 2n‑1
        maxb = b_full.max()
        grad = _subgradient_max_convolution(a, b_full)

        # --- Adam update -------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)

        # linear decay of the base learning rate
        frac = (time.monotonic() - start) / budget_s
        lr = lr0 * (1.0 - frac)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        # project back onto the simplex (preserves non‑negativity and sum)
        a_candidate = _project_to_simplex(a_candidate, target_sum)

        # --------------------------------------------------------------
        # 4. Evaluate the candidate
        # --------------------------------------------------------------
        b_cand = _fft_convolve(a_candidate, a_candidate)
        cand_score = 2.0 * n * b_cand.max() / (a_candidate.sum() ** 2)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = a_candidate.copy()
            a = a_candidate
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"[{iteration:7d}] new best  score = {best_score:.6f}")
        else:
            a = a_candidate

        # --------------------------------------------------------------
        # 5. Periodic random multiplicative perturbation
        # --------------------------------------------------------------
        if iteration % 500 == 0:
            pert = np.exp(rng.normal(0.0, 0.07, size=n))
            a_pert = best_seq * pert
            a_pert = _project_to_simplex(a_pert, target_sum)
            b_pert = _fft_convolve(a_pert, a_pert)
            pert_score = 2.0 * n * b_pert.max() / (a_pert.sum() ** 2)
            if pert_score < best_score:
                best_score = pert_score
                best_seq = a_pert.copy()
                a = a_pert
                m.fill(0.0)
                v.fill(0.0)
                last_improve = iteration
                print(f"[{iteration:7d}] perturb improved  score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 6. Stagnation restart
        # --------------------------------------------------------------
        if iteration - last_improve > 3000:
            a = rng.random(n)
            a = _project_to_simplex(a, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"[{iteration:7d}] restart due to stagnation")

    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    # Return as a plain Python list of floats
    return [float(v) for v in best_seq]
