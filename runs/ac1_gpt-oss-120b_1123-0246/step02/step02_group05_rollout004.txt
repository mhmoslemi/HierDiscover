"""
Hybrid deterministic–stochastic optimiser for the self‑convolution bound.

We start from the analytically good “arcsine” sequence (a[i] = C(2i,i)/4**i)
and iteratively improve it.
The main deterministic step uses an exponential re‑weighting that reduces
the contribution of indices that are most responsible for the current
maximum convolution entry.
Contributions are obtained from the convolution of the sequence with a
binary mask that marks the (near‑)maximal positions.
The update preserves the ℓ₁‑norm by renormalising to sum = 1.
If a deterministic step does not improve the exact objective we fall back
to cheap stochastic moves (block scaling, mass transfers, early‑to‑late
shifts).  The best sequence seen so far is remembered and returned.
The whole search respects a time budget (default ≈ 900 s) and prints
progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: arcsine (inverse‑square‑root) sequence
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence: a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def renormalise(seq: np.ndarray, target_sum: float = 1.0) -> np.ndarray:
    """Scale seq so that its sum equals target_sum (clipping first)."""
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur == 0.0:
        # fall back to a uniform distribution
        seq[:] = target_sum / seq.size
        return seq
    seq *= target_sum / cur
    return seq


# ----------------------------------------------------------------------
# Stochastic move building blocks (identical to the previous solution)
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                 # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return renormalise(cand, target_sum)


def transfer_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """Move a random fraction (≤30 %) of mass from one index to another."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1                 # guarantee i != j
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


def early_to_late_move(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Move a fraction of mass from the early part of the sequence to its
    late part (keeps the total sum unchanged)."""
    n = base.size
    early_cut = max(1, n // 10)          # first 10 % of positions
    late_start = max(early_cut, n - early_cut)   # last 10 %
    i = int(rng.integers(0, early_cut))
    j = int(rng.integers(late_start, n))
    if i == j:
        return base.copy()
    frac = rng.random() * 0.40           # up to 40 % of the chosen early entry
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


# ----------------------------------------------------------------------
# Core deterministic step (exponential re‑weighting)
# ----------------------------------------------------------------------
def exponential_reweight(a: np.ndarray,
                         max_b: float,
                         mask: np.ndarray,
                         lr: float) -> np.ndarray:
    """
    Reduce the current maximum convolution entry.
    *a* – current sequence (sum = 1)
    *max_b* – current maximal self‑convolution value
    *mask* – binary mask (1 on positions attaining the max, 0 elsewhere)
    *lr* – step size (larger → stronger re‑weighting)
    Returns a new candidate (renormalised to sum = 1).
    """
    n = a.size
    # convolution of a with mask (first n entries) – gives per‑index
    # contribution to the maximal convolution positions.
    L = 1 << ((3 * n - 2).bit_length())
    Fa = np.fft.rfft(a, n=L)
    Fmask = np.fft.rfft(mask.astype(np.float64), n=L)
    conv_full = np.fft.irfft(Fa * Fmask, n=L)[:n]   # only first n entries needed
    contrib = a * conv_full                         # element‑wise product
    mean_c = contrib.mean()
    # exponential re‑weighting
    delta = contrib - mean_c
    # Clip exponent to avoid overflow/underflow
    ex = -lr * delta
    ex = np.clip(ex, -20.0, 20.0)
    factor = np.exp(ex)
    cand = a * factor
    return renormalise(cand, 1.0)


# ----------------------------------------------------------------------
# Objective evaluation (exact, same formula as evaluate_sequence for sum=1)
# ----------------------------------------------------------------------
def exact_score(a: np.ndarray, L_fft: int) -> float:
    """
    Compute the exact objective 2·n·max(b) for a sequence a of
    ℓ₁‑norm = 1 using an FFT‑based self‑convolution.
    """
    n = a.size
    Fa = np.fft.rfft(a, n=L_fft)
    b_full = np.fft.irfft(Fa * Fa, n=L_fft)[: 2 * n - 1]
    max_b = float(b_full.max())
    return 2.0 * n * max_b

# ----------------------------------------------------------------------

    finally returned.

    # ------------------------------------------------------------------
    global_vars = globals()

    # ------------------------------------------------------------------

    # Initial score
    cur_score = exact_score(a, L_fft)
    best_seq = a.copy()
    best_score = cur_score

    # Hyper‑parameters
    # ------------------------------------------------------------------

    end_lr = 0.5             # minimal strength (approaches zero)

                             # the deterministic step fails to improve

    no_improve_counter = 0

        # --------------------------------------------------------------
        cand_score = exact_score(cand, L_fft)

            cur_score = cand_score
            no_improve_counter = 0
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{iteration:>6}] new best (det) = {best_score:.6f}")
            continue          # move to next iteration

        # --------------------------------------------------------------
        # 4. If deterministic step failed, try a random perturbation
        # --------------------------------------------------------------
        no_improve_counter += 1
        if rng.random() < prob_random or no_improve_counter >= max_no_improve:
            # pick a random move type

                cur_score = cand2_score

                    print(f"[{iteration:>6}] new best (rand) = {best_score:.6f}")

        # Nothing to do; the loop will recompute with the same a.

        # Periodic progress report (every ~10 000 iters or every 30 s)
        if iteration % 10000 == 0:
            print(f"[{iteration:>6}] current_score={cur_score:.6f}, best_score={best_score:.6f}")

        # early exit if we already beat the target comfortably
        if best_score < 1.503:
            print(f"[early stop] target reached with score {best_score:.6f}")
            break

    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    # Return a plain Python list – the harness expects a list of floats.
    return [float(x) for x in best_seq]
