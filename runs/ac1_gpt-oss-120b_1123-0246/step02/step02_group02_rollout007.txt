Implementation of projection onto simplex:

def project_simplex(v, S):
    if S <= 0: return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        tau = 0
    else:
        rho = rho[-1]
        tau = cssv[rho] / (rho + 1)
    w = np.maximum(v - tau, 0)
    return w

Note: This yields sum(w) = S (subject to floating error). Use double precision.

def subgradient_optimize(a0, S, deadline, max_iters=20000):
    n = len(a0)
    fft_len = next_pow2(2*n - 1)
    a = a0.copy()
    best_local = a.copy()
    best_score = evaluate_sequence(a.tolist())
    for t in range(max_iters):
        now = time.monotonic()
        if now >= deadline: break
        # compute conv
        b = _self_convolution_fft(a, fft_len)  # length 2n-1
        max_b = b.max()
        # if new improvement:
        if max_b < best_score - 1e-12:
            best_score = max_b
            best_local = a.copy()
            print(f"[subgrad iter {t}] new best = {best_score:.6f}")
        # compute gradient max indices
        # tolerance: near max within 1e-12 relative maybe.
        K = np.where(b >= max_b - 1e-12)[0]
        # compute gradient
        grad = np.zeros_like(a)
        for k in K:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1)
            j = k - i
            # add contributions
            grad[i] += 2.0 * a[j]
        # step size schedule
        step = 0.5 / np.sqrt(t + 1)
        cand = a - step * grad
        cand = np.clip(cand, 0.0, 1000.0)
        sum_c = cand.sum()
        if sum_c > 0:
            cand *= (S / sum_c)
        # optionally small random perturb after some iterations
        if t % 500 == 0 and t > 0:
            # random multiplicative noise
            mult = np.exp(rng.normal(0.0, 0.02, size=n))
            cand = cand * mult
            cand = np.clip(cand, 0.0, 1000.0)
            sum_c = cand.sum()
            if sum_c > 0:
                cand *= (S / sum_c)
        a = cand
    # final evaluation of best
    best_score_exact = evaluate_sequence(best_local.tolist())
    return best_local, best_score_exact

"""
Sub‑gradient optimisation for the step‑function minimisation problem.

The algorithm works directly on the non‑smooth objective
    f(a) = max_k (a * a)_k
under the constraints a_i ≥ 0 and Σ a_i = √(2 n).
For a given index k where the maximum is attained the sub‑gradient is
    g_i = 2·a_{k‑i}   (out‑of‑range indices contribute 0).
A projected sub‑gradient step (with a simple ℓ₂‑normalisation to the
target sum) is performed, together with occasional random kicks to
escape plateaus.  Several diverse initial vectors (uniform, random,
U‑shaped, arcsine‑shaped, √‑difference‑shaped and the previous best)
are tried; the best feasible sequence found before the time budget
expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using real FFTs.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)
    return conv_full[: 2 * a.size - 1]


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the scaled simplex
    { w ≥ 0 , Σ w = s }.
    """
    if s <= 0:
        return np.zeros_like(v)

    # work on a copy to avoid side‑effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_arc_sine_shape(n: int) -> np.ndarray:
    """Arcsine‑type shape: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = np.clip(a, 0.0, 1000.0)
    return a


def _make_sqrt_diff_shape(n: int) -> np.ndarray:
    """
    Shape with large values near the edges:
    a_i = √(d) – √(d‑1) where d = distance to the nearest edge + 1.
    """
    i = np.arange(n)
    d = np.minimum(i, n - 1 - i) + 1.0
    a = np.sqrt(d) - np.sqrt(d - 1.0)
    a = np.clip(a, 0.0, 1000.0)
    return a


def _subgradient_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    max_iters: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for a single start vector.
    Returns the best vector seen and its exact evaluation score.
    """
    n = a0.size
    fft_len = _next_pow2(2 * n - 1)

    # initialise the working vector (clip, normalise)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0)
    if a.sum() > 0.0:
        a *= (target_sum / a.sum())

    best_seq = a.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))  # exact score

    # main loop
    it = 0
    while it < max_iters:
        now = time.monotonic()
        if now >= deadline:
            break

        # --- convolution and max index set ---------------------------------
        b = _self_convolution_fft(a, fft_len)          # length 2n‑1
        max_b = b.max()
        # Update best (the score equals max_b because Σa = √(2n))
        if max_b < best_score - 1e-12:
            best_score = max_b
            best_seq = a.copy()
            print(f"[subgrad] iter {it:5d} new best = {best_score:.6f}")

        # Identify all indices attaining the maximum (within tolerance)
        K = np.where(b >= max_b - 1e-12)[0]

        # --- sub‑gradient of the max (vectorised via np.add.at) -----------
        grad = np.zeros_like(a)
        for k in K:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=np.int64)
            j = k - i
            np.add.at(grad, i, 2.0 * a[j])

        # --- step size (diminishing) --------------------------------------
        step = 0.5 / np.sqrt(it + 1)   # 0.5 is a modest initial step

        # --- gradient step + feasibility projection ------------------------
        cand = a - step * grad
        cand = np.clip(cand, 0.0, 1000.0)
        # simple ℓ₂‑projection onto the scaled simplex (no sorting)
        s = cand.sum()
        if s > 0.0:
            cand *= (target_sum / s)

        # occasional random kick (helps escape flat plateaus)
        if it > 0 and it % 500 == 0:
            mult = np.exp(rng.normal(0.0, 0.02, size=n))
            cand = cand * mult
            cand = np.clip(cand, 0.0, 1000.0)
            s = cand.sum()
            if s > 0.0:
                cand *= (target_sum / s)

        a = cand
        it += 1

    # final exact evaluation of the best candidate from this run
    final_score = float(evaluate_sequence(best_seq.tolist()))
    return best_seq, final_score
