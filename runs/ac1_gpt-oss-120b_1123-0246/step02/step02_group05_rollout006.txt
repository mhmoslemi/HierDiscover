def project_onto_simplex(v, s=1.0):
    # v is numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: all zero? then uniform distribution
        w = np.full_like(v, s / n)
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def fast_score(seq, total_sum=None):
    if total_sum is None:
        total_sum = seq.sum()
    if total_sum < 0.01:
        return np.inf
    n = seq.shape[0]
    # compute convolution via FFT
    n_conv = 2*n - 1
    n_fft = 1 << ((n_conv).bit_length())
    fft_seq = np.fft.rfft(seq, n=n_fft)
    conv = np.fft.irfft(fft_seq * fft_seq, n=n_fft)[:n_conv]
    max_b = conv.max()
    return 2.0 * n * max_b / (total_sum * total_sum)

"""
Convolution‑bound minimizer.

The algorithm combines three ideas:

1.  A small deterministic grid‑search over simple parametric families
    (sin‑power and shifted‑reciprocal‑sqrt shapes) that already beat the
    arcsine baseline on many lengths.

2.  A projected gradient descent on a smooth approximation of the
    ``max(conv(a,a))`` term.  The gradient is obtained by a weighted
    correlation of the current sequence with the convolution, computed
    via FFTs for speed.  After each gradient step we project onto the
    probability simplex (fixed ℓ₁‑norm, non‑negative entries) to keep the
    search in the feasible domain.

3.  Occasional random perturbations to escape plateaus.

The routine runs until the allotted time (``budget_s``) expires and
returns the best sequence found.  All numbers are non‑negative, the sum
is fixed to 1.0, and values are far below the 1000‑limit imposed by the
security checks in ``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Utility: simple ℓ₁‑simplex projection (non‑negative, sum = s)
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries will be zero; return uniform distribution
        return np.full_like(v, s / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Normalisation helper (clips to [0,1000] and rescales to a target sum)
# ----------------------------------------------------------------------
def renormalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000] and scale so that sum(seq)=target_sum."""
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur == 0.0:
        return seq
    seq = seq * (target_sum / cur)
    return seq


# ----------------------------------------------------------------------
# Scoring (fast via FFT) – the objective is scale‑invariant,
# so we fix the ℓ₁‑norm to 1.0 (or any positive constant).
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray, target_sum: float = 1.0) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)²).  Uses FFT for speed."""
    total = seq.sum()
    if total < 0.01:
        return np.inf
    n = seq.shape[0]
    conv_len = 2 * n - 1
    # length for zero‑padding – next power of two ≥ conv_len
    fft_len = 1 << ((conv_len).bit_length())
    fft_seq = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fft_seq * fft_seq, n=fft_len)[:conv_len]
    max_b = conv.max()
    return 2.0 * n * max_b / (total * total)


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Exact arcsine (inverse‑square‑root) coefficients."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


def build_sin_power_sequence(length: int, p: float, delta: float) -> np.ndarray:
    """
    a_i = sin(θ_i + δ)^{-p},   θ_i = (i+0.5)·π/length.
    The shape is symmetric, heavy at the ends for p>0.
    """
    theta = (np.arange(length) + 0.5) * np.pi / length
    vals = np.sin(theta + delta) ** (-p)
    return vals


def build_offset_sqrt_sequence(length: int, alpha: float) -> np.ndarray:
    """
    a_i = 1 / sqrt((i+α)*(length‑1‑i+α)).
    For α=0 this coincides (up to a constant) with the continuous
    arcsine law; α>0 smooths the singularity at the ends.
    """
    i = np.arange(length, dtype=np.float64)
    denom = np.sqrt((i + alpha) * (length - 1 - i + alpha))
    # prevent division by zero – α>0 guarantees positivity
    vals = 1.0 / denom
    return vals


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    the evaluation function.  The routine spends the allotted time on
    a quick parametric sweep followed by a projected gradient descent.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256  # fallback for testing

    target_sum = 1.0  # fixing the ℓ₁‑norm; scale‑invariant objective

    # ------------------------------------------------------------------
    # 1.  Baseline: arcsine
    # ------------------------------------------------------------------
    best_seq = renormalise(build_arcsine_sequence(n), target_sum)
    best_score = fast_score(best_seq, target_sum)
    print(f"[init] length={n}, arcsine score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Small parametric grid‑search
    # ------------------------------------------------------------------
    # (p,delta) family – sin‑power
    p_vals = np.linspace(0.30, 0.80, 11)  # 0.30 … 0.80
    delta_vals = np.linspace(0.0, np.pi / (4 * n), 5)  # tiny shift up to π/(4n)

    for p in p_vals:
        for delta in delta_vals:
            seq = build_sin_power_sequence(n, p, delta)
            seq = renormalise(seq, target_sum)
            sc = fast_score(seq, target_sum)
            if sc < best_score:
                best_seq = seq.copy()
                best_score = sc
                print(f"[grid‑sin] p={p:.3f}, δ={delta:.5e} → {sc:.6f}")

    # (alpha) family – shifted reciprocal‑sqrt
    alpha_vals = [0.0, 0.2, 0.5, 1.0, 2.0, 5.0]
    for alpha in alpha_vals:
        seq = build_offset_sqrt_sequence(n, alpha)
        seq = renormalise(seq, target_sum)
        sc = fast_score(seq, target_sum)
        if sc < best_score:
            best_seq = seq.copy()
            best_score = sc
            print(f"[grid‑off] α={alpha:.3f} → {sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Projected gradient descent refinement
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    # FFT length sufficient for both convolution (2n‑1) and correlation (3n‑2)
    fft_len = 1 << ((3 * n - 2).bit_length())

    # hyper‑parameters
    step_factor = 0.02          # basic step size (as a fraction of target_sum)
    beta = 1.0                  # smoothing for the soft‑max over the convolution
    beta_max = 30.0
    it = 0
    no_improve = 0

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)  # small safety slack

    while time.monotonic() < deadline:
        it += 1

        # ---- current convolution (via FFT) ----
        fft_a = np.fft.rfft(cur_seq, n=fft_len)
        conv_full = np.fft.irfft(fft_a * fft_a, n=fft_len)[: (2 * n - 1)]
        max_c = conv_full.max()

        # ---- soft‑max weights over convolution entries ----
        # gradually focus on the true maximum
        beta = min(beta_max, beta * 1.015)
        w = np.exp(beta * (conv_full - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient = 2·corr(w, cur_seq) (valid part only) ----
        fft_w = np.fft.rfft(w, n=fft_len)
        # correlation = irfft(fft_w * conj(fft_a))
        corr_full = np.fft.irfft(fft_w * np.conj(fft_a), n=fft_len)[:n]  # first n entries = valid
        grad = 2.0 * corr_full

        # ---- try a gradient step (projected onto the simplex) ----
        step = step_factor * target_sum
        cand = cur_seq - step * grad
        cand = project_onto_simplex(cand, target_sum)
        cand_score = fast_score(cand, target_sum)

        if cand_score < cur_score:
            cur_seq, cur_score = cand, cand_score
            no_improve = 0
            if cand_score < best_score:
                best_seq, best_score = cand.copy(), cand_score
                print(f"[grad] it={it:>6} new best = {best_score:.6f}")
        else:
            # try a half‑size step as a cheap back‑track
            cand2 = cur_seq - 0.5 * step * grad
            cand2 = project_onto_simplex(cand2, target_sum)
            cand2_score = fast_score(cand2, target_sum)
            if cand2_score < cur_score:
                cur_seq, cur_score = cand2, cand2_score
                no_improve = 0
                if cand2_score < best_score:
                    best_seq, best_score = cand2.copy(), cand2_score
                    print(f"[grad‑half] it={it:>6} new best = {best_score:.6f}")
            else:
                no_improve += 1

        # ---- occasional random jitter to escape flat regions ----
        if rng.random() < 0.015:
            jitter = rng.normal(scale=0.02, size=n)
            cand3 = cur_seq + jitter
            cand3 = project_onto_simplex(cand3, target_sum)
            cand3_score = fast_score(cand3, target_sum)
            if cand3_score < cur_score:
                cur_seq, cur_score = cand3, cand3_score
                no_improve = 0
                if cand3_score < best_score:
                    best_seq, best_score = cand3.copy(), cand3_score
                    print(f"[jitter] it={it:>6} new best = {best_score:.6f}")

        # ---- adapt step size if we have been stuck for a while ----
        if no_improve > 2000:
            step_factor *= 0.5
            no_improve = 0
            if step_factor < 1e-5:
                step_factor = 1e-5

        # early exit if we already beat a reasonable target
        if best_score < 1.503:
            break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return as plain Python list for the harness.
    return [float(v) for v in best_seq]
