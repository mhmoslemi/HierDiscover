"""
Optimiser for the convolution‐norm objective.

The routine starts from the analytically optimal “arcsine’’ sequence,
then performs an extensive low–dimensional parameter sweep:
   – mixing the arcsine with its reverse,
   – adding a uniform background,
   – applying a power‐law deformation.
All combinations of the three simple families are evaluated (few
hundred candidates only – cheap compared with the convolution).

If the sweep does not already beat the target, a fast directed‐mass–
transfer local search is launched.  The move removes a fraction of the
mass from the indices that contribute most to the current maximal
convolution entry and deposits it on low–value positions, keeping the
non‑negativity and the sum unchanged (scaling is irrelevant for the
objective).  The search respects a time budget (default 800 s) and
stops as soon as a score below 1.503 is found.

The function `propose_candidate` returns the best list of heights found.
"""
import numpy as np
import time


def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the “arcsine’’ (inverse‑square‑root) non‑negative sequence."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0, σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    # a slightly larger sigma than the reference code (more exploration)
    factor = float(np.exp(rng.normal(0.0, 0.20)))
    cand[start:stop] *= factor
    # occasional symmetrisation – useful for the arcsine family
    if rng.random() < 0.30:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def directed_transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Move a moderate fraction of mass from the index that contributes most
    to the current maximal convolution coefficient to a low‑value index.
    """
    n = base.size
    conv = np.convolve(base, base)
    # locate the (first) index where the convolution is maximal
    max_idx = int(np.argmax(conv[:n]))

    # contributions a[i] * a[max_idx‑i] (zero where out of range)
    idx = np.arange(n)
    partner = max_idx - idx
    mask = (partner >= 0) & (partner < n)
    contrib = np.zeros_like(base)
    contrib[mask] = base[idx[mask]] * base[partner[mask]]

    # source: index with the largest contribution
    src = int(np.argmax(contrib))
    # target: one of the 20 % smallest values (far from the source)
    low_thr = np.percentile(base, 20)
    low_candidates = np.where(base <= low_thr)[0]
    if low_candidates.size == 0:
        # fallback – random other index
        tgt = int(rng.integers(0, n))
        if tgt == src:
            tgt = (tgt + 1) % n
    else:
        tgt = int(rng.choice(low_candidates))

    # fraction to move – a bit larger than the reference (up to 30 %)
    frac = rng.random() * 0.30 + 0.05          # 5 % … 35 %
    delta = frac * base[src]
    if delta <= 0.0:
        return base.copy()

    cand = base.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)


def propose_candidate(seed: int = 42, budget_s: float = 800, **kwargs):
    """
    Return a list of heights – the best step function found within the
    allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        # fall back to a moderate size – still large enough for a good bound
        n = 3000

    # --------------------------------------------------------------
    # 2. Baseline arcsine sequence
    # --------------------------------------------------------------
    base_seq = build_arcsine_sequence(n)
    best_seq = base_seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] length={n}, arcsine score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Helper families for the sweep
    # --------------------------------------------------------------
    rev_seq = base_seq[::-1]
    uniform_seq = np.full(n, np.mean(base_seq))  # same L1 norm as arcsine

    # --------------------------------------------------------------
    # 4. Exhaustive low‑dimensional sweep
    # --------------------------------------------------------------
    alphas = np.linspace(0.0, 0.5, 11)          # blend with reversed arcsine
    betas  = np.linspace(0.0, 0.5, 11)          # blend with uniform background
    powers = np.linspace(0.4, 1.0, 13)          # power‑law deformation (p≤1)

    for alpha in alphas:
        for beta in betas:
            if alpha + beta > 1.0:
                continue
            # linear combination of the three basic shapes
            mixed = (1.0 - alpha - beta) * base_seq + alpha * rev_seq + beta * uniform_seq
            for p in powers:
                cand = np.power(mixed, p)
                cand = np.clip(cand, 0.0, 1000.0)
                sc = float(evaluate_sequence(cand.tolist()))
                if sc < best_score:
                    best_seq, best_score = cand, sc
                    print(f"[sweep] α={alpha:.2f}, β={beta:.2f}, p={p:.2f} → score={sc:.6f}")

    # --------------------------------------------------------------
    # 5. If needed, a fast directed stochastic local search
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # keep a small slack
    iteration = 0
    cur_seq = best_seq.copy()
    cur_score = best_score

    while time.monotonic() < deadline:
        iteration += 1

        # 70 % directed transfer, 30 % generic block perturbation
        if rng.random() < 0.7:
            cand = directed_transfer_move(cur_seq, rng)
        else:
            cand = block_perturbation(cur_seq, rng)

        sc = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(sc):
            continue

        # accept only improving moves (simple hill climbing)
        if sc < cur_score:
            cur_seq, cur_score = cand, sc

        # keep the best ever seen
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"[{iteration:>6}] new best = {best_score:.6f}")

        # early exit when target is reached
        if best_score < 1.503:
            print("[stop] target achieved")
            break

        # optional occasional random restart to avoid stagnation
        if iteration % 2000 == 0 and rng.random() < 0.25:
            cur_seq = best_seq.copy()
            cur_score = best_score

    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best_seq]
```
