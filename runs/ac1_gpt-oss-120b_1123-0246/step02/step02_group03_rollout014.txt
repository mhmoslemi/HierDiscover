"""
Optimisation of a non‑negative step‑function for the max‑self‑convolution
objective (the same objective used by the reference implementation).

The algorithm combines three complementary ideas:

1. **Soft‑max smoothing** –  the raw objective  max_k (a*a)_k  is replaced by
   a differentiable soft‑max approximation controlled by a temperature β.
   As β grows the approximation becomes tighter.

2. **Adam‑style projected gradient descent** –  the gradient of the smoothed
   objective is obtained analytically (exactly as in the reference code)
   and updated with Adam momentum.  After each update the vector is projected
   onto the feasible set:
   * non‑negative,
   * ≤ 1000,
   * fixed ℓ¹‑norm  Σ a_i = √(2·n).

3. **Exploration heuristics** –  a multiplicative log‑normal perturbation is
   applied every few hundred iterations; if no improvement is seen for a
   while the search restarts from a fresh random Dirichlet‑like point.
   Both the temperature β and the step‑size are annealed linearly in time.

The routine runs for at most ``budget_s`` seconds (default 1000 s) and
returns the best sequence found.  It works with any starting vector that
is accessible as the global variable ``height_sequence_1``; otherwise a
uniform vector of moderate size (4000 entries) is used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level (no closures) as required
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Fast linear convolution using a power‑of‑two real FFT.
    The result has length len(x) + len(y) – 1.
    """
    n = x.size + y.size - 1
    N = _next_pow2(n)
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv


def _softmax_gradient(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the soft‑max approximation to max_k (a*a)_k.
    Returns (gradient, exact_max), where ``exact_max`` is the true maximal
    convolution value (required for the final score).
    """
    # exact convolution and maximal entry
    b = _fft_convolve(a, a)
    maxb = b.max()

    # soft‑max weights (stabilised by subtracting maxb)
    w = np.exp(beta * (b - maxb))
    w_sum = w.sum()
    if w_sum == 0.0:        # numerical safeguard
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum

    # gradient of Σ_k w_k (a*a)_k  = 2 Σ_k w_k a_{k‑i}
    w_rev = w[::-1]                     # reversed weights
    conv_rev = _fft_convolve(a, w_rev)  # length = n + (2n‑1) – 1 = 3n‑2
    idx = (2 * a.size - 2) - np.arange(a.size)
    grad = 2.0 * conv_rev[idx]

    # project onto the hyper‑plane Σ a_i = const (zero‑mean gradient)
    grad -= grad.mean()
    return grad, maxb


def _renormalise(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to the allowed box, renormalise to ``target_sum`` and clip again."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    return a

    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # keep the length reasonable – very long vectors slow down the FFT
        if base.size > 12000:
            lo = (base.size - 12000) // 2
            base = base[lo:lo + 12000]
    else:
        base = np.ones(4000, dtype=np.float64)

    n = base.size
    a = np.clip(base, 0.0, 1000.0)

    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n) → eval = max_b
    if a.sum() == 0.0:
        a[:] = 1.0
    a = _renormalise(a, target_sum)

    # ------------------------------------------------------------------
    # 2. Book‑keeping of the best solution seen so far
    # ------------------------------------------------------------------
    best_seq = a.copy()
    _, best_maxb = _softmax_gradient(a, beta=1.0)   # exact maximal convolution
    best_score = float(best_maxb)                  # under normalisation = eval
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # temperature (soft‑max β) schedule
    beta_start, beta_end = 0.5, 20.0
    # step‑size schedule (will be scaled by Adam’s adaptive term)
    lr_start = target_sum * 0.02
    lr_end = target_sum * 0.001

    iteration = 0
    last_improve = 0
    stagnation_limit = 4000   # iterations without improvement → restart

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac = min(1.0, elapsed / budget_s)

        # annealing of β and of the base learning rate
        beta = beta_start + (beta_end - beta_start) * frac
        base_lr = lr_start + (lr_end - lr_start) * frac

        # ---- gradient step (Adam) ------------------------------------
        grad, cur_maxb = _softmax_gradient(a, beta)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Adam update (we minimise → move opposite to the gradient)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _renormalise(a_new, target_sum)

        # evaluate the candidate (exact max of the convolution)
        cand_maxb = _fft_convolve(a_new, a_new).max()
        cand_score = float(cand_maxb)

        # accept improvement outright
        if cand_score < best_score:
            best_seq = a_new.copy()
            best_score = cand_score
            a = a_new
            last_improve = iteration
            print(f"[{iteration:6d}] new best  score = {best_score:.6f}")
        else:
            # Metropolis‑type acceptance to escape shallow plateaux
            T = max(1e-8, 1.0 - frac)          # artificial temperature
            if np.exp((best_score - cand_score) / T) > rng.random():
                a = a_new
            else:
                # keep the previous point if the proposal is rejected
                pass

        # ---- occasional multiplicative perturbation --------------------
        if iteration % 500 == 0 and iteration > 0:
            perturb = np.exp(rng.normal(0.0, 0.08, size=n))
            pert = a * perturb
            pert = _renormalise(pert, target_sum)
            pert_score = _fft_convolve(pert, pert).max()
            if pert_score < best_score:
                best_seq = pert.copy()
                best_score = float(pert_score)
                a = pert
                last_improve = iteration
                print(f"[{iteration:6d}] rand‑pert improved score = {best_score:.6f}")

        # ---- stagnation restart ----------------------------------------
        if iteration - last_improve > stagnation_limit:
            a = rng.random(n)                     # fresh Dirichlet‑like start
            a = _renormalise(a, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = iteration
            print(f"[{iteration:6d}] restart due to stagnation")

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Return the best sequence encountered
    # ------------------------------------------------------------------
    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    return [float(v) for v in best_seq]
