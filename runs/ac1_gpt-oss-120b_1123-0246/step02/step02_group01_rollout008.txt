    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.1)

    best_score = np.inf
    best_n = None
    best_alpha = None

    # initial guess: choose a relatively large n to approach limit.
    n_large = 131072  # 2^17
    # Optionally add other n smaller if time permits.
    n_candidates = [65536, n_large]  # increasing order; will try both.

    for n in n_candidates:
        if time.monotonic() >= deadline:
            break
        # First try symmetric arcsine (α=0.5)
        alpha0 = 0.5
        seq = _beta_sequence(n, alpha0)
        score = _fast_objective(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_n = n
            best_alpha = alpha0
            print(f"[init] n={n}, α={alpha0:.3f} → score={score:.6f}")

        # Now refine α around 0.5 using golden-section search.
        lo, hi = max(0.1, alpha0 - 0.4), min(1.5, alpha0 + 0.4)

        f_c = _fast_objective(seq_c)
        seq_d = _beta_sequence(n, d)

                d = c

"""
Propose a low‑score step‑function.

The evaluation function is
    2·n·max(conv(a,a)) / (sum(a)**2)
which is invariant to scaling of *a*.  Consequently we can fix the
normalisation sum(a)=sqrt(2·n) and minimise simply max(conv(a,a)).  The
continuous arcsine density (Beta(½,½)) has a very flat self‑convolution,
so a discretisation of this density gives scores close to the theoretical
limit 1.5.  The bound improves roughly like C/n, therefore we use a
large sequence (≈1.3·10⁵ terms) to push the score below the target
1.503.  The algorithm works as follows

1.  For a few large values of *n* generate a symmetric Beta(α,α) shape.
    α is refined by a cheap golden‑section search using a fast FFT‑based
    objective (the objective equals max(conv) because of the fixed sum).
2.  Keep the best candidate seen so far.
3.  Perform a tiny stochastic mass‑transfer hill‑climb: move a small
    amount of probability mass from a random index *i* to *j*,
    re‑normalise, and accept the change if the fast objective improves.
5.  Finally evaluate the best sequence with the official
    `evaluate_sequence` (which also applies clipping) and return it.

The whole procedure respects the provided time budget (default 500 s)
and never exceeds the security checks of `evaluate_sequence`.  The
result is a list of non‑negative floats ready for submission.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Fast evaluation of the objective (max‑self‑convolution) for a fixed sum.
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(a,a)) / (sum(a)**2) using an FFT.
    The input is assumed to be a 1‑D NumPy array of non‑negative numbers.
    This function is cheap (O(n log n)) and is used during the search.
    """
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    # convolution via FFT (zero‑pad to the next power of two)
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Generate a symmetric Beta(α,α) shaped sequence of length n.
# The sequence is scaled so that sum = sqrt(2·n) (the normalisation used
# in the definition of the evaluation function).
# ----------------------------------------------------------------------
def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Return a length‑n array whose entries are proportional to
        p^{α‑1} (1‑p)^{α‑1},   p = (i+0.5)/n .
    The array is normalised to have sum sqrt(2·n) and clipped to
    the hard bounds enforced by `evaluate_sequence`.
    """
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, alpha - 1.0)
    target_sum = sqrt(2.0 * n)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    a = w * (target_sum / w_sum)
    a = np.clip(a, 0.0, 1000.0)
    return a
