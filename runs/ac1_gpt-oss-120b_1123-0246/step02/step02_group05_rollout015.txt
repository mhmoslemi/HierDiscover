"""
Hybrid simulated‑annealing + stochastic local search for the
self‑convolution upper‑bound problem.

The algorithm:
  • Initialise with the analytically optimal “arcsine” sequence and
    normalise its ℓ₁‑norm to 1 (the objective is scale‑invariant).
  • Repeatedly generate a candidate by one of several cheap stochastic
    moves (block scaling, mass transfer, early‑to‑late transfer, a simple
    gradient move and a random mixing move).  The move magnitude is
    intentionally large to be able to escape the flat plateau of the
    arcsine shape.
  • Accept the candidate either if it improves the exact objective or
    probabilistically according to a temperature schedule (simulated
    annealing).  The temperature slowly decays to zero, turning the
    search into a hill‑climber after an exploration phase.
  • Keep the best sequence seen (according to the exact objective) and
    return it when the time budget expires.

All convolutions are performed with FFTs (O(n log n)) so thousands of
iterations easily fit into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast real‑valued convolution using FFT (zero‑padding to the next power
# of two).  Returns a NumPy array of length len(x)+len(y)-1.
# ----------------------------------------------------------------------
def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = len(x) + len(y) - 1
    L = 1 << ((n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    Z = X * Y
    conv = np.fft.irfft(Z, L)[:n]
    return conv


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence but without safety
# checks).  The ℓ₁‑norm is kept exactly 1, therefore the denominator
# disappears and the objective reduces to 2·n·max(b).
# ----------------------------------------------------------------------
def exact_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a sequence with sum(seq)=1."""
    n = seq.shape[0]
    b = fft_convolve(seq, seq)
    max_b = float(b.max())
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
# Simple arcsine (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Normalise a sequence to a given ℓ₁‑sum (clipping to the allowed range
# before scaling, as required by evaluate_sequence).
# ----------------------------------------------------------------------
def renormalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur == 0.0:
        return seq
    seq *= target_sum / cur
    return seq


# ----------------------------------------------------------------------
# Stochastic perturbation operators (all keep the total sum fixed via
# renormalisation).  The fractions have been enlarged compared to the
# reference code to make the search more aggressive.
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    # larger log‑normal factor (σ≈0.20) for broader exploration
    factor = float(np.exp(rng.normal(0.0, 0.20)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:               # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return renormalise(cand, target_sum)


def transfer_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    frac = rng.random() * 0.70              # up to 70 % transfer
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


def early_to_late_move(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    n = base.size
    early_cut = max(1, n // 10)            # first 10 %
    late_start = max(early_cut, n - early_cut)   # last 10 %
    i = int(rng.integers(0, early_cut))
    j = int(rng.integers(late_start, n))
    if i == j:
        return base.copy()
    frac = rng.random() * 0.80            # up to 80 % transfer
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


def gradient_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """Shift a fraction of the coefficient that contributes most to the
    current maximal convolution entries toward a light index."""
    n = base.size
    b = fft_convolve(base, base)
    max_b = b.max()
    tol = max_b * (1.0 - 1e-12)
    mask = (b >= tol).astype(np.float64)
    contrib_conv = fft_convolve(base, mask)
    contrib = base * contrib_conv[:n]
    heavy = int(np.argmax(contrib))
    # choose a light index with small contribution
    light_candidates = np.where(contrib < contrib.mean())[0]
    if light_candidates.size == 0:
        light = (heavy + n // 2) % n
    else:
        light_candidates = light_candidates[light_candidates != heavy]
        if light_candidates.size == 0:
            light = (heavy + 1) % n
        else:
            light = int(rng.choice(light_candidates))
    frac = rng.random() * 0.40
    delta = frac * base[heavy]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[heavy] -= delta
    cand[light] += delta
    return renormalise(cand, target_sum)


def mix_with_random(base: np.ndarray, rng: np.random.Generator,
                    target_sum: float) -> np.ndarray:
    """Blend the current sequence with a fresh random Dirichlet vector."""
    n = base.size
    t = rng.random() * 0.15                 # up to 15 % mixing
    rand = rng.random(n)
    rand /= rand.sum()
    cand = (1.0 - t) * base + t * rand
    return renormalise(cand, target_sum)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of floats – the best step‑function heights found within
    ``budget_s`` seconds.  The search starts from the arcsine shape and
    performs a simulated‑annealing hill‑climber that combines several
    stochastic moves.  The best sequence (according to the exact
    evaluate_sequence objective) is returned.
    """
    rng = np.random.default_rng(seed)

    # Length of the problem – use the provided reference length if
    # available; otherwise fall back to a moderate size.
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 2048

    # ------------------------------------------------------------------
    # 1.  Starting point – the analytically derived arcsine shape.
    # ------------------------------------------------------------------
    base_seq = build_arcsine_sequence(n)
    base_seq = renormalise(base_seq, 1.0)      # enforce ℓ₁‑norm = 1
    cur_seq = base_seq.copy()
    cur_score = exact_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Search parameters
    # ------------------------------------------------------------------
    # Simulated‑annealing schedule (temperature decays linearly)
    T_start = 0.10
    T_end = 1e-6
    # Smoothing parameter for the “soft‑max” weight (increments slowly)
    C_start = 5.0
    C_end = 45.0
    # Probabilities for the different move types
    move_probs = np.array([0.30, 0.25, 0.20, 0.15, 0.10])  # sum to 1
    move_funcs = [
        block_perturbation,
        transfer_move,
        early_to_late_move,
        gradient_move,
        mix_with_random,
    ]

    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)  # small slack
    it = 0
    last_improve_it = 0

    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - (deadline - budget_s)
        frac = min(1.0, elapsed / budget_s)

        # temperature and smoothing evolve with time
        temperature = T_start * (1.0 - frac) + T_end * frac
        C = C_start * (1.0 - frac) + C_end * frac

        # ---- pick a move -------------------------------------------------
        r = rng.random()
        idx = np.searchsorted(np.cumsum(move_probs), r)
        move = move_funcs[idx]

        cand = move(cur_seq, rng, 1.0)

        # ---- evaluate candidate -------------------------------------------
        cand_score = exact_score(cand)

        # ---- accept / reject (simulated annealing) -----------------------
        delta = cand_score - cur_score
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(temperature, 1e-12)):
            cur_seq = cand
            cur_score = cand_score

        # ---- keep the global best -----------------------------------------
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            last_improve_it = it
            print(f"[{it:>7}] new best = {best_score:.6f}")

            # early exit if we have already beaten a generous target
            if best_score < 1.503:
                break

        # occasional full re‑normalisation to avoid drift due to rounding
        if it % 5000 == 0:
            cur_seq = renormalise(cur_seq, 1.0)

        # if no improvement for a long time, inject a random perturbation
        if it - last_improve_it > 20000:
            # re‑mix with a fresh random vector to escape stagnation
            cur_seq = mix_with_random(cur_seq, rng, 1.0)
            cur_score = exact_score(cur_seq)
            last_improve_it = it

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return a plain Python list – the harness expects a list of floats.
    return [float(x) for x in best_seq]
