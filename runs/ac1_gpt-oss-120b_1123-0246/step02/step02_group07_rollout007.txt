def directed_transfer_move(seq, rng):
    n = seq.size
    conv = np.convolve(seq, seq)
    # find index of max conv (first occurrence)
    k0 = int(conv.argmax())
    # compute contributions for that k0
    i_start = max(0, k0 - (n - 1))
    i_end = min(k0, n - 1)
    i_range = np.arange(i_start, i_end+1)
    j_range = k0 - i_range
    contrib = seq[i_range] * seq[j_range]
    src_idx = i_range[np.argmax(contrib)]
    # choose target index from low percentiles
    low_thr = np.percentile(seq, 20)
    low_candidates = np.where(seq <= low_thr)[0]
    if low_candidates.size == 0:
        tgt = int(rng.integers(0, n))
        # ensure different
        while tgt == src_idx:
            tgt = int(rng.integers(0, n))
    else:
        tgt = int(rng.choice(low_candidates))
        if tgt == src_idx:
            # fallback
            tgt = (src_idx + n//2) % n
    # transfer fraction
    frac = rng.random() * 0.2 + 0.01
    delta = frac * seq[src_idx]
    if delta <= 0:
        return seq.copy()
    cand = seq.copy()
    cand[src_idx] -= delta
    cand[tgt] += delta
    # ensure nonnegative
    cand = np.clip(cand, 0.0, 1000.0)
    # renorm? sum unchanged anyway.
    return cand
```

"""
Convolution–bound optimiser.

The optimiser works in three stages:
1.  Build the classic arcsine (inverse‑square‑root) distribution and
    normalise it to unit ℓ¹‑norm (the value of the score is invariant
    under scaling, so keeping sum = 1 is convenient).
2.  Exhaustively explore a low‑dimensional parametric family:
    – apply a power transform a↦a**p,
    – symmetrise it by mixing with its reversal,
    – optionally blend with a tiny uniform component.
    This cheap sweep already beats the naïve arcsine bound for many
    lengths.
3.  Perform a long stochastic local search (simulated‑annealing style)
    using several complementary moves:
    – directed mass transfer from the index that contributes most to the
      maximal convolution coefficient,
    – random block scaling,
    – smoothing by a moving‑average filter,
    – random “reset’’ of a few entries,
    – a power‑law perturbation,
    – a random convex combination of the original arcsine, its reversal
      and the uniform vector,
    – a linear tilt of the whole profile.
    The temperature decays linearly to zero, allowing occasional
    uphill moves early on and strict hill‑climbing later.
The routine respects the time budget (default 500 s, the harness provides
up to 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1.  Construction of the analytical “arcsine’’ sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the discrete inverse‑square‑root (arcsine) sequence of
    the given length.  The values already lie in [0, 1] and sum to ≈1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    return a

# ----------------------------------------------------------------------
# 2.  Move primitives (all accept a 1‑D np.ndarray and a rng, return a copy)
# ----------------------------------------------------------------------
def directed_transfer_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small fraction of mass from the index that contributes most
    to the current maximal convolution entry towards a low‑mass index."""
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())                # index of the current maximum
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i_range = np.arange(i_start, i_end + 1)
    j_range = k - i_range
    contrib = seq[i_range] * seq[j_range]
    src = int(i_range[np.argmax(contrib)])    # most harmful source index

    # target – pick from the 20 % smallest entries
    low_thr = np.percentile(seq, 20)
    low_cands = np.where(seq <= low_thr)[0]
    if low_cands.size == 0:
        tgt = int(rng.integers(0, n))
        while tgt == src:
            tgt = int(rng.integers(0, n))
    else:
        tgt = int(rng.choice(low_cands))
        if tgt == src:                     # extremely unlikely but guard
            tgt = (src + n // 2) % n

    # transfer a random fraction (1 % … 21 %)
    frac = rng.random() * 0.20 + 0.01
    delta = frac * seq[src]
    if delta <= 0.0:
        return seq.copy()

    cand = seq.copy()
    cand[src] -= delta
    cand[tgt] += delta
    # sum is preserved; just clip for safety
    return np.clip(cand, 0.0, 1000.0)


def block_perturbation(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    n = seq.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = seq.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def smoothing_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Replace the sequence by a convex combination with its moving‑average."""
    n = seq.size
    max_w = max(3, n // 10)
    w = int(rng.integers(3, max_w + 1))
    kernel = np.ones(w) / w
    smoothed = np.convolve(seq, kernel, mode='same')
    lam = rng.random() * 0.25          # mixing weight
    cand = (1.0 - lam) * seq + lam * smoothed
    cand = cand / cand.sum()           # renormalise (preserves scale invariance)
    return np.clip(cand, 0.0, 1000.0)


def random_reset_move(seq: np.ndarray, rng: np.random.Generator,
                      n_reset: int = 5) -> np.ndarray:
    """Re‑draw a few coordinates from a Dirichlet distribution while keeping the
    total mass unchanged."""
    n = seq.size
    idx = rng.choice(n, size=n_reset, replace=False)
    total = seq[idx].sum()
    if total <= 0.0:
        return seq.copy()
    new_weights = rng.random(n_reset)
    new_weights /= new_weights.sum()
    cand = seq.copy()
    cand[idx] = new_weights * total
    return np.clip(cand, 0.0, 1000.0)


def power_move(seq: np.ndarray, rng: np.random.Generator,
               p_min: float = 0.55, p_max: float = 1.15) -> np.ndarray:
    """Raise all entries to a random power p∈[p_min,p_max] and renormalise."""
    p = rng.random() * (p_max - p_min) + p_min
    cand = np.power(seq, p)
    if cand.sum() == 0.0:
        return seq.copy()
    cand = cand / cand.sum()
    return np.clip(cand, 0.0, 1000.0)


def tilt_move(seq: np.ndarray, rng: np.random.Generator,
              max_tilt: float = 0.25) -> np.ndarray:
    """Apply a small linear tilt across the index range."""
    n = seq.size
    tilt = (rng.random() * 2 - 1) * max_tilt   # in [‑max_tilt, max_tilt]
    idx = np.arange(n) / (n - 1)               # in [0, 1]
    factor = 1.0 + tilt * (2 * idx - 1)        # symmetric linear factor
    cand = seq * factor
    cand = np.clip(cand, 0.0, 1000.0)
    cand = cand / cand.sum()
    return cand


def mix_with_base_move(seq: np.ndarray, rng: np.random.Generator,
                       base: np.ndarray, rev: np.ndarray,
                       uniform: np.ndarray) -> np.ndarray:
    """Create a fresh convex combination of the original arcsine,
    its reversal and the uniform vector (random coefficients)."""
    a = rng.random()
    b = rng.random() * (1.0 - a)
    c = 1.0 - a - b
    cand = c * base + a * rev + b * uniform
    cand = cand / cand.sum()
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# 3.  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of floats – the best step‑function heights found
    within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the problem size (length of the known sequence)
    # --------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256                     # safety fallback

    # ------------------------------------------------------------------
    # Stage 1 – analytical starting point (unit ℓ¹‑norm)
    # ------------------------------------------------------------------
    raw = build_arcsine_sequence(n)
    base = raw / raw.sum()                 # sum == 1
    rev = base[::-1]
    uniform = np.full(n, 1.0 / n)

    best = base.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, arcsine score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – systematic low‑dimensional sweep
    # ------------------------------------------------------------------
    alphas = np.linspace(0.0, 1.0, 41)      # mixing weight for the reversal
    powers = np.linspace(0.55, 1.15, 13)    # power‑law exponents
    uniforms = np.linspace(0.0, 0.25, 6)    # amount of uniform blending

    for p in powers:
        pow_base = np.power(base, p)
        if pow_base.sum() == 0.0:
            continue
        pow_base /= pow_base.sum()
        pow_rev = pow_base[::-1]

        for alpha in alphas:
            mixed = (1.0 - alpha) * pow_base + alpha * pow_rev
            mixed /= mixed.sum()

            for u in uniforms:
                cand = (1.0 - u) * mixed + u * uniform
                cand /= cand.sum()
                sc = float(evaluate_sequence(cand.tolist()))
                if sc < best_score:
                    best, best_score = cand, sc
                    print(f"[sweep] p={p:.3f}, α={alpha:.2f}, u={u:.2f} → {sc:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – directed stochastic local search with simulated annealing
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    current = best.copy()
    current_score = best_score
    it = 0
    T0 = 0.008                                # initial temperature

    while time.monotonic() < deadline:
        it += 1
        # linearly decaying temperature
        t_elapsed = it * 0.0  # dummy placeholder (temperature based on time)
        T = T0 * (1.0 - (time.monotonic() - (deadline - budget_s)) / budget_s)
        T = max(T, 1e-8)

        # choose a move type
        r = rng.random()
        if r < 0.22:
            cand = directed_transfer_move(current, rng)
        elif r < 0.42:
            cand = block_perturbation(current, rng)
        elif r < 0.60:
            cand = smoothing_move(current, rng)
        elif r < 0.70:
            cand = random_reset_move(current, rng, n_reset=5)
        elif r < 0.80:
            cand = power_move(current, rng, p_min=0.55, p_max=1.15)
        elif r < 0.90:
            cand = tilt_move(current, rng, max_tilt=0.22)
        else:
            cand = mix_with_base_move(
                current, rng, base=base, rev=rev, uniform=uniform
            )

        # normalise to unit sum (keeping the scale invariance of the score)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand = cand / cand_sum
        cand = np.clip(cand, 0.0, 1000.0)

        sc = float(evaluate_sequence(cand.tolist()))
        if not np.isfinite(sc):
            continue

        # simulated‑annealing acceptance
        delta = sc - current_score
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            current, current_score = cand, sc
            if sc < best_score:
                best, best_score = cand, sc
                print(f"[{it:>6}] new best = {best_score:.6f}")

                if best_score < 1.503:
                    print("[stop] target reached")
                    break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    return [float(x) for x in best]
