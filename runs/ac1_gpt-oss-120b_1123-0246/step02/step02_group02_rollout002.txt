"""
Sub‑gradient descent on the true ∞‑norm of the linear self‑convolution.

The algorithm works with a fixed L¹‑mass  Σaᵢ = √(2 n) (the normalisation used in
the reference code).  For any sequence a the evaluation function reduces to
the maximal entry of the linear convolution a∗a, because
2 n·max(a∗a)/Σa² = max(a∗a) under this normalisation.
Hence we directly minimise max(a∗a).

For a given a we compute the full convolution by FFT, locate the index k
where it attains its maximum, and calculate the exact sub‑gradient of that
max‑entry:
    ∂/∂aⱼ (cₖ) = a_{k‑j}  (if 0≤k‑j<n) + a_{j‑k} (if 0≤j‑k<n).
Moving a a small step opposite to this gradient reduces the worst convolution
value.  After each step we project onto the feasible set:
    – clip to [0,1000],
    – renormalise the L¹‑mass to √(2 n).

Several diverse initialisations (uniform, sorted random, exponential tails,
and the previously discovered best sequence interpolated to the current length)
are explored.  A simple linear step‑size schedule, occasional random
perturbations and a tiny acceptance probability for uphill moves provide
exploration.  The best sequence found is finally evaluated with the exact
`evaluate_sequence` routine.

The routine respects the supplied time‑budget (default 900 s, leaving a 2 s
margin for finalisation) and returns the best sequence as a plain list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a∗a using real FFTs.
    Returns the full array of length (2*len(a)-1).
    """
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)
    return conv_full[: 2 * a.size - 1]


def _max_convolution(a: np.ndarray, fft_len: int, n: int) -> tuple[np.ndarray, float]:
    """
    Compute the full convolution a∗a and return (conv_array, max_value).
    """
    conv = _self_convolution_fft(a, fft_len)
    return conv, float(conv.max())


def _gradient_of_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of the maximal convolution entry c_{k_max} w.r.t a.
    For each j:
        g_j = a_{k_max‑j} (if that index exists) + a_{j‑k_max} (if exists).
    """
    n = a.size
    g = np.zeros_like(a)

    # part coming from a_{k_max‑j}
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    if lo <= hi:
        idx = np.arange(lo, hi + 1, dtype=int)
        g[idx] += a[k_max - idx]

    # part coming from a_{j‑k_max}
    if k_max <= n - 1:
        idx2 = np.arange(k_max, n, dtype=int)
        g[idx2] += a[idx2 - k_max]

    return g


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 900) -> list[float]:
    """
    Search for a low‑score step function within the allotted time.
    Returns the best sequence found (list of floats).
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a small safety margin for the final printing
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # 1. Choose a working length (n) and the normalisation constant
    # --------------------------------------------------------------
    try:
        n0 = int(np.asarray(height_sequence_1).size)   # type: ignore[name-defined]
    except Exception:
        n0 = 6000

    # use a length a little larger than the previous one if time permits
    n = int(np.clip(max(n0, 4000), 2000, 9000))

    target_sum = math.sqrt(2.0 * n)          # Σa = √(2 n)  →  objective = max(conv)

    fft_len = _next_pow2(2 * n - 1)          # enough for linear self‑convolution

    # --------------------------------------------------------------
    # 2. Build a collection of diverse starting points
    # --------------------------------------------------------------
    starts: list[np.ndarray] = []

    # (a) uniform vector
    starts.append(np.full(n, target_sum / n, dtype=float))

    # (b) pure random positive vector, then renormalise
    starts.append(rng.random(n, dtype=float) * target_sum)

    # (c) sorted random (decreasing) – encourages low early values
    dec = np.sort(rng.random(n, dtype=float))[::-1]
    dec = dec / dec.sum() * target_sum
    starts.append(dec)

    # (d) exponential shapes with random bases
    for base in rng.uniform(0.80, 0.98, size=3):
        exp_seq = (base ** np.arange(n, dtype=float))
        exp_seq = exp_seq / exp_seq.sum() * target_sum
        starts.append(exp_seq)

    # (e) interpolation of the globally best known sequence, if any
    try:
        prev = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            old_x = np.linspace(0.0, 1.0, prev.size)
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, prev)
            interp = np.clip(interp, 0.0, 1000.0)
            if interp.sum() > 0.0:
                interp = interp * (target_sum / interp.sum())
                starts.append(interp)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Core optimisation loop over the different starts
    # --------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_score: float = np.inf          # this stores the maximal convolution value
    total_iters = 0

    for start_id, a0 in enumerate(starts):
        if time.monotonic() >= deadline:
            break

        a = np.clip(a0, 0.0, 1000.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)

        # exact evaluation of the starting point
        init_score = float(evaluate_sequence(a.tolist()))
        if init_score < best_score:
            best_score = init_score
            best_seq = a.copy()
            print(f"[start {start_id}] initial exact score = {best_score:.6f}")

        # parameters for this run
        max_iters = 4000
        # initial step size – a few times the average entry
        eta0 = 3.0 * (target_sum / n)
        # linear decay of eta
        for it in range(max_iters):
            # time guard
            if time.monotonic() >= deadline:
                break

            # ----------------------------------------------------------
            # (a) compute full convolution and locate the worst index
            # ----------------------------------------------------------
            conv, cur_max = _max_convolution(a, fft_len, n)
            k_max = int(np.argmax(conv))

            # ----------------------------------------------------------
            # (b) sub‑gradient of the active max entry
            # ----------------------------------------------------------
            g = _gradient_of_max(a, k_max)

            # ----------------------------------------------------------
            # (c) step opposite to the gradient, then project
            # ----------------------------------------------------------
            eta = eta0 * max(0.0, 1.0 - it / max_iters)
            if eta == 0.0:
                break

            a_new = a - eta * g
            a_new = np.clip(a_new, 0.0, 1000.0)

            # renormalise the L1‑mass exactly
            s_new = a_new.sum()
            if s_new > 0.0:
                a_new = a_new * (target_sum / s_new)

            # ----------------------------------------------------------
            # (d) evaluate the candidate (fast max‑based)
            # ----------------------------------------------------------
            _, cand_max = _max_convolution(a_new, fft_len, n)

            # monotone acceptance – keep the move if it does not increase the max
            if cand_max <= cur_max * (1.0 + 1e-12):
                a = a_new
                cur_max = cand_max

                # update the global best using the cheap bound
                if cand_max < best_score:
                    best_score = cand_max
                    best_seq = a.copy()
                    # occasional exact check to avoid drift caused by clipping
                    if total_iters % 100 == 0:
                        exact = float(evaluate_sequence(a.tolist()))
                        if exact < best_score:
                            best_score = exact
                    print(f"[{total_iters:6d}] new best (max) = {best_score:.6f}")

            else:
                # tiny probability to accept a worsening move (helps escape plateaus)
                if rng.random() < 0.001:
                    a = a_new
                    cur_max = cand_max

            total_iters += 1

            # ----------------------------------------------------------
            # (e) diversification – random multiplicative perturbation
            # ----------------------------------------------------------
            if it % 500 == 0 and it > 0:
                pert = np.exp(rng.normal(0.0, 0.05, size=n))
                a_pert = a * pert
                a_pert = np.clip(a_pert, 0.0, 1000.0)
                s_pert = a_pert.sum()
                if s_pert > 0.0:
                    a_pert = a_pert * (target_sum / s_pert)
                _, max_pert = _max_convolution(a_pert, fft_len, n)
                if max_pert < cur_max:
                    a = a_pert
                    cur_max = max_pert
                    if max_pert < best_score:
                        best_score = max_pert
                        best_seq = a.copy()
                        print(f"[{total_iters:6d}] perturbed improvement = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Final refinement on the current best (tiny step size)
    # --------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        # a few extra passes with a much smaller eta
        fine_eta0 = (target_sum / n) * 0.5
        fine_iters = 2000
        for it in range(fine_iters):
            if time.monotonic() >= deadline:
                break
            conv, cur_max = _max_convolution(a, fft_len, n)
            k_max = int(np.argmax(conv))
            g = _gradient_of_max(a, k_max)
            eta = fine_eta0 * max(0.0, 1.0 - it / fine_iters)
            a_new = a - eta * g
            a_new = np.clip(a_new, 0.0, 1000.0)
            s_new = a_new.sum()
            if s_new > 0.0:
                a_new = a_new * (target_sum / s_new)
            _, cand_max = _max_convolution(a_new, fft_len, n)
            if cand_max <= cur_max:
                a = a_new
                cur_max = cand_max
                if cand_max < best_score:
                    best_score = cand_max
                    best_seq = a.copy()
        # replace best_seq with the refined version
        best_seq = a

    # --------------------------------------------------------------
    # 5. Final exact evaluation and return
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (should never happen)
        best_seq = np.full(n, target_sum / n, dtype=float)

    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] exact score = {final_score:.6f}")
    return [float(v) for v in best_seq]
