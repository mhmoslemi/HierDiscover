"""
Soft‑max projected gradient descent (full convolution) for the max‑autocorrelation
step‑function problem.

The original implementation used only the first half of the linear convolution,
which does not match the evaluation function (which uses the full convolution).
This version computes the full convolution both for the objective
`max_k (a*a)[k]` and for the soft‑max gradient, and projects the update onto
the simplex {a_i ≥ 0, Σ a_i = √(2 n)}.  A linear temperature schedule is used
for the soft‑max temperature β and a linearly decaying step‑size.
Occasional multiplicative perturbations and stagnation restarts help escape
plateaus.  The routine respects the 1000 s time budget, prints progress and
returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures)
# ----------------------------------------------------------------------


def _full_fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a power‑of‑two FFT.
    Returns the full result of length len(x)+len(y)-1.
    """
    n = x.size + y.size - 1
    N = 1 << (n - 1).bit_length()           # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv


def _grad_and_maxb_full(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Compute the gradient of the soft‑max approximation to
    max_k (a*a)[k] using the *full* convolution (length 2n‑1), and also
    return the exact maximal autocorrelation value (the true score).
    The gradient is projected onto the hyper‑plane Σ a_i = const
    (zero‑sum gradient) before being returned.
    """
    # exact autocorrelation (full convolution)
    b = _full_fft_convolve(a, a)          # length 2n‑1
    maxb = b.max()

    # soft‑max weights (stable: subtract max)
    exp_term = np.exp(beta * (b - maxb))
    w = exp_term / exp_term.sum()         # length 2n‑1, sums to 1

    # gradient of Σ_k w_k (a*a)[k]  =  2 Σ_k w_k a_{k‑i}
    # This is a convolution of w with the reversed a.
    a_rev = a[::-1]
    conv = _full_fft_convolve(w, a_rev)   # length 3n‑2
    n = a.size
    # extract the slice corresponding to i = 0 … n‑1
    grad = 2.0 * conv[n - 1 : n - 1 + n]

    # project out the component along the all‑ones vector (preserve total mass)
    grad -= grad.mean()
    return grad, float(maxb)


def _normalise(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1000], then renormalise to the exact target sum."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    a = np.clip(a, 0.0, 1000.0)
    # second renormalisation in case clipping changed the total mass
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    return a


# ----------------------------------------------------------------------
# Main search routine – this is the function that will be called.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a low‑score height sequence within the allotted time.
    Returns a list of floats (the best sequence found).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (default 1000 s).
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a tiny safety margin for the caller
    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------
    # 1. Choose a starting length and initialise the sequence
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # If the provided construction is huge, truncate to a reasonable size.
        if base.size > 12000:
            lo = (base.size - 12000) // 2
            base = base[lo:lo + 12000]
    else:
        # fallback: a modest uniform vector
        base = np.ones(4000, dtype=np.float64)

    n = base.size
    target_sum = np.sqrt(2.0 * n)

    a = _normalise(base, target_sum)

    # ------------------------------------------------------------------
    # 2. Bookkeeping
    # ------------------------------------------------------------------
    # initial exact score
    _, init_maxb = _grad_and_maxb_full(a, beta=1.0)
    best_seq = a.copy()
    best_score = init_maxb
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # algorithmic hyper‑parameters
    beta_start = 0.5
    beta_end = 20.0                     # final temperature (sharper soft‑max)
    init_lr = target_sum * 0.025        # base step size (≈2.5 % of the total mass)
    min_lr = init_lr * 0.01
    iteration = 0
    last_improve = 0

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / budget_s

        # temperature schedule (soft‑max “β”)
        beta = beta_start + (beta_end - beta_start) * frac

        # gradient w.r.t. the smooth approximation of max
        grad, _ = _grad_and_maxb_full(a, beta)

        # unit‑norm direction (already zero‑sum)
        norm = np.linalg.norm(grad) + 1e-12
        direction = -grad / norm

        # linearly decaying step size
        lr = max(min_lr, init_lr * (1.0 - frac))

        # propose a new candidate
        cand = a + lr * direction
        cand = _normalise(cand, target_sum)

        # evaluate the candidate (true objective = max of the full convolution)
        b_cand = _full_fft_convolve(cand, cand)
        cand_score = float(b_cand.max())

        # keep the best ever seen
        if cand_score + 1e-12 < best_score:
            best_seq = cand.copy()
            best_score = cand_score
            a = cand                    # move to the improved point
            last_improve = iteration
            print(f"[{iteration:6d}] new best  score = {best_score:.6f}")
        else:
            # even if not improving, continue the walk
            a = cand

        # ------------------------------------------------------------------
        # 4. Random multiplicative perturbations (every few hundred steps)
        # ------------------------------------------------------------------
        if iteration % 400 == 0 and iteration > 0:
            perturb = np.exp(rng.normal(0.0, 0.07, size=n))
            pert = a * perturb
            pert = _normalise(pert, target_sum)
            pert_score = float(_full_fft_convolve(pert, pert).max())
            if pert_score + 1e-12 < best_score:
                best_seq = pert.copy()
                best_score = pert_score
                a = pert
                last_improve = iteration
                print(f"[{iteration:6d}] perturb improved score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 5. Stagnation restart (if no improvement for a while)
        # ------------------------------------------------------------------
        if iteration - last_improve > 5000:
            a = rng.random(n)
            a = _normalise(a, target_sum)
            last_improve = iteration
            print(f"[{iteration:6d}] restart due to stagnation (score = {best_score:.6f})")

        iteration += 1

    # ------------------------------------------------------------------
    # 6. Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    return [float(v) for v in best_seq]
