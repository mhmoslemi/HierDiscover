```
def conv_full_fft(x, y):
    m = x.size + y.size - 1
    n = 1 << ((m-1).bit_length())
    fx = np.fft.rfft(x, n)
    fy = np.fft.rfft(y, n)
    fprod = fx * fy
    conv = np.fft.irfft(fprod, n)
    return conv[:m]
```

"""
Simulated‑annealing optimizer with a smooth‑max gradient.

The algorithm works on a normalized non‑negative vector `a` (∑a = 1).
For a given `a` the convolution `c = a*a` (full length 2n‑1) satisfies
∑c = 1.  The score returned by ``evaluate_sequence`` is
    2 n · max(c)      (because the sum is already 1).

We minimise `max(c)` by repeatedly
  1. evaluating `c` (FFT‑based O(n log n) convolution),
  2. turning the max into a smooth approximation:
         w_k ∝ exp(c_k / τ)   (soft‑max, τ is a temperature)
     which yields the gradient  g = 2 · Σ_k w_k·shift_k(a).
  3. applying a projected gradient step
         a ← a – η·(g – mean(g))
     (the subtraction of the mean keeps the total mass unchanged).
  4. occasional random block perturbations to escape local minima.
Both the step size η and the soft‑max temperature τ are annealed
exponentially, and a Metropolis rule with a separate temperature allows
up‑hill moves early on.

The routine keeps the best vector found within the time budget and
returns it as a plain Python list.  It starts from the known best
construction (``height_sequence_1``) when available, otherwise from a
grid of power‑law seeds.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------
def _evaluate_fast(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence for a normalized vector (sum>0)."""
    total = seq.sum()
    if total < 1e-12:
        return float("inf")
    # full convolution (FFT for speed)
    conv = _conv_full_fft(seq, seq)
    max_c = conv.max()
    n = seq.size
    # sum is 1 after normalisation, so denominator drops out
    return float(2.0 * n * max_c)


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using an FFT of length
    next power‑of‑two.  Returns a real array of length len(x)+len(y)-1.
    """
    m = x.size + y.size - 1
    nfft = _next_pow2(m)
    fx = np.fft.rfft(x, nfft)
    fy = np.fft.rfft(y, nfft)
    conv = np.fft.irfft(fx * fy, nfft)
    return conv[:m]


def _softmax_weights(c: np.ndarray, tau: float) -> np.ndarray:
    """Return a soft‑max distribution over `c` with temperature `tau`."""
    if tau < 1e-14:
        # As τ→0 the distribution collapses onto the maximal entries.
        max_val = c.max()
        mask = (c == max_val)
        w = np.zeros_like(c, dtype=np.float64)
        w[mask] = 1.0 / mask.sum()
        return w
    max_c = c.max()
    # subtract the max for numerical stability
    exp_vals = np.exp((c - max_c) / tau)
    w = exp_vals / exp_vals.sum()
    return w


def _gradient_direction(seq: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the descent direction for a given soft‑max weight vector `w`.

    The gradient of the smooth maximum is
        g = 2 * Σ_k w_k * shift_k(seq)
    where shift_k(seq)[i] = seq[k‑i] (zero outside the range).
    We return `-g + mean(g)` which has zero sum.
    """
    # reverse w to turn the weighted sum of shifts into a convolution
    w_rev = w[::-1]
    conv_full = _conv_full_fft(seq, w_rev)               # length n + (2n‑1) - 1 = 3n‑2
    # Extract d[i] = Σ_j seq[j] * w[i+j]  (i = 0 … n‑1)
    #   d[i] = conv_full[(len(w_rev)-1) - i]
    idx = (len(w_rev) - 1) - np.arange(seq.size)
    d = conv_full[idx]
    # Zero‑sum direction (the factor 2 is absorbed into the learning rate)
    direction = -d + d.mean()
    return direction


def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    Occasionally the whole vector is symmetrised to keep the shape balanced.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)          # minimal block length (~0.5 % of n)
    hi = max(2, n // 12)           # maximal block length (~8 % of n)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:        # mild symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

# Main entry point
# ----------------------------------------------------------------------

    betas = [0.0, 0.25, 0.5, 1.0, 2.0]

    for alpha in alphas:
        for beta in betas:
            seq = 1.0 / ((idx + beta) ** alpha)
            seq = np.clip(seq, 0.0, 1000.0)
            s = seq.sum()
            if s == 0:
                continue
            seq /= s
            sc = _evaluate_fast(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()

    # uniform seed
    seq = np.ones(n, dtype=np.float64)
    seq /= seq.sum()
    sc = _evaluate_fast(seq)
    if sc < best_score:
        best_score = sc
        best_seq = seq.copy()

    # try the known best construction if it exists
    if init_from_base:
        base_norm = np.clip(base, 0.0, 1000.0)
        total = base_norm.sum()
        if total > 0:
            base_norm /= total
            sc = _evaluate_fast(base_norm)
            if sc < best_score:
                best_score = sc
                best_seq = base_norm.copy()

    # fallback – should never be needed
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_seq /= best_seq.sum()
        best_score = _evaluate_fast(best_seq)

    # --------------------------------------------------------------
    # 2) Main optimisation loop (annealed soft‑max gradient + occasional
    #    block jumps).  All vectors are kept normalised (∑=1).
    # --------------------------------------------------------------
    current_seq = best_seq.copy()
    current_score = best_score
    start_time = time.monotonic()
    it = 0

    # annealing schedules (exponential)
    eta_start = 0.14      # initial step size
    eta_end   = 3e-5
    tau_start = 5e-3      # soft‑max temperature (smoothness)
    tau_end   = 1e-7
    temp_start = 2e-2    # Metropolis temperature
    temp_end   = 2e-6
    block_prob = 0.22     # probability of a random block perturbation

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        total   = deadline - start_time
        prog = min(1.0, elapsed / total)

        # Parameter schedules
        eta   = eta_start * (eta_end / eta_start) ** prog
        tau   = tau_start * (tau_end / tau_start) ** prog
        temper = temp_start * (temp_end / temp_start) ** prog

        # ----------------------------------------------------------
        # Compute the current convolution and the smooth‑max weights
        # ----------------------------------------------------------
        conv_full = _conv_full_fft(current_seq, current_seq)   # length 2n‑1
        max_c = conv_full.max()
        cur_score = 2.0 * n * max_c          # (sum is 1)

        weights = _softmax_weights(conv_full, tau)

        # ----------------------------------------------------------
        # Propose a candidate
        # ----------------------------------------------------------
        if rng.random() < block_prob:
            cand = _make_block_perturbation(current_seq, rng)
        else:
            direction = _gradient_direction(current_seq, weights)
            cand = current_seq + eta * direction
            cand = np.clip(cand, 0.0, 1000.0)

        # occasional extra symmetrisation (helps flatten the convolution)
        if rng.random() < 0.12:
            cand = 0.5 * (cand + cand[::-1])

        # Renormalise to keep the total mass at 1
        s = cand.sum()
        if s > 0:
            cand /= s
        else:
            cand = current_seq.copy()

        # ----------------------------------------------------------
        # Evaluate candidate
        # ----------------------------------------------------------
        cand_conv = _conv_full_fft(cand, cand)
        cand_score = 2.0 * n * cand_conv.max()

        # ----------------------------------------------------------
        # Acceptance (Metropolis style)
        # ----------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temper, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            current_seq = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"iter {it:07d} – new best: {best_score:.6f}")

        it += 1

    # ----------------------------------------------------------------------
    # Final sanity checks (the external evaluator will clip again)
    # ----------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = best_seq.sum()
    if s > 0:
        best_seq /= s

    print(f"finished after {it:,} iterations, best score={best_score:.6f}")
    return best_seq.tolist()
