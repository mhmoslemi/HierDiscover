"""
Gradient‑descent optimizer for the self‑convolution bound.

The evaluator is scale‑invariant, therefore we optimise only the shape of the
step function.  Starting from the analytically optimal “arcsine’’ (inverse‑square‑root)
profile we iteratively reduce the maximum convolution value.

The algorithm uses a smoothed maximum:
    w_k ∝ exp( λ·(c_k – max(c)) ),   c = a * a
and updates the coefficients by a multiplicative mirror‑descent step

    a_i ← a_i·exp( –η·g_i ),   g_i = 2·∑_k w_k·a_{k‑i},

which is exactly the gradient of the soft‑max of the convolution.  After each
update the vector is clipped to the evaluator’s admissible range and renormalised
to the target ℓ₁‑norm `S = sqrt(2·n)`.  The step size η and the smoothing
parameter λ are linearly annealed so that early iterations perform a strong
global smoothing while later iterations focus on the current worst shift.

The best sequence encountered during the run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Analytic arcsine start (central binomial coefficients)
# ----------------------------------------------------------------------
def arcsine_sequence(length: int) -> np.ndarray:
    """Return the discrete arcsine (inverse‑square‑root) sequence of given length.

    a[0] = 1 and a[i] = a[i‑1]·(2·i‑1)/(2·i) for i≥1.
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# 2.  Helper: evaluate the objective without calling the heavyweight routine.
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray) -> float:
    """Return the same score as `evaluate_sequence` for a valid, already‑clipped
    sequence.  The evaluator’s clipping is already accounted for.
    """
    n = seq.shape[0]
    if np.isnan(seq).any() or np.isinf(seq).any():
        return np.inf
    total = float(np.sum(seq))
    if total < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    max_conv = float(np.max(conv))
    return float(2.0 * n * max_conv / (total * total))


# ----------------------------------------------------------------------
# 3.  Core optimisation loop
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 500,
                     **kwargs) -> list[float]:
    """
    Gradient‑descent optimiser for the convolution bound.
    Runs for at most `budget_s` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size – respect the reference sequence if present.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        ref = globals()["height_sequence_1"]
        n = int(len(ref))
        try:
            init = np.array(ref, dtype=np.float64)
            if init.shape[0] != n:
                raise ValueError
        except Exception:
            init = arcsine_sequence(n)
    else:
        n = 256
        init = arcsine_sequence(n)

    # Normalise the starting shape to the target ℓ₁‑norm.
    target_sum = float(np.sqrt(2.0 * n))
    a = np.clip(init, 0.0, 1000.0)
    a = a * (target_sum / np.sum(a))

    # Initial score.
    best = a.copy()
    best_score = fast_score(best)
    cur = a.copy()
    cur_score = best_score
    print(f"[init] n={n}, score={best_score:.6f}")

    # Annealing schedule parameters.
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # tiny safety margin

    # eta (multiplicative step) goes from 0.5 → 0.001
    eta_start = 0.5
    eta_end = 0.001
    # lambda (soft‑max sharpness) goes from 0.1 → 5.0
    lam_start = 0.1
    lam_end = 5.0

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----------------------------------------------------------
        # 1) compute the convolution and the smoothed weight vector.
        # ----------------------------------------------------------
        conv = np.convolve(cur, cur)                # length 2n‑1
        max_c = float(np.max(conv))

        # linear schedule for λ
        elapsed = time.time() - start_time
        frac_time = elapsed / float(budget_s)
        lam = lam_start * (1.0 - frac_time) + lam_end * frac_time

        # soft‑max weights (stable by subtracting max)
        w = np.exp(lam * (conv - max_c))
        w_sum = np.sum(w)
        if w_sum == 0.0:
            # fallback to uniform weights
            w = np.ones_like(conv) / conv.size
        else:
            w = w / w_sum

        # ----------------------------------------------------------
        # 2) gradient of the smoothed max (mirror‑descent direction)
        # ----------------------------------------------------------
        #   g_i = 2 * Σ_k w_k * a_{k‑i}
        rev_cur = cur[::-1]
        grad_full = np.convolve(w, rev_cur)         # length 3n‑2
        # slice the part that corresponds to indices 0…n‑1
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]

        # ----------------------------------------------------------
        # 3) multiplicative update, then projection & renorm.
        # ----------------------------------------------------------
        eta = eta_start * (1.0 - frac_time) + eta_end * frac_time
        # protect against overflow – cap the exponent magnitude
        exp_arg = -eta * grad
        # limit the update factor to a reasonable range
        exp_arg = np.clip(exp_arg, -20.0, 20.0)
        cur = cur * np.exp(exp_arg)

        # clip to evaluator limits and renormalise to target ℓ₁ norm
        np.clip(cur, 0.0, 1000.0, out=cur)
        total = float(np.sum(cur))
        if total < 1e-12:               # numerical safety – restart from arcsine
            cur = arcsine_sequence(n)
            cur = cur * (target_sum / np.sum(cur))
            total = target_sum
        cur = cur * (target_sum / total)

        # ----------------------------------------------------------
        # 4) evaluate and keep the best.
        # ----------------------------------------------------------
        cur_score = fast_score(cur)
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
            print(f"[{iteration:>6}] new best = {best_score:.6f}")

        # ----------------------------------------------------------
        # 5) occasional random perturbation to avoid stagnation.
        # ----------------------------------------------------------
        if iteration % 500 == 0 and rng.random() < 0.25:
            # a mild random block re‑scale
            w_min = max(1, n // 250)
            w_max = max(2, n // 25)
            width = int(rng.integers(w_min, w_max + 1))
            start_idx = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.07)))
            cur[start_idx:start_idx + width] *= factor
            np.clip(cur, 0.0, 1000.0, out=cur)
            cur = cur * (target_sum / np.sum(cur))

    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    # Return a plain Python list as required by the harness.
    return [float(x) for x in best]
