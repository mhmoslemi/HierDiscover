def project_simplex(v):

    u = np.sort(v)[::-1]

    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def auto_convolve(a):
    n = a.shape[0]
    L = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[:2*n-1]
    return conv

def block_perturbation(base, rng):
    n = base.size
    # width: choose maybe n/100 to n/5.
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand = base.copy()
    cand[start:start+width] *= factor
    cand = np.clip(cand, 0.0, None)
    # renorm
    cand_sum = cand.sum()
    if cand_sum > 0: cand = cand / cand_sum
    else: cand = np.full_like(cand, 1.0/n)
    return cand

"""
Gradient‑based flattening of the auto‑convolution.

The evaluator is scale‑invariant, therefore we work with a probability
distribution (sum = 1).  Starting from a discretised arcsine law (the exact
continuous solution whose self‑convolution is uniform) we repeatedly:

1. Compute the auto‑convolution a*a with an FFT (O(n log n)).
2. Locate the index k where the convolution attains its maximum.
3. The gradient of the maximal entry w.r.t. a is
       g_i = 2 * a_{k-i}  (only when the partner index is in range).
4. Perform an exponentiated‑gradient step
       a_i ← a_i * exp(–η * g_i)      (η decays linearly with time)
   and renormalise to keep sum(a)=1.
5. Occasionally apply cheap stochastic perturbations (block scaling,
   smoothing) to avoid local minima and accept moves with a simple
   Metropolis rule (temperature also linearly cooled).

The procedure runs for the allotted time budget and returns the best
sequence seen.  All numbers are non‑negative and far below the 0‑1000
clipping limits, so the evaluator will not modify them.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Utility functions
# ----------------------------------------------------------------------
def auto_convolve(a: np.ndarray) -> np.ndarray:
    """Return the (full) auto‑convolution a*a using a power‑of‑two FFT."""
    n = a.size
    # length of the linear convolution
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    return conv


def arcsine_integral_sequence(n: int) -> np.ndarray:
    """
    Discretise the continuous arcsine distribution on [0,1] into n equal
    bins: a_i = F((i+1)/n) - F(i/n) with F(x) = (2/π) * arcsin(sqrt(x)).
    The resulting vector sums to 1.
    """
    x = np.arange(n + 1) / n
    # CDF of the arcsine law (Beta(1/2,1/2))
    F = (2.0 / np.pi) * np.arcsin(np.sqrt(x))
    a = np.diff(F)
    # Numerical safety: ensure non‑negative and normalise
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand = base.copy()
    cand[start : start + width] *= factor
    # renormalise to keep sum = 1
    s = cand.sum()
    if s > 0:
        cand /= s
    else:
        cand = np.full_like(cand, 1.0 / n)
    return cand


def smooth_sequence(seq: np.ndarray) -> np.ndarray:
    """Apply a small 3‑point linear smoothing filter and renormalise."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(seq, kernel, mode="same")
    sm = np.maximum(sm, 0.0)
    s = sm.sum()
    if s > 0:
        sm /= s
    else:
        sm = np.full_like(seq, 1.0 / seq.size)
    return sm


def compute_score(a: np.ndarray, n: int) -> float:
    """Score for a normalised vector (sum = 1)."""
    conv = auto_convolve(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# 2.  Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs) -> list[float]:
    """
    Gradient‑based search for a low‑max‑convolution step function.
    Returns a list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (use the reference sequence if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = int(len(globals()["height_sequence_1"]))
        try:
            init = np.array(height_sequence_1, dtype=np.float64)
            if init.shape[0] != n:
                raise ValueError
            # normalise to a probability vector
            init = np.maximum(init, 0.0)
            s = init.sum()
            if s > 0:
                init /= s
            else:
                init = arcsine_integral_sequence(n)
        except Exception:
            init = arcsine_integral_sequence(n)
    else:
        # fallback – a modest size if no reference is provided
        n = 256
        init = arcsine_integral_sequence(n)

    # ------------------------------------------------------------------
    # Baseline – discretised arcsine law
    # ------------------------------------------------------------------
    a = init.copy()
    cur_score = compute_score(a, n)

    # Keep a copy of the best solution found
    best = a.copy()
    best_score = cur_score

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Parameters for the schedule
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # safety margin
    it = 0

    # Linear cooling for step size (η) and temperature (T)
    eta_start = 5.0
    eta_end = 0.05
    T_start = 0.02
    T_end = 0.0005

    # How often to apply extra smoothing / random perturbations
    smooth_interval = max(1, n // 10)       # ~ few hundred for our n
    perturb_interval = max(1, n // 5)

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1.  Compute convolution of the current candidate
        # --------------------------------------------------------------
        conv = auto_convolve(a)
        k_max = int(np.argmax(conv))

        # --------------------------------------------------------------
        # 2.  Gradient of the maximal entry (sub‑gradient)
        #    g_i = 2 * a_{k_max-i}  for valid indices
        # --------------------------------------------------------------
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[partner[mask]]   # sub‑gradient w.r.t. a_i

        # --------------------------------------------------------------
        # 3.  Determine step size and temperature (linear cooling)
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        frac = elapsed / float(budget_s)
        eta = eta_start * (1.0 - frac) + eta_end * frac
        T = T_start * (1.0 - frac) + T_end * frac

        # --------------------------------------------------------------
        # 4.  Propose a new candidate (exponentiated‑gradient step)
        # --------------------------------------------------------------
        cand = a * np.exp(-eta * g)
        # renormalise to keep sum = 1
        s = cand.sum()
        if s > 0:
            cand /= s
        else:
            cand = np.full_like(cand, 1.0 / n)

        # Occasionally smooth the candidate (helps prevent spikes)
        if it % smooth_interval == 0:
            cand = smooth_sequence(cand)

        # Occasionally apply a random block perturbation
        if it % perturb_interval == 0 and rng.random() < 0.4:
            cand = block_perturbation(cand, rng)

        # --------------------------------------------------------------
        # 5.  Evaluate the candidate
        # --------------------------------------------------------------
        cand_score = compute_score(cand, n)

        # --------------------------------------------------------------
        # 6.  Acceptance test (Metropolis rule)
        # --------------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability to accept a worsening move
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[{it:>7}] new best = {best_score:.6f}")

                # Early stop if we beat the target bound
                if best_score < 1.503:
                    print("[early] target reached")
                    break

        # ------------------------------------------------------------------
        # 7.  Simple stagnation handling – re‑initialise from the arcsine if
        #     no improvement for a long period.
        # ------------------------------------------------------------------
        if it % 20000 == 0:
            # occasional global reset (keeps the best seen)
            a = arcsine_integral_sequence(n)

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return a plain Python list of floats
    return [float(x) for x in best]
