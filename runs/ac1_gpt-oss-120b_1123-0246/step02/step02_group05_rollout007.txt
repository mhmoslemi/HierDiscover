        init_seq = np.array(arr0, dtype=np.float64)
        # ensure nonnegative and not all zeros.

        if init_seq.sum() < 0.01:
            # fallback to arcsine
            init_seq = build_arcsine_sequence(n)

    # Renormalize sum to S0 = init_seq.sum()

    # division factor for step size

def block_perturbation(base, rng, target_sum):
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 10)  # bigger block
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.5))) # larger sigma
    cand[start:start+width] *= factor
    # optional symmetrical
    if rng.random() < 0.2:
        cand = 0.5 * (cand + cand[::-1])
    # renorm
    cand_sum = cand.sum()
    if cand_sum > 0:
        cand *= target_sum / cand_sum
    return cand

def transfer_move(base, rng, target_sum):
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n-1))
    if j >= i:
        j+=1
    # Transfer up to 50% mass
    frac = rng.random() * 0.5
    delta = frac * base[i]
    if delta == 0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    cand_sum = cand.sum()
    # renorm to target sum (scale close)
    # sum unchanged after delta transfer; but due to floating errors may drift small; we can keep as is
    # Ensure sum small drift corrected
    if cand_sum != target_sum:
        cand *= target_sum / cand_sum
    return cand

    w /= w.sum()
    # compute gradient efficiently (conv of reversed seq with w)
    seq_rev = seq[::-1]
    size = 1 << ((len(seq_rev) + len(w) - 1).bit_length())
    fa_rev = np.fft.rfft(seq_rev, n=size)
    fw = np.fft.rfft(w, n=size)
    fg = np.fft.irfft(fa_rev * fw, n=size)[:len(seq_rev) + len(w) - 1]
    g = 2.0 * fg[n-1:n-1+n]

    # Determine step magnitude
    # compute typical scale: avg = target_sum / n
    avg = target_sum / n
    # compute scaling factor = avg / (max(|g|)+eps)
    max_abs_g = np.max(np.abs(g)) + 1e-12
    step = avg / max_abs_g  # this ensures max change per entry ~ avg
    # add random factor to step magnitude
    step *= rng.random() * 0.5 + 0.5  # between 0.5 and 1.0

    cand = seq - step * g
    # ensure nonnegative
    cand = np.clip(cand, 0.0, 1000.0)
    # renorm to target sum
    cur_sum = cand.sum()
    if cur_sum > 0:
        cand *= target_sum / cur_sum
    return cand

"""
Hybrid simulated‑annealing + soft‑max gradient descent for the
    2·n·max(a*a)/(sum(a)^2)  minimisation problem.

The algorithm:
    • Initialise with the analytically good “arcsine’’ sequence (or the
      provided height_sequence_1 if it exists).
    • Keep the ℓ₁‑norm (sum of the coefficients) fixed – the objective is
      scale‑invariant, this only stabilises the numerics.
    • In each iteration propose a candidate by one of several stochastic
      moves:
          – block scaling,
          – transfer of mass between two entries,
          – early‑to‑late transfer,
          – a gradient‑guided move obtained from a soft‑max of the current
            self‑convolution (β is increased during the run).
    • Accept improving candidates unconditionally and occasionally accept
      worsening ones under a temperature schedule (simulated annealing).
    • Remember the best sequence seen and return it once the time budget
      expires (or as soon as the target 1.503 is reached).

All heavy work (convolution, gradient) uses FFT‑based convolution which
runs in O(n log n) and comfortably fits into the 1000 s budget for
n≈5 000.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the classic arcsine (inverse‑square‑root) sequence
    a[i] = binom(2i,i) / 4**i  (i=0,…,length‑1)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


def renormalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Rescale seq so that its sum equals target_sum (clipping first)."""
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur == 0.0:
        return seq
    seq *= target_sum / cur
    return seq


# ----------------------------------------------------------------------
# FFT based convolution (real → real) – self‑convolution only
# ----------------------------------------------------------------------


def compute_self_convolution(seq: np.ndarray) -> np.ndarray:
    """Return a*a using an FFT of size power‑of‑two (real input)."""
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(seq, n=size)
    fc = fa * fa                               # point‑wise product
    conv = np.fft.irfft(fc, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Objective (exactly the same as evaluate_sequence)
# ----------------------------------------------------------------------


def objective(seq: np.ndarray) -> float:
    """2·n·max(b)/(sum(a)²) where b = a*a (linear convolution)."""
    n = seq.shape[0]
    s = float(seq.sum())
    if s < 0.01:
        return np.inf
    b = compute_self_convolution(seq)
    max_b = float(b.max())
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Stochastic perturbations (keep ℓ₁‑norm fixed)
# ----------------------------------------------------------------------


def block_perturbation(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ))."""
    n = base.size
    # block width – from 0.5 % up to 20 % of the sequence
    w_min = max(1, n // 200)
    w_max = max(2, n // 5)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.5)))   # fairly large σ
    cand[start:start + width] *= factor
    # occasional symmetrisation – helps when the optimum is roughly symmetric
    if rng.random() < 0.2:
        cand = 0.5 * (cand + cand[::-1])
    return renormalise(cand, target_sum)


def transfer_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """Move a random fraction (≤50 %) of mass from one index to another."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    frac = rng.random() * 0.5               # up to half of the source entry
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    # sum is unchanged up to floating‑point noise; renormalise for safety
    return renormalise(cand, target_sum)


def early_to_late_move(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Shift a fraction of mass from the first 10 % to the last 10 %."""
    n = base.size
    early_cut = max(1, n // 10)
    late_start = max(early_cut, n - early_cut)
    i = int(rng.integers(0, early_cut))
    j = int(rng.integers(late_start, n))
    if i == j:
        return base.copy()
    frac = rng.random() * 0.5
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


# ----------------------------------------------------------------------
# Soft‑max gradient‑guided move
# ----------------------------------------------------------------------


def gradient_descent_move(seq: np.ndarray, cur_conv: np.ndarray,
                         beta: float, target_sum: float,
                         rng: np.random.Generator) -> np.ndarray:
    """
    Perform one gradient‑descent step on the smooth surrogate
    L = log‑sum‑exp(β·(a*a)).  The step size adapts to the magnitude of the
    gradient and the current average coefficient.
    """
    n = seq.size
    max_b = cur_conv.max()

    # soft‑max weights for each convolution index
    w = np.exp(beta * (cur_conv - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:                # should never happen, but be defensive
        w[:] = 1.0 / w.size
    else:
        w /= w_sum

    # gradient:  ∂L/∂a_j = 2 * Σ_i w_{i+j} * a_i  (cross‑correlation)
    a_rev = seq[::-1]
    size = 1 << ((len(a_rev) + len(w) - 1).bit_length())
    fa_rev = np.fft.rfft(a_rev, n=size)
    fw = np.fft.rfft(w, n=size)
    corr = np.fft.irfft(fa_rev * fw, n=size)[: len(a_rev) + len(w) - 1]
    # slice the part that corresponds to indices 0…n‑1
    grad = 2.0 * corr[n - 1:n - 1 + n]

    # step magnitude – aim at moving roughly a fraction of the average entry
    avg = target_sum / n
    max_abs = np.max(np.abs(grad)) + 1e-12
    step = avg / max_abs                       # makes the biggest change ≈ avg
    step *= (rng.random() * 0.5 + 0.5)         # randomise a bit (0.5–1.0)

    cand = seq - step * grad
    cand = np.clip(cand, 0.0, 1000.0)
    return renormalise(cand, target_sum)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    2·n·max(a*a)/(sum(a)²).  The routine mixes stochastic perturbations,
    a soft‑max gradient descent, and simulated‑annealing acceptance.
    It stops when the time budget expires (or when the target 1.503 is
    reached) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size n
    # ------------------------------------------------------------------
    init_from_global = globals().get("height_sequence_1", [])
    if isinstance(init_from_global, (list, tuple, np.ndarray)) and len(init_from_global) > 0:
        n = len(init_from_global)
        base = np.array(init_from_global, dtype=np.float64)
        base = np.clip(base, 0.0, 1000.0)
        if base.sum() < 0.01:                # degenerate – fall back to arcsine
            base = build_arcsine_sequence(n)
    else:
        # modest fallback size – 1024 usually gives a good trade‑off
        n = 1024
        base = build_arcsine_sequence(n)

    # ------------------------------------------------------------------
    # Normalise the ℓ₁‑norm (scale does not affect the objective)
    # ------------------------------------------------------------------
    target_sum = float(base.sum())
    cur_seq = renormalise(base, target_sum)
    cur_score = objective(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Annealing schedule
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    # temperature parameters – start relatively hot, cool to zero
    T0 = max(0.01 * best_score, 1e-4)
    # β‑schedule for the soft‑max (start smooth, end sharp)
    beta0, beta_max = 0.5, 30.0

    it = 0
    last_report = start_time

    # keep the convolution of the current best for gradient moves
    cur_conv = compute_self_convolution(cur_seq)

    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start_time
        frac = elapsed / budget_s

        # annealing temperature (linear cooling)
        T = T0 * (1.0 - frac)

        # soft‑max β schedule
        beta = beta0 + frac * (beta_max - beta0)

        # ------------------------------------------------------------------
        # Choose a move type
        # ------------------------------------------------------------------
        r = rng.random()
        if r < 0.45:                                   # gradient step (most frequent)
            cand_seq = gradient_descent_move(cur_seq, cur_conv,
                                             beta, target_sum, rng)
        elif r < 0.65:                                 # block scaling
            cand_seq = block_perturbation(cur_seq, rng, target_sum)
        elif r < 0.85:                                 # mass transfer
            cand_seq = transfer_move(cur_seq, rng, target_sum)
        else:                                          # early‑to‑late shift
            cand_seq = early_to_late_move(cur_seq, rng, target_sum)

        cand_score = objective(cand_seq)

        # ------------------------------------------------------------------
        # Acceptance (simulated annealing)
        # ------------------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        elif T > 0.0:
            prob = np.exp(-(cand_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_seq = cand_seq
            cur_score = cand_score
            cur_conv = compute_self_convolution(cur_seq)  # needed for next gradient

        # ------------------------------------------------------------------
        # Record the best sequence ever seen
        # ------------------------------------------------------------------
        if cand_score < best_score:
            best_seq = cand_seq.copy()
            best_score = cand_score
            print(f"[{it:>7}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

            if best_score < 1.503:                # target already met
                break

        # occasional progress report
        if it % 1000 == 0 and time.monotonic() - last_report > 5.0:
            print(f"[{it}] cur={cur_score:.6f}, best={best_score:.6f}, "
                  f"β={beta:.2f}, T={T:.5f}")
            last_report = time.monotonic()

    total_time = time.monotonic() - start_time
    print(f"[final] best score {best_score:.6f} after {it} iterations, "
          f"elapsed {total_time:.2f}s")
    # Return a plain Python list – the harness expects that form.
    return [float(x) for x in best_seq]
