"""
Simple stochastic optimizer for the convolution‑bound problem.

The evaluator is scale‑invariant, therefore we only need to optimise the
*shape* of the non‑negative step function.  The algorithm starts from the
analytically optimal “arcsine” (inverse‑square‑root) sequence and then runs
a simulated‑annealing loop with several cheap moves:

* block_perturbation – multiply a random contiguous block by a log‑normal factor.
* transfer_move      – transfer a random fraction of mass from one index to another.
* swap_move          – exchange two entries.
* targeted_move      – move a tiny amount of mass away from the pair that
                       currently contributes most to the maximal convolution.
* edge_balancing_move – move mass from the pair with the **largest**
                       product on the maximal diagonal to the pair with the
                       **smallest** product (helps to flatten the convolution).
* gradient_move      – deterministic step that pushes the sequence towards a
                       state where all contributions to the maximal diagonal are
                       equal (the direction of the true gradient of the score).

All moves preserve the total L1‑mass (so the denominator of the evaluation
function stays constant) and respect the evaluator’s clipping rules.
A simple Metropolis acceptance rule with a linearly cooled temperature allows
temporary worsening steps and helps escape local minima.

The routine runs until the time budget expires (or the target score of 1.503 is
reached) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Analytic start – the arcsine (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i  (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# 2.  Simple stochastic moves.
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.3:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another (sum unchanged)."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def swap_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Exchange two entries – useful for reshuffling large values."""
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    cand = base.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand


def targeted_move(base: np.ndarray,
                  conv: np.ndarray,
                  k_max: int,
                  rng: np.random.Generator) -> np.ndarray:
    """
    Reduce the dominant convolution entry.
    - Find the pair (i, j) with i+j = k_max that gives the largest product.
    - Move a small amount of mass away from that pair to two random positions.
    The total sum stays unchanged.
    """
    n = base.size
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idxs = np.arange(i_start, i_end + 1)
    partners = k_max - idxs
    products = base[idxs] * base[partners]
    max_pos = int(np.argmax(products))
    i = int(idxs[max_pos])
    j = int(partners[max_pos])

    l1, l2 = rng.choice(n, size=2, replace=False)

    max_move = min(base[i], base[j])
    if max_move == 0:
        return base.copy()
    delta = rng.random() * 0.20 * max_move

    cand = base.copy()
    if i == j:
        cand[i] -= 2.0 * delta
        cand[l1] += delta
        cand[l2] += delta
    else:
        cand[i] -= delta
        cand[j] -= delta
        cand[l1] += delta
        cand[l2] += delta
    return np.clip(cand, 0.0, 1000.0)


def edge_balancing_move(base: np.ndarray,
                        conv: np.ndarray,
                        k_max: int,
                        rng: np.random.Generator) -> np.ndarray:
    """
    Move mass from the pair (i,j) that *maximises* a[i]*a[j] on the
    maximal diagonal to the pair (u,v) that *minimises* a[u]*a[v] on the same
    diagonal.  This explicitly tries to flatten the dominant convolution.
    """
    n = base.size
    idx = np.arange(n)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)
    idx = idx[mask]
    partner = partner[mask]

    products = base[idx] * base[partner]
    if products.size == 0:
        return base.copy()

    max_pos = int(np.argmax(products))
    min_pos = int(np.argmin(products))

    i, j = int(idx[max_pos]), int(partner[max_pos])
    u, v = int(idx[min_pos]), int(partner[min_pos])

    max_move = min(base[i], base[j])
    if max_move == 0:
        return base.copy()
    delta = rng.random() * 0.25 * max_move   # take at most 25 % of the smaller entry

    cand = base.copy()
    if i == j:
        cand[i] -= 2.0 * delta
    else:
        cand[i] -= delta
        cand[j] -= delta

    if u == v:
        cand[u] += 2.0 * delta
    else:
        cand[u] += delta
        cand[v] += delta

    return np.clip(cand, 0.0, 1000.0)


def gradient_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Deterministic gradient‑like step.
    Let k* be the index of the maximal convolution value.
    For each i define g_i = a_{k*-i} (or 0 if out of range).
    The gradient of the objective (up to a positive factor) is
        sum_a * g_i - max_b .
    We move a little in the opposite direction while preserving non‑negativity.
    """
    n = base.size
    conv = np.convolve(base, base)
    maxb = float(conv.max())
    k_max = int(np.argmax(conv))
    sum_a = float(base.sum())

    # g_i = a_{k_max-i} when the index is valid
    idx = np.arange(n)
    partner = k_max - idx
    valid = (partner >= 0) & (partner < n)
    g = np.zeros_like(base)
    g[valid] = base[partner[valid]]

    delta = sum_a * g - maxb          # direction that would increase the score
    # take a tiny step opposite to delta
    step = 0.02 * delta
    cand = base - step
    return np.clip(cand, 0.0, 1000.0)

        elapsed = time.time() - start_time
        frac = elapsed / float(budget_s)
        T = T_start * (1.0 - frac) + T_end * frac   # linear cooling

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                print(f"[{it:>7}] new best = {best_score:.6f}")

                if best_score < 1.503:
                    print("[early] target reached")
                    break

        # occasional global perturbation to avoid deep stagnation
        if it % 5000 == 0 and rng.random() < 0.5:
            cur = np.clip(build_arcsine_sequence(n), 0.0, 1000.0)

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    return [float(x) for x in best]
