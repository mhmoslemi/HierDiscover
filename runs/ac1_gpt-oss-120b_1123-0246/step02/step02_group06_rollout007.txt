def solve_lp_max_sum(f, M):
    n = len(f)
    # Build matrix A_ub: shape (2n-1, n)
    offsets = np.arange(n)
    data = [np.full(n, f[d]) for d in range(n)]
    A = sp.diags(data, offsets, shape=(2*n-1, n), format='csc')
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if not res.success:
        return None
    g = res.x
    # ensure nonnegativity
    g = np.maximum(g, 0.0)
    return g

"""
Iterative linear‑programming optimisation for the step‑function problem.

We keep the total ℓ1‑norm fixed (∑a = √(2·n)); with that
normalisation the evaluation score coincides with the maximum of the
self‑convolution (M = max(a*a)).  Starting from the analytically optimal
“arcsine” shape we repeatedly:

1. Compute M = max(conv(f,f)).
2. Solve the linear program
       maximise   1ᵀ g
       subject to  f * g ≤ M   (element‑wise, linear in g)
                  g ≥ 0
   The constraint matrix is the lower‑triangular Toeplitz matrix
   generated from f, built with a sparse diagonal representation.
3. Form a convex combination (1‑t)·f + t·g for several mixing
   factors t∈{0.5,0.3,0.2,0.1,0.05}, renormalise the result to the
   fixed ℓ1‑norm and keep the best improvement.
4. If a number of iterations passes without improvement we perform a
   tiny random mass‑transfer to escape possible stagnation.

Each iteration strictly reduces the maximal convolution (unless the LP
fails), driving the score towards the theoretical lower bound 1.
The routine runs for the allotted time budget (default 500 s, the
harness may allow up to 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


def _build_arcsine_sequence(length: int) -> np.ndarray:
    """Arcsine (central‑binomial) sequence a[i] = C(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    return a


def _score(seq: np.ndarray) -> float:
    """Exact objective (identical to evaluate_sequence for valid seq)."""
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = np.max(b)
    s = np.sum(seq)
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def _build_conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """
    Return the (2n‑1)×n matrix A such that A @ g = f * g (linear in g).
    A is lower‑triangular Toeplitz: each sub‑diagonal d (0≤d<n) is constant f[d].
    """
    n = f.shape[0]
    offsets = np.arange(n)               # 0,1,…,n‑1
    # every diagonal consists of the same value f[d] repeated n times
    data = [np.full(n, f[d]) for d in range(n)]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise 1ᵀg   subject to  f*g ≤ M   and g≥0.
    Returns the optimal g (as a dense ndarray) or None if LP fails.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)                       # (2n‑1)×n matrix
    b_ub = np.full(2 * n - 1, M)                    # right‑hand side
    c = -np.ones(n)                                 # minimise -sum(g)
    bounds = [(0.0, None)] * n                       # g ≥ 0
    # 'highs' is the modern performant LP solver in SciPy
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        return None
    g = np.maximum(res.x, 0.0)                      # eliminate tiny negatives
    return g


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    Returns a plain list of floats (the heights).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a manageable length.  If the harness supplies a reference
    # sequence we honour its size up to a safe ceiling (512).
    # ------------------------------------------------------------------
    ref = globals().get("height_sequence_1", [])
    n = int(len(ref))
    if n == 0:
        n = 256
    else:
        n = min(n, 512)               # keep the LP tractable

    # ------------------------------------------------------------------
    # Initialise with the analytically optimal arcsine shape and fix its
    # ℓ1‑norm to √(2·n) (the same normalisation used in the reference paper).
    # ------------------------------------------------------------------
    f = _build_arcsine_sequence(n)
    f = np.clip(f, 0.0, 1000.0)

    target_sum = math.sqrt(2.0 * n)
    f = f * (target_sum / np.sum(f))

    best = f.copy()
    best_score = _score(best)

    print(f"[init] length={n}, initial score={best_score:.6f}")

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)   # keep 1 s safety margin
    it = 0
    no_improve_counter = 0

    # mixing factors we will try each iteration (larger first)
    mix_factors = [0.5, 0.3, 0.2, 0.1, 0.05]

    while time.monotonic() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # Normalise at the start of the loop (protect against drift).
        # ------------------------------------------------------------------
        f = f * (target_sum / np.sum(f))

        # current max convolution M
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)

        # ------------------------------------------------------------------
        # Linear‑programming step: find a direction g that can increase the
        # ℓ1‑mass without violating the current max bound.
        # ------------------------------------------------------------------
        g = _solve_lp_max_sum(f, M)
        if g is None:
            # LP failed – fall back to a tiny random perturbation.
            i, j = rng.integers(0, n, size=2)
            if f[i] > 0:
                delta = 0.01 * f[i]
                f[i] -= delta
                f[j] += delta
            continue

        # ------------------------------------------------------------------
        # Try several convex combinations and keep the best improvement.
        # After mixing we re‑normalise to the fixed ℓ1‑norm.
        # ------------------------------------------------------------------
        improved = False
        for t in mix_factors:
            cand = (1.0 - t) * f + t * g
            cand = cand * (target_sum / np.sum(cand))   # keep normalisation
            cand_score = _score(cand)

            if cand_score < best_score - 1e-9:
                best = cand.copy()
                best_score = cand_score
                f = cand.copy()                         # continue from here
                improved = True
                no_improve_counter = 0
                print(f"[{it:>6}] t={t:.2f} new best = {best_score:.6f}")
                break

        if not improved:
            no_improve_counter += 1
            # occasional random mass‑transfer to escape flat plateaux
            if no_improve_counter % 150 == 0:
                i, j = rng.integers(0, n, size=2)
                if f[i] > 0:
                    delta = 0.02 * f[i]                # move 2 % of a[i]
                    f[i] -= delta
                    f[j] += delta

        # ------------------------------------------------------------------
        # Early exit if we have already beaten the target comfortably.
        # ------------------------------------------------------------------
        if best_score < 1.5:
            break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Convert to plain Python floats for the harness.
    return [float(x) for x in best]
