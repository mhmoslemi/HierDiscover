Pseudo:

```
def project_simplex(v, S):
    # v is array of any real numbers.
    # Return projection onto {x >=0, sum(x)=S}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=42, budget_s=950, **kwargs):
    import numpy as np, time

    rng = np.random.default_rng(seed)

    n = len(globals().get("height_sequence_1", []))
    if n == 0:
        n = 256

    # compute target sum (unscaled) - trivial using baseline arcsine sum
    def arcsine_seq():
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i-1] * ((2*i - 1)*(2*i)) / (4.0*i*i)
        return a

    seq_candidates = []
    # arcsine
    a0 = arcsine_seq()
    seq_candidates.append(a0)
    # power law: a_i = (i+1)^-0.5
    idx = np.arange(n) + 1
    a_power = idx.astype(np.float64) ** -0.5
    seq_candidates.append(a_power)
    # random Dirichlet
    random_dir = rng.random(n)
    random_dir = random_dir / random_dir.sum() * a0.sum()  # scale to same sum
    seq_candidates.append(random_dir)

    best = None
    best_score = np.inf

    for seq in seq_candidates:
        sc = evaluate_sequence(seq.tolist())
        if sc < best_score:
            best = seq.copy()
            best_score = float(sc)
            print(f"[init] candidate score {best_score:.6f}")

    # Projection onto simplex ensure sum fixed
    S = best.sum()  # sum from best candidate

    def project_simplex(v, S):
        # Implementation O(n log n)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - S
        rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
        if rho.size == 0:
            theta = 0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0)
        return w

    # subgradient descent
    max_iters = 200000
    t0 = 0.1  # step size factor
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        # compute convolution via FFT
        # obtain full conv
        m = 1 << ((2 * n - 1).bit_length())
        fft_a = np.fft.rfft(best, n=m)
        conv = np.fft.irfft(fft_a * fft_a, n=m)[:2 * n - 1]
        # Determine maximal entry
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # compute gradient
        g = np.zeros_like(best)
        low = max(0, k_max - (n - 1))
        high = min(k_max, n - 1)
        idx_range = np.arange(low, high + 1)
        g[idx_range] = 2.0 * best[k_max - idx_range]

        # step size decreasing
        eta = t0 / np.sqrt(it)
        cand = best - eta * g
        # project onto simplex (sum = S)
        cand = project_simplex(cand, S)

        # evaluate
        sc = evaluate_sequence(cand.tolist())
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = float(sc)
            print(f"[{it}] new best {best_score:.6f}")

            # early stop if below target
            if best_score < 1.503:
                print("[stop] target reached")
                break

        # optional: occasional random perturbation to escape
        if it % 5000 == 0:
            # small random noise
            noise = rng.normal(scale=0.01, size=n)
            cand_noise = best + noise
            cand_noise = np.maximum(cand_noise, 0.0)
            cand_noise = project_simplex(cand_noise, S)
            scn = evaluate_sequence(cand_noise.tolist())
            if np.isfinite(scn) and scn < best_score:
                best = cand_noise
                best_score = float(scn)
                print(f"[{it}] noise improved to {best_score:.6f}")

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    return [float(x) for x in best]

"""
Projected sub‑gradient descent for minimizing the convolution‐bound.

The routine builds a few simple initial sequences (the exact arcsine
distribution, a 1/√i power law and a random Dirichlet vector), picks the
best one and then repeatedly:
    * computes the full self‑convolution (FFT for speed),
    * finds the index where the convolution is maximal,
    * builds the sub‑gradient of that maximal entry,
    * takes a diminishing step and projects back onto the simplex
      (non‑negative entries with the original total sum).

The projection guarantees feasibility for ``evaluate_sequence``.
Occasional small random perturbations are added to avoid stagnation.
The loop stops when the time budget is almost exhausted or when the
target value 1.503 is reached.
"""

import time
import numpy as np


def build_arcsine_sequence(length: int) -> np.ndarray:
    """Arcsine (inverse‑square‑root) coefficients:
    a[i] = binom(2i,i) / 4**i  (exact up to floating point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    return a


def project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    { x | x_i >= 0, sum_i x_i = target_sum }.
    """
    # Implementation follows the algorithm of
    # Duchi et al., “Efficient Projections onto the l1‑Ball for Learning in
    # High Dimensions”, ICML 2008.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(a: np.ndarray, fft_size: int) -> tuple[float, int, np.ndarray]:
    """
    Returns (max_value, argmax_index, full_convolution) of a*a using FFT.
    ``fft_size`` should be a power‑of‑two >= 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=fft_size)
    conv = np.fft.irfft(fa * fa, n=fft_size)[: 2 * a.size - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx, conv


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that minimizes
    ``evaluate_sequence``.  The method combines a few analytic starts,
    a projected sub‑gradient descent on the maximal convolution entry
    and occasional random perturbations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256                     # safe fallback

    # ------------------------------------------------------------------
    # Initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # (1) arcsine
    a_arcs = build_arcsine_sequence(n)
    candidates.append(a_arcs)

    # (2) simple power‑law  a_i = (i+1)^{-0.5}
    idx = np.arange(n, dtype=np.float64) + 1.0
    a_pow = idx ** -0.5
    candidates.append(a_pow)

    # (3) random Dirichlet (scaled to same total mass)
    rnd = rng.random(n)
    a_dir = rnd / rnd.sum() * a_arcs.sum()
    candidates.append(a_dir)

    # pick the best start according to the true evaluator
    best = None
    best_score = np.inf
    for seq in candidates:
        sc = evaluate_sequence(seq.tolist())
        if sc < best_score:
            best = seq.copy()
            best_score = float(sc)
            print(f"[init] candidate score {best_score:.6f}")

    # keep the sum fixed for the whole optimisation (scale invariance)
    S = best.sum()
    if S < 0.01:
        # safeguard – shouldn't happen with the builds above
        S = 1.0
        best = best / best.sum()

    # Pre‑compute the FFT length (next power of two)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent on the maximal convolution entry
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # leave a small slack
    it = 0
    step0 = 0.5                     # initial step size factor

    # Helper: compute objective from convolution maximum (avoids extra eval)
    def objective_from_max(max_val: float) -> float:
        return 2.0 * n * max_val / (S * S)

    while time.monotonic() < deadline:
        it += 1

        # current max convolution entry and its index
        cur_max, k_max, conv = max_convolution(best, fft_len)

        # occasional cheap status report
        if it % 5000 == 0:
            cur_score = objective_from_max(cur_max)
            print(f"[{it:>6}] cur_score={cur_score:.6f}   max_idx={k_max}")

        # sub‑gradient of the active (maximal) convolution entry
        g = np.zeros_like(best)
        low = max(0, k_max - (n - 1))
        high = min(k_max, n - 1)
        idx_range = np.arange(low, high + 1)
        g[idx_range] = 2.0 * best[k_max - idx_range]

        # diminishing step size
        eta = step0 / np.sqrt(it)

        # projected gradient step
        cand = best - eta * g
        cand = project_onto_simplex(cand, S)

        # evaluate new candidate (fast path)
        new_max, _, _ = max_convolution(cand, fft_len)
        new_score = objective_from_max(new_max)

        # accept if strictly better
        if new_score < best_score:
            best = cand
            best_score = new_score
            print(f"[{it}] new best {best_score:.6f}")

            if best_score < 1.503:
                print("[stop] target reached")
                break

        # --------------------------------------------------------------
        # Simple diversification: every few hundred iterations add a tiny
        # random jitter and re‑project.  This helps to escape shallow plateaus.
        # --------------------------------------------------------------
        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            jitter = best + noise
            jitter = np.maximum(jitter, 0.0)
            jitter = project_onto_simplex(jitter, S)
            j_max, _, _ = max_convolution(jitter, fft_len)
            j_score = objective_from_max(j_max)
            if j_score < best_score:
                best = jitter
                best_score = j_score
                print(f"[{it}] jitter improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check using the official evaluator (cheap)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best.tolist())
    if final_score < best_score:
        best_score = float(final_score)
    print(f"[final] best score {best_score:.6f} after {it} iterations")
    return [float(x) for x in best]
