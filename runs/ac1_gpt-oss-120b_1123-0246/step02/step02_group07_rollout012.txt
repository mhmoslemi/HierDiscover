def propose_candidate(seed=...), budget_s=1000):
    rng = np.random.default_rng(seed)
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 256
    base_seq = build_arcsine_sequence(n)
    best = base_seq.copy()
    best_score = evaluate_sequence(best.tolist())
    # Phase A: random combos
    start_time = time.time()
    # allocate some time for random search
    while time.time() - start_time < budget_s * 0.4:
        # generate random parameters
        alpha = rng.random()
        p = rng.uniform(0.5, 1.2)
        q = rng.uniform(0.5, 1.2)
        seq = (1-alpha) * np.power(base_seq, p) + alpha * np.power(base_seq[::-1], q)
        # add small random noise to avoid being stuck
        if rng.random() < 0.3:
            # block perturbation
            seq = block_perturbation(seq, rng)
        # clamp
        seq = np.clip(seq, 0.0, 1000.0)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best = seq
            best_score = score
            print("[random] new best", best_score)
    # Phase B: directed local search with SA
    deadline = time.monotonic() + budget_s * 0.6
    it = 0
    cur_seq = best.copy()
    cur_score = best_score
    while time.monotonic() < deadline:
        it+=1
        # propose move
        if rng.random() < 0.7:
            cand = directed_transfer_move(cur_seq, rng)
        else:
            cand = block_perturbation(cur_seq, rng)
        cand_score = evaluate_sequence(cand.tolist())
        # accept if better or with probability exp(-(cand_score-cur_score)/T)
        # define temperature T schedule decreasing over time
        elapsed = time.monotonic() - (deadline - budget_s * 0.6)
        frac = (deadline - time.monotonic()) / (budget_s * 0.6)   # decreasing
        T = max(0.0001, 0.1 * frac)   # start at ~0.1, decay
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / T):
            cur_seq, cur_score = cand, cand_score
        # update best
        if cur_score < best_score:
            best, best_score = cur_seq, cur_score
            print(f"[{it}] new best = {best_score:.6f}")
        # early stop if below target
        if best_score < 1.503:
            break
    print("[final] best score", best_score, "after", it, "iterations")
    return best.tolist()

def random_parameter_search(base_seq, rev_seq, rng, budget):
    best = base_seq.copy()
    best_score = evaluate_sequence(best.tolist())
    start = time.monotonic()
    while time.monotonic() - start < budget:
        alpha = rng.random()
        p = rng.uniform(0.1, 1.5)
        q = rng.uniform(0.1, 1.5)   # for rev sequence exponent
        mix1 = np.power(base_seq, p)
        mix2 = np.power(rev_seq, q)
        cand = (1-alpha) * mix1 + alpha * mix2
        cand = np.clip(cand, 0.0, 1000.0)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[rand] α={alpha:.3f}, p={p:.3f}, q={q:.3f} -> {sc:.6f}")
    return best, best_score
```

def gradient_move(base, rng):
    n = base.size
    conv = np.convolve(base, base)
    max_val = conv[:n].max()
    # compute weights for smooth max
    beta = 10.0
    # subtract max to avoid overflow
    w = np.exp(beta * (conv[:n] - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w = w / w_sum  # normalize to sum to 1
    
    # compute gradient approx
    g = np.empty(n, dtype=np.float64)
    for i in range(n):
        # dot product of w[i:] with base[:n-i]
        if i == n-1:
            g[i] = 0.0
        else:
            g[i] = 2.0 * np.dot(w[i:], base[:n-i])
    # find source index (high gradient)
    src = int(np.argmax(g))
    # find target index (low gradient)
    # we could pick index with minimal value among low percentiles
    low_thr = np.percentile(g, 20)
    low_candidates = np.where(g <= low_thr)[0]
    if low_candidates.size == 0:
        tgt = int(rng.integers(0, n))
        if tgt == src:
            tgt = (src + 1) % n
    else:
        tgt = int(rng.choice(low_candidates))
    # fraction to move
    frac = rng.random() * 0.10 + 0.01
    delta = frac * base[src]
    if delta <= 0.0:
        return base.copy()
    cand = base.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)

"""
Convolution‑bound optimiser – extended version.

The optimiser builds the exact arcsine (inverse‑square‑root) sequence,
then performs a broader deterministic sweep over tilt and power‑law
parameters, followed by a random sampling of such parameters.
Finally a stochastic local search with three move types (gradient‑guided
transfer, block perturbation and index swap) and a simple simulated‑annealing
acceptance scheme is run until the time budget expires.
The aim is to increase the total sum of the coefficients while keeping the
maximum auto‑convolution low, thereby reducing the value returned by
``evaluate_sequence``.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# 1.  Build the “arcsine’’ (inverse‑square‑root) sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the arcsine sequence a_i = binom(2i,i)/4**i for i=0..length‑1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a  # already in [0,1]


# ----------------------------------------------------------------------
# 2.  Simple perturbations used in the search.
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0, σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def swap_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap two random positions."""
    n = base.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return base.copy()
    cand = base.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand


def gradient_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Gradient‑guided mass transfer.
    Uses a soft‑max over all convolution entries to approximate the gradient
    of the maximal convolution value. Mass is moved from the index with the
    largest gradient (most harmful) to one with a small gradient.
    """
    n = base.size
    # full auto‑convolution
    conv_full = np.convolve(base, base)               # length 2n‑1
    max_val = conv_full.max()
    # smooth weights – larger for entries close to the maximum
    beta = 6.0
    w = np.exp(beta * (conv_full - max_val))          # values in (0,1]
    # gradient = 2 * (w * base)  (convolution with reversed base)
    base_rev = base[::-1]
    grad_full = np.convolve(w, base_rev)              # length 3n‑2
    # extract the part that corresponds to each index i
    grad = 2.0 * grad_full[n - 1:n - 1 + n]           # size n
    # source = index with highest gradient, target = low‑gradient index
    src = int(np.argmax(grad))
    low_thr = np.percentile(grad, 15)
    low_candidates = np.where(grad <= low_thr)[0]
    if low_candidates.size == 0:
        tgt = int(rng.integers(0, n))
        if tgt == src:
            tgt = (src + 1) % n
    else:
        tgt = int(rng.choice(low_candidates))
    # move a modest fraction of the source mass
    frac = rng.random() * 0.07 + 0.01                 # 1 % … 8 %
    delta = frac * base[src]
    if delta <= 0.0:
        return base.copy()
    cand = base.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# 3.  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a
    better bound than the analytical arcsine construction.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size – use the known construction if present.
    # --------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256                     # safe fallback

    start_time = time.monotonic()
    total_budget = max(0.0, float(budget_s))

    # ------------------------------------------------------------------
    # Phase 0 – analytical starting point (arcsine)
    # ------------------------------------------------------------------
    base_seq = build_arcsine_sequence(n)
    rev_seq = base_seq[::-1]
    best = base_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, arcsine score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic parameter sweeps (expanded ranges)
    # ------------------------------------------------------------------
    # tilt (mix arcsine & reversed) – up to 80 %
    for alpha in np.linspace(0.0, 0.80, 41):
        cand = (1.0 - alpha) * base_seq + alpha * rev_seq
        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[sweep] tilt α={alpha:.3f} → score={sc:.6f}")

    # power‑law deformation – exponents down to 0.30
    for p in np.linspace(0.30, 1.20, 19):
        cand = np.power(base_seq, p)
        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[sweep] power p={p:.3f} → score={sc:.6f}")

    # combined tilt + power
    for alpha in np.linspace(0.0, 0.60, 13):
        mixed = (1.0 - alpha) * base_seq + alpha * rev_seq
        for p in np.linspace(0.40, 1.00, 13):
            cand = np.power(mixed, p)
            cand = np.clip(cand, 0.0, 1000.0)
            sc = float(evaluate_sequence(cand.tolist()))
            if sc < best_score:
                best, best_score = cand, sc
                print(f"[sweep] α={alpha:.2f}, p={p:.2f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – randomised exploration of the same low‑dimensional family
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.20 * total_budget   # 20 % of the budget
    while time.monotonic() < phase2_deadline:
        alpha = rng.random()                         # mix weight
        p = rng.uniform(0.20, 1.40)                  # power on forward
        q = rng.uniform(0.20, 1.40)                  # power on reversed
        forward = np.power(base_seq, p)
        backward = np.power(rev_seq, q)
        cand = (1.0 - alpha) * forward + alpha * backward
        cand = np.clip(cand, 0.0, 1000.0)
        # occasional small block perturbation to escape flat regions
        if rng.random() < 0.15:
            cand = block_perturbation(cand, rng)
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[rand] α={alpha:.3f}, p={p:.3f}, q={q:.3f} → {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – directed stochastic local search (gradient + SA)
    # ------------------------------------------------------------------
    deadline = start_time + total_budget
    cur_seq = best.copy()
    cur_score = best_score
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # pick a move type
        choice = rng.random()
        if choice < 0.5:
            cand = gradient_move(cur_seq, rng)
        elif choice < 0.8:
            cand = block_perturbation(cur_seq, rng)
        else:
            cand = swap_move(cur_seq, rng)

        sc = float(evaluate_sequence(cand.tolist()))
        # simple simulated‑annealing acceptance
        if sc < cur_score:
            accept = True
        else:
            # temperature decreases linearly to a tiny value
            remaining = deadline - time.monotonic()
            total = deadline - start_time
            T = max(1e-4, 0.12 * (remaining / total))
            accept = rng.random() < np.exp(-(sc - cur_score) / T)

        if accept:
            cur_seq, cur_score = cand, sc
            if cur_score < best_score:
                best, best_score = cur_seq, cur_score
                print(f"[{it:>6}] new best = {best_score:.6f}")

                if best_score < 1.503:            # target reached
                    print("[stop] target reached")
                    break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return plain Python floats (the harness expects a list)
    return [float(x) for x in best]
