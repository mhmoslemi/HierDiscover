"""
Propose a candidate step‑function by locally optimizing the
`2·n·max(conv)/sum²` objective with a smooth approximation of the
maximum.  The algorithm works as follows:

1.  Gather a good initial shape (the best Beta‑shaped sequence
    found by the previous runner, or fall back to a simple exponential).

2.  Re‑scale it so that Σa = √(2·n) (the scaling does not affect the
    evaluation value but keeps the numbers in a convenient range).

3.  Repeatedly minimise a differentiable surrogate
    `L(a)= (1/τ)·log Σ exp(τ·conv(a,a))`,
    whose gradient can be expressed via a single convolution.
    An Adam optimiser (projected onto the non‑negative orthant and
    re‑normalised after each step) drives the surrogate down,
    which in practice reduces the true `max(conv)`.

4.  The temperature τ is slowly increased, tightening the surrogate
    towards the true maximum.

5.  Periodically the exact objective `2·n·max(conv)/sum²`
    is evaluated; the best sequence seen so far is retained.

6.  A short random‑perturbation sweep is performed at the end
    (tiny mass moves) to squeeze out any remaining improvement.

The whole routine respects the supplied time budget and returns the
best sequence found as a plain Python list.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Fast exact objective (used for comparison and final evaluation)
# ----------------------------------------------------------------------
def _fast_objective(seq: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(conv)/sum² (with clipping)."""
    seq = np.clip(seq, 0.0, 1000.0)
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return np.inf
    # convolution via FFT (next power of two)
    size = 2 * n - 1
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Helper: one gradient step for the smoothed max surrogate
# ----------------------------------------------------------------------
def _smooth_max_step(a: np.ndarray, tau: float, lr: float,
                     m1: int, m2: int,
                     target_sum: float,
                     adam_state):
    """
    Perform one Adam step on the surrogate
        L(a) = (1/τ)·log Σ_k exp(τ·conv_k)
    Returns the updated sequence and the current surrogate loss.
    """
    n = a.shape[0]
    L = 2 * n - 1                     # length of a*a convolution
    # ------------------------------------------------------------------
    # 1. FFT of a (used both for conv and gradient)
    # ------------------------------------------------------------------
    A = np.fft.rfft(a, n=m2)          # size m2 >= 3·n

    # ------------------------------------------------------------------
    # 2. exact self‑convolution (needed for the softmax weights)
    # ------------------------------------------------------------------
    conv = np.fft.irfft(A * A, n=m2)[:L]   # length L
    max_c = conv.max()

    # ------------------------------------------------------------------
    # 3. soft‑max weights over the convolution entries
    # ------------------------------------------------------------------
    # shifted exponent to avoid overflow
    exp_term = np.exp(tau * (conv - max_c))
    sum_exp = exp_term.sum()
    w = exp_term / sum_exp           # shape (L,)

    # ------------------------------------------------------------------
    # 4. gradient of the surrogate:
    #    ∂L/∂a_i = 2 * Σ_j w_{i+j} * a_j
    #    This is a convolution of a with the reversed weight vector.
    # ------------------------------------------------------------------
    w_rev = w[::-1]                  # length L
    W_rev = np.fft.rfft(w_rev, n=m2)
    conv_rev = np.fft.irfft(A * W_rev, n=m2)   # length m2

    # extract the needed slice: indices L-1‑i, i=0..n‑1
    idx_center = L - 1
    grad = 2.0 * conv_rev[idx_center - np.arange(n)]

    # ------------------------------------------------------------------
    # 5. Adam update (projected onto a≥0 and re‑normalised)
    # ------------------------------------------------------------------
    m, v, beta1, beta2, eps, t = adam_state
    t += 1
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

    # projection onto the feasible set
    a_new = np.clip(a_new, 0.0, 1000.0)

    # re‑normalise to keep the prescribed sum (scaling does not change
    # the objective, but it prevents the numbers from drifting).
    s = a_new.sum()
    if s > 0.0:
        a_new *= (target_sum / s)

    # The surrogate loss (for monitoring)
    surrogate = max_c + np.log(sum_exp) / tau

    adam_state[:] = [m, v, beta1, beta2, eps, t]   # mutate in‑place
    return a_new, surrogate

    except NameError:
        pass

    # ------------------------------------------------------------------
    # 1.  Choose a working length
    # ------------------------------------------------------------------
    # Prefer the length of the existing best sequence; otherwise try a few
    # powers of two.
    if best_seq is not None:
        n = best_seq.size
    else:
        n_candidates = [1024, 2048, 4096, 8192, 16384]
        n = n_candidates[0]
        # quick coarse trial to pick a reasonable n (very fast)
        for cand_n in n_candidates:
            if time.monotonic() >= deadline:
                break
            target_sum = sqrt(2.0 * cand_n)
            # simple exponential start
            a0 = np.exp(-np.linspace(0, 5, cand_n))
            a0 = a0 * (target_sum / a0.sum())
            sc = _fast_objective(a0)
            if sc < best_score:
                best_score = sc
                best_seq = a0.copy()
                n = cand_n
                print(f"[len‑scan] n={cand_n} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Build a refined starter (Beta‑shape with a tuned α)
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        # try a short α‑grid around the previous best (if known)
        target_sum = sqrt(2.0 * n)
        alphas = np.linspace(0.25, 0.75, 11)
        for alpha in alphas:
            if time.monotonic() >= deadline:
                break
            p = (np.arange(n, dtype=np.float64) + 0.5) / n
            w = np.power(p, alpha - 1.0) * np.power(1.0 - p, alpha - 1.0)
            a = w * (target_sum / w.sum())
            a = np.clip(a, 0.0, 1000.0)
            sc = _fast_objective(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[beta‑grid] α={alpha:.3f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Gradient‑based smoothing of the max‑convolution
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall back to a plain exponential shape
        target_sum = sqrt(2.0 * n)
        best_seq = np.exp(-np.linspace(0, 5, n))
        best_seq = best_seq * (target_sum / best_seq.sum())
    a = best_seq.copy()
    target_sum = sqrt(2.0 * n)

    # FFT sizes (pre‑computed for speed)
    conv_len = 2 * n - 1
    grad_len = 3 * n - 2
    m_fft = 1 << ((max(conv_len, grad_len) - 1).bit_length())

    # Adam state: m, v, beta1, beta2, eps, t   (all stored in a list for mutability)
    adam_state = [
        np.zeros_like(a),          # m
        np.zeros_like(a),          # v
        0.9,                       # beta1
        0.999,                     # beta2
        1e-8,                      # eps
        0                          # iteration counter t
    ]

    # optimisation hyper‑parameters
    tau = 5.0          # start temperature (soft max)
    lr = 0.5           # initial learning rate
    max_iters = 2000   # upper bound; the loop also respects the time budget

    for it in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        a, surrogate = _smooth_max_step(a, tau, lr, m_fft, m_fft, target_sum, adam_state)

        # tighten the temperature slowly (approach true max)
        if it % 30 == 0 and tau < 40.0:
            tau *= 1.07

        # decay learning rate a little
        if it % 50 == 0:
            lr *= 0.93

        # Periodically evaluate the exact objective and keep the best one
        if it % 10 == 0:
            sc = _fast_objective(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[gd] iter {it:4d}  surrogate={surrogate:.6f}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 4.  Optional fine‑grained random perturbation sweep
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        a = best_seq.copy()
        step_scale = target_sum / (30.0 * n)   # tiny mass amount
        while time.monotonic() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * step_scale
            if a[i] < delta:
                continue
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            # re‑normalise (keeps sum ≈ target_sum)
            s = cand.sum()
            if s > 0.0:
                cand *= (target_sum / s)
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _fast_objective(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                a = cand
                print(f"[perturb] improved → score={best_score:.6f}")
            else:
                # still keep walking with the candidate (helps exploration)
                a = cand

    # ------------------------------------------------------------------
    # 5.  Final accurate evaluation (using the provided wrapper)
    # ------------------------------------------------------------------
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] evaluated with evaluate_sequence → score = {final_score:.6f}")

    return [float(v) for v in best_seq]
