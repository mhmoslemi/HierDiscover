def simplex_projection(v, S):
    """Project v onto the probability simplex scaled by S (nonnegative, sum=S)."""

    # algorithm from Duchi et al. 2008 O(n log n)
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, S):
    """Project onto simplex {x >=0, sum=S}."""
    if S < 0:
        raise ValueError
    # if sum(v) == S and all nonnegative, return v early
    # else compute projection
    # Standard algorithm:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]

"""
Hybrid optimizer for the convolution‑bound problem.

The algorithm proceeds in three stages:
1. Build the analytically good “arcsine” sequence.
2. Scan a few low‑dimensional parametric families
   (power‑law, shifted power‑law, exponential decay) and keep the
   best shape found – this often already beats the arcsine bound.
3. Refine the current best by a projected sub‑gradient descent
   on the objective 2·n·max(conv(a,a))/‖a‖₁².
   The sub‑gradient at the active convolution index k is
       g_i = 2·a_{k-i}  (zero outside the support).
   After a step we project back onto the non‑negative simplex
   (fixed ℓ₁‑norm) using a simple renormalisation.
The whole search stops when the supplied time budget expires, and the
best sequence discovered so far is returned.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Basic utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def build_arcsine_sequence(length: int) -> np.ndarray:
    """Arcsine (inverse‑square‑root) shape a[i] = C(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


def renormalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to the allowed range and scale so that sum(seq) == target_sum."""
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur == 0.0:
        # fall back to a uniform distribution (unlikely to happen)
        seq[:] = target_sum / len(seq)
        return seq
    seq *= target_sum / cur
    return seq


def sequence_score(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(b)/(sum(a)²)."""
    s = float(seq.sum())
    if s < 0.01:
        return np.inf
    b = np.convolve(seq, seq)
    max_b = float(b.max())
    n = seq.shape[0]
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Parameterised families (used for a quick global scan)
# ----------------------------------------------------------------------


def power_law_shape(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """a[i] = 1 / (i+1+shift)**p   (non‑negative, decreasing)."""
    idx = np.arange(n, dtype=np.float64) + 1.0 + shift
    # avoid division by zero – shift >= 0 guarantees idx>0
    return 1.0 / (idx ** p)


def exponential_shape(n: int, r: float) -> np.ndarray:
    """Geometric decay a[i] = r**i   (0<r<1)."""
    return r ** np.arange(n, dtype=np.float64)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_step(a: np.ndarray, max_idx: int, step: float,
                     target_sum: float) -> np.ndarray:
    """
    Perform one sub‑gradient update.
    g_i = 2·a[max_idx‑i] (if the index is in range) else 0.
    Then a ← a – step·g  and project onto the ℓ₁‑simplex.
    """
    n = a.size
    g = np.zeros_like(a)

    if max_idx < n:
        # indices 0 … max_idx receive contributions
        g[: max_idx + 1] = 2.0 * a[max_idx :: -1]   # reversed slice
    else:
        # max_idx ≥ n → only the tail part receives contributions
        d = max_idx - (n - 1)          # first index that gets a non‑zero entry
        # from d … n‑1 we have a[max_idx‑i] = a[(n‑1)‑(i‑d)]
        g[d:] = 2.0 * a[: n - d][::-1]

    cand = a - step * g
    cand = renormalise(cand, target_sum)   # projection onto non‑negative simplex
    return cand


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """Return a list of floats – the best step‑function heights found."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:                     # fallback when the global is absent
        n = 256

    # ------------------------------------------------------------------
    # 1. Start with the analytically derived arcsine sequence
    # ------------------------------------------------------------------
    base_seq = build_arcsine_sequence(n)
    target_sum = float(base_seq.sum())           # ≈ sqrt(2·n)
    cur_seq = renormalise(base_seq, target_sum)
    cur_score = sequence_score(cur_seq)

    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"[init] length={n}, arcsine score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Quick sweep over a few low‑dimensional families
    # ------------------------------------------------------------------
    # (a) Shifted power‑law a_i = 1/(i+1+β)^p
    for p in np.linspace(0.35, 0.80, 10):          # 10 values
        for beta in (0.0, 0.5, 1.0, 2.0, 5.0):
            seq = power_law_shape(n, p, shift=beta)
            seq = renormalise(seq, target_sum)
            sc = sequence_score(seq)
            if sc < best_score:
                best_seq = seq.copy()
                best_score = sc
                cur_seq = seq.copy()
                cur_score = sc
                print(f"[scan] power p={p:.3f}, β={beta:.1f} → score {sc:.6f}")

    # (b) Exponential decay a_i = r^i
    for r in np.linspace(0.90, 0.999, 12):
        seq = exponential_shape(n, r)
        seq = renormalise(seq, target_sum)
        sc = sequence_score(seq)
        if sc < best_score:
            best_seq = seq.copy()
            best_score = sc
            cur_seq = seq.copy()
            cur_score = sc
            print(f"[scan] exponential r={r:.3f} → score {sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient descent from the best‑found start
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)  # a safety slack
    it = 0
    eta0 = 0.5                     # initial step size (will decay)

    # helper to occasionally inject a random perturbation (helps escape plateaus)
    def jitter(seq):
        jitter_vec = np.exp(rng.normal(0.0, 0.06, size=seq.shape))
        cand = seq * jitter_vec
        return renormalise(cand, target_sum)

    last_improve_it = 0

    while time.monotonic() < deadline:
        it += 1

        # --- compute the current maximal convolution index ---
        b = np.convolve(cur_seq, cur_seq)
        max_b = float(b.max())
        max_idx = int(np.argmax(b))

        # --- deterministic sub‑gradient step ---
        step = eta0 / np.sqrt(it)          # decreasing stepsize
        cand = subgradient_step(cur_seq, max_idx, step, target_sum)

        cand_score = sequence_score(cand)

        # accept the new point unconditionally (the step size shrinks)
        cur_seq = cand
        cur_score = cand_score

        # keep track of the best ever seen
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            last_improve_it = it
            print(f"[{it:>6}] new best = {best_score:.6f}")

            # early exit if we have already passed the target bound
            if best_score < 1.503:
                break

        # --- occasional random jitter if stuck for a while ---
        if it % 500 == 0 and (it - last_improve_it) > 2000:
            cur_seq = jitter(best_seq)   # re‑centre the walk around the best
            cur_score = sequence_score(cur_seq)
            # reset iteration counter for the step‑size schedule (optional)
            # it = 0  # not resetting to keep decay smooth

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    return [float(x) for x in best_seq]
