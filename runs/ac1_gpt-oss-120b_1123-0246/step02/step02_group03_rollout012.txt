def project_to_simplex(v, z=1.0):
    # v is 1D numpy array non-negative allowed.
    # Implements algorithm from 'Efficient Projections onto the l1-ball for Learning in High Dimensions'.
    # But we project onto sum = z, not onto L1-norm <= z.
    # ... Actually we can use known algorithm.

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_to_simplex(v, z):

"""
Propose a low‑score step‑function using a soft‑max‑smoothed
max‑autocorrelation objective.  The algorithm is an Adam‑based
projected gradient descent on the smoothed objective, with a linearly
increasing temperature (β) and periodic random multiplicative
perturbations.  After each update the vector is projected onto the
non‑negative simplex Σa = √(2·n) (the sum is fixed, which makes the
evaluation function equal to the maximal convolution value).  The
procedure runs until the allocated time budget expires and returns the
best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: linear convolution using real FFT (fast for many iterations)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via power‑of‑two FFT."""
    n = x.size + y.size - 1
    N = 1 << (n - 1).bit_length()           # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv


# ----------------------------------------------------------------------
# Helper: projection onto the non‑negative simplex {z ≥ 0, Σz = Z}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, Z: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {z ≥ 0, Σz = Z}.
    Implementation of the algorithm from Duchi et al. (2008).
    """
    if Z < 0:
        raise ValueError("Simplex sum must be non‑negative")
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - Z
    ind = np.arange(1, len(v) + 1)
    # Find rho = max {j | u_j - cssv_j / j > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Helper: compute soft‑max gradient of the smoothed max convolution
# ----------------------------------------------------------------------
def _softmax_grad_and_maxb(a: np.ndarray, beta: float):
    """
    Returns (grad, max_b):
        grad – gradient of the soft‑max approximation of max_k (a∗a)_k,
               projected onto the zero‑sum hyper‑plane.
        max_b – the exact maximal autocorrelation value (needed for logging).
    """
    n = a.size
    # exact convolution
    b = _fft_convolve(a, a)
    max_b = b.max()

    # soft‑max weights over all lags (stable because we subtract max_b)
    exp_term = np.exp(beta * (b - max_b))
    w = exp_term / exp_term.sum()

    #   ∂/∂a_i = 2 Σ_k w_k a_{k-i}
    w_rev = w[::-1]
    conv_rev = _fft_convolve(a, w_rev)                # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)                  # indices n‑1 … 2n‑2
    grad = 2.0 * conv_rev[idx]

    # project onto Σ a_i = 0 (zero‑sum) hyper‑plane
    grad -= grad.mean()
    return grad, max_b


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 990.0) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The optimiser runs for up to `budget_s`
    seconds and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # safety margin

    # --------------------------------------------------------------
    # 1. Choose starting length and initialise the vector
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        # keep a moderate size for speed (max 12000 entries)
        if base.size > 12000:
            lo = (base.size - 12000) // 2
            base = base[lo:lo + 12000]
    else:
        # uniform fallback of moderate length
        base = np.ones(4000, dtype=np.float64)

    n = base.size
    a = np.clip(base, 0.0, 1000.0)

    # Normalise sum to √(2 n).  Under this normalisation the evaluation
    # equals the maximal convolution entry.
    target_sum = np.sqrt(2.0 * n)
    if a.sum() == 0.0:
        a[:] = 1.0
    a *= (target_sum / a.sum())
    a = _project_to_simplex(a, target_sum)

    # --------------------------------------------------------------
    # 2. Book‑keeping
    # --------------------------------------------------------------
    best_seq = a.copy()
    _, best_maxb = _softmax_grad_and_maxb(a, beta=1.0)
    best_score = float(best_maxb)          # with normalisation, score = max_b
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule parameters
    beta_start = 0.5
    beta_end = 30.0               # sharper soft‑max than the previous run
    init_lr = target_sum * 0.02   # base step size (≈2 % of total mass)
    iteration = 0
    last_improve = 0

    # --------------------------------------------------------------
    # 3. Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / budget_s                     # 0 → 1 over the run

        # Temperature schedule (smoothly increase β)
        beta = beta_start + (beta_end - beta_start) * (frac ** 0.8)

        # Gradient of the smoothed objective
        grad, cur_maxb = _softmax_grad_and_maxb(a, beta)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        t = iteration + 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Direction = - (bias‑corrected first moment) / sqrt(second moment)
        direction = -m_hat / (np.sqrt(v_hat) + eps)

        # Decaying learning rate – 1/√t works well for this scale‑free problem
        lr = init_lr / np.sqrt(t)

        # Propose a new candidate and project onto the simplex
        cand = a + lr * direction
        cand = _project_to_simplex(cand, target_sum)

        # Enforce the hard upper bound (rarely active, but kept for safety)
        cand = np.clip(cand, 0.0, 1000.0)

        # Evaluate the true objective (maximal convolution entry)
        conv = _fft_convolve(cand, cand)
        cand_maxb = conv.max()
        cand_score = float(cand_maxb)

        # Always keep the best ever seen
        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score
            last_improve = iteration
            print(f"[{iteration:7d}] new best  score = {best_score:.6f}")

        # Walk continues regardless of immediate improvement
        a = cand

        # --------------------------------------------------------------
        # 4. Random multiplicative perturbations (every few hundred steps)
        # --------------------------------------------------------------
        if iteration % 250 == 0 and iteration > 0:
            perturb = np.exp(rng.normal(0.0, 0.12, size=n))
            pert = a * perturb
            pert = _project_to_simplex(pert, target_sum)
            pert = np.clip(pert, 0.0, 1000.0)
            pert_maxb = _fft_convolve(pert, pert).max()
            if pert_maxb < best_score:
                best_seq = pert.copy()
                best_score = float(pert_maxb)
                a = pert
                last_improve = iteration
                print(f"[{iteration:7d}] perturb improved score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 5. Stagnation restart (if no improvement for a while)
        # --------------------------------------------------------------
        if iteration - last_improve > 4000:
            # fresh random Dirichlet‑like start (concentration < 1 gives sparsity)
            a = rng.random(n) ** 2          # bias towards smaller numbers
            a = _project_to_simplex(a, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"[{iteration:7d}] restart due to stagnation (new random start)")

        iteration += 1

    # --------------------------------------------------------------
    # 6. Return the best sequence found
    # --------------------------------------------------------------
    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    return [float(v) for v in best_seq]
