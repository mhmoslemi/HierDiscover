    rng = np.random.default_rng(seed)
    n = len(globals().get("height_sequence_1", []))
    if n == 0:

    best_seq = cur_seq.copy()
    best_val = cur_val

    deadline = time.monotonic() + budget_s * 0.95
    iteration = 0
    steps_since_improve = 0

    while time.monotonic() < deadline:
        iteration += 1
        # get FFT and conv
        A = np.fft.rfft(cur_seq, L)
        conv = np.fft.irfft(A * A, L)[:2*n-1]
        max_b = conv.max()

        tol = max_b * (1 - 1e-12)
        mask = (conv >= tol).astype(np.float64)
        mask_sum = mask.sum()
        if mask_sum == 0:

        else:

            s = cand.sum()

                cur_val = cand_val
                improved = True
                if cand_val < best_val - 1e-12:

"""
Projected gradient descent + occasional random perturbations.

The algorithm works on a fixed‐length non‑negative vector `a` whose
ℓ₁‑norm is kept constant (≈√(2 n)).  For a given shape the objective is
simply the maximum of the self‑convolution `b = a * a`,
because 2·n·max(b)/(∑a)² = max(b) when ∑a = √(2 n).

* The gradient of the (smoothed) maximum is obtained analytically:
    ∂/∂aᵢ  Σₖ wₖ·bₖ = 2·(w ⊛ a)ᵢ,
  where `w` are normalized weights on the convolution positions
  (soft‑max of `b` or a hard mask on the maximal entries) and `⊛`
  denotes the discrete cross‑correlation.
  The cross‑correlation is computed with a single FFT, so each iteration
  costs only O(n log n).

* A simple projected gradient step `a ← a – η·grad` is performed,
  followed by clipping to non‑negative values and renormalisation to the
  target ℓ₁‑norm.  A tiny back‑tracking line search guarantees monotone
  improvement.

* When the gradient step stalls, a cheap stochastic “block scaling” move
  (as used in the reference code) is applied to escape flat regions.
  After a long period without improvement the current state is replaced
  by a fresh random Dirichlet vector – a cheap multi‑restart strategy.

The routine runs until the allotted time expires (≈95 % of the budget)
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def renormalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to the allowed range, then scale to the given ℓ₁‑norm."""
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur == 0.0:
        return seq
    seq *= target_sum / cur
    return seq


def build_arcsine_sequence(length: int) -> np.ndarray:
    """Analytical arcsine shape a[i] = C(2i,i)/4**i (≈1/√(π i))."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


def block_perturbation(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                      # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return renormalise(cand, target_sum)


def max_convolution(seq: np.ndarray, L: int) -> float:
    """Return max(b) where b = seq * seq (computed via FFT of length L)."""
    A = np.fft.rfft(seq, L)
    b = np.fft.irfft(A * A, L)
    n = seq.shape[0]
    return float(b[: 2 * n - 1].max())


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of floats – the best step‑function heights found within
    ``budget_s`` seconds.  The method:
        1. Starts from the analytically optimal arcsine shape (ℓ₁‑norm √(2 n)).
        2. Repeatedly performs a projected gradient step that reduces the
           (soft‑max) maximum of the self‑convolution, using FFT‑based
           correlation for the gradient.
        3. If the gradient step cannot improve the objective, a cheap
           stochastic block‑perturbation is tried; after long stagnation a
           full random restart (Dirichlet) is performed.
        4. The best sequence seen (according to the exact objective
           2·n·max(b)/(∑a)²) is remembered and finally returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    height_seq = globals().get("height_sequence_1", [])
    n = int(len(height_seq))
    if n == 0:
        n = 256

    # target ℓ₁‑norm (kept fixed throughout the search)
    target_sum = float(np.sqrt(2.0 * n))

    # FFT length sufficient for both convolution (2n‑1) and correlation
    # (2n‑1 + n‑1 = 3n‑2).  A power of two speeds up the transforms.
    L = next_pow2(3 * n - 1)

    # ------------------------------------------------------------------
    # initialise with the arcsine sequence
    # ------------------------------------------------------------------
    cur_seq = renormalise(build_arcsine_sequence(n), target_sum)
    cur_val = max_convolution(cur_seq, L)          # equals the objective
    best_seq = cur_seq.copy()
    best_val = cur_val

    print(f"[init] n={n}, initial objective={best_val:.6f}")

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, budget_s * 0.95)
    it = 0
    no_improve = 0
    momentum = np.zeros_like(cur_seq)               # for a simple momentum term
    mom_coeff = 0.9

    while time.monotonic() < deadline:
        it += 1

        # ---- FFT of the current sequence --------------------------------
        A = np.fft.rfft(cur_seq, L)
        b_full = np.fft.irfft(A * A, L)[: 2 * n - 1]
        max_b = float(b_full.max())

        # ---- build a weight vector on the convolution positions ----------
        # use a hard mask on values that are (almost) maximal
        tol = max_b * (1.0 - 1e-12)
        mask = (b_full >= tol).astype(np.float64)
        mask_sum = mask.sum()
        if mask_sum == 0.0:
            # fallback – uniform weights (should never happen)
            mask = np.full_like(mask, 1.0 / mask.size)
        else:
            mask /= mask_sum

        # ---- gradient = 2·(mask ⊛ cur_seq) --------------------------------
        W = np.fft.rfft(mask, L)
        corr_full = np.fft.irfft(W * np.conj(A), L)[: mask.size + n - 1]
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]    # length n

        # ---- simple momentum ---------------------------------------------
        momentum = mom_coeff * momentum + (1.0 - mom_coeff) * grad

        # ---- back‑tracking line search -----------------------------------
        eta = 0.08                                   # initial step size
        improved = False
        for _ in range(6):
            cand = cur_seq - eta * momentum
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                break
            cand = cand * (target_sum / s)

            cand_val = max_convolution(cand, L)
            if cand_val < cur_val - 1e-12:
                # accept the candidate
                cur_seq = cand
                cur_val = cand_val
                improved = True
                break
            eta *= 0.5

        if improved:
            no_improve = 0
            if cur_val < best_val - 1e-12:
                best_seq = cur_seq.copy()
                best_val = cur_val
                print(f"[{it:>6}] new best = {best_val:.6f}")
                # early exit if we beat the target comfortably
                if best_val < 1.503:
                    break
            continue

        # ---- no improvement – try a cheap stochastic move ---------------
        no_improve += 1
        cand = block_perturbation(cur_seq, rng, target_sum)
        cand_val = max_convolution(cand, L)
        if cand_val < cur_val:
            cur_seq = cand
            cur_val = cand_val
            no_improve = 0
            if cand_val < best_val - 1e-12:
                best_seq = cand.copy()
                best_val = cand_val
                print(f"[{it:>6}] new best (after perturb) = {best_val:.6f}")
            continue

        # ---- long stagnation – full random restart ----------------------
        if no_improve > 5000:
            cur_seq = renormalise(
                rng.dirichlet(np.ones(n), size=1)[0], target_sum
            )
            cur_val = max_convolution(cur_seq, L)
            momentum.fill(0.0)
            no_improve = 0
            print(f"[{it:>6}] restart (objective={cur_val:.6f})")

    print(f"[final] best objective {best_val:.6f} after {it} iterations")
    return [float(x) for x in best_seq]
