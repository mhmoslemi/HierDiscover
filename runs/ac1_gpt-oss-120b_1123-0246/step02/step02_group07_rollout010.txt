def compute_obj(seq):
    seq = np.asarray(seq, dtype=np.float64)
    seq = np.clip(seq, 0.0, 1000.0)  # ensure constraints
    n = seq.shape[0]
    # Use next power of two for length for speed (or just n*2)
    m = 1 << ((2*n -1).bit_length())
    # zero-pad
    fft_seq = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft_seq * fft_seq, n=m)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return 2 * n * max_b / (sum_a ** 2)

"""
Propose a step‑function sequence that beats the arcsine baseline.

The algorithm consists of three stages:

1. **Base families** – Build three simple families with sum=1:
   * the exact (unnormalised) arcsine sequence,
   * its reversal,
   * the uniform sequence.
   All three are normalised to have ℓ₁–norm = 1 so that any convex
   combination also has ℓ₁–norm = 1 (scaling does not change the objective).

2. **Extensive parameter sweep** – Explore
   * a pure convex combination of forward / reversed arcsine,
   * a three‑way mixture (forward + reversed + uniform),
   * after each mixture optionally apply a power transform `a → a**p`
     (renormalised to ℓ₁=1) and a short moving‑average smoothing.
   The best candidate of this cheap search is kept.

3. **Fast gradient‑guided local search** – Using the current best sequence,
   repeatedly identify the convolution index attaining the maximal value,
   compute the sub‑gradient of that entry (2·a_{k−i}) and perform a
   small projected gradient step (clipping to non‑negative values).
   Accept the move only if the true objective improves.
   Occasionally a random block perturbation is tried to escape plateaus.

All heavy work is done with FFT‑based convolution (`O(n log n)`) which
allows thousands of evaluations well within the 1000 s budget.
The final sequence is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Building the analytical arcsine sequence (inverse‑square‑root law)
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a length‑`length` array a[i] = C(2i,i)/4**i (exact up to fp noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    return a


# ----------------------------------------------------------------------
# 2.  Fast objective (same formula as evaluate_sequence, but O(n log n))
# ----------------------------------------------------------------------
def fast_objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2) with convolution via FFT."""
    s = np.clip(seq, 0.0, 1000.0)
    n = s.size
    total = s.sum()
    if total < 0.01:
        return np.inf
    # length for a linear convolution (next power of two for speed)
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(s, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (total * total))


# ----------------------------------------------------------------------
# 3.  Simple stochastic perturbations (kept from the reference code)
# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by exp(N(0,σ))."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# 4.  Directed gradient step on the current maximal convolution entry
# ----------------------------------------------------------------------
def gradient_step(seq: np.ndarray, step_factor: float = 0.1) -> np.ndarray:
    """
    Perform a single projected gradient step aimed at reducing the
    largest element of the auto‑convolution.
    """
    n = seq.size
    # FFT convolution (fast, exact up to floating point noise)
    m = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    k = int(np.argmax(conv))                     # index of the maximal entry

    # sub‑gradient of conv[k] w.r.t. each a[i] is 2·a[k−i] (if the partner exists)
    grad = np.zeros_like(seq)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    i = np.arange(lo, hi + 1)
    j = k - i
    grad[i] = 2.0 * seq[j]

    # step size – chosen relative to the current maximal value
    norm = np.linalg.norm(grad) + 1e-12
    eta = step_factor * conv[k] / norm
    cand = seq - eta * grad
    cand = np.clip(cand, 0.0, 1000.0)
    return cand


# ----------------------------------------------------------------------
# 5.  Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of non‑negative heights – the best step‑function found
    within the allotted time.  The routine starts from the analytical arcsine,
    performs a dense combinatorial sweep (mixing forward / reversed arcsine
    and a uniform baseline, plus optional power‐law and smoothing), then
    refines the best candidate with a fast gradient‑guided local search.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (length of the reference solution, if any)
    # --------------------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256  # a harmless fallback

    # --------------------------------------------------------------
    # Pre‑compute the three elementary families (all ℓ₁‑normalised)
    # --------------------------------------------------------------
    base_arc = build_arcsine_sequence(n)
    base_arc = base_arc / np.sum(base_arc)          # ℓ₁ = 1
    base_rev = base_arc[::-1].copy()                # reversed arcsine
    uniform = np.full(n, 1.0 / n)                   # uniform, already ℓ₁ = 1

    # Start with the pure arcsine (objective ≈ π/2)
    best = base_arc.copy()
    best_score = fast_objective(best)
    print(f"[init] length={n}, arcsine score={best_score:.6f}")

    # -----------------------------------------------------------------
    # Stage 1 – exhaustive mixing of the three families
    # -----------------------------------------------------------------
    # 1. Two‑way mix (forward / reversed) – sweep whole [0,1] interval
    for alpha in np.linspace(0.0, 1.0, 101):
        cand = (1.0 - alpha) * base_arc + alpha * base_rev
        sc = fast_objective(cand)
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[mix2] α={alpha:.3f} → {sc:.6f}")

    # 2. Three‑way convex combinations (coarse grid, then refined)
    steps = 21
    for w1 in np.linspace(0.0, 1.0, steps):
        for w2 in np.linspace(0.0, 1.0 - w1, steps):
            w3 = 1.0 - w1 - w2
            cand = w1 * base_arc + w2 * base_rev + w3 * uniform
            sc = fast_objective(cand)

                best, best_score = cand, sc
                print(f"[mix3] w1={w1:.2f}, w2={w2:.2f}, w3={w3:.2f} → {sc:.6f}")

    # 3. Power‑law deformations after mixing (p in [0.5, 1.5])
    for alpha in np.linspace(0.0, 1.0, 41):
        mixed = (1.0 - alpha) * base_arc + alpha * base_rev
        for p in np.linspace(0.5, 1.5, 11):
            cand = np.power(mixed, p)
            # renormalise because power changes the ℓ₁‑norm
            cand = cand / np.sum(cand)
            sc = fast_objective(cand)
            if sc < best_score:
                best, best_score = cand, sc
                print(f"[pow] α={alpha:.2f}, p={p:.2f} → {sc:.6f}")

    # 4. Light moving‑average smoothing (helps a bit for some mixes)
    for w in [3, 5, 7, 11, 15]:
        kernel = np.ones(w) / w
        cand = np.convolve(best, kernel, mode='same')
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand / np.sum(cand)                      # keep ℓ₁ = 1
        sc = fast_objective(cand)
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[smooth] width={w} → {sc:.6f}")

    # -----------------------------------------------------------------
    # Stage 2 – fast gradient‑guided local refinement
    # -----------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # keep a small slack
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # 70 % gradient step, 30 % random block move (to escape shallow plateaus)
        if rng.random() < 0.7:
            cand = gradient_step(best, step_factor=0.12)
        else:
            cand = block_perturbation(best, rng)

        sc = fast_objective(cand)
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc
            print(f"[{it:>6}] new best = {best_score:.6f}")

            if best_score < 1.503:
                print("[stop] target reached")
                break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # The harness expects a plain list of floats
    return [float(x) for x in best]
