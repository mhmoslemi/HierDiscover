assistantfinal**Solution Overview**

The evaluator is *scale‑invariant*: only the *shape* of the step function matters.  
If we normalise the sequence to  

\[
\sum_i a_i = \sqrt{2\,n},
\]

the evaluation reduces to the maximal entry of the self‑convolution

\[
M = \max_k (a*a)_k .
\]

Thus minimising the score is equivalent to minimising `M`.

The algorithm below is a deterministic variant of the “alternating linear programme”
described in the statement.  

* For a fixed shape `f` we compute its current worst convolution value `M = max(f*f)`.  
* Keeping `M` fixed we solve the linear programme  

\[
\max\; \mathbf{1}^T g \qquad
\text{s.t.}\;\; f * g \le M,\; g\ge 0 .
\]

  The variables are the new heights `g`. Because `f*g` is linear in `g` this
  is a genuine linear programme (≈ 2 n constraints, n variables).

* The solution `g` is rescaled to the required normalisation  
  (`sum(g)=√(2 n)`) and mixed with the old shape:

\[
f_{\text{new}} = (1-t)f + t\,g ,
\]

  where the mixing factor `t∈[0,1]` is chosen (by a cheap line‑search) to minimise
  the maximal convolution of the mixture.

* If the new maximal convolution `M_new` is smaller we accept the update,
  otherwise we keep the old shape and try a small random perturbation.

The procedure is repeated until the time budget expires (or the target
value `1.503` is reached).  The whole loop is fast: for `n=256` the linear
programme has 256 variables and 511 constraints and solves in a few
milliseconds with the ECOS solver.

The final sequence is returned as a plain list of floats; the evaluator’s
clipping (`0 ≤ x ≤ 1000`) guarantees validity.

```python
"""
Projected alternating‑LP optimiser for the self‑convolution bound.

The routine starts from the analytically optimal “arcsine” (inverse‑square‑root)
shape, normalises it to sum = sqrt(2·n) and repeatedly:
  1. solves a linear programme  max 1ᵀg   s.t.   f * g ≤ M, g ≥ 0,
  2. rescales the LP solution to the required L1‑norm,
  3. mixes the old and the new shape, choosing a step size t that minimises
     the maximal entry of the resulting self‑convolution,
  4. accepts the mix if it strictly improves the bound.

The loop runs for at most *budget_s* seconds (default 950 s, leaving a safety
margin) and returns the best sequence found.  The evaluation score equals the
maximal convolution value because the sequence is always normalised.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Inverse‑square‑root (arcsine) sequence a[i] = C(2i,i)/4**i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def normalise_to_target_sum(seq: np.ndarray, target: float) -> np.ndarray:
    """Scale `seq` so that its L1‑norm equals `target`."""
    s = np.sum(seq)
    if s == 0:
        return seq
    return seq * (target / s)


def convolution_max(seq: np.ndarray) -> float:
    """Return max_k (seq * seq)[k] (self‑convolution)."""
    conv = np.convolve(seq, seq)
    return float(np.max(conv))


def build_toeplitz_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A of shape (2n‑1, n) such that A @ g = f * g.
    Entry A[k, j] = f[k-j] when 0 ≤ k-j < n, otherwise 0.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        # f[i] contributes to rows i ... i+n-1 in column j
        A[i:i + n, :] += np.diag(f[i] * np.ones(n), k=i)[:, :rows]
    # The above loop is simple but a little wasteful; for modest n it
    # is perfectly fine (O(n²) work).
    # Simpler (and fast enough) explicit double loop:
    # for i in range(n):
    #     for j in range(n):
    #         A[i + j, j] = f[i]
    return A


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear programme:
        maximise   1ᵀ g
        subject to  (f * g) ≤ M   (elementwise)
                    g ≥ 0
    Returns the optimal g as a NumPy array, or None on failure.
    """
    n = f.size
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    # explicit construction – cheap for n ≤ 512
    for i in range(n):
        for j in range(n):
            A[i + j, j] = f[i]

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M + 1e-12]          # tiny tolerance
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver="ECOS", warm_start=True, verbose=False, max_iters=5000)
    except Exception:          # solver failure
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    return np.asarray(g.value).flatten()


def optimal_mixing_factor(f: np.ndarray, g: np.ndarray) -> float:
    """
    Given two normalised shapes f and g, find a mixing t ∈ [0,1]
    that minimises the maximal entry of ((1‑t)f + t g) * ((1‑t)f + t g).
    A cheap grid search (step 0.05) followed by a local refinement is used.
    """
    conv_f = np.convolve(f, f)
    conv_g = np.convolve(g, g)
    conv_fg = np.convolve(f, g)

    def M_of_t(t: float) -> float:
        # (1‑t)²·conv_f + 2·t·(1‑t)·conv_fg + t²·conv_g
        return float(np.max((1 - t) ** 2 * conv_f +
                            2 * t * (1 - t) * conv_fg +
                             t ** 2 * conv_g))

    # coarse grid
    ts = np.linspace(0.0, 1.0, 21)
    Ms = [M_of_t(t) for t in ts]
    t_best = ts[int(np.argmin(Ms))]
    # refine around the best point
    lo = max(0.0, t_best - 0.05)
    hi = min(1.0, t_best + 0.05)
    ts_ref = np.linspace(lo, hi, 41)
    Ms_ref = [M_of_t(t) for t in ts_ref]
    t_best = ts_ref[int(np.argmin(Ms_ref))]
    return float(t_best)

        it += 1

        # --------------------------------------------------------------
        # 3.1 Solve LP to obtain a candidate g (maximise L1 under f*g ≤ M)
        # --------------------------------------------------------------
        g_candidate = solve_lp_max_sum(cur_f, cur_M)
        if g_candidate is None:
            # fall back to a random small move
            cur_f = cur_f * (1.0 + 0.01 * (rng.random(n) - 0.5))
            cur_f = np.clip(cur_f, 0.0, 1000.0)
            cur_f = normalise_to_target_sum(cur_f, target_sum)
            cur_M = convolution_max(cur_f)
            continue

        # --------------------------------------------------------------
        # 3.2 Rescale g to the target L1‑norm
        # --------------------------------------------------------------
        sum_g = np.sum(g_candidate)
        if sum_g <= 0:
            break
        g = g_candidate * (target_sum / sum_g)

        # --------------------------------------------------------------
        # 3.3 Choose a mixing factor t that minimises the resulting max‑conv.
        # --------------------------------------------------------------
        t = optimal_mixing_factor(cur_f, g)

        # --------------------------------------------------------------
        # 3.4 Form the new shape and evaluate its maximal convolution.
        # --------------------------------------------------------------
        new_f = (1.0 - t) * cur_f + t * g
        new_f = np.clip(new_f, 0.0, 1000.0)
        new_f = normalise_to_target_sum(new_f, target_sum)

        new_M = convolution_max(new_f)

        # --------------------------------------------------------------
        # 3.5 Acceptance test
        # --------------------------------------------------------------
        if new_M < cur_M - 1e-7:          # strict improvement
            cur_f = new_f
            cur_M = new_M
            no_improve = 0
            if new_M < best_score:
                best_seq = new_f.copy()
                best_score = new_M
                print(f"[{it:>6}] new best M={best_score:.6f}")
                if best_score < 1.503:
                    print("[early] target reached")
                    break
        else:
            # No progress – perform a tiny random perturbation to
            # escape a possible plateau.
            no_improve += 1
            if no_improve % 10 == 0:
                cur_f = cur_f * (1.0 + 0.02 * (rng.random(n) - 0.5))
                cur_f = np.clip(cur_f, 0.0, 1000.0)
                cur_f = normalise_to_target_sum(cur_f, target_sum)
                cur_M = convolution_max(cur_f)

        # occasional global reset to the pure arcsine shape (helps restart)
        if it % 3000 == 0 and rng.random() < 0.3:
            cur_f = build_arcsine_sequence(n)
            cur_f = normalise_to_target_sum(cur_f, target_sum)
            cur_M = convolution_max(cur_f)

    # ------------------------------------------------------------------
    # 4. Final reporting
    # ------------------------------------------------------------------
    print(f"[final] best score (M)={best_score:.6f} after {it} iterations")
    # The evaluator is scale‑invariant; we return the normalised shape.
    return [float(x) for x in best_seq]
```
