def propose_candidate(seed=1234, budget_s=1000.0):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 0.5)  # margin

    # choose n
    try:
        prev_arr = np.asarray(height_sequence_1, dtype=float)
        n0 = prev_arr.size
    except Exception:
        n0 = 6000
    # maybe adjust n to be between 4000 and 12000
    n = int(np.clip(n0, 4000, 12000))

    # initialize a
    if prev_arr.size == n:
        a = prev_arr.copy()
    else:
        # interpolate if needed
        old_x = np.linspace(0,1,prev_arr.size)
        new_x = np.linspace(0,1,n)
        a = np.interp(new_x, old_x, prev_arr)

    # ensure nonnegatives
    a = np.clip(a, 0.0, 1000.0)
    # normalize sum to 1
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= s

    # precompute fft length
    fft_len = _next_pow2(2*n - 1)  # must have >= len conv
    # compute initial conv
    cur_max = _max_convolution(a, fft_len)
    cur_score = 2.0 * n * cur_max
    best_a = a.copy()
    best_max = cur_max
    best_score = cur_score

    print(f"initial: n={n}, score={best_score:.6f}, max_b={best_max:.8f}")

    # simulated annealing parameters
    total_iters = 0
    max_iters = 2000000  # large bound; time check will stop earlier
    T0 = cur_max * 0.5   # initial temperature
    Tf = cur_max * 0.001  # final temperature

    while time.monotonic() < deadline and total_iters < max_iters:
        # schedule temperature
        t = total_iters / max_iters
        T = T0 * (Tf/T0)**t  # exponential decay

        # generate move
        # pick source index weighted by a
        i = rng.choice(n, p=a)  # biased to larger mass
        # pick target index uniform
        j = rng.integers(n)
        if i == j:
            continue

        # choose fraction to move
        # amount = epsilon * a[i]; epsilon random in [0, max_frac]
        max_frac = 0.2
        eps = rng.random() * max_frac
        amount = eps * a[i]
        if amount <= 0:
            continue

        # propose new a
        cand = a.copy()
        cand[i] -= amount
        cand[j] += amount

        # ensure nonnegative (cand[i] >= 0)
        # (by construction ok)
        # compute new max
        cand_max = _max_convolution(cand, fft_len)

        # acceptance
        if cand_max < cur_max:
            a = cand
            cur_max = cand_max
        else:
            # probability
            if rng.random() < np.exp(-(cand_max - cur_max) / T):
                a = cand
                cur_max = cand_max

        # update best
        if cur_max < best_max:
            best_a = a.copy()
            best_max = cur_max
            best_score = 2.0 * n * best_max

            a = a * perturb

Simulated‑annealing + targeted moves for the step‑function minimisation.

The algorithm starts from the best known construction (global variable
``height_sequence_1``) if it exists, interpolates it to a moderately larger
length (≈10 000) and then performs a long simulated‑annealing run.  In each

index k where the convolution attains its maximum is identified, and the

fraction of its mass to a random index.  The move is accepted if it lowers

best sequence ever seen is remembered and returned after a final exact

import numpy as np

# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a using real FFTs.
    Returns an array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)
    return conv_full[: 2 * a.size - 1]


def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the autocorrelation a*a."""
    return _self_convolution_fft(a, fft_len).max()


def _normalise_one(a: np.ndarray) -> np.ndarray:
    """Scale a to have L1‑norm = 1 (leaving zero vector unchanged)."""
    s = a.sum()
    if s == 0.0:
        return a
    return a / s

        # --- keep the global best -------------------------------------------
        if cur_max < best_max:
            best_a = a.copy()
            best_max = cur_max
            best_score = 2.0 * n * best_max
            print(f"[{total_iters:7d}] new best score={best_score:.6f} (max={best_max:.8e})")

        # --- occasional global perturbation to escape plateaus ---------------
        if total_iters % 2000 == 0 and total_iters > 0:
            # multiplicative jitter on all coordinates
            jitter = np.exp(rng.normal(0.0, 0.015, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, 1000.0)
            a = _normalise_one(a)
            # recompute the peak after perturbation
            cur_max = _max_convolution(a, fft_len)

        total_iters += 1

    # ------------------------------------------------------------------
    # 6. Final exact evaluation of the best candidate
    # ------------------------------------------------------------------
    final_score = float(evaluate_sequence(best_a.tolist()))
    print(f"[final] best exact score = {final_score:.6f}")

    # Return the sequence as a plain Python list of floats
    return [float(v) for v in best_a]
