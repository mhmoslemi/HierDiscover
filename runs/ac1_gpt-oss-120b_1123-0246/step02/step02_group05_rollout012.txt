"""
Hybrid LP‑guided search for the step‑function convolution bound.

The algorithm starts from the analytically optimal arcsine sequence,
then repeatedly solves a small linear program that maximises the ℓ₁‑norm
of a candidate `g` under the *current* convolution bound `M = max(f*f)`.
The LP uses only the most critical convolution positions (the K
largest entries of `f*f`) as constraints, which keeps the problem
tiny and fast.  The candidate `g` is normalised to the original ℓ₁‑norm
and mixed with the current sequence `f` using a convex combination;
the mix is accepted only if it lowers the exact objective
`2·n·max(b)/(sum(a)²)`.  Occasionally stochastic moves
(block‑scaling, mass transfers and symmetrisation) are applied to
escape local minima.

The loop runs until the supplied time budget expires or the target
value 1.503 is reached.  The best sequence seen is returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# Try importing SciPy – optional, fall back to pure stochastic search.
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    SCIPY_AVAILABLE = False

# Helper: arcsine sequence (central binomial coefficients)
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return the discrete arcsine sequence (binom(2i,i)/4**i)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        # recurrence a[i] = a[i‑1] * ((2i‑1)(2i)) / (4·i²)
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Scaling utilities
# ----------------------------------------------------------------------
def renormalise(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale `seq` so that its ℓ₁‑norm equals `target_sum`."""
    seq = np.clip(seq, 0.0, 1000.0)
    cur = seq.sum()
    if cur == 0.0:
        return seq
    seq *= target_sum / cur
    return seq


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence but without safety checks)
# ----------------------------------------------------------------------
def sequence_score(seq: np.ndarray) -> float:
    """Return 2·n·max(b)/(sum(a)²) for the given sequence."""
    s = float(seq.sum())
    if s < 0.01:
        return float('inf')
    b = np.convolve(seq, seq)
    max_b = float(b.max())
    n = seq.shape[0]
    return 2.0 * n * max_b / (s * s)

# ----------------------------------------------------------------------
def block_perturbation(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:start+width] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return renormalise(cand, target_sum)


def transfer_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    n = base.size
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n - 1))
    if j >= i:
        j += 1
    frac = rng.random() * 0.30
    delta = frac * base[i]
    if delta == 0.0:
        return base.copy()
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return renormalise(cand, target_sum)


def early_to_late_move(base: np.ndarray, rng: np.random.Generator,
                       target_sum: float) -> np.ndarray:
    n = base.size
    early_cut = max(1, n // 10)
    late_start = max(early_cut, n - early_cut)

def gradient_move(base: np.ndarray, rng: np.random.Generator,
                  target_sum: float) -> np.ndarray:
    """Move mass from the index contributing most to the current max."""
    n = base.size
    b = np.convolve(base, base)
    max_b = b.max()
    tol = max_b * (1.0 - 1e-12)
    mask = (b >= tol).astype(np.float64)
    contrib_conv = np.convolve(base, mask)
    contrib = base * contrib_conv[:n]
    heavy_idx = int(np.argmax(contrib))
    light_candidates = np.where(contrib < (contrib.mean()))[0]
    if light_candidates.size == 0:
        light_idx = (heavy_idx + 1) % n
    else:
        light_candidates = light_candidates[light_candidates != heavy_idx]

        return base.copy()
    cand = base.copy()
    cand[heavy_idx] -= delta
    cand[light_idx] += delta
    return renormalise(cand, target_sum)


# ----------------------------------------------------------------------
# Core LP‑based move (requires SciPy)
# ----------------------------------------------------------------------
def lp_max_sum_move(f: np.ndarray, M: float, target_sum: float,
                    rng: np.random.Generator,
                    top_k: int = 20) -> np.ndarray:
    """
    Solve a small linear program:
        maximise   sum(g_i)
        subject to  (f * g)_k <= M   for the `top_k` largest convolution points.
                    g_i >= 0
    Return the normalised solution g (sum = target_sum).  If the LP fails,
    fall back to the current sequence.
    """
    n = f.size
    # convolution f*f already computed externally, we only need the indices
    # of the largest entries.
    # Use a fixed number of constraints `top_k` (or all if fewer).
    # `top_k` should be small enough to keep the LP cheap.
    # The returned vector g has the same length as f.
    if not SCIPY_AVAILABLE:
        return f.copy()   # fallback

    # Build active constraint rows: k indices with largest (f*f) values.
    # The caller will pass the current max value M; we need the positions.
    # Since we do not have the full convolution vector here, we compute it.
    b = np.convolve(f, f)
    # pick the `top_k` distinct indices with largest values
    if top_k >= len(b):
        active = np.arange(len(b))
    else:
        # np.argpartition gives unsorted top K
        idx_part = np.argpartition(-b, top_k - 1)[:top_k]
        # sort them for a more stable matrix (optional)
        active = np.sort(idx_part)

    rows = []
    cols = []
    data = []

    for row_idx, k in enumerate(active):
        # col range where f_{k-i} exists (i = col)
        col_start = max(0, k - (n - 1))
        col_end = min(k, n - 1)
        # columns involved in this constraint
        cols_row = np.arange(col_start, col_end + 1, dtype=np.int32)
        rows_row = np.full(cols_row.shape, row_idx, dtype=np.int32)
        # coefficient = f_{k - col}
        data_row = f[k - cols_row]
        rows.append(rows_row)
        cols.append(cols_row)
        data.append(data_row)

    # concatenate all rows
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)

    A_ub = csr_matrix((data, (rows, cols)), shape=(len(active), n))
    b_ub = np.full(len(active), M, dtype=np.float64)

    # maximise sum(g) == minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    # Bounds g_i >= 0 (no explicit upper bound)
    bounds = [(0.0, None)] * n

    # Solve LP with HiGHS (fast for sparse problems)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})

    if not res.success or np.any(np.isnan(res.x)):
        # fallback to original
        return f.copy()

    g0 = np.asarray(res.x, dtype=np.float64)

    # Normalise to the required ℓ₁‑norm.
    s = g0.sum()
    if s <= 0.0:
        return f.copy()
    g = g0 * (target_sum / s)
    return g

def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of non‑negative heights (step‑function) that minimises
    `evaluate_sequence`.  The search runs for at most `budget_s` seconds.
    """
    rng = np.random.default_rng(seed)

    # Determine problem size – fall back to 256 if unknown.
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256

    # Target ℓ₁‑norm.  Any positive value works because the objective is
    # scale‑invariant; we use sqrt(2·n) to stay close to the analytic
    # normalisation used in the literature.
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1. Initial sequence – either the provided construction or the arcsine.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and len(height_sequence_1) == n:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        base_seq = build_arcsine_sequence(n)

    cur_seq = renormalise(base_seq, target_sum)
    cur_score = sequence_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Search loop
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    it = 0
    # Adaptive step size for convex combination with LP‑candidate.
    mix_t = 0.4
    while time.monotonic() < deadline:
        it += 1

        # Occasionally (5 % chance) apply a cheap stochastic move
        if rng.random() < 0.05:
            # pick a random move among the simple ones
            if rng.random() < 0.5:
                cand = block_perturbation(cur_seq, rng, target_sum)
            else:
                cand = transfer_move(cur_seq, rng, target_sum)
            cand_score = sequence_score(cand)
            if cand_score < cur_score:
                cur_seq, cur_score = cand, cand_score
                if cand_score < best_score:
                    best_seq, best_score = cand.copy(), cand_score
                    print(f"[{it}] new best (random) = {best_score:.6f}")
            continue

        # ----- LP‑guided improvement -----
        # compute current convolution and its maximum
        b = np.convolve(cur_seq, cur_seq)
        M = float(b.max())

        # Solve small LP to obtain a better candidate `g`.
        g_candidate = lp_max_sum_move(cur_seq, M, target_sum, rng, top_k=30)

        # Mix the new candidate with the current sequence.
        # Reduce the mixing weight gradually if we do not see improvement.
        new_seq = (1.0 - mix_t) * cur_seq + mix_t * g_candidate
        new_seq = renormalise(new_seq, target_sum)

        new_score = sequence_score(new_seq)

        if new_score < cur_score:
            # Accept improvement
            cur_seq, cur_score = new_seq, new_score
            if new_score < best_score:
                best_seq, best_score = new_seq.copy(), new_score
                print(f"[{it}] new best (LP) = {best_score:.6f}")
                # tighten mixing weight after a good step
                mix_t = max(0.05, mix_t * 0.9)
                if best_score < 1.503:
                    break
        else:
            # No improvement – try a gradient‑type move as a fallback.
            grad_cand = gradient_move(cur_seq, rng, target_sum)
            grad_score = sequence_score(grad_cand)
            if grad_score < cur_score:
                cur_seq, cur_score = grad_cand, grad_score
                if grad_score < best_score:
                    best_seq, best_score = grad_cand.copy(), grad_score
                    print(f"[{it}] new best (grad) = {best_score:.6f}")
            else:
                # Reduce mixing step to become more conservative
                mix_t = min(0.5, mix_t * 1.05)

        # Occasionally enforce a mild symmetrisation – the optimal shape
        # is expected to be roughly symmetric.
        if rng.random() < 0.12:
            sym_seq = 0.5 * (cur_seq + cur_seq[::-1])
            sym_seq = renormalise(sym_seq, target_sum)
            sym_score = sequence_score(sym_seq)
            if sym_score < cur_score:
                cur_seq, cur_score = sym_seq, sym_score
                if sym_score < best_score:
                    best_seq, best_score = sym_seq.copy(), sym_score
                    print(f"[{it}] new best (sym) = {best_score:.6f}")

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return as plain Python list – the harness expects a list of floats.
    return [float(x) for x in best_seq]

# LP‑guided move (only used when SciPy is available)
# ----------------------------------------------------------------------
def lp_max_sum_move(f: np.ndarray,
                    M: float,
                    target_sum: float,
                    rng: np.random.Generator,
                    top_k: int = 30) -> np.ndarray:
    """
    Solve a tiny linear programme:
        maximise   sum(g_i)
        subject to  (f * g)_k <= M   for the `top_k` largest convolution points.
                    g_i >= 0
    Return the normalised solution (ℓ₁‑norm = target_sum).  If the LP fails,
    the current sequence `f` is returned unchanged.
    """
    if not SCIPY_AVAILABLE:
        return f.copy()

    n = f.size
    # Full convolution to locate the largest entries (cheap relative to the LP).
    full_b = np.convolve(f, f)

    # Determine the indices of the `top_k` largest values.
    if top_k >= len(full_b):
        active = np.arange(len(full_b))
    else:
        idx_part = np.argpartition(-full_b, top_k - 1)[:top_k]
        active = np.sort(idx_part)               # a stable ordering

    rows, cols, data = [], [], []

    for row_idx, k in enumerate(active):
        col_start = max(0, k - (n - 1))
        col_end = min(k, n - 1)
        cols_row = np.arange(col_start, col_end + 1, dtype=np.int32)
        rows_row = np.full(cols_row.shape, row_idx, dtype=np.int32)
        data_row = f[k - cols_row]               # coefficient = f_{k‑i}
        rows.append(rows_row)
        cols.append(cols_row)
        data.append(data_row)

    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)

    A_ub = csr_matrix((data, (rows, cols)), shape=(len(active), n))
    b_ub = np.full(len(active), M, dtype=np.float64)

    # maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(c,
                  A_ub=A_ub,
                  b_ub=b_ub,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 5.0})

    if not res.success or np.any(np.isnan(res.x)):
        return f.copy()

    g0 = np.asarray(res.x, dtype=np.float64)
    s = g0.sum()
    if s <= 0.0:
        return f.copy()
    g = g0 * (target_sum / s)
    return g

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Return a list of non‑negative step‑function heights that minimises the
    evaluation function.  The search runs for at most ``budget_s`` seconds.
    """
    rng = np.random.default_rng(seed)

    # ----- problem size -------------------------------------------------
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 256

    # Target ℓ₁‑norm – any positive value works because the objective is
    # scale‑invariant; we choose the classic √(2n).
    target_sum = math.sqrt(2.0 * n)

    # ----- initial sequence ---------------------------------------------
    if "height_sequence_1" in globals() and len(height_sequence_1) == n:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        base_seq = build_arcsine_sequence(n)

    cur_seq = renormalise(base_seq, target_sum)
    cur_score = sequence_score(cur_seq)

    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ----- search loop --------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)
    it = 0
    mix_t = 0.4                     # initial convex‑combination weight

    while time.monotonic() < deadline:
        it += 1

        # Occasionally use a cheap random move (5 % of iterations)
        if rng.random() < 0.05:
            if rng.random() < 0.5:
                cand = block_perturbation(cur_seq, rng, target_sum)
            else:
                cand = transfer_move(cur_seq, rng, target_sum)
            cand_score = sequence_score(cand)
            if cand_score < cur_score:
                cur_seq, cur_score = cand, cand_score
                if cand_score < best_score:
                    best_seq, best_score = cand.copy(), cand_score
                    print(f"[{it}] new best (rand) = {best_score:.6f}")
            continue

        # ---- LP‑guided improvement ------------------------------------
        b = np.convolve(cur_seq, cur_seq)
        M = float(b.max())

        g_candidate = lp_max_sum_move(cur_seq, M, target_sum, rng, top_k=30)

        # Mix the candidate with the current sequence.
        new_seq = (1.0 - mix_t) * cur_seq + mix_t * g_candidate
        new_seq = renormalise(new_seq, target_sum)   # technically a no‑op
        new_score = sequence_score(new_seq)

        if new_score < cur_score:
            cur_seq, cur_score = new_seq, new_score
            if new_score < best_score:
                best_seq, best_score = new_seq.copy(), new_score
                print(f"[{it}] new best (LP) = {best_score:.6f}")
                mix_t = max(0.05, mix_t * 0.9)       # become more conservative
                if best_score < 1.503:
                    break
        else:
            # LP step did not improve – try a gradient‑type move.
            grad_cand = gradient_move(cur_seq, rng, target_sum)
            grad_score = sequence_score(grad_cand)
            if grad_score < cur_score:
                cur_seq, cur_score = grad_cand, grad_score
                if grad_score < best_score:
                    best_seq, best_score = grad_cand.copy(), grad_score
                    print(f"[{it}] new best (grad) = {best_score:.6f}")
            else:
                # No improvement; relax the mixing weight a little.
                mix_t = min(0.5, mix_t * 1.05)

        # Occasionally enforce a mild symmetrisation – the optimum is expected
        # to be roughly symmetric.
        if rng.random() < 0.12:
            sym_seq = 0.5 * (cur_seq + cur_seq[::-1])
            sym_seq = renormalise(sym_seq, target_sum)
            sym_score = sequence_score(sym_seq)
            if sym_score < cur_score:
                cur_seq, cur_score = sym_seq, sym_score
                if sym_score < best_score:
                    best_seq, best_score = sym_seq.copy(), sym_score
                    print(f"[{it}] new best (sym) = {best_score:.6f}")

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    return [float(x) for x in best_seq]
