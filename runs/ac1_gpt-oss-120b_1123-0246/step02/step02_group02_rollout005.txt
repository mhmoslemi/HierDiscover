def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)  # leave 5 sec safety
    # Determine n
    try:
        prev_seq = np.asarray(height_sequence_1, dtype=float)
        n = prev_seq.size
    except Exception:
        n = 6000
    n = int(np.clip(n, 2000, 9000))  # ensure moderate
    # maintain sum=1
    # compute fft length
    fft_len = _next_pow2(2*n - 1)
    # starting candidates
    starts = []
    # uniform
    starts.append(np.full(n, 1.0/n))
    # random Dirichlet
    starts.append(rng.random(n))
    starts[-1] /= starts[-1].sum()
    # U-shaped exponents
    for p in rng.uniform(-4.0, -0.2, size=5):
        idx = np.arange(n, dtype=float)
        a = (idx + 1) ** p + (n - idx) ** p
        a = np.maximum(a, 0)
        a /= a.sum()
        starts.append(a)
    # include previous if not too small
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.size > 0:
            if prev.size != n:
                old_x = np.linspace(0, 1, prev.size)
                new_x = np.linspace(0, 1, n)
                prev = np.interp(new_x, old_x, prev)
            prev = np.clip(prev, 0.0, 1000.0)
            if prev.sum() > 0:
                prev = prev / prev.sum()
                starts.append(prev)
    except Exception:
        pass

    best_seq = None
    best_score = np.inf
    total_iters = 0

    # function to approximate score rapidly
    def approx_score(a):
        b = _self_convolution_fft(a, fft_len)
        maxb = b.max()
        return 2.0 * n * maxb

    # main loops
    for start_idx, a0 in enumerate(starts):
        if time.monotonic() >= deadline:
            break
        a = a0.copy()
        # ensure sum=1
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s
        # evaluate initial
        sc = approx_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[start {start_idx}] init approx score = {best_score:.6f}")
        # gradient descent loop
        max_iters = 20000
        eta = 0.8  # initial step size
        eta_decay = 0.9995
        min_eta = 1e-6
        for it in range(max_iters):
            now = time.monotonic()
            if now >= deadline:
                break
            # compute conv & max index
            conv = _self_convolution_fft(a, fft_len)
            maxb = conv.max()
            kmax = int(conv.argmax())
            # if maxb is already near lower bound: perhaps break? Let's compute lower bound: average = 1/(2*n-1). So minimal possible maxb >= 1/(2*n-1). Score lower bound = 2*n * 1/(2*n-1) ≈ 1.0
            # maybe early stop if improvement small
            # compute gradient
            grad = np.zeros_like(a)
            start_i = max(0, kmax - (n - 1))
            end_i = min(n - 1, kmax)
            if start_i <= end_i:
                idxs = np.arange(start_i, end_i + 1)
                grad[idxs] = 2.0 * a[kmax - idxs]
            # take step with backtracking line search
            step_accepted = False
            cur_eta = eta
            for trial in range(10):
                cand = a - cur_eta * grad
                # clip and renorm
                cand = np.maximum(cand, 0.0)
                ss = cand.sum()
                if ss > 0:
                    cand = cand / ss
                else:
                    # all zeros -> skip
                    cand = a.copy()
                cand_maxb = _self_convolution_fft(cand, fft_len).max()
                if cand_maxb <= maxb + 1e-12:
                    a = cand
                    maxb = cand_maxb
                    step_accepted = True
                    break
                else:
                    cur_eta *= 0.5
                    if cur_eta < min_eta:
                        break
            # update eta globally
            eta = max(min_eta, eta * eta_decay)
            # record progress
            total_iters += 1
            # periodic evaluation
            if total_iters % 500 == 0:
                sc = approx_score(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[iter {total_iters}] new best approx score = {best_score:.6f} (eta={eta:.6f})")
            # occasional random perturbation
            if it % 2000 == 0 and it > 0:
                # add small multiplicative noise
                noise = np.exp(rng.normal(0.0, 0.05, size=n))
                a = a * noise
                a = np.maximum(a, 0.0)
                a /= a.sum()
                # recompute convergence after perturb
                # reset eta to larger maybe
                eta = max(eta, 0.4)

    # final evaluation using exact evaluate_sequence function
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] best exact score = {final_score:.6f}")
    return [float(v) for v in best_seq]

"""
Propose a low‑score step‑function by direct sub‑gradient descent on the
max‑convolution objective.

The algorithm works with the scale‑invariant formulation
    minimize   max_k (a * a)[k]      (a_i ≥ 0 , Σ a_i = 1)
which is equivalent to the original `evaluate_sequence` score
    2·n·max_k (a*a)[k]  (n = len(a)).

For a given sequence a the sub‑gradient of   f(a)=max_k (a*a)[k}
is  g = 2·corr(a, e_{k*}) where k* is any index attaining the maximum.
A simple projected gradient step (clip, renormalise) is applied.
The loop is run on several diverse initialisations (uniform, random Dirichlet,
U‑shaped power laws, and the previous best if available).  A small amount of
multiplicative noise is injected periodically to escape plateaus.
Fast FFT‑based self‑convolutions give cheap estimates of the objective; the
exact `evaluate_sequence` is called only for the final answer (and occasional
updates of the best‑so‑far).

The routine respects the supplied time budget (leaving a small safety margin)
and returns the best sequence it has found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using real FFTs.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)
    return conv_full[: 2 * a.size - 1]


def _approx_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast approximation of the evaluation score (sum(a) is assumed to be 1)."""
    b = _self_convolution_fft(a, fft_len)
    return 2.0 * n * b.max()


def _symmetrise(a: np.ndarray) -> np.ndarray:
    """Force exact symmetry a[i] = a[-i-1] (preserves the sum)."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function returns a list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a safety margin for final evaluation / printing
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1. Choose a working length (n) and pre‑compute FFT length
    # --------------------------------------------------------------
    try:

    except Exception:
        n = 6000               # a reasonable default
    n = int(np.clip(n, 2000, 9000))

    fft_len = _next_pow2(2 * n - 1)   # sufficient for all linear convolutions

    # --------------------------------------------------------------
    # 2. Build a collection of starting points
    # --------------------------------------------------------------
    starts = []

    # (a) uniform vector
    starts.append(np.full(n, 1.0 / n))

    # (b) Dirichlet‑like random vector (Gamma(1,1) draws)
    rand = rng.random(n)
    rand /= rand.sum()
    starts.append(rand)

    # (c) Several “U‑shaped’’ power‑law profiles
    for p in rng.uniform(-4.0, -0.2, size=6):
        idx = np.arange(n, dtype=float)
        a = (idx + 1) ** p + (n - idx) ** p
        a = np.maximum(a, 0.0)
        a /= a.sum()
        starts.append(a)

    # (d) Previous best (interpolated if length differs)
    try:
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.ndim == 1 and prev.size > 0:

                new_x = np.linspace(0.0, 1.0, n)
                prev = np.interp(new_x, old_x, prev)
            prev = np.clip(prev, 0.0, 1000.0)
            if prev.sum() > 0.0:
                prev = prev / prev.sum()
                starts.append(prev)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Core optimisation loop
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    total_iters = 0

    for start_id, base_a in enumerate(starts):
        if time.monotonic() >= deadline:
            break

        a = base_a.copy()
        a = np.maximum(a, 0.0)

        if a_sum > 0.0:
            a /= a_sum

        # enforce exact symmetry from the start (the optimum is known to be symmetric)

                cand_maxb = _self_convolution_fft(cand, fft_len).max()
                if cand_maxb <= maxb + 1e-14:
                    a = cand
                    maxb = cand_maxb
                    accepted = True
                    break
                cur_eta *= 0.5
                if cur_eta < min_eta:
                    break

            # ----------------------------------------------------------
            # d) step‑size update
            # ----------------------------------------------------------
            eta = max(min_eta, eta * eta_decay)

            # ----------------------------------------------------------
            # e) bookkeeping
            # ----------------------------------------------------------
            total_iters += 1
            if total_iters % 500 == 0:
                approx = 2.0 * n * maxb
                if approx < best_score:
                    best_score = approx
                    best_seq = a.copy()
                    print(f"[iter {total_iters:5d}] new best approx score = {best_score:.6f}")

            # ----------------------------------------------------------
            # f) occasional random perturbation (diversification)
            # ----------------------------------------------------------
            if it % 2000 == 0 and it > 0:

                sigma = 0.08
                noise = np.exp(rng.normal(0.0, sigma, size=n))
                a = a * noise
                a = np.maximum(a, 0.0)
                a /= a.sum()
                a = _symmetrise(a)
                # reset a modest step size after a kick
                eta = max(eta, 0.4)

    # 4. Final exact evaluation (once) and return
    # --------------------------------------------------------------
