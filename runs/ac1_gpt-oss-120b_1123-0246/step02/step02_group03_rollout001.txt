"""
Propose a low–autocorrelation step function.

The algorithm consists of two stages:

1.  **Soft‑max gradient descent with Adam momentum**  
    – The objective `max_k (a∗a)_k` is approximated by a
      temperature‑annealed soft‑max.  
    – Gradients are computed analytically (via FFT) and an
      Adam‑style update with a linearly decaying learning rate
      keeps the total mass fixed (`∑a = √(2 n)`) by clipping and
      renormalising after every step.

2.  **Targeted local search**  
    – After the time budget allocated to the first stage, the
      current worst lag `k*` (where the convolution attains its
      maximum) is identified.  
    – A pair of indices `(i, j)` that contributes most to this
      lag (`a_i·a_j`) is selected and a small amount of mass is
      transferred from them to two positions with the smallest
      heights.  The move preserves the total sum and respects the
      `[0, 1000]` bounds.  
    – Greedy acceptance (and occasional log‑normal perturbations)
      is used to drive the maximum autocorrelation down.

Both stages respect the constraints required by `evaluate_sequence`
and stop before the provided time budget expires.
"""

import time
import numpy as np
from typing import List

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a power‑of‑two real FFT."""
    n = x.size + y.size - 1
    N = 1 << (n - 1).bit_length()  # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv


def _max_autocorr(a: np.ndarray) -> float:
    """Exact maximal autocorrelation (a∗a)."""
    return float(_fft_convolve(a, a).max())


def _grad_and_maxb(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Gradient of the soft‑max approximation of max_k (a∗a)_k.
    Returns the projected gradient (zero‑sum) and the exact max value.
    """
    n = a.size
    b = _fft_convolve(a, a)               # exact autocorrelation
    maxb = b.max()

    # soft‑max weights (stability via subtraction of maxb)
    exp_term = np.exp(beta * (b - maxb))
    w = exp_term / exp_term.sum()

    # gradient of Σ_k w_k (a∗a)_k  :  ∂/∂a_i = 2 Σ_k w_k a_{k‑i}
    w_rev = w[::-1]
    conv_rev = _fft_convolve(a, w_rev)   # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)      # indices picking the needed slice
    grad = 2.0 * conv_rev[idx]

    # project onto the hyper‑plane Σ a_i = const (zero‑mean)
    grad -= grad.mean()
    return grad, maxb


def _local_search(
    a: np.ndarray,
    best_score: float,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing focused on the current worst lag.
    Moves a small amount of mass from a high‑contributing pair to
    two minima.  Returns the (possibly) improved vector and its
    maximal autocorrelation.
    """
    n = a.size
    while time.monotonic() < deadline:
        # current autocorrelation and worst lag
        b = _fft_convolve(a, a)
        kmax = int(np.argmax(b))

        # indices that actually participate in lag kmax
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        i_all = np.arange(lo, hi + 1, dtype=int)
        j_all = kmax - i_all

        # consider only distinct pairs (i<j) to avoid the i=j case
        mask = i_all < j_all
        i_candidates = i_all[mask]
        j_candidates = j_all[mask]

        if i_candidates.size == 0:
            # fall back to a random distinct pair
            i_idx, j_idx = rng.integers(0, n, size=2)
            while j_idx == i_idx:
                j_idx = rng.integers(0, n)
        else:
            contrib = a[i_candidates] * a[j_candidates]
            if contrib.sum() == 0:
                # unlikely, but pick uniformly
                idx = rng.integers(0, i_candidates.size)
            else:
                probs = contrib / contrib.sum()
                idx = rng.choice(i_candidates.size, p=probs)
            i_idx = i_candidates[idx]
            j_idx = j_candidates[idx]

        # two positions with the smallest values (receive mass)
        low_order = np.argsort(a)[:2]
        t1, t2 = low_order[0], low_order[1]

        # amount to transfer – up to 10 % of the smaller donor
        max_transfer = min(a[i_idx], a[j_idx])
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random() * 0.10  # ≤10 % move

        cand = a.copy()
        cand[i_idx] -= delta
        cand[j_idx] -= delta
        cand[t1] += delta
        cand[t2] += delta

        # enforce bounds (clipping) – sum stays unchanged, but clipping may
        # introduce a tiny deviation; re‑normalise if needed
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s != 0.0:
            cand *= (target_sum / s)

        new_score = _max_autocorr(cand)
        if new_score < best_score:
            a = cand
            best_score = new_score
            # occasional log: improvement found
            print(f"[ls] iter {int(time.monotonic() - deadline):5d} "
                  f"improved score = {best_score:.6f}")

        # occasional random log‑normal shake‑up (probability ≈0.001)
        if rng.random() < 0.001:
            pert = np.exp(rng.normal(0.0, 0.07, size=n))
            cand2 = a * pert
            cand2 = np.clip(cand2, 0.0, 1000.0)
            cand2 = cand2 * (target_sum / cand2.sum())
            cand2 = np.clip(cand2, 0.0, 1000.0)
            cand2 = cand2 * (target_sum / cand2.sum())
            cand2_score = _max_autocorr(cand2)
            if cand2_score < best_score:
                a = cand2
                best_score = cand2_score
                print(f"[ls‑pert] improved score = {best_score:.6f}")

    return a, best_score

    start_time = time.monotonic()

    deadline = start_time + max(0.0, float(budget_s) - 0.2)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    n = base.size
    target_sum = np.sqrt(2.0 * n)

    a = np.clip(base, 0.0, 1000.0)
    if a.sum() == 0.0:
        a[:] = 1.0
    a *= (target_sum / a.sum())
    a = np.clip(a, 0.0, 1000.0)
    a *= (target_sum / a.sum())               # second pass after clipping

    best_seq = a.copy()
    best_score = _max_autocorr(a)
    print(f"[init] n={n}, initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Soft‑max gradient descent with Adam (≈70 % of the time budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.70 * budget_s
    # hyper‑parameters
    beta_start = 0.5
    beta_end = 12.0
    init_lr = target_sum * 0.02                # base step size (~2 % of total mass)
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    last_improve_iter = 0

    while time.monotonic() < phase1_deadline:
        t += 1
        elapsed = time.monotonic() - start_time
        frac = elapsed / budget_s

        # temperature (soft‑max sharpness)
        beta = beta_start + (beta_end - beta_start) * frac
        # learning rate schedule – linear decay
        lr = max(init_lr * (1.0 - frac), init_lr * 0.02)

        # Adam update (descent direction is -grad)
        m = beta1 * m + (1.0 - beta1) * grad

        m_hat = m / (1.0 - beta1 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # renormalise to the exact target sum (preserving the simplex)

        cand_score = _max_autocorr(cand)

            best_seq = cand.copy()
            a = cand
            last_improve_iter = t
            print(f"[gd] iter {t:6d}  new best = {best_score:.6f}")
        else:
            a = cand

        # random multiplicative perturbation every few hundred iterations
        if t % 500 == 0:
            pert = np.exp(rng.normal(0.0, 0.07, size=n))
            cand2 = a * pert
            cand2 = np.clip(cand2, 0.0, 1000.0)
            cand2 = cand2 * (target_sum / cand2.sum())
            cand2 = np.clip(cand2, 0.0, 1000.0)
            cand2 = cand2 * (target_sum / cand2.sum())
            cand2_score = _max_autocorr(cand2)
            if cand2_score < best_score:

                best_seq = cand2.copy()
                a = cand2
                print(f"[pert] iter {t:6d}  improved = {best_score:.6f}")

        # simple stagnation restart
        if t - last_improve_iter > 3000:
            a = rng.random(n)
            a = np.clip(a, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # 2. Targeted local search (remaining time)
    # ------------------------------------------------------------------
    a, best_score = _local_search(

    print(f"[done] total time {time.monotonic() - start_time:.2f}s, "
          f"best score = {best_score:.6f}")
