def _conv_fft(a, b):
    size = 1 << ((len(a) + len(b) - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[:len(a)+len(b)-1]

"""
Propose a low‑score step‑function using a smooth‑gradient descent on a
soft‑max approximation of the max‑convolution objective, followed by a fast
random‑perturbation fine‑tune.  The method works for any length n: we start
from a Beta(½,½) (arcsine) shape, iteratively minimise a smooth version of
2·n·max(b)/sum(a)^2 (which, after normalising sum(a)=√(2n), is simply the
maximum entry of the self‑convolution).  Gradient information is obtained
analytically (∂b_i/∂a_j=2·a_{i‑j}) and computed efficiently with FFT‑based
convolutions.  A continuation on the soft‑max temperature c moves the
optimisation from a global smoothing to the true max‑norm.  After the
gradient phase we apply tiny random mass transfers to escape possible local
minima.  The best sequence seen is finally evaluated with the provided
`evaluate_sequence` and returned.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Helper: FFT‑based convolution (real inputs)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution of real arrays x and y using FFT."""
    size = 1 << ((len(x) + len(y) - 2).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Helper: fast evaluation of the original objective (max of self‑convolution)
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray) -> float:
    """Compute 2·n·max(b)/sum(a)^2 using an FFT convolution."""
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = _conv_fft(a, a)
    max_b = b.max()
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Helper: generate a symmetric Beta(α,α)‑shaped sequence for a given n
# ----------------------------------------------------------------------
def _beta_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Build a non‑negative sequence of length n with shape
    p^{α‑1}(1‑p)^{α‑1} where p = (i+0.5)/n.
    The sequence is scaled so that its sum equals sqrt(2·n).
    """
    p = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(p, alpha - 1.0) * np.power(1.0 - p, alpha - 1.0)
    target_sum = sqrt(2.0 * n)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w)
        w_sum = w.sum()
    a = w * (target_sum / w_sum)
    a = np.clip(a, 0.0, 1000.0)
    return a


# ----------------------------------------------------------------------
# Core optimiser for a single length n
# ----------------------------------------------------------------------
def _optimize_sequence(n: int, a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a projected gradient descent on a soft‑max approximation of the
    max‑convolution objective.  Returns the best sequence found and its fast
    objective value.
    """
    desired_sum = sqrt(2.0 * n)
    a = a0.copy()
    best_a = a.copy()
    best_score = _fast_objective(a)

    # start with a modest smoothing parameter
    c = 0.5
    iteration = 0

    while time.monotonic() < deadline and iteration < 5000:
        # --- self‑convolution and soft‑max weights ---
        b = _conv_fft(a, a)               # length 2n‑1
        max_b = b.max()
        # stabilise the exponent
        exp_term = np.exp(c * (b - max_b))
        w = exp_term / exp_term.sum()     # softmax over convolution entries

        # --- gradient of the smoothed objective ---
        # correlation w ⋆ a = conv(w, a[::-1]) evaluated at shift j = n‑1‑j
        conv2 = _conv_fft(w, a[::-1])    # length 3n‑2
        g = 2.0 * conv2[n - 1 :: -1][:n]  # shape (n,)

        # --- backtracking line‑search (only accept improvements) ---
        step = 1.0
        improved = False
        for _ in range(5):
            cand = a - step * g
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0.0:
                cand *= desired_sum / s
            cand = np.clip(cand, 0.0, 1000.0)
            sc = _fast_objective(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                a = cand                     # continue from the improved point
                improved = True
                break
            step *= 0.5                     # reduce step and retry

        if not improved:
            # no improving step found – keep the current a (might be at a plateau)
            pass

        # increase the soft‑max concentration slowly
        c = min(c * 1.02 + 0.01, 200.0)

        iteration += 1
        if iteration % 100 == 0:
            print(f"[opt] n={n:6d} iter={iteration:5d} c={c:6.3f} best={best_score:.6f}")

        # early exit if we already beat the target by a comfortable margin
        if best_score < 1.45:
            break

    return best_a, best_score
