    # base candidates: uniform; best existing; random.
    candidates = []
    uniform = np.full(n, target_sum / n)
    candidates.append(uniform)
    # if previous seq exists, interp to n and scale
    if 'height_sequence_1' in globals():
        prev = np.asarray(height_sequence_1, dtype=float)

            prev = prev * (target_sum / s)
        candidates.append(prev)
    # random initial

    rand_vec = rng.random(n)
    rand_vec = rand_vec / rand_vec.sum() * target_sum
    candidates.append(rand_vec)

    for cand in candidates:
        seq = cand.copy()
        init_max = compute_max_conv(seq, fft_len)
        init_score = float(evaluate_sequence(seq.tolist()))
        if init_score < best_score:
            best_score = init_score
            best_seq = seq.copy()

        # Track best for this start
        cur_seq = seq
        cur_max = init_max
        # Temperature schedule: start T = maybe 0.5 * cur_max? Actually T measures tolerance for accepting worse.
        # We'll use exponential cooling.
        start_T = cur_max * 0.5
        end_T = cur_max * 0.001
        max_iters = 200000  # adapt to time
        for it in range(max_iters):
            now = time.monotonic()
            if now >= start_time + time_limit:
                break
            # compute temperature as linear or exponential:
            frac = it / max_iters
            T = start_T * ((end_T / start_T) ** frac)

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:

"""
Propose a low‑score step‑function using a hybrid of simulated‑annealing
and soft‑max gradient descent.

The routine works with a fixed length n (taken from the previous best
construction if available, otherwise a default).  The mass is normalised
to √(2 n) – the same normalisation used in the evaluation – and every
candidate is kept non‑negative and bounded by 1000.

* Simulated‑annealing phase *
    – Randomly transfer a small amount of mass between two positions while
      preserving the total sum.
    – The new convolution peak is computed by FFT.  Moves that improve the
      peak are always accepted; uphill moves are accepted with probability
      exp(−Δ / T) where T is a temperature that cools linearly.
    – Every few thousand accepted moves a random multiplicative perturbation
      (log‑normal “kick”) is applied to diversify the search.

* Gradient‑descent phase *
    – After the SA budget is exhausted a few soft‑max gradient steps are
      performed.  The smooth approximation to max (log‑sum‑exp) yields a
      cheap gradient that is computed with two FFTs.  The step size is
      decayed and the sequence is re‑projected onto the simplex (mass
      fixed, non‑negative).

Both phases keep track of the best exact score (via evaluate_sequence)
and finally return the best sequence found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using real FFTs.
    Returns array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)
    return conv_full[: 2 * a.size - 1]


def _max_self_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of a*a (full convolution)."""
    return _self_convolution_fft(a, fft_len).max()


def _correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full correlation (w ⋆ a) = convolution(w[::-1], a) using FFTs.
    Returns length len(w)+len(a)-1.
    """
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, fft_len)
    fa = np.fft.rfft(a, fft_len)
    corr_full = np.fft.irfft(fw * fa, fft_len)
    return corr_full[: len(w) + len(a) - 1]


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 314159, budget_s: float = 950.0, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluate_sequence() score.  The function respects a wall‑clock budget
    (default ~950 s; a safety margin is left for the final evaluation).
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # 2 s safety margin

    # ------------------------------------------------------------------
    # 1. Determine working length n and normalisation constant
    # ------------------------------------------------------------------
    prev_global = globals().get("height_sequence_1")
    if isinstance(prev_global, (list, tuple, np.ndarray)):
        n = int(np.asarray(prev_global).size)
    else:
        n = 6000                # sensible default
    n = int(np.clip(n, 2000, 9000))   # keep runtime manageable

    target_sum = np.sqrt(2.0 * n)      # the “√(2 n)” normalisation
    fft_len = _next_pow2(2 * n - 1)    # length for all needed FFTs

    # ------------------------------------------------------------------
    # 2. Build a few starting points
    # ------------------------------------------------------------------
    starts = []

    # (a) uniform (baseline)
    starts.append(np.full(n, target_sum / n))

    # (b) random positive vector
    rand_vec = rng.random(n)
    rand_vec = rand_vec / rand_vec.sum() * target_sum
    starts.append(rand_vec)

    # (c) exponential decreasing profile (often good for anti‑correlation)
    r = 0.998   # very slow decay, keeps all entries non‑tiny but breaks symmetry
    idx = np.arange(n, dtype=float)
    expo = r ** idx
    expo = expo / expo.sum() * target_sum
    starts.append(expo)

    # (d) reuse the previous best (interpolated / re‑scaled if needed)
    if isinstance(prev_global, (list, tuple, np.ndarray)):
        prev = np.asarray(prev_global, dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            if prev.size != n:
                # linear interpolation to required length
                old_x = np.linspace(0.0, 1.0, prev.size)
                new_x = np.linspace(0.0, 1.0, n)
                prev = np.interp(new_x, old_x, prev)
            prev = np.clip(prev, 0.0, 1000.0)
            s = prev.sum()
            if s > 0.0:
                prev = prev * (target_sum / s)
            starts.append(prev)

    best_seq = None
    best_score = np.inf
    total_moves = 0

    # ------------------------------------------------------------------
    # 3. Iterate over starting points
    # ------------------------------------------------------------------
    for start_id, seq0 in enumerate(starts):
        if time.monotonic() >= deadline:
            break

        a = np.asarray(seq0, dtype=float).copy()
        # enforce bounds and normalise exactly
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)

        cur_max = _max_self_convolution(a, fft_len)

        # exact score (may be slightly different because of clamping in evaluate_sequence)
        exact = float(evaluate_sequence(a.tolist()))
        if exact < best_score:
            best_score = exact
            best_seq = a.copy()
            print(f"[start {start_id}] initial exact score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Simulated‑annealing loop (mass‑transfer moves)
        # ------------------------------------------------------------------
        avg = target_sum / n
        # Temperature schedule: start relatively hot, finish cold
        T_start = max(cur_max * 0.5, 1e-4)
        T_end = cur_max * 0.001 + 1e-8

        # Determine a rough number of moves we can afford (depends on time)
        # We'll just run until the time budget expires.
        while time.monotonic() < deadline:
            # linear cooling based on total_moves
            frac = total_moves / 500000.0   # arbitrary scaling; larger => cooler
            T = T_start * (1 - frac) + T_end * frac

            # ---- propose a mass transfer between two indices ----
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            # draw delta from a zero‑mean normal distribution, scale with avg
            delta = rng.normal(scale=0.05 * avg)  # about ±5 % of avg
            vi, vj = a[i], a[j]
            new_vi = vi + delta
            new_vj = vj - delta
            # enforce bounds
            if new_vi < 0.0 or new_vi > 1000.0 or new_vj < 0.0 or new_vj > 1000.0:
                continue

            a_new = a.copy()
            a_new[i] = new_vi
            a_new[j] = new_vj

            # sum is preserved exactly by construction
            new_max = _max_self_convolution(a_new, fft_len)

            # ---- acceptance test ----
            if new_max < cur_max:
                a = a_new
                cur_max = new_max
                # exact evaluation only when improvement is seen
                exact = float(evaluate_sequence(a.tolist()))
                if exact < best_score:
                    best_score = exact
                    best_seq = a.copy()
                    print(f"[{total_moves:8d}] new best exact = {best_score:.6f}")
            else:
                # uphill move: accept probabilistically
                if T > 0.0 and rng.random() < np.exp(-(new_max - cur_max) / T):
                    a = a_new
                    cur_max = new_max

            total_moves += 1

            # ---- occasional diversification (log‑normal kick) ----
            if total_moves % 20000 == 0 and total_moves > 0:
                mult = np.exp(rng.normal(0.0, 0.08, size=n))
                a_kick = a * mult
                a_kick = np.clip(a_kick, 0.0, 1000.0)
                s_kick = a_kick.sum()
                if s_kick > 0.0:
                    a_kick *= (target_sum / s_kick)
                a = a_kick
                cur_max = _max_self_convolution(a, fft_len)
                # check if this kick produced improvement
                exact = float(evaluate_sequence(a.tolist()))
                if exact < best_score:
                    best_score = exact
                    best_seq = a.copy()
                    print(f"[{total_moves:8d}] post‑kick best exact = {best_score:.6f}")

            # ---- occasional soft‑max gradient refinement (fast) ----
            if total_moves % 50000 == 0 and total_moves > 0:
                # perform a handful of gradient steps
                a = _softmax_gradient_refine(a, target_sum, fft_len, rng)

    # ------------------------------------------------------------------
    # 4. Final exact evaluation and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (should never happen)
        best_seq = np.full(n, target_sum / n)

    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] best exact score = {final_score:.6f}")
    return [float(v) for v in best_seq]


# ----------------------------------------------------------------------
# Soft‑max gradient refinement (few inner iterations)
# ----------------------------------------------------------------------


def _softmax_gradient_refine(a: np.ndarray, target_sum: float,
                             fft_len: int, rng: np.random.Generator,
                             inner_iters: int = 100) -> np.ndarray:
    """
    Apply a short soft‑max gradient descent starting from ``a``.
    The routine returns the refined sequence (still respecting the sum).
    """
    n = a.size
    # hyper‑parameters
    beta_start = 0.8
    beta_end = 6.0
    eta_start = 0.03 * target_sum

    seq = a.copy()
    for it in range(inner_iters):
        # self‑convolution
        b = _self_convolution_fft(seq, fft_len)
        cur_max = b.max()
        beta = beta_start + (beta_end - beta_start) * (it / inner_iters)

        # stable soft‑max weights on the convolution values
        shifted = b - cur_max
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.zeros_like(w)
            w_norm[np.argmax(b)] = 1.0
        else:
            w_norm = w / w_sum

        # gradient = 2·β·corr(L‑1‑i)
        corr = _correlation_fft(w_norm, seq, fft_len)
        L = len(w)
        grad = 2.0 * beta * corr[L - 1 - np.arange(n)]

        # normalise direction, take a step, re‑project onto simplex
        norm = np.linalg.norm(grad)
        if norm > 0.0:
            direction = grad / norm
        else:
            direction = np.zeros_like(grad)

        eta = eta_start * max(0.0, 1.0 - it / inner_iters)
        cand = seq - eta * direction
        cand = np.clip(cand, 0.0, 1000.0)

        # re‑normalise the total mass exactly
        s = cand.sum()
        if s > 0.0:
            cand = cand * (target_sum / s)

        # accept if it improves the max‑convolution
        cand_max = _max_self_convolution(cand, fft_len)
        if cand_max <= cur_max:
            seq = cand
        else:
            # occasional uphill acceptance (small probability)
            if rng.random() < 0.001:
                seq = cand
    return seq
