"""
Optimization of step‑function heights for the convolution bound.
The routine starts from a fine grid of power‑law seeds (a_i ∝ (i+β)^{‑α}),
picks the best seed, and then runs a projected gradient descent that
minimises a smooth surrogate of the maximum convolution value.
The surrogate is  L = log ∑_k exp(γ·(a∗a)_k)  .
Its gradient can be computed by a single extra convolution, giving a
fast O(n²) update while respecting non‑negativity and the sum‑to‑one
constraint.
A simulated‑annealing style acceptance rule with an exponential
cool‑down, together with occasional random block perturbations, helps
to escape local minima.  The best sequence found before the time budget
expires is returned.
"""

import time
import numpy as np

def _evaluate_fast(seq: np.ndarray) -> float:
    """Fast analogue of the provided evaluate_sequence (sum>0 assumed)."""
    total = seq.sum()
    if total < 1e-12:
        return float('inf')
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    n = seq.size
    return float(2.0 * n * max_b / (total * total))

def _make_block_perturbation(base: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal noise on a random contiguous block."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, n - width + 1))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:   # optional mild symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

def _smooth_max_gradient(seq: np.ndarray, gamma: float) -> np.ndarray:
    """
    Compute the gradient of L(γ)=log∑_k exp(γ·(seq*seq)_k) w.r.t. ``seq``.
    The returned vector has the same size as ``seq``.
    """
    n = seq.size
    # full auto‑convolution a*a
    conv = np.convolve(seq, seq)               # length 2n‑1
    # Log‑sum‑exp weights (numerically stable)
    max_c = conv.max()
    w_unnorm = np.exp(gamma * (conv - max_c))
    w = w_unnorm / w_unnorm.sum()
    # weighted sum of shift vectors = (w * rev(seq)) convolution
    rev = seq[::-1]
    weighted = np.convolve(w, rev)             # length 3n‑2
    grad = weighted[n-1:n-1 + n]                # slice of length n
    # The true gradient of L is γ·2·grad, but the scalar factor is irrelevant
    return grad

    if total > 0:

    return best_seq.tolist()
