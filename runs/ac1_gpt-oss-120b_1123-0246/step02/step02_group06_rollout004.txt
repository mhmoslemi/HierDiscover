    T_start = 0.5

        # compute temperature
        elapsed = (time.monotonic() - start) / budget_s
        T = T_start * (1 - elapsed) + T_end * elapsed
        # pick a move
        r = rng.random()
        if r < 0.4:

"""
Simulated‑Annealing with permutation‑aware moves.

The algorithm starts from a shuffled “arcsine’’ distribution (the analytically
optimal decreasing shape).  It then explores the space of sequences by a
combination of

* simple swaps of two positions,
* reversal of a random block,
* random shuffling of a block,
* transfer of mass from one index to another,
* scaling of a block by a random factor.

These moves change both the multiset of values and their ordering, which
influences the aperiodic autocorrelation (the convolution).  A Metropolis
acceptance rule with a linearly‑cooled temperature is used.  The best
candidate seen before the time budget expires is returned.

The evaluation uses exactly the same logic as the supplied
``evaluate_sequence`` function but is implemented locally with FFT for speed.
"""

import time
import numpy as np
from math import exp, log, ceil

# ----------------------------------------------------------------------
# Helper to build the analytically optimal arcsine sequence.
# ----------------------------------------------------------------------
def build_arcsine_sequence(length: int) -> np.ndarray:
    """Return a non‑negative sequence a of given length with
    a[i] = binom(2i,i)/4**i  (exact up to floating‑point noise)."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        num = (2 * i - 1) * (2 * i)
        den = 4.0 * i * i
        a[i] = a[i - 1] * (num / den)
    return a


# ----------------------------------------------------------------------
# Score evaluation (exact clone of evaluate_sequence, but using FFT)
# ----------------------------------------------------------------------
def compute_score(seq: np.ndarray) -> float:
    """Score equivalent to evaluate_sequence for a NumPy array."""
    # protect against negatives / huge values
    seq = np.clip(seq, 0.0, 1000.0)

    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")

    n = seq.size
    # convolution via FFT (real‑to‑real)
    size = 1 << int(ceil(log(2 * n - 1, 2)))
    fft_a = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[: 2 * n - 1]
    max_b = float(conv.max())

    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Move primitives
# ----------------------------------------------------------------------
def swap_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap two random indices."""
    n = base.size
    i, j = rng.choice(n, size=2, replace=False)
    cand = base.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand


def block_reverse_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Reverse a random contiguous block."""
    n = base.size
    if n < 2:
        return base.copy()
    start = rng.integers(0, n - 1)
    end = rng.integers(start + 2, n + 1)  # at least length 2
    cand = base.copy()
    cand[start:end] = cand[start:end][::-1]
    return cand


def block_shuffle_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Shuffle the contents of a random block."""
    n = base.size
    if n < 2:
        return base.copy()
    start = rng.integers(0, n - 2)
    end = rng.integers(start + 2, n + 1)
    block = base[start:end].copy()
    rng.shuffle(block)
    cand = base.copy()
    cand[start:end] = block
    return cand


def transfer_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = base.size
    i = rng.integers(0, n)          # source
    j = rng.integers(0, n)          # destination (can be equal, harmless)
    if i == j or base[i] == 0.0:
        return base.copy()
    frac = rng.random() * 0.9       # up to 90 % of the source mass
    delta = frac * base[i]
    cand = base.copy()
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def block_scale_move(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random block by a log‑normal factor."""
    n = base.size
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.13)))
    cand[start:stop] *= factor
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 500, **kwargs):
    """
    Simulated‑Annealing search for a step‑function that minimises
    ``evaluate_sequence``.  The routine runs inside the supplied time budget
    and returns the best sequence (list of floats) found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the sequence length.
    # ------------------------------------------------------------------
    # If a reference sequence is present, use its length; otherwise fall back
    # to a moderate size that still allows a rich search.
    n = int(len(globals().get("height_sequence_1", [])))
    if n == 0:
        n = 2000

    # ------------------------------------------------------------------
    # 1. Starting point – shuffled arcsine values.
    # ------------------------------------------------------------------
    base_vals = build_arcsine_sequence(n)
    # Create an initial permutation (high / low interleaving)
    # interleave largest and smallest values to break monotonicity.
    sorted_vals = np.sort(base_vals)[::-1]           # descending
    interleaved = np.empty_like(sorted_vals)
    left, right = 0, n - 1
    pos = 0
    while left <= right:
        interleaved[pos] = sorted_vals[left]
        pos += 1
        left += 1
        if left <= right:
            interleaved[pos] = sorted_vals[right]
            pos += 1
            right -= 1
    cur = interleaved.astype(np.float64)

    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Simulated annealing loop.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # 2 s safety margin
    it = 0

    # Linear temperature schedule
    T_start = 0.5
    T_end = 1e-6

    while time.monotonic() < deadline:
        it += 1
        elapsed = (time.monotonic() - start_time) / budget_s
        T = T_start * (1.0 - elapsed) + T_end * elapsed

        # Choose a move
        r = rng.random()
        if r < 0.20:
            cand = swap_move(cur, rng)
        elif r < 0.35:
            cand = block_reverse_move(cur, rng)
        elif r < 0.50:
            cand = block_shuffle_move(cur, rng)
        elif r < 0.70:
            cand = transfer_move(cur, rng)
        else:
            cand = block_scale_move(cur, rng)

        cand_score = compute_score(cand)
        if not np.isfinite(cand_score):
            continue

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            prob = exp(-(cand_score - cur_score) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[{it:>7}] new best = {best_score:.6f}")

        # Periodic reset to best (helps keep diversity)
        if it % 60000 == 0:
            cur = best.copy()
            cur_score = best_score

        # Early exit if we beat a generous target
        if best_score < 1.45:      # well below the target 1.503
            break

    print(f"[final] best score {best_score:.6f} after {it} iterations")
    # Return plain Python list as required by the harness.
    return [float(x) for x in best]
