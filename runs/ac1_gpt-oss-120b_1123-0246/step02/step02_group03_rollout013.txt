"""
Propose a low‑score step‑function using a hybrid optimisation:
- multiple random initialisations (uniform, exponential, linear)
- soft‑max‑smoothed gradient descent with an Adam‑style optimiser
- occasional hard sub‑gradient steps that directly target the current
  maximal convolution entry (the “peak” of a*a)
- periodic log‑normal perturbations and restarts to escape plateaus

The sequence length is taken from the existing `height_sequence_1`
(if present) and limited to 12 000 entries to keep the FFT fast.
All updates keep the L1‑norm fixed to √(2 n) (the normalisation used
by `evaluate_sequence`).  The routine runs for the supplied time
budget (≈ 1000 s) and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (linear, not circular)
# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a power‑of‑two real FFT."""
    n = x.size + y.size - 1
    N = 1 << (n - 1).bit_length()            # next power of two
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution objective
# ----------------------------------------------------------------------
def _softmax_grad_and_maxb(a: np.ndarray, beta: float) -> tuple[np.ndarray, float]:
    """
    Returns
      grad : gradient of the soft‑max approximation to max_k (a*a)_k,
      maxb: exact maximal convolution value.
    """
    n = a.size
    # exact convolution and its maximum
    b = _fft_convolve(a, a)
    maxb = b.max()

    # soft‑max weights (more stable with subtraction of maxb)
    exp_term = np.exp(beta * (b - maxb))
    w = exp_term / exp_term.sum()

    # gradient of Σ_k w_k (a*a)_k = 2 Σ_k w_k a_{k‑i}
    w_rev = w[::-1]
    conv_rev = _fft_convolve(a, w_rev)            # length 3n‑2
    # indices that correspond to the centred part of the convolution
    idx = (2 * n - 2) - np.arange(n)               # n‑1 … 2n‑2
    grad = 2.0 * conv_rev[idx]

    # make the gradient zero‑sum (project onto Σ a_i = const hyper‑plane)
    grad -= grad.mean()
    return grad, maxb


# ----------------------------------------------------------------------
# Hard sub‑gradient for the true max‑convolution (peak reduction)
# ----------------------------------------------------------------------
def _hard_subgrad(a: np.ndarray, max_index: int) -> np.ndarray:
    """
    Sub‑gradient of the term (a*a)_k with k = max_index.
    The gradient is 2 * reversed(a) shifted by the lag d = n‑1‑k.
    """
    n = a.size
    d = n - 1 - max_index                # lag for which the peak occurs
    rev = a[::-1]
    g = np.zeros_like(a)

    if d >= 0:
        # shift rev to the left by d
        g[: n - d] = 2.0 * rev[d:]
    else:
        # shift rev to the right by |d|
        d = -d
        g[d:] = 2.0 * rev[: n - d]
    # zero‑sum projection (the update will later be re‑normalised anyway)
    g -= g.mean()
    return g


# ----------------------------------------------------------------------
# Random initialisation helpers
# ----------------------------------------------------------------------
def _random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a random non‑negative vector of length n.
    Several families are mixed to give diverse starting points.
    """
    choice = rng.integers(0, 4)
    if choice == 0:                                   # flat Dirichlet‑like
        a = rng.random(n)
    elif choice == 1:                                 # exponential decay
        lam = rng.uniform(0.0005, 0.02)                # control decay speed
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
    elif choice == 2:                                 # linear drop
        a = np.linspace(1.0, 0.0, n, dtype=np.float64)
    else:                                             # quadratic drop
        a = (np.arange(n, dtype=np.float64) + 1) ** -1.0
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 990.0) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The routine respects the 1000 s time limit
    and returns the best sequence discovered when the budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)   # safety margin

    # --------------------------------------------------------------
    # 1. Determine problem size (n) and normalisation
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = base.size
        # keep a manageable size – the solver works well up to ~12 k
        if n > 12000:
            lo = (n - 12000) // 2
            base = base[lo: lo + 12000]
            n = base.size
    else:
        n = 8000                     # a comfortable default
    target_sum = np.sqrt(2.0 * n)    # the “√(2n)” normalisation used by the evaluator

    # --------------------------------------------------------------
    # 2. Optimiser hyper‑parameters
    # --------------------------------------------------------------
    beta_start = 0.5
    beta_end = 13.0                     # final temperature (sharp soft‑max)
    init_lr = target_sum * 0.02         # base learning‑rate (≈2 % of total mass)
    min_lr = init_lr * 0.02
    restart_gap = 4000                  # iterations without improvement ⇒ restart
    max_iters_without_print = 5000

    # Adam‑style moments
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_seq = None
    best_score = np.inf
    iteration = 0
    last_improve = -1
    last_restart = -1

    # placeholders for Adam state (initialised at the first restart)
    m = None
    v = None
    t = 0

    # --------------------------------------------------------------
    # 3. Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- restart if we have stagnated or at the very first step -----
        if iteration == 0 or (iteration - last_improve) > restart_gap:
            a = _random_initial(n, rng)
            a = np.clip(a, 0.0, 1000.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.ones_like(a)
                a_sum = a.sum()
            a *= (target_sum / a_sum)                # enforce Σ a_i = √(2n)
            a = np.clip(a, 0.0, 1000.0)
            a *= (target_sum / a.sum())               # second normalisation after clipping
            # reset Adam moments
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            t = 0
            last_restart = iteration
            print(f"[{iteration:6d}] restart (new random seed) – sum={a.sum():.3f}")

        # ----- schedule for the soft‑max temperature β -----
        elapsed = time.monotonic() - start_time
        frac = min(1.0, elapsed / budget_s)
        beta = beta_start + (beta_end - beta_start) * frac

        # ----- compute gradient of the smooth surrogate -----
        grad, _ = _softmax_grad_and_maxb(a, beta)

        # ----- Adam update (projected onto zero‑sum hyper‑plane) -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = max(min_lr, init_lr * (1.0 - frac))
        step = - lr * m_hat / (np.sqrt(v_hat) + eps)
        step -= step.mean()                # keep the sum unchanged
        a = a + step

        # ------ enforce the problem’s simple bounds ------
        a = np.clip(a, 0.0, 1000.0)

        # ------ re‑normalise total mass exactly to √(2n) ------
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)
        a = np.clip(a, 0.0, 1000.0)        # a tiny extra safety clip

        # ----- evaluate the true objective (max convolution) -----
        conv = _fft_convolve(a, a)
        cur_maxb = conv.max()
        cur_score = float(cur_maxb)        # sum is fixed ⇒ score = max_b

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{iteration:6d}] NEW BEST  score={best_score:.6f}")

        # ----- hard sub‑gradient step every few thousand iterations -----
        if iteration % 3000 == 0 and iteration > 0:
            max_idx = conv.argmax()
            subg = _hard_subgrad(a, max_idx)
            # use a modest step to nudge the peak downwards
            sub_lr = init_lr * 0.4 * (1.0 - frac)
            a = a - sub_lr * subg
            a = np.clip(a, 0.0, 1000.0)
            a *= (target_sum / a.sum())

        # ----- occasional multiplicative log‑normal perturbations -----
        if iteration % 500 == 0 and iteration > 0:
            perturb = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * perturb
            a = np.clip(a, 0.0, 1000.0)
            a *= (target_sum / a.sum())

        # occasional progress printout (helps debugging on long runs)
        if iteration % max_iters_without_print == 0 and iteration > 0:
            print(f"[{iteration:6d}]  current score={cur_score:.6f}  best={best_score:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # 4. Return the best sequence discovered
    # --------------------------------------------------------------
    print(f"[DONE] iterations={iteration}, best score={best_score:.6f}")
    if best_seq is None:                     # should never happen, but be safe
        raise RuntimeError("No feasible sequence was produced.")
    return [float(v) for v in best_seq]
