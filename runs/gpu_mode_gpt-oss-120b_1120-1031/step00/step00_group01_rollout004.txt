                  stride_bb, stride_bj, stride_bk, stride_bh,

def trimul_kernel(
    A_ptr, B_ptr, C_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(3)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    # early exit if hidden out of bounds
    if pid_h >= H:
        return

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    off_h = pid_h * stride_a_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        a_ptr = A_ptr + pid_b * stride_a_batch \
                + offs_i[:, None] * stride_a_i \
                + offs_k[None, :] * stride_a_k \
                + off_h

        b_ptr = B_ptr + pid_b * stride_b_batch \
                + offs_k[:, None] * stride_b_k \
                + offs_j[None, :] * stride_b_j \
                + off_h
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0, dtype=tl.float16)

            + offs_i[:, None] * stride_c_i \
            + offs_j[None, :] * stride_c_j \

def trimul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """Compute out[b,i,j,h] = sum_k A[b,i,k,h] * B[b,j,k,h] (A and B are half,
    accumulator is fp32). Each program instance processes a (i,j) tile for a
    single hidden channel h."""
    pid_b = tl.program_id(3)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    # Guard against hidden dim overflow
    if pid_h >= H:
        return

    # Offsets for the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Hidden offset (stride for the H dimension)
    off_h = pid_h * stride_a_h

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

                + offs_i[:, None] * stride_a_i \

"""
TriMul “outgoing” implementation (AlphaFold3) with a custom Triton kernel.

The operator works on a 4‑D tensor x ∈ ℝ^{B×N×N×C} and a binary mask
M ∈ ℝ^{B×N×N}.  The forward pass consists of:

1. Layer‑norm over the channel dimension C.
2. Two linear projections (left/right) plus three gating linear layers.
   The mask is applied to the projected tensors before gating.
3. Core “triangle” multiplication:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply (N×N × N×Nᴛ) for each hidden
   channel h.  The heavy O(N³·H) work is performed by a custom Triton
   kernel that computes a block‑wise GEMM using FP16 inputs and a
   FP32 accumulator.
4. Layer‑norm over the hidden dimension, multiplication with an output
   gate and a final linear projection back to the original channel size.

All heavy computation (step 3) is off‑loaded to Triton, while the
light‑weight linear/layer‑norm/gating steps are executed with PyTorch.
The kernel processes one hidden channel per program instance (grid dim‑2)
and computes a tile of size BLOCK_M×BLOCK_N for the (i,j) output plane.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """TriMul core: out[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h].

    A and B are FP16 tensors, the accumulator is FP32.  Each program instance
    processes a (BLOCK_M × BLOCK_N) tile for a single hidden channel h.
    """
    pid_b = tl.program_id(3)      # batch dimension
    pid_i = tl.program_id(0)      # i (row) block
    pid_j = tl.program_id(1)      # j (col) block
    pid_h = tl.program_id(2)      # hidden‑channel block (size 1)

    # Guard against out‑of‑range hidden indices (necessary only when H is not a
    # multiple of the grid size; harmless otherwise).
    if pid_h >= H:
        return

    # Tile offsets for the output (i, j) plane
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Offset for the selected hidden channel
    off_h = pid_h * stride_a_h

    # FP32 accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Reduction over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (M, K) – layout (B, i, k, h)
        a_ptr = A_ptr + pid_b * stride_a_batch \
                + offs_i[:, None] * stride_a_i \
                + offs_k[None, :] * stride_a_k \
                + off_h
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Load B tile transposed: shape (K, N) – layout (B, j, k, h) but we

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        hidden_dim,                     # 1 program per hidden channel
        B,
    )

    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out,
        B, N, hidden_dim,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,    # works well on H100
    )

    # ------------------------------------------------------------------
    # 4. Output normalisation, gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Apply the output gate
    out = out * out_gate  # both are float32

    # Final linear projection back to original channel size C
    out = F.linear(out, to_out_weight)  # shape [B, N, N, dim]

    return out
```
