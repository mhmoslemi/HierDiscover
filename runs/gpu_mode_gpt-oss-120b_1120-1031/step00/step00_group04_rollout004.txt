def custom_kernel(data):
    input_tensor, mask, weights, config = data
    batch_size, N, _, dim = input_tensor.shape

    # get config dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # layernorm on input
    # Using torch.nn.functional.layer_norm: want weight & bias. Set eps=1e-5
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            normalized_shape=(dim,),
                                            weight=weights["norm.weight"],
                                            bias=weights["norm.bias"])

    # Linear transforms
    # left_proj = F.linear(x_norm, left_proj_weight) (weight shape hidden_dim x dim)
    # Use torch.nn.functional.linear
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left_proj.dtype)
        left_proj = left_proj * mask_
        right_proj = right_proj * mask_

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Cast left and right to float16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Allocate output tensor
    out = torch.empty((batch_size, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # Choose block sizes
    BLOCK_M = 32  # tile rows
    BLOCK_N = 32  # tile cols
    BLOCK_K = 32  # sum dimension tile
    BLOCK_D = 16  # hidden dim chunk per launch

    # Compute number of hidden dim blocks
    H = hidden_dim
    num_d_blocks = (H + BLOCK_D - 1) // BLOCK_D

    # Get strides
    stride_left = left.stride()  # (stride_b, stride_i, stride_k, stride_h)
    stride_right = right.stride()
    stride_out = out.stride()

    # Determine the total number of hidden dim blocks by launching kernel multiple times per d block
    for d_blk in range(num_d_blocks):
        d_start = d_blk * BLOCK_D
        # compute actual BLOCK_D for last block
        cur_BLOCK_D = min(BLOCK_D, H - d_start)

        # compute pointers offset for left and right data corresponding to d_start
        # We can offset pointer by d_start * stride_left_h
        left_ptr = left.data_ptr() + d_start * stride_left[3] * left.element_size()
        right_ptr = right.data_ptr() + d_start * stride_right[3] * right.element_size()
        out_ptr = out.data_ptr() + d_start * stride_out[3] * out.element_size()

        # launch kernel
        grid = ( (N + BLOCK_M - 1) // BLOCK_M,
                 (N + BLOCK_N - 1) // BLOCK_N,
                 batch_size)
        trimul_kernel[grid](
            left_ptr, right_ptr, out_ptr,
            batch_size, N, H,
            stride_left[0], stride_left[1], stride_left[2], stride_left[3],
            stride_right[0], stride_right[1], stride_right[2], stride_right[3],
            stride_out[0], stride_out[1], stride_out[2], stride_out[3],
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
            BLOCK_D=cur_BLOCK_D, NUM_D=H, # maybe we need NUM_D for mask
            # Also pass type hints: tl.float16 for left/right
        )
    # After kernel, out is fp32 (since we accumulate in fp32 and store)

    # Now apply post gating and linear transforms:
    # out_norm = layer_norm(out, hidden_dim, weight=to_out_norm_weight, bias=to_out_norm_bias)
    out_norm = torch.nn.functional.layer_norm(out,
                                              normalized_shape=(hidden_dim,),
                                              weight=weights["to_out_norm.weight"],
                                              bias=weights["to_out_norm.bias"])
    # out_gate was already computed earlier (sigmoid). But shape [B,N,N,hidden_dim]; we need to multiply after out_norm.
    out_mul = out_norm * out_gate

    # final linear

    return output

"""
TriMul (outgoing) forward pass fused with Triton.

Algorithm
---------
1. Layer‑norm the input 4‑D tensor (B, N, N, C) using the provided
   `norm.weight` and `norm.bias`.
2. Apply three linear projections (no bias):
   * left_proj  : (C → H)
   * right_proj : (C → H)
   * three gates  (C → H)  + sigmoid
   * optional element‑wise mask on the projections.
3. Fuse the core O(N³·H) computation:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   The accumulation is performed by a custom Triton kernel that
   - tiles the i‑ and j‑axes (BLOCK_M × BLOCK_N threads per program)
   - tiles the reduction axis k in BLOCK_K chunks
   - processes the hidden dimension H in small blocks (BLOCK_D) that
     fit into registers (each thread holds a vector of length BLOCK_D).
   Accumulators are kept in FP32 for numerical stability; the input
   tensors are stored in FP16 to reduce memory traffic.
4. Layer‑norm the intermediate result (H‑dim), apply the output gate,
   and a final linear projection back to the original channel dimension C.
5. Return the tensor of shape (B, N, N, C).

The kernel implements only the O(N³·H) part; all cheap point‑wise
operations are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the core triple‑multiplication
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, H,
    # strides for left (b,i,k,d)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right (b,j,k,d) – note that the second axis is the “j” index
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for out (b,i,j,d)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # runtime offset for the hidden‑dim block processed by this launch
    D_OFFSET,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(2)               # batch dimension
    pid_i = tl.program_id(0)               # i‑tile index
    pid_j = tl.program_id(1)               # j‑tile index

    # ------------------------------------------------------------------
    #  tile offsets
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)
    d_range = tl.arange(0, BLOCK_D)

    # absolute hidden‑dim indices for this block
    d_idx = D_OFFSET + d_range
    mask_d = d_idx < H                     # hide out‑of‑range channels

    mask_i = i_base < N
    mask_j = j_base < N

    # accumulator: one FP32 vector (BLOCK_D) per (i,j) pair
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  reduction over k
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k_off = k_start + k_range
        mask_k = k_off < N

        # ----- left tile  (M, K, D) -------------------------------------------------
        left_ptrs = (left_ptr
                     + pid_b * stride_left_b
                     + i_base[:, None, None] * stride_left_i
                     + k_off[None, :, None] * stride_left_k
                     + d_idx[None, None, :] * stride_left_h)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                       other=0.0,
                       dtype=tl.float16).to(tl.float32)

        # ----- right tile (K, N, D) – note that the second axis is “j” ---------------
        right_ptrs = (right_ptr
                      + pid_b * stride_right_b
                      + k_off[:, None, None] * stride_right_k
                      + j_base[None, :, None] * stride_right_i
                      + d_idx[None, None, :] * stride_right_h)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                        other=0.0,
                        dtype=tl.float16).to(tl.float32)

        # ----- accumulate per hidden dimension --------------------------------------
        for d in range(BLOCK_D):
            # each slice is a (M,K) · (K,N) matrix‑multiply
            acc[:, :, d] += tl.dot(left[..., d], right[..., d])

    # ------------------------------------------------------------------
    #  store result
    # ------------------------------------------------------------------
    out_ptrs = (out_ptr
                + pid_b * stride_out_b
                + i_base[:, None, None] * stride_out_i
                + j_base[None, :, None] * stride_out_j
                + d_idx[None, None, :] * stride_out_h)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

# ----------------------------------------------------------------------
#  Python entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights      : dict of torch.Tensors containing all linear weights
            - config       : dict with keys "dim", "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension (often called `c`)

    # ------------------------------------------------------------------
    #  1. Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B,N,N,C]

    # ------------------------------------------------------------------
    #  2. Linear projections + gates
    # ------------------------------------------------------------------
    # proj: (C -> H)  – weight shape [H, C]
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # optional mask (broadcast over the hidden dimension)
    if mask is not None:
        mask_broadcast = mask.unsqueeze(-1).to(left_proj.dtype)
        left_proj = left_proj * mask_broadcast
        right_proj = right_proj * mask_broadcast

    # apply gates
    left = left_proj * left_gate               # [B,N,N,H]
    right = right_proj * right_gate            # [B,N,N,H]

    # ------------------------------------------------------------------
    #  3. Core O(N³·H) multiplication (Triton)
    # ------------------------------------------------------------------
    # Cast to FP16 for bandwidth‑limited compute
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # output buffer (FP32 accumulator)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=input_tensor.device)

    # Tiling parameters (tuned for H100)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 16            # must be a compile‑time constant

    # Strides (in elements) for the three tensors
    stride_left = left.stride()     # (b,i,k,d)
    stride_right = right.stride()   # (b,i,k,d) – same layout, we use the 2nd axis as “j”
    stride_out = out.stride()       # (b,i,j,d)

    # Number of blocks along the hidden dimension
    num_d_blocks = (H + BLOCK_D - 1) // BLOCK_D

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B )

    for d_block in range(num_d_blocks):
        d_offset = d_block * BLOCK_D
        # Launch kernel for a single hidden‑dim chunk
        trimul_kernel[grid](
            left.data_ptr(),
            right.data_ptr(),
            out.data_ptr(),
            B, N, H,
            stride_left[0], stride_left[1], stride_left[2], stride_left[3],
            stride_right[0], stride_right[1], stride_right[2], stride_right[3],
            stride_out[0], stride_out[1], stride_out[2], stride_out[3],
            d_offset,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # ------------------------------------------------------------------
    #  4. Post‑processing (LayerNorm → gate → final linear)
    # ------------------------------------------------------------------
    # to_out_norm : layer‑norm over hidden dimension H
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )  # [B,N,N,H]

    out_gated = out_norm * out_gate                     # apply output gate (broadcast OK)

    # final projection back to C channels (no bias)
    output = F.linear(out_gated, weights["to_out.weight"])   # [B,N,N,C]

    return output
