def custom_kernel(data):
    """Perform TriMul forward using Triton for the central triple multiplication.
    This implementation follows the AlphaFold3 TriMul outgoing version.
    It normalizes the input, applies linear projections and gates, then computes
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d] via a custom Triton
    kernel that performs a batch of independent N×N matrix multiplies over the
    hidden dimension. The result is normalized, gated, projected back to the
    original dimension and returned.
    """
    import torch, triton, triton.language as tl

    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # ensure mask shape
    if mask is None or mask.numel() == 0 or config.get("nomask", False):
        mask = torch.ones((B, N, N), dtype=input_tensor.dtype, device=input_tensor.device)
    # cast mask to same dtype
    mask = mask.to(input_tensor.dtype)

    # layernorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B,N,N,dim]

    # linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # [B,N,N,hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_weight)  # bias None
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask
    mask_unsq = mask.unsqueeze(-1)  # [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel (optional)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # allocate output
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # launch kernel
    # block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # reduce register pressure

    # calculate strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_k, stride_o_h = out.stride()

    # grid
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_head = B * hidden_dim
    grid = (grid_m, grid_n, grid_head)

    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_k, stride_o_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        # program ids
        pid_i = tl.program_id(0)   # block row index (i)
        pid_j = tl.program_id(1)   # block col index (j)
        pid_head = tl.program_id(2)  # batch * hidden dimension

        b = pid_head // H
        h = pid_head % H

        # compute global offsets
        offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

        # initialize accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # loop over k
        for start_k in range(0, tl.cdiv(N, BLOCK_K)):
            cur_k = start_k * BLOCK_K + tl.arange(0, BLOCK_K)

            # compute masks
            mask_i = offs_i < N
            mask_j = offs_j < N
            mask_k = cur_k < N

            # pointers for left: shape (M, K)
            a_ptr = left_ptr + b * stride_l_b + h * stride_l_h \
                     + offs_i[:, None] * stride_l_i \
                     + cur_k[None, :] * stride_l_k
            # pointers for right (transposed) shape (K, N)
            b_ptr = right_ptr + b * stride_r_b + h * stride_r_h \
                     + cur_k[:, None] * stride_r_k \
                     + offs_j[None, :] * stride_r_i

            a = tl.load(a_ptr, mask=(mask_i[:, None] & mask_k[None, :]), other=0.0, dtype=tl.float16)
            b = tl.load(b_ptr, mask=(mask_k[:, None] & mask_j[None, :]), other=0.0, dtype=tl.float16)

            # dot accumulation
            acc += tl.dot(a, b)

        # store result
        c_ptr = out_ptr + b * stride_o_b + h * stride_o_h \
                + offs_i[:, None] * stride_o_i \
                + offs_j[None, :] * stride_o_k
        mask_c = (offs_i[:, None] < N) & (offs_j[None, :] < N)
        tl.store(c_ptr, acc, mask=mask_c)

    # Launch Triton kernel
    tri_mul_kernel[grid](
        left_fp16, right_fp16, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_k, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Synchronize (optional)
    torch.cuda.synchronize()

    # out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate

    # final projection
    output = torch.nn.functional.linear(out, to_out_weight)  # bias None

    return output
```

"""
TriMul (outgoing) forward implementation with Triton.

The heavy “pairwise multiplication” is performed by a custom Triton kernel that
computes, for every batch `b` and hidden channel `h`:

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is equivalent to `out_h = left_h @ right_h.T` for each hidden slice `h`.
The kernel launches a 2‑D grid of blocks over the (i, j) plane and a third
dimension that combines the batch and hidden‑channel axes.  LayerNorms,
linear projections and gating are done in PyTorch; the kernel works in
float16 for the matrix multiply and accumulates in float32 to retain precision.
The final result is returned in float32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel for the core triple‑multiplication (batched N×N matmul)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                     # const float16 *
    right_ptr,                    # const float16 *
    out_ptr,                      # float32 *
    B, N, H,                      # int32 meta-parameters
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,
    BLOCK_M: tl.constexpr,        # tile size in i‑dimension
    BLOCK_N: tl.constexpr,        # tile size in j‑dimension
    BLOCK_K: tl.constexpr,        # tile size in reduction (k)
):
    # ---------------------------------------------------------------
    # program IDs
    # ---------------------------------------------------------------
    pid_i   = tl.program_id(0)                # block row index (i)
    pid_j   = tl.program_id(1)                # block col index (j)
    pid_head = tl.program_id(2)               # combines batch & hidden dim

    # decode batch and hidden‑channel number
    b = pid_head // H
    h = pid_head % H

    # global offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # reduction over k
    # ---------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tile_k in range(num_k_tiles):
        start_k = tile_k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds accesses
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # -----------------------------------------------------------
        #  Load A = left[b, i, k, h]   (shape BLOCK_M × BLOCK_K)
        # -----------------------------------------------------------
        a_ptr = (left_ptr
                 + b * stride_l_b
                 + h * stride_l_h
                 + offs_i[:, None] * stride_l_i
                 + offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptr,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0,
                    dtype=tl.float16)

        # -----------------------------------------------------------
        #  Load B = right[b, j, k, h] transposed as (k, j)
        # -----------------------------------------------------------
        b_ptr = (right_ptr
                 + b * stride_r_b
                 + h * stride_r_h
                 + offs_k[:, None] * stride_r_k
                 + offs_j[None, :] * stride_r_i)
        b = tl.load(b_ptr,
                    mask=(mask_k[:, None] & mask_j[None, :]),
                    other=0.0,
                    dtype=tl.float16)

        # accumulate the dot product (FP32 accumulator)
        acc += tl.dot(a, b)   # a,b are FP16 → acc FP32

    # ---------------------------------------------------------------
    #  Store the result: out[b, i, j, h]
    # ---------------------------------------------------------------
    c_ptr = (out_ptr
             + b * stride_o_b
             + h * stride_o_h
             + offs_i[:, None] * stride_o_i
             + offs_j[None, :] * stride_o_k)
    mask_store = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptr, acc, mask=mask_store)


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    # Shapes / configuration
    B, N, _, dim = input_tensor.shape          # input: [B, N, N, dim]
    hidden_dim = config["hidden_dim"]          # usually 128
    eps = 1e-5

    # ------------------------------------------------------------------
    # Prepare mask (if nomask, treat as all‑ones)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    mask = mask.to(dtype).unsqueeze(-1)        # [B, N, N, 1]

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim‑last)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                          # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gating
    # ------------------------------------------------------------------
    # Project to hidden dimension
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask before the gates (as in the reference implementation)
    left  = left * mask
    right = right * mask

    # Compute gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy multiplication (saves memory bandwidth)
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Allocate output buffer (FP32, as required for subsequent layers)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128   # tile size for i‑dimension
    BLOCK_N = 128   # tile size for j‑dimension
    BLOCK_K = 32    # reduction tile size

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_h = B * hidden_dim
    grid = (grid_m, grid_n, grid_h)

    # Strides (convert to int for Triton)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_k, stride_o_h = out.stride()

    # ------------------------------------------------------------------
    # Launch the kernel
    # ------------------------------------------------------------------
    tri_mul_kernel[grid](
        left_fp16, right_fp16, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_k, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before moving on
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                           # gating after norm

    # Final linear back to original dim
    output = F.linear(out, weights["to_out.weight"])   # bias is None

    return output
