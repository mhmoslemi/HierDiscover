def custom_kernel(data):
    x, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']

    # Convert mask to float32 if exists else all ones.
    if mask is None:
        mask = torch.ones(x.shape[:3], dtype=x.dtype, device=x.device)
    mask = mask.to(x.dtype) # keep same dtype.

    # layernorm
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # projections and gates
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']

    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # apply mask
    mask_exp = mask.unsqueeze(-1)  # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # gating
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel

"""
TriMul (outgoing) forward pass implemented with Triton.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,dim]).
2. Linear projections (no bias) to a hidden dimension `H`:
   * left = Linear(x, left_proj)        -> [B,N,N,H]
   * right = Linear(x, right_proj)      -> [B,N,N,H]
3. Apply an optional pairwise mask (broadcast on the hidden dim).
4. Compute three sigmoidal gates from `x`:
   left_gate , right_gate , out_gate  -> [B,N,N,H]
5. Modulate the projected tensors with the gates:
   left  = left  * left_gate
   right = right * right_gate
6. **Heavy part** – compute the tensor product over the second sequence
   dimension:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix‑multiply (L @ R^T) performed per hidden
   channel with a Triton kernel (fp16 inputs, fp32 accumulation).
7. Layer‑norm over the hidden dimension, apply `out_gate`,
   and a final linear projection back to `dim`.

The kernel fuses the N³ multiplication for all batches and hidden
channels, using block‑wise tiling (M/N/K) and tensor‑core friendly
fp16 arithmetic.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const __fp16 *
    right_ptr,         # const __fp16 *
    out_ptr,           # float *
    B, N, H,           # int64 scalars
    stride_b, stride_i, stride_j, stride_h,   # int64 strides of tensors
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    for a given batch `b` and hidden channel `h`.
    Grid layout:
        axis 0 : combined (batch, hidden)   -> pid = b*H + h
        axis 1 : tile row (i)
        axis 2 : tile column (j)
    """
    pid = tl.program_id(0)
    batch_id = pid // H
    hid = pid % H

    pid_m = tl.program_id(1)   # i‑tile
    pid_n = tl.program_id(2)   # j‑tile

    # ----- tile start indices -----
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch_id * stride_b + hid * stride_h
    right_base = right_ptr + batch_id * stride_b + hid * stride_h
    out_base   = out_ptr   + batch_id * stride_b + hid * stride_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K)  -> rows i, cols k
        a_ptrs = left_base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_j
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right tile: we need (K, N) i.e. rows k, cols j
        # This is effectively the transpose needed for the dot.
        b_ptrs = right_base + offs_j[None, :] * stride_i + offs_k[:, None] * stride_j
        b = tl.load(b_ptrs,
                    mask=mask_j[None, :] & mask_k[:, None],
                    other=0.0).to(tl.float32)

        # Accumulate the product
        acc += tl.dot(a, b)

    # Write the result (only for in‑bounds i,j)
    c_ptrs = out_base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config) where
        - x       : torch.Tensor [B, N, N, dim]
        - mask    : torch.Tensor [B, N, N] (float, 0/1)
        - weights : dict of torch.Tensors containing all linear / LN weights
        - config  : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as input)
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    device = x.device
    dtype = x.dtype

    # -------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # -------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4. Gating values (sigmoid)
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Cast to fp16 for the heavy N³ kernel
    # -------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = x.shape
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # -------------------------------------------------
    # 6. Triton kernel launch
    # -------------------------------------------------
    # Block sizes – chosen to fit in shared memory and expose enough parallelism
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    stride_b, stride_i, stride_j, stride_h = left_fp16.stride()  # (b,i,j,h)

    grid = (
        B * hidden,                         # axis 0 : batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,       # axis 1 : i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N        # axis 2 : j‑tiles
    )

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden,
        stride_b,
        stride_i,
        stride_j,
        stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=3,           # Triton defaults; may be tuned
        num_warps=4,
    )
    # -------------------------------------------------
    # 7. Post‑processing: LayerNorm over hidden, gating, final linear
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
