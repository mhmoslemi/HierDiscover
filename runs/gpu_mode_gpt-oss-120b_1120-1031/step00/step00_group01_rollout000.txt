```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute bs and hd indices
    b = pid // H
    hd = pid % H

    # Compute row and column offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers base for this (b, hd)
    left_base = left_ptr + b * stride_lb + hd * stride_ld
    right_base = right_ptr + b * stride_rb + hd * stride_rd
    out_base = out_ptr + b * stride_ob + hd * stride_od

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile [M,K]
        left_ptrs = left_base + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right tile transposed to [K,N]
        # right original shape [j,k], we need transpose => [k,j]
        right_T_ptrs = right_base + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_rj
        right_T = tl.load(right_T_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # compute dot product
        acc += tl.dot(left, right_T)

    # Store result
    out_ptrs = out_base + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) implementation with a fused batched GEMM written in Triton.

The heavy part of TriMul is the Einstein‑summation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each batch `b` and hidden channel `d` this is exactly a matrix
multiplication `out_{b,d} = left_{b,d} @ right_{b,d}.T` where the
matrices have shape (N, N).  The kernel below evaluates many independent
(b,d) multiplications in parallel, tiling the output matrices with
BLOCK_M × BLOCK_N tiles and accumulating over the K dimension with
BLOCK_K.  All other operations (LayerNorm, linear projections, gating,
masking, final projection) are performed with PyTorch and only the
core N³ reduction is offloaded to Triton.

The implementation works for the shapes required by the test suite
(e.g. B∈{1,2}, N up to 1024, hidden_dim=128) and returns a Float32
tensor of shape [B, N, N, dim].
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM kernel:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    The first program id encodes (b, h), while the second and third ids
    tile the i- and j‑dimensions respectively.
    """
    pid = tl.program_id(0)          # encodes (b, h)
    pid_m = tl.program_id(1)        # tile along i
    pid_n = tl.program_id(2)        # tile along j

    # ------------------------------------------------------------
    # decode batch and hidden index
    b = pid // H
    h = pid % H

    # ------------------------------------------------------------
    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------
    # base pointers for the selected (b, h) slice
    left_base = left_ptr + b * stride_lb + h * stride_ld
    right_base = right_ptr + b * stride_rb + h * stride_rd
    out_base = out_ptr + b * stride_ob + h * stride_od

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------
    # loop over k dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  [M, K]  (i, k)
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile transposed  [K, N]  (k, j)
        # we load right in (k, j) order by swapping the stride usage.
        right_T_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        right_T = tl.load(right_T_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0).to(tl.float32)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left, right_T)

    # ------------------------------------------------------------
    # write back the tile
    out_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (may be all‑ones)
        - weights      : dict of model weights (all Tensor)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Retrieve weights (all assumed to be on the same device as the input)
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm on the input tensor (normalize over the last dim = dim)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # Linear projections (no bias)
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Mask handling (mask may be None or omitted when `nomask=True`)
    if mask is None:
        mask = torch.ones_like(input_tensor[..., 0], dtype=dtype, device=device)
    # ensure mask is float for broadcasting
    mask = mask.to(dtype).unsqueeze(-1)    # [B, N, N, 1]

    # Apply mask to the projected tensors
    left  = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid of a linear projection of x)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Ensure tensors are contiguous for the Triton kernel
    left  = left.contiguous()
    right = right.contiguous()

    # Allocate output buffer for the N³ reduction
    out = torch.empty_like(left)  # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Triton kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 128

    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # Strides (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    grid = (
        B * H,                                 # program id 0 : (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,          # program id 1 : i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N           # program id 2 : j‑tiles
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Synchronize to make sure the result is ready before further ops
    torch.cuda.synchronize(device)

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden_dim, gating, final projection
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate

    # Final linear mapping back to `dim`
    out = F.linear(out, to_out_weight)

    return out
