import torch
import triton
import triton.language as tl

# Triton kernel applying mask and gates elementwise.
@triton.jit
def mask_gate_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                     total_elems, hidden_dim,
                     BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    # Load projection and gate values
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # Compute the mask index, broadcasting over hidden_dim
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
    # Elementwise multiply
    out = proj * gate * mask_val
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    TritMul "outgoing" forward pass.
    Implements:
        1. LayerNorm on input.
        2. Linear projections (left, right) and per‑position gates.
        3. Optional mask applied together with the gates via a small Triton kernel.
        4. Batched matrix multiplication over the k‑dimension (sum_k left[i,k] * right[j,k]).
        5. LayerNorm, out‑gate, and final linear projection back to the original dimension.
    All heavy arithmetic is performed in FP16 for speed; output is cast back to FP32.
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Whether the mask should be ignored
    nomask = config.get("nomask", False)

    # Cast everything to FP16 (half) for the core computation
    compute_dtype = torch.float16

    # Load and cast weight tensors
    norm_w = weights["norm.weight"].to(compute_dtype)
    norm_b = weights["norm.bias"].to(compute_dtype)

    left_proj_w = weights["left_proj.weight"].to(compute_dtype)
    right_proj_w = weights["right_proj.weight"].to(compute_dtype)

    left_gate_w = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w = weights["out_gate.weight"].to(compute_dtype)

    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)

    to_out_w = weights["to_out.weight"].to(compute_dtype)

    # --------------------------------------------------------------
    # 1) LayerNorm over the feature dimension `dim`
    # --------------------------------------------------------------
    eps = 1e-5
    x = input_tensor.to(compute_dtype)
    x = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # shape [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections and gate values (all in FP16)
    # --------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)          # [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # 3) Apply mask (if present) together with the gates via Triton
    # --------------------------------------------------------------
    if not nomask:
        # Ensure tensors are contiguous for pointer arithmetic
        mask = mask_tensor.to(compute_dtype).contiguous()
        # left_f = left_proj * left_gate * mask
        left_f = torch.empty_like(left_proj)
        right_f = torch.empty_like(right_proj)

        total_elements = left_proj.numel()      # B * N * N * hidden_dim
        BLOCK = 1024

        grid = lambda meta: (triton.cdiv(total_elements, meta['BLOCK_SIZE']),)

        mask_gate_kernel[grid](
            left_proj,
            left_gate,
            mask,
            left_f,
            total_elements,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
        )
        mask_gate_kernel[grid](
            right_proj,
            right_gate,
            mask,
            right_f,
            total_elements,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
        )
    else:
        # No mask → simple elementwise fusion in PyTorch
        left_f = left_proj * left_gate
        right_f = right_proj * right_gate

    # --------------------------------------------------------------
    # 4) TriMul contraction: out[i,j] = Σ_k left[i,k] * right[j,k]
    #    Implemented as a batched GEMM on the (hidden_dim) batch dimension.
    # --------------------------------------------------------------
    # Shape to [B, hidden_dim, N, N] to expose the hidden dimension as batch
    left_perm = left_f.permute(0, 3, 1, 2).contiguous()
    right_perm = right_f.permute(0, 3, 1, 2).contiguous()

    # Batched matrix multiplication: (B*hidden_dim) matrices N×N times N×N^T
    # Right matrix needs to be transposed on the last two axes.
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    # Restore original layout [B, N, N, hidden_dim]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5) Output layernorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast back to FP32 as required by the interface
    return out.to(torch.float32)

TriMul (outgoing) forward pass – highly‑optimized version.

Algorithm
---------
1. Layer‑norm over the last feature dimension (`dim`).
2. Linear projections → left / right (shape B,N,N,hidden_dim).
3. Compute three element‑wise gates (left, right, out) from the same
   normalized tensor.
4. If a mask is present, fuse “mask * projection * gate” using a small
   Triton kernel (the only required Triton part).  When `nomask=True`
   the fusion is performed with a plain PyTorch elementwise product.
5. Triangular multiplicative contraction:
        out_{i j d} = Σ_k left_{i k d} * right_{j k d}
   Implemented as a batched GEMM – we reshape to
   (B·hidden_dim, N, N) and call `torch.matmul`, which on H100
   uses tensor‑core kernels.
6. Layer‑norm over the hidden dimension, apply the out‑gate and a final
   linear projection back to `dim`.
All heavy arithmetic is performed in FP16 for speed; the result is
returned as FP32.
"""

    hidden_dim,             # i32    size of the hidden dimension
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)               # [BLOCK_SIZE]

    # mask for the tail of the array
    valid = offs < total_elems

    # load projection and gate values
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)   # fp16
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # each contiguous block of `hidden_dim` entries shares the same mask entry
    mask_index = offs // hidden_dim
    m_val = tl.load(mask_ptr + mask_index, mask=valid, other=0.0)

    out = proj * gate * m_val
    tl.store(out_ptr + offs, out, mask=valid)


# --------------------------------------------------------------
# Entry point required by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    TritMul outgoing forward pass.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    Tensor of shape [bs, seq_len, seq_len, dim] (FP32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Helper to cast weights to the compute dtype (FP16)
    # ------------------------------------------------------------------
    ct = torch.float16
    norm_w = weights["norm.weight"].to(ct)
    norm_b = weights["norm.bias"].to(ct)

    left_proj_w = weights["left_proj.weight"].to(ct)      # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(ct)

    left_gate_w = weights["left_gate.weight"].to(ct)
    right_gate_w = weights["right_gate.weight"].to(ct)
    out_gate_w = weights["out_gate.weight"].to(ct)

    to_out_norm_w = weights["to_out_norm.weight"].to(ct)  # (hidden_dim,)
    to_out_norm_b = weights["to_out_norm.bias"].to(ct)

    to_out_w = weights["to_out.weight"].to(ct)            # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the feature dimension `dim`
    # ------------------------------------------------------------------
    eps = 1e-5
    x = input_tensor.to(ct)
    x = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Projections & gate values (all FP16)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_w)          # [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 3) Fuse mask, projection and gate (Triton kernel) if needed
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure contiguous layout for pointer arithmetic inside Triton
        mask = mask_tensor.to(ct).contiguous()
        left_f = torch.empty_like(left_proj)
        right_f = torch.empty_like(right_proj)

        total = left_proj.numel()          # B * N * N * hidden_dim
        BLOCK = 1024                       # reasonable work‑group size

        grid = lambda meta: (triton.cdiv(total, meta['BLOCK_SIZE']),)

        # left side
        mask_gate_kernel[grid](
            left_proj,
            left_gate,
            mask,
            left_f,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
        )
        # right side
        mask_gate_kernel[grid](
            right_proj,
            right_gate,
            mask,
            right_f,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
        )
    else:
        # No mask → simple elementwise multiplication
        left_f = left_proj * left_gate
        right_f = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4) Triangular multiplicative contraction (batched GEMM)
    # ------------------------------------------------------------------
    # [B,N,N,hidden_dim] → [B, hidden_dim, N, N]
    left_perm = left_f.permute(0, 3, 1, 2).contiguous()
    right_perm = right_f.permute(0, 3, 1, 2).contiguous()

    # Batched matmul: (B·hidden_dim, N, N) @ (B·hidden_dim, N, N)^T
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    # Back to original layout [B,N,N,hidden_dim]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Output normalisation, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # → [B,N,N,dim]

    # ------------------------------------------------------------------
    # Return FP32 tensor as required by the interface
    # ------------------------------------------------------------------
    return out.to(torch.float32)
