def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah, # a batch stride: offset for each batch
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_STAGES: tl.constexpr,
):
    pid = tl.program_id(0)
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_idx = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))

    pid_n = block_idx % tl.cdiv(N, BLOCK_N)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # convert config: dim, hidden_dim
    # use torch.nn.functional.layer_norm
    # weight and bias for norm: shape [dim].
  
    # Norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (config["dim"],), weight=norm_weight, bias=norm_bias)
    # Left proj etc.
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # weight shape (hidden_dim, dim)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))
    # Apply mask if any
    if mask is not None and not config.get('nomask', True):
        mask = mask.unsqueeze(-1) # shape (B,N,N,1)
        left = left * mask
        right = right * mask
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    # Convert to half for buffering? Might help.
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    
    # Permute for kernel
    B, N, _, _ = left.shape
    hidden_dim = config["hidden_dim"]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous() # transpose k dimension

    # Compute out using triton kernel
    out = triton_gemm_batch(left_perm, right_perm, B, hidden_dim, N)  # returns tensor shape (B, hidden_dim, N, N)
    # Convert out to float32
    out = out.to(torch.float32)
    # Permute back to [B,N,N,hidden]
    out = out.permute(0, 2, 3, 1).contiguous()

    # Apply to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate
    out = out * out_gate

    # to_out linear
    out = torch.nn.functional.linear(out, to_out_weight)  # weight shape (dim, hidden_dim)
    return out
```

    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

"""
TriMul (outgoing) custom kernel.

The implementation follows the reference PyTorch TriMul module but off‑loads the
core N³ reduction (the einsum ``... i k d, ... j k d -> ... i j d``) to a
high‑performance Triton kernel.  The tensor layout is transformed so that each
hidden‑dimension slice becomes an independent batch element, turning the
einsum into a batched matrix multiplication:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
                = (left_slice[d] @ right_slice[d].T)[i,j]

The Triton kernel evaluates this batched GEMM in FP16 for the operands and
accumulates in FP32, yielding a FP32 result that is then processed by the
remaining LayerNorm / gating / linear layers (implemented with PyTorch for
convenience).  The kernel works for any batch size (B ≤ 2), sequence length
(N ∈ {128,256,512,1024}) and hidden dimension (c_z ∈ {128,384,768}) used in the
tests.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (A @ Bᵀ) where A and B have shape
# (batch, M, K) and (batch, K, N) respectively.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)          # batch * hidden channel
    pid_m = tl.program_id(1)            # block row
    pid_n = tl.program_id(2)            # block column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * stride_ah
    b_base = b_ptr + batch_id * stride_bh
    c_base = c_ptr + batch_id * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k_iter in range(num_tiles_k):
        k_offset = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A[i, k]   (shape BLOCK_M x BLOCK_K)
        a = tl.load(
            a_base + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B_T[k, j] (shape BLOCK_K x BLOCK_N)
        b = tl.load(
            b_base + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Cast to FP32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Store the result (FP32)
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A : (batch, M, K)   FP16
    B : (batch, K, N)   FP16
    Returns C : (batch, M, N) FP32
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Block sizes – chosen to give good occupancy on H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[
        grid
    ](
        A,
        B,
        C,
        M,
        N,
        K,
        A.stride(1),  # stride_am
        A.stride(2),  # stride_ak
        A.stride(0),  # stride_ah (batch stride)
        B.stride(1),  # stride_bk
        B.stride(2),  # stride_bn
        B.stride(0),  # stride_bh
        C.stride(1),  # stride_cm
        C.stride(2),  # stride_cn
        C.stride(0),  # stride_ch
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return C


# ----------------------------------------------------------------------
# Entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator with a Triton‑accelerated
    N³ reduction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : [B, N, N, dim] torch.Tensor (float32)
        - mask        : [B, N, N] torch.Tensor (bool or float) – may be all ones
        - weights     : dict of model parameters (float32 tensors)
        - config      : dict with at least 'dim', 'hidden_dim', and 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool; convert to same dtype as projections
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()   # fp32
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()  # fp32
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()      # fp32

    # Cast gates to FP16 for cheap element‑wise multiplication
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left.to(torch.float16) * left_gate_h
    right = right.to(torch.float16) * right_gate_h

    # ------------------------------------------------------------------
    # 6. Prepare tensors for the Triton batched GEMM:
    #    left:  (B, N, N, hidden) -> (B, hidden, N, N)
    #    right: (B, N, N, hidden) -> (B, hidden, K=N, J=N) but transposed on the
    #           last two axes so that we compute left @ rightᵀ.
    # ------------------------------------------------------------------
    # (B, hidden, N, N) contiguous layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose last two dims

    # Collapse batch & hidden into a single leading dimension
    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)    # (batch, M, K)
    right_flat = right_perm.view(B * H, N, N)  # (batch, K, N)

    # ------------------------------------------------------------------
    # 7. Triton batched matmul: out_flat = left_flat @ right_flat
    #    Result dtype is FP32.
    # ------------------------------------------------------------------
    out_flat = _batched_matmul_triton(left_flat, right_flat)  # (batch, N, N) FP32

    # Reshape back to (B, N, N, hidden)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 8. to_out_norm (LayerNorm over hidden dimension)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 9. Multiply by out_gate (still FP32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out
