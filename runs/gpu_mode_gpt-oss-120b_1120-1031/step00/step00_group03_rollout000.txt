def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    # layernorm
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    # using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)
    # left proj
    left_proj_weight = weights["left_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # mask apply if provided
    if mask is not None:
        # mask shape [B,N,N]; unsqueeze(-1) for broadcasting
        left = left * mask.unsqueeze(-1)
        right = right * mask.unsqueeze(-1)
    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate
    # Prepare for kernel
    B, N, _, _ = left.shape
    # reshape for kernel
    left_perm = left.permute(0,3,1,2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0,3,1,2).contiguous()

    left_flat = left_perm.view(batch_hidden, N, N)

@triton.jit
def tri_mul_kernel(a_ptr, b_ptr, c_ptr,
                   batch, M, N, K,
                   stride_a_batch, stride_a_m, stride_a_k,
                   stride_b_batch, stride_b_n, stride_b_k,
                   stride_c_batch, stride_c_m, stride_c_n,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    # compute block indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_blocks_m * num_blocks_n
    batch_id = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    pid_m = tile_id // num_blocks_n
    pid_n = tile_id % num_blocks_n

    # offsets for current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Determine run-time masks for out-of-bound
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load block from A (a_ptr)
        a_ptrs = a_ptr + batch_id * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load block from B transposed (b_ptr)
        b_ptrs = b_ptr + batch_id * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_ptr + batch_id * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

The heavy part of the TriMul module is the Einstein‑summation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden channel `d` (and each batch) this is a matrix
multiply `left_d @ right_dᵀ` where `left_d, right_d ∈ ℝ^{N×N}`.
We fuse this N²×N work into a single Triton kernel that processes
a tiled `M×N` block (M = rows = i, N = cols = j) while iterating over
the reduction dimension `k`.  The kernel works on the flattened batch
dimension `B*hidden_dim`, so the whole computation is a batch of
independent GEMMs (`B*hidden_dim` ≈ up to 1536 for the largest
configuration).

All other components (LayerNorm, linear projections, gates, final
linear) are performed with PyTorch which already provides highly‑optimized
implementations.  The kernel returns a tensor of shape
`[B, N, N, hidden_dim]` which is then Layer‑normalized,
gated and projected back to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,                         # dimensions
    stride_a_batch, stride_a_m, stride_a_k,  # strides for A
    stride_b_batch, stride_b_n, stride_b_k,  # strides for B (original layout)
    stride_c_batch, stride_c_m, stride_c_n,  # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ
    A : (batch, M, K)   (here M = N = seq_len, K = seq_len)
    B : (batch, N, K)   (we read it transposed as K×N)
    C : (batch, M, N)
    """
    pid = tl.program_id(0)

    # number of tiles in each dimension
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_m * num_n

    # decode which batch / tile we are processing
    batch_id = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    pid_m = tile_id // num_n
    pid_n = tile_id % num_n

    # offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- A block (M × K) -------------------------------------------------
        a_ptrs = (
            a_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ---- B block transposed (K × N)  ------------------------------------
        # B is stored as (batch, N, K).  We need Bᵀ → (K, N).
        b_ptrs = (
            b_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k    # stride over K (now rows)
            + offs_n[None, :] * stride_b_n    # stride over N (now columns)
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # matrix multiply–accumulate
        acc += tl.dot(a, b)

    # ---- store the result ----------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (x, mask, weights, config)
        - x            : Tensor[bs, seq_len, seq_len, dim]
        - mask         : Tensor[bs, seq_len, seq_len]  (may be None)
        - weights      : dict of model parameters (see docstring)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    out : Tensor[bs, seq_len, seq_len, dim]   (float32, CUDA)
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------- #
    # 1. Input layer‑norm (LayerNorm over ``dim``)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias)                                         #
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------------- #
    if mask is not None and mask.numel() > 0:
        mask_ = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------------------------- #
    # 4. Gating (linear + sigmoid)                                           #
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Core TriMul – batched matmul (left @ rightᵀ)                         #
    #    Prepare tensors in (batch*hidden_dim, N, N) layout
    # --------------------------------------------------------------------- #
    B, N, _, _ = left.shape

    # bring the hidden dimension in front so we can flatten batch*hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_hidden = B * hidden_dim
    left_flat = left_perm.view(batch_hidden, N, N)
    right_flat = right_perm.view(batch_hidden, N, N)

    # output buffer
    out_flat = torch.empty_like(left_flat)   # FP32, CUDA

    # --------------------------------------------------------------------- #
    # 6. Launch Triton kernel
    # --------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = batch_hidden * grid_m * grid_n

    tri_mul_kernel[
        (total_programs,)
    ](
        left_flat,
        right_flat,
        out_flat,
        batch_hidden,
        N,
        N,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 7. Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------------- #
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------------- #
    # 8. Output LayerNorm + gating
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # --------------------------------------------------------------------- #
    # 9. Final linear projection back to `dim`
    # --------------------------------------------------------------------- #
    out = F.linear(out, weights["to_out.weight"])

    return out
