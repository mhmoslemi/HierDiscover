def layernorm_kernel(
    x_ptr, out_ptr, weight_ptr, bias_ptr,
    total_rows, dim,
    stride_x_row, stride_x_dim,
    stride_out_row, stride_out_dim,
    eps: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    if row_idx >= total_rows: return
    offs_d = tl.arange(0, BLOCK_D)
    # iterate over dim in blocks
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)
    num = 0
    for start in range(0, dim, BLOCK_D):
        cur_D = min(BLOCK_D, dim-start)
        offs = start + offs_d

        x = tl.load(x_ptr + row_idx*stride_x_row + offs*stride_x_dim, mask=mask, other=0.0).to(tl.float32)

@triton.jit
def layernorm_kernel(
    X_ptr, Y_ptr,
    weight_ptr, bias_ptr,
    ROWS, COLS,
    stride_x_row, stride_x_col,
    stride_y_row, stride_y_col,
    eps: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    row = tl.program_id(0)
    if row >= ROWS:
        return

    col_offsets = tl.arange(0, BLOCK_C)
    mask = col_offsets < COLS

    # base pointers for this row
    X_row_ptr = X_ptr + row * stride_x_row
    Y_row_ptr = Y_ptr + row * stride_y_row

    x = tl.load(X_row_ptr + col_offsets * stride_x_col, mask=mask, other=0.0).to(tl.float32)
    sum_val = tl.sum(x)
    sum_sq = tl.sum(x * x)
    mean = sum_val / COLS
    var = sum_sq / COLS - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)
    y = (x - mean) * inv_std
    # scale and bias
    w = tl.load(weight_ptr + col_offsets, mask=mask, other=0.0)
    b = tl.load(bias_ptr + col_offsets, mask=mask, other=0.0)
    y = y * w + b
    tl.store(Y_row_ptr + col_offsets * stride_y_col, y, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)  # bool

    # Get weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Normalize input with Triton kernel
    B, N, _, _ = input_tensor.shape
    # flatten first three dimensions
    x_flat = input_tensor.reshape(-1, dim)
    x_norm_flat = torch.empty_like(x_flat)
    rows = x_flat.shape[0]
    cols = dim

    # Launch layernorm kernel
    BLOCK_C = 384  # max dim
    layernorm_kernel[(rows,)](
        x_flat,
        x_norm_flat,
        norm_weight,
        norm_bias,
        rows,
        cols,
        x_flat.stride(0),
        x_flat.stride(1),
        x_norm_flat.stride(0),
        x_norm_flat.stride(1),
        eps=1e-5,
        BLOCK_C=BLOCK_C,
    )
    # reshape back
    x_norm = x_norm_flat.view(B, N, N, dim)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    if not nomask:
        mask = mask_tensor.unsqueeze(-1)  # shape B,N,N,1
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # matmul over k dimension per hidden channel
    B, N, _, H = left.shape
    left_permuted = left.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H, N, N)
    right_permuted = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # compute using half precision for speed
    left_f16 = left_permuted.to(torch.float16)
    right_f16 = right_permuted.to(torch.float16)
    out_f16 = torch.bmm(left_f16, right_f16.transpose(1, 2))  # (B*H,N,N)
    out = out_f16.to(torch.float32).reshape(B, H, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # to_out_norm (LayerNorm over hidden dim)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # apply out_gate
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, to_out_weight)  # shape B,N,N,dim

    return out
```

"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1. **Input LayerNorm** – a custom Triton kernel normalises the last dimension
   of the 4‑D tensor `x ∈ ℝ^{B×N×N×C}` using the provided `norm.weight` and
   `norm.bias`.  The kernel works on a flattened view `x_flat ∈ ℝ^{(B·N·N)×C}`
   and processes one row per program; each program spawns `BLOCK_C` threads
   that load the whole channel dimension at once (max dim = 384).

2. **Linear projections & gates** (PyTorch)  
   ```
   left  = Linear(x_norm, left_proj.weight)   # → (B,N,N,HD)
   right = Linear(x_norm, right_proj.weight)
   left_gate  = sigmoid(Linear(x_norm, left_gate.weight))
   right_gate = sigmoid(Linear(x_norm, right_gate.weight))
   out_gate   = sigmoid(Linear(x_norm, out_gate.weight))
   ```
   Optional mask (`mask ∈ {0,1}`) is applied to `left` and `right` before the
   gate multiplications.

3. **Channel‑wise outer‑product sum**  
   The Einstein sum
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`  
   is equivalent to a batched matrix multiplication per hidden channel:
   ```
   L_d = left[..., :, :, d]          # (B,N,N)
   R_d = right[..., :, :, d]         # (B,N,N)
   out[..., d] = L_d @ R_dᵀ
   ```
   We fuse the three batch & channel dimensions (`B·HD`) and use `torch.bmm`
   on half‑precision tensors (FP16) for speed, converting the result back to
   FP32.

4. **Output LayerNorm + gating** (PyTorch)  
   Normalise over the hidden dimension with `to_out_norm`, multiply by the
   `out_gate`, and finally project back to the original feature dimension with
   `to_out`.

The heavy element‑wise normalisation is executed on the GPU via Triton, while
the bulk of linear algebra leverages highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: LayerNorm over the last dimension of a 2‑D view (rows×C)
# -------------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    X_ptr,                # *float32 input   (rows, C)
    Y_ptr,                # *float32 output  (rows, C)
    weight_ptr,           # *float32  weight (C)
    bias_ptr,             # *float32    bias (C)
    ROWS,                 # int total rows = B * N * N
    COLS,                 # int channel dimension C
    stride_x_row,         # int stride between rows of X
    stride_x_col,         # int stride between columns of X  (normally 1)
    stride_y_row,         # int stride between rows of Y
    stride_y_col,         # int stride between columns of Y
    eps: tl.constexpr,    # float epsilon for numerical stability
    BLOCK_C: tl.constexpr # compile‑time block size (>= max C, 384 works)
):
    row = tl.program_id(0)
    if row >= ROWS:
        return

    # offsets for the channel dimension
    col = tl.arange(0, BLOCK_C)
    mask = col < COLS

    # base pointers for the selected row
    X_row_ptr = X_ptr + row * stride_x_row
    Y_row_ptr = Y_ptr + row * stride_y_row

    # -----------------------------------------------------------------
    # Load the whole row, compute mean/variance, then write normalised
    # -----------------------------------------------------------------
    x = tl.load(X_row_ptr + col * stride_x_col, mask=mask, other=0.0).to(tl.float32)

    # mean & variance (single‑pass reduction)
    sum_val = tl.sum(x)                     # Σ x
    sum_sq  = tl.sum(x * x)                 # Σ x²
    mean = sum_val / COLS
    var  = sum_sq / COLS - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # normalisation + affine transform
    y = (x - mean) * inv_std
    w = tl.load(weight_ptr + col, mask=mask, other=0.0)
    b = tl.load(bias_ptr   + col, mask=mask, other=0.0)
    y = y * w + b

    # store result
    tl.store(Y_row_ptr + col * stride_y_col, y, mask=mask)


# -------------------------------------------------------------------------
# Public entry point used by the evaluator
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape (B, N, N, dim) – the transformed pairwise features.
    """
    # -----------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]          # input feature dimension C
    hidden_dim = config["hidden_dim"]   # projection dimension H
    nomask     = config.get("nomask", False)

    # -----------------------------------------------------------------
    # retrieve model weights
    # -----------------------------------------------------------------
    norm_weight        = weights["norm.weight"]
    norm_bias          = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm -> Triton kernel
    # -----------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N   # B·N·N rows after flattening the first three dims
    # flatten to (rows, C) – contiguous view
    x_flat   = input_tensor.reshape(total_rows, dim).contiguous()
    x_norm_flat = torch.empty_like(x_flat)

    # Launch one program per row; each program spawns BLOCK_C threads.
    BLOCK_C = 384   # maximum possible channel dimension
    layernorm_kernel[(total_rows,)](
        x_flat,
        x_norm_flat,
        norm_weight,
        norm_bias,
        total_rows,
        dim,
        x_flat.stride(0),  # row stride
        x_flat.stride(1),  # column stride (==1 for contiguous)
        x_norm_flat.stride(0),
        x_norm_flat.stride(1),
        eps=1e-5,
        BLOCK_C=BLOCK_C,
    )
    # restore original 4‑D shape
    x_norm = x_norm_flat.view(B, N, N, dim)

    # -----------------------------------------------------------------
    # 2) Linear projections & gating (PyTorch – dense kernels)
    # -----------------------------------------------------------------
    left  = F.linear(x_norm, left_proj_weight)    # (B,N,N,HD)
    right = F.linear(x_norm, right_proj_weight)

    if not nomask:
        # mask shape (B,N,N) → (B,N,N,1) for broadcasting
        mask = mask_tensor.unsqueeze(-1)
        left  = left * mask
        right = right * mask

    # gated scalars
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 3) Core TriMul -> batched matrix multiplication per hidden channel
    # -----------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    # reshape to (B·H, N, N) so a single torch.bmm covers all batches+heads
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # FP16 matmul for speed, then back to FP32
    left_fp16  = left_mat.to(torch.float16)
    right_fp16 = right_mat.to(torch.float16)
    out_fp16   = torch.bmm(left_fp16, right_fp16.transpose(1, 2))
    out_fp32   = out_fp16.to(torch.float32)

    # reshape back to (B,N,N,HD)
    out = out_fp32.view(B, H, N, N).permute(0, 2, 3, 1)   # (B,N,N,HD)

    # -----------------------------------------------------------------
    # 4) Output LayerNorm + out‑gate + final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate
    out = F.linear(out, to_out_weight)      # → (B,N,N,dim)

    return out
