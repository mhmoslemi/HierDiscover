```
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_aa, stride_ai, stride_ak,
    stride_bb, stride_bi, stride_bk,
    stride_cc, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
    # Program IDs
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start pointers for batch
    a_batch_ptr = a_ptr + pid_batch * stride_aa
    b_batch_ptr = b_ptr + pid_batch * stride_bb
    c_batch_ptr = c_ptr + pid_batch * stride_cc

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile (M, K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ai) + (offs_k[None, :] * stride_ak)
        # a_ptrs shape (BLOCK_M, BLOCK_K)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load b tile (K, N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bi) + (offs_n[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute
        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc.to(tl.float16) if acc.dtype==tl.float32 else acc, mask=mask_m[:, None] & mask_n[None, :])
```

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bi, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets for i and j
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # batch pointer offsets
    a_batch_ptr = a_ptr + pid_batch * stride_am
    b_batch_ptr = b_ptr + pid_batch * stride_bm
    c_batch_ptr = c_ptr + pid_batch * stride_cm

    # masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ai) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bi) + (offs_n[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # compute dot product
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
Implementation of TriMul "outgoing" update using a custom Triton kernel for the
pairwise reduction (einsum) step.
The kernel performs a batched GEMM of shape (B*hidden_dim, N, N) @ (B*hidden_dim, N, N)^T,
equivalent to einsum('... i k d, ... j k d -> ... i j d').
All other operations (LayerNorms, Linear projections, gates) are performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bi, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM kernel: C = A @ B^T  (A,B shape = [batch, M, K])"""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # batch offsets
    a_batch_ptr = a_ptr + pid_batch * stride_am
    b_batch_ptr = b_ptr + pid_batch * stride_bm
    c_batch_ptr = c_ptr + pid_batch * stride_cm

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ai) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B tile (K x N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bi) + (offs_n[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product (accumulate in fp32)
        acc += tl.dot(a, b)

    # write back result (cast to fp16 – output tensor is half precision)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the einsum '... i k d, ... j k d -> ... i j d' using a custom Triton kernel.
    left/right: tensors of shape (B, N, N, hidden_dim) (float32).
    Returns:
        out: tensor of shape (B, N, N, hidden_dim) (float32).

    total = B * H
    left_flat = left_fp.view(total, N, N)
    right_flat = right_fp.view(total, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # strides (in number of elements)
    a_batch_stride, a_row_stride, a_col_stride = left_flat.stride()
    b_batch_stride, b_row_stride, b_col_stride = right_flat.stride()
    c_batch_stride, c_row_stride, c_col_stride = out_flat.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        a_batch_stride, a_row_stride, a_col_stride,
        b_batch_stride, b_row_stride, b_col_stride,
        c_batch_stride, c_row_stride, c_col_stride,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back to (B, N, N, H) and cast to float32
    out_fp = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out_fp.to(torch.float32)
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be None)
            - weights: dict of model weights (see description)
            - config: dict containing 'dim', 'hidden_dim', and 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # shape [B, N, N, dim]

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Mask (if present) ----------
    if not nomask and mask is not None:
        # mask shape [B, N, N] -> broadcast to hidden dim
        mask_exp = mask.unsqueeze(-1).type_as(left)
        left = left * mask_exp
        right = right * mask_exp

    # ---------- Gating ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Core TriMul (einsum) ----------
    # Cast to half for the kernel to reduce memory traffic.
    out = tri_mul_gemm(left, right)   # shape [B, N, N, hidden_dim], float32

    # ---------- Output LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ---------- Final projection ----------
    out = F.linear(out, weights["to_out.weight"])
    return out

which is equivalent to a batched matrix multiplication
    C = A @ Bᵀ
where for each batch element and each hidden channel d,
A = left[..., i, k, d]  (shape N×N) and
B = right[..., j, k, d] (shape N×N).

All LayerNorms, Linear projections and gating are performed in PyTorch.
The core pair‑wise reduction is implemented with a custom Triton kernel
that computes many independent GEMMs (B·hidden_dim of size N×N).  The
kernel works on half‑precision data (FP16) and accumulates in FP32 for
numerical stability.  After the kernel the result is cast back to FP32
for the remaining operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ  (batched GEMM)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,                     # left  matrix pointer (FP16)
    b_ptr,                     # right matrix pointer (FP16), already transposed (K×N)
    c_ptr,                     # output pointer (FP16)
    M, N, K,                   # matrix dimensions (all = seq_len)
    stride_am, stride_ai, stride_ak,   # strides for A: batch, row (i), col (k)
    stride_bm, stride_bi, stride_bk,   # strides for B: batch, row (k), col (j)
    stride_cm, stride_ci, stride_cj,   # strides for C: batch, row (i), col (j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel: C = A @ Bᵀ.
    A and B have shape (batch, M, K) and (batch, K, N) respectively.
    """
    # program IDs
    pid_batch = tl.program_id(0)   # batch index = B * hidden_dim
    pid_m = tl.program_id(1)       # block row index (i)
    pid_n = tl.program_id(2)       # block column index (j)

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # batch base pointers
    a_batch = a_ptr + pid_batch * stride_am
    b_batch = b_ptr + pid_batch * stride_bm
    c_batch = c_ptr + pid_batch * stride_cm

    # mask for out‑of‑bounds (only needed on the fringe)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- load tile of A (M × K) ----
        a_ptrs = a_batch + (offs_m[:, None] * stride_ai) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # ---- load tile of B (K × N) ----
        b_ptrs = b_batch + (offs_k[:, None] * stride_bi) + (offs_n[None, :] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # ---- dot product and accumulate (FP32) ----
        acc += tl.dot(a, b)

    # ---- write result back (store as FP16) ----
    c_ptrs = c_batch + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that wraps the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the einsum ``... i k d, ... j k d -> ... i j d`` using the
    Triton kernel above.

    Parameters
    ----------
    left, right : torch.Tensor
        Tensors of shape (B, N, N, hidden) and dtype float32.

    Returns
    -------
    torch.Tensor
        Tensor of shape (B, N, N, hidden) and dtype float32.
    """
    B, N, N2, H = left.shape
    assert N == N2, "TriMul expects square pairwise dimensions"

    # --------------------------------------------------------------
    # 1) Rearrange tensors to (B, H, N, N) and cast to FP16.
    #    For the right tensor we also transpose the last two axes so
    #    that its layout is (K, J) = (k, j) – i.e. the transposed matrix.
    # --------------------------------------------------------------
    left_fp  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)          # (B, H, i, k)
    right_fp = right.permute(0, 3, 2, 1).contiguous().to(torch.float16)         # (B, H, k, j)

    # --------------------------------------------------------------
    # 2) Collapse batch and hidden dimensions -> (B*H, N, N)
    # --------------------------------------------------------------
    total = B * H
    left_flat  = left_fp.view(total, N, N)   # (batch, M, K) where M=N, K=N
    right_flat = right_fp.view(total, N, N)  # (batch, K, N)

    # --------------------------------------------------------------
    # 3) Allocate output buffer (FP16)
    # --------------------------------------------------------------
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # --------------------------------------------------------------
    # 4) Extract strides (in elements) for the kernel.
    # --------------------------------------------------------------
    a_batch_stride, a_row_stride, a_col_stride = left_flat.stride()
    b_batch_stride, b_row_stride, b_col_stride = right_flat.stride()
    c_batch_stride, c_row_stride, c_col_stride = out_flat.stride()

    # --------------------------------------------------------------
    # 5) Kernel launch configuration
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        total,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # --------------------------------------------------------------
    # 6) Launch the Triton kernel
    # --------------------------------------------------------------
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        a_batch_stride, a_row_stride, a_col_stride,
        b_batch_stride, b_row_stride, b_col_stride,
        c_batch_stride, c_row_stride, c_col_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H) and convert to FP32
    # --------------------------------------------------------------
    out_fp = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out_fp.to(torch.float32)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of model parameters (see problem statement)
        - config       : dict containing 'dim', 'hidden_dim' and optional 'nomask'

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B, N, N, hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(left)   # (B, N, N, 1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid after a bias‑less linear)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul contraction (batched GEMM via Triton)
    # ------------------------------------------------------------------
    # Cast to half precision for the kernel to reduce memory traffic.
    out = tri_mul_gemm(left, right)                     # (B, N, N, hidden) float32

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + final gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])
    return out
