def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # (1) LayerNorm (norm)
    # norm has weight and bias: shape [dim]
    # Use torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)
    # (2) left_proj = linear without bias: weight shape (hidden_dim, dim)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    # (3) mask: shape [B,N,N], we need to unsqueeze last dim to broadcast to hidden_dim
    if mask is not None:

        left = left * mask_exp
        right = right * mask_exp
    # (4) Gating: left_gate = sigmoid(linear(x_norm, left_gate_weight)) (no bias)
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    # (5) multiply gating
    left = left * left_gate
    right = right * right_gate
    # (6) compute out via custom triton kernel over left,right
    out = tri_mul_outgoing_kernel(left, right)   # shape [B,N,N,hidden_dim]

    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    # (8) Multiply by out_gate
    out = out * out_gate
    # (9) final linear to_out: weight shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]
    return out

def tri_mul_matmul(left_fp16, right_fp16):
    # left_fp16 shape [B,N,N,H], right same
    B, N, _, H = left_fp16.shape
    # permute
    left_perm = left_fp16.permute(0,3,1,2).contiguous()  # [B,H,N,N]
    right_perm = right_fp16.permute(0,3,2,1).contiguous()  # [B,H,N,N] (k/j swapped)
    # reshape to (B*H,N,N)
    batch_hidden = B * H
    left_flat = left_perm.reshape(batch_hidden, N, N)
    right_flat = right_perm.reshape(batch_hidden, N, N)
    # output empty
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)  # shape [B*H,N,N]
    # compute strides
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)
    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)
    stride_b_n = right_flat.stride(2)
    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)
    M = N
    N_out = N
    K = N
    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N), batch_hidden)
    # launch
    tri_mul_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        M, N_out, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape back
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1)
    return out
```

"""
TriMul (outgoing) – Triton‐accelerated core

This implementation follows the reference PyTorch TriMul module but offloads the
heavy N³ reduction to a custom Triton kernel.  The kernel computes, for each
batch element *b* and hidden dimension *d*,

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication
    L_d @ R_dᵀ   (L_d,R_d ∈ ℝ^{N×N})

The kernel is a tiled GEMM (BLOCK_M×BLOCK_N×BLOCK_K) that works on the
left/right tensors cast to float16 for bandwidth efficiency while accumulating
in float32.  The surrounding operations (LayerNorm, Linear, gating, mask) are
implemented with regular PyTorch ops.

The entry point expected by the grader is `custom_kernel(data)`.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM computing  Σ_k A[i,k] * B[j,k]
# --------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (all equal to seq_len)
    stride_ab, stride_am, stride_ak,         # A: batch, row, col strides
    stride_bb, stride_bk, stride_bn,         # B: batch, row (k), col (j) strides
    stride_cb, stride_cm, stride_cn,         # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    """
    Compute for a single (batch*t_hidden) slice:
        C[i, j] = Σ_k A[i, k] * B[k, j]
    where A has shape (M, K) and B has shape (K, N).
    All tensors are stored in row‑major order.
    """
    batch_id = tl.program_id(1)   # batch * hidden dimension
    pid = tl.program_id(0)        # tile id over the (M, N) output plane

    # ----------------------------------------------------------
    # Tile coordinates
    # ----------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_blocks_n
    pid_n = pid % num_blocks_n

    # ----------------------------------------------------------
    # Offsets within the tile
    # ----------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current tile (before the K‑loop)
    a_base = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_base = b_ptr + batch_id * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------
    # Loop over the reduction dimension K
    # ----------------------------------------------------------
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_iter * BLOCK_K
        # Load a sub‑matrix of A and B.  The masks guarantee we never read OOB.
        a = tl.load(a_base,
                    mask=mask_m[:, None] & (cur_k + offs_k < K),
                    other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_base,
                    mask=(cur_k + offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)   # (BLOCK_K, BLOCK_N)

        # Accumulate the dot product; a,b are fp16 → acc is fp32.
        acc += tl.dot(a, b)

        # Move pointers to the next K‑block.
        a_base += BLOCK_K * stride_ak
        b_base += BLOCK_K * stride_bk

    # ----------------------------------------------------------
    # Write the result back to C
    # ----------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------
# Helper that runs the Triton kernel on the left/right tensors
# --------------------------------------------------------------
def tri_mul_matmul(left_fp16: torch.Tensor, right_fp16: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, H] (fp16)
    Returns: out [B, N, N, H] (fp32)
    """
    B, N, _, H = left_fp16.shape
    # ------------------------------------------------------------------
    # Rearrange to (batch * hidden, N, N) where the second operand will be
    # transposed on its inner (k) dimension.
    # ------------------------------------------------------------------
    # left[i,k]   →  (i,k)   matrix
    # right[j,k]  →  (j,k)   matrix → need (k,j) for GEMM
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()      # [B, H, N, N] (k ↔ j)

    batch_hidden = B * H
    left_mat = left_perm.reshape(batch_hidden, N, N)   # (BH, M, K)
    right_mat = right_perm.reshape(batch_hidden, N, N) # (BH, K, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float32)  # (BH, M, N)

    # --------------------------------------------------------------
    # Strides (in elements) for the three tensors
    # --------------------------------------------------------------
    stride_ab = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bb = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)

    stride_cb = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    M = N
    K = N
    N_out = N

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (tiles over M×N, batch_hidden)
    grid = (triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N), batch_hidden)

    tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M, N_out, K,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout
    out = out_mat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    return out  # fp32


# --------------------------------------------------------------
# Main entry point used by the grader
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Implements the forward pass of the TriMul “outgoing” update.
    Input tuple:
        - input_tensor:  torch.Tensor, shape [B, N, N, dim] (float32)
        - mask:          torch.Tensor, shape [B, N, N] (float32/bool) or None
        - weights:       dict of model parameters
        - config:        dict containing at least "dim" and "hidden_dim"
    Output:
        - torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm before any projection
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask may be bool; cast to float for multiplication
        mask = mask.to(dtype=x.dtype)
        mask_exp = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gates (sigmoid after a linear projection, no bias)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_weight).sigmoid()     # [B,N,N,hidden]
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy N³ reduction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # Triton kernel: compute Σ_k left[...,i,k,d] * right[...,j,k,d]
    # ------------------------------------------------------------------
    out = tri_mul_matmul(left_fp16, right_fp16)   # [B,N,N,hidden] in fp32

    # ------------------------------------------------------------------
    # Final LayerNorm + gating + linear projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by the output gate (same hidden dim)
    out = out * out_gate

    # Linear projection to original dimensionality
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
