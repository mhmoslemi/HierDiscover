@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_j, stride_left_d,
    stride_right_b, stride_right_i, stride_right_j, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # Define i and j indices
    i_idx = i_offset + tl.arange(0, BLOCK_M)
    j_idx = j_offset + tl.arange(0, BLOCK_N)

    i_mask = i_idx < N
    j_mask = j_idx < N

    # Loop over hidden dimension blocks
    for d_offset in range(0, hidden_dim, BLOCK_D):
        d_idx = d_offset + tl.arange(0, BLOCK_D)
        d_mask = d_idx < hidden_dim

        # Initialize accumulator
        acc = tl.zeros([BLOCK_M, BLOCK_N, BLOCK_D], dtype=tl.float32)

        # Loop over K dimension (seq_len)
        for k_offset in range(0, N, BLOCK_K):
            k_idx = k_offset + tl.arange(0, BLOCK_K)
            k_mask = k_idx < N

            # Compute left pointer offsets: shape (BLOCK_M, BLOCK_K, BLOCK_D)
            left_ptrs = left_ptr + pid_b * stride_left_b \
                + i_idx[:, None, None] * stride_left_i \
                + k_idx[None, :, None] * stride_left_j \
                + d_idx[None, None, :] * stride_left_d
            left_mask = i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]
            left = tl.load(left_ptrs, mask=left_mask, other=0.0)  # shape (M, K, D)

            # Compute right pointer offsets: shape (BLOCK_N, BLOCK_K, BLOCK_D)
            right_ptrs = right_ptr + pid_b * stride_right_b \
                + j_idx[:, None, None] * stride_right_i \
                + k_idx[None, :, None] * stride_right_j \
                + d_idx[None, None, :] * stride_right_d
            right_mask = j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]
            right = tl.load(right_ptrs, mask=right_mask, other=0.0)

            # Now compute outer product for each k slice
            # we can iterate over k within BLOCK_K
            for kk in range(0, BLOCK_K):
                # scalar mask maybe zero if out-of-range handled above
                left_kk = left[:, kk, :]   # (M, D)
                right_kk = right[:, kk, :] # (N, D)
                # outer product via broadcasting
                # left_kk[:, None, :] shape (M,1,D); right_kk[None, :, :] shape (1,N,D)
                # product shape (M,N,D)
                acc += left_kk[:, None, :] * right_kk[None, :, :]

        # Store results to output
        out_ptrs = out_ptr + pid_b * stride_out_b \
            + i_idx[:, None, None] * stride_out_i \
            + j_idx[None, :, None] * stride_out_j \
            + d_idx[None, None, :] * stride_out_d
        out_mask = i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]
        tl.store(out_ptrs, acc, mask=out_mask)
```

def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    # Pt layernorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    
    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    
    # Optional mask
    if mask is not None and not config.get("nomask", False):
        mask_unsqueeze = mask.unsqueeze(-1)
        left = left * mask_unsqueeze
        right = right * mask_unsqueeze
    
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))
    
    left = left * left_gate
    right = right * right_gate
    
    # Core product via Triton kernel
    B, N, _, _ = left.shape # shape B,N,N,hd
    out = torch.empty_like(left) # same shape
    # launch kernel
    grid = (B, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    trimul_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=B,
        N=N,
        hidden_dim=hidden_dim,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_j=left.stride(2),
        stride_left_d=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_i=right.stride(1),
        stride_right_j=right.stride(2),
        stride_right_d=right.stride(3),
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_d=out.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D
    )
    # out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_j, stride_left_d,
    stride_right_b, stride_right_i, stride_right_j, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    i_idx = i_offset + tl.arange(0, BLOCK_M)
    j_idx = j_offset + tl.arange(0, BLOCK_N)
    i_mask = i_idx < N
    j_mask = j_idx < N

    # handle out-of-bounds for i and j:
    # If out-of-range, skip compute
    # We'll compute whole loops regardless but store only valid.

    for d_offset in range(0, hidden_dim, BLOCK_D):
        d_idx = d_offset + tl.arange(0, BLOCK_D)
        d_mask = d_idx < hidden_dim

        acc = tl.zeros([BLOCK_M, BLOCK_N, BLOCK_D], dtype=tl.float32)

        for k_offset in range(0, N, BLOCK_K):
            k_idx = k_offset + tl.arange(0, BLOCK_K)
            k_mask = k_idx < N

            # Load left: shape (M, K, D)
            left_ptrs = left_ptr + pid_b * stride_left_b \
                + i_idx[:, None, None] * stride_left_i \
                + k_idx[None, :, None] * stride_left_j \
                + d_idx[None, None, :] * stride_left_d
            left_mask = i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]
            left = tl.load(left_ptrs, mask=left_mask, other=0.0)

            # Load right: shape (N, K, D)
            right_ptrs = right_ptr + pid_b * stride_right_b \
                + j_idx[:, None, None] * stride_right_i \
                + k_idx[None, :, None] * stride_right_j \
                + d_idx[None, None, :] * stride_right_d
            right_mask = j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]
            right = tl.load(right_ptrs, mask=right_mask, other=0.0)

            # Loop over k in block
            for kk in range(0, BLOCK_K):
                # left_kk: (M, D)
                left_kk = left[:, kk, :]  # (M, D)
                right_kk = right[:, kk, :]  # (N, D)
                # Broadcast multiply: (M,1,D) * (1,N,D) -> (M,N,D)
                acc += left_kk[:, None, :] * right_kk[None, :, :]

        # Store acc
        out_ptrs = out_ptr + pid_b * stride_out_b \
            + i_idx[:, None, None] * stride_out_i \
            + j_idx[None, :, None] * stride_out_j \
            + d_idx[None, None, :] * stride_out_d
        out_mask = i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]
        tl.store(out_ptrs, acc, mask=out_mask)

"""
TriMul (outgoing version) – Triton‑accelerated forward pass.

The heavy part of TriMul is the tensor contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
where `left` and `right` have shape [B, N, N, H] (H = hidden_dim).
We implement this contraction with a tiled Triton kernel.
* The i‑ and j‑axes are tiled with BLOCK_M × BLOCK_N tiles.
* The hidden dimension is tiled with BLOCK_D and processed sequentially.
* The reduction over the sequence axis k is performed in blocks of size
  BLOCK_K, loading a (M×K×D) and (N×K×D) slice, then accumulating
  the outer‑product across the K dimension:
        acc[i, j, d] += left[i, k, d] * right[j, k, d]
  for every k in the tile.  All arithmetic is kept in FP32.

The surrounding LayerNorm, linear projections and gating are performed
with regular PyTorch ops (highly optimized).  Only the O(N³·H) core
operation is off‑loaded to Triton.
"""

import math
import torch
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_d,
    stride_right_b, stride_right_i, stride_right_j, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute out = Σ_k left[...,i,k,:] * right[...,j,k,:].

    Tiling:
      * i‑axis: BLOCK_M
      * j‑axis: BLOCK_N
      * hidden d‑axis: BLOCK_D (processed in a sequential loop)
      * reduction k‑axis: BLOCK_K (looped over the whole sequence length)
    """
    pid_b = tl.program_id(0)  # batch block
    pid_i = tl.program_id(1)  # i‑tile
    pid_j = tl.program_id(2)  # j‑tile

    # Tile offsets
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Index vectors for the current tile
    i_idx = i0 + tl.arange(0, BLOCK_M)
    j_idx = j0 + tl.arange(0, BLOCK_N)
    i_mask = i_idx < N
    j_mask = j_idx < N

    # Loop over hidden‑dim tiles
    d_offset = 0
    while d_offset < H:
        # Hidden‑dim indices within the tile
        d_idx = d_offset + tl.arange(0, BLOCK_D)
        d_mask = d_idx < H

        # Accumulator for this (i,j,d) tile
        acc = tl.zeros([BLOCK_M, BLOCK_N, BLOCK_D], dtype=tl.float32)

        # Loop over the reduction axis k
        k_offset = 0
        while k_offset < N:
            k_idx = k_offset + tl.arange(0, BLOCK_K)
            k_mask = k_idx < N

            # -------------------- load left[i, k, d] --------------------
            # shape (M, K, D)
            left_ptrs = (
                left_ptr
                + pid_b * stride_left_b
                + i_idx[:, None, None] * stride_left_i
                + k_idx[None, :, None] * stride_left_j
                + d_idx[None, None, :] * stride_left_d
            )
            left_mask = i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]
            left = tl.load(left_ptrs, mask=left_mask, other=0.0)

            # -------------------- load right[j, k, d] -------------------
            # shape (N, K, D)
            right_ptrs = (
                right_ptr
                + pid_b * stride_right_b
                + j_idx[:, None, None] * stride_right_i
                + k_idx[None, :, None] * stride_right_j
                + d_idx[None, None, :] * stride_right_d
            )
            right_mask = j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :]
            right = tl.load(right_ptrs, mask=right_mask, other=0.0)

            # -------------- outer‑product over the current k --------------
            # Iterate over the K values inside the tile (compile‑time)
            for kk in range(0, BLOCK_K):
                # left_kk: (M, D),  right_kk: (N, D)
                left_kk = left[:, kk, :]    # (M, D)
                right_kk = right[:, kk, :]  # (N, D)
                # Broadcast multiply -> (M, N, D)
                acc += left_kk[:, None, :] * right_kk[None, :, :]

            k_offset += BLOCK_K

        # -------------------- store the result --------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + i_idx[:, None, None] * stride_out_i
            + j_idx[None, :, None] * stride_out_j
            + d_idx[None, None, :] * stride_out_d
        )
        out_mask = i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :]
        tl.store(out_ptrs, acc, mask=out_mask)

        d_offset += BLOCK_D


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple containing
            - input tensor x of shape [B, N, N, dim]
            - mask tensor of shape [B, N, N] (may be None)
            - weights: dict with all learned parameters
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, same device as inputs)
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----- LayerNorm over the last dimension -----
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----- Linear projections (no bias) -----
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ----- Optional mask (broadcast on the hidden dim) -----
    if mask is not None and not nomask:
        mask = mask.to(x.dtype).unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----- Gating (sigmoid) -----
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ----- Core O(N³·H) contraction via Triton -----
    B, N, _, _ = left.shape
    out = torch.empty_like(left)   # shape [B, N, N, hidden_dim]

    # Tile sizes – tuned for H100 / fp32 registers
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 8
    BLOCK_D = 8

    grid = (
        B,
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )

    _trimul_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_j=left.stride(2),
        stride_left_d=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_i=right.stride(1),
        stride_right_j=right.stride(2),
        stride_right_d=right.stride(3),
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_d=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ----- Output normalisation, gating, and final projection -----
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
