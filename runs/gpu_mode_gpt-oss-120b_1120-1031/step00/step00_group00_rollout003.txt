def custom_kernel(data):
    x, mask, weights, config = data
    B,N,_,C = x.shape  # dim = C
    hidden = config["hidden_dim"]

    eps = 1e-5

    # layernorm
    # compute mean and var across last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # Linear projections (weights shape: hidden, dim)
    # Use torch.nn.functional.linear: F.linear(input, weight, bias=None)
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # left_gate, right_gate, out_gate
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # mask expand
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # shape [B,N,N,1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp
    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # reorder for matmul
    left_mat = left.permute(0,3,1,2).contiguous()   # [B, hidden, N, N]
    right_mat_T = right.permute(0,3,2,1).contiguous()  # [B, hidden, N, N] with k as row

    # reshape to batch dimension
    B_hidden = B * hidden
    left_mat_flat = left_mat.reshape(B_hidden, N, N)
    right_mat_flat = right_mat_T.reshape(B_hidden, N, N)

    # allocate out_mat_flat
    out_mat_flat = torch.empty_like(left_mat_flat)  # float32

    # launch triton kernel
    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = ( (B_hidden * ((N + BLOCK_M - 1)//BLOCK_M)), ( (N + BLOCK_N - 1)//BLOCK_N ) )
    batched_matmul_kernel[grid](
        left_mat_flat, right_mat_flat, out_mat_flat,
        B_hidden, N, N, N,
        left_mat_flat.stride(0), left_mat_flat.stride(1), left_mat_flat.stride(2),
        right_mat_flat.stride(0), right_mat_flat.stride(1), right_mat_flat.stride(2),
        out_mat_flat.stride(0), out_mat_flat.stride(1), out_mat_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K)

    # reshape out to (B, hidden, N, N)
    out_mat = out_mat_flat.reshape(B, hidden, N, N)

    # permute back to (B,N,N,hidden)
    out = out_mat.permute(0,2,3,1)  # (B, N, N, hidden)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    # gating
    out = out * out_gate   # shape (B,N,N,hidden)
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul “outgoing” forward pass (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the 4‑D input tensor `x ∈ ℝ^{B×N×N×C}`.
2. Linear projections (`left_proj`, `right_proj`) and three sigmoidal gates
   (`left_gate`, `right_gate`, `out_gate`) are computed from the normalized tensor.
3. The masks (if present) are applied to the projected tensors before gating.
4. The heavy triangular reduction
       out[b,i,j,d] = Σ_k left[b,i,k,d] · right[b,j,k,d]
   is performed as a batch of independent matrix multiplications
   L_d = left[...,i,k,d]      (shape B×D×N×N)
   R_dᵀ = right[...,k,j,d]    (shape B×D×N×N)   # transposed over the two sequence axes
   and realised in a custom Triton kernel (`batched_matmul_kernel`).
5. The result is layer‑normed (`to_out_norm`), modulated by `out_gate`,
   and linearly projected back to the original channel dimension (`to_out`).

The Triton kernel fuses the O(N³·D) reduction while the remaining
operations are performed with regular PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (B·D matrices of size N×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,                # pointer to left matrices   (B·D, M, K)
    b_ptr,                # pointer to right matrices  (B·D, K, N) – already transposed
    c_ptr,                # pointer to output matrices (B·D, M, N)
    batch,                # total number of matrices = B·D
    M, N, K,              # matrix dimensions (here M=N=K)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ------------------------------------------------------------------
    # 1) Program‑ID layout
    # ------------------------------------------------------------------
    pid = tl.program_id(axis=0)      # encodes (batch_id, block_row)
    pid_n = tl.program_id(axis=1)    # block column index

    num_blocks_m = tl.cdiv(M, BLOCK_M)          # blocks per matrix along the M axis
    batch_id = pid // num_blocks_m               # which (b·d) matrix we are processing
    block_m = pid % num_blocks_m                 # which block row inside that matrix

    # Abort if we are out of range (should never happen)
    if batch_id >= batch:
        return

    # ------------------------------------------------------------------
    # 2) Offsets for this block
    # ------------------------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)    # [BLOCK_N]

    # Masks for boundary handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 3) Accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) Loop over the reduction axis K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)       # [BLOCK_K]
        mask_k = offs_k < K

        # ---- Load A block (M × K) ------------------------------------
        a_ptrs = (a_ptr
                  + batch_id * stride_a_batch
                  + offs_m[:, None] * stride_a_m
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # ---- Load B block (K × N) ------------------------------------
        # B is stored transposed (K dimension first, N second)
        b_ptrs = (b_ptr
                  + batch_id * stride_b_batch
                  + offs_k[:, None] * stride_b_k
                  + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # ---- Compute partial product ----------------------------------
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # 5) Write back the result
    # ------------------------------------------------------------------
    c_ptrs = (c_ptr
              + batch_id * stride_c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, device‑matched to inputs)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (over the last dimension)
    # ------------------------------------------------------------------
    x = input_tensor
    # mean & variance are computed in float32 for numerical stability
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # 2) Linear projections and gates (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])          # → [B,N,N,hidden]
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 3) Optional mask (mask shape = [B, N, N])
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)       # [B,N,N,1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # ------------------------------------------------------------------
    # 4) Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Rearrange tensors so that we can treat each (b, d) pair as a
    #    separate matrix‑multiplication problem.
    #    left  : (B, N, N, hidden) → (B, hidden, N, N)
    #    right : (B, N, N, hidden) → (B, hidden, N, N) and then transpose
    # ------------------------------------------------------------------
    left_mat = left.permute(0, 3, 1, 2).contiguous()        # [B, hidden, N, N]
    right_mat_T = right.permute(0, 3, 2, 1).contiguous()   # [B, hidden, N, N] (k as first axis)

    B, H, N, _ = left_mat.shape
    batch = B * H                                           # total independent matrices

    # Collapse batch & hidden dimensions → (batch, N, N)
    left_flat = left_mat.reshape(batch, N, N)
    right_flat = right_mat_T.reshape(batch, N, N)

    # Allocate output buffer (float32)
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 6) Launch the Triton kernel for the batched N×N matmul reduction.
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch * ((N + BLOCK_M - 1) // BLOCK_M),      # axis 0  → (batch, block_row)
        (N + BLOCK_N - 1) // BLOCK_N                 # axis 1  → block column
    )

    batched_matmul_kernel[
        grid
    ](
        left_flat,
        right_flat,
        out_flat,
        batch,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Reshape back to (B, N, N, hidden)
    # ------------------------------------------------------------------
    out_mat = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 8) Post‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(out_mat,
                            normalized_shape=(hidden_dim,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"],
                            eps=eps)
    out_gated = out_norm * out_gate
    out = F.linear(out_gated, weights["to_out.weight"])

    return out
