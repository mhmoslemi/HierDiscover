    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_i = tl.program_id(0) # across i tiles
    pid_j = tl.program_id(1) # across j tiles
    pid_b = tl.program_id(2) # batch dimension

    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N
    b_offset = pid_b

    # pointers to start of left tile: left[b, i, k, h]
    # we will iterate over k dimension blocks

        left_i = i_offset + tl.arange(0, BLOCK_M)[:, None, None]

def compute_tri_mul_outgoing(left, right):
    B,N,N,H = left.shape
    # left: (B,N,N,H)
    # right: (B,N,N,H)
    # reshape to (B*H, N, N)
    left_reshape = left.permute(0,3,1,2).reshape(B*H, N, N)

    return out

    left_ptr, right_ptr,

    mask_ptr,

    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_left_b, stride_out_left_i, stride_out_left_j, stride_out_left_h,
    stride_out_right_b, stride_out_right_i, stride_out_right_j, stride_out_right_h,

    to_out_weight = weights["to_out.weight"]

    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections and gates
    # Linear: output = input @ weight.T (weight shape (out_dim, in_dim))
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [bs,N,N,hidden_dim]

    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    stride_out_right = right_out.stride()

    # Compute out = left @ right^T
    # We can cast to fp16 for speed with fp32 accumulation; use torch.matmul with dtype torch.float16 then acc in float32? torch supports autocasting

"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Input `x` → LayerNorm (`dim`).
2. Linear projections (no bias) → `left_proj`, `right_proj` (shape B,N,N, hidden).
3. Gates (linear + sigmoid) → `left_gate`, `right_gate`, `out_gate`.
4. Fuse mask, left‑gate and right‑gate with a tiny Triton kernel:
   left  = left_proj  * mask * left_gate
   right = right_proj * mask * right_gate
5. Compute the core O(N³·H) term:
      out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   This is performed as a batched GEMM:
      (B*H, N, N) @ (B*H, N, N)ᵀ
6. LayerNorm over the hidden dimension, apply `out_gate`.
7. Final linear projection back to `dim`.
All heavy algebra (step 5) uses cuBLAS via `torch.bmm`; the element‑wise
mask/gate fusion (step 4) is done in Triton to satisfy the “kernel‑part”
requirement.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr,               # float32 * (B,N,N,H)
    right_ptr,              # float32 * (B,N,N,H)
    left_gate_ptr,          # float32 * (B,N,N,H)
    right_gate_ptr,         # float32 * (B,N,N,H)
    mask_ptr,               # float32 * (B,N,N)
    out_left_ptr,           # float32 * (B,N,N,H) – destination
    out_right_ptr,          # float32 * (B,N,N,H) – destination
    B, N, H,                # int32 – dimensions
    # strides (in element units, not bytes)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_left_b, stride_out_left_i, stride_out_left_j, stride_out_left_h,
    stride_out_right_b, stride_out_right_i, stride_out_right_j, stride_out_right_h,
    BLOCK_SIZE: tl.constexpr
):
    """
    One‑dimensional elementwise kernel that fuses:
        out_left  = left_proj  * mask * left_gate
        out_right = right_proj * mask * right_gate
    """
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)   # [0, BLOCK_SIZE)

    total_elements = B * N * N * H
    mask_valid = offset < total_elements

    # ------------------------------------------------------------------
    # Decode linear offset → (b,i,j,h)
    #   idx = ((b*N + i)*N + j)*H + h
    # ------------------------------------------------------------------
    idx = offset
    h = idx % H
    idx1 = idx // H
    j = idx1 % N
    idx2 = idx1 // N
    i = idx2 % N
    b = idx2 // N

    # ------------------------------------------------------------------
    # Load values (masked loads return 0.0 for out‑of‑bounds threads)
    # ------------------------------------------------------------------
    left = tl.load(
        left_ptr
        + b * stride_left_b
        + i * stride_left_i
        + j * stride_left_j
        + h * stride_left_h,
        mask=mask_valid,
        other=0.0,
    )
    right = tl.load(
        right_ptr
        + b * stride_right_b
        + i * stride_right_i
        + j * stride_right_j
        + h * stride_right_h,
        mask=mask_valid,
        other=0.0,
    )
    left_gate = tl.load(
        left_gate_ptr
        + b * stride_gate_b
        + i * stride_gate_i
        + j * stride_gate_j
        + h * stride_gate_h,
        mask=mask_valid,
        other=0.0,
    )
    right_gate = tl.load(
        right_gate_ptr
        + b * stride_gate_b
        + i * stride_gate_i
        + j * stride_gate_j
        + h * stride_gate_h,
        mask=mask_valid,
        other=0.0,
    )
    mask_val = tl.load(
        mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j,
        mask=mask_valid,
        other=0.0,
    )

    # ------------------------------------------------------------------
    # Apply mask and gates
    # ------------------------------------------------------------------
    out_left = left * mask_val * left_gate
    out_right = right * mask_val * right_gate

    # ------------------------------------------------------------------
    # Store results
    # ------------------------------------------------------------------
    tl.store(
        out_left_ptr
        + b * stride_out_left_b
        + i * stride_out_left_i
        + j * stride_out_left_j
        + h * stride_out_left_h,
        out_left,
        mask=mask_valid,
    )
    tl.store(
        out_right_ptr
        + b * stride_out_right_b
        + i * stride_out_right_i
        + j * stride_out_right_j
        + h * stride_out_right_h,
        out_right,
        mask=mask_valid,
    )


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (or None)
        - weights      : dict of pre‑loaded model parameters
        - config       : dict containing 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same device as inputs)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Gating (linear -> sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # ------------------------------------------------------------------
    # Prepare mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask = mask.to(dtype=dtype)

    # ------------------------------------------------------------------
    # Fuse mask & gates with Triton (produces gated, masked tensors)
    # ------------------------------------------------------------------
    left_gated = torch.empty_like(left_proj)
    right_gated = torch.empty_like(right_proj)

    # Strides (in element units, not bytes)
    s_left = left_proj.stride()
    s_right = right_proj.stride()
    s_gate = left_gate.stride()          # gates share the same layout as proj
    s_mask = mask.stride()
    s_out_left = left_gated.stride()
    s_out_right = right_gated.stride()

    total_elements = B * N * N * hidden_dim
    BLOCK_SIZE = 1024
    grid = ( (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    _mask_gate_fusion_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left_gated,
        right_gated,
        B,
        N,
        hidden_dim,
        # strides for left
        s_left[0], s_left[1], s_left[2], s_left[3],
        # strides for right
        s_right[0], s_right[1], s_right[2], s_right[3],
        # strides for gates (identical layout as left/right)
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        # strides for mask (B,N,N)
        s_mask[0], s_mask[1], s_mask[2],
        # strides for output tensors
        s_out_left[0], s_out_left[1], s_out_left[2], s_out_left[3],
        s_out_right[0], s_out_right[1], s_out_right[2], s_out_right[3],
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # Core O(N³·H) multiplication:
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # Implemented as batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N)
    left_mat = left_gated.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_gated.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matrix multiply (float32).  // can be autocast to fp16 if desired
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm (over hidden dimension) + out‑gate
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate  # broadcast over hidden dim

    # ------------------------------------------------------------------
    # Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_final = torch.nn.functional.linear(
        out_norm, weights["to_out.weight"]
    )  # [B, N, N, dim]

    return out_final
