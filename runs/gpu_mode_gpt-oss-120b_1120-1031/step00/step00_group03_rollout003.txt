"""
TriMul "outgoing" kernel implementation.

The input tensor x (B, N, N, D) is layer‑normed, projected to a hidden
dimension H with two linear layers and gated. The core operation is the
einsum

    out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]

which can be rewritten as a batch of matrix multiplications
 left[:,:,k] @ right[:,:,k]^T for each hidden channel.  We fuse this
contraction into a Triton kernel that processes a batch of (B*H) square
matrices of shape (N,N).  The remainder of the forward pass (layer‑norms,
gates and final linear projection) is performed with PyTorch.

The kernel uses a tiling strategy (BLOCK_M, BLOCK_N, BLOCK_K) and
accumulates in FP32.  It works for all test configurations and
produces a float32 output of shape (B, N, N, D).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def batched_gemm_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # shape parameters
    batch, M, N, K,
    # strides for a
    stride_a_batch, stride_a_m, stride_a_k,
    # strides for b
    stride_b_batch, stride_b_k, stride_b_n,
    # strides for c
    stride_c_batch, stride_c_m, stride_c_n,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: for each b in [0, batch), compute C[b] = A[b] @ B[b]"""
    pid_b = tl.program_id(2)          # batch index
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # Offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch_ptr = a_ptr + pid_b * stride_a_batch
    b_batch_ptr = b_ptr + pid_b * stride_b_batch
    c_batch_ptr = c_ptr + pid_b * stride_c_batch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (K x N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Matrix multiply–accumulate
        acc += tl.dot(a, b)

    # Write back result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated implementation of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input, mask, weights, config)
        - input : torch.Tensor, shape [B, N, N, D]
        - mask  : torch.Tensor, shape [B, N, N] (may be all‑ones)
        - weights: dict with the layer parameters
        - config : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False)

    # ---------- 1. LayerNorm over the feature dimension ----------
    # LayerNorm expects normalized_shape = (dim,)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ---------- 2. Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # -> [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- 3. Optional mask ----------
    if use_mask:
        # mask shape (B,N,N) -> (B,N,N,1)
        mask_exp = mask.to(dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # ---------- 4. Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Core contraction via Triton ----------
    B, N, _, _ = left.shape

    # Re‑order to (B, hidden_dim, N, N) and make contiguous
    left_t = left.permute(0, 3, 1, 2).contiguous()
    # right transpose over the two N dimensions before flattening
    right_t = right.permute(0, 3, 2, 1).contiguous()

    # Collapse batch and hidden_dim -> total_mat = B * hidden_dim
    left_mat = left_t.view(-1, N, N)   # [BH, N, N]
    right_mat = right_t.view(-1, N, N) # [BH, N, N]

    BH = left_mat.shape[0]  # B * hidden_dim

    # Allocate output buffer
    out_mat = torch.empty_like(left_mat)

    # Tile sizes – chosen to fit H100 shared memory comfortably
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BH,                # batch count
        N, N, N,           # M, N, K (square)
        # strides for A
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        # strides for B
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        # strides for C
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 6. Output LayerNorm ----------
    out_norm = F.layer_norm(out,
                            (hidden_dim,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"],
                            eps=1e-5)

    # ---------- 7. Apply output gate ----------
    out_norm = out_norm * out_gate

    # ---------- 8. Final linear back to original dimension ----------
    output = F.linear(out_norm, weights["to_out.weight"])

    return output

"""
TriMul “outgoing” implementation (AlphaFold‑3 style).

The forward pass consists of

  1. Layer‑norm over the feature dimension `dim`.
  2. Two linear projections (`left_proj`, `right_proj`) + optional masking.
  3. Three sigmoidal gates (`left_gate`, `right_gate`, `out_gate`).
  4. Core contraction
         out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
     which can be rewritten as a batched matrix multiplication:
         for each hidden channel h:
             out[b,:,:,h] = left[b,:,:,h] @ right[b,:,:,h].T
     The batched GEMM is executed by a custom Triton kernel that
     processes a batch of `B*hidden_dim` square matrices of size `N×N`.
  5. Layer‑norm over the hidden dimension, multiplication with `out_gate`,
     and a final linear projection back to `dim`.

Only the heavy O(N³) contraction is written in Triton; the remaining
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def batched_gemm_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # size parameters
    batch, M, N, K,
    # strides for A (batch, M, K)
    stride_a_batch, stride_a_m, stride_a_k,
    # strides for B (batch, K, N)
    stride_b_batch, stride_b_k, stride_b_n,
    # strides for C (batch, M, N)
    stride_c_batch, stride_c_m, stride_c_n,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM:  C[b] = A[b] @ B[b]   (all matrices are square)."""
    pid_b = tl.program_id(2)          # which matrix in the batch
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this matrix in the batch
    a_batch_ptr = a_ptr + pid_b * stride_a_batch
    b_batch_ptr = b_ptr + pid_b * stride_b_batch
    c_batch_ptr = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K sized tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile (K × N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Matrix multiply‑accumulate
        acc += tl.dot(a, b)

    # Write the result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m) + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, D] (float32/float16)
        - mask         : torch.Tensor, shape [B, N, N] (may be all‑ones)
        - weights      : dict of model parameters (no bias on linear layers)
        - config       : dict containing ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1. unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # 2. Input LayerNorm over the feature dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if use_mask:
        mask_exp = mask.to(dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core contraction – batched GEMM in Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Rearrange to (B, hidden_dim, N, N) and make contiguous
    left_t = left.permute(0, 3, 1, 2).contiguous()
    # Right needs the two N axes swapped before flattening
    right_t = right.permute(0, 3, 2, 1).contiguous()

    # Collapse batch and channel dimensions:   BH = B * hidden_dim
    left_mat = left_t.view(-1, N, N)       # [BH, N, N]
    right_mat = right_t.view(-1, N, N)     # [BH, N, N]
    BH = left_mat.shape[0]

    # Output buffer
    out_mat = torch.empty_like(left_mat)

    # Tile sizes – chosen to fit comfortably in H100 shared memory
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # tiles along M
        triton.cdiv(N, BLOCK_N),   # tiles along N
        BH,                        # one grid entry per matrix in the batch
    )

    batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BH,                     # batch count
        N, N, N,                # M, N, K (square)
        # strides for A
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        # strides for B
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        # strides for C
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 8. Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return output
