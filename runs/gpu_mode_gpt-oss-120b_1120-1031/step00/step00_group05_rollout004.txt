import torch
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N: tl.int32,
    # strides: assume contiguous layout of shape (BC, N, N)
    # leading stride for batch = N * N
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bc = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offset_batch = pid_bc * N * N
    # Tile offsets within N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Masks for boundary
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers
    left_base = left_ptr + offset_batch
    right_base = right_ptr + offset_batch
    out_base = out_ptr + offset_batch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: rows=offs_m, cols=offs_k
        left_ptrs = left_base + offs_m[:, None] * N + offs_k[None, :]
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load right block transposed: rows=offs_k, cols=offs_n
        right_ptrs = right_base + offs_k[:, None] * N + offs_n[None, :]
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Dot product (M x K) * (K x N) -> (M x N)
        acc += tl.dot(left, right).to(tl.float32)

    out_ptrs = out_base + offs_m[:, None] * N + offs_n[None, :]

"""
TriMul (outgoing) – fast Triton implementation.

The module applies a LayerNorm + linear projections + gating to the input
tensor X ∈ ℝ^{B×N×N×C}.  After the projections we obtain two tensors
L,R ∈ ℝ^{B×N×N×H} (H = hidden_dim).  The core operation is the
tri‑multiplicative update

    out[b,i,j,h] = Σ_k L[b,i,k,h] · R[b,j,k,h]

i.e. a batched matrix‑multiply L_h @ R_hᵀ for every channel h.
The heavy O(B·H·N³) reduction is performed by a custom Triton kernel that
processes the (B·H) matrices in parallel, using block‑wise dot products
with FP16 inputs and FP32 accumulation.  The result is then normalised,
gated, and projected back to the original channel dimension C.

The kernel works for any batch size (1‑2), sequence length N up to 1024,
and hidden dimension H = 128 (as used in AlphaFold3).  A mask (optional)
is applied before gating.
"""

import torch
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]
# L and R are stored as (BC, N, N) where BC = B * H.
# The kernel treats each (b,h) pair as an independent matrix‑multiply.
# -------------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const half*  (BC, N, N)
    right_ptr,         # const half*  (BC, N, N)
    out_ptr,           #       float* (BC, N, N)
    N: tl.int32,       # sequence length
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_bc = tl.program_id(0)      # index over (batch * hidden_dim)
    pid_m  = tl.program_id(1)      # tile row i
    pid_n  = tl.program_id(2)      # tile col j

    # -----------------------------------------------------------------
    # Compute offsets for this (bc,i,j) tile
    # -----------------------------------------------------------------
    offset_batch = pid_bc * N * N                     # start of this matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M) # i‑indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N) # j‑indices
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this matrix (BC dimension already accounted for)
    left_base  = left_ptr  + offset_batch
    right_base = right_ptr + offset_batch
    out_base   = out_ptr   + offset_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in BLOCK_K sized chunks
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load a tile from L (M × K) ---------------------------------
        # offset: i * N + k
        left_ptrs = left_base + offs_m[:, None] * N + offs_k[None, :]
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)               # half

        # ---- load a tile from Rᵀ (K × N) ---------------------------------
        # we need R transposed: access element (k, j) → original (j, k)
        right_ptrs = right_base + offs_k[:, None] * N + offs_n[None, :]
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)              # half

        # ---- block‑wise dot product (M×K)·(K×N) -------------------------
        # tl.dot works on half inputs, result is half; cast to float32
        acc += tl.dot(left, right).to(tl.float32)

    # store the result tile
    out_ptrs = out_base + offs_m[:, None] * N + offs_n[None, :]
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# -------------------------------------------------------------------------
# Entry point used by the test harness
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config) where
            input_tensor : torch.Tensor  [B, N, N, C]
            mask          : torch.Tensor or None  [B, N, N]
            weights       : dict of torch.Tensor (model parameters)
            config        : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dev = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # layer‑norm on the input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    eps = 1e-5
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,dim]

    # -----------------------------------------------------------------
    # linear projections (no bias) and mask handling
    # -----------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (H, dim)
    right_proj_weight = weights["right_proj.weight"]  # (H, dim)

    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,H]

    # optional mask (mask shape [B,N,N] or None)
    if mask is None:
        mask = torch.ones(input_tensor.shape[:3],
                          dtype=input_tensor.dtype,
                          device=dev)
    mask = mask.to(input_tensor.dtype).unsqueeze(-1)   # [B,N,N,1]
    left  = left * mask
    right = right * mask

    # -----------------------------------------------------------------
    # gating networks (sigmoid)
    # -----------------------------------------------------------------
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))  # [B,N,N,H]

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Prepare tensors for the Triton kernel
    #   – bring the hidden dimension to the second axis (BC, N, N)
    #   – cast to half for the matmul kernel
    # -----------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    bc = B * hidden_dim
    left_bc  = left.view(bc, N, N).to(torch.float16)   # half
    right_bc = right.view(bc, N, N).to(torch.float16)  # half
    out_bc   = torch.empty((bc, N, N), dtype=torch.float32, device=dev)

    # -----------------------------------------------------------------
    # launch Triton kernel
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        bc,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    trimul_outgoing_kernel[grid](
        left_bc,
        right_bc,
        out_bc,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # synchronize to make sure the kernel is finished before next ops
    torch.cuda.synchronize()

    # -----------------------------------------------------------------
    # Reshape back to (B, N, N, H)
    # -----------------------------------------------------------------
    out = out_bc.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # -----------------------------------------------------------------
    # post‑processing: norm → gate → final linear
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))  # [B,N,N,H]

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate  # apply output gate

    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    return out
