    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    # mask
    nomask = config.get("nomask", False)

        mask = torch.ones((B, N, N), dtype=input_tensor.dtype, device=input_tensor.device)
    else:

    left_fused, right_fused = _apply_mask_and_gate_fuse(left_proj, right_proj, left_gate, right_gate, mask, hidden_dim)
    # Compute out via batched matmul
    # reshape to (B, hidden, N, N)
    left_fused_t = left_fused.permute(0, 3, 1, 2).contiguous()
    right_fused_t = right_fused.permute(0, 3, 1, 2).contiguous()
    B_, H_, N_, _ = left_fused_t.shape
    left_flat = left_fused_t.view(B_ * H_, N_, N_)
    right_flat = right_fused_t.view(B_ * H_, N_, N_)
    # bmm
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))
    # reshape back to (B,N,N,hidden)

        bias=weights["to_out_norm.bias"],

    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
Triton‑accelerated forward pass of the “outgoing” TriMul block (AlphaFold3).

Algorithm
---------
1️⃣ Layer‑normalize the 4‑D input  (B,N,N,C) using the provided LN weights.
2️⃣ Linear projections → left_proj, right_proj (C → H) and three gates
   (left_gate, right_gate, out_gate) with sigmoid activation.
3️⃣ Fuse mask multiplication, left‑gate and right‑gate application in a
   custom Triton kernel (elementwise: out = proj * gate * mask).
4️⃣ Compute the pairwise multiplicative update:
        out_{i,j,d} = Σ_k left[i,k,d] * right[j,k,d]
   This is a batched matrix‑multiply for each hidden channel d.
   We reshape to (B·H, N, N) and use `torch.bmm`, which maps to cuBLAS
   Tensor‑core kernels on an H100.
5️⃣ Layer‑norm the result over the hidden dimension, multiply by `out_gate`,
   and apply the final output linear projection (H → C).

Only the mask‑gate fusion is implemented in Triton; the rest uses
high‑performance PyTorch primitives.  All tensors stay on the GPU and the
final output is float32.
"""

import torch
import triton
import triton.language as tl

# ------------------------------------------------------------------------
# Triton kernel: fuse mask, left/right gates with the projected tensors.
# ------------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                # float32 *
    right_ptr,               # float32 *
    left_gate_ptr,           # float32 *
    right_gate_ptr,          # float32 *
    mask_ptr,                # float32 *   (shape: B*N*N)
    out_left_ptr,            # float32 *
    out_right_ptr,           # float32 *
    total_elems,             # int64  total number of elements in left/right (B*N*N*H)
    hidden_dim,              # int64  H
    BLOCK_SIZE: tl.constexpr # compile‑time block size
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # guard for out‑of‑range threads
    mask = offs < total_elems

    # each hidden channel repeats the same mask entry
    mask_idx = offs // hidden_dim

    # load values
    left   = tl.load(left_ptr   + offs, mask=mask, other=0.0)
    right  = tl.load(right_ptr  + offs, mask=mask, other=0.0)
    l_gate = tl.load(left_gate_ptr  + offs, mask=mask, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m_val  = tl.load(mask_ptr   + mask_idx, mask=mask, other=0.0)

    # fused multiply‑add
    out_left  = left * l_gate * m_val
    out_right = right * r_gate * m_val

    # store results
    tl.store(out_left_ptr  + offs, out_left,  mask=mask)
    tl.store(out_right_ptr + offs, out_right, mask=mask)


def _apply_mask_and_gate_fuse(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Apply `mask * gate` to `left` and `right` using a Triton kernel.

    Arguments
    ---------
    left, right      : (B,N,N,H) tensors (float32)
    left_gate, right_gate : (B,N,N,H) tensors (float32)
    mask             : (B,N,N) tensor (float32) – already broadcast‑compatible
    hidden_dim       : int (== H)

    Returns
    -------
    left_fused, right_fused : (B,N,N,H) tensors
    """
    assert left.is_contiguous() and right.is_contiguous()
    assert left_gate.is_contiguous() and right_gate.is_contiguous()
    assert mask.is_contiguous()

    total = left.numel()                # B * N * N * H
    out_left  = torch.empty_like(left)
    out_right = torch.empty_like(right)

    # Typical block size for 1‑D kernels; 1024 gives good occupancy.
    BLOCK_SIZE = 1024
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask,
        out_left,
        out_right,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return out_left, out_right


# ------------------------------------------------------------------------
# Public entry point
# ------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]  (float32, on CUDA)
        - mask_tensor  : torch.Tensor [B, N, N]    (bool/float, on CUDA) or None
        - weights_dict : dict of parameter tensors
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor : [B, N, N, C] (float32, on CUDA)
    """
    # --------------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the 4‑D tensor
    # --------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------------
    # 2️⃣ Linear projections (C → H) – bias‑free
    # --------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid) – also C → H
    # --------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # --------------------------------------------------------------------
    # 4️⃣ Prepare mask (float32) – broadcasted over the hidden dimension
    # --------------------------------------------------------------------
    if mask_tensor is None or nomask:
        mask = torch.ones((B, N, N), dtype=x.dtype, device=device)
    else:
        mask = mask_tensor.to(dtype=x.dtype, device=device)
    mask = mask.contiguous()

    # --------------------------------------------------------------------
    # 5️⃣ Fuse mask * gate multiplication via Triton
    # --------------------------------------------------------------------
    left_fused, right_fused = _apply_mask_and_gate_fuse(
        left_proj, right_proj, left_gate, right_gate, mask, hidden_dim
    )

    # --------------------------------------------------------------------
    # 6️⃣ Pairwise multiplicative update (einsum) → batched GEMM per hidden dim
    # --------------------------------------------------------------------
    # reshape to (B, H, N, N) to treat H as a batch dimension
    left_t = left_fused.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right_fused.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    B_, H_, N_, _ = left_t.shape
    left_flat = left_t.view(B_ * H_, N_, N_)      # (B*H, N, N)
    right_flat = right_t.view(B_ * H_, N_, N_)    # (B*H, N, N)

    # Batched matrix multiplication: Σ_k left[i,k,d] * right[j,k,d]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------
    # 7️⃣ Output layer‑norm over hidden dimension
    # --------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------------
    # 8️⃣ Apply final output gate
    # --------------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------------
    # 9️⃣ Final linear (H → C)
    # --------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
