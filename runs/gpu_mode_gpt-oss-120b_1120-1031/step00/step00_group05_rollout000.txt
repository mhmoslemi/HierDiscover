"""
TriMul (outgoing) – fast Triton implementation.

Given a 4‑D pair‑wise tensor X ∈ ℝ^{B×N×N×C},
the outgoing TriMul computes for each hidden channel d
    out_{b,i,j,d} = Σ_{k=0}^{N-1} L_{b,i,k,d} · R_{b,j,k,d}
where L = (X·W_left) ⊙ σ(X·W_left_gate) and
      R = (X·W_right) ⊙ σ(X·W_right_gate).
A mask (optional) is applied before gating.
The heavy N³ reduction is performed by a fused Triton kernel that
tiles the i‑ and j‑axes (BLOCK_M×BLOCK_N) and reduces over k in
BLOCK_K‑sized chunks.  The hidden dimension is tiled with BLOCK_D=1,
so each program instance computes a 2‑D tile for a single hidden
channel.  Afterwards a LayerNorm, an output gate, and a final linear
projection are applied in PyTorch.

The kernel works for any batch size (B≤2), sequence length N (≤1024) and
hidden size H (≤384).  All tensors are expected to be on the CUDA device.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,                 # pointers
    N, H,                                         # runtime sizes
    stride_bl, stride_i, stride_k, stride_h,      # left strides
    stride_br, stride_j, stride_kr, stride_hr,    # right strides
    stride_bo, stride_i_out, stride_j_out, stride_h_out,  # output strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
       where d iterates in blocks of size BLOCK_D (here =1).
    """
    pid_i = tl.program_id(0)          # tile over i dimension
    pid_j = tl.program_id(1)          # tile over j dimension
    pid_z = tl.program_id(2)          # combined batch & hidden‑tile dimension

    # ---- decode batch and hidden index ---------------------------------
    num_hidden_tiles = tl.cdiv(H, BLOCK_D)               # = H when BLOCK_D=1
    batch_id = pid_z // num_hidden_tiles
    hidden_tile = pid_z % num_hidden_tiles
    d = hidden_tile * BLOCK_D                            # hidden channel (scalar)

    # ---- offsets for the tile -------------------------------------------
    i_offsets = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_in_bounds = i_offsets < N
    j_in_bounds = j_offsets < N

    # ---- base pointers (account for batch and hidden channel) -----------
    left_base  = left_ptr  + batch_id * stride_bl  + d * stride_h
    right_base = right_ptr + batch_id * stride_br  + d * stride_hr
    out_base   = out_ptr   + batch_id * stride_bo  + d * stride_h_out

    # ---- accumulator ----------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- reduction over the k dimension ---------------------------------
    for k in range(0, N, BLOCK_K):
        k_offsets = k + tl.arange(0, BLOCK_K)
        k_in_bounds = k_offsets < N

        # left block : shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (i_offsets[:, None] * stride_i) + (k_offsets[None, :] * stride_k)
        left = tl.load(left_ptrs,
                       mask=i_in_bounds[:, None] & k_in_bounds[None, :],
                       other=0.0).to(tl.float32)

        # right block (transposed) : shape (BLOCK_K, BLOCK_N)
        # note: we load right[b, j, k, d] but arrange as (k, j)
        right_ptrs = right_base + (k_offsets[:, None] * stride_kr) + (j_offsets[None, :] * stride_j)
        right = tl.load(right_ptrs,
                        mask=k_in_bounds[:, None] & j_in_bounds[None, :],
                        other=0.0).to(tl.float32)

        # matrix multiplication for this k‑chunk
        acc += tl.dot(left, right)          # (BLOCK_M, BLOCK_N) result

    # ---- write back ------------------------------------------------------
    out_ptrs = out_base + (i_offsets[:, None] * stride_i_out) + (j_offsets[None, :] * stride_j_out)
    tl.store(out_ptrs, acc, mask=i_in_bounds[:, None] & j_in_bounds[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-6,
    )

    # ------------------------------------------------------------------
    # Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (mask is shape [B,N,N]; when `nomask` == True it can be None)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        # ensure same dtype and broadcast over the feature dimension
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)            # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel for the N³ reduction
    # ------------------------------------------------------------------
    # ensure contiguous layout for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)               # [B,N,N,hidden_dim]

    # tile sizes – tuned for H100
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64
    BLOCK_D = 1   # one hidden channel per program

    # compute grid dimensions
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    hidden_tiles = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    grid_z = B * hidden_tiles
    grid = (grid_i, grid_j, grid_z)

    # stride information (int64 values)
    stride_bl, stride_i, stride_k, stride_h = left.stride()
    stride_br, stride_j, stride_kr, stride_hr = right.stride()
    stride_bo, stride_i_out, stride_j_out, stride_h_out = out.stride()

    # launch kernel
    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        N,
        hidden_dim,
        stride_bl,
        stride_i,
        stride_k,
        stride_h,
        stride_br,
        stride_j,
        stride_kr,
        stride_hr,
        stride_bo,
        stride_i_out,
        stride_j_out,
        stride_h_out,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Output LayerNorm + gating + final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-6,
    )
    out = out * out_gate                         # element‑wise gating
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
