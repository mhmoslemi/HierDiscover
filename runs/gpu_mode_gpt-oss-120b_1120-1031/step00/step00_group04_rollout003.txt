def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']

    # compute normalized input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # linear projections
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])

    # apply mask if needed - mask shape [B,N,N], broadcast to hidden_dim
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype) # broadcast
        left = left * mask_exp
        right = right * mask_exp

    # compute gates
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight']).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # convert to half for kernel
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # allocate output
    out_fp32 = torch.empty_like(left_h, dtype=torch.float32, device=device)  # shape B,N,N,hidden_dim

    # launch triton kernel
    # define block sizes: we can pick 64,64,32 typical values for H100 for N up to 1024
    TILE_M = 64
    TILE_N = 64
    TILE_K = 32
    # hidden block size = 1
    BLOCK_H = 1

    B, N, _, H = left_h.shape

    # compute strides
    stride_bl, stride_il, stride_kl, stride_hl = left_h.stride()
    stride_br, stride_jr, stride_kr, stride_hr = right_h.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out_fp32.stride()

    # grid dimensions
    grid_x = (N + TILE_M - 1) // TILE_M
    grid_y = (N + TILE_N - 1) // TILE_N
    hidden_blocks = (H + BLOCK_H - 1) // BLOCK_H

        stride_bo, stride_io, stride_jo, stride_ho,

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    H_TILE_SIZE: tl.constexpr):
    
    pid_m = tl.program_id(0) # tile row i
    pid_n = tl.program_id(1) # tile col j
    pid_z = tl.program_id(2) # batch*hidden tile

    # compute batch index and hidden tile index
    hidden_tiles_per_batch = tl.cdiv(H, H_TILE_SIZE)
    batch_idx = pid_z // hidden_tiles_per_batch
    hidden_tile_idx = pid_z % hidden_tiles_per_batch
    # start indices
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N
    h_start = hidden_tile_idx * H_TILE_SIZE

    # compute # of valid elements
    # Offsets for i,j axes
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    # mask for i,j inside N
    i_mask = i_range < N
    j_mask = j_range < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_start in range(0, N, BLOCK_K):
        # K indices for this tile
        k_range_block = k_start + k_range
        k_mask = k_range_block < N

        # compute mask for left and right loads
        mask_left = i_mask[:, None] & k_mask[None, :]
        mask_right = k_mask[:, None] & j_mask[None, :]

        # left pointer offsets for shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            batch_idx * stride_bl + \
            (i_range[:, None] * stride_il) + \
            (k_range_block[None, :] * stride_kl) + \
            (h_start * stride_hl)

        # right pointer offsets for shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            batch_idx * stride_br + \
            (k_range_block[:, None] * stride_kr) + \
            (j_range[None, :] * stride_jr) + \
            (h_start * stride_hr)

        # load
        a = tl.load(left_ptrs, mask=mask_left, other=0.0, dtype=tl.float16)
        b = tl.load(right_ptrs, mask=mask_right, other=0.0, dtype=tl.float16)

        # dot product
        acc += tl.dot(a, b)   # result float32

    # store to output
    out_ptrs = out_ptr + \
        batch_idx * stride_bo + \
        (i_range[:, None] * stride_io) + \
        (j_range[None, :] * stride_jo) + \
        (h_start * stride_ho)

    # mask for output store
    mask_out = i_mask[:, None] & j_mask[None, :]

    tl.store(out_ptrs, acc, mask=mask_out)

"""
TriMul (outgoing) – Triton implementation of the expensive
pair‑wise multiplication in AlphaFold3.

Algorithm
---------
1. Layer‑norm the input tensor (B, N, N, C).
2. Linear projections → left / right (shape (B,N,N,H)).
3. Optional mask (broadcasted on the hidden dimension).
4. Gating: sigmoid(linear(...)) and element‑wise gating of left/right.
5. Core operation:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiply per hidden channel.
   Implemented with a custom Triton kernel that tiles the (i,j) output
   plane (BLOCK_M × BLOCK_N) and reduces over K in BLOCK_K steps.
   The hidden dimension is tiled with BLOCK_H = 1 (one channel per
   program).  The kernel works in FP16 for the inputs and accumulates
   in FP32.
6. Layer‑norm over the hidden dimension, gating, and final linear
   projection back to the original channel size.
The code uses only the forward pass – gradients are not required.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # problem sizes
    stride_bl, stride_il, stride_kl, stride_hl, # left strides
    stride_br, stride_jr, stride_kr, stride_hr, # right strides
    stride_bo, stride_io, stride_jo, stride_ho, # out strides
    hid_tiles_per_batch,                        # int = ceil(H / BLOCK_H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,                      # =1 (one hidden channel per program)
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)   # tile row   (i)
    pid_n = tl.program_id(1)   # tile col   (j)
    pid_z = tl.program_id(2)   # combined batch & hidden tile

    # --------------------------------------------------------------
    # Decode batch index and hidden‑tile index
    # --------------------------------------------------------------
    batch_idx = pid_z // hid_tiles_per_batch
    hidden_tile_idx = pid_z % hid_tiles_per_batch
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N
    h_start = hidden_tile_idx * BLOCK_H          # BLOCK_H = 1 ⇒ scalar

    # --------------------------------------------------------------
    # Index ranges for this tile
    # --------------------------------------------------------------
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    i_mask = i_range < N
    j_mask = j_range < N

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        k_offset = k_start + k_range
        k_mask = k_offset < N

        # ---- left (shape BLOCK_M × BLOCK_K) ----
        left_ptrs = (
            left_ptr
            + batch_idx * stride_bl
            + i_range[:, None] * stride_il
            + k_offset[None, :] * stride_kl
            + h_start * stride_hl
        )
        mask_left = i_mask[:, None] & k_mask[None, :]
        a = tl.load(left_ptrs, mask=mask_left, other=0.0, dtype=tl.float16)

        # ---- right (shape BLOCK_K × BLOCK_N) ----
        # note the transposition: we need right[j,k] → we load (k, j)
        right_ptrs = (
            right_ptr
            + batch_idx * stride_br
            + k_offset[:, None] * stride_kr
            + j_range[None, :] * stride_jr
            + h_start * stride_hr
        )
        mask_right = k_mask[:, None] & j_mask[None, :]
        b = tl.load(right_ptrs, mask=mask_right, other=0.0, dtype=tl.float16)

        # ---- dot product accumulation (FP32) ----
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Write back the result (FP32)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch_idx * stride_bo
        + i_range[:, None] * stride_io
        + j_range[None, :] * stride_jo
        + h_start * stride_ho
    )
    mask_out = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float)
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack and prepare
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over the last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask may be None when config["nomask"] is True)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=left.dtype, device=device).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy multiplication
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 6. Allocate output tensor (FP32) for the kernel result
    # ------------------------------------------------------------------
    out = torch.empty_like(left_h, dtype=torch.float32, device=device)  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 7. Triton kernel launch
    # ------------------------------------------------------------------
    B, N, _, H = left_h.shape
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32   # tuned for H100 FP16 matmul
    BLOCK_H = 1                               # one hidden channel per program

    # Strides (in elements, not bytes)
    stride_bl, stride_il, stride_kl, stride_hl = left_h.stride()
    stride_br, stride_jr, stride_kr, stride_hr = right_h.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    hidden_tiles = (H + BLOCK_H - 1) // BLOCK_H
    grid_x = (N + BLOCK_M - 1) // BLOCK_M
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * hidden_tiles

    tri_mul_kernel[(grid_x, grid_y, grid_z)](
        left_h,
        right_h,
        out,
        B,
        N,
        H,
        stride_bl,
        stride_il,
        stride_kl,
        stride_hl,
        stride_br,
        stride_jr,
        stride_kr,
        stride_hr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_ho,
        hidden_tiles,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        BLOCK_H,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 8. Post‑processing: LayerNorm over hidden_dim, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # (B,N,N,hidden_dim)

    out = out * out_gate                       # gating (FP32)

    # final projection back to original channel size
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    return out
```
