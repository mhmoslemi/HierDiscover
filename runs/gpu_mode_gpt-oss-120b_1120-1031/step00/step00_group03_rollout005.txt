analysisThe task: we need to translate the forward pass of TriMul module into a custom kernel that uses Triton for at least part of the operations. The problem states to implement "outgoing" version of the TriMul operator from the AlphaFold3 paper. The provided reference code for TriMul is a module that does:

- Input x: shape [B, N, N, dim]
- Apply layer norm (norm on dim).
- left = left_proj(x)   -> shape [B,N,N,hidden_dim]
- right = right_proj(x) -> shape same
- mask: shape [B,N,N]; unsqueeze to ...1 to broadcast to dim.
- left = left * mask
- right = right * mask

- left_gate = sigmoid(left_gate(x)) (size [B,N,N,hidden_dim]) same as left.
- right_gate = sigmoid(right_gate(x))
- out_gate = sigmoid(out_gate(x))  shape [B,N,N,hidden_dim] (actually out_gate will later applied after to_out_norm; but note config says out_gate weight is linear projecting original x dim-> hidden_dim, applied with sigmoid, then multiplied after to_out_norm.

- left = left * left_gate
- right = right * right_gate

- out = einsum('... i k d, ... j k d -> ... i j d', left, right). Which is: for each b, i, j, d: sum over k (0..N-1) left[b,i,k,d] * right[b,j,k,d].

So the outer product shape is [B,N,N, hidden_dim].

- Then apply to_out_norm (layernorm on hidden_dim)
- out = out * out_gate
- out = to_out(out) -> shape [B,N,N,dim]

Thus the final shape matches input shape.

                    stride_right_j, stride_right_k, stride_right_h,
                    stride_out_i, stride_out_j, stride_out_h,

    j0 = j_block * BLOCK_J

    mask_j = offs_j < N

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    # assume dtype float32
    x = input_tensor

    # layernorm
    x = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # mask
    if mask is not None:
        # mask type may be bool or float
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp
    # gating
    left = left * left_gate
    right = right * right_gate

    # allocate out intermediate
    B, N, _, _ = left.shape
    out_inter = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=left.device)

    # get strides in number of elements
    l_strides = left.stride()
    r_strides = right.stride()
    o_strides = out_inter.stride()
    # Convert strides to number of elements (already number-of-elements)
    # confirm dtype size: use .element_size() later in pointer offsets.

    # Kernel launch
    import triton
    import triton.language as tl

    @triton.jit
    def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                       N, H,
                       stride_left_i, stride_left_k, stride_left_h,
                       stride_right_j, stride_right_k, stride_right_h,
                       stride_out_i, stride_out_j, stride_out_h,
                       BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
                       BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr):
        pid = tl.program_id(0)
        num_i = tl.cdiv(N, BLOCK_I)
        i_block = pid % num_i
        j_block = pid // num_i
        i0 = i_block * BLOCK_I
        j0 = j_block * BLOCK_J

        # range offsets
        range_i = tl.arange(0, BLOCK_I)
        range_j = tl.arange(0, BLOCK_J)
        range_h = tl.arange(0, BLOCK_H)

        # offsets for constant axes
        offs_i = i0 + range_i[:, None, None]   # (BLOCK_I,1,1)
        offs_j = j0 + range_j[None, :, None]   # (1,BLOCK_J,1)
        offs_h = range_h[None, None, :]        # (1,1,BLOCK_H)

        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_h = offs_h < H

        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # k loop
        for k0 in range(0, N, BLOCK_K):
            cur_k = k0 + tl.arange(0, BLOCK_K)
            mask_k = cur_k < N

            # left tile: (i,k,h)
            left_ptrs = left_ptr + offs_i * stride_left_i + cur_k[None, :, None] * stride_left_k + offs_h * stride_left_h
            left_mask = mask_i & mask_k[:, None, None] & mask_h
            left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

            # right tile: (k,j,h)
            right_ptrs = right_ptr + cur_k[:, None, None] * stride_right_k + offs_j * stride_right_j + offs_h * stride_right_h
            right_mask = mask_k[:, None, None] & mask_j & mask_h
            right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

            # accumulate dot per hidden slice
            for h_idx in tl.static_range(0, BLOCK_H):
                # left_slice: (BLOCK_I, BLOCK_K)
                left_slice = left_tile[:, :, h_idx]
                # right_slice: (BLOCK_K, BLOCK_J)
                right_slice = right_tile[:, :, h_idx]
                acc[:, :, h_idx] += tl.dot(left_slice, right_slice)

        # store
        out_ptrs = out_ptr + offs_i * stride_out_i + offs_j * stride_out_j + offs_h * stride_out_h
        out_mask = mask_i & mask_j & mask_h
        tl.store(out_ptrs, acc, mask=out_mask)

    # Compute grid
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 16  # choose 16; covers up to 128/384 hidden dims with loops

    grid_i = (N + BLOCK_I - 1) // BLOCK_I
    grid_j = (N + BLOCK_J - 1) // BLOCK_J
    grid = (grid_i * grid_j,)

    # pointer offset per batch
    for b in range(B):
        left_ptr = left.data_ptr() + b * l_strides[0] * left.element_size()
        right_ptr = right.data_ptr() + b * r_strides[0] * right.element_size()
        out_ptr = out_inter.data_ptr() + b * o_strides[0] * out_inter.element_size()
        tri_mul_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            hidden_dim,
            l_strides[1],
            l_strides[2],
            l_strides[3],
            r_strides[1],
            r_strides[2],
            r_strides[3],
            o_strides[1],
            o_strides[2],
            o_strides[3],
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H
        )
    # Wait for kernels to finish
    torch.cuda.synchronize()

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(out_inter, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    # multiply out_gate
    out_norm = out_norm * out_gate
    # final linear
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return out

"""
custom_kernel
==============

Implements the **outgoing** TriMul operator (AlphaFold3) using a hybrid
PyTorch/Triton approach.

Algorithm
---------
1. Layer‑norm the input `x` on the feature dimension (`dim`).
2. Linear projections → tensors `left` and `right` of shape
   `[B, N, N, H]` (`H = hidden_dim`).
3. Optional binary `mask` is broadcasted and applied to both tensors.
4. Gated projections:
   * `left_gate  = sigmoid(Linear(x))`
   * `right_gate = sigmoid(Linear(x))`
   * `out_gate   = sigmoid(Linear(x))`
   * `left  = left * left_gate`
   * `right = right * right_gate`
5. **Heavy N³ multiplication**  
   Compute for each batch `b`, output tile `(i0:i0+Bi, j0:j0+Bj, h0:h0+Bh)` :

   ```
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   ```

   This is a fused “matrix‑multiply‑over‑k” for all hidden channels.
   The computation is performed by `tri_mul_kernel` written in Triton,
   tiling the `i`, `j` and hidden (`h`) dimensions while looping over
   the `k` dimension in blocks.  The kernel accumulates in FP32
   (inputs may be FP16/F32) and writes the result back in FP32.
6. Layer‑norm the intermediate result (`out`) on the hidden dimension,
   multiply with `out_gate`, and apply the final linear projection
   back to `dim` features.

Only the O(N³) part is executed on GPU via Triton; everything else uses
standard PyTorch operations (already highly optimized).  The implementation
supports both the “nomask” and masked variants, any batch size ≤ 2, and
hidden dimensions up to 384.

The kernel is written for Triton 3.3.1 and targets NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # const float*  shape [B,i,k,h]
    right_ptr,         # const float*  shape [B,j,k,h]
    out_ptr,           #       float*  shape [B,i,j,h]
    N,                 # seq_len (int32)
    H,                 # hidden_dim (int32)
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_j, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute
        out[i, j, h] = Σ_k left[i, k, h] * right[j, k, h]
    for a single batch (batch dimension is handled outside the kernel).
    The kernel tiles the i, j and hidden (h) axes; k is reduced in a
    blocked loop.
    """
    # --------------- program ids (tile indices) -----------------
    i_block = tl.program_id(0)                     # tile index along I
    j_block = tl.program_id(1)                     # tile index along J
    h_block = tl.program_id(2)                     # tile index along hidden

    i0 = i_block * BLOCK_I
    j0 = j_block * BLOCK_J
    h0 = h_block * BLOCK_H

    # --------------- index ranges -------------------------------
    range_i = tl.arange(0, BLOCK_I)
    range_j = tl.arange(0, BLOCK_J)
    range_k = tl.arange(0, BLOCK_K)
    range_h = tl.arange(0, BLOCK_H)

    offs_i = i0 + range_i[:, None, None]               # (Bi,1,1)
    offs_j = j0 + range_j[None, :, None]               # (1,Bj,1)
    offs_h = h0 + range_h[None, None, :]               # (1,1,Bh)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator for a (Bi, Bj, Bh) tile
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------- reduction over k -----------------------
    for k0 in range(0, N, BLOCK_K):
        cur_k = k0 + range_k                               # (Bk)
        mask_k = cur_k < N

        # -------- left tile (i, k, h) -------------------------
        left_ptrs = left_ptr + \
                    offs_i * stride_left_i + \
                    cur_k[None, :, None] * stride_left_k + \
                    offs_h * stride_left_h
        left_mask = mask_i & mask_k[:, None, None] & mask_h
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # -------- right tile (k, j, h) ------------------------
        right_ptrs = right_ptr + \
                     cur_k[:, None, None] * stride_right_k + \
                     offs_j * stride_right_j + \
                     offs_h * stride_right_h
        right_mask = mask_k[:, None, None] & mask_j & mask_h
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # ---- dot‑product for every hidden slice in the tile ---
        for hh in tl.static_range(0, BLOCK_H):
            # (Bi, Bk) x (Bk, Bj) -> (Bi, Bj)
            lhs = left_tile[:, :, hh]          # (Bi, Bk)
            rhs = right_tile[:, :, hh]         # (Bk, Bj)
            acc[:, :, hh] += tl.dot(lhs, rhs)  # accumulate in FP32

    # -------------------- write back ---------------------------
    out_ptrs = out_ptr + \
               offs_i * stride_out_i + \
               offs_j * stride_out_j + \
               offs_h * stride_out_h
    out_mask = mask_i & mask_j & mask_h
    tl.store(out_ptrs, acc, mask=out_mask)


def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing version).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        # mask has shape [B, N, N]; make it [B, N, N, 1] for broadcasting
        mask_exp = mask_tensor.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ensure contiguous memory for the kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 5. Triton kernel for the N³ multiplicative update
    # ------------------------------------------------------------------
    # allocate intermediate result (shape [B, N, N, hidden_dim])
    out_inter = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=left.device)

    # strides in **elements** (not bytes)
    l_str = left.stride()   # (B, i, k, h)
    r_str = right.stride()  # (B, j, k, h)
    o_str = out_inter.stride()

    # kernel launch configuration
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 16                         # hidden‑dim tiling factor

    grid_i = (N + BLOCK_I - 1) // BLOCK_I
    grid_j = (N + BLOCK_J - 1) // BLOCK_J
    grid_h = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid = (grid_i, grid_j, grid_h)

    # launch one kernel per batch (B ≤ 2, cheap loop)
    for b in range(B):
        left_ptr  = left.data_ptr()  + b * l_str[0]  * left.element_size()
        right_ptr = right.data_ptr() + b * r_str[0] * right.element_size()
        out_ptr   = out_inter.data_ptr() + b * o_str[0] * out_inter.element_size()

        tri_mul_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            hidden_dim,
            l_str[1], l_str[2], l_str[3],
            r_str[1], r_str[2], r_str[3],
            o_str[1], o_str[2], o_str[3],
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # make sure the kernel finished before proceeding
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 6. Post‑processing: layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_inter,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final projection → [B,N,N,dim]

    return out
