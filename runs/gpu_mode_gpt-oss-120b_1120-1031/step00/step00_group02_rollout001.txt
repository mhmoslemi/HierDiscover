@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(0)  # batch*H index
    b = pid_z // H
    h = pid_z % H

    pid_i = tl.program_id(1)  # block row index
    pid_j = tl.program_id(2)  # block col index

    # Compute start offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute offsets for left and right
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: left[b,i,k,h], shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_j)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile (will be transposed): right[b,j,k,h] -> need right[j,k] so that we afterwards dot left * right, sum over k.
        # Since we want right_T (k,j) we can load right[b, j, k, h] and then do dot: left (i,k) dot right (j,k) across k.

        right_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + (offs_j[:, None] * stride_right_i) + (offs_k[None, :] * stride_right_j)
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Compute: sum over k left[i,k] * right[j,k] -> matrix multiplication using dot product
        # Use tl.dot with reduction dimension BLOCK_K
        acc += tl.dot(left, right, allow_tf32=False)  # shape (BLOCK_M, BLOCK_N) 

        # Increment offsets automatically via loop index not needed as we computed taking absolute offsets each iteration.

    # After loop, store result
    out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16) if out_ptr.dtype == tl.float16 else acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – fast Triton implementation

The module computes for each batch `b` and hidden channel `h`

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where

    left  = (x @ W_left )            * mask   * σ( x @ W_left_gate )
    right = (x @ W_right)            * mask   * σ( x @ W_right_gate )
    out_gate = σ( x @ W_out_gate )

All LayerNorms use the supplied weight / bias tensors.
The heavy O(N³·H) part (`out = left @ rightᵀ`) is performed by a
custom Triton kernel.  The rest (LayerNorm, linear layers, sigmoid,
masking) is done with regular PyTorch ops (they are cheap compared to
the cubic term).  The kernel works on a tile of `BLOCK_M × BLOCK_N`
output elements for a single batch & hidden channel and accumulates
over `k` in tiles of size `BLOCK_K`.  It is fully vectorised, uses
float‑32 accumulation and stores the result in float‑32.

The kernel is launched with a 3‑D grid:
    dim0 = B * H        (batch × hidden channel)
    dim1 = ceil(N/BLOCK_M)   (output‑row tiles)
    dim2 = ceil(N/BLOCK_N)   (output‑col tiles)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # const *float32
    right_ptr,         # const *float32
    out_ptr,           # *float32
    B, N, H,           # int32  (batch, seq_len, hidden_dim)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    # 1) program ids
    # --------------------------------------------------------------
    pid_z = tl.program_id(0)               # batch * hidden
    pid_i = tl.program_id(1)               # row tile
    pid_j = tl.program_id(2)               # col tile

    # decode batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # --------------------------------------------------------------
    # 2) tile coordinates
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) loop over the reduction dimension k
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left[i,k] -------------------------------------------------
        # left layout: (b, i, k, h)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_j
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- load rightᵀ[k,j] (i.e. right[j,k]) -----------------------------
        # right layout: (b, j, k, h) when viewed as transposed
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_j   # k → first dimension
            + offs_j[None, :] * stride_right_i   # j → second dimension
        )
        right_t = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- rank‑k update --------------------------------------------------
        acc += tl.dot(left, right_t)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # 4) write back the tile
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of parameter tensors (no gradients needed)
            - config: dict containing at least "dim" and "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0) unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1) LayerNorm over the feature dimension
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) and gating preprocessors
    # --------------------------------------------------------------
    # ---- projections -------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ---- gating (sigmoid) --------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_unsq = mask.unsqueeze(-1).to(dtype)          # (B, N, N, 1)
        left = left * mask_unsq
        right = right * mask_unsq

    # --------------------------------------------------------------
    # 4) Apply gates (element‑wise)
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Compute the cubic term: out_mul = left @ rightᵀ   (per hidden channel)
    # --------------------------------------------------------------
    # allocate output buffer for the intermediate hidden‑dim tensor
    out_mul = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ensure contiguous layout for simple stride extraction
    left = left.contiguous()
    right = right.contiguous()
    out_mul = out_mul.contiguous()

    # strides (int64) – order: (batch, seq_i, seq_j, hidden)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out_mul.stride()

    # choose a reasonable tile size (tuned for H100, FP32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _trimul_outgoing_kernel[grid](
        left,
        right,
        out_mul,
        B,
        N,
        hidden_dim,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # enough warps for these tile sizes on H100
    )

    # --------------------------------------------------------------
    # 6) Post‑processing: LayerNorm over hidden, gating, final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out_mul,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                                   # broadcast over hidden
    out = F.linear(out, weights["to_out.weight"])          # (dim, hidden_dim) weight

    return out
