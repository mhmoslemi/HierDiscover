"""
propose_candidate – diversified search for a low‑value step function

The routine builds a pool of simple parametric families (exponential,
power‑law, Dirichlet draws) and scans a small grid of their parameters.
The few best families are refined with three stages:

   * Sub‑gradient “exponentiated‑gradient” mirror descent on the exact
     max‑convolution (fast, many iterations).
   * A continuation L‑BFGS optimiser that minimises a smooth
     log‑sum‑exp surrogate of the max‑convolution.
   * A short permutation‑annealing phase that only re‑orders the final
     multiset (swap / short‑segment reversal moves).

All phases respect the 1000 s overall budget and store the best sequence
found in the global ``height_sequence_1`` variable for the next run.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple families for a quick “good enough’’ start
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)   (i = 0..n‑1)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _dirichlet_sample(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.gamma(alpha, 1.0, size=n)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _mix(seq1: np.ndarray, seq2: np.ndarray, w: float) -> np.ndarray:
    """convex combination w·seq1 + (1‑w)·seq2 (both already normalised)."""
    m = w * seq1 + (1.0 - w) * seq2
    s = m.sum()
    return m / s if s != 0 else _uniform(len(m))


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a probability vector (sum = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax utilities (used by exponentiated‑gradient)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 2_000_000,
                      eta0: float = 5.0):
    """Mirror‑descent with the true max‑convolution sub‑gradient."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space to stay feasible.
    """
    try:
        import scipy.optimize as _opt
    except Exception:       # pragma: no cover
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smoothed max
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)      # chain rule for softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Permutation annealing (swap + segment‑reverse moves)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 n: int,
                 deadline: float,
                 rng: np.random.Generator,
                 max_iters: int = 300_000):
    """
    Simulated‑annealing style search that only re‑orders the entries.
    Swaps and short segment reversals are proposed; the algorithm keeps
    the best sequence found.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-8 * cur_val

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.6:          # swap move (most frequent)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                           # short segment reversal
            i = rng.integers(0, n)
            max_len = min(50, n - i)
            seg_len = rng.integers(1, max_len + 1)
            j = i + seg_len - 1
            ns = cur_seq.copy()
            ns[i:j + 1] = ns[i:j + 1][::-1]

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The algorithm proceeds in three stages:

    1. **Exploration** – a cheap grid scan over a few parametric families
       (exponential, power‑law, Dirichlet) plus a few simple mixtures.
       The 8 best candidates (lowest surrogate) are retained.

    2. **Polishing** – each retained candidate is refined by
       a long exponentiated‑gradient run followed by a continuation
       L‑BFGS optimisation with a dense β‑schedule.

    3. **Final annealing** – the very best polished candidate is
       further improved by a short permutation‑annealing phase.

    The globally best sequence is stored in ``height_sequence_1`` and
    returned as a plain ``list[float]``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 2.0)   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size (reuse previous length if possible)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            seq_prev = np.asarray(prev, dtype=np.float64)
            n = seq_prev.shape[0]
            if n == 0:
                raise ValueError
        except Exception:
            seq_prev = None
            n = 4000
    else:
        seq_prev = None
        n = 4000

    # ------------------------------------------------------------------
    # 1. QUICK EXPLORATION over simple families
    # ------------------------------------------------------------------
    candidates = []

    # keep the previous solution if it exists
    if seq_prev is not None:
        candidates.append(seq_prev / seq_prev.sum())

    # exponential decays
    for rate in np.linspace(0.001, 0.010, 10):
        cand = _exp_decay(n, rate)
        candidates.append(cand)

    # power‑law familes
    for p in np.linspace(0.5, 2.5, 9):
        candidates.append(_power_law(n, p))

    # Dirichlet draws (heavy‑tailed)
    for alpha in (0.02, 0.05, 0.1, 0.2):
        for _ in range(2):   # two samples each
            candidates.append(_dirichlet_sample(n, alpha, rng))

    # mixtures with the uniform distribution
    uni = _uniform(n)
    for w in (0.3, 0.5, 0.7):
        for rate in (0.003, 0.005, 0.007):
            mix = _mix(_exp_decay(n, rate), uni, w)
            candidates.append(mix)

    # evaluate surrogates (cheap, O(#candidates) FFTs)
    scored = []
    for seq in candidates:
        val = _objective_surrogate(seq)
        scored.append((val, seq))
    scored.sort(key=lambda x: x[0])

    # retain at most K best families for polishing
    K = min(8, len(scored))
    top_candidates = [seq for _, seq in scored[:K]]
    print(f"[Exploration] best surrogate among families = {scored[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2. POLISHING – EG + L‑BFGS on each of the top candidates
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    # allocate time proportionally: ~60 % of the budget for polishing
    polish_deadline = deadline - 200.0   # leave ~200 s for final annealing
    for idx, cand in enumerate(top_candidates, 1):
        if time.time() > polish_deadline:
            break
        print(f"[Polish] candidate {idx}/{len(top_candidates)} – start EG")
        # 2a – exponentiated‑gradient (many iterations)
        seq_eg, val_eg = _exp_gradient_opt(cand, n, polish_deadline,
                                           max_iters=1_800_000,
                                           eta0=5.0)
        print(f"   EG surrogate = {val_eg:.6f}")

        # 2b – L‑BFGS continuation with an extended β‑schedule
        if time.time() < polish_deadline:
            seq_lb = _lbfgs_smooth_optimize(seq_eg, n, polish_deadline,
                                            max_iter_per_beta=5000)
            val_lb = _objective_surrogate(seq_lb)
            print(f"   L‑BFGS surrogate = {val_lb:.6f}")
            if val_lb < val_eg:
                seq_final, val_final = seq_lb, val_lb
            else:
                seq_final, val_final = seq_eg, val_eg
        else:
            seq_final, val_final = seq_eg, val_eg

        if val_final < best_val:
            best_val = val_final
            best_seq = seq_final.copy()
            print(f"[Polish] new global best surrogate = {best_val:.6f}")

    # If no candidate survived (unlikely), fall back to exponential decay
    if best_seq is None:
        best_seq = _exp_decay(n, 0.003)

    # ------------------------------------------------------------------
    # 3. FINAL PERMUTATION‑ANNEALING (re‑ordering only)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        seq_perm, val_perm = _perm_anneal(best_seq, n, deadline, rng,
                                         max_iters=250_000)
        if val_perm < best_val:
            best_val = val_perm
            best_seq = seq_perm
            print(f"[FinalAnneal] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. FINAL exact evaluation via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("[Result] evaluate_sequence raised an exception:", exc)

    # store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
