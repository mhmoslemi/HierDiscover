"""
Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence().  The algorithm is a hybrid of

  1. diverse random / structured seeds (uniform, Dirichlet,
     sparse Dirichlet, Gaussian mixtures, Fourier‑softmax),

  2. a global low‑dimensional search (SciPy differential evolution
     or pure random sampling) on the parametric families,

  3. a long L‑BFGS optimisation of a smooth log‑sum‑exp surrogate
     (variables live in the log‑softmax space),

  4. an exact exponentiated‑gradient (mirror‑descent) polishing that
     minimises the true max‑convolution objective,

  5. a few cheap stochastic perturbations of the current best.

All phases respect the overall 1000 s budget; the best sequence found
is stored in the global variable ``height_sequence_1`` and returned as a
plain ``list[float]``.
"""

import time
import math
import numpy as np

# optional SciPy – used for differential evolution
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for a probability vector:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full‑length Dirichlet draw (conc>0)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sparse_dirichlet(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Pick k random positions and draw a Dirichlet on them."""
    idx = rng.choice(n, size=k, replace=False)
    w = rng.dirichlet(np.full(k, 1.0)).astype(np.float64)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def _mixture_gaussian(params: np.ndarray, n: int, K: int = 8) -> np.ndarray:
    """
    Build a probability vector from a mixture of K Gaussian windows.
    params layout (K components):
        w_0 … w_{K-1}      (non‑negative, later normalised)
        μ_0 … μ_{K-1}      (in [0,1])
        σ_0 … σ_{K-1}      (positive)
    """
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    if w.sum() > 0.0:
        w = w / w.sum()
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)   # normalised coordinate
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _fourier_softmax(theta: np.ndarray, basis: np.ndarray) -> np.ndarray:
    """
    Build a probability vector by applying a softmax to a Fourier
    series: a_i ∝ exp( Σ_k θ_k φ_k(i) ).
    """
    logits = basis @ theta
    a = _softmax(logits)
    return a


# ----------------------------------------------------------------------
# Smooth L‑BFGS optimiser (log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 15000):
    """L‑BFGS‑B on the smooth surrogate (variables = log‑softmax)."""

    if _opt is None:
        return start_vec.copy()          # nothing to do without SciPy

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0,
                         1000000.0, 2000000.0, 5000000.0,
                         10000000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 3.0):
    """Mirror‑descent on the true max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        # each unordered pair contributes twice, except diagonal
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val
