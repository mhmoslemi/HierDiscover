"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------

1. **Seed generation** – many analytic families (exponential, power‑law,
   random Dirichlet) and many random *permutations* of a steeply decreasing
   base curve.  Every vector is normalised to the required sum
   ``S = √(2·n)`` (the scale of the original objective).

2. **Surrogate filtering** – the exact surrogate
   ``2·n·max(conv(a,a))`` (which equals ``evaluate_sequence`` for the
   normalised vectors) is evaluated for all seeds; the best ``top_k``
   candidates are kept.

3. **Permutation annealing** – for each candidate a cheap simulated‑annealing
   on the permutation of the entries is performed.  Moves are either a swap
   of two positions or a small Gaussian perturbation of the values.  The
   temperature is linearly cooled and the best sequence seen is remembered.

4. **Exponentiated‑gradient (mirror‑descent) polishing** – starting from the
   best permutation a mirror‑descent on the *exact* max‑convolution is run.
   The sub‑gradient is the active convolution entry (the one attaining the
   maximum) and the update is performed in log‑space (the exponentiated‑gradient
   step) which automatically respects non‑negativity and the simplex constraint.

5. **Optional L‑BFGS smoothing** – if SciPy is available a few L‑BFGS‑B steps
   on the smooth log‑sum‑exp surrogate are performed to obtain a final tiny
   improvement.

6. **Global final polishing** – the globally best sequence undergoes a long
   exponentiated‑gradient run with the remaining time budget; a short final
   permutation tweak is also applied.

All steps respect the overall 1000 s budget; the final sequence is stored in
the global variable ``height_sequence_1`` for possible continuation runs.
"""

# Basic numeric utilities (FFT‑based full convolution)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are assumed to have *any* sum – we normalise
# to 1 internally because the true objective is scale invariant).
# ----------------------------------------------------------------------
def _surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(convolution) for a vector (any sum)."""
    if seq.sum() == 0.0:
        return np.inf
    norm = seq / seq.sum()
    n = len(norm)
    conv = _conv_fft(norm, norm)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return vectors *with sum = 1* – we later scale to S)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    cen = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - cen))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_dirichlet(n: int, conc: float,
                             rng: np.random.Generator) -> np.ndarray:
    """Keep only the 20 % largest components of a Dirichlet draw."""
    v = rng.dirichlet(np.full(n, conc))
    thresh = np.percentile(v, 80)
    v = np.where(v >= thresh, v, 0.0)
    if v.sum() == 0.0:
        return _uniform(n)
    return v / v.sum()


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.004) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    if noisy.sum() == 0.0:
        return seq.copy()
    return noisy / noisy.sum()


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation on the *exact* max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Works in log‑space so the iterate always stays on the simplex.
    """
    a = start_seq.copy()
    n = len(a)
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # gradient of the active entry   (sub‑gradient of max)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev]

        # step size decreasing like 1/√t
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing on permutations (swap + tiny Gaussian perturbation)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 rng: np.random.Generator,
                 max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Local search that swaps two positions or perturbs the whole vector.
    Accepts worsening moves with a simple temperature schedule.
    """
    cur = start_seq.copy()
    cur_val = _surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    n = len(cur)
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap move
            i, j = rng.integers(0, n, size=2)
            if i != j:
                ns = cur.copy()
                ns[i], ns[j] = ns[j], ns[i]
            else:
                continue
        else:                                   # tiny Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s * cur.sum()            # keep the original scale

        val = _surrogate(ns)

        if val < cur_val:
            cur = ns
            cur_val = val
            if val < best_val:
                best = ns.copy()
                best_val = val
        else:
            # metropolis acceptance
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur = ns
                    cur_val = val

    return best, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS optimisation of the smooth log‑sum‑exp surrogate.
# ----------------------------------------------------------------------
def _lbfgs_opt(start_vec: np.ndarray,
               deadline: float,
               beta_schedule=None,
               max_iter_per_beta: int = 2500):
    """Minimise the smoothed max‑convolution using L‑BFGS‑B (if SciPy is present)."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
                         2_000.0, 5_000.0, 20_000.0, 50_000.0,
                         100_000.0, 200_000.0, 500_000.0, 1_000_000.0]

    v = np.log(start_vec + 1e-30)          # log‑space variables

    n = len(start_vec)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_f = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_f * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_f)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break

    return _softmax(v)

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem dimension – a compromise between speed and expressiveness.
    # --------------------------------------------------------------
    n = 4000
    S = math.sqrt(2.0 * n)                         # required L1‑norm

    # --------------------------------------------------------------
    # Helper to bring any vector to the required scale (L1 = S).
    # --------------------------------------------------------------
    def _to_scale(v: np.ndarray) -> np.ndarray:
        v = np.maximum(v, 0.0)
        total = v.sum()
        if total == 0.0:
            return np.full_like(v, S / n)
        return v * (S / total)

    # --------------------------------------------------------------
    # 1. Massive seed generation (many families + permutations)
    # --------------------------------------------------------------
    seeds = []

    # a few deterministic families (always useful)
    seeds.append(_to_scale(_uniform(n)))
    seeds.append(_to_scale(_exp_decay(n, 0.002)))     # steep decay
    seeds.append(_to_scale(_exp_decay(n, 0.02)))      # mild decay
    seeds.append(_to_scale(_sym_exp_decay(n, 0.005)))

    # power‑law families
    for p in (0.2, 0.5, 1.0, 1.5, 2.0):
        seeds.append(_to_scale(_power(n, p)))

    # random Dirichlet draws (wide concentration sweep)
    for conc in (0.01, 0.02, 0.05, 0.1, 0.5, 1.0):
        for _ in range(12):
            seeds.append(_to_scale(_random_dirichlet(n, conc, rng)))

    # very sparse Dirichlet draws (tiny support)
    for conc in (0.001, 0.005, 0.01):
        for _ in range(8):
            seeds.append(_to_scale(_random_sparse_dirichlet(n, conc, rng)))

    # random permutations of a steeply decreasing curve – the most promising
    base_decay = _exp_decay(n, 0.003)               # shape that works well
    base_scaled = _to_scale(base_decay)
    for _ in range(250):
        seeds.append(rng.permutation(base_scaled))

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(20, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seed surrogate values (2·n·max‑conv) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best = top_seeds[0].copy()
    global_best_val = _surrogate(global_best)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Refine every promising seed
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget to this seed
        remaining = deadline - time.time()
        share = remaining * 0.10 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] time budget ≈ {share:.1f}s")

        # ---- (a) permutation‑based simulated annealing ----
        seq_perm, val_perm = _perm_anneal(seed_vec.copy(),
                                          sub_deadline,
                                          rng,
                                          max_iters=250_000)

        # ---- (b) exponentiated‑gradient polishing ----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_perm,
                                               sub_deadline,
                                               max_iters=400_000,
                                               eta0=6.0)
        else:
            seq_eg, val_eg = seq_perm, val_perm

        # ---- (c) optional L‑BFGS smoothing ----
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_opt(seq_eg,
                                   sub_deadline,
                                   beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                  200.0, 500.0, 1_000.0,
                                                  2_000.0, 5_000.0,
                                                  20_000.0, 50_000.0,
                                                  100_000.0, 200_000.0,
                                                  500_000.0, 1_000_000.0],
                                   max_iter_per_beta=1500)
            val_lbfgs = _surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_eg, val_eg

        # ---- (d) a quick second permutation pass (helps escape local traps) ----
        if time.time() < sub_deadline:
            seq_perm2, val_perm2 = _perm_anneal(seq_lbfgs.copy(),
                                               sub_deadline,
                                               rng,
                                               max_iters=80_000)
        else:
            seq_perm2, val_perm2 = seq_lbfgs, val_lbfgs

        # pick the best of all refinements
        candidates = [
            (seed_vec, _surrogate(seed_vec)),
            (seq_perm, val_perm),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_perm2, val_perm2)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long exponentiated‑gradient polishing (use all remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best,
                                                 deadline,
                                                 max_iters=2_500_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Very short final permutation tweak (may shave a little)
    # --------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 30.0
        seq_last, val_last = _perm_anneal(global_best.copy(),
                                          short_deadline,
                                          rng,
                                          max_iters=120_000)
        if val_last < global_best_val:
            global_best = seq_last
            global_best_val = val_last
            print(f"[Final permutation] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Evaluate with the real (security‑checked) function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best.tolist()
    return global_best.tolist()
