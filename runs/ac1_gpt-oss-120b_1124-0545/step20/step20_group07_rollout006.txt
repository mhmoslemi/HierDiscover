"""
Propose a non‑negative step‑function (list of heights) that improves the
objective ``evaluate_sequence``.

The method consists of three phases:

1. **Initialisation** – re‑use the best known sequence (global
   ``height_sequence_1``) together with a few simple random starts.

2. **Projected sub‑gradient descent** – the objective is
   ``2·n·max(conv(a,a))`` for a vector with ℓ₁‑norm = 1.  The sub‑gradient
   of the max‑convolution at the active offset ``k*`` is
   ``g_i = 2·a_{k*‑i}``.  After a step in the direction ``‑g`` we project
   back onto the simplex (non‑negative entries that sum to one).  The step
   size decays as ``η / sqrt(t)``.  This phase runs as long as time permits
   and refines every start candidate.

3. **Fine‑grained polishing** – a short L‑BFGS optimisation of a smooth
   max (log‑sum‑exp) and a simulated‑annealing style mass‑transfer hill‑climber
   are applied to the best vector found in phase 2.

All steps respect the 1000 s wall‑time limit.  The final vector (still
normalised to sum = 1) is stored in ``height_sequence_1`` and returned.
"""

import time
import math
import numpy as np

# optional SciPy – used only for the L‑BFGS continuation stage
try:
    import scipy.optimize as _opt
except Exception:   # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Low‑level numeric utilities (no closures, no lambdas)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (fast for n≈4000)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with ℓ₁‑norm = 1:
       2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Reference: Duchi et al., ICML 2008.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries are zero
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 2 – projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 300_000,
                         step0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Minimise ``2·n·max(conv(a,a))`` with a simple Euclidean projected
    sub‑gradient method.  Returns the best vector seen and its surrogate.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # remove the component parallel to (1,…,1) → keeps Σa constant
        g_centered = g - np.mean(g)

        step = step0 / math.sqrt(it)
        a_next = a - step * g_centered
        a_next = _project_simplex(a_next)

        cur_val = _objective_surrogate(a_next)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_next.copy()

        a = a_next

        # occasional status line
        if it % 50_000 == 0:
            print(f"  sub‑grad iter {it:7d}: cur {cur_val:.6f}, best {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase 3 – continuation L‑BFGS on a smoothed max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 20_000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space so the simplex constraint is
    automatically satisfied.
    """
    if _opt is None:
        # SciPy not available – return the input unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1, 5, 20, 80, 200, 500, 1_000, 2_000, 5_000,
            10_000, 20_000, 50_000, 100_000, 200_000,
            500_000, 1_000_000, 2_000_000
        ]

    n = len(start_vec)
    # work in log‑softmax space (strictly positive)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            m = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = m
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

            # gradient w.r.t. a (softmax space)
            w = exp_beta / sum_exp                    # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)               # chain rule through softmax
            return obj, grad_v

        def callback(_):
            # interrupt when we run out of time
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta}")
            break

    return _softmax(v)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# ----------------------------------------------------------------------
# Phase 3 – cheap stochastic polishing (simulated annealing)
# ----------------------------------------------------------------------
def _stochastic_anneal(start_seq: np.ndarray,
                       deadline: float,
                       target_surrogate: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing hill‑climber.
    Moves a random fraction of mass from one index to another and
    accepts worsening moves with probability exp(-Δ/T).
    """
    n = start_seq.shape[0]
    rng = np.random.default_rng()
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    # simple exponential cooling
    T = 1e-3
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a modest fraction of a[i] to a[j]
        frac = 0.05 * (0.5 + rng.random())   # ~[0.025,0.075]
        delta = frac * a[i]
        if delta <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_val = _objective_surrogate(new_a)
        diff = new_val - cur_val
        if diff < 0 or rng.random() < math.exp(-diff / (T + 1e-15)):
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
                print(f"  anneal iter {it:7d}: new best {best_val:.6f}")

            # slowly tighten the temperature
            T *= 0.99999
        else:
            # if we reject, still cool a little
            T *= 0.999995

        it += 1
        if it % 20_000 == 0:
            print(f"  anneal status {it:7d}: cur {cur_val:.6f}, best {best_val:.6f}, T {T:.2e}")
        if best_val <= target_surrogate:
            break
    return best_seq, best_val

# Main entry point required by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The search combines several optimisation ideas
    (projected sub‑gradient, L‑BFGS continuation and simulated annealing)
    and respects the 1000 s wall‑time limit.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a few seconds margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Determine problem size (reuse previous length if possible)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple)) and len(prev) > 0:
        n = len(prev)
    else:
        n = 4000

    # ------------------------------------------------------------------
    # Helper: normalise any vector to ℓ₁‑norm = 1 (and clip negatives)
    # ------------------------------------------------------------------
    def _norm(v: np.ndarray) -> np.ndarray:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return np.full(n, 1.0 / n, dtype=np.float64)
        return v / s

    # ------------------------------------------------------------------
    # 1. Build a small set of diverse start vectors
    # ------------------------------------------------------------------
    starts = []

    # uniform baseline
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))

    # previous best (if any)
    if prev is not None:
        starts.append(_norm(np.asarray(prev, dtype=np.float64)))

    # a few random Dirichlet draws (different concentration parameters)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        starts.append(_norm(rng.dirichlet(np.full(n, conc))))

    # a couple of low‑frequency sinusoidal curves (different phases)
    t = np.arange(n, dtype=np.float64)
    for phase in (0.0, math.pi / 3, math.pi / 2):
        a = np.cos(2.0 * math.pi * t / n + phase)
        a = np.maximum(a, 0.0)
        starts.append(_norm(a))

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient descent on every start vector
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    target_surrogate = 1.503   # we stop early once we hit the target

    # allocate roughly equal time to each start (but keep some budget for later phases)
    remaining_for_sub = max(0.0, deadline - time.time() - 40.0)   # leave ~40 s for L‑BFGS & anneal
    per_start_deadline = time.time() + remaining_for_sub / max(1, len(starts))

    print("\n--- Sub‑gradient descent phase ---")
    for idx, s in enumerate(starts):
        if time.time() > deadline:
            break
        # give each start its own time slice (roughly)
        sub_deadline = min(per_start_deadline, deadline)
        seq, val = _subgradient_descent(s, sub_deadline,
                                        max_iters=400_000,
                                        step0=0.5)
        print(f"  start #{idx+1}: best surrogate {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            if best_val <= target_surrogate:
                print("  Target reached during sub‑gradient phase.")
                break

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS (smooth max) – fine tuning
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0 and _opt is not None:
        print("\n--- L‑BFGS continuation smoothing ---")
        # Ensure strictly positive entries for the log‑softmax representation
        start_vec = np.maximum(best_seq, 1e-12)
        start_vec = start_vec / start_vec.sum()
        seq_lbfgs = _lbfgs_smooth_optimize(start_vec,
                                           deadline,
                                           beta_schedule=[1, 5, 20, 80, 200,
                                                          500, 1_000, 2_000, 5_000,
                                                          10_000, 20_000, 50_000,
                                                          100_000, 200_000,
                                                          500_000, 1_000_000,
                                                          2_000_000],
                                           max_iter_per_beta=20_000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"  L‑BFGS surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print("  L‑BFGS improved the best surrogate.")

    # ------------------------------------------------------------------
    # 4. Local stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\n--- Simulated‑annealing polishing ---")
        seq_ann, val_ann = _stochastic_anneal(best_seq,
                                              deadline,
                                              target_surrogate=target_surrogate)
        if val_ann < best_val:
            best_val, best_seq = val_ann, seq_ann.copy()
            print(f"  Annealing further improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final true evaluation and bookkeeping
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (should never happen)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Ensure the final vector is a plain Python list of floats
    result = best_seq.tolist()

    # evaluate with the real (security‑checked) function
    try:
        true_val = evaluate_sequence(result)
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
          f"true evaluate_sequence = {true_val}")

    globals()["height_sequence_1"] = result
    return result
