"""
Propose a non‑negative step‑function that minimizes
    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

Algorithm
---------
1. Load the previous best sequence (if any) and normalise it to L1‑norm 1.
2. Perform a fast random‑parameter search over several analytic families
   (beta, Gaussian, exponential, sine‑window, mixtures, low‑frequency cosine
   sums, Dirichlet draws).  Keep the best shape seen according to the exact
   surrogate 2·n·max(conv) (identical to evaluate_sequence for L1‑norm 1).
3. Refine the best shape with a large‑scale exponentiated‑gradient (mirror‑descent)
   optimisation of the exact max‑convolution objective.  The method works in
   log‑space, guaranteeing non‑negativity and unit L1‑norm.
4. Optional “restart” passes: perturb the current best solution and run a
   short EG optimisation; keep any improvement.
5. Final light simulated‑annealing polishing.
6. Evaluate the final sequence with the supplied evaluate_sequence() and store
   it in the global variable ``height_sequence_1`` for future runs.

The routine respects the supplied time budget (default 1000 s) and prints
progress information.  All helper functions are top‑level as required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for an L1‑normalised vector: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return L1‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mixture_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two Beta‑shaped densities."""
    w = rng.random()
    a1 = _beta_shape(n,
                     alpha=rng.uniform(0.2, 5.0),
                     beta=rng.uniform(0.2, 5.0))
    a2 = _beta_shape(n,
                     alpha=rng.uniform(0.2, 5.0),
                     beta=rng.uniform(0.2, 5.0))
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Perturbation utilities
# ----------------------------------------------------------------------


def _perturb(seq: np.ndarray,
             rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation (exact max‑convolution)
# ----------------------------------------------------------------------


def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its exact objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # Log‑space representation (automatically keeps L1‑norm = 1)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # indices where conv is within a tiny relative margin of the maximum
        eps = max_val * 1e-7
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient accumulation across active indices
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 150_000,
                         start_sigma: float = 0.001,
                         T_start: float = 5e-3,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Light SA: Gaussian perturbations with linear cooling.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear temperature schedule
        t_frac = attempts / max_attempts
        T = (1 - t_frac) * T_start + t_frac * T_end
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random‑parameter search
# ----------------------------------------------------------------------


def _random_parameter_search(n: int,
                             rng: np.random.Generator,
                             deadline: float,
                             init_seq: np.ndarray,
                             init_val: float) -> tuple[np.ndarray, float]:
    """
    Explore a variety of analytic families with random parameters.
    Returns the best shape found.
    """
    best_seq = init_seq.copy()
    best_val = init_val
    families = [
        # (function, sampler)
        ("beta", lambda: _beta_shape(n,
                                    alpha=rng.uniform(0.1, 5.0),
                                    beta=rng.uniform(0.1, 5.0))),
        ("gaussian", lambda: _gaussian(n, sigma_frac=rng.uniform(0.05, 0.5))),
        ("exp_decay", lambda: _exp_decay(n, rate=rng.uniform(0.005, 0.1))),
        ("sym_exp", lambda: _sym_exp_decay(n, rate=rng.uniform(0.005, 0.1))),
        ("sine", lambda: _sine_window(n, gamma=rng.uniform(0.4, 2.0))),
        ("mixture_beta", lambda: _mixture_beta(n, rng)),
        ("lowfreq", lambda: _lowfreq_curve(n,
                                           num_freq=rng.integers(1, 5),
                                           rng=rng)),
        ("dirichlet", lambda: _random_dirichlet(n,
                                                conc=rng.uniform(0.1, 5.0),
                                                rng=rng)),
        ("exp_random", lambda: _random_exponential(n, rng)),
    ]

    # Simple round‑robin with early‑exit when time runs out
    idx = 0
    while time.time() < deadline - 5.0:
        name, sampler = families[idx % len(families)]
        seq = sampler()
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random] New best via {name}: {best_val:.6f}")
        idx += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best sequence in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0   # safety margin

    # --------------------------------------------------------------
    # Problem size (can be overridden via kwargs)
    # --------------------------------------------------------------
    n = kwargs.get("n", 4000)

    # --------------------------------------------------------------
    # Load previous solution if present
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0.0:
                    prev_arr = prev_arr / s
                else:
                    prev_arr = None
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # --------------------------------------------------------------
    # Initial sequence
    # --------------------------------------------------------------
    if prev_arr is not None:
        init_seq = prev_arr.copy()
        print("[Info] Loaded previous best sequence.")
    else:
        init_seq = _sine_window(n, gamma=1.2)
        print("[Info] No prior solution – starting from sine window.")

    init_val = _objective_surrogate(init_seq)
    print(f"[Info] Initial surrogate value = {init_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – random parameter exploration
    # --------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("\n--- Random parameter search ---")
        best_seq, best_val = _random_parameter_search(
            n=n,
            rng=rng,
            deadline=deadline,
            init_seq=init_seq,
            init_val=init_val
        )
    else:
        best_seq, best_val = init_seq.copy(), init_val

    # --------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient polishing (large scale)
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:
        print("\n--- Exponentiated‑gradient polishing (large) ---")
        seq_eg, val_eg = _exp_gradient_opt_multi(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=2_500_000,
            eta0=5.0
        )
        if val_eg < best_val:
            best_seq, best_val = seq_eg, val_eg
            print(f"[EG] Improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – short restart passes (perturb + EG)
    # --------------------------------------------------------------
    for restart in range(3):
        if time.time() > deadline - 15.0:
            break
        print(f"\n--- Restart pass {restart + 1} ---")
        pert = _perturb(best_seq, rng, sigma=0.02)
        seq2, val2 = _exp_gradient_opt_multi(
            start_seq=pert,
            n=n,
            deadline=deadline,
            max_iters=500_000,
            eta0=4.5
        )
        if val2 < best_val:
            best_seq, best_val = seq2, val2
            print(f"[Restart] Improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – simulated annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n--- Simulated annealing fine‑tuning ---")
        seq_sa, val_sa = _simulated_annealing(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            rng=rng,
            max_attempts=150_000,
            start_sigma=0.001,
            T_start=5e-3,
            T_end=1e-9
        )
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[SA] Improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("[Error] evaluate_sequence raised an exception:", exc)

    # Store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
