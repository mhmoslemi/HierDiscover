def _adam_exp_gradient_opt(start_seq, n, deadline, max_iters=None, lr=0.5):
    a = start_seq.copy()
    v = np.log(a + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # compute max possible iterations based on time, if max_iters None.
    while True:
        if time.time()>deadline:
            break
        if max_iters is not None and t>=max_iters:
            break

        # compute a
        a = _softmax(v)
        # compute conv and gradient
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # gradient wrt a
        g = np.zeros_like(a)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        idx = np.arange(i_start, i_end+1)
        rev_idx = k - idx
        g[idx] = 2 * a[rev_idx]

        # gradient wrt v
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam update
        t+=1
        m = beta1 * m + (1-beta1) * grad_v
        v2 = beta2 * v2 + (1-beta2) * (grad_v * grad_v)
        m_hat = m / (1 - beta1**t)
        v_hat = v2 / (1 - beta2**t)
        v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate occasionally
        if t % 500 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    return best_seq, best_val

"""
Propose a step‑function (non‑negative sequence) that improves the
evaluation metric

    2·n·max(conv(a, a)) / (sum(a))²

for a fixed length n (default 4000).  The algorithm combines
* diverse cheap seeds (uniform, Dirichlet draws, tiny perturbations);
* a massive random search in a low‑dimensional Gaussian‑mixture family;
* an extended L‑BFGS continuation on a smooth (log‑sum‑exp) surrogate;
* a large‑scale exponentiated‑gradient (Adam) optimisation of the true
  max‑convolution objective;
* a final random‑local‑search polishing phase.

All stages respect the overall time budget (default 1000 s).  The best
sequence found is stored in the global variable ``height_sequence_1``
and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (only needed for the L‑BFGS continuation)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate / true objective (the surrogate is exact when sum(a)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) for an ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Parametric mixture of Gaussian windows (fixed K=5)
# ----------------------------------------------------------------------
def _mixture_gaussian(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a probability vector from ``params``.
    Layout (K=5 components):
        w_0 … w_4      ≥0   (relative weights)
        μ_0 … μ_4      in [0,1] (centres)
        σ_0 … σ_4      in [0.001,0.5] (widths)
    The resulting vector is renormalised to sum to 1.
    """
    K = 5
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    w_sum = w.sum()
    if w_sum > 0.0:
        w = w / w_sum
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)   # normalised coordinate
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser on the smooth (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 20000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space.
    """
    if _opt is None:
        # SciPy unavailable – fall back to a simple return.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0,
                         2000000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Adam‑based exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_exp_gradient_opt(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = None,
                           lr: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution using Adam in the
    log‑softmax space.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    v = np.log(a + 1e-30)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # estimate a safe upper bound for iterations if None is given
    if max_iters is None:
        # rough guess: 200 µs per iteration → 5 M iterations ≈ 1000 s
        max_iters = 5_000_000

    while time.time() < deadline and t < max_iters:
        # current probability vector
        a = _softmax(v)

        # full convolution and active index
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx
        g[idx] = 2.0 * a[rev_idx]

        # gradient w.r.t logits (softmax Jacobian)
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v2 / (1.0 - beta2 ** t)

        v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional evaluation
        if t % 500 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val
