"""
Propose a non‑negative step‑function (height sequence) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The algorithm consists of three phases:

1. **Exploratory sampling** – generate a collection of analytic families
   (uniform, sine windows, Gaussians, exponential decays, beta‑shapes) and a
   set of random Dirichlet / low‑frequency curves.  All candidates are
   ℓ₁‑normalised (sum = 1) and the best surrogate value
   2·n·max(conv) is kept.

2. **Mirror‑descent (exponentiated‑gradient) polishing** – starting from the
   best candidate, run a projected sub‑gradient optimiser on the exact
   max‑convolution objective.  The iterates are kept in log‑space (soft‑max)
   which automatically enforces non‑negativity and unit sum.  The step‑size
   follows η_t = η₀ / √t.  The loop runs until the global time budget is
   exhausted (or a very large iteration cap is reached).  Several short
   random restarts are performed as time permits.

3. **Simulated‑annealing fine‑tuning** – a lightweight SA pass makes occasional
   worsening moves to escape possible shallow minima.  The final sequence
   is stored in the global variable ``height_sequence_1`` for future runs.

The implementation respects the 1000‑second budget, works fully with
NumPy (SciPy is optional) and never uses filesystem/network I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max -> probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact max‑convolution for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add small Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 2_000_000,
                            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)        # log‑space (ensures positivity)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # indices whose value is very close to the maximum
        eps = max_val * 1e-6
        active = np.where(conv >= max_val - eps)[0]

        # accumulate sub‑gradient contributions from each active index
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            g = 2.0 * g_tot / active.size   # sub‑gradient of max‑conv
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 150_000,
                         start_sigma: float = 0.001,
                         T_start: float = 5e-3,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linearly decreasing temperature.
    Returns the best sequence observed before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        t_frac = attempts / max_attempts
        T = T_start * (1.0 - t_frac) + T_end * t_frac
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, stores the best sequence in the
    global variable ``height_sequence_1`` and finally returns it as a list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size (adjustable; larger n gives finer resolution but costs more)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Helper to safely ℓ₁‑normalise a vector
    # --------------------------------------------------------------
    def _normalise(v: np.ndarray) -> np.ndarray:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return _uniform(n)
        return v / s

    # --------------------------------------------------------------
    # Load previous best if available
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = _normalise(arr)
                val = _objective_surrogate(arr)
                if val < best_val:
                    best_val = val
                    best_seq = arr.copy()
                print("[Info] Loaded previous best sequence.")
        except Exception:
            pass

    # --------------------------------------------------------------
    # Build a library of candidate families
    # --------------------------------------------------------------
    candidates = []

    # analytic families
    candidates.append(_uniform(n))
    candidates.append(_sine_window(n, gamma=0.8))
    candidates.append(_sine_window(n, gamma=1.0))
    candidates.append(_sine_window(n, gamma=1.2))
    candidates.append(_gaussian(n, sigma_frac=0.10))
    candidates.append(_gaussian(n, sigma_frac=0.15))
    candidates.append(_exp_decay(n, rate=0.004))
    candidates.append(_exp_decay(n, rate=0.008))
    candidates.append(_sym_exp_decay(n, rate=0.005))
    candidates.append(_beta_shape(n, alpha=0.8, beta=1.2))
    candidates.append(_beta_shape(n, alpha=1.2, beta=0.8))
    candidates.append(_beta_shape(n, alpha=2.0, beta=2.0))

    # random Dirichlet draws with various concentrations
    for conc in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0]:
        for _ in range(4):
            candidates.append(_random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for _ in range(8):
        candidates.append(_lowfreq_curve(n, num_freq=3, rng=rng))

    # --------------------------------------------------------------
    # Evaluate all candidates and keep the best one seen so far
    # --------------------------------------------------------------
    for seq in candidates:
        if time.time() > deadline:
            break
        seq = _normalise(seq)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[Info] Initial best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Main exponentiated‑gradient polishing (uses the exact max‑conv)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)

    if time.time() < deadline - 30.0:
        print("\n--- Exponentiated‑gradient polishing (main run) ---")
        seq_eg, val_eg = _exp_gradient_opt_multi(
            best_seq, n, deadline, max_iters=2_000_000, eta0=5.0
        )
        if val_eg < best_val:
            best_val = val_eg
            best_seq = seq_eg.copy()
            print(f"[EG] Improved to {best_val:.6f}")

        # ----------------------------------------------------------
        # A few short random restarts (time‑permitting)
        # ----------------------------------------------------------
        for restart_idx in range(3):
            if time.time() > deadline - 5.0:
                break
            start_seq = _perturb(best_seq, rng, sigma=0.01)
            seq_r, val_r = _exp_gradient_opt_multi(
                start_seq, n, deadline, max_iters=300_000, eta0=5.0
            )
            if val_r < best_val:
                best_val = val_r
                best_seq = seq_r.copy()
                print(f"[EG restart {restart_idx}] Improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Final simulated‑annealing fine‑tuning (if time remains)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing fine‑tuning ---")
        seq_sa, val_sa = _simulated_annealing(
            best_seq, n, deadline, rng,
            max_attempts=150_000,
            start_sigma=0.001,
            T_start=5e-3,
            T_end=1e-9
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"[SA] Improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Official evaluation via the supplied `evaluate_sequence`
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Error] evaluate_sequence raised:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
