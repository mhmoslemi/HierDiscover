"""
Propose a (potentially) better step‐function heights vector.

Algorithm
---------
1. Choose a relatively long vector (default n=8000).  The longer the
   vector the lower the theoretical bound 2 n/(2 n‑1)≈1, so we exploit
   a larger search space while still staying within the 1000 s budget.

2. Generate a modest collection of *seed* vectors that are very
   diverse:
   * uniform,
   * “U‑shaped’’ beta distributions (α,β < 1) – heavy mass at the edges,
   * edge‑power laws `a_i ∝ (min(i,n‑1‑i)+ε)^{-p}`,
   * simple alternating patterns,
   * a few random Dirichlet draws,
   * low‑frequency cosine curves,
   * mixtures of the above with the uniform vector.
   All seeds are ℓ₁‑normalised.

3. Evaluate the cheap surrogate objective
        F(a) = 2·n·max(conv(a,a))      (a already ℓ₁‑normalised)
   and keep the best few (default 5).

4. **Exponentiated‑gradient (mirror descent) refinement**.
   Starting from each top seed we run a long EG optimisation in log‑softmax
   space, using the exact sub‑gradient of the active convolution entry.
   The iteration count is limited only by the remaining wall‑time.
   The best vector seen across all seeds is retained.

5. **Continuation L‑BFGS smoothing** (if SciPy is present).
   The non‑smooth max is approximated by a log‑sum‑exp with a schedule of
   increasing β (temperature).  Optimisation is performed in log‑softmax
   variables, guaranteeing non‑negativity and the simplex constraint.

6. **Stochastic local search** – a cheap hill‑climbing phase that moves a
   small random amount of mass between two coordinates; any improvement
   (according to the exact surrogate) is kept.

7. Evaluate the final candidate with the provided ``evaluate_sequence``.
   The resulting list is stored in the global variable ``height_sequence_1``
   for possible reuse.

All steps respect the overall time budget; a small safety margin is kept
to guarantee a return before the deadline.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Basic numeric utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector ℓ₁‑normalised to 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_u_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution (α,β) in [0,1] – U‑shaped when α,β < 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_power_shape(n: int, p: float, eps: float = 1e-6) -> np.ndarray:
    """a_i ∝ (min(i, n‑1‑i) + eps)^{-p}, i = 0..n‑1."""
    idx = np.arange(n, dtype=np.float64)
    d = np.minimum(idx, n - 1 - idx) + eps
    a = d ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _alternating_pattern(n: int, high_on_even: bool = True) -> np.ndarray:
    """Mass only on alternating indices, equally split among them."""
    a = np.zeros(n, dtype=np.float64)
    if high_on_even:
        a[::2] = 1.0
    else:
        a[1::2] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_cosine(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ------------------------------------------------------------
# Mixing / projection helpers
# ------------------------------------------------------------
def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


def _ternary_mix(a: np.ndarray, b: np.ndarray, deadline: float,
                 steps: int = 30):
    """Best convex combination of a and b by ternary search on α∈[0,1]."""
    lo, hi = 0.0, 1.0
    best_seq, best_val = None, float('inf')
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)
        if val1 < val2:
            hi = m2
            if val1 < best_val:
                best_val, best_seq = val1, seq1
        else:
            lo = m1
            if val2 < best_val:
                best_val, best_seq = val2, seq2
    mid = (lo + hi) / 2.0
    seq_mid = _mix_shapes(a, b, mid)
    val_mid = _objective_surrogate(seq_mid)
    if val_mid < best_val:
        best_seq, best_val = seq_mid, val_mid
    return best_seq, best_val


# ------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on exact max
# ------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        # active index (any that attains the max)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (softmax space)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)          # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Very cheap stochastic local search
# ------------------------------------------------------------
def _local_search(seq: np.ndarray,
                  deadline: float,
                  moves: int = 30000,
                  step_scale: float = 0.015):
    """
    Simple random mass‑transfer hill‑climbing.
    At each move a small amount of probability is moved from one index
    to another; improvements (according to the exact surrogate) are kept.
    """
    rng = np.random.default_rng()
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    n = len(seq)

    for _ in range(moves):
        if time.time() > deadline:
            break
        eps = step_scale * rng.random()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = min(eps, best_seq[i])
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # renorm for numeric safety
        new_seq /= new_seq.sum()
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq
    return best_seq, best_val

        if time.time() > deadline:
            break

        # Final local stochastic search
        local_deadline = deadline - 1.0
        seq_loc, val_loc = _local_search(best_overall_seq,
                                         local_deadline,
                                         moves=25000,
                                         step_scale=0.012)
        if val_loc < best_overall_val:
            best_overall_val, best_overall_seq = val_loc, seq_loc.copy()
            print(f"  Local‑search improvement: {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 4. Official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised", exc)

    print(f"\n=== Finished ===\nBest surrogate ≈ {best_overall_val:.6f}"
          f", true evaluate_sequence ≈ {true_val}\nLength = {len(best_overall_seq)}")

    # store for reuse by a later run
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
