    # a: numpy array shape (n,)
    # conv: numpy array shape (2n-1,)
    n = a.shape[0]

"""
Propose a non‑negative step function minimizing `evaluate_sequence`.

Algorithm
---------
1. **Broad parametric sweep** – generate a few hundred analytic
   windows (uniform, triangular, sine, Gaussian, Kaiser, Tukey,
   Beta‑shape, power‑law, random Dirichlet / sparse vectors, plus the
   SciPy signal windows).  All vectors are ℓ₁‑normalised and the cheap
   surrogate 2·n·max(conv(a,a)) (exact for normalised vectors) is used
   to keep the `top_k` best candidates.

2. **Exponentiated‑gradient (mirror‑descent)** – starting from every
   top candidate run a mirror‑descent on the exact max‑convolution
   sub‑gradient (the same update as the previous state‑of‑the‑art).
   Several random restarts are also performed.  The best vector found
   is retained.

3. **Continuation L‑BFGS** – the best vector is refined by a
   continuation L‑BFGS optimiser on the smooth log‑sum‑exp approximation
   of the max.  A long schedule of β values and a larger per‑β
   iteration budget are used.

4. **Mass‑transfer annealing** – a fast in‑place update of the full
   convolution is used to perform millions of tiny “move δ mass from i
   to j” proposals.  Improvements are accepted greedily; the step size
   is adapted during the run.

5. **Final evaluation** – the true `evaluate_sequence` is called once
   and the result is stored globally as ``height_sequence_1``.
"""

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with ℓ₁‑norm = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Modified Bessel function I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Cosine‑tapered window.  α=0 ⇒ uniform, α=1 ⇒ Hann."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution (support on {1,…,n})."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k random non‑zero positions with exponential masses."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    a[idx] = rng.exponential(scale=1.0, size=k)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution objective.
    Returns best sequence and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space so we stay on the simplex
    automatically.
    """
    if _opt is None:
        # SciPy not available – fall back to the input vector unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1, 5, 20, 80, 200, 500, 1000,
                         2000, 5000, 20000, 50000, 100000,
                         200000, 500000, 1000000, 2000000]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)                # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L-BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Fast in‑place mass‑transfer move (used by annealing)
# ----------------------------------------------------------------------
def _apply_move(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """Move a fraction δ of mass from index i to j.
    Updates `a` and the full convolution `conv` in place.
    """
    n = a.shape[0]
    # slice updates (use current a – which is the pre‑move vector)
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic correction terms
    idx2i = 2 * i
    idx2j = 2 * j
    idxij = i + j
    delta2 = delta * delta
    conv[idx2i] += delta2
    conv[idx2j] += delta2
    conv[idxij] -= 2.0 * delta2

    # finally update the probabilities
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
# Local stochastic polishing – tiny mass transfers with greedy acceptance
# ----------------------------------------------------------------------
def _local_search_anneal(a: np.ndarray,
                         conv: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> np.ndarray:
    """
    Greedy annealing that repeatedly transfers a small amount of mass
    between two positions, keeping the convolution up‑to‑date.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_conv = conv.copy()
    cur_val = 2.0 * n * np.max(conv)
    best_val = cur_val

    step_frac = 0.08  # initial fraction of source mass to move

    while time.time() < deadline:
        # pick source with non‑negligible mass
        i = rng.integers(0, n)
        if a[i] <= 1e-15:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        max_move = step_frac * a[i]
        if max_move <= 0.0:
            continue
        delta = rng.uniform(0.0, max_move)

        # apply move
        _apply_move(a, conv, i, j, delta)
        new_val = 2.0 * n * np.max(conv)

        if new_val < cur_val:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                best_conv = conv.copy()
            # slight increase of aggressiveness after success
            step_frac = min(0.25, step_frac * 1.005)
        else:
            # revert (apply opposite move)
            _apply_move(a, conv, i, j, -delta)
            # decay step size on failure
            step_frac = max(1e-5, step_frac * 0.998)

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The pipeline consists of a parametric
    sweep, several EG runs (with random restarts), continuation L‑BFGS
    smoothing and a final mass‑transfer annealing stage.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – reuse the previous length if a solution already exists
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple)) and len(prev) > 0:
        n = len(prev)
    else:
        n = 4000

    # ------------------------------------------------------------------
    # Helper to safely normalise a vector to unit ℓ₁‑norm
    # ------------------------------------------------------------------
    def _norm(v: np.ndarray) -> np.ndarray:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return _uniform(n)
        return v / s

    # ------------------------------------------------------------------
    # 1. Broad parametric sweep (cheap surrogate evaluation)
    # ------------------------------------------------------------------
    print(f"\n--- Parametric sweep (n={n}) ---")
    top_k = 20
    best_heap: list[tuple[float, np.ndarray]] = []  # (value, vec)

    def _maybe_insert(vec: np.ndarray):
        """Insert vec into best‑heap if it belongs to the top‑k."""
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            best_heap.append((val, vec))
            best_heap.sort(key=lambda x: x[0])
        else:
            if val < best_heap[-1][0]:
                best_heap[-1] = (val, vec)
                best_heap.sort(key=lambda x: x[0])

    # uniform baseline
    _maybe_insert(_uniform(n))

    # analytic families (moderately dense grids)
    for gamma in np.linspace(0.01, 1.0, 20):
        _maybe_insert(_sine_window(n, gamma))

    for sigma_frac in np.linspace(0.07, 0.30, 12):
        _maybe_insert(_gaussian(n, sigma_frac))

    for beta in np.linspace(0.5, 20.0, 15):
        _maybe_insert(_kaiser_window(n, beta))

    for alpha in np.linspace(0.0, 1.0, 11):
        _maybe_insert(_tukey_window(n, alpha))

    alist = [0.1, 0.3, 0.7, 1.0, 2.0, 5.0, 10.0]
    for a_ in alist:
        for b_ in alist:
            _maybe_insert(_beta_shape(n, a_, b_))

    for p in np.linspace(0.0, 2.5, 12):
        _maybe_insert(_power(n, p))

    # low‑frequency sinusoids
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            _maybe_insert(_lowfreq_curve(n, nf, rng))

    # random Dirichlet draws
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            _maybe_insert(_random_dirichlet(n, conc, rng))

    # random sparse vectors
    for k in (5, 10, 20):
        for _ in range(2):
            _maybe_insert(_random_sparse(n, k, rng))

    # SciPy signal windows (if available)
    if _get_window is not None:
        for wtype in ['hann', 'hamming', 'blackman', 'nuttall', 'flattop']:
            w = _get_window(wtype, n, fftbins=False)
            _maybe_insert(np.asarray(w, dtype=np.float64))
        # Kaiser with a few betas
        for beta in np.linspace(0.5, 20.0, 8):
            w = _get_window(('kaiser', beta), n, fftbins=False)
            _maybe_insert(np.asarray(w, dtype=np.float64))
        # Tukey varies alpha
        for alpha in np.linspace(0.0, 1.0, 9):
            w = _get_window(('tukey', alpha), n, fftbins=False)
            _maybe_insert(np.asarray(w, dtype=np.float64))
        # Gaussian windows
        for sigma in np.linspace(0.07, 0.30, 7):
            w = _get_window(('gaussian', sigma), n, fftbins=False)
            _maybe_insert(np.asarray(w, dtype=np.float64))

    # include previous best (if any) and perturbations
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.size == n and prev_arr.sum() > 0:
            base = _norm(prev_arr)
            _maybe_insert(base)
            for sigma in (0.001, 0.003, 0.005):
                for _ in range(2):
                    pert = base + sigma * rng.normal(size=n)
                    pert = np.maximum(pert, 0.0)
                    _maybe_insert(_norm(pert))

    # sort the heap (ascending by surrogate value)
    best_heap.sort(key=lambda x: x[0])
    top_seqs = [vec for _, vec in best_heap[:top_k]]
    best_val, best_seq = best_heap[0]
    print("\nTop candidates after sweep (surrogate):")
    for i, (val, _) in enumerate(best_heap[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # ------------------------------------------------------------------
    # 2. Exponentiated‑gradient refinement (multiple restarts)
    # ------------------------------------------------------------------
    eg_time = max(0.0, deadline - time.time() - 120.0)  # keep ~120 s for later stages
    eg_deadline = time.time() + eg_time
    print(f"\n--- EG refinement (budget ≈ {eg_time:.1f}s) ---")
    # run EG on the sweep winners
    for idx, seq in enumerate(top_seqs):
        if time.time() > eg_deadline:
            break
        remain = eg_deadline - time.time()
        # allocate up to 250 k iterations (scaled to remaining time)
        max_iters = min(250_000, int(200_000 * (remain / eg_time + 0.1)))
        seq_opt, val_opt = _exp_gradient_opt(seq.copy(),
                                             n,
                                             eg_deadline,
                                             max_iters=max_iters,
                                             eta0=7.0)
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print(f"  EG from cand {idx+1}: new best surrogate = {best_val:.6f}")

    # random restarts while time permits
    while time.time() < eg_deadline - 5.0:
        conc = 0.05 + 4.95 * rng.random()
        start_vec = rng.dirichlet(np.full(n, conc))
        seq_opt, val_opt = _exp_gradient_opt(start_vec,
                                             n,
                                             eg_deadline,
                                             max_iters=150_000,
                                             eta0=7.0)
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print(f"  Random EG restart: new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS smoothing (smooth max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0 and _opt is not None:
        print("\n--- L‑BFGS continuation smoothing ---")
        start_vec = np.maximum(best_seq, 1e-12)
        start_vec /= start_vec.sum()
        seq_lbfgs = _lbfgs_smooth_optimize(start_vec,
                                           n,
                                           deadline,
                                           beta_schedule=[1, 5, 20, 80,
                                                          200, 500, 1000,
                                                          2000, 5000,
                                                          20000, 50000,
                                                          100000, 200000,
                                                          500000, 1000000,
                                                          2000000],
                                           max_iter_per_beta=8000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print(f"  L‑BFGS improvement: surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Local stochastic polishing (tiny mass moves)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Local stochastic annealing ---")
        a = best_seq.copy()
        conv = _conv_fft(a, a)
        anneal_deadline = time.time() + remaining * 0.9
        best_seq = _local_search_anneal(a, conv, anneal_deadline, rng)
        best_val = _objective_surrogate(best_seq)

    # ------------------------------------------------------------------
    # 5. Final true evaluation and bookkeeping
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
