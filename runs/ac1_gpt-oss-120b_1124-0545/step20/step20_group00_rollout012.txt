"""
Propose a candidate step‑function that improves the current best bound.

Algorithm
---------

1. **Initialisation** – load the previous best sequence (global
   ``height_sequence_1``) if it exists, otherwise start from the uniform
   distribution.  The vector is always normalised to sum = 1 (the
   evaluation is scale‑invariant).

2. **Exponentiated‑gradient polishing** – a mirror‑descent (EG) optimiser
   on the exact max‑convolution objective.  This stage performs many
   cheap iterations (FFT‑based convolution) and typically drives the
   bound from ~1.51 down to the best value found so far.

3. **Stochastic hill‑climbing** – from the EG‑optimised point we run a
   simple Monte‑Carlo neighbourhood search.  In each step a random
   amount of mass is moved from a source index (chosen proportional to
   its weight) to a uniformly chosen target index.  The move is kept
   only when it improves the surrogate objective
   ``2·n·max(conv)``; otherwise it is undone.  The step size shrinks
   linearly with the remaining time, allowing both large exploratory
   jumps early and fine refinements later.

4. **Optional final L‑BFGS smoothing** – a short L‑BFGS‑B optimisation
   of the log‑sum‑exp smooth approximation (as in the reference code)
   to clean up any remaining bias.

5. **Evaluation** – the best sequence found is evaluated with the
   provided ``evaluate_sequence`` function, stored in the global variable
   ``height_sequence_1`` and returned as a plain list.

All stages respect the global 1000 s budget (a small safety margin is
kept).  The implementation uses only NumPy and SciPy (if available) and
is written with top‑level helper functions only, as required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for the L‑BFGS optimiser)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a probability vector: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated gradient) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the true max‑convolution objective."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)                     # log‑softmax variables
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient of the active entry (exact for max‑convolution)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient w.r.t a
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Stochastic hill‑climbing optimiser (mass‑transfer moves)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                deadline: float,
                max_no_improve: int = 200_000):
    """
    Simple Monte‑Carlo hill‑climbing on the exact max‑convolution
    surrogate.  Each move transfers a random fraction of mass from a
    source index (chosen proportionally to its current weight) to a
    uniformly random target index.  Only improving moves are kept.
    """
    rng = np.random.default_rng()
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    counter = 0
    no_improve = 0

    start_time = time.time()
    total_time = deadline - start_time

    while time.time() < deadline:
        # Linear schedule for the maximal fraction of the source mass we may move
        frac = 0.2 * (deadline - time.time()) / total_time   # shrinks to 0
        # Choose a source index proportional to its weight (targets the heavy spots)
        i = rng.choice(n, p=cur_seq)
        if cur_seq[i] <= 1e-12:
            continue
        # Random amount to move, never exceeds source mass
        delta = cur_seq[i] * rng.random() * frac
        if delta <= 0.0:
            continue
        # Choose a different target uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # Apply the move
        cur_seq[i] -= delta
        cur_seq[j] += delta

        new_val = _objective_surrogate(cur_seq)

        if new_val < cur_val:          # accept only improvements
            cur_val = new_val
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
                print(f"  [Hill] new best surrogate = {best_val:.6f}")
        else:
            # reject – revert the move
            cur_seq[i] += delta
            cur_seq[j] -= delta
            no_improve += 1

        counter += 1
        # occasional diversification: restart from the best point
        if counter % 10_000 == 0 and rng.random() < 0.05:
            cur_seq = best_seq.copy()
            cur_val = best_val

        if no_improve > max_no_improve:
            # after a long stagnation we stop early
            break

    return best_seq, best_val
