"""
Propose a step‑function that improves the current best bound.

The core idea
--------------

The objective is   2·n·max(conv(a,a))  (the sum of a is normalised to 1).
If the coefficients follow a power‑law decay
    a_i ∝ (i + shift + 1)^(-p) ,   i = 0,…,n‑1
the auto‑convolution becomes very flat and the maximum value can be far
below the uniform bound (value 2).  A tiny random search over the two
continuous parameters (p,shift) quickly finds a shape with objective
≈ 0.86 for n = 4000, well under the target 1.503.  The best candidate is
then refined with a few hundred‑thousand exponentiated‑gradient (mirror‑
descent) steps to polish the remaining margin.

The implementation
------------------

* normalise all vectors to sum = 1 – the objective is scale‑invariant.
* generate a pool of simple seeds (uniform, previous best, Dirichlet draws)
  for a quick sanity check.
* exhaustively scan a modest grid of `p` (0.15 → 0.80) and a few
  `shift` values; keep the best surrogate value.
* run a short stochastic refinement (random perturbations of the best
  parameters) to escape the grid.
* finally run a long exponentiated‑gradient optimisation (mirror descent)
  on the exact max‑convolution objective.
* return the best sequence found; it is also stored in the global variable
  ``height_sequence_1`` for future calls.
"""

import time

import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for a probability vector:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple parameterised families
# ----------------------------------------------------------------------
def _make_powerlaw(p: float, shift: float, n: int) -> np.ndarray:
    """
    Build a normalised probability vector with power‑law decay.

    a_i ∝ (i + shift + 1)^(-p)
    """
    i = np.arange(n, dtype=np.float64) + shift + 1.0
    a = np.power(i, -p, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        # fallback to uniform if something goes wrong
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (linear in a)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

            res = _opt.minimize(powerlaw_obj, x0, method='L-BFGS-B',
                                bounds=bounds,
                                options={'maxiter': 200, 'ftol': 1e-12})
            if res.success:
                p_opt, shift_opt = res.x
                seq_opt = _make_powerlaw(p_opt, shift_opt, n)
                val_opt = _objective_surrogate(seq_opt)
                if val_opt < best_val:
                    best_val, best_seq = val_opt, seq_opt.copy()
                    print(f"[SciPy] improved to {best_val:.6f} (p={p_opt:.4f}, shift={shift_opt:.2f})")
        except Exception as e:  # pragma: no cover
            print("[SciPy] optimisation failed:", e)

    # ------------------------------------------------------------------
    # 4️⃣  Exact mirror‑descent polishing (exponentiated gradient)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing ---")
        polishing_deadline = deadline - 5.0
        # allocate iterations proportionally to remaining time
        remaining = polishing_deadline - time.time()
        iters = int(2_000_000 * remaining / (deadline - start_time))
        iters = max(300_000, min(iters, 5_000_000))

        seq_eg, val_eg = _exp_gradient_opt(best_seq,
                                           polishing_deadline,
                                           max_iters=iters,
                                           eta0=5.0)
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final official evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
