```python
"""
Search for a non‑negative step‑function (list of heights) that minimises
the objective ``evaluate_sequence``.

The algorithm has three stages:

1.  A thorough parametric sweep over many analytic families (including a
    dense grid of Beta‑shaped windows) and a handful of random Dirichlet /
    low‑frequency curves.  Each candidate is ℓ₁‑normalised and scored with the
    exact surrogate 2·n·max(conv(a,a)) (the true objective for a normalised
    vector).  The best ``top_k`` candidates are kept.

2.  Exponentiated‑gradient (mirror‑descent) refinement on the exact
    max‑convolution objective for every candidate from step 1.  The iteration
    budget is allocated dynamically according to the remaining wall‑time.

3.  A fast stochastic polishing phase – tiny mass is moved between two random
    coordinates; improving moves are kept, the step‑size is adapted
    automatically.  The best vector found is finally evaluated with
    ``evaluate_sequence`` and stored in the global ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a vector with ℓ₁‑norm = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                     # Modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Cosine‑tapered window.  α=0 ⇒ uniform, α=1 ⇒ Hann."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta distribution on {0,…,n‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Local stochastic polishing (hill‑climber)
# ----------------------------------------------------------------------
def _polish(seq: np.ndarray,
            n: int,
            deadline: float,
            rng: np.random.Generator,
            step_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """Tiny mass‑moves between random coordinates until time runs out."""
    best_seq = seq.copy()
    best_val = _objective(best_seq)

    cur_seq = seq.copy()
    cur_val = best_val
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a fraction of a[i] to a[j]
        delta = step_frac * cur_seq[i] * rng.uniform(0.3, 1.7)
        delta = min(delta, cur_seq[i])
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # re‑normalise (prevent drift)
        s = new_seq.sum()
        if s <= 0.0:
            new_seq = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            new_seq = new_seq / s

        val_new = _objective(new_seq)
        if val_new < cur_val:
            cur_seq = new_seq
            cur_val = val_new
            if cur_val < best_val:
                best_val, best_seq = cur_val, cur_seq.copy()
                # slightly increase aggressiveness
                step_frac = min(0.2, step_frac * 1.02)
        else:
            # slowly decay step size
            step_frac = max(1e-4, step_frac * 0.998)

        it += 1
        if it % 5000 == 0:
            print(f"    polish iter {it:6d}: cur surrogate = {cur_val:.6f}")

    return best_seq, best_val

            best_val, best_seq = val_opt, seq_opt.copy()
            print(f"  candidate {idx+1}: new best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Local stochastic polishing (hill‑climbing)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Local stochastic polishing ---")
        best_seq, best_val = _polish(best_seq, n, deadline, rng, step_frac=0.05)

    # ------------------------------------------------------------------
    # 4. Final true evaluation and bookkeeping
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
          f"true evaluate_sequence = {true_val}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
