"""
Improved minimiser for `evaluate_sequence`.

The algorithm consists of four stages:

1. **Dense parametric sweep** – generate thousands of candidates from
   rich analytic families (sine‑power, Gaussian, Kaiser, Tukey, Beta,
   power‑law) and many random Dirichlet / Beta draws.  All vectors are
   ℓ₁‑normalised, and the exact surrogate objective
       2·n·max(conv(a,a))        (a has unit sum)
   is evaluated with a fast FFT‑based convolution.

2. **Mirror‑descent (exponentiated‑gradient) refinement** – the best
   few candidates from the sweep are refined by many EG steps.
   The update lives in log‑softmax space, guaranteeing non‑negativity
   and the simplex constraint.

3. **Continuation L‑BFGS on a smooth max** – starting from the best EG
   result we run a continuation of L‑BFGS on the log‑sum‑exp smooth
   approximation of the max.  The β‑schedule is longer and the per‑β
   iteration budget is larger than in the previous baseline.

4. **Simulated‑annealing polishing** – a short SA loop explores larger
   mass‑transfer moves while accepting occasional uphill steps.
   This helps escape the tiny plateau that EG/L‑BFGS often settle in.

The whole procedure respects the 1000 s wall‑time limit; the best
sequence found is stored in the global ``height_sequence_1`` variable
and returned as a plain Python list.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Optional SciPy import (used only for L‑BFGS)
# --------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# --------------------------------------------------------------
# Exact surrogate objective (scale‑invariant)
# --------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Assumes sum(seq) == 1.
    Returns 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(np.max(conv))


# --------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_power(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s


def _kaiser(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1.0 - y * y)
    a = np.i0(t)            # modified Bessel I0
    s = a.sum()
    return a / s


def _tukey(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return w / w.sum()
    w = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:                 # left cosine taper
            w[i] = 0.5 * (1 + np.cos(np.pi *
                        (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            w[i] = 1.0
        else:                       # right cosine taper
            w[i] = 0.5 * (1 + np.cos(np.pi *
                        (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    return w / w.sum()


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a_vec = (i ** (a - 1.0)) * ((n - i + 1) ** (b - 1.0))
    s = a_vec.sum()
    return a_vec / s


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a_vec = 1.0 / (idx ** p)
    s = a_vec.sum()
    return a_vec / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a Beta(α,β) density on a finite grid and normalise."""
    a = rng.gamma(shape=rng.uniform(0.1, 5.0), scale=1.0)
    b = rng.gamma(shape=rng.uniform(0.1, 5.0), scale=1.0)
    i = np.arange(1, n + 1, dtype=np.float64)
    vec = (i ** (a - 1.0)) * ((n - i + 1) ** (b - 1.0))
    s = vec.sum()
    if s <= 0.0:
        return _uniform(n)
    return vec / s


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _perturb(seq: np.ndarray,
             rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    return noisy / s if s > 0.0 else seq.copy()


def _mix(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mix = alpha * a + (1.0 - alpha) * b
    s = mix.sum()
    return mix / s if s > 0.0 else a.copy()


# --------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation
# --------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (handle edges)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# --------------------------------------------------------------
# L‑BFGS optimiser on smooth max (log‑sum‑exp)
# --------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1, 5, 20, 80, 200, 500, 1000, 2000,
                         5000, 20000, 50000, 100000, 200000,
                         500000, 1000000, 2000000, 5000000]

    # work in log‑softmax space (guarantees sum‑one, non‑negativity)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)

        def fun(v_loc):
            a = _softmax(v_loc)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                # for numerical stability
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            smooth_max = shift + np.log(sum_exp) / beta
            obj = 2.0 * len(a) * smooth_max

            # gradient w.r.t. a
            w = exp_beta / sum_exp                     # weight for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[len(a) - 1:len(a) - 1 + len(a)]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta}")
            break

    return _softmax(v)


# --------------------------------------------------------------
# Simulated‑annealing polishing
# --------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         init_temp: float = 1e-3,
                         final_temp: float = 1e-7,
                         steps: int = 200_000) -> tuple[np.ndarray, float]:
    """Simple exponential‑schedule SA that moves a fraction of mass
    between two random coordinates.  Returns best sequence seen."""
    rng = np.random.default_rng()
    seq = start_seq.copy()
    cur_val = _objective_surrogate(seq)
    best_seq = seq.copy()
    best_val = cur_val

    n = len(seq)
    T0 = init_temp
    Tf = final_temp
    decay = (Tf / T0) ** (1.0 / max(1, steps - 1))

    T = T0
    for it in range(steps):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose moving a fraction of a[i] to a[j]
        frac = rng.uniform(0.001, 0.05)          # step size
        delta = frac * seq[i] * rng.uniform(0.5, 1.5)
        delta = min(delta, seq[i])
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective_surrogate(new_seq)
        improvement = cur_val - new_val
        accept = False
        if improvement > 0:
            accept = True
        else:
            prob = math.exp(improvement / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            seq = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()

        T *= decay
        if it % 10_000 == 0 and it > 0:
            print(f"  SA iter {it:6d}: cur = {cur_val:.6f}, best = {best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats).  The implementation follows
    the four‑stage plan described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse previous length if available, else 4000
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    n = len(prev) if isinstance(prev, (list, tuple)) and len(prev) > 0 else 4000

    # ------------------------------------------------------------------
    # Helper: normalise to ℓ₁‑norm = 1
    # ------------------------------------------------------------------
    def _norm(v: np.ndarray) -> np.ndarray:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return _uniform(n)
        return v / s

    # ------------------------------------------------------------------
    # 1️⃣  Parametric sweep (cheap surrogate)
    # ------------------------------------------------------------------
    print(f"\n--- Parametric sweep (n={n}) ---")
    top_k = 20
    best_heap: list[tuple[float, np.ndarray]] = []   # (value, vec)

    def _maybe_insert(vec: np.ndarray):
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            best_heap.append((val, vec))
            best_heap.sort(key=lambda x: x[0])
        else:
            if val < best_heap[-1][0]:
                best_heap[-1] = (val, vec)
                best_heap.sort(key=lambda x: x[0])

    # Uniform baseline
    _maybe_insert(_uniform(n))

    # Analytic families – finer grids
    # Sine‑power
    for gamma in np.linspace(0.05, 3.0, 40):
        _maybe_insert(_sine_power(n, gamma))

    # Gaussian
    for sigma_frac in np.linspace(0.04, 0.35, 30):
        _maybe_insert(_gaussian(n, sigma_frac))

    # Kaiser
    for beta in np.linspace(0.1, 25.0, 30):
        _maybe_insert(_kaiser(n, beta))

    # Tukey
    for alpha in np.linspace(0.0, 1.0, 25):
        _maybe_insert(_tukey(n, alpha))

    # Power‑law
    for p in np.linspace(0.0, 3.0, 30):
        _maybe_insert(_power_law(n, p))

    # Random Beta shape (many draws)
    for _ in range(120):
        a = rng.uniform(0.1, 10.0)
        b = rng.uniform(0.1, 10.0)
        _maybe_insert(_beta_shape(n, a, b))

    # Random Dirichlet
    for conc in np.geomspace(0.01, 10.0, 12):
        for _ in range(25):
            _maybe_insert(_random_dirichlet(n, conc, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4):

            _maybe_insert(_lowfreq_curve(n, nf, rng))

    # Perturb previous solution (if any)
    if isinstance(prev, (list, tuple)) and len(prev) == n:
        prev_arr = _norm(np.asarray(prev, dtype=np.float64))

    best_seq = top_seqs[0].copy()
    best_val = _objective_surrogate(best_seq)

    # ------------------------------------------------------------------

                                              eg_deadline,

                                                          2_000_000, 5_000_000],
                                           max_iter_per_beta=8000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < best_val:

            print(f"  L‑BFGS improvement: surrogate = {best_val:.6f}")

          f"true evaluate_sequence = {true_val}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
