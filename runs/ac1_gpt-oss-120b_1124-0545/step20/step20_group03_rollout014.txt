def _smooth_obj_small(u):

    conv = _conv_from_weights(idx, w)  # returns conv array length 2n-1
    shift = np.max(conv)
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_beta)
    obj = 2 * n * (shift + np.log(sum_exp) / beta)
    p = exp_beta / sum_exp
    # compute gradient w.r.t w
    sum_mat = idx[:,None] + idx[None,:]
    p_mat = p[sum_mat]
    grad_w = 2 * p_mat @ w
    grad_u = w * (grad_w - np.dot(grad_w, w))
    return obj, grad_u

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # determine n
    n = 4000
    prev = globals().get("height_sequence_1")
    if prev is not None and len(prev) == n:
        seq0 = np.asarray(prev, dtype=np.float64)
        # ensure nonnegative, normalize to sum 1
        seq0 = np.maximum(seq0, 0.0)
        total = seq0.sum()
        if total > 0:
            seq0 = seq0 / total
        else:
            seq0 = np.full(n, 1.0/n)
    else:
        seq0 = np.full(n, 1.0/n)  # uniform start

    best_seq = seq0.copy()
    best_val = _objective_surrogate(best_seq)

    # Phase 1: run Adam optimizer from current best
    seq_opt, val_opt = _adam_optimize(best_seq, n, deadline)
    if val_opt < best_val:
        best_seq, best_val = seq_opt, val_opt

    # Phase 2: random restarts
    attempts = 0
    while time.time() < deadline and attempts < 20:
        attempts += 1
        # random Dirichlet with alpha=1 (uniform)
        rand_seq = rng.dirichlet(alpha=np.ones(n))
        seq_opt, val_opt = _adam_optimize(rand_seq, n, deadline)
        if val_opt < best_val:
            best_seq, best_val = seq_opt, val_opt
            print(f"[Restart {attempts}] new best surrogate {best_val:.6f}")

    # Phase 3: Sidon set approach
    # compute best Sidon set
    best_idx_set = None
    best_set_size = 0
    sidon_deadline = deadline - 10.0
    while time.time() < sidon_deadline:
        idx = _sidon_greedy(n, rng)
        if idx.size > best_set_size:
            best_set_size = idx.size
            best_idx_set = idx
            print(f"[Sidon] new best set size {best_set_size}")
            if best_set_size >= 100:  # maybe good enough
                break

    if best_idx_set is not None:
        # Initialize uniform weights on set
        m = best_idx_set.size
        w0 = np.full(m, 1.0/m)
        # Optimize weights using Adam optimizer on support
        w_opt, val_opt = _adam_optimize_support(best_idx_set, w0, n, deadline)
        # Build full sequence
        seq_full = np.zeros(n, dtype=np.float64)
        seq_full[best_idx_set] = w_opt
        val_full = _objective_surrogate(seq_full)
        if val_full < best_val:
            best_seq, best_val = seq_full, val_full
            print(f"[Sidon] improved surrogate {best_val:.6f}")

    # Evaluate final using evaluate_sequence
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] best surrogate = {best_val:.6f}, true evaluate = {true_val:.6f}")
    except Exception as exc:
        print("evaluate_sequence exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
propose_candidate – fast global search for a low‐max‑convolution step function

The routine follows three complementary ideas:

1. **Smooth max‑convolution optimisation** – the true objective
   2·n·max(a∗a) is non‑smooth.  We replace the max by a
   log‑sum‑exp (temperature β) and minimise the smooth surrogate with
   Adam (exponentiated‑gradient in log‑space).  β is increased gradually
   (continuation) so that the solution slowly approaches the true
   non‑smooth optimum.

2. **Random restarts** – different random Dirichlet starts are optimised
   independently.  Because the problem is highly non‑convex this greatly
   improves the chance of escaping poor local minima.

3. **Sparse “Sidon‑like’’ support** – a set of indices whose pairwise sums
   collide as little as possible is built by a greedy random algorithm.
   With this support the convolution is naturally flatter; the weights on
   the support are further refined by the same smooth Adam optimiser
   (now in the reduced dimension).

All phases respect the global 1000 s budget, keep the globally best
sequence (also stored in the global name ``height_sequence_1`` for the next
run) and finally return the best list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Softmax utilities (numerically stable)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate used for quick evaluation (sum = 1 → 2·n·max)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate (exact max‑convolution) for a sequence with ℓ₁‑norm = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser for the smooth max‑convolution (full‑dimensional)
# ----------------------------------------------------------------------
def _adam_optimize_full(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        schedule=None):
    """
    Adam optimiser on the smooth max‑convolution surrogate.
    start_seq must already be ℓ₁‑normalised (sum = 1) and non‑negative.
    schedule is a list of (beta, iters) pairs.
    Returns (best_seq, best_val) with best_val = 2·n·max(conv) (exact).
    """
    if schedule is None:
        schedule = [
            (1.0, 2000), (5.0, 2000), (20.0, 2000), (80.0, 2000),
            (200.0, 2000), (500.0, 2000), (1000.0, 2000), (2000.0, 2000),
            (5000.0, 2000), (10000.0, 2000), (20000.0, 2000),
            (50000.0, 2000), (100000.0, 2000)
        ]

    # log‑space variables
    v = np.log(start_seq + 1e-30)
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_seq = np.copy(start_seq)
    best_val = _objective_surrogate(start_seq)

    for beta, iters in schedule:
        for _ in range(iters):
            if time.time() > deadline:
                return best_seq, best_val
            t += 1
            a = _softmax(v)                 # current probabilities, sum = 1

            # exact convolution and smooth max weights
            conv = _conv_fft(a, a)           # length 2n‑1
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            p = exp_beta / sum_exp           # smooth‑max probabilities

            # gradient w.r.t. a (see derivation in the comment header)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, p)      # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule to log‑space (softmax)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update (learning‑rate decays with √t)
            lr = 0.5 / (np.sqrt(t) + 1.0)
            m1 = beta1 * m1 + (1.0 - beta1) * grad_v
            m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)
            m1_hat = m1 / (1.0 - beta1 ** t)
            m2_hat = m2 / (1.0 - beta2 ** t)
            v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

            # exact objective (no smoothing) – keep the best ever seen
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser for a dense support (low dimensional)
# ----------------------------------------------------------------------
def _adam_optimize_support(idx: np.ndarray,
                           w0: np.ndarray,
                           n: int,
                           deadline: float,
                           schedule=None):
    """
    Optimise the weights on a fixed sparse support ``idx`` using the same
    smooth‑max surrogate as the full optimiser.  ``w0`` must be a probability
    vector (sum = 1) of length ``len(idx)``.
    Returns (optimal_weights_on_support, best_surrogate_value).
    """
    m = len(idx)
    if schedule is None:
        schedule = [
            (1.0, 5000), (5.0, 4000), (20.0, 3000), (80.0, 2000),
            (200.0, 1500), (500.0, 1200), (1000.0, 1000), (2000.0, 800),
            (5000.0, 600), (10000.0, 500), (20000.0, 400), (50000.0, 300)
        ]

    # pre‑compute the matrix of pairwise sums (shape m×m)
    sum_mat = idx[:, None] + idx[None, :]  # integer matrix, values in 0..2n‑2

    # initialise log‑space variables
    u = np.log(w0 + 1e-30)
    m1 = np.zeros_like(u)
    m2 = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_w = np.copy(w0)
    # build full vector for surrogate evaluation
    full_vec = np.zeros(n, dtype=np.float64)
    full_vec[idx] = best_w
    best_val = _objective_surrogate(full_vec)

    for beta, iters in schedule:
        for _ in range(iters):
            if time.time() > deadline:
                return best_w, best_val
            t += 1
            w = _softmax(u)                                 # probabilities on support

            # exact convolution on the support (via bincount)
            prod = np.outer(w, w).ravel()
            conv = np.bincount(sum_mat.ravel(), weights=prod,
                               minlength=2 * n - 1)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            p = exp_beta / sum_exp                           # smooth‑max weights

            # gradient w.r.t. w
            p_mat = p[sum_mat]                               # shape m×m
            grad_w = 2.0 * (p_mat @ w)

            # chain rule to log‑space
            dot = np.dot(grad_w, w)
            grad_u = w * (grad_w - dot)

            # Adam step
            lr = 0.5 / (np.sqrt(t) + 1.0)
            m1 = beta1 * m1 + (1.0 - beta1) * grad_u
            m2 = beta2 * m2 + (1.0 - beta2) * (grad_u ** 2)
            m1_hat = m1 / (1.0 - beta1 ** t)
            m2_hat = m2 / (1.0 - beta2 ** t)
            u = u - lr * m1_hat / (np.sqrt(m2_hat) + eps)

            # exact surrogate on the full length‑n vector
            w_cur = _softmax(u)
            full_vec[:] = 0.0
            full_vec[idx] = w_cur
            cur_val = _objective_surrogate(full_vec)
            if cur_val < best_val:
                best_val = cur_val
                best_w = w_cur.copy()

    return best_w, best_val


# ----------------------------------------------------------------------
# Greedy construction of a (near) Sidon set – maximal size, few collisions
# ----------------------------------------------------------------------
def _sidon_greedy(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random greedy construction of a set S⊂{0,…,n‑1} such that all unordered
    sums i+j (i≤j) are distinct as far as possible.
    Returns the indices of the set (sorted).
    """
    candidates = np.arange(n, dtype=int)
    rng.shuffle(candidates)
    used = np.zeros(2 * n - 1, dtype=bool)   # sum i+j occupied ?
    chosen = []

    for i in candidates:
        if chosen:
            sums = i + np.array(chosen, dtype=int)
            if used[sums].any():
                continue
        if used[2 * i]:
            continue
        # accept i
        chosen.append(i)
        for j in chosen:          # mark all new sums, including i+i
            used[i + j] = True
    return np.array(sorted(chosen), dtype=int)

                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The algorithm runs three complementary phases
    (see the module docstring) and respects the allotted time budget.
    The best sequence found is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin for final I/O

    # ------------------------------------------------------------------
    # Problem size – use the previously stored length if available.
    # ------------------------------------------------------------------
    n = 4000
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) == n:
        seq0 = np.asarray(prev, dtype=np.float64)
        seq0 = np.maximum(seq0, 0.0)
        total = seq0.sum()
        if total > 0.0:
            seq0 = seq0 / total
        else:
            seq0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # fallback – start from the uniform distribution (ℓ₁‑norm = 1)
        seq0 = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = seq0.copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] surrogate of initial sequence = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimiser from the existing (or uniform) start.
    # ------------------------------------------------------------------
    seq_opt, val_opt = _adam_optimize_full(best_seq, n, deadline)
    if val_opt < best_val:

        print(f"[Full‑Adam] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – a handful of random Dirichlet restarts.
    # ------------------------------------------------------------------
    restarts = 5
    for r in range(1, restarts + 1):
        if time.time() > deadline:
            break
        rand_seq = rng.dirichlet(alpha=np.ones(n))
        seq_r, val_r = _adam_optimize_full(rand_seq, n, deadline)
        if val_r < best_val:
            best_seq, best_val = seq_r, val_r
            print(f"[Restart {r}] new best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – sparse Sidon‑like support + specialised optimisation.
    # ------------------------------------------------------------------
    # Build a large (near‑Sidon) support set.
    sidon_deadline = deadline - 20.0
    best_idx = None
    best_sz = 0
    while time.time() < sidon_deadline:
        cand = _sidon_greedy(n, rng)
        if cand.size > best_sz:
            best_sz = cand.size
            best_idx = cand
            print(f"[Sidon] new best support size = {best_sz}")
        # stop if we already have a reasonably large set (≈100)
        if best_sz >= 110:
            break

    if best_idx is not None and best_idx.size > 0:
        m = best_idx.size

        w_opt, val_w = _adam_optimize_support(best_idx, w0, n, deadline)
        seq_full = np.zeros(n, dtype=np.float64)
        seq_full[best_idx] = w_opt
        val_full = _objective_surrogate(seq_full)
        if val_full < best_val:
            best_seq, best_val = seq_full, val_full
            print(f"[Sidon‑opt] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final exact evaluation (the only place evaluate_sequence is called).
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] final surrogate = {best_val:.6f}, true value = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
