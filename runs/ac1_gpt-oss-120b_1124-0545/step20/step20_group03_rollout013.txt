def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0
    n = 4000
    # Init best
    best_seq = None
    best_val = float('inf')
    # Check previous
    prev = globals().get('height_sequence_1')
    if prev is not None:
        seq = np.asarray(prev, dtype=np.float64)
        if seq.shape != (n,):
            # ignore if mismatched length
            seq = None
        else:
            # Normalize to sum 1
            seq = seq.clip(min=0)
            tot = seq.sum()
            if tot>0:
                seq = seq / tot
                best_seq = seq.copy()
                best_val = _objective_surrogate(seq)
    # else define fallback
    if best_seq is None:
        # fallback: generate moderate exponential shape with rate approx 0.001
        seq = _exp_decay(n, rate=0.001) # placeholder but we'll produce via function
        best_seq = seq.copy()
        best_val = _objective_surrogate(seq)

    print(f"[Init] initial best surrogate = {best_val:.6f}")

    # Phase 1 random parameter search
    phase1_deadline = deadline - 200.0   # reserve time for later phases
    while time.time() < phase1_deadline:
        # Choose family
        family = rng.integers(0, 3)
        if family == 0:
            rate = 10 ** rng.uniform(-5, -2)  # from 1e-5 to 1e-2
            cand = generate_exp(n, rate)
        elif family == 1:
            p = rng.uniform(0.0, 2.5)
            cand = generate_power(n, p)
        else:
            rate = 10 ** rng.uniform(-5, -2)
            p = rng.uniform(0.0, 2.5)
            cand = generate_combined(n, rate, p)
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Random] new best surrogate {best_val:.6f} (family {family})")
    # Phase 2 L-BFGS smoothing if time
    if _opt is not None and time.time() < deadline - 100:
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline)
        val_lb = _objective_surrogate(seq_lb)
        if val_lb < best_val:
            best_val = val_lb
            best_seq = seq_lb.copy()
            print(f"[L-BFGS] improved surrogate {best_val:.6f}")

    # Phase 3 Exponentiated gradient (mirror descent)
    if time.time() < deadline - 30:
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, deadline,
                                            max_iters=500_000, eta0=5.0)
        if val_eg < best_val:
            best_val = val_eg
            best_seq = seq_eg.copy()
            print(f"[EG] improved surrogate {best_val:.6f}")

    # Phase 4 local random perturbations (swap + delta)
    while time.time() < deadline:
        # propose a small transfer of mass
        a = best_seq.copy()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to move: random proportion up to 20% of smaller entry
        delta = rng.uniform(0, min(a[i], a[j]) * 0.2)
        a[i] -= delta
        a[j] += delta
        # Re-normalize (should sum = unchanged, but rounding error maybe)
        # no need.
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Perturb] improved surrogate {best_val:.6f}")

    # final evaluation:
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] final surrogate = {best_val:.6f}, true value = {true_val:.6f}")
    except Exception as e:
        print("Eval error", e)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
propose_candidate – hybrid random‑search + smoothing + mirror‑descent

This implementation combines several ideas that were not used together
in the previous code:

1.  Random sampling of simple parametric families (exponential,
    power‑law, a product of both and an “arcsine” shape).  Each candidate
    is normalised to ℓ₁‑norm = 1 and evaluated with the exact
    max‑convolution objective (the same surrogate used by
    ``evaluate_sequence``).  Good candidates are kept as a fresh global
    best.

2.  A continuation L‑BFGS optimiser that minimises a smooth
    approximation of the max‑convolution (log‑sum‑exp).  The continuation
    schedule is longer than in the reference code (more β‑steps and
    more iterations per step) which gives the optimiser more time to
    converge.

3.  Exponentiated‑gradient (mirror‑descent) polishing on the exact
    objective.  The optimiser works in the log‑softmax space so the
    probability constraints are satisfied automatically.

4.  A final cheap stochastic perturbation phase that moves small amounts
    of mass between random positions; this can escape shallow local
    minima that the deterministic steps miss.

The whole search respects the 1000 second budget, keeps the globally
best sequence and stores it in ``height_sequence_1`` for the next run.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:       # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact max‑convolution, assumes ℓ₁‑norm = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) – the exact value returned by evaluate_sequence
    after normalising the ℓ₁‑norm to one."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax utilities (used by exponentiated‑gradient & L‑BFGS)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Simple “starter families” – they generate probability vectors.
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), i=0…n‑1, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _combined_exp_power(n: int, rate: float, p: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)·(i+1)^{-p}, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) * (i + 1) ** (-p)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _arcsine_shape(n: int) -> np.ndarray:
    """
    Discrete analogue of the arcsine distribution on {0,…,n‑1}:
        a_i ∝ 1/√((i+½)·(n‑i‑½))
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n - i))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5_000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space to stay feasible.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Strategy:

    1. Load the previously best sequence (global ``height_sequence_1``).  If it
       does not exist, initialise with a cheap random family.
    2. Randomly sample from several parametric families (exponential,
       power‑law, combined, arcsine).  Keep the globally best surrogate.
    3. Apply a long continuation L‑BFGS optimisation on a smooth max‑convolution
       surrogate.
    4. Polish the result with exponentiated‑gradient mirror‑descent on the exact
       max‑convolution.
    5. Final cheap stochastic “move‑mass” perturbations.
    6. Return (and store) the best sequence.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size – keep a reasonable size for the budget.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 1. Load previous solution or initialise a fallback.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            seq0 = np.asarray(prev, dtype=np.float64)
            if seq0.shape != (n,):
                raise ValueError
        except Exception:
            seq0 = None
    else:
        seq0 = None

    if seq0 is None:
        # start from a very mild exponential (rate ≈ 1e‑3) – cheap to compute.
        seq0 = _exp_decay(n, rate=0.001)

    # Normalise to ℓ₁‑norm = 1 and protect against tiny sums.
    seq0 = np.maximum(seq0, 0.0)
    tot = seq0.sum()
    if tot == 0.0:
        seq0 = _uniform(n)
    else:
        seq0 = seq0 / tot

    best_seq = seq0.copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Random search in low‑dimensional families.
    # --------------------------------------------------------------
    phase1_deadline = deadline - 250.0   # leave ~250 s for polishing phases
    families = ['exp', 'power', 'combined', 'arcsine']
    while time.time() < phase1_deadline:
        fam = rng.choice(families)
        if fam == 'exp':
            rate = 10 ** rng.uniform(-5, -2)        # 1e‑5 … 1e‑2
            cand = _exp_decay(n, rate)
        elif fam == 'power':
            p = rng.uniform(0.0, 2.5)
            cand = _power_law(n, p)
        elif fam == 'combined':
            rate = 10 ** rng.uniform(-5, -2)
            p = rng.uniform(0.0, 2.5)
            cand = _combined_exp_power(n, rate, p)
        else:   # arcsine
            cand = _arcsine_shape(n)

        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Random] new best surrogate {best_val:.6f} (family={fam})")

    # --------------------------------------------------------------
    # 3. L‑BFGS smoothing (continuation)
    # --------------------------------------------------------------
    if _opt is not None and time.time() < deadline - 120.0:
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                        max_iter_per_beta=5_000)
        val_lb = _objective_surrogate(seq_lb)
        if val_lb < best_val:
            best_val = val_lb
            best_seq = seq_lb.copy()
            print(f"[L‑BFGS] improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Exponentiated‑gradient polishing (exact max‑convolution)
    # --------------------------------------------------------------
    if time.time() < deadline - 40.0:
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, deadline,
                                           max_iters=1_500_000,
                                           eta0=5.0)
        if val_eg < best_val:
            best_val = val_eg
            best_seq = seq_eg.copy()
            print(f"[EG] improved surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final cheap stochastic “move mass” perturbations.
    # --------------------------------------------------------------
    while time.time() < deadline:
        a = best_seq.copy()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = rng.uniform(0.0, min(a[i], a[j]) * 0.2)
        a[i] -= delta
        a[j] += delta
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Perturb] new best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] final surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Result] evaluate_sequence raised an exception:", exc)

    # Store for the next run (facilitates incremental improvement)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
