def adam_opt(seq_init, n, deadline, max_iters=500000, lr=0.1):
    a = seq_init.copy()
    v = np.log(a + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v) # second moment
    beta1=0.9; beta2=0.999; eps=1e-8
    for t in range(1, max_iters+1):
        if time.time() > deadline: break
        a = _softmax(v)   # recompute a each step
        conv = _conv_fft(a, a)   # O
        k = np.argmax(conv)
        max_val = conv[k]
        # gradient g_j = 2 * a[k-j] for indices with 0 <= k-j < n
        g = np.zeros_like(a)
        # compute shift
        start = max(0, k - (n-1))
        # i index range in a for which k-i in range
        # Actually j in [0,n-1], then k-j in [0,n-1] => j ∈ [k-(n-1), k]
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        idx = np.arange(j_start, j_end+1)
        rev = k - idx
        g[idx] = 2.0 * a[rev]  # vector assignment
        # compute grad w.r.t logits:
        dot = np.dot(g, a)
        grad_v = a * (g - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v2 = beta2 * v2 + (1 - beta2) * (grad_v**2)
        m_hat = m / (1 - beta1**t)
        v2_hat = v2 / (1 - beta2**t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)
        # track best
        cur_val = 2 * n * max_val  # since sum=1, objective = 2*n*max_val
        if cur_val < best_val:
            best_val = cur_val; best_seq = a.copy()
    return best_seq, best_val

"""
propose_candidate
-----------------

Searches for a non‑negative height sequence that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The method combines:
  * a large collection of analytic and random initial families,
  * many short Adam‑optimisation runs on the *exact* max‑convolution
    (mirror‑descent on the log‑softmax parametrisation),
  * a final high‑precision exponentiated‑gradient polish,
  * a brief simulated‑annealing fine‑tune.

All work respects the global 1000 s budget and stores the best sequence in
the global variable ``height_sequence_1`` for future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (the true objective when the sum is 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) for a sequence with sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑shape (i=1..n)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Adam optimisation on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 500_000,
              lr: float = 0.1,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Adam) on the exact max‑convolution objective.
    Works in the log‑softmax (probability‑simplex) representation.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    v = np.log(a + 1e-30)                     # logits
    m = np.zeros_like(v)                     # first moment
    v2 = np.zeros_like(v)                    # second moment
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- probability vector -----
        a = _softmax(v)

        # ----- convolution & max index -----
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        max_val = conv[k]
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ----- sub‑gradient of max(conv) w.r.t. a (single active index) -----
        g = np.zeros_like(a)                  # d (max) / d a
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]                 # ∂conv_k/∂a_j = 2·a_{k‑j}
        g *= (2.0 * n)                        # factor from the outer objective

        # ----- chain rule to logits (softmax Jacobian) -----
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # ----- Adam parameter update -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        # keep logits in a sane range (prevents overflow)
        if not np.isfinite(v).all():
            v = np.log(np.maximum(a, 1e-30))
        else:
            v = np.clip(v, -30.0, 30.0)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) polishing – same as the
# reference implementation but kept here for stand‑alone use.
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    High‑precision EG polishing using the exact max‑convolution.
    Returns best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        eps = max_val * 1e-6
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient (averaged over all active indices)
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (lightweight)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.01,
                         T_end: float = 1e-6) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

                                              T_end=1e-9)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"[SA] Improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the supplied function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print("\n=== Finished ===")
        print(f"Surrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Error] evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
