"""
propose_candidate
-----------------

Search for a non‑negative step function (probability vector) that minimises
``evaluate_sequence``.  The core idea is to start from the analytically
optimal *arcsine* (Beta(0.5,0.5)) shape – the continuous distribution whose
self‑convolution is the flattest possible – and then improve it with a few
iterations of exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
objective, followed by a smooth L‑BFGS optimisation (log‑sum‑exp surrogate).
A modest sweep around the arcsine parameters (β‑shapes) and a handful of
random Dirichlet / low‑frequency seeds are also evaluated; the best few are
refined.  The whole procedure respects a 1000 s wall‑time budget and returns
the best vector found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric helpers (FFT‑based convolution, softmax, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a vector already ℓ₁‑normalised."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) shape on {1,…,n} (ℓ₁‑normalised)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth low‑frequency cosine mixture, forced non‑negative and normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() <= 0.0:
        return _uniform(n)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration ‘conc’ (already normalised)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Convex‑combination helpers
# ----------------------------------------------------------------------
def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     eta0: float = 5.0):
    """Mirror‑descent on max‑convolution (exact). Returns best vector seen."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)               # stay inside simplex via softmax

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))           # active index (max entry)

        # sub‑gradient of the active entry: g_i = 2*a_{k-i} for i that contribute
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000):
    """Smooth‑max L‑BFGS‑B optimisation (variables live in log‑softmax space)."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1000.0, 2000.0, 5000.0,
                         20000.0, 50000.0, 100000.0]

    v = np.log(start_vec + 1e-30)          # log‑softmax coordinates

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # numeric stabilisation
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

            # gradient wrt a (in softmax space)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine:
        * builds a wide set of analytic and random seeds (including the exact arcsine shape);
        * evaluates the cheap surrogate (2·n·max(conv));
        * refines the best few seeds with exponentiated‑gradient descent;
        * optionally smooths with L‑BFGS;
        * returns the best vector found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.5      # keep a safety margin

    # --------------------------------------------------------------
    # 1. Choose a problem size (larger n gives a tighter theoretical bound)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 2. Generate a diverse collection of seed vectors
    # --------------------------------------------------------------
    seeds = []

    # plain families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # the analytically motivated arcsine (Beta(0.5,0.5))
    seeds.append(_beta_shape(n, 0.5, 0.5))

    # fine sweep around the arcsine parameters – this is the main novelty
    alphas = np.arange(0.35, 0.66 + 1e-9, 0.02)   # 0.35,0.37,…,0.65
    betas  = np.arange(0.35, 0.66 + 1e-9, 0.02)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # a handful of random Dirichlet draws (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency cosine mixtures – smooth but non‑parametric
    for nf in (2, 3, 4, 5):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # include the previous best if it exists
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                arr = arr / arr.sum()
                seeds.append(arr)
                for _ in range(2):
                    seeds.append(_mix_shapes(arr, _uniform(n), 0.1))
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. Cheap surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top surrogate values after seeding ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 4. Refine the top candidates
    # --------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Refinement for candidate #{idx+1} ---")

        # Exponentiated‑gradient (exact max) – allocate ~40 % of remaining budget
        remaining = deadline - time.time()
        eg_deadline = time.time() + max(4.0, 0.4 * remaining)
        seq_eg, val_eg = _exp_gradient_opt(seed_seq.copy(),
                                           n,
                                           eg_deadline,
                                           max_iters=2_000_000,
                                           eta0=5.0)
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print(f"  EG improvement: {val_eg:.6f}")

        if time.time() > deadline:
            break

        # Smooth L‑BFGS refinement (if SciPy is available)
        lbfgs_deadline = deadline - 1.0
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, lbfgs_deadline,
                                        beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                       200.0, 500.0, 1000.0,
                                                       2000.0, 5000.0,
                                                       20000.0, 50000.0,
                                                       100000.0, 200000.0,
                                                       500000.0, 1000000.0],
                                        max_iter_per_beta=3000)
        val_lb = _objective_surrogate(seq_lb)
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()
            print(f"  L‑BFGS improvement: {val_lb:.6f}")

    # --------------------------------------------------------------
    # 5. Official evaluation with the provided function
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        true_val = None
        print("Warning: evaluate_sequence raised", exc)

    print("\n=== Finished ===")
    print(f"Best surrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val}")
    print(f"Length = {len(best_seq)}")

    # store for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
