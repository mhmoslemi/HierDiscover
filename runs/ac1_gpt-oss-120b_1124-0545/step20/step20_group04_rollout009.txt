"""
Improved step‑function search for the evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works as follows

1. **Rich seed pool** – analytic windows (uniform, exponential,
   power‑law, Gaussian, beta‑shapes, Dirichlet draws, etc.) and a few
   perturbations of the previous best sequence (if any).

2. **Fast surrogate evaluation** – the objective for ℓ₁‑normalised
   vectors is `2·n·max(conv)`.  Convolution is carried out by a real FFT.

3. **Continuation Adam optimisation** – a smooth approximation of the
   max via a log‑sum‑exp is minimised in log‑softmax space using the
   Adam optimiser.  The smoothing parameter β is increased
   (continuation) and the best vector after each stage is kept.

4. **Exponentiated‑gradient polishing** – exact sub‑gradient descent on
   the true max‑convolution (the same EG used in the reference code).

5. **Fine‑grained random perturbations** – short stochastic hill‑climbing
   with tiny Gaussian moves to escape possible plateaus.

All steps respect the global time budget (default 1000 s).  The best
sequence found is stored in ``height_sequence_1`` and returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv.astype(np.float64)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors of length n
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sample from a symmetric Dirichlet."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray,
             rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence found and stores it in the global variable
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size (keep the same dimension as the reference run)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if any) to seed the search
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_norm = None
    if isinstance(previous, list) and len(previous) == n:
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            prev_norm = prev_arr / prev_arr.sum()

    # ------------------------------------------------------------------
    # 1. Seed generation (rich family)
    # ------------------------------------------------------------------
    seeds = []

    # Analytic families (all are already ℓ₁‑normalised)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Exponential decays – a broader grid than before
    for gamma in np.linspace(0.05, 12.0, 25):
        seeds.append(_exp_decay(n, gamma))

    # Power‑law shapes
    for p in np.linspace(0.0, 3.0, 25):
        seeds.append(_power(n, p))

    # Gaussian windows
    for sigma in np.linspace(0.04, 0.30, 15):
        seeds.append(_gaussian(n, sigma))

    # Beta‑shapes – coarse grid
    alphas = np.linspace(0.2, 5.0, 6)
    betas = np.linspace(0.2, 5.0, 6)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # Dirichlet random draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Perturbations of the previous best (if available)
    if prev_norm is not None:
        seeds.append(prev_norm)
        for _ in range(8):
            seeds.append(_perturb(prev_norm, rng, sigma=0.004))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    best_seq, best_val = evaluated[0][0].copy(), evaluated[0][1]
    print(f"\n[Init] best surrogate after seeding = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation Adam optimisation on the smooth max‑convolution
    # ------------------------------------------------------------------
    # Smooth max via log‑sum‑exp, continuation on β.
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                     1000.0, 2000.0, 5000.0, 10000.0, 20000.0]
    total_iters_per_beta = 4000          # a few thousand Adam steps per β

    # Work in log‑softmax space to stay on the simplex.
    log_a = np.log(best_seq + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # Adam hyper‑parameters
        lr = 0.04          # base learning rate (works well for all β)
        beta1, beta2 = 0.9, 0.999
        m = np.zeros_like(log_a)
        v = np.zeros_like(log_a)
        t_it = 0

        while t_it < total_iters_per_beta and time.time() < deadline:
            t_it += 1

            a = _softmax(log_a)                         # current probabilities
            conv = _conv_fft(a, a)                       # full convolution
            max_c = np.max(conv)
            shift = max_c
            # soft‑max weights over the convolution entries
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp

            # Gradient of the smooth objective w.r.t. a
            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

            # Convert to gradient w.r.t. log‑softmax variables
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_v
            v = beta2 * v + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t_it)
            v_hat = v / (1 - beta2 ** t_it)
            log_a -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # Periodic evaluation of the surrogate to keep the global best
            if t_it % 250 == 0:
                cur_a = _softmax(log_a)
                cur_val = _objective_surrogate(cur_a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_a.copy()
                    print(f"[Adam β={beta:.0e} iter={t_it:5d}] new best surrogate = {best_val:.6f}")

        # Reset the optimiser state for the next β, but keep the best vector.
        log_a = np.log(best_seq + 1e-30)

    print(f"\n[Adam] best surrogate after continuation = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Random local search (tiny perturbations) – quick hill‑climbing
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # generate a tiny perturbation of the current best
        cand = _perturb(best_seq, rng, sigma=0.0015)
        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand
            print(f"[Perturb] improved surrogate = {best_val:.6f}")
        # stop after a few hundred attempts without improvement
        if time.time() > deadline - 0.5:
            break

    # ------------------------------------------------------------------
    # 5. Final EG polishing on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_eg, val_eg = _exp_gradient_opt(best_seq,
                                           n,
                                           deadline,
                                           max_iters=600_000,
                                           eta0=6.0)
        if val_eg < best_val:
            best_seq, best_val = seq_eg, val_eg
            print(f"[EG] improved surrogate after polishing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation (exact objective)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
