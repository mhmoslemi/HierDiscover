def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5

    n = 4000

    seeds = []

    # deterministic families:
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    seeds.append(_alternating_dirichlet(n, rng))

    # add new families
    # sine
    for gamma in np.linspace(0.1, 5.0, 180):
        seeds.append(_sine_window(n, gamma))

    # Kaiser
    for beta in np.linspace(0.1, 30.0, 180):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian
    for sigma_frac in np.linspace(0.02, 0.5, 100):
        seeds.append(_gaussian(n, sigma_frac))

    # Double Gaussian near edges
    for sigma in (0.05, 0.08, 0.12, 0.15, 0.20):
        for shift in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.45, 0.5):
            seeds.append(_double_gaussian(n, sigma, shift))

    # Power law
    for p in np.linspace(0.0, 2.5, 150):
        seeds.append(_power(n, p))

    # Beta shape grid
    alphas = np.linspace(0.05, 5.0, 40)  # extended to low 0.05
    betas = np.linspace(0.05, 5.0, 40)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # Tukey
    for a_ in np.linspace(0.0, 1.0, 51):
        seeds.append(_tukey_window(n, a_))

    # Edge reciprocal
    for p in np.linspace(0.1, 2.0, 60):
        seeds.append(_edge_reciprocal(n, p))

    # Edge exponential
    for lam in np.linspace(0.0005, 0.05, 60):
        seeds.append(_edge_exponential(n, lam))

    # Random Dirichlet low concentration
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential distribution
    for _ in range(30):
        seeds.append(_random_exponential(n, rng))

    # Low frequency curves
    for nf in (2,3,4,5,6):
        for _ in range(10):

    if rng is None:
        rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    total_time = deadline - time.time()
    start_time = time.time()
    for move in range(max_moves):
        if time.time() > deadline:
            break
        elapsed = time.time() - start_time
        # linear temperature schedule
        t_frac = elapsed / total_time if total_time > 0 else 1.0
        T = init_temp * (1 - t_frac) + final_temp * t_frac
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:

        new_seq[j] += delta

            cur_val = new_val

                best_val = cur_val

"""
Propose a step‑function (non‑negative heights) that minimises the
``evaluate_sequence`` objective.

Algorithm (≈ 1000 s budget)

1. **Very broad seeding** – generate thousands of ℓ₁‑normalised candidates
   from analytic families (sine, Kaiser, Gaussian, power‑law, beta,
   Tukey, …) plus a new collection of *U‑shaped* families:
   *edge‑reciprocal*, *edge‑exponential* and *double‑Gaussian* with peaks
   near the borders.  Random Dirichlet, exponential, low‑frequency cosine
   curves and mixtures with uniform are also added.  All are normalised so
   that their sum is 1, making the surrogate objective
   ``_objective_surrogate`` exact for the true score.

2. **Surrogate filtering** – evaluate the cheap surrogate
   ``2·n·max(conv)`` for every seed, retain the best ~15 candidates.

3. **Refinement** – for each top seed:
   * Exponentiated‑gradient (mirror‑descent) optimisation on the exact
     max‑convolution (sub‑gradient of the active entry) for a few
     seconds.
   * Continuation L‑BFGS on a smooth log‑sum‑exp approximation of the max,
     with an extended temperature schedule.
   * Convex ternary mixing with the uniform distribution and with the
     reversed vector (both can lower the peak).
   * Simple stochastic hill‑climbing (mass‑transfer moves).
   * Simulated‑annealing (SA) that performs the same mass‑transfer moves
     but accepts worsening moves with a temperature schedule, allowing
     escape from local minima.

   The best vector seen over all refinements is kept.

4. **Final evaluation** – the best vector is passed to the provided
   ``evaluate_sequence`` function, stored globally as ``height_sequence_1``,
   and returned as a plain ``list[float]``.

All heavy work respects the global deadline; the routine never exceeds
the allotted 1000 s.  No file or network I/O is performed.
"""

# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector ℓ₁‑normalised to 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _edge_reciprocal(n: int, p: float = 1.0) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p + 1/(n-i)^p, symmetric, higher at the borders."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / ((i + 1) ** p) + 1.0 / ((n - i) ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _edge_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    """a_i ∝ exp(-lam·i) + exp(-lam·(n‑1‑i)), symmetric."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Random families

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


def _ternary_mix(a: np.ndarray, b: np.ndarray, deadline: float,
                 steps: int = 30):
    """Best convex combination of a and b by ternary search on α∈[0,1]."""
    lo, hi = 0.0, 1.0
    best_seq, best_val = None, float('inf')
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)
        if val1 < val2:
            hi = m2
            if val1 < best_val:
                best_val, best_seq = val1, seq1
        else:
            lo = m1
            if val2 < best_val:
                best_val, best_seq = val2, seq2
    mid = (lo + hi) / 2.0
    seq_mid = _mix_shapes(a, b, mid)
    val_mid = _objective_surrogate(seq_mid)
    if val_mid < best_val:
        best_seq, best_val = seq_mid, val_mid
    return best_seq, best_val

# Exponentiated‑gradient (mirror‑descent) optimisation
# ------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     eta0: float = 5.0):
    """Mirror‑descent on exact max‑convolution. Returns best sequence seen."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0,
                         5000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (softmax space)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)          # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ------------------------------------------------------------
# Simple stochastic local search (hill climbing)
# ------------------------------------------------------------
def _local_search(seq: np.ndarray,
                  deadline: float,
                  moves: int = 20000,
                  step_scale: float = 0.02):
    """
    Random mass‑transfer hill‑climbing.
    """
    rng = np.random.default_rng()
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    n = len(seq)

    for _ in range(moves):
        if time.time() > deadline:
            break
        eps = step_scale * rng.random()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = min(eps, best_seq[i])
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq /= new_seq.sum()
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq
    return best_seq, best_val

# Simulated annealing (exploratory stochastic search)
# ------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         max_moves: int = 50000,
                         init_temp: float = 1e-3,
                         final_temp: float = 1e-8,
                         step_scale: float = 0.02,
                         rng: np.random.Generator = None):
    """
    Mass‑transfer SA that can escape local minima.
    """
    if rng is None:
        rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    n = len(start_seq)

    total_time = max(0.0, deadline - time.time())
    start_time = time.time()

    for _ in range(max_moves):
        if time.time() > deadline:
            break
        elapsed = time.time() - start_time
        t_frac = elapsed / total_time if total_time > 0 else 1.0
        # linear cooling
        T = init_temp * (1.0 - t_frac) + final_temp * t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if cur_seq[i] <= 0.0:
            continue
        delta = step_scale * rng.random() * cur_seq[i]
        if delta <= 0.0:
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq /= new_seq.sum()
        new_val = _objective_surrogate(new_seq)
        delta_val = new_val - cur_val
        if delta_val < 0.0 or rng.random() < math.exp(-delta_val / max(T, 1e-20)):
            cur_seq = new_seq
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
    return best_seq, best_val

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  Returns the best sequence found within the
    allocated time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # Fixed length (larger n gives finer discretisation, but the objective
    # depends mainly on the shape).  Empirically n = 4000 performed well.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Broad seed generation (many analytic and random families)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n = {n}) ---")
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    seeds.append(_alternating_dirichlet(n, rng))

    # sine window – dense grid
    for gamma in np.linspace(0.1, 5.0, 180):
        seeds.append(_sine_window(n, gamma))

    # Kaiser window – wide grid
    for beta in np.linspace(0.1, 30.0, 180):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian – medium grid
    for sigma_frac in np.linspace(0.02, 0.5, 100):
        seeds.append(_gaussian(n, sigma_frac))

    # Double‑Gaussian – various edge‑shifts
    for sigma in (0.05, 0.08, 0.12, 0.15, 0.20):
        for shift in (0.05, 0.10, 0.15, 0.20, 0.30, 0.40, 0.45, 0.50):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))

    # Power‑law
    for p in np.linspace(0.0, 2.5, 150):
        seeds.append(_power(n, p))

    # Beta‑shape – extended grid (including very small alphas/betas)
    alphas = np.linspace(0.05, 5.0, 40)
    betas = np.linspace(0.05, 5.0, 40)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # Tukey window
    for a_ in np.linspace(0.0, 1.0, 51):
        seeds.append(_tukey_window(n, a_))

    # NEW edge‑heavy families
    for p in np.linspace(0.1, 2.0, 60):
        seeds.append(_edge_reciprocal(n, p))
    for lam in np.linspace(0.0005, 0.05, 60):
        seeds.append(_edge_exponential(n, lam))

    # Random Dirichlet (very low concentrations)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential
    for _ in range(30):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(10):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mix low‑freq curves with uniform
    for mix_alpha in (0.1, 0.3, 0.5, 0.7):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous best (if any)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                arr = arr / arr.sum()
                seeds.append(arr)
                for _ in range(5):
                    seeds.append(_perturb(arr, rng, sigma=0.002))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Surrogate evaluation of all seeds (cheap)
    # ------------------------------------------------------------------
    print("\n--- Evaluating seeds (surrogate) ---")
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(15, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop surrogate values after seeding:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1}: {v:.9f}")

    # ------------------------------------------------------------------
    # 3. Refinement of top seeds
    # ------------------------------------------------------------------
    best_overall_seq = top_seeds[0].copy()
    best_overall_val = _objective_surrogate(best_overall_seq)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Refinement for seed #{idx + 1} ---")

        # --- Exponentiated‑gradient stage ---
        remaining = deadline - time.time()
        eg_time = min(30.0, 0.25 * remaining)   # allocate at most 30 s
        seq_eg, val_eg = _exp_gradient_opt(seed_seq.copy(),
                                           n,
                                           time.time() + eg_time,
                                           max_iters=1_500_000,
                                           eta0=8.0)
        if val_eg < best_overall_val:
            best_overall_val, best_overall_seq = val_eg, seq_eg.copy()
            print(f"  EG improvement: {val_eg:.9f}")

        if time.time() > deadline:
            break

        # --- Continuation L‑BFGS smoothing ---
        seq_lb = _lbfgs_smooth_optimize(best_overall_seq,
                                        n,
                                        deadline,
                                        beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                       200.0, 500.0, 1000.0,
                                                       2000.0, 5000.0,
                                                       20000.0, 50000.0,
                                                       100000.0, 200000.0,
                                                       500000.0, 1000000.0,
                                                       5000000.0],
                                        max_iter_per_beta=3000)
        val_lb = _objective_surrogate(seq_lb)
        if val_lb < best_overall_val:
            best_overall_val, best_overall_seq = val_lb, seq_lb.copy()
            print(f"  L‑BFGS improvement: {val_lb:.9f}")

        if time.time() > deadline:
            break

        # --- Mixing with uniform (ternary search) ---
        mixed_seq, mixed_val = _ternary_mix(best_overall_seq,
                                            _uniform(n),
                                            deadline,
                                            steps=30)
        if mixed_seq is not None and mixed_val < best_overall_val:
            best_overall_val, best_overall_seq = mixed_val, mixed_seq.copy()
            print(f"  Mix‑uniform improvement: {mixed_val:.9f}")

        # --- Mixing with reversed vector (symmetry) ---
        rev_seq = best_overall_seq[::-1]
        mixed_seq2, mixed_val2 = _ternary_mix(best_overall_seq,
                                              rev_seq,
                                              deadline,
                                              steps=20)
        if mixed_seq2 is not None and mixed_val2 < best_overall_val:
            best_overall_val, best_overall_seq = mixed_val2, mixed_seq2.copy()
            print(f"  Mix‑reversed improvement: {mixed_val2:.9f}")

        if time.time() > deadline:
            break

        # --- Simple stochastic local search ---
        seq_loc, val_loc = _local_search(best_overall_seq,
                                         deadline,
                                         moves=20000,
                                         step_scale=0.015)
        if val_loc < best_overall_val:
            best_overall_val, best_overall_seq = val_loc, seq_loc.copy()
            print(f"  Local‑search improvement: {val_loc:.9f}")

        if time.time() > deadline:
            break

        # --- Simulated annealing ---
        seq_sa, val_sa = _simulated_annealing(best_overall_seq,
                                              deadline,
                                              max_moves=50000,
                                              init_temp=1e-3,
                                              final_temp=1e-8,
                                              step_scale=0.02,
                                              rng=rng)
        if val_sa < best_overall_val:
            best_overall_val, best_overall_seq = val_sa, seq_sa.copy()
            print(f"  Simulated‑annealing improvement: {val_sa:.9f}")

    # ------------------------------------------------------------------
    # 4. Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised", exc)

    print(f"\n=== Finished ===\nBest surrogate ≈ {best_overall_val:.9f}"
          f", true evaluate_sequence = {true_val}\nLength = {len(best_overall_seq)}")

    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
