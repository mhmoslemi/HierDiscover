def _adam_single_active_opt(start_seq, n, deadline, max_iters=1_000_000, eta0=5.0):
    a = start_seq.copy()
    log_a = np.log(a+1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t=0
    beta1=0.9
    beta2=0.999
    eps=1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    while t < max_iters and time.time() < deadline:
        t+=1
        conv = _conv_fft(a,a)
        k = int(conv.argmax())
        # compute gradient of conv_k with respect to a: g_i = 2 * sum_{j | i+j = k} a_j
        # Equivalent to for each i, j = k - i if in [0, n-1]; sum contributions
        # As earlier in exp_gradient_opt, they compute g_tot by adding a[rev_idx] for each i; if we use approach:
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        idx_i = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = k - idx_i
        g = np.zeros_like(a)
        g[idx_i] = a[rev_idx]  # this adds a[rev_idx] for each i
        # But need factor 2? Since derivative includes both terms; we can include factor 2
        g = 2.0 * g
        # Compute gradient for log_a
        dot = np.dot(g, a)
        grad_log = a * (g - dot)
        # Adam updates
        m = beta1 * m + (1-beta1) * grad_log
        v = beta2 * v + (1-beta2) * (grad_log**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # Learning rate schedule: maybe eta = eta0 / (np.sqrt(t) + 1)
        eta = eta0 / (np.sqrt(t) + 1.0)
        log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)
        # evaluate surrogate occasionally (every 1000 iterations)
        if t % 1000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    return best_seq, best_val

"""
optimize_step_function.py

Search for a non‑negative step‑function (sequence of heights) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The algorithm:
1. Build a diverse pool of analytic seeds (uniform, triangular, beta‑skew,
   exponential, power‑law, random Dirichlet, plus small perturbations of the
   previous best if available).  All seeds are ℓ₁‑normalised.
2. Keep the top‑K seeds according to the cheap surrogate
       surrogate(a) = 2·n·max(conv(a,a))   (a already normalised, sum = 1)
3. For each top seed run a **single‑active Adam optimiser** in the
   log‑softmax (mirror‑descent) space.  The gradient is the exact sub‑gradient
   of the active convolution index (the one attaining the current maximum).
   Adam’s momentum accelerates convergence compared with plain exponentiated‑
   gradient.
4. Keep the best solution among all runs.
5. Polish the global best with a long Adam run (up to the remaining budget)
   and a short simulated‑annealing pass for fine‑grained stochastic
   improvement.
6. Return the final sequence (list of floats).  The result is stored in the
   global variable ``height_sequence_1`` for future runs.

The whole routine respects a 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i), i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sparse Dirichlet draws (conc < 1 gives heavy sparsity)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser for the *single‑active* max‑convolution sub‑gradient
# ----------------------------------------------------------------------
def _adam_single_active_opt(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 500_000,
                            eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent in log‑softmax space with Adam momentum,
    using the exact sub‑gradient of the currently active convolution index.
    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # main loop
    while t < max_iters and time.time() < deadline:
        t += 1

        # compute full convolution and locate the active index
        conv = _conv_fft(a, a)
        k = int(conv.argmax())

        # gradient of conv_k w.r.t. a (exact single‑active sub‑gradient)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g = np.zeros_like(a)
        g[idx_i] = a[rev_idx]          # contribution a[rev_idx] for each i
        g = 2.0 * g                    # factor 2 from the derivative

        # chain rule for log‑softmax (mirror‑descent)
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam update (learning‑rate decays like 1/√t)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = eta0 / (math.sqrt(t) + 1.0)
        log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)

        a = _softmax(log_a)

        # occasional evaluation (every 1000 steps)
        if t % 1000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (lightweight)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 80_000,
                         start_sigma: float = 0.001,
                         T_start: float = 0.01,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA that perturbs the probability vector with a diminishing
    Gaussian noise and a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best found sequence in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5          # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_seq = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,) and np.all(arr >= 0):
                s = arr.sum()
                if s > 1e-12:
                    prev_seq = arr / s
        except Exception:
            prev_seq = None

    # ------------------------------------------------------------------
    # 1. Build a large pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))

    # beta‑skewed families (both left‑ and right‑skew)
    alphas = np.linspace(0.2, 5.0, 12)
    betas = np.linspace(0.2, 5.0, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))
            seeds.append(_beta_shape(n, b, a))   # flipped version

    # exponential decays (fast and slow)
    rates = np.logspace(-4, -0.5, 20)
    for r in rates:
        exp_seq = _exp_decay(n, r)
        seeds.append(exp_seq)
        seeds.append(exp_seq[::-1])               # reversed (right‑skew)

    # power‑law families
    ps = np.linspace(0.0, 3.0, 15)
    for p in ps:
        p_seq = _power(n, p)
        seeds.append(p_seq)
        seeds.append(p_seq[::-1])

    # random Dirichlet draws (several concentration values)
    for conc in (0.001, 0.005, 0.01, 0.05, 0.1):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # perturbations of the previous best, if any
    if prev_seq is not None:
        seeds.append(prev_seq)
        for _ in range(6):
            pert = prev_seq + 0.01 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0.0:
                seeds.append(pert / s)

    # a few random shifts of an exponential shape (to explore boundary effects)
    for _ in range(20):
        r = rng.choice(rates)
        base = _exp_decay(n, r)
        shift = rng.integers(0, n // 4)
        shifted = np.concatenate([np.zeros(shift), base[: n - shift]])
        shifted = shifted / shifted.sum()
        seeds.append(shifted)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        raise RuntimeError("No valid seeds produced within the time budget.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(30, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values (lower is better):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement with Adam (single active)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a modest slice of the remaining budget to this seed
        remaining = deadline - time.time()
        # give each remaining seed roughly the same fraction (but not less than 0.5 s)
        per_seed = max(0.5, remaining * 0.12 / max(1, (top_k - idx)))
        sub_deadline = time.time() + per_seed

        seq_opt, val_opt = _adam_single_active_opt(seed_vec.copy(),
                                                   n,
                                                   sub_deadline,
                                                   max_iters=200_000,
                                                   eta0=8.0)

        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Seed {idx+1}/{top_k}] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing – long Adam run + brief SA
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long Adam polishing (single‑active) ---")
        final_seq, final_val = _adam_single_active_opt(global_best_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=2_000_000,
                                                       eta0=6.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final Adam] improved surrogate to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final simulated‑annealing polishing ---")
        sa_seq, sa_val = _simulated_annealing(global_best_seq,
                                              n,
                                              deadline,
                                              rng,
                                              max_attempts=100_000,
                                              start_sigma=0.001,
                                              T_start=0.008,
                                              T_end=1e-9)
        if sa_val < global_best_val:
            global_best_seq = sa_seq
            global_best_val = sa_val
            print(f"[Final SA] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
