"""
propose_candidate – multi‑restart Adam‑smoothed optimisation

The routine builds a few sensible initial probability vectors (uniform,
exponential decays, random Dirichlet draws and – when available – the
previous best sequence).  For every start it runs an Adam optimiser on a
smooth approximation of the exact objective

    F(a) = 2·n·max_k (a∗a)_k ,   a_i ≥ 0 ,   Σ a_i = 1

(the smoothing uses a log‑sum‑exp with a schedule of increasing β).
After the smooth phase a cheap greedy “mass‑transfer” local search tries
tiny redistributions of probability mass.  The best sequence seen over
all restarts is finally evaluated with the provided ``evaluate_sequence``
function and stored globally as ``height_sequence_1`` for the next run.

The optimiser works entirely on the log‑softmax parameters to keep the
probability simplex feasible, uses FFT‑based convolutions (O(n log n)),
and respects the 1000 s budget.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (fast for n≈4000)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helper utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _uniform(n: int) -> np.ndarray:
    """Uniform probability vector (ℓ₁‑norm = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a probability vector: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser on a β‑smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          beta_schedule=None,
                          steps_per_beta: int = 5000,
                          lr0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed objective.
    Parameters
    ----------
    start_seq : np.ndarray
        Initial probability vector (already normalised).
    n : int
        Length of the vector.
    deadline : float
        Absolute time at which the optimisation must stop.
    beta_schedule : list[float] | None
        Sequence of β values – larger β makes the smooth max approach the
        true max.  If ``None`` a default schedule is used.
    steps_per_beta : int
        Number of Adam steps for each β.
    lr0 : float
        Initial learning‑rate (scaled by 1/√t internally).
    Returns
    -------
    best_seq, best_val : (np.ndarray, float)
        The best probability vector discovered and its exact surrogate value.
    """
    if beta_schedule is None:
        beta_schedule = [
            0.1, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0,
            100.0, 200.0, 500.0, 1_000.0, 2_000.0, 5_000.0,
            10_000.0, 20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0
        ]

    # work in log‑softmax space – this automatically enforces non‑negativity
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)   # first moment (momentum)
    s = np.zeros_like(v)   # second moment (RMS)
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(start_seq)

    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return best_seq, best_val
            t += 1

            # ----- forward pass -----
            a = _softmax(v)                     # probability vector (ℓ₁ = 1)
            conv = _conv_fft(a, a)              # length 2n‑1
            max_c = np.max(conv)

            # smooth weighting w = softmax(β·(conv‑max_c))
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp               # shape 2n‑1, sums to 1

            # ----- gradient w.r.t. a -----
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # ∂ (β‑smooth max) / ∂ a

            # ----- chain rule back to v (logits) -----
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ----- Adam update (no bias‑correction – cheap & works) -----
            m = 0.9 * m + 0.1 * grad_v
            s = 0.999 * s + 0.001 * (grad_v ** 2)
            lr = lr0 / math.sqrt(t)             # decaying learning‑rate
            v = v - lr * m / (np.sqrt(s) + 1e-12)

            # ----- occasional exact evaluation (for bookkeeping) -----
            if t % 500 == 0:
                cur_a = _softmax(v)
                cur_val = _objective_surrogate(cur_a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_a.copy()

    # final exact check
    final_a = _softmax(v)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_val = final_val
        best_seq = final_a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small greedy local‑search that redistributes a tiny amount of mass
# ----------------------------------------------------------------------
def _local_improve(seq: np.ndarray,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing that repeatedly moves a small random fraction of
    probability mass from one index to another.  It accepts a move only
    if it strictly improves the exact surrogate; an occasional tiny
    uphill jump (probability 0.001) helps to escape shallow plateaux.
    """
    a = seq.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue

        # move a random fraction (up to 5 %) of a[i] to position j
        frac = rng.random() * 0.05
        delta = a[i] * frac
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur_val = _objective_surrogate(new_a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = new_a.copy()
            a = new_a
        else:
            # occasional uphill acceptance
            if rng.random() < 0.001:
                a = new_a

        # every few thousand iterations also try a deterministic ordering‑step:
        if (it + 1) % 10_000 == 0:
            # sort decreasing – this is cheap and often improves the peak
            sorted_a = np.sort(best_a)[::-1]
            sorted_val = _objective_surrogate(sorted_a)
            if sorted_val < best_val:
                best_val = sorted_val
                best_a = sorted_a.copy()
                a = sorted_a

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The method combines several random restarts, an Adam optimiser on a
    smoothed max‑convolution, and a tiny greedy local‑mass‑transfer phase.
    The best sequence found (as a plain ``list[float]``) is stored in the
    global variable ``height_sequence_1`` for future runs.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # problem size – keep the same dimension as the previous best
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. generate a collection of plausible initial probability vectors
    # ------------------------------------------------------------------
    start_vectors = []

    # uniform and a few exponential shapes
    start_vectors.append(_uniform(n))
    for rate in (0.001, 0.002, 0.003, 0.005, 0.008, 0.01):
        start_vectors.append(_exp_decay(n, rate))

    # a handful of random Dirichlet draws (the Dirichlet with α=1 is uniform)
    for _ in range(3):
        sample = rng.random(n)
        sample /= sample.sum()
        start_vectors.append(sample)

    # add the previous solution (if any) and a few random permutations of it
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape == (n,):
                start_vectors.append(prev_arr / prev_arr.sum())
                for _ in range(4):
                    perm = rng.permutation(n)
                    start_vectors.append(prev_arr[perm])
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. run the optimiser on every start (time‑aware)
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")

    for idx, vec in enumerate(start_vectors):
        if time.time() > deadline:
            break

        print(f"[Run {idx+1}/{len(start_vectors)}] initial surrogate = "
              f"{_objective_surrogate(vec):.6f}")

        # a) smooth Adam optimisation
        seq_opt, val_opt = _adam_smooth_optimize(
            start_seq=vec,
            n=n,
            deadline=deadline,
            steps_per_beta=5000,
            lr0=0.7
        )
        print(f"    [Adam] best surrogate after smoothing = {val_opt:.6f}")

        # b) cheap greedy local improvement
        seq_loc, val_loc = _local_improve(
            seq=seq_opt,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=150_000
        )
        print(f"    [Local] surrogate after mass‑transfer = {val_loc:.6f}")

        # c) keep the best of this run
        if val_loc < global_best_val:
            global_best_val = val_loc
            global_best_seq = seq_loc.copy()
            print(f"    ** New global best surrogate = {global_best_val:.6f}")

    if global_best_seq is None:
        # fallback – should never happen, but keep the programme alive
        global_best_seq = _uniform(n)
        global_best_val = _objective_surrogate(global_best_seq)

    # ------------------------------------------------------------------
    # 3. final evaluation with the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"[Result] surrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # ------------------------------------------------------------------
    # 4. store for the next run and return
    # ------------------------------------------------------------------
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
