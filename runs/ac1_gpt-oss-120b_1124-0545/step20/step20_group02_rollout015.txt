def project_simplex(v):
    # Input: vector v (any real values).
    # Output: projection onto simplex {x>=0, sum=1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, len(v)+1) > 0)[0]
    if len(rho)==0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step‑function (height sequence) that minimises
``evaluate_sequence``.

The idea is simple but powerful: a linearly decreasing probability vector
(a_i ∝ n‑i) has a much flatter self‑convolution than the uniform vector.
For length n the maximal convolution value is ≈2/(3n) instead of 1/n,
giving an objective ≈4/3 ≈ 1.33 for n≈4000, well below the previous 1.51.

The code builds a small portfolio of analytic candidates (uniform,
linear, quadratic, fractional powers, mixtures) and also a few random
Dirichlet draws.  All candidates are normalised to a sum of 1, their
surrogate objective 2·n·max(conv) is evaluated (via an FFT‑based
convolution) and the best one is kept.  Finally a short exponentiated‑
gradient polishing pass (mirrored on the exact max‑convolution) is run
starting from the best candidate to squeeze out any remaining gain.

The whole search respects the supplied time budget (default 1000 s) and
stores the final sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(L log L))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for a vector whose sum = 1)
# ----------------------------------------------------------------------
def _surrogate_value(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)), assuming Σ seq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_descending(n: int) -> np.ndarray:
    """a_i ∝ (n‑i)  →  strictly decreasing linear shape."""
    a = np.arange(n, 0, -1, dtype=np.float64)          # n, n‑1, …, 1
    return a / a.sum()


def _power_descending(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n‑i)^p  (p≥0)."""
    a = np.arange(n, 0, -1, dtype=np.float64) ** p
    return a / a.sum()


def _mixed_linear_uniform(n: int, lam: float) -> np.ndarray:
    """λ·linear_descending + (1‑λ)·uniform."""
    lin = _linear_descending(n)
    uni = _uniform(n)
    return lam * lin + (1.0 - lam) * uni


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (exact max‑convolution)
# ----------------------------------------------------------------------
def _eg_polish(start_seq: np.ndarray,
               n: int,
               deadline: float,
               max_iter: int = 200_000,
               eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    v = np.log(a + 1e-30)            # log‑softmax parametrisation
    best_seq = a.copy()
    best_val = _surrogate_value(a)

    for it in range(1, max_iter + 1):
        if time.time() > deadline:
            break

        # -------- exact max‑convolution and its active set ----------
        a = _softmax(v)
        conv = _conv_fft(a, a)
        m = conv.max()
        eps = m * 1e-12
        active = np.where(conv >= m - eps)[0]

        # -------- sub‑gradient of the max (averaged over actives) -----
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]           # size n

        # chain rule to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # -------- step (decreasing learning‑rate) ------------------
        eta = eta0 / math.sqrt(it)
        v = v - eta * grad_v

        # -------- keep the best ------------------------------------
        cur_val = _surrogate_value(_softmax(v))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(v).copy()
            # occasional progress report
            if it % 20_000 == 0:
                print(f"[EG] iter {it:,} – surrogate = {best_val:.6f}")

    return best_seq, best_val

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best sequence in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # --------------------------------------------------------------
    # Problem size – we keep the same size as the previous best solution
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Build a (small) collection of candidate sequences
    # --------------------------------------------------------------
    candidates = []

    # Analytic families
    candidates.append(_uniform(n))
    candidates.append(_linear_descending(n))
    candidates.append(_power_descending(n, 2.0))   # quadratic

    for p in (0.5, 1.0, 1.5, 2.0):
        candidates.append(_power_descending(n, p))

    for lam in np.linspace(0.0, 1.0, 6):
        candidates.append(_mixed_linear_uniform(n, lam))

    # Random Dirichlet draws (different concentration levels)
    for conc in (0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            candidates.append(_random_dirichlet(n, conc, rng))

    # --------------------------------------------------------------
    # Evaluate all candidates (surrogate) and keep the best one
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    for idx, seq in enumerate(candidates):
        if time.time() > deadline:
            break
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s == 0.0:
            continue
        seq = seq / s                     # ℓ₁‑normalise

        val = _surrogate_value(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Info] New best surrogate {best_val:.6f} (candidate {idx})")

    # --------------------------------------------------------------
    # Short exponential‑gradient polishing starting from the best point
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.2:
        print("\n--- EG polishing (fast) ---")
        seq_eg, val_eg = _eg_polish(best_seq, n, deadline,
                                   max_iter=200_000, eta0=5.0)
        if val_eg < best_val:
            best_val = val_eg
            best_seq = seq_eg
            print(f"[EG] Polishing improved surrogate to {best_val:.6f}")

    # --------------------------------------------------------------
    # Final true evaluation
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Error] evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
