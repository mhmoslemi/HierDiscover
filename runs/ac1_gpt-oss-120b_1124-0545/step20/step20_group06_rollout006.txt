def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    n = 4000 # possibly change??

    # load previous if exists:
    prev = globals().get('height_sequence_1')
    prev_arr = None
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.shape != (n,):
            prev_arr = None

    # generate seeds:

    seeds = []

    # Uniform
    seeds.append(_uniform(n))
    # Triangular
    seeds.append(_triangular(n))
    # Beta shape grid (coarser)
    alphas = np.linspace(0.2, 5.0, 12)  # coarse
    betas = np.linspace(0.2, 5.0, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))
    # Exponential left with rates
    rates = np.logspace(-4, -0.1, 25)
    for r in rates:
        seeds.append(_exp_decay(n, r))
    # Mixture left exp + uniform
    for r in rates[::5]:
        exp_seq = _exp_decay(n, r)
        for al in np.linspace(0.05, 0.5, 5):
            seeds.append(_mix_shapes(_uniform(n), exp_seq, al))
    # Mixture left & right exponentials
    for r in rates[::5]:
        left = _exp_decay(n, r)
        right = left[::-1]
        for al in np.linspace(0.1, 0.9, 5):
            mixed = (1-al)*left + al*right
            mixed /= mixed.sum()
            seeds.append(mixed)
    # Power law
    for p in np.linspace(0.0, 4.0, 15):
        seeds.append(_power(n, p))
    # Random Dirichlet
    for conc in (0.005, 0.02, 0.05, 0.1, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # include previous best if any
    if prev_arr is not None and prev_arr.sum()>0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(4):
            pert = prev_arr + 0.01 * rng.normal(size=n)
            pert = np.maximum(pert, 0)
            s = pert.sum()
            if s>0:
                seeds.append(pert / s)

    # filter seeds to ensure sum >0
    # Possibly duplicates.

    # Evaluate seeds quickly:
    evaluated = []
    for vec in seeds:
        if time.time()>deadline:
            break
        vec = np.maximum(vec, 0)
        s = vec.sum()
        if s == 0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec,_ in evaluated[:top_k]]
    print(f"Top seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"{i+1}: {v:.6f}")

    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"Initial global best surrogate: {global_best_val:.6f}")

    # For each top seed, run Adam optimizer
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # allocate sub_deadline
        remaining_time = deadline - time.time()
        sub_deadline = time.time() + min(remaining_time*0.12, 30) # at most 30 sec per seed perhaps
        # run Adam
        seq_adam, val_adam = _adam_opt(seed_vec, n, sub_deadline,
                                      max_iters=200_000,
                                      lr=0.5,
                                      single_active=True)
        # also maybe try non single active once
        # Run smoothing of best few seeds using L-BFGS if SciPy available
        if time.time() < sub_deadline and _opt is not None:
            seq_lb = _lbfgs_smooth_optimize(seq_adam, n, sub_deadline, max_iter_per_beta=600)
            val_lb = _objective_surrogate(seq_lb)
        else:
            seq_lb, val_lb = seq_adam, val_adam

        # pick best among seed, adam, lbfgs
        candidate = min([(seed_vec,_objective_surrogate(seed_vec)),
                         (seq_adam,val_adam),
                         (seq_lb,val_lb)], key=lambda x: x[1])

        cand_seq, cand_val = candidate
        print(f"[Seed {idx+1}/{top_k}] local best surrogate: {cand_val:.6f}")

        if cand_val < global_best_val:
            global_best_seq = cand_seq.copy()
            global_best_val = cand_val
            print(f"   *** New global best: {global_best_val:.6f}")

    # final polishing with longer Adam
    if time.time() < deadline:
        print("\n--- Final polishing with long Adam ---")
        final_seq, final_val = _adam_opt(global_best_seq, n, deadline,
                                         max_iters=1_200_000,
                                         lr=0.6,
                                         single_active=True)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved to {global_best_val:.6f}")

    # final SA (optional)
    if time.time() < deadline:
        print("\n--- Final SA polishing ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                          n,
                                                          deadline,
                                                          rng,
                                                          max_attempts=150_000,
                                                          start_sigma=0.001,
                                                          T_start=0.01,

            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # evaluate true objective
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
    except Exception as e:
        true_val = None
        print("Evaluate sequence exception:", e)

    print(f"\n=== Result ===\nSurrogate = {global_best_val:.6f}, true = {true_val}")

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()

"""
Algorithm: multi‑stage optimizer for the step‑function bound

We search a probability vector a (non‑negative, sum = 1) that minimises

    2·n·max(conv(a,a))                     (the exact objective when sum=1)

The pipeline is:

1. **Deterministic & random seeding** – a rich set of analytic families
   (uniform, exponential, beta, power‑law, mixtures, Dirichlet draws) is
   generated and normalised.  Their surrogate values are computed with a fast
   FFT‑based convolution and the best few seeds are kept.

2. **Local refinement** – each top seed is optimised by a mirror‑descent
   optimiser in log‑space (Adam).  The sub‑gradient is taken w.r.t. the
   *single* most active convolution index (the shift that currently attains
   the maximum).  The Adam step‑size schedule provides a more aggressive
   descent than the plain 1/√t schedule used before.

3. **Optional LBFGS smoothing** – on the result of Adam we run a short
   continuation LBFGS on a smooth log‑sum‑exp surrogate (if SciPy is
   available).

4. **Final polishing** – a long Adam run on the current global best and a
   brief simulated‑annealing pass are performed until the global time budget
   expires.

All steps respect the 1000 s overall deadline.  The best sequence found is
stored in the global variable ``height_sequence_1`` and returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for the LBFGS refinement)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Adam (mirror‑descent) optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 300_000,
              lr: float = 0.5,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              single_active: bool = True):
    """
    Mirror‑descent with Adam momentum on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space to keep positivity and the ℓ₁‑norm via soft‑max
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # ----- active index -------------------------------------------------
        if single_active:
            active_idx = int(conv.argmax())
            active = np.array([active_idx], dtype=np.int64)
        else:
            eps_act = max_val * 1e-12
            active = np.where(conv >= max_val - eps_act)[0]

        # ----- sub‑gradient for the active shift(s) ------------------------
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # ----- Adam update --------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing feasibility.
    """
    if _opt is None:
        return start_vec  # SciPy not available

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()                     # stability shift
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp                 # weights for each convolution index
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice back to length n

            # chain rule for log‑softmax (mirror‑descent)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.02,
                         T_end: float = 1e-7):
    """
    Simple SA working directly on the exact surrogate.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

    # --------------------------------------------------------------
    # 1. Deterministic + random seeding
    # --------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # Basic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Beta‑skewed families (coarse grid)
    alphas = np.linspace(0.2, 5.0, 12)
    betas = np.linspace(0.2, 5.0, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Exponential decays (left‑hand)
    rates = np.logspace(-4, -0.1, 25)   # 0.0001 .. 0.79
    for r in rates:

    # Mixture: uniform + left exponential (asymmetric bias)
    for r in rates[::5]:
        exp_seq = _exp_decay(n, r)
        for w in np.linspace(0.05, 0.5, 5):
            seeds.append(_mix_shapes(_uniform(n), exp_seq, w))

    # Mixture: left exponential + right exponential (symmetric but with
    # different rates – useful for fine‑tuning the overlap)
    for r in rates[::5]:
        left = _exp_decay(n, r)
        right = left[::-1]                     # mirror image
        for w in np.linspace(0.1, 0.9, 5):

            mixed /= mixed.sum()
            seeds.append(mixed)

    # Power‑law families

        seeds.append(_power(n, p))

    for conc in (0.001, 0.005, 0.02, 0.05, 0.1, 0.5, 1.0):
