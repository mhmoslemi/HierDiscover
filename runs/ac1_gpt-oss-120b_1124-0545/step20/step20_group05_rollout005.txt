def adam_optimize(v0, n, beta_schedule, num_iters_per_beta, lr, eps, deadline):
    v = v0.copy()
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9; beta2 = 0.999
    t = 0
    for beta in beta_schedule:
        # compute iterations until time cut.
        for _ in range(num_iters_per_beta):
            if time.time()>deadline:
                break
            t += 1
            a = softmax(v)
            conv = _conv_fft(a,a)
            shift = np.max(conv)
            expbeta = np.exp(beta*(conv - shift))
            sumexp = expbeta.sum()
            w = expbeta / sumexp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            grad_v = a * (grad_a - np.dot(grad_a, a))
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1**t)
            v_hat_corr = v_hat / (1 - beta2**t)
            v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)
    return softmax(v)
```

"""
Propose a step‑function (non‑negative probability vector) that minimises
``evaluate_sequence``.

Algorithm
---------

*   Create a modest pool of diverse seed vectors (uniform, Dirichlet,
    low‑frequency cosine curves, beta‑shaped windows, a simple linear ramp,
    and the previous best if available).

*   For each seed run a *smooth‑max Adam* optimiser:
    the vector is represented as a soft‑max of log‑weights `v` (guarantees
    non‑negativity and ℓ₁‑normalisation).  The objective is
        2·n·smooth_max( a*a , β )
    where `smooth_max(x,β) = (1/β)·log Σ exp(β·x)` approximates the true
    max.  β is increased in a schedule (1 → … → 10⁵) so that the optimiser
    first sees a smooth landscape and then concentrates on the true max.
    Adam’s momentum and adaptive steps help escape local plateaus.

*   Keep the best sequence seen (according to the cheap surrogate
    ``2·n·max(conv)``).  After the Adam phase a very cheap stochastic local
    search (random mass‐transfer moves) is performed for a few seconds to
    polish any remaining improvement.

*   Finally evaluate the best candidate with the official ``evaluate_sequence``
    and store it globally for possible reuse.

All heavy work is guarded by a hard deadline (``budget_s`` seconds) so that
the function always returns before the system terminates it.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Basic utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max: returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate for ℓ₁‑normalised vectors: 2·n·max(conv(a,a))."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Seed families
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decay(n: int) -> np.ndarray:
    """Simple decreasing ramp: a_i ∝ (n-i)."""
    w = np.arange(n, 0, -1, dtype=np.float64)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   total_iters: int = 200_000,
                   lr: float = 0.12,
                   beta_schedule=None) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of 2·n·smooth_max(conv(a,a),β) with a = softmax(v).

    Returns the best vector found (ℓ₁‑normalised) and its surrogate value.
    """
    if beta_schedule is None:
        # increasingly sharp approximations of the true max
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0,
                         500.0, 1_000.0, 2_000.0,
                         5_000.0, 10_000.0, 20_000.0,
                         50_000.0, 100_000.0]

    # log‑softmax variables
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)          # first moment
    v_hat = np.zeros_like(v)      # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        beta_val = float(beta)

        # run a chunk of iterations (bounded by time & total count)
        while time.time() < deadline and t_iter < total_iters:
            t_iter += 1

            a = _softmax(v)                     # current probability vector
            conv = _conv_fft(a, a)               # full auto‑convolution
            shift = np.max(conv)                 # numeric stability trick
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                # soft‑assignment to the max term

            # gradient w.r.t. a (exact for the smooth max)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through the soft‑max
            grad_v = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat_corr = v_hat / (1.0 - beta2 ** t_iter)
            step = lr * m_hat / (np.sqrt(v_hat_corr) + eps)
            v -= step

            # track best surrogate value seen
            cur_seq = _softmax(v)
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

    return best_seq, best_val


# ------------------------------------------------------------
# Very cheap stochastic local search (mass‑transfer hill‑climbing)
# ------------------------------------------------------------
def _local_search(seq: np.ndarray,
                  deadline: float,
                  moves: int = 25_000,
                  step_scale: float = 0.012) -> tuple[np.ndarray, float]:
    """Randomly move a tiny amount of probability between two bins."""
    rng = np.random.default_rng()
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    n = len(seq)

    for _ in range(moves):
        if time.time() > deadline:
            break
        eps = step_scale * rng.random()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = min(eps, best_seq[i])
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq /= new_seq.sum()           # renormalise for safety
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq
    return best_seq, best_val
