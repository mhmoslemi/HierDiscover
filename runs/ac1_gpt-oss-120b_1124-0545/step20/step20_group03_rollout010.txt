"""
Propose a step‑function with a low max‑convolution value.

The algorithm:
  1. Build a few diverse starting distributions (exponential decay,
     an “edge‑heavy’’ shape, a random Dirichlet, and the previous best
     if it exists).
  2. For each start run a mirror‑descent (exponentiated‑gradient) on the
     exact non‑smooth objective  f(a)=2·n·max(conv(a,a)).
  3. Polish the result with a smooth log‑sum‑exp surrogate using
     L‑BFGS‑B (if SciPy is available).
  4. Perform a short local mass‑transfer refinement.
  5. Keep the best sequence found before the 1000 s budget expires.
All sequences are kept in ℓ₁‑norm = 1 (scaling does not change the
objective).  The routine stores the best sequence in the global variable
``height_sequence_1`` for the next run.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used for the final L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:       # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level FFT utilities – full linear convolution via a real FFT.
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, N: int, L: int) -> np.ndarray:
    """
    Full linear convolution a*a using a pre‑computed FFT length ``N``
    (the next power of two ≥ L = 2·len(a)‑1).
    Returns the first ``L`` entries.
    """
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact max‑convolution) – used everywhere.
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray, N: int, L: int) -> float:
    """2·n·max(conv(a,a)), where n = len(a).  a must sum to 1."""
    n = a.shape[0]
    conv = _conv_fft(a, N, L)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax utilities (used by exponentiated‑gradient).
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic starter families.
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), i=0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _edge_up(n: int) -> np.ndarray:
    """Higher weight near the two ends, lower in the centre."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (i + 1.0) + 1.0 / (n - i)
    return a / a.sum()


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     N: int,
                     L: int,
                     deadline: float,
                     max_iters: int = 800_000,
                     eta0: float = 8.0):
    """
    Mirror‑descent on f(a)=2·n·max(conv(a,a)).
    Returns the best sequence encountered and its surrogate value.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a, N, L)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_fft(a, N, L)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (scaled by 2 – step size accounts)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a, N, L)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp).
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 1000):
    """
    Optimises the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space to stay feasible.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    L_full = 2 * n - 1
    N_fft = _next_pow2(L_full)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, N_fft, L_full)
            max_c = np.max(conv)
            shift = max_c                     # stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient in the original space (see derivation in the prompt)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, N_fft, L_full + n - 1)   # length >= 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simple local refinement – transfer a tiny mass from one index to another.
# ----------------------------------------------------------------------
def _local_refine(seq: np.ndarray,
                  N: int,
                  L: int,
                  deadline: float,
                  rng: np.random.Generator,
                  steps: int = 5000,
                  eps: float = 1e-4):
    """
    Very cheap hill‑climbing that moves a small amount of probability mass.
    Returns the best sequence seen and its surrogate value.
    """
    a = seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, N, L)
    n = a.shape[0]

    it = 0
    while time.time() < deadline and it < steps:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = min(a[i], eps)
        if delta <= 0.0:
            continue
        ns = a.copy()
        ns[i] -= delta
        ns[j] += delta
        val = _objective_surrogate(ns, N, L)
        if val < best_val:
            best_val = val
            best_seq = ns.copy()
            a = ns                     # accept improvement
        else:
            # occasional uphill move to escape plateaus
            if rng.random() < 0.001:
                a = ns
        it += 1

    return best_seq, best_val

    # --------------------------------------------------------------

    # d) random Dirichlet‑like start
