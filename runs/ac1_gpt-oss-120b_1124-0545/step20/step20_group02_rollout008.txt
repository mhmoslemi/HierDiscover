Propose a non‑negative step‑function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm works in three phases:

1. **LP‑flattening (coarse optimisation)**
   – Work with a modest length `n_small` (default 1024) so that a dense
     linear program can be solved quickly.
   – Starting from the uniform distribution, repeatedly:
        * compute the current maximum of the self‑convolution (M);
        * solve a linear programme that maximises the ℓ₁‑norm of a
          candidate `b` under the *linear* constraints
            (f * b)(k) ≤ M   for all k,
          i.e. the convolution of the current distribution `f` with `b`
          never exceeds the present peak `M`.

          normalised version of `b` into `f`.  This strictly reduces the
          admissible peak, because after scaling the new distribution
          inherits the smaller bound `M / S`.
        * keep the best `f` (smallest observed `M`).

   described in the statement (max‑convolution LP).  Because it operates
   on a small problem it can be performed several times within the

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (ℓ₁=1, v≥0). """
    # based on algorithm by Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a probability vector (ℓ₁‑normalised)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(seq: np.ndarray,
                      deadline: float,
                      max_iters: int = 500_000,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its (exact) objective.
    """
    a = seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)               # log‑space ensures positivity

    # pre‑allocate helpers to avoid inner allocations
    active_tol = 1e-12
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        eps = max_val * 1e-9 + active_tol
        active = np.where(conv >= max_val - eps)[0]

        # compute sub‑gradient of the (exact) max
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # exponentiated‑gradient update (mirror descent on the simplex)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming flattening step (works on a small problem size)
# ----------------------------------------------------------------------
def _lp_flatten(f: np.ndarray, M: float) -> tuple[np.ndarray, bool]:
    """
    Solve the LP:
        maximise   Σ b_i
        subject to (f * b)_k ≤ M  for all k,
                   b_i ≥ 0.

    Returns the optimal b and a flag indicating whether the LP succeeded.
    """
    n = len(f)
    rows = 2 * n - 1

    # Build dense constraint matrix A where A[k, j] = f[k-j] if 0≤k-j<n else 0
    A = np.zeros((rows, n), dtype=np.float64)
    j_idx = np.arange(n, dtype=np.int64)
    for i in range(n):
        A[i + j_idx, j_idx] = f[i]

    # Linear programme (SciPy's HiGHS solver)
    c = -np.ones(n)                     # minimise -Σb_i  <=> maximise Σb_i
    b_ub = M * np.ones(rows)
    bounds = [(0.0, None)] * n

    res = _opt.linprog(c,
                       A_ub=A,
                       b_ub=b_ub,
                       bounds=bounds,
                       method='highs',
                       options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return np.zeros_like(f), False
    b = res.x
    return b, True

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n_target: int = 4000,
                     n_small: int = 1024,
                     lp_iters: int = 6,
                     eg_iters: int = 1_200_000,
                     mix_factor: float = 0.5) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best sequence in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Load previous solution if present (use it as warm start)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.ndim == 1 and np.all(prev_arr >= 0):
                s = prev_arr.sum()
                if s > 0:
                    prev_arr = prev_arr / s
                else:
                    prev_arr = None
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # --------------------------------------------------------------
    # Initial distribution at the small scale
    # --------------------------------------------------------------
    if prev_arr is not None and len(prev_arr) == n_small:
        f = prev_arr.copy()
        print("[Info] Loaded previous fine‑grained sequence.")
    else:
        # start from a smooth sine window – a decent generic shape
        i = np.arange(n_small, dtype=np.float64)
        f = np.sin(np.pi * (i + 1) / (n_small + 1)) ** 1.2
        f = f / f.sum()
        print("[Info] Starting from sine window (small scale).")

    # --------------------------------------------------------------
    # Phase 1 – LP flattening (coarse optimisation)
    # --------------------------------------------------------------
    if _opt is not None:
        print("\n--- Phase 1: LP flattening ---")
        for it in range(lp_iters):
            if time.time() > deadline:
                break
            conv = _conv_fft(f, f)
            M = conv.max()
            print(f" LP step {it+1}/{lp_iters}: current peak = {M:.6e}")

            b, ok = _lp_flatten(f, M)
            if not ok:
                print("  LP solver failed – aborting LP phase.")
                break

            S = b.sum()
            if S <= 1.0 + 1e-9:
                print("  No mass increase (Σb ≤ 1) – LP phase converged.")
                break    # cannot improve further

            # Normalise b and mix
            b_norm = b / S
            f = (1.0 - mix_factor) * f + mix_factor * b_norm
            # re‑normalise to avoid numerical drift
            f = f / f.sum()
            print(f"  Σb = {S:.6f} → mixed, new peak = {_conv_fft(f,f).max():.6e}")

    else:
        print("\n[Warning] SciPy not available – skipping LP phase.")

    # --------------------------------------------------------------
    # Upsample to target length
    # --------------------------------------------------------------
    if n_target != n_small:
        x_small = np.linspace(0.0, 1.0, n_small)
        x_target = np.linspace(0.0, 1.0, n_target)
        f = np.interp(x_target, x_small, f)
        f = f / f.sum()
        print(f"\n[Info] Upsampled to n={n_target}")

    # --------------------------------------------------------------
    # Phase 2 – high‑precision exponentiated‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 2: Exponentiated‑gradient polishing ---")
        seq_eg, val_eg = _exp_gradient_opt(f,
                                           deadline,
                                           max_iters=eg_iters,
                                           eta0=4.5)
        if val_eg < _objective_surrogate(f):
            f = seq_eg
            print(f"[EG] Improved objective to {val_eg:.6f}")
        else:
            print("[EG] No improvement over LP‑derived start.")
    else:
        print("[Info] Time budget exhausted before EG polishing.")

    # --------------------------------------------------------------
    # Final evaluation with the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(f.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {_objective_surrogate(f):.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:       # pragma: no cover
        print("[Error] evaluate_sequence raised:", exc)

    # Store for possible future warm‑starts
    globals()["height_sequence_1"] = f.tolist()
    return f.tolist()
