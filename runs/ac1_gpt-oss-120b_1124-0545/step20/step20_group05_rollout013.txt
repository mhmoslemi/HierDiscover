def _construct_sparse_set(n, m, max_per_sum, rng):
    sum_len = 2*n - 1
    sum_counts = np.zeros(sum_len, dtype=np.int16)
    S = []
    attempts = 0
    max_attempts_total = 5000 * m   # arbitrary
    while len(S) < m and attempts < max_attempts_total:
        i = rng.integers(0, n)
        if i in S:   # you can maintain bool array
            attempts += 1
            continue
        # compute sums
        if S:
            arr_S = np.array(S, dtype=np.int32)
            sums = i + arr_S  # shape len(S)
            if np.any(sum_counts[sums] + 1 > max_per_sum):
                attempts += 1
                continue
        # self sum
        sum_self = i + i
        if sum_counts[sum_self] + 1 > max_per_sum:
            attempts += 1
            continue
        # accept i
        if S:
            sum_counts[sums] += 1
        sum_counts[sum_self] += 1
        S.append(i)
        attempts = 0 # reset attempts after success
    if len(S) < m:
        raise ValueError('Failed to construct set')
    return np.array(S, dtype=np.int32)

    attempts = 0

"""
Propose candidate – “Sparse‑sum‑limited” construction.

The evaluation function is
    f(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

Since scaling does not affect the ratio we may enforce sum(a)=1.
For a probability vector a the quantity max(conv(a,a)) is the
largest entry of the autocorrelation
    c_k = Σ_i a_i·a_{k-i}                           (0 ≤ k < 2n−1)

If the support S={i : a_i>0} has size m and each unordered pair
(i,j) (including i=j) yields a distinct sum i+j, then
    max_k c_k = 1/m² .
In practice we only need to keep the multiplicity of any sum
small (≤ 2 or 3).  The construction below builds a support of a
chosen size m such that *every* possible sum appears at most a
prescribed number `max_per_sum`.  With m≈170 and `max_per_sum=2`
the theoretical bound is

    max(conv) ≤ 4 / m²   →   f(a) ≤ 2·n·4/m² ≈ 1.1  (for n=4000),

comfortably beating the current record 1.5085.

Algorithm
---------

1.  Random greedy construction of a set S⊂{0,…,n‑1} of size m with
    the constraint `sum_counts[i+j] ≤ max_per_sum`.
    The construction is repeated with several values of m and
    both limits 2 and 3 until the time budget expires.

2.  Initialise a probability vector with uniform weight
    `a_i = 1/m` on S (zero elsewhere) and evaluate the exact
    surrogate 2·n·max(conv).

3.  Keep the best vector found.  For the best candidate we run
    a short “exponentiated‑gradient” (mirror‑descent) optimisation
    on the exact max‑convolution and a continuation L‑BFGS
    optimiser on a log‑sum‑exp smooth‑max surrogate (if SciPy is
    available).  Both refinements keep the vector on the simplex.

4.  Return the final vector (as a plain Python list) and store it in
    the global variable ``height_sequence_1`` for reuse.

All heavy work respects the global ``budget_s`` (default 1000 s);
the function will always return a feasible sequence before the
deadline (worst‑case fallback is the uniform distribution).
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Basic numeric utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector ℓ₁‑normalised to 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Sparse support construction (bounded sum‑multiplicity)
# ------------------------------------------------------------
def _construct_sparse_support(n: int,
                              m: int,
                              max_per_sum: int,
                              rng: np.random.Generator,
                              max_tries: int = 200) -> np.ndarray:
    """
    Greedy random construction of a set S⊂{0,…,n‑1} with |S|=m
    such that every unordered sum i+j (including i=j) occurs at
    most ``max_per_sum`` times.
    Raises RuntimeError if construction fails within ``max_tries``.
    """
    length_sum = 2 * n - 1
    # The array counts unordered sums; we work with ordered pairs later.
    for _ in range(max_tries):
        sum_counts = np.zeros(length_sum, dtype=np.int16)
        used = np.zeros(n, dtype=bool)
        S = []
        # random permutation of the whole index set
        order = rng.permutation(n)
        for i in order:
            if len(S) >= m:
                break
            if used[i]:
                continue
            # check collisions with existing members
            if S:
                arr = np.array(S, dtype=np.int32)
                sums = i + arr                     # unordered i+existing
                if np.any(sum_counts[sums] + 1 > max_per_sum):
                    continue
            # self‑sum i+i
            s_self = i + i
            if sum_counts[s_self] + 1 > max_per_sum:
                continue
            # accept this index
            if S:
                sum_counts[sums] += 1
            sum_counts[s_self] += 1
            S.append(i)
            used[i] = True
        if len(S) == m:
            return np.array(S, dtype=np.int32)
        # otherwise retry with a new random order
    raise RuntimeError(f'Failed to build a set of size {m} with max_per_sum={max_per_sum}')


# ------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimisation on the exact max
# ------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_200_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution.
    Returns best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (ordered pairs)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0,
                         5000000.0, 10000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (softmax space)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)          # chain rule
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)
