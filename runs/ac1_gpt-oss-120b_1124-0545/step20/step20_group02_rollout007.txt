"""
Propose a non‑negative step‑function (height sequence) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

Algorithm (≈ 1000 s budget)

1. Load the previous best sequence (if any) and keep it as a candidate.
2. Randomly generate many analytic / stochastic families (beta‑shape,
   sine, Gaussian, exponential, Chebyshev, Kaiser, Dirichlet, etc.).
   All candidates are ℓ₁‑normalised (sum = 1).  Keep the best one according
   to the *exact* max‑convolution objective (fast FFT surrogate).
3. Starting from the best candidate run a short L‑BFGS refinement on the
   smooth log‑sum‑exp surrogate of the max‑convolution (a schedule of
   increasing β).  This gives a high‑quality “smooth” local optimum.
4. Polish the result with many iterations of exponentiated‑gradient
   (mirror‑descent) on the *exact* max‑convolution (sub‑gradient from the
   currently maximal index).  The loop is time‑controlled, so it uses the
   remaining budget.
5. A final lightweight simulated‑annealing pass (Gaussian perturbations
   with a linear cooling schedule) is applied to escape any remaining
   local minima.
6. The final sequence is stored in the global variable ``height_sequence_1``
   and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional imports
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

try:

except Exception:  # pragma: no cover

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective for ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective (2·n·max(conv(seq,seq))) for a sequence with sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cheb_window(n: int, att: float) -> np.ndarray:
    """Chebyshev window (attenuation in dB)."""
    if _signal is None:
        return _uniform(n)
    w = _signal.chebwin(n, at=att)
    w = np.maximum(w, 0.0)
    s = w.sum()
    return w / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window (beta parameter)."""
    if _signal is None:
        return _uniform(n)
    w = _signal.kaiser(n, beta)
    w = np.maximum(w, 0.0)
    s = w.sum()
    return w / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max‑conv
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 5_000_000,
                            eta0: float = 4.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space; this automatically enforces positivity and sum‑1.
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # pick *one* maximal index (deterministic tie‑break)
        k = int(np.argmax(conv))

        # sub‑gradient for that index (the 2·a_{k‑i} rule)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g = np.zeros_like(a)
        # add contributions a[rev_idx]; interior points get factor 2 automatically
        g[idx_i] = a[rev_idx]
        # compensate for the missing symmetric term at the borders
        # (when k-i out of range, the term appears only once)
        # The factor 2 works for all interior positions; for border it is still correct.
        g = 2.0 * g

        # chain rule to log‑space (log‑softmax)
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_v

        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth max surrogate (for L‑BFGS)
# ----------------------------------------------------------------------
class _SmoothMax:
    """
    Callable object for the smooth max‑convolution surrogate.
    The objective is
        2·n·(shift + log(sum(exp(beta·(conv‑shift))))/beta)
    where shift = max(conv) for numerical stability.
    Gradient is provided w.r.t. the log‑softmax (v) variables.
    """
    __slots__ = ("n", "beta")

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, v: np.ndarray) -> tuple[float, np.ndarray]:
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        shift = max_c
        exp_beta = np.exp(self.beta * (conv - shift))
        sum_exp = exp_beta.sum()
        obj = 2.0 * self.n * (shift + np.log(sum_exp) / self.beta)

        # gradient wrt a
        w = exp_beta / sum_exp
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3·n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # chain rule to v‑space (log‑softmax)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return obj, grad_v


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 180_000,
                         start_sigma: float = 0.001,
                         T_start: float = 5e-3,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Lightweight SA: Gaussian perturbations with a linearly decreasing temperature.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best sequence in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Problem size (can be tweaked – larger n gives finer resolution)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous solution if any
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0.0:
                    prev_arr = prev_arr / s
                else:
                    prev_arr = None
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # --------------------------------------------------------------
    # Initial candidate set
    # --------------------------------------------------------------
    if prev_arr is not None:
        best_seq = prev_arr.copy()
        best_val = _objective_surrogate(best_seq)
        print("[Info] Loaded previous best sequence (objective ≈ {:.6f})".format(best_val))
    else:
        best_seq = _sine_window(n, gamma=1.2)
        best_val = _objective_surrogate(best_seq)
        print("[Info] No previous sequence – starting from sine window (objective ≈ {:.6f})".format(best_val))

    # --------------------------------------------------------------
    # Random search over analytic families
    # --------------------------------------------------------------
    print("\n--- Random family search (fast) ---")
    families = [
        "uniform", "sine", "gauss", "exp", "symexp",
        "beta", "cheb", "kaiser", "rand_dir", "rand_exp"
    ]
    # allocate roughly 20 % of the total budget to this phase
    rs_deadline = start_time + budget_s * 0.20
    it = 0
    while time.time() < rs_deadline and it < 30_000:
        fam = rng.choice(families)
        if fam == "uniform":
            cand = _uniform(n)
        elif fam == "sine":
            gamma = rng.uniform(0.2, 3.2)
            cand = _sine_window(n, gamma=gamma)
        elif fam == "gauss":
            sigma_frac = rng.uniform(0.02, 0.45)
            cand = _gaussian(n, sigma_frac=sigma_frac)
        elif fam == "exp":
            rate = rng.uniform(0.0005, 0.02)
            cand = _exp_decay(n, rate)
        elif fam == "symexp":
            rate = rng.uniform(0.0005, 0.02)
            cand = _sym_exp_decay(n, rate)
        elif fam == "beta":
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            cand = _beta_shape(n, alpha=alpha, beta=beta)
        elif fam == "cheb":
            att = rng.uniform(20.0, 200.0)  # dB attenuation
            cand = _cheb_window(n, att=att)
        elif fam == "kaiser":
            beta = rng.uniform(0.5, 15.0)
            cand = _kaiser_window(n, beta=beta)
        elif fam == "rand_dir":
            conc = rng.uniform(0.001, 10.0)
            cand = _random_dirichlet(n, conc=conc, rng=rng)
        elif fam == "rand_exp":
            cand = _random_exponential(n, rng=rng)
        else:
            cand = _uniform(n)

        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print("[Random] New best {:.6f} (family={}, it={})".format(best_val, fam, it))
        it += 1

    print("[Info] Best after random search {:.6f}".format(best_val))

    # --------------------------------------------------------------
    # Smooth‑max L‑BFGS refinement (if SciPy is available)
    # --------------------------------------------------------------
    if _opt is not None and time.time() < deadline:
        print("\n--- L‑BFGS refinement (smooth max) ---")
        v = np.log(best_seq + 1e-30)
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0
        ]
        best_seq_lb = best_seq.copy()
        best_val_lb = best_val

        for beta in beta_schedule:
            if time.time() > deadline:
                break
            print(f"  * beta = {beta:.0e}")
            smooth_obj = _SmoothMax(n, beta)

            # callback to abort when out of time
            def _cb(_):
                if time.time() > deadline:
                    raise KeyboardInterrupt()

            try:
                res = _opt.minimize(
                    fun=smooth_obj,
                    x0=v,
                    method="L-BFGS-B",
                    jac=True,
                    callback=_cb,
                    options={"maxiter": 2500, "disp": False, "ftol": 1e-15}
                )
                v = res.x
            except KeyboardInterrupt:
                print("[L‑BFGS] Time limit reached – exiting early.")
                break
            except Exception as e:  # pragma: no cover
                print(f"[L‑BFGS] Exception at beta={beta}: {e}")
                break

            cur_seq = _softmax(v)
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val_lb:
                best_val_lb = cur_val
                best_seq_lb = cur_seq.copy()
                print(f"    > Improved surrogate to {best_val_lb:.6f}")

        best_seq = best_seq_lb
        best_val = best_val_lb
        print(f"\n[After L‑BFGS] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # Exponentiated‑gradient polishing (high‑precision)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (high‑precision) ---")
        seq_eg, val_eg = _exp_gradient_opt_multi(
            best_seq,
            n,
            deadline,
            max_iters=3_000_000,
            eta0=5.0
        )
        if val_eg < best_val:
            best_seq = seq_eg
            best_val = val_eg
            print(f"[EG] Improved surrogate to {best_val:.6f}")
        else:
            print(f"[EG] No improvement (best still {best_val:.6f})")

    # --------------------------------------------------------------
    # Final simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final Simulated Annealing fine‑tuning ---")
        seq_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            deadline,
            rng,
            max_attempts=200_000,
            start_sigma=0.001,
            T_start=5e-3,
            T_end=1e-9
        )
        if val_sa < best_val:
            best_seq = seq_sa
            best_val = val_sa
            print(f"[SA] Improved surrogate to {best_val:.6f}")
        else:
            print(f"[SA] No improvement (best still {best_val:.6f})")

    # --------------------------------------------------------------
    # Official evaluation via the supplied function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Error] evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
