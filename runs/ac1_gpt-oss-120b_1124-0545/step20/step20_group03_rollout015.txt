    for name, params in families:
        for param in params:
            if time.time() > deadline:
                break
            if name == 'exp':
                seq = _exp_decay(n, rate=param)
            elif name == 'pow':
                seq = _powerlaw(n, p=param)
            elif name == 'desc_pow':

                seq = _beta_shape(n, a_param, b_param)

            val = _objective_surrogate(seq)
            if val < best_val:

                best_seq = seq
    # After scanning families, optionally try random Dirichlet draws for a while.

    if best_seq is not None:

    # optional L-BFGS smoothing
    if time.time() < deadline - 5.0 and _opt is not None:

        if val_lb < best_val:

            best_val = val_lb
    # store globally
    globals()["height_sequence_1"] = best_seq.tolist()
    # compute true evaluate_sequence
    true_val = evaluate_sequence(best_seq.tolist())

def _next_pow2(x):
    return 1<<(x-1).bit_length()
def _conv_fft(a,b):
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa*fb, n=N)[:L]
    return conv
```

def _objective_surrogate(seq):
    seq = np.asarray(seq, dtype=np.float64)
    # ensure sum >0
    sum_seq = seq.sum()
    if sum_seq <= 0:
        return float('inf')
    seq_norm = seq / sum_seq
    n = len(seq_norm)
    conv = _conv_fft(seq_norm, seq_norm)
    return 2.0 * n * np.max(conv)

    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ∑a = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for a vector with ℓ₁‑norm = 1:
        2·n·max(conv(seq, seq)).
    The input is assumed non‑negative.  Normalisation is performed
    automatically, so the function may be called with an arbitrary
    non‑zero vector.
    """
    seq = np.asarray(seq, dtype=np.float64)
    total = seq.sum()
    if total <= 0.0:
        return float("inf")
    a = seq / total
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple families of candidate shapes (all return a normalised vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _powerlaw(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = i ** (-p)
    return a / a.sum()


def _desc_power(n: int, s: float) -> np.ndarray:
    """a_i ∝ (n‑i)^(s)   (decreasing polynomial)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i) ** s
    return a / a.sum()


def _mix_exp_power(n: int, rate: float, p: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)·(i+1)^(‑p)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) * ((i + 1.0) ** (-p))
    return a / a.sum()


def _gaussian_decay(n: int, beta: float) -> np.ndarray:
    """a_i ∝ exp(-beta·i²)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-beta * (i ** 2))
    return a / a.sum()


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution density on the grid i=0..n‑1 (α=a, β=b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n          # centre of each bin
    vals = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    if vals.sum() == 0.0:
        return _uniform(n)
    return vals / vals.sum()


# ----------------------------------------------------------------------
# Softmax utilities (used by exponentiated‑gradient)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient mirror‑descent on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution (sub‑gradient at the
    active shift).  Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)                     # work in log‑space

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # gradient of the active entry (see the original code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space to stay feasible.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Strategy:
      1. Scan a handful of low‑dimensional parametric families (exponential,
         power‑law, mixed, Gaussian, beta) and keep the best shape.
      2. Perform a short random Dirichlet exploration to diversify the search.
      3. Refine the current best vector with exponentiated‑gradient mirror‑descent.
      4. If time permits, run a continuation L‑BFGS on a log‑sum‑exp surrogate.
      5. Store and return the best sequence (also saved in the global variable
         ``height_sequence_1``).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size – 4000 proved a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 0. Start from a previous run if it exists (might already be very good).
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            cand = np.asarray(prev, dtype=np.float64)
            if cand.shape == (n,):
                val = _objective_surrogate(cand)
                best_seq, best_val = cand, val
                print(f"[Init] loaded previous best surrogate = {val:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1. Coarse grid search over a few analytic families.
    # --------------------------------------------------------------
    print("[Search] scanning exponential family")
    for rate in np.logspace(np.log10(3e-4), np.log10(1e-2), num=80):
        if time.time() > deadline:
            break
        seq = _exp_decay(n, rate)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    print("[Search] scanning power‑law family")
    for p in np.linspace(0.2, 2.0, 80):
        if time.time() > deadline:
            break
        seq = _powerlaw(n, p)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    print("[Search] scanning descending‑polynomial family")
    for s in np.linspace(0.5, 5.0, 60):
        if time.time() > deadline:
            break
        seq = _desc_power(n, s)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    print("[Search] scanning mixed exponential‑power family")
    for rate in np.logspace(np.log10(3e-4), np.log10(5e-3), num=30):
        for p in np.linspace(0.2, 1.0, 30):
            if time.time() > deadline:
                break
            seq = _mix_exp_power(n, rate, p)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val, best_seq = val, seq

    print("[Search] scanning Gaussian‑like family")
    for beta in np.logspace(np.log10(1e-7), np.log10(5e-6), num=30):
        if time.time() > deadline:
            break
        seq = _gaussian_decay(n, beta)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    print("[Search] scanning beta‑distribution family")
    alphas = np.linspace(0.2, 2.5, 15)
    betas  = np.linspace(0.2, 2.5, 15)
    for a_param in alphas:
        for b_param in betas:
            if time.time() > deadline:
                break
            seq = _beta_shape(n, a_param, b_param)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val, best_seq = val, seq

    # --------------------------------------------------------------
    # 2. Random Dirichlet draws (adds exploration beyond the analytic families).
    # --------------------------------------------------------------
    print("[Search] random Dirichlet exploration")
    while time.time() < deadline - 30.0:          # keep ~30 s for polishing
        # draw a concentration parameter uniformly in [0.1, 3.0]
        alpha = rng.uniform(0.1, 3.0)
        seq = rng.dirichlet(np.full(n, alpha))
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    # --------------------------------------------------------------
    # 3. Exponentiated‑gradient polishing on the exact max‑convolution.
    # --------------------------------------------------------------
    if best_seq is not None:
        print(f"[Polish] start EG polishing – current surrogate = {best_val:.6f}")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, deadline,
                                           max_iters=2_000_000,
                                           eta0=5.0)
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg
        print(f"[Polish] EG result surrogate = {val_eg:.6f}")

    # --------------------------------------------------------------
    # 4. Optional L‑BFGS smoothing (if time remains)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0 and _opt is not None:
        print("[L‑BFGS] smoothing start")
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline)
        val_lb = _objective_surrogate(seq_lb)
        print(f"[L‑BFGS] after smoothing surrogate = {val_lb:.6f}")
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb

    # --------------------------------------------------------------
    # 5. Final evaluation via the provided function
    # --------------------------------------------------------------
    if best_seq is None:
        raise RuntimeError("No candidate sequence generated.")
    true_val = evaluate_sequence(best_seq.tolist())
    print(f"[Result] best surrogate = {best_val:.6f}, evaluate_sequence = {true_val:.6f}")

    # Store for the next run (facilitates incremental improvement)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
