"""
Propose a low‑upper‑bound step‑function.

The evaluation function
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2)
is invariant to scaling, so we work with L¹‑normalised vectors.
For a fixed length n the problem is to minimise max(conv(seq,seq)).
The algorithm is a lightweight multi‑stage search:

1.  Produce a diverse pool of “seeds’’ (uniform, triangular,
    sine, Gaussian, power‑law, β‑shape and, crucially, many
    exponential‑decay families a_i ∝ exp(-λ·i)).
2.  Keep the best few seeds according to the cheap surrogate
    2·n·max(conv(seq,seq)).
3.  For each seed run a long exponentiated‑gradient (mirror‑descent)
    optimisation on the exact max (sub‑gradient is the active
    convolution entry).  The step‑size η decays as 1/√t.
4.  Refine the result with a continuation L‑BFGS‑B optimiser that
    smooths the max via a log‑sum‑exp surrogate and uses a
    schedule of increasing β (≈ 1,…,5·10⁶).  Variables live in
    log‑softmax space → non‑negative simplex is respected.
5.  Finish with a cheap simulated‑annealing mass‑transfer hill‑climb
    that can escape shallow local minima.
6.  Return the best sequence found (also stored as
    ``height_sequence_1`` for reuse).

All steps honour the global time budget (default 1000 s).  The code
uses only NumPy and SciPy (if available) and contains no I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector ℓ₁‑normalised to 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, lam: float) -> np.ndarray:
    """Exponential decreasing shape a_i ∝ exp(-lam·i)."""
    if lam <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random families
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Mixing / projection helpers
# ----------------------------------------------------------------------
def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex blend α·a + (1‑α)·b, re‑normalised."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution.
    Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space to keep simplex constraint automatically
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (see analysis)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
except Exception:  # pragma: no cover
    _opt = None
    _sp = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0, 2000000.0,
                         5000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                      # stability shift
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (softmax space)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)          # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing mass‑transfer local search
# ----------------------------------------------------------------------
def _simulated_annealing(seq: np.ndarray,
                         deadline: float,
                         moves: int = 20000,
                         scale: float = 0.02):
    """
    Random mass‑transfer annealing.  A small amount of probability is moved
    from one index to another; downhill moves are always accepted, uphill
    moves with probability exp(-Δ/T).  Temperature T linearly cools.
    """
    rng = np.random.default_rng()
    cur_seq = seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    n = len(seq)

    T0 = 0.02  # initial temperature (relative to objective scale)

    for step in range(moves):
        if time.time() > deadline:
            break
        T = T0 * (1.0 - step / moves)   # simple linear schedule

        eps = scale * rng.random()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = min(eps, cur_seq[i])
        if delta <= 0.0:
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq /= new_seq.sum()
        new_val = _objective_surrogate(new_seq)
        diff = new_val - cur_val
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
            cur_seq, cur_val = new_seq, new_val

    return cur_seq, cur_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` within the provided time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.5          # safety margin

    # moderate length – enough to reach the theoretical bound, cheap enough
    n = 2000

    # --------------------------------------------------------------
    # 1. Broad seed generation (deterministic + parametric families)
    # --------------------------------------------------------------
    print(f"\n--- Generating seeds (n={n}) ---")
    seeds = []

    # classic deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.12))
    seeds.append(_power(n, p=0.5))

    # exponential families – dense sweep over λ
    for lam in np.linspace(0.0002, 0.0045, 30):
        seeds.append(_exp_decay(n, lam))

    # power‑law sweep (covers a different tail behaviour)
    for p in np.linspace(0.0, 2.0, 30):
        seeds.append(_power(n, p))

    # beta‑shape sweep – moderate grid
    alphas = np.linspace(0.3, 4.0, 12)
    betas  = np.linspace(0.3, 4.0, 12)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # random Dirichlet seeds (different concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # bring in the previous best if it exists (down‑sample / interpolate)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, arr.size)
            x_new = np.linspace(0.0, 1.0, n)
            arr = np.interp(x_new, x_old, arr)
        arr = np.maximum(arr, 0.0)
        if arr.sum() > 0.0:
            arr /= arr.sum()
            seeds.append(arr)

    # --------------------------------------------------------------
    # 2. Evaluate all seeds with the cheap surrogate
    # --------------------------------------------------------------
    print("\n--- Evaluating seeds (surrogate) ---")
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop surrogate values after seeding:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Refine each top seed
    # --------------------------------------------------------------
    best_overall_seq = top_seeds[0].copy()
    best_overall_val = _objective_surrogate(best_overall_seq)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Refinement for seed #{idx+1} ---")
        # ----------------------------------------------------------
        # (a) Mirror‑descent (exponentiated gradient)
        # ----------------------------------------------------------
        eg_time = time.time() + max(5.0, 0.35 * (deadline - time.time()))
        seq_eg, val_eg = _exp_gradient_opt(seed_seq.copy(), n,
                                           eg_time,
                                           max_iters=2_500_000,
                                           eta0=6.0)
        if val_eg < best_overall_val:
            best_overall_val, best_overall_seq = val_eg, seq_eg.copy()
            print(f"  EG improvement: {val_eg:.6f}")

        # ----------------------------------------------------------
        # (b) Continuation L‑BFGS smoothing
        # ----------------------------------------------------------
        if _opt is not None:
            lbfgs_time = time.time() + max(5.0, 0.30 * (deadline - time.time()))
            seq_lb = _lbfgs_smooth_optimize(best_overall_seq,
                                            n,
                                            lbfgs_time,
                                            beta_schedule=[1, 5, 20, 80,
                                                           200, 500, 1000,
                                                           2000, 5000, 20000,
                                                           50000, 100000,
                                                           200000, 500000,
                                                           1000000, 2000000,
                                                           5000000],
                                            max_iter_per_beta=4000)
            val_lb = _objective_surrogate(seq_lb)
            if val_lb < best_overall_val:
                best_overall_val, best_overall_seq = val_lb, seq_lb.copy()
                print(f"  L‑BFGS improvement: {val_lb:.6f}")

        # ----------------------------------------------------------
        # (c) Simulated‑annealing mass‑transfer hill‑climb
        # ----------------------------------------------------------
        anneal_time = time.time() + max(3.0, 0.15 * (deadline - time.time()))
        seq_sa, val_sa = _simulated_annealing(best_overall_seq,
                                              anneal_time,
                                              moves=25000,
                                              scale=0.02)
        if val_sa < best_overall_val:
            best_overall_val, best_overall_seq = val_sa, seq_sa.copy()
            print(f"  Annealing improvement: {val_sa:.6f}")

    # --------------------------------------------------------------
    # 4. Final official evaluation
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised", exc)

    print(f"\n=== Finished ===\nBest surrogate ≈ {best_overall_val:.6f}"
          f", true evaluate_sequence = {true_val}\nLength = {len(best_overall_seq)}")

    # expose for the next run
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
