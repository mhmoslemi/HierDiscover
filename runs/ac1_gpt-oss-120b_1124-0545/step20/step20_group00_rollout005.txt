"""
Propose a better step‑function sequence.

Algorithm
---------

1. **Diverse seed generation** – create many ℓ₁‑normalised candidates:
   * uniform distribution,
   * previous best (if any),
   * Dirichlet draws with several concentration parameters,
   * Beta‑shaped PDFs (α,β ∈ (0.1,5)),
   * Power‑U shapes a_i ∝ ((i+½)(n‑i‑½))^γ with γ ∈ (‑2,2),
   * Edge‑block shapes (high weight on the two borders).

   All candidates are normalised to sum to 1 and evaluated with the exact
   surrogate `2·n·max(conv(a,a))`.  The best few are kept.

2. **Mirror‑descent polishing** – for each of the top‑k seeds run a long
   exponentiated‑gradient (mirror‑descent) optimisation on the true
   max‑convolution objective.  The update works in log‑softmax space,
   uses a simple 1/√t step‑size and respects the global deadline.
   The best sequence seen during polishing is retained.

3. **Optional L‑BFGS smoothing** – if SciPy is available a short
   continuation of the smoothed log‑sum‑exp objective (as in the
   reference code) is performed on the current global best.

4. **Final evaluation** – the best sequence is passed to the provided
   `evaluate_sequence` function and stored globally as `height_sequence_1`.

The routine is fully self‑contained, uses only NumPy (and SciPy when
available) and respects the 1000‑second budget.  It deliberately explores
a wider family of shapes than the previous approach, which helps to
break the local minimum around 1.5085 and pushes the bound towards the
target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for the final L‑BFGS stage)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a probability vector:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray:
    """Beta(α,β) density sampled on the grid i=0,…,n‑1 and normalised."""
    # shift to avoid exact 0/1 which would give infinities
    eps = 1e-12
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # clamp away from 0/1 for numerical safety
    x = np.clip(x, eps, 1.0 - eps)
    raw = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    total = raw.sum()
    if total <= 0.0 or not np.isfinite(total):
        raw = np.ones_like(raw)
        total = raw.sum()
    return raw / total


def _power_u_shape(gamma: float, n: int) -> np.ndarray:
    """U‑shaped power law: a_i ∝ ((i+½)(n‑i‑½))^γ."""
    i = np.arange(n, dtype=np.float64) + 0.5
    left = i
    right = n + 0.5 - i
    base = left * right
    # ensure non‑negative base (it always is)
    raw = np.exp(gamma * np.log(base + 1e-30))
    total = raw.sum()
    if total <= 0.0 or not np.isfinite(total):
        raw = np.ones_like(raw)
        total = raw.sum()
    return raw / total


def _edge_block(k: int, ratio: float, n: int) -> np.ndarray:
    """Two high‑weight edge blocks of width k, weight = ratio."""
    arr = np.ones(n, dtype=np.float64)
    if k > 0:
        arr[:k] = ratio
        arr[-k:] = ratio
    return arr / arr.sum()

# Mirror‑descent (exponentiated‑gradient) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the true max‑convolution.
    Operates in log‑softmax space (exponentiated gradient).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx
        g[idx] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space.
    """
    if _opt is None:
        # SciPy not available – return the input unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient w.r.t a
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

    best_seq = None
    best_val = np.inf
    top_seqs = []   # list of (val, seq) tuples, sorted ascending

    print("\n--- Evaluating diversified seeds ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        # keep a tiny sliding window of the three best sequences
        if len(top_seqs) < 3:
            top_seqs.append((val, vec.copy()))
            top_seqs.sort(key=lambda x: x[0])
        elif val < top_seqs[-1][0]:
            top_seqs[-1] = (val, vec.copy())
            top_seqs.sort(key=lambda x: x[0])

        if val < best_val:
            best_val = val
            best_seq = vec.copy()
        if (idx + 1) % 250 == 0:
            print(f"  evaluated {idx+1:4d}/{len(seeds)} seeds – current best surrogate = {best_val:.6f}")

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Mirror‑descent polishing of the top candidates
    # ------------------------------------------------------------------
    # Allocate roughly half of the remaining time to polishing
    time_now = time.time()
    time_left = deadline - time_now
    if time_left > 10.0 and best_seq is not None:
        polish_budget = time_left * 0.55
        per_candidate = polish_budget / max(1, len(top_seqs))
        print("\n--- Exponentiated‑gradient polishing (multi‑start) ---")
        for rank, (seed_val, seed_seq) in enumerate(top_seqs, start=1):
            pol_deadline = time.time() + per_candidate
            print(f"  Polishing candidate #{rank} (surrogate {seed_val:.6f})")
            seq_opt, val_opt = _exp_gradient_opt(seed_seq, n, pol_deadline,
                                                 max_iters=10_000_000,
                                                 eta0=5.0)
            print(f"    polished surrogate = {val_opt:.6f}")
            if val_opt < best_val:
                best_val, best_seq = val_opt, seq_opt.copy()
                print(f"    >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Optional L‑BFGS smoothing (exact same as reference)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- L‑BFGS smoothing of current best ---")
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.35
        seq_lbfgs = _lbfgs_smooth_optimize(best_seq, n, sub_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1000.0,
                                                          2000.0, 5000.0,
                                                          20000.0, 50000.0,
                                                          100000.0, 200000.0],
                                           max_iter_per_beta=6000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print(f"    >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final true evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the result for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

The richer initialisation and longer polishing give a strong chance of beating the previous bound of ≈ 1.5085 and moving toward the target 1.503.
