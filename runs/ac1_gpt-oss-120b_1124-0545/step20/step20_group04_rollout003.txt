"""
A Frank–Wolfe style optimizer for the max‑convolution objective.

The algorithm
  1. Starts from an existing high‑quality sequence (height_sequence_1) or a
     random Dirichlet draw.
  2. Repeatedly solves the linear program
        maximise   1ᵀ g   s.t.  a ∗ g ≤ M,   g ≥ 0
     where a is the current distribution and M = max(a ∗ a).
     The solution g tries to place as much probability mass as possible
     without breaking the current worst‑case convolution bound.
  3. A short line‑search mixes a and g (a ← (1‑t)a + t g); a small t
     provably reduces the maximum convolution.
  4. After a few LP steps a short exponentiated‑gradient (mirror‑descent)
     polishing on the exact max‑convolution is performed.
  5. Finally a long EG run is executed until the time budget expires.
"""

"""
Frank‑Wolfe + Exponentiated‑Gradient optimizer for the
max‑convolution objective

    f(a) = 2·n·max(conv(a, a)),   a ≥ 0,   Σ a = 1.

The algorithm:
  1. Initialise from the previously stored best sequence (``height_sequence_1``)
     or from a random Dirichlet draw.
  2. Repeatedly solve the linear programme

        maximise   1ᵀ g            (i.e. maximise total mass)
        subject to   a ∗ g ≤ M ,   g ≥ 0 ,

     where M = max(a ∗ a) and “∗” is the linear convolution.
     The LP puts as much probability mass as possible without breaking the
     current worst‑case convolution bound.
  3. Perform a short line‑search on the convex mixture
        a ← (1‑t)·a + t·g   (t∈[0,0.3])
     and keep the mixture that gives the smallest surrogate value.
  4. After each LP step a few iterations of exponentiated‑gradient (mirror‑
     descent) are executed on the exact max‑convolution to polish the point.
  5. When the time budget is almost exhausted a long EG run finishes the
     optimisation.
  6. The final sequence is stored in ``height_sequence_1`` and returned.

The whole routine respects a user‑supplied time budget (default 1000 s) and
never returns a sequence that violates the non‑negativity or normalisation
requirements enforced by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (required for the LP)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt_lp  # linprog
    import scipy.sparse as _sp        # sparse matrices for the LP
    _LP_AVAILABLE = True
except Exception:  # pragma: no cover
    _LP_AVAILABLE = False
    _opt_lp = None
    _sp = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate objective for a ℓ₁‑normalised vector:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Random families (used for initialisation and occasional restarts)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sample from a symmetric Dirichlet and normalise to sum = 1."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add small Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated gradient) on the true max‑convolution.
    Returns the best sequence seen (as a probability vector) and its surrogate.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry (index k)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Frank‑Wolfe style LP step
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray):
    """Return a sparse (2n‑1)×n matrix representing the linear map g↦a∗g."""
    n = len(a)
    # each diagonal offset d (0 … n‑1) is constant a[d]
    diags = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=int)
    return _sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")


def _frank_wolfe_opt(a0: np.ndarray, deadline: float,
                     max_iter: int = 8) -> tuple[np.ndarray, float]:
    """Run a few Frank‑Wolfe LP steps starting from ``a0``."""
    if not _LP_AVAILABLE:
        return a0, _objective_surrogate(a0)

    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_surrogate(a)
    n = len(a)
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() > deadline:
            break

        # 1) current bound M
        conv_aa = _conv_fft(a, a)
        M = float(np.max(conv_aa))

        # 2) linear programme: maximise 1ᵀg  s.t.  a∗g ≤ M , g≥0
        A = _build_conv_matrix(a)                     # (2n‑1)×n sparse
        b_vec = np.full(2 * n - 1, M, dtype=np.float64)
        c_obj = -np.ones(n, dtype=np.float64)         # -1ᵀg  (maximise sum)

        try:
            res = _opt_lp.linprog(c=c_obj,
                                  A_ub=A,
                                  b_ub=b_vec,
                                  bounds=[(0, None)] * n,
                                  method="highs",
                                  options={"presolve": True})
        except Exception as exc:                       # pragma: no cover
            print(f"[FW] linprog exception at iter {it}: {exc}")
            break

        if not res.success:
            print(f"[FW] linprog failed at iter {it}: {res.message}")
            break

        g_raw = res.x
        if g_raw.sum() <= 0:
            print("[FW] LP returned zero mass – aborting.")
            break
        g = g_raw / g_raw.sum()                       # normalise to Σ=1

        # 3) line‑search on the convex combination a←(1‑t)a + t g
        best_local_val = best_val
        best_local_seq = a.copy()
        best_t = 0.0
        for t in np.linspace(0.0, 0.3, 7)[1:]:        # 0.05,0.10,…0.30
            a_t = (1.0 - t) * a + t * g
            val_t = _objective_surrogate(a_t)
            if val_t < best_local_val:
                best_local_val = val_t
                best_local_seq = a_t.copy()
                best_t = t

        if best_t > 0.0:
            a = best_local_seq
            if best_local_val < best_val:
                best_val = best_local_val
                best_a = a.copy()
                print(f"[FW] iter {it}: improved surrogate to {best_val:.6f} (t={best_t:.3f})")
        else:
            # No obvious improvement – add a tiny random perturbation
            a = _perturb(a, rng, sigma=0.005)
            print(f"[FW] iter {it}: no improvement, perturbed.")

        # 4) short EG polishing (quick local refinement)
        time_left = deadline - time.time()
        if time_left > 5.0:
            sub_dead = time.time() + min(30.0, 0.2 * time_left)
            a_eg, eg_val = _exp_gradient_opt(a, n, sub_dead, max_iters=50_000, eta0=5.0)
            if eg_val < best_val:
                a = a_eg
                best_val = eg_val
                best_a = a.copy()
                print(f"[FW] iter {it}: EG polishing improved to {best_val:.6f}")

        # free the sparse matrix early (helps the GC)
        del A

    return best_a, best_val

# Main entry point

                v = np.log(start_vec + 1e-30)
                beta_schedule = [1.0, 5.0, 20.0, 80.0,
