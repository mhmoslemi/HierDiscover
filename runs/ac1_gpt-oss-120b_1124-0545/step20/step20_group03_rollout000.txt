def subgradient(seq):
    conv = _conv_fft(seq, seq)
    max_val = conv.max()
    # tolerance relative
    tol = max_val * 1e-12
    # get active indices where conv >= max_val - tol
    active = np.where(conv >= max_val - tol)[0]
    g = np.zeros_like(seq)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = k - idx_i
        g[idx_i] += 2.0 * seq[rev_idx]
    g /= len(active)
    return g
```

def proj_simplex(v):
    """Project onto the probability simplex: v_i >= 0, sum = 1."""
    # standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
propose_candidate – multi‑restart simplex optimisation for the max‑convolution
objective.

The algorithm proceeds as follows:

1. Load the previously best sequence (global ``height_sequence_1``) if it
   exists, otherwise start from a handful of analytic families (exponential,
   power‑law, uniform) and a few random Dirichlet draws.

2. Very cheap 1‑D search over the power‑law exponent provides a good U‑shaped
   starter (large values near the ends, small in the centre).

3. Each starter is refined by a short exponentiated‑gradient (mirror‑descent)
   run that uses a *sub‑gradient* averaged over all indices whose convolution
   value is within machine tolerance of the current maximum.  The best
   intermediate solution is kept.

4. The best candidate from step 3 is polished with a long exponentiated‑gradient
   run (up to a few million iterations) and afterwards with a projected‑gradient
   descent on the same sub‑gradient.  Both methods keep the vector on the
   probability simplex (non‑negative, sum = 1).

5. If SciPy is available and time permits, a smooth‑max L‑BFGS optimisation
   (log‑softmax variables) is performed as a final touch.

6. The resulting sequence is evaluated with the supplied ``evaluate_sequence``,
   stored globally as ``height_sequence_1`` and returned.

All helper functions are defined at the top level; no closures or file I/O are
used.  The implementation respects the 1000 s budget by checking the deadline
frequently.
"""

import math
import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:  # pragma: no cover
    import scipy.optimize as _opt
except Exception:  # pragma: no cover

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when sum(seq)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a sequence already normalised to L1‑norm 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax utilities (used by exponentiated‑gradient)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic starter families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), i = 0..n‑1, then normalised to L1=1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _power_law(n: int, p: float) -> np.ndarray:
    """U‑shaped law: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1.0) ** (-p) + (n - i) ** (-p)
    s = w.sum()
    if s == 0.0:
        return _uniform(n)
    return w / s


# ----------------------------------------------------------------------
# Project onto the probability simplex (Euclidean projection)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform
        return _uniform(n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (averaged over all active indices)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of max(conv(a,a)) for a ≥ 0, sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    # absolute tolerance – a few ULPs are enough
    tol = max_val * 1e-12
    active = np.where(conv >= max_val - tol)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] += 2.0 * a[rev]
    g /= max(1, len(active))
    return g

# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,

                            eta0: float = 5.0) -> tuple[np.ndarray, float]:

    best_val = _objective_surrogate(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g

# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth‑max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """Optimise a smooth approximation of the max‑convolution using L‑BFGS‑B."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)        # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

    ``evaluate_sequence`` objective.

    The search combines:
      * a set of analytic starters (exponential, power‑law, uniform, random)
      * a short exponentiated‑gradient polish for each starter
      * a long exponentiated‑gradient refinement of the best candidate
      * an optional projected‑gradient refinement
      * a final L‑BFGS smoothing phase if time permits.

    The best sequence found is stored globally as ``height_sequence_1`` and
    returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the same dimension as the previous best.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 1. Gather initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # a) previous best, if compatible
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            seq = np.asarray(prev, dtype=np.float64)
            if seq.shape == (n,):
                seq = np.clip(seq, 0.0, 1000.0)
                s = seq.sum()
                if s > 0:
                    seq = seq / s
                else:
                    seq = _uniform(n)
                candidates.append(seq)
        except Exception:
            pass

    # b) analytic starters
    candidates.append(_exp_decay(n, rate=0.003))
    candidates.append(_uniform(n))
    # c) power‑law – a cheap 1‑D search gives a strong U‑shaped starter
    candidates.append(_best_power_law(n, deadline))

    # d) a few random Dirichlet draws
    for _ in range(5):
        candidates.append(rng.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # 2. Short EG polish for each starter (cheap, 5 × 10⁴ iterations)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    for idx, seq0 in enumerate(candidates):
        if time.time() > deadline - 30:
            break
        # quick polish – limit iterations so that we keep budget
        seq_opt, val_opt = _exp_gradient_opt_multi(seq0, n, deadline,
                                                   max_iters=50_000,
                                                   eta0=5.0)
        print(f"[Init {idx}] surrogate after short EG = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()

    # If for some reason we have no candidate, fall back to uniform
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective_surrogate(best_seq)

    # ------------------------------------------------------------------
    # 3. Long EG refinement (up to ~2 M iterations)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20:
        seq_fine, val_fine = _exp_gradient_opt_multi(best_seq, n, deadline,
                                                     max_iters=2_200_000,
                                                     eta0=5.0)
        print(f"[Fine EG] surrogate after long run = {val_fine:.6f}")
        if val_fine < best_val:
            best_val, best_seq = val_fine, seq_fine.copy()

    # ------------------------------------------------------------------
    # 4. Projected‑gradient polishing (optional, cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        seq_pg, val_pg = _pgd_opt(best_seq, n, deadline,
                                  max_iters=200_000,
                                  eta0=1.0)
        print(f"[PGD] surrogate after projected‑gradient = {val_pg:.6f}")
        if val_pg < best_val:
            best_val, best_seq = val_pg, seq_pg.copy()

    # ------------------------------------------------------------------
    # 5. Optional L‑BFGS smoothing (if SciPy available)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5 and _opt is not None:
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline)
        val_lb = _objective_surrogate(seq_lb)
        print(f"[L‑BFGS] surrogate after smoothing = {val_lb:.6f}")
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()

    # ------------------------------------------------------------------
    # 6. Final evaluation via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for the next run (facilitates incremental improvement)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
