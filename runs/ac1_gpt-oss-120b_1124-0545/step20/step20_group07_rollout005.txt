"""
Extended optimizer for the step‑function problem.

The algorithm improves on the previous pipeline by:
  1. A richer parametric sweep (exponential, inverse‑power families) and a
     cheap golden‑section refinement of the most promising 1‑D parameters.
  2. Exponentiated‑gradient (mirror‑descent) with a longer schedule.
  3. A continuation L‑BFGS on the smooth max, now with many more
     iterations per β (up to 15 000) and a denser β‑schedule.
  4. A lightweight 2‑way mixture search between the best vectors.
  5. A simulated‑annealing polishing stage that moves a small amount of mass
     between random coordinates while a temperature slowly cools.

All steps respect the 1000 s wall‑time limit and keep the current best
vector in the global ``height_sequence_1`` variable.
"""

import time
import math
import numpy as np

# Optional SciPy – required for the L‑BFGS stage

try:

except Exception:  # pragma: no cover

# Basic numerical helpers

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector, sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# -------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# -------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) for a vector whose ℓ₁‑norm = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic families (all return ℓ₁‑normalised vectors)
# -------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i).  lam ≥ 0."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_linear(n: int, b: float) -> np.ndarray:
    """a_i ∝ 1/(i+1+b).  b > -1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (i + b)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_sqrt(n: int, b: float) -> np.ndarray:
    """a_i ∝ 1/sqrt(i+1+b).  b > -1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i + b)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    # still normalised because α+(1‑α)=1
    return mixed

# Golden‑section search for a single scalar parameter
# -------------------------------------------------------------
def _golden_section(func, a, b, tol=1e-4, max_iter=30):
    """Return (x_opt, f_opt) that approximately minimises func on [a,b]."""
    phi = (np.sqrt(5) - 1) / 2  # 0.618…
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - phi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + phi * (b - a)
            fd = func(d)
    x_opt = (a + b) / 2.0
    f_opt = func(x_opt)
    return x_opt, f_opt

# Exponentiated‑gradient (mirror‑descent) optimiser
# -------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (see comment in original code)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Continuation L‑BFGS on smoothed max
# -------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000):
    """
    Smooth‑max L‑BFGS‑B optimiser.
    Variables live in log‑softmax space (guarantees non‑negativity &
    ℓ₁‑norm = 1).
    """
    if _opt is None:
        return start_vec  # SciPy unavailable – return unchanged

    if beta_schedule is None:
        beta_schedule = [1, 5, 20, 80,
                         200, 500, 1000,
                         2000, 5000,
                         20000, 50000,
                         100000, 200000,
                         500000, 1000000]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (full vector)
            w = exp_beta / sum_exp                     # weighting of each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)                # back‑prop through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

        def func(param):
            vec = base_func(param)
            return _objective_surrogate(vec)

        opt_param, opt_val = _golden_section(func,
                                             bounds[0],
                                             bounds[1],
                                             tol=1e-5,
                                             max_iter=25)
        vec_opt = base_func(opt_param)
        _maybe_insert(vec_opt)

    # Re‑sort after refinement
    best_heap.sort(key=lambda x: x[0])
    top_seqs = [vec for _, vec in best_heap[:top_k]]
    best_val, best_seq = best_heap[0]

    # ---------------------------------------------------------
    # 2. Exponentiated‑gradient refinement
    # ---------------------------------------------------------
    time_left = deadline - time.time()
    # Reserve ~30 % of remaining time for the later stages
    eg_budget = max(0.0, time_left * 0.35)
    eg_deadline = time.time() + eg_budget
    print(f"\n--- EG refinement (budget ≈ {eg_budget:.1f}s) ---")
    for idx, seq in enumerate(top_seqs):
        if time.time() > eg_deadline:
            break
        remaining = eg_deadline - time.time()
        # allocate up to 400 k iterations or what fits in remaining time
        # (heuristic: 1 µs per iteration ≈ generous)
        max_iters = min(400_000, int(2e5 * (remaining / eg_budget + 0.1)))
        seq_opt, val_opt = _exp_gradient_opt(seq.copy(),
                                             n,
                                             eg_deadline,
                                             max_iters=max_iters,
                                             eta0=5.0)
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print(f"  Candidate {idx+1}: new best surrogate = {best_val:.6f}")

    # ---------------------------------------------------------
    # 3. Continuation L‑BFGS with many iterations
    # ---------------------------------------------------------
    if time.time() < deadline - 10.0 and _opt is not None:
        print("\n--- L‑BFGS continuation (high‑precision) ---")
        start_vec = np.maximum(best_seq, 1e-12)
        start_vec = start_vec / start_vec.sum()
        # Very generous iteration budget for the final betas
        seq_lbfgs = _lbfgs_smooth_optimize(start_vec,
                                           n,
                                           deadline,
                                           beta_schedule=[1, 5, 20, 80,
                                                          200, 500, 1000,
                                                          2000, 5000,
                                                          20000, 50000,
                                                          100000, 200000,
                                                          500000, 1000000,
                                                          2000000, 5000000],
                                           max_iter_per_beta=15000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print(f"  L‑BFGS improvement: surrogate = {best_val:.6f}")

    # ---------------------------------------------------------
    # 4. Two‑way mixture optimisation (linear combinations)
    # ---------------------------------------------------------
    # Try mixing the current best with a few other top candidates.
    print("\n--- Mixture refinement ---")
    mix_candidates = best_heap[1:6]   # five alternatives
    for mix_val, mix_vec in mix_candidates:
        # sample α in [0,1] – a simple grid is sufficient
        best_mix_val = best_val
        best_mix_seq = best_seq
        for alpha in np.linspace(0.0, 1.0, 21):
            mixed = _mix_shapes(best_seq, mix_vec, alpha)
            val = _objective_surrogate(mixed)
            if val < best_mix_val:
                best_mix_val = val
                best_mix_seq = mixed.copy()
        if best_mix_val < best_val:
            best_val, best_seq = best_mix_val, best_mix_seq.copy()
            print(f"  Mixed with alt (α≈{alpha:.2f}) → surrogate = {best_val:.6f}")

    # ---------------------------------------------------------
    # 5. Simulated‑annealing polishing
    # ---------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Simulated‑annealing polishing ---")
        seq = best_seq.copy()
        cur_val = best_val
        T = 0.001          # initial temperature
        T_final = 1e-9
        decay = (T_final / T) ** (1.0 / (remaining * 2000))  # approx 2000 updates per sec
        it = 0
        while time.time() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # move a fraction of mass from i to j
            delta = 0.03 * seq[i] * rng.uniform(0.5, 1.5)
            delta = min(delta, seq[i])
            new_seq = seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            val_new = _objective_surrogate(new_seq)
            if val_new < cur_val or rng.uniform() < math.exp(-(val_new - cur_val) / max(T, 1e-12)):
                seq = new_seq
                cur_val = val_new
                if cur_val < best_val:
                    best_val, best_seq = cur_val, seq.copy()
                    print(f"  iter {it:7d}: new best surrogate = {best_val:.6f}")
            T = max(T_final, T * decay)
            it += 1
            if it % 5000 == 0:
                print(f"  iter {it:7d}: T={T:.2e}, cur surrogate={cur_val:.6f}")

    # ---------------------------------------------------------
    # 6. Final true evaluation & bookkeeping
    # ---------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
          f"true evaluate_sequence = {true_val}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
