def _adam_optimize(start_seq, n, deadline, beta_schedule, iters_per_beta):
    log_v = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_v)
    v = np.zeros_like(log_v)
    t = 0
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    for beta in beta_schedule:
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                return best_seq, best_val
            # compute a, conv, shift, w
            a = _softmax(log_v)  # compute a vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # compute softmax w over conv with beta
            # shift for stability
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            if sum_exp == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = exp_beta / sum_exp

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            # full length = n + (2n-1) -1 = 3n-2
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            t += 1
            lr = step_size * np.sqrt(1 - beta2**t) / (1 - beta1**t)   # typical bias correction
            # update moments
            m = beta1 * m + (1 - beta1) * grad_log
            v = beta2 * v + (1 - beta2) * (grad_log**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            log_v = log_v - lr * m_hat / (np.sqrt(v_hat) + eps)

            # compute surrogate value
            val = _objective_surrogate(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
    return best_seq, best_val

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:

def _triangular(n: int) -> np.ndarray:

    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _gaussian(n: int, sigma_frac: float) -> np.ndarray:

def _piecewise_linear(n: int, num_breaks: int,
                      rng: np.random.Generator) -> np.ndarray:

    points = np.concatenate(([0], points, [n - 1]))

    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)

def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:

    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution (mirror‑descent in log‑softmax space)
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule,
                   max_iters_per_beta: int = 2000,
                   step_size: float = 0.10,
                   beta1: float = 0.90,
                   beta2: float = 0.999,
                   eps: float = 1e-12):
    """
    Optimise the smooth approximation of max(conv(a,a)) using Adam.
    All vectors are ℓ₁‑normalised internally.
    Returns the best sequence (as ndarray) and its surrogate value.
    """
    log_v = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_v)
    v = np.zeros_like(log_v)

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    t = 0  # iteration counter for bias correction

    for beta in beta_schedule:
        for _ in range(max_iters_per_beta):
            if time.time() > deadline:
                return best_seq, best_val

            # current probabilities
            a = _softmax(log_v)

            # self‑convolution
            conv = _conv_fft(a, a)

            # smooth‑max weighting
            shift = np.max(conv)               # numeric stabiliser
            exp_beta = np.exp(beta * (conv - shift))
            w = exp_beta / exp_beta.sum()       # shape (2n‑1,)

            # gradient of smoothed max w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)         # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t. log‑softmax (mirror‑descent)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr = step_size * math.sqrt(1.0 - beta2 ** t) / (1.0 - beta1 ** t)
            log_v = log_v - lr * m_hat / (np.sqrt(v_hat) + eps)

            # evaluate the new point
            a_new = _softmax(log_v)
            cur_val = _objective_surrogate(a_new)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a_new.copy()

    return best_seq, best_val

# ----------------------------------------------------------------------
# Permutation‑based hill‑climbing (simulated annealing flavour)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 120000) -> tuple[np.ndarray, float]:
    """
    Swaps two positions or adds tiny Gaussian noise.
    Accepts worsening moves with a linearly cooling temperature.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                     # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                        # Gaussian perturb
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val

# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine follows the multi‑stage scheme described in the module
    docstring and respects the overall 1000 s time budget.
    The best sequence is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # Problem dimension – 4000 provides a good trade‑off between quality
    # and speed for the given budget.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if present (useful for continuation runs)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Build a large pool of candidate seeds
    # ------------------------------------------------------------------
    seeds = []

    # Core analytic families (grid‑search over a few parameters)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.2, 4.0, 15):
        seeds.append(_sine_window(n, gamma=gamma))

    for sigma_frac in np.linspace(0.05, 0.35, 15):
        seeds.append(_gaussian(n, sigma_frac))

    for p in np.linspace(0.1, 4.0, 15):
        seeds.append(_power(n, p))

    # Exponential decays (wide log‑space sweep)
    for rate in np.logspace(-5, -0.3, 20):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Blackman and a few Kaiser windows
    seeds.append(_blackman_window(n))
    for beta in (2.0, 5.0, 10.0):
        seeds.append(_kaiser_window(n, beta))

    # DPSS windows (if SciPy is available)
    if _signal is not None and hasattr(_signal.windows, "dpss"):
        for NW in (2.0, 3.0, 4.0):
            seeds.append(_dpss_window(n, NW))

    # Beta‑shape samples
    for _ in range(80):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws with a wide range of concentrations
    for conc in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Very sparse Dirichlet draws (tiny support)
    for conc in [0.001, 0.005, 0.01]:
        for _ in range(5):
            seeds.append(_random_sparse_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(30):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4):
        for _ in range(6):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random piecewise‑linear curves
    for breaks in (3, 5, 8):
        for _ in range(8):
            seeds.append(_piecewise_linear(n, breaks, rng))

    # Mixtures of exponentials (asymmetric shapes)
    for rate1, rate2, alpha in [(0.003, 0.02, 0.3),
                                (0.01, 0.004, 0.6),
                                (0.002, 0.001, 0.5)]:
        a = _exp_decay(n, rate1)
        b = _sym_exp_decay(n, rate2)
        seeds.append(_mix_shapes(a, b, alpha))

    # Random permutations of a few structured bases (break symmetry)
    for base in [_exp_decay(n, 0.003), _sym_exp_decay(n, 0.004),
                 _beta_shape(n, 2.0, 2.0), _gaussian(n, 0.18)]:
        for _ in range(6):
            seeds.append(np.random.permutation(base))

    # Include the previous best and a few perturbed copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            seeds.append(_perturb(prev_arr, rng, sigma=0.005))

    # ------------------------------------------------------------------
    # 2. Surrogate evaluation and selection of the most promising seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seed surrogate values ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Local refinement for every top seed
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a modest slice of the remaining budget
        remaining = deadline - time.time()
        share = remaining * 0.15 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # (a) quick permutation‑based hill climbing
        seq_hc, val_hc = _perm_hill_climb(seed_vec.copy(),
                                          n, sub_deadline, rng,
                                          max_iters=100_000)

        # (b) Adam optimisation on the smoothed max‑convolution
        if time.time() < sub_deadline:
            beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 2000.0, 5000.0, 20000.0,
                         50000.0, 100000.0, 200000.0,
                         500000.0, 1000000.0]
            seq_adam, val_adam = _adam_optimize(seq_hc, n,
                                                sub_deadline,
                                                beta_sched,
                                                max_iters_per_beta=1500,
                                                step_size=0.10)
        else:
            seq_adam, val_adam = seq_hc, val_hc

        # (c) pick the best among the local candidates
        cand = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_hc, val_hc),
            (seq_adam, val_adam)
        ]
        local_best_seq, local_best_val = min(cand, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long Adam polishing using the remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print("\n--- Final global Adam polishing ---")
        beta_sched = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                     1000.0, 2000.0, 5000.0, 20000.0,
                     50000.0, 100000.0, 200000.0,
                     500000.0, 1000000.0]
        # a few thousand iterations per β give a deep final refinement
        global_best_seq, global_best_val = _adam_optimize(
            global_best_seq, n, deadline,
            beta_sched,
            max_iters_per_beta=3000,
            step_size=0.08)

    # ------------------------------------------------------------------
    # 6. Optional short permutation tweak (≈30 s)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 30.0
        seq_last, val_last = _perm_hill_climb(global_best_seq.copy(),
                                              n, short_deadline, rng,
                                              max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final permutation] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Evaluate the true objective and store the result
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
