def _two_block_shape(n, block_size, gap):
    seq = np.zeros(n, dtype=np.float64)
    if block_size <= 0:
        return _uniform(n)
    # first block from 0 to block_size - 1
    seq[:block_size] = 1.0 / (2 * block_size)
    # second block start at block_size + gap
    start2 = block_size + gap
    if start2 + block_size <= n:
        seq[start2:start2+block_size] = 1.0 / (2 * block_size)
    else:
        # if overflow, adjust second block size to remaining
        size2 = max(0, n - start2)
        if size2 > 0:
            seq[start2:] = 1.0 / (block_size + size2)  # not exact but ensures sum=1
    return seq

"""
search for a non‑negative step function that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three stages:

1. **Massive seeding** – a large, diverse collection of analytically‑defined
   shapes (uniform, triangular, beta‑skewed, exponential, power‑law,
   two‑block, mixtures, random Dirichlet draws).  All vectors are ℓ₁‑normalised.
   The top few (by the exact surrogate objective) survive.

2. **Local optimisation** – each survivor is refined by
   *exponentiated‑gradient* (mirror descent) on the exact non‑smooth
   objective, followed by a *smooth‑max* L‑BFGS (log‑sum‑exp) pass.
   Only a modest fraction of the time budget is spent per seed.

3. **Polishing** – a longer exponentiated‑gradient run on the best
   candidate and a final low‑temperature simulated‑annealing pass.

All work respects the given time budget.  The resulting best sequence is
stored in the global variable ``height_sequence_1`` and returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# optional SciPy for L‑BFGS
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate objective (for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective when sum(seq)=1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i), i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _two_block_shape(n: int, block: int, gap: int) -> np.ndarray:
    """
    Two uniform blocks of equal total mass (0.5 each) separated by `gap`.
    `block` is the length of each block.  If the second block would exceed the
    vector length it is omitted.
    """
    seq = np.zeros(n, dtype=np.float64)
    if block <= 0:
        return _uniform(n)
    if block * 2 + gap > n:
        # can't fit two blocks – fall back to a single block
        seq[:block] = 1.0 / block
        seq /= seq.sum()
        return seq
    seq[:block] = 0.5 / block
    start2 = block + gap
    seq[start2:start2 + block] = 0.5 / block
    return seq


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 5.0,
                      single_active: bool = False):
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
    If ``single_active`` is True only the most active index is used for the
    sub‑gradient; otherwise all indices whose value is within a tiny tolerance
    of the maximum are used (capped to 200 sampled indices for speed).
    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space (soft‑max guarantees positivity & ℓ₁‑norm)
    log_a = np.log(a + 1e-30)

    max_active = 200  # cap to keep each iteration cheap

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # ----- active set -------------------------------------------------
        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]
            if active.size > max_active:
                # random subset when far too many indices are active
                rng = np.random.default_rng()
                active = rng.choice(active, max_active, replace=False)

        # ----- sub‑gradient ------------------------------------------------
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            # exact sub‑gradient of the max is the *average* over the active ones
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # ----- learning‑rate (1/√t decay) --------------------------------
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 1500):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing feasibility.
    """
    if _opt is None:
        return start_vec  # SciPy unavailable – return unchanged

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0,
            20_000.0, 50_000.0,
            100_000.0, 200_000.0,
            500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()                     # stability shift
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp                 # weights over each convolution index
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice back to length n

            # chain rule for log‑softmax (mirror‑descent)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.004,
                         T_start: float = 0.02,
                         T_end: float = 1e-7):
    """
    Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    The routine respects the time budget and stores the best found sequence in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more flexibility at modest cost.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Load a previous solution if available (helps continuity)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Deterministic & random seeding
    # ------------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # simple analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_triangular(n)[::-1])
    for gamma in (0.5, 1.0, 2.0):
        seeds.append(_sine_window(n, gamma))

    # exponential decays
    for r in np.logspace(-5, -0.5, 25):
        if time.time() > deadline:
            break
        seeds.append(_exp_decay(n, r))

    # power‑law families
    for p in np.logspace(-2, 2, 25):
        if time.time() > deadline:
            break
        seeds.append(_power(n, p))

    # beta‑skewed families – log‑spaced extreme values
    alphas = np.logspace(-3, 2, 20)
    betas = np.logspace(-3, 2, 20)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            seeds.append(_beta_shape(n, a, b))

    # two‑block shapes (two separated uniform blocks)
    for block_frac in np.linspace(0.02, 0.25, 8):
        block = max(1, int(n * block_frac))
        max_gap_frac = max(0.0, (n - 2 * block) / n)
        for gap_frac in np.linspace(block_frac, max_gap_frac, 5):
            if time.time() > deadline:
                break
            gap = int(n * gap_frac)
            seeds.append(_two_block_shape(n, block, gap))

    # random Dirichlet draws (highly sparse)
    for conc in (1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 5e-2, 0.1):
        for _ in range(3):
            if time.time() > deadline:
                break
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixture uniform + exponential
    for r in np.logspace(-5, -0.5, 10):
        if time.time() > deadline:
            break
        exp_seq = _exp_decay(n, r)
        seeds.append(_mix_shapes(_uniform(n), exp_seq, 0.3))

    # mixture of left/right beta shapes
    for a_val in (0.2, 0.5, 1.0, 2.0):
        for b_val in (0.2, 0.5, 1.0, 2.0):
            beta1 = _beta_shape(n, a_val, b_val)
            beta2 = _beta_shape(n, b_val, a_val)
            seeds.append(_mix_shapes(beta1, beta2, 0.4))

    # add previous best and gentle perturbations
    if prev_arr is not None and prev_arr.sum() > 0.0:
        base = prev_arr / prev_arr.sum()
        seeds.append(base)
        for _ in range(5):
            pert = base + 0.002 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        raise RuntimeError("No valid seeds generated within the time budget.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start local refinement (exp‑gradient → L‑BFGS → SA)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a fraction of the remaining time for this seed
        remaining_time = deadline - time.time()
        remaining_seeds = top_k - si
        sub_deadline = time.time() + remaining_time * 0.6 / max(1, remaining_seeds)

        # (a) exponentiated‑gradient (multi‑active)
        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                           n,
                                           sub_deadline,
                                           max_iters=150_000,
                                           eta0=6.0,
                                           single_active=False)

        # (b) L‑BFGS smoothing (if time permits)
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               max_iter_per_beta=2000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs = seq_eg
            val_lbfgs = val_eg

        # (c) short simulated‑annealing polish
        hill_deadline = sub_deadline - 0.2
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_lbfgs,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=80_000,
                                                  start_sigma=0.0015,
                                                  T_start=0.02,
                                                  T_end=5e-8)
        else:
            seq_sa, val_sa = seq_lbfgs, val_lbfgs

        # pick the best among the three refinements and the original seed
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing – longer EG + final SA
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (multi‑active) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=800_000,
                                                 eta0=5.0,
                                                 single_active=False)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final short SA polishing ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                          n,
                                                          deadline,
                                                          rng,
                                                          max_attempts=100_000,
                                                          start_sigma=0.001,
                                                          T_start=0.01,
                                                          T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
