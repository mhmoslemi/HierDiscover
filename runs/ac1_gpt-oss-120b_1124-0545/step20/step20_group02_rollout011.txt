def _smooth_gradient_descent(seq, n, deadline, beta, eta0=1.0, max_iters=2000):
    a = seq.copy()
    log_a = np.log(a+1e-30)
    for it in range(max_iters):
        if time.time()>deadline: break
        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        shift = max_c
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp
        # compute gradient in a space
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length 3n-2
        grad_a = 2.0 * full[n-1:n-1+n]
        # chain rule to v-space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        eta = eta0 / np.sqrt(it+1)
        log_a -= eta * grad_v
    return _softmax(log_a)
```

```
def _exp_gradient_opt_until(seq, n, deadline):
    a = seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        eps = max_val * 1e-6
        active = np.where(conv >= max_val - eps)[0]
        # compute gradient g_tot
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]
        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)
        # learning rate schedule
        eta = 5.0 / max(1.0, math.sqrt(iteration))
        log_a -= eta * g
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[EG] iter {iteration}, improved {best_val:.6f}")
    return best_seq, best_val

"""
Search for a non‑negative step‑function (height sequence) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The algorithm works in three phases:

1. **Broad exploration** – generate many candidates from simple analytic families
   (uniform, sine window, exponential decays, beta‑shapes, low‑frequency random
   curves) and from random Dirichlet draws.  All candidates are normalised to a
   unit ℓ¹‑norm and the best according to the surrogate
   `2·n·max(conv)` is kept.

2. **Mirror‑descent polishing** – starting from the best candidate we run a
   long exponentiated‑gradient (mirror‑descent) loop that directly minimizes the
   exact `max(conv)` on the probability simplex.  The loop runs until the time
   budget is exhausted.

3. **Fine‑tuning** – a lightweight simulated‑annealing pass makes occasional
   uphill moves to escape any remaining local minima.

The final sequence (scaled to sum = 1) is stored in the global variable
`height_sequence_1` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate/objective (sum = 1 → denominator is 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ¹‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_until(start_seq: np.ndarray,
                            deadline: float,
                            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Runs until ``deadline`` and returns the best sequence seen.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0

    while True:
        if time.time() > deadline:
            break
        it += 1

        conv = _conv_fft(a, a)
        max_val = conv.max()
        eps = max_val * 1e-6
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient of the max over the active indices
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]
        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # decreasing step size (similar to 1/√t)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 50000 == 0:
                print(f"[EG] iter {it:,} – improved surrogate to {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (accept occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 150_000,
                         start_sigma: float = 0.001,
                         T_start: float = 5e-3,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linear cooling schedule.
    Returns the best sequence discovered before ``deadline``.
    """
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Candidate handling utilities
# ----------------------------------------------------------------------
def _maybe_update_best(seq: np.ndarray,
                       best_info: dict,
                       name: str = "") -> None:
    """
    Normalises ``seq`` (non‑negative, sum = 1), evaluates the surrogate,
    and updates the entry stored in ``best_info`` if an improvement is found.
    """
    seq = np.maximum(seq, 0.0)
    ss = seq.sum()
    if ss == 0.0:
        return
    seq = seq / ss
    val = _objective_surrogate(seq)
    if val < best_info["val"]:
        best_info["val"] = val
        best_info["seq"] = seq.copy()
        best_info["name"] = name
        print(f"[Info] New best ({name}) → {val:.6f}")

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem size (fixed; can be changed if desired)
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous solution if available
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0.0:
                    prev_arr /= s
                else:
                    prev_arr = None
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # --------------------------------------------------------------
    # Container for the best sequence discovered so far
    # --------------------------------------------------------------
    best_info = {"seq": None, "val": float("inf"), "name": ""}

    if prev_arr is not None:
        _maybe_update_best(prev_arr, best_info, name="previous")
    else:
        print("[Info] No previous sequence found – starting from scratch.")

    # --------------------------------------------------------------
    # 1) Broad exploration over analytic families and random draws
    # --------------------------------------------------------------
    print("\n--- Exploration of analytic families ---")
    # deterministic families
    families = [
        (_uniform, {}, "uniform"),
        (_sine_window, {"gamma": 1.0}, "sine_g1.0"),
        (_sine_window, {"gamma": 1.2}, "sine_g1.2"),
        (_sine_window, {"gamma": 1.4}, "sine_g1.4"),
        (_exp_decay, {"rate": 5e-4}, "exp_r5e-4"),
        (_exp_decay, {"rate": 1e-3}, "exp_r1e-3"),
        (_exp_decay, {"rate": 2e-3}, "exp_r2e-3"),
        (_exp_decay, {"rate": 5e-3}, "exp_r5e-3"),
        (_sym_exp_decay, {"rate": 1e-3}, "symexp_r1e-3"),
        (_sym_exp_decay, {"rate": 2e-3}, "symexp_r2e-3"),
    ]

    for func, kwargs_f, name in families:
        if time.time() > deadline:
            break
        seq = func(n, **kwargs_f)
        _maybe_update_best(seq, best_info, name=name)

    # grid over exponential rates (log‑space)
    rates = np.logspace(np.log10(5e-4), np.log10(5e-2), num=25)
    for rate in rates:
        if time.time() > deadline:

        seq = _exp_decay(n, rate)
        _maybe_update_best(seq, best_info, name=f"exp_rate{rate:.1e}")

    # grid over beta‑shape parameters
    alphas = np.linspace(0.4, 5.0, 12)
    betas = np.linspace(0.4, 5.0, 12)
    for alpha in alphas:
        for beta in betas:
            if time.time() > deadline:
                break
            seq = _beta_shape(n, alpha, beta)
            _maybe_update_best(seq, best_info, name=f"beta_a{alpha:.1f}_b{beta:.1f}")

    # random low‑frequency curves
    for nf in (3, 5, 7):
        for _ in range(15):
            if time.time() > deadline:
                break
            seq = _lowfreq_curve(n, nf, rng)
            _maybe_update_best(seq, best_info, name=f"lfc_{nf}")

    # random Dirichlet draws with various concentration parameters
    concs = [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]
    for conc in concs:

            if time.time() > deadline:

            seq = _random_dirichlet(n, conc, rng)
            _maybe_update_best(seq, best_info, name=f"dir_conc{conc}")

    # --------------------------------------------------------------

    if best_info["seq"] is not None and time.time() < deadline:

        seq_eg, val_eg = _exp_gradient_opt_until(best_info["seq"], deadline, eta0=5.0)
        if val_eg < best_info["val"]:
            best_info["seq"] = seq_eg
            best_info["val"] = val_eg
            best_info["name"] = "eg_polished"
            print(f"[Info] EG improved surrogate to {val_eg:.6f}")

    # --------------------------------------------------------------
    # 3) Final simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if best_info["seq"] is not None and time.time() < deadline:
        print("\n--- Simulated‑annealing fine‑tuning ---")
        seq_sa, val_sa = _simulated_annealing(best_info["seq"], deadline, rng,
                                              max_attempts=180_000,
                                              start_sigma=0.0005,
                                              T_start=1e-2,
                                              T_end=1e-9)
        if val_sa < best_info["val"]:
            best_info["seq"] = seq_sa
            best_info["val"] = val_sa
            best_info["name"] = "sa_polished"
            print(f"[Info] SA improved surrogate to {val_sa:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    final_seq = best_info["seq"]
    if final_seq is None:
        raise RuntimeError("No sequence was generated – this should not happen.")

    try:
        true_val = evaluate_sequence(final_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate {best_info['val']:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Error] evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for future runs
    globals()["height_sequence_1"] = final_seq.tolist()
    return final_seq.tolist()
