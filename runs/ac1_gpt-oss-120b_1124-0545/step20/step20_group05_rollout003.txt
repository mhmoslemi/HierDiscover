def build_A_ub(f, active_indices):
    n = len(f)

        cols = np.arange(col_start, col_end + 1)

    return A

def solve_max_sum_lp(f, M, active_indices):
    A = build_A_ub(f, active_indices)
    b_ub = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(len(f))
    bounds = [(0, None)] * len(f)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    # Normalize
    s = b.sum()
    if s <= 0:
        return None
    b = b / s
    return b
```

"""
propose_candidate
-----------------

A two‑stage optimiser for the step‑height problem.

Stage 1 – Cutting‑plane linear programming:
   * Start from a good seed (the previous best, or a double‑Gaussian).
   * Compute M = max(conv(f,f)).
   * Solve max ∑b   s.t.   f * b ≤ M (component‑wise)  and b≥0.
     The constraint set is huge (2 n‑1 rows), so we start with only the
     indices where conv(f,f) is close to M and iteratively add any
     violated inequalities (a simple cutting‑plane scheme).  The LP is
     tiny (≤ 200 constraints) and solved with SciPy's high‑performance
     “highs’’ driver.

   * After each LP we renormalise b to a probability vector.
     If max(conv(b,b)) < M we accept b as the new f and repeat.

Stage 2 – Mirror‑descent (exponentiated‑gradient) fine‑tuning:
   * Run the exact sub‑gradient method on the max‑convolution objective
     (the same routine used in the previous solution) for the remaining
     time budget.

Both stages respect the total wall‑time budget (default 1000 s) and
always keep the best vector seen.  The final vector is evaluated with
``evaluate_sequence`` and stored in the global ``height_sequence_1``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Core numeric utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – turns a vector into a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a length‑n vector with ℓ₁‑norm = 1:
       2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Linear‑programming cutting‑plane helpers
# ------------------------------------------------------------
def _build_A_ub(f: np.ndarray, act_idx: list[int]) -> np.ndarray:
    """
    Build the (|act_idx| × n) matrix A such that (A @ b)[k] = (f * b)[act_idx[k]].
    Each row k corresponds to a convolution index.
    """
    n = len(f)
    K = len(act_idx)
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(act_idx):
        col_start = max(0, k - (n - 1))
        col_end = min(k, n - 1)
        cols = np.arange(col_start, col_end + 1, dtype=int)
        # b[col] is multiplied by f[k-col]
        A[row, cols] = f[k - cols]
    return A


def _solve_lp_cutting(f: np.ndarray, M: float,
                      init_active: list[int],
                      deadline: float,
                      max_constraints: int = 200) -> tuple[np.ndarray | None, bool]:
    """
    Solve
        maximise   1ᵀ b
        subject to conv(f,b)[k] ≤ M   for k ∈ active,
                   b ≥ 0
    using a cutting‑plane approach: after solving, any violated
    inequality (conv(f,b)[k] > M) is added to the active set.
    Returns a feasible b (or None) and a flag indicating whether all
    constraints are satisfied.
    """
    n = len(f)
    active = set(init_active)

    while True:
        if time.time() > deadline:
            return None, False

        A = _build_A_ub(f, list(active))
        b_ub = np.full(A.shape[0], M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)               # maximise sum(b)
        bounds = [(0, None)] * n

        # linear programme
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception as e:          # pragma: no cover
            print(f"Linprog exception: {e}")
            return None, False

        if not res.success:
            # LP infeasible (should not happen because b=f is feasible)
            return None, False

        b = res.x
        # Normalise to a probability vector (the LP maximises the *unscaled* sum)
        s = b.sum()
        if s <= 0.0:
            return None, False
        b = b / s

        # Check all (2n‑1) convolution constraints
        conv_fb = _conv_fft(f, b)
        viol = np.where(conv_fb > M + 1e-12)[0]
        if viol.size == 0:
            # feasible for the full set of constraints
            return b, True

        # add a few violating indices and continue
        for idx in viol[:10]:
            active.add(int(idx))
        if len(active) > max_constraints:
            # give up enlarging the system – return the current b (may be infeasible)
            return b, False


# ------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser
# ------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 2.5

    # --------------------------------------------------------
    # 1. Initialise a decent seed (previous best or a double‑Gaussian)
    # --------------------------------------------------------
    n = 4000
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                f = arr / arr.sum()
                print("Using previously stored height_sequence_1 as start.")
            else:
                raise ValueError
        except Exception:
            f = None
    else:
        f = None

    if f is None:
        # build a double‑Gaussian shape – a reasonably good generic start
        sigma_frac = 0.07
        shift_frac = 0.30
        sigma = sigma_frac * n
        mu1 = shift_frac * (n - 1)
        mu2 = (1.0 - shift_frac) * (n - 1)
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
        a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
        s = a.sum()
        f = a / s if s > 0 else np.full(n, 1.0 / n)

    # keep the best seen so far
    best_f = f.copy()
    best_val = _objective_surrogate(best_f)
    print(f"Initial surrogate value: {best_val:.8f}")

    # --------------------------------------------------------
    # 2. Cutting‑plane LP stage
    # --------------------------------------------------------
    lp_deadline = deadline - 200.0   # reserve ~200 s for the final EG stage
    iteration = 0
    while time.time() < lp_deadline:
        iteration += 1
        conv_f = _conv_fft(f, f)
        M_f = float(np.max(conv_f))
        # choose active indices where the convolution is within a tiny
        # relative tolerance of the maximum (normally just a handful)
        eps_active = 1e-5
        active = np.where(conv_f >= M_f * (1.0 - eps_active))[0].tolist()
        if not active:                     # fallback – always include the argmax
            active = [int(np.argmax(conv_f))]

        # solve the LP with cutting‑plane enrichment
        sub_deadline = lp_deadline - time.time()
        if sub_deadline <= 0:
            break
        b, feasible = _solve_lp_cutting(f, M_f, active, sub_deadline)
        if b is None:
            print("LP failed – terminating LP stage.")
            break

        # normalise (already normalised inside the solver, but keep safe)
        b = b / b.sum()
        M_b = float(np.max(_conv_fft(b, b)))

        # improvement test
        if M_b < M_f - 1e-12:
            f = b
            cur_val = 2.0 * n * M_b
            if cur_val < best_val:
                best_val, best_f = cur_val, b.copy()
                print(f"LP iter {iteration:3d}: improved – M={M_b:.8e}, val={cur_val:.8f}")
            continue
        else:
            # no progress – exit the LP loop
            print(f"LP iter {iteration:3d}: no further improvement (M={M_b:.8e})")
            break

    # --------------------------------------------------------
    # 3. Mirror‑descent fine‑tuning (exponentiated gradient)
    # --------------------------------------------------------
    if time.time() < deadline - 5.0:
        eg_deadline = deadline - 2.0   # a tiny cushion before final evaluation
        f_eg, val_eg = _exp_gradient_opt(best_f.copy(), n, eg_deadline,
                                         max_iters=2_000_000, eta0=5.0)
        if val_eg < best_val:
            best_val, best_f = val_eg, f_eg.copy()
            print(f"EG refinement improved value to {val_eg:.8f}")

    # --------------------------------------------------------
    # 4. Final official evaluation and return
    # --------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_f.tolist())
    except Exception as exc:          # pragma: no cover
        print(f"Warning: evaluate_sequence raised {exc}")

    print("\n=== Finished ===")
    print(f"Best surrogate ≈ {best_val:.8f}, true evaluate_sequence = {true_val}")
    print(f"Length = {len(best_f)}")

    # store for reuse by a subsequent run
    globals()["height_sequence_1"] = best_f.tolist()
    return best_f.tolist()
