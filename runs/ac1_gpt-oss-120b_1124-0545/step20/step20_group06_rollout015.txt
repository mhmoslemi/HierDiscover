   best_val = math.inf

   p_values = np.concatenate((np.linspace(0,10,101), np.linspace(10,30,41))) # fine near low region

      seq = base ** p
      seq_sum = seq.sum()
      if seq_sum == 0: continue
      seq = seq / seq_sum
      val = _objective_surrogate(seq)    # surrogate = 2*n*max(conv)
      if val < best_val:
          best_val = val
          best_seq = seq.copy()

   # Additional search for mixture with uniform:
   # try mixing factor α ∈ [0,0.5] maybe:
   # base shape best_seq + uniform mixture:
   uniform = np.full(n, 1.0/n)
   for alpha in np.linspace(0,0.5,11):
      mix = (1-alpha)*best_seq + alpha*uniform
      mix = mix / mix.sum()
      val = _objective_surrogate(mix)
      if val < best_val:
           best_val=val
           best_seq = mix

   # final evaluation with evaluate_sequence:
   try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"Best objective from evaluate_sequence: {true_val:.6f}")
   except Exception as e:
        true_val = None
        print("Evaluation error", e)

   # store globally.
   globals()["height_sequence_1"] = best_seq.tolist()
   return best_seq.tolist()

"""
optimiser_v2.py – Fast, high‑resolution search for a step‑function with
minimal ``evaluate_sequence`` value.

Algorithm
---------
1. **Parameterised seeds** – a large random pool of simple analytic families
   (uniform, exponential decay, beta‑shaped, mixtures of the latter).  For each
   seed we normalise the vector (ℓ₁‑norm = 1) and evaluate the cheap surrogate

        surrogate(seq) = 2 * n * max(conv(seq, seq))

   (exact for normalised vectors).  The best few seeds are retained.

2. **Refinement** – Starting from the current best seed we run two complementary
   optimisers:
   * *Exact mirror‑descent* on the true max‑convolution (full active set,
     exponentiated‑gradient in log‑space).  This directly reduces the true
     ℓ∞‑norm of the auto‑convolution.
   * *Smooth L‑BFGS* – minimise the log‑sum‑exp surrogate with an increasing
     temperature schedule (β up to 1e7).  The smooth objective converges
     reliably to a point where the true max is minimal.

3. **Polishing** – a short simulated‑annealing pass (Gaussian perturbations)
   that can escape shallow local minima.

All work respects a hard 1000 s deadline (a safety margin of 2 s).  The final
sequence is stored in the global ``height_sequence_1`` variable and returned as
a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper – FFT based linear convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) – the exact evaluate_sequence value
    when sum(seq) == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return raw (non‑normalised) vectors)
# ----------------------------------------------------------------------
def _uniform_raw(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64)


def _exp_decay_raw(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-rate * i)


def _beta_shape_raw(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a


def _mix_raw(weights: list[float], components: list[np.ndarray]) -> np.ndarray:
    """Convex mixture Σ w_j·comp_j (weights should sum to 1)."""
    mixed = np.zeros_like(components[0])
    for w, comp in zip(weights, components):
        mixed += w * comp
    return mixed


# ----------------------------------------------------------------------
# Exact mirror‑descent (exponentiated‑gradient) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      eta0: float = 5.0,
                      single_active: bool = False):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    If ``single_active`` is True only the *most* active index is used for the
    sub‑gradient; otherwise all indices within a small tolerance are used.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space; the soft‑max guarantees positivity & ℓ₁‑norm
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # ----- active set -------------------------------------------------
        if single_active:
            k = int(conv.argmax())
            active = np.array([k], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        # ----- sub‑gradient (exact) --------------------------------------
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # ----- learning‑rate (1/√t decay) -------------------------------
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = np.exp(log_a)
        a /= a.sum()                     # maintain ℓ₁‑norm = 1

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    The optimisation works in log‑softmax space, guaranteeing feasibility.
    """
    try:
        import scipy.optimize as _opt
    except Exception:                     # pragma: no cover
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 5_000_000.0, 1_000_0000.0
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = np.exp(v_local)
            a /= a.sum()
            conv = _conv_fft(a, a)
            shift = conv.max()                     # stability shift
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp                 # weights over each convolution index
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice back to length n

            # chain rule for log‑softmax (mirror‑descent)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    a = np.exp(v)
    a /= a.sum()
    return a

# Simulated‑annealing (Gaussian perturbations, accepts occasional worsenings)

def _simulated_annealing(start_seq: np.ndarray,

    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and stores
    the best found sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # problem size – 4000 is a good trade‑off between detail and speed
    n = 4000

    # ------------------------------------------------------------------
    # 1. Randomised seeding across many analytic families
    # ------------------------------------------------------------------
    print("\n--- Random seeding (n={}) ---".format(n))
    best_seq = None
    best_val = math.inf

    # a few deterministic starters
    candidates = []

    # uniform
    candidates.append(_uniform_raw(n))

    # exponential decays (log‑uniform rates)
    for _ in range(30):
        rate = 10 ** rng.uniform(-5, -0.1)   # roughly 1e‑5 … 0.79
        candidates.append(_exp_decay_raw(n, rate))

    # beta shapes (log‑uniform alphas/betas)
    for _ in range(80):
        alpha = 10 ** rng.uniform(-2, 1)    # 0.01 … 10
        beta = 10 ** rng.uniform(-2, 1)
        candidates.append(_beta_shape_raw(n, alpha, beta))

    # mixtures of two exponentials
    for _ in range(80):
        rate1 = 10 ** rng.uniform(-5, -0.1)
        rate2 = 10 ** rng.uniform(-5, -0.1)
        mix = rng.uniform(0.0, 1.0)
        comp1 = _exp_decay_raw(n, rate1)
        comp2 = _exp_decay_raw(n, rate2)
        candidates.append(_mix_raw([mix, 1 - mix], [comp1, comp2]))

    # mixtures of beta and uniform
    for _ in range(40):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        mix = rng.uniform(0.0, 0.5)
        beta_seq = _beta_shape_raw(n, alpha, beta)
        uniform_seq = _uniform_raw(n)
        candidates.append(_mix_raw([mix, 1 - mix], [beta_seq, uniform_seq]))

    # random Dirichlet draws (extreme sparsity)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1):
        for _ in range(6):
            cand = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            candidates.append(cand)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    for vec in candidates:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()

    print(f"\n[Seed] best surrogate after seeding = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Refinement – mirror‑descent + smooth L‑BFGS
    # ------------------------------------------------------------------
    if best_seq is None:                     # very unlikely, but guard
        best_seq = _uniform_raw(n) / n

    # a. Exact mirror‑descent (full active set, many iterations)
    sub_deadline = time.time() + (deadline - time.time()) * 0.40
    seq_md, val_md = _exp_gradient_opt(best_seq.copy(),
                                       n,
                                       sub_deadline,
                                       max_iters=800_000,
                                       eta0=8.0,
                                       single_active=False)
    if val_md < best_val:
        best_val = val_md
        best_seq = seq_md.copy()
        print(f"[MD] improved surrogate = {best_val:.6f}")

    # b. Smooth L‑BFGS (aggressive β schedule)
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.30
        seq_lbfgs = _lbfgs_smooth_optimize(best_seq.copy(),
                                           n,
                                           sub_deadline,
                                           beta_schedule=[
                                               1.0, 5.0, 20.0, 80.0,
                                               200.0, 500.0, 1_000.0,
                                               2_000.0, 5_000.0,
                                               20_000.0, 50_000.0,
                                               100_000.0, 200_000.0,
                                               500_000.0, 1_000_000.0,
                                               5_000_000.0, 1_000_0000.0
                                           ],
                                           max_iter_per_beta=4000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < best_val:
            best_val = val_lbfgs
            best_seq = seq_lbfgs.copy()
            print(f"[L‑BFGS] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final polishing – brief simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.20
        seq_sa, val_sa = _simulated_annealing(best_seq,
                                              n,
                                              sub_deadline,
                                              rng,
                                              max_attempts=120_000,
                                              start_sigma=0.001,
                                              T_start=5e-3,
                                              T_end=1e-9)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"[SA] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Official evaluation with the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for continuity across runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
