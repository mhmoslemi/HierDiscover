"""
Simulated‑annealing local optimisation of a step‑function.

Starting from the best sequence already stored in the global
``height_sequence_1`` (or a uniform distribution if none exists) the
algorithm repeatedly:
  1. evaluates the full self‑convolution,
  2. selects a source index with probability proportional to its
     contribution to the current maximal convolution entry,
  3. moves a small random fraction of the source mass to a uniformly
     chosen different target index,
  4. accepts the move if it improves the objective or, with a
     Metropolis probability, if it is slightly worse (temperature
     cools linearly).

All moves keep the vector non‑negative and on the unit ℓ₁‑simplex,
hence the evaluation function is scale‑invariant.  The search runs
until the time budget (default 1000 s) is exhausted, after which the
best sequence found is stored in ``height_sequence_1`` and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric helpers (must be top‑level – no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective identical to ``evaluate_sequence`` for a
    unit‑sum vector:  2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method is a targeted simulated‑annealing
    local search that focuses on the entries contributing to the current
    maximal convolution value.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a few seconds margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Initialise from the best known construction (if any)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple)) and len(prev) > 0:
        seq = np.asarray(prev, dtype=np.float64)
    else:
        # fall back to a uniform distribution of length 4000
        n = 4000
        seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Normalise to a unit ℓ₁‑norm and clip possible negatives
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total <= 0.0:
        seq = np.full_like(seq, 1.0 / seq.size)
    else:
        seq = seq / total

    n = len(seq)

    # ------------------------------------------------------------------
    # Current best bookkeeping
    # ------------------------------------------------------------------
    cur_obj = _objective_surrogate(seq)
    best_seq = seq.copy()
    best_val = cur_obj
    print(f"Start: n={n}, objective={best_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters
    # ------------------------------------------------------------------
    start_temp = 0.02               # initial temperature for Metropolis rule
    step_frac  = 0.10                # initial fraction of source mass to move
    it = 0

    # ------------------------------------------------------------------
    # Main SA loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # 1. compute full convolution of the current sequence
        conv = _conv_fft(seq, seq)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        cur_obj = 2.0 * n * max_val    # objective for the current state

        # 2. pick a source index weighted by its contribution to the max entry
        i_start = max(0, max_idx - (n - 1))
        i_end   = min(n - 1, max_idx)
        idx_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        rev_idx   = max_idx - idx_range
        contrib   = seq[idx_range] * seq[rev_idx]
        tot_contrib = contrib.sum()
        if tot_contrib > 0.0:
            probs = contrib / tot_contrib
            source = rng.choice(idx_range, p=probs)
        else:
            source = rng.integers(0, n)

        # 3. choose a distinct target uniformly at random
        target = rng.integers(0, n - 1)
        if target >= source:
            target += 1   # ensure target ≠ source

        # 4. propose moving a random fraction of the source mass
        delta = step_frac * seq[source] * rng.uniform(0.5, 1.5)
        if delta > seq[source]:
            delta = seq[source]

        new_seq = seq.copy()
        new_seq[source] -= delta
        new_seq[target] += delta

        # 5. evaluate the new objective
        new_obj = _objective_surrogate(new_seq)

        # 6. Metropolis acceptance rule
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            elapsed = time.time() - start_time
            total   = deadline - start_time
            # linear cooling (temperature never reaches zero)
            temp = start_temp * max(0.0, 1.0 - elapsed / total)
            temp = max(temp, 1e-8)
            prob = math.exp(-(new_obj - cur_obj) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            seq = new_seq
            cur_obj = new_obj
            # slowly increase step size after a success
            step_frac = min(0.25, step_frac * 1.02)
            if new_obj < best_val - 1e-12:
                best_val = new_obj
                best_seq = new_seq.copy()
                print(f"Iter {it:6d}: new best = {best_val:.6f}")
        else:
            # shrink step size after a failure
            step_frac = max(1e-5, step_frac * 0.998)

        it += 1
        # occasional status line
        if it % 50000 == 0:
            print(f"Iter {it:6d}: cur={cur_obj:.6f}, best={best_val:.6f}, step={step_frac:.5f}")

    # ------------------------------------------------------------------
    # Final true evaluation and record keeping
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nSurrogate = {best_val:.6f}, evaluate_sequence = {true_val}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
