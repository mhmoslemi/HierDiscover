def _adam_optimize(start_a, deadline, steps_per_beta=2000):
    n = len(start_a)
    v = np.log(start_a + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1=0.9; beta2=0.999; eps=1e-8
    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)
    t = 0
    beta_schedule = [...]
    # For each beta, run up to steps_per_beta steps
    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time()>deadline:
                return best_seq, best_val
            a = _softmax(v)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            t += 1
            m = beta1 * m + (1-beta1)*grad_v
            v_hat = beta2 * v_hat + (1-beta2)*(grad_v*grad_v)
            m_hat = m / (1 - beta1**t)
            v_hat_corr = v_hat / (1 - beta2**t)
            eta = 5.0 / math.sqrt(t)   # step size
            v = v - eta * m_hat / (np.sqrt(v_hat_corr) + eps)
            cur_val = _objective_surrogate(_softmax(v))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(v).copy()
    return best_seq, best_val

        j = rng.integers(0, n)

            prob = np.exp(-(val - cur_val) * 100) # small temperature
            if rng.random() < prob:
                cur_seq = new_seq
                cur_val = val
    return best_seq, best_val

"""
propose_candidate – hybrid random‑restart Adam optimiser for the max‑convolution
objective.

The routine
1️⃣ builds a collection of plausible initial step‑functions (exponential
   decays with several rates, a uniform shape, Dirichlet draws and the previously
   stored best sequence if any);
2️⃣ runs a short “exploration’’ phase: each starter is polished by a few
   Adam‑updates on the smooth log‑sum‑exp surrogate of max (a∗a).  The best
   survivor is retained;
3️⃣ launches a longer “exploitation’’ phase that continues the Adam optimisation
   from the current best vector with a finer schedule;
4️⃣ finishes with a cheap local‑transfer hill‑climb which moves a tiny amount of
   mass between two positions whenever it improves the exact surrogate
   (2·n·max(conv(a,a)));
5️⃣ returns the final sequence (ℓ₁‑normalised to 1, all entries ≥0) and stores it
   as ``height_sequence_1`` for the next run.

The optimisation works completely in the probability‑simplex via the
softmax‑parametrisation v↦softmax(v) and uses a schedule of increasing β
values to gradually turn the smooth surrogate into the true max.  Adam’s
adaptive steps together with the schedule tend to escape the shallow local
minima that the plain exponentiated‑gradient method left behind, giving a
sub‑0.5 % improvement on the benchmark target.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Soft‑max and surrogate objective (sum of entries = 1)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv(seq,seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic starter families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), i=0..n‑1, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Clip negatives and renormalise to ℓ₁‑norm = 1."""
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total == 0.0:
        return _uniform(len(seq))
    return seq / total


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_a: np.ndarray,
                   deadline: float,
                   steps_per_beta: int = 2000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate of max(conv(a,a)).
    Operates in the log‑softmax space to keep a on the simplex.
    Returns the best sequence seen and its exact surrogate value.
    """
    n = len(start_a)
    # initialise log‑probabilities
    v = np.log(start_a + 1e-30)

    # Adam moments
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # best‑so‑far bookkeeping
    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    # schedule of smoothing parameters – slowly morph to the true max
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
        2000.0, 5000.0, 20000.0, 50000.0,
        200000.0, 500000.0, 1_000_000.0
    ]

    t = 0  # global step counter

    for beta in beta_schedule:
        # run a bounded number of inner iterations for the current beta
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return best_seq, best_val

            a = _softmax(v)                     # current probability vector
            conv = _conv_fft(a, a)              # a∗a
            # smooth weights w_k = softmax(β·conv_k)
            shift = np.max(conv)                # numerical stabiliser
            exp_beta = np.exp(beta * (conv - shift))
            w = exp_beta / exp_beta.sum()

            # gradient w.r.t. a (see derivation in the description)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t. v (soft‑max Jacobian)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            # step size – similar decay to the original EG implementation
            eta = 5.0 / math.sqrt(t)

            v = v - eta * m_hat / (np.sqrt(v2_hat) + eps)

            # exact surrogate (no smoothing) – used for bookkeeping
            cur_val = _objective_surrogate(_softmax(v))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(v).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny local‑transfer hill‑climbing (exact surrogate)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray,
                  deadline: float,
                  steps: int = 2000) -> tuple[np.ndarray, float]:
    """
    Very cheap neighbourhood search: move a small fraction of mass from one
    position to another.  Accept improvements (and occasional uphill moves)
    based on the exact surrogate 2·n·max(conv).
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    for _ in range(steps):
        if time.time() > deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # transfer 1 % of the source mass (or the whole if tiny)
        amount = 0.01 * cur_seq[i]
        if amount <= 0.0:
            continue
        # ensure non‑negativity
        if cur_seq[i] < amount:
            continue
        ns = cur_seq.copy()
        ns[i] -= amount
        ns[j] += amount
        val = _objective_surrogate(ns)
        if val < cur_val:
            cur_seq, cur_val = ns, val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # very mild Metropolis acceptance to escape flat plateaux
            # temperature is deliberately tiny – we only need occasional jumps
            prob = math.exp(-(val - cur_val) * 200.0)
            if rng.random() < prob:
                cur_seq, cur_val = ns, val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the 1000‑second budget, stores the best sequence in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5               # safety margin

    # ------------------------------------------------------------------
    # Determine problem size (reuse the length of a previously stored best)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            seq_prev = np.asarray(prev, dtype=np.float64)
            n = seq_prev.shape[0]
        except Exception:
            seq_prev = None
            n = 4000
    else:
        seq_prev = None
        n = 4000

    # ------------------------------------------------------------------
    # Helper to assemble a pool of plausible starters
    # ------------------------------------------------------------------
    starters = []

    # 1) previously best (if available)
    if seq_prev is not None and seq_prev.shape[0] == n:
        starters.append(("prev_best", seq_prev.copy()))

    # 2) several exponential decays (different rates)
    for rate in (0.001, 0.002, 0.003, 0.004, 0.005, 0.007, 0.01):
        starters.append((f"exp_rate_{rate:.3f}", _exp_decay(n, rate)))

    # 3) uniform
    starters.append(("uniform", _uniform(n)))

    # 4) Dirichlet draws with a few concentration parameters
    for conc in (0.2, 0.5, 1.0, 2.0):
        raw = rng.gamma(shape=conc, scale=1.0, size=n)
        starters.append((f"dirichlet_{conc:.1f}", _normalize(raw)))

    # ------------------------------------------------------------------
    # Exploration phase – cheap Adam runs on every starter
    # ------------------------------------------------------------------
    explore_deadline = deadline - 250.0           # reserve ~4 min for deep polishing
    best_seq = None
    best_val = float("inf")
    best_name = None

    for name, seq in starters:
        if time.time() > explore_deadline:
            break
        seq = _normalize(seq)
        init_val = _objective_surrogate(seq)
        print(f"[Explore] starter {name:>15}  init surrogate = {init_val:.6f}")
        seq_opt, val_opt = _adam_optimize(seq, explore_deadline, steps_per_beta=400)
        print(f"[Explore] {name:>15}  after short Adam = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq, best_name = val_opt, seq_opt.copy(), name

    # ------------------------------------------------------------------
    # Randomised restarts while time permits (still in exploration window)
    # ------------------------------------------------------------------
    while time.time() < explore_deadline:
        # random choice between exponential and Dirichlet start
        if rng.random() < 0.5:
            rate = rng.uniform(0.001, 0.012)
            seq = _exp_decay(n, rate)
            name = f"rand_exp_{rate:.4f}"
        else:
            conc = rng.uniform(0.1, 3.0)
            raw = rng.gamma(shape=conc, scale=1.0, size=n)
            seq = _normalize(raw)
            name = f"rand_dir_{conc:.2f}"
        seq_opt, val_opt = _adam_optimize(seq, explore_deadline, steps_per_beta=400)
        if val_opt < best_val:
            best_val, best_seq, best_name = val_opt, seq_opt.copy(), name
            print(f"[Explore] new champion {best_name:>15}  surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Exploitation phase – long Adam run from the current champion
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _exp_decay(n, 0.003)
    print(f"[Exploit] starting long Adam from {best_name or 'fallback'}")
    seq_opt, val_opt = _adam_optimize(best_seq, deadline, steps_per_beta=2000)
    print(f"[Exploit] long Adam achieved surrogate = {val_opt:.6f}")
    if val_opt < best_val:
        best_val, best_seq = val_opt, seq_opt.copy()

    # ------------------------------------------------------------------
    # Final cheap local‑transfer polishing (if time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:

        if val_opt < best_val:

    # ------------------------------------------------------------------
    # Final evaluation via the provided (security‑checked) function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] final surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                       # pragma: no cover
        print("[Error] evaluate_sequence raised an exception:", exc)

    # Store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
