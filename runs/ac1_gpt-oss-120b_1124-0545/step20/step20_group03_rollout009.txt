def _exp_gradient_opt(start_seq, n, deadline, max_iters=500_000, eta0=5.0):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # compute gradient as per conv[max_idx] w.r.t a.
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        # gradient contribution 2*a[rev_idx]
        # For entries where idx_i == rev_idx (i.e., max_idx even and idx_i = max_idx/2), derivative should be a[rev_idx] not double, but ignore small.

"""
propose_candidate – multi‑shape initialization + focused mirror‑descent

The routine first builds a small library of candidate step‑functions (exponential,
power‑law, Gaussian, semi‑circular, symmetric exponential, two‑level, sinusoid
and the previous best if any).  Each candidate is normalised to ℓ₁‑norm = 1 and
its surrogate value 2·n·max(conv) is evaluated.  The best candidate is taken as
a start point and refined with three optimisation phases:

1. *Permutation annealing* – stochastic swaps & short‑segment reversals while
   respecting the simplex (the objective is evaluated on the exact max‑convolution).

2. *Exponentiated‑gradient mirror descent* – sub‑gradient descent on the exact
   max‑convolution (active index) in log‑softmax (simplex) coordinates.

3. *L‑BFGS smoothing* – a continuation method that minimises a log‑sum‑exp
   approximation of the max‑convolution, starting from a low smoothing
   parameter and ending at a very large β (≈10⁶).  This final polish often
   yields a modest extra reduction.

All phases respect the global 1000 s budget; the best sequence found is stored
in the global ``height_sequence_1`` for reuse in subsequent runs.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:       # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple surrogate objective (exact when ∑a = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) assuming the ℓ₁‑norm of seq equals 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _true_objective(seq: np.ndarray) -> float:
    """Exact value of evaluate_sequence (no external call)."""
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return float('inf')
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv) / (s ** 2))


# ----------------------------------------------------------------------
# Numerically stable softmax (used by exponentiated‑gradient)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Return a probability vector (sum = 1) from the log‑domain vector v."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Shape generators – each returns a non‑negative vector of length n,
# normalised to ℓ₁‑norm = 1.
# ----------------------------------------------------------------------
def _shape_exp_decay(n: int, rate: float) -> np.ndarray:
    a = np.exp(-rate * np.arange(n, dtype=np.float64))
    return a / a.sum()


def _shape_power_law(n: int, alpha: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-alpha))
    return a / a.sum()


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _shape_semicircle(n: int, R: float) -> np.ndarray:
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64) - mu
    a = np.sqrt(np.maximum(0.0, 1.0 - (i / R) ** 2))
    return a / a.sum()


def _shape_sinusoid(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1))
    return a / a.sum()


def _shape_sym_exponential(n: int, rate: float) -> np.ndarray:
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * np.abs(i - mu))
    return a / a.sum()


def _shape_two_step(n: int, p: float, ratio: float) -> np.ndarray:
    """First ⌈p·n⌉ entries have weight = ratio·B, the rest weight = B."""
    m = max(1, min(n - 1, int(round(p * n))))
    B = 1.0 / (m * ratio + (n - m))
    A = ratio * B
    a = np.full(n, B, dtype=np.float64)
    a[:m] = A
    return a / a.sum()


def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Permutation annealing (swap + short‑segment reverse moves)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 n: int,
                 deadline: float,
                 rng: np.random.Generator,
                 max_iters: int = 300_000):
    """
    Simulated‑annealing that re‑orders the entries of ``start_seq``.
    Only swaps and short segment reversals are used; the best sequence seen
    (lowest surrogate) is returned.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-8 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.6:                # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                 # short‑segment reversal
            i = rng.integers(0, n)
            max_len = min(50, n - i)
            seg_len = rng.integers(1, max_len + 1)
            j = i + seg_len - 1
            ns = cur_seq.copy()
            ns[i:j + 1] = ns[i:j + 1][::-1]

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution.  The variables live in the
    log‑softmax (simplex) space; the update uses the sub‑gradient of the
    currently active convolution entry.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of conv[max_idx] w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimises a smooth approximation of the max‑convolution using L‑BFGS‑B.
    Operates in log‑softmax coordinates to stay on the probability simplex.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp on the convolution vector
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            # convolution of rev_a (length n) with w (length 2n‑1) -> length 3n‑2,
            # the relevant slice is n‑1 … 2n‑2
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Refinement pipeline – runs all three phases within the given deadline.
# ----------------------------------------------------------------------
def _refine_sequence(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator):
    """
    Given a normalized start sequence, run permutation annealing, exponentiated
    gradient mirror‑descent and (if time permits) L‑BFGS smoothing.
    Returns the best sequence seen and its surrogate value.
    """
    # ----------- 1. Permutation annealing (re‑ordering only) -----------
    now = time.time()
    time_left = deadline - now
    if time_left > 30.0:
        perm_deadline = now + min(120.0, 0.25 * time_left)   # cap 2 min
        seq_perm, val_perm = _perm_anneal(start_seq, n, perm_deadline,
                                          rng, max_iters=300_000)
    else:
        seq_perm, val_perm = start_seq.copy(), _objective_surrogate(start_seq)

    # ----------- 2. Exponentiated‑gradient polishing --------------------
    now = time.time()
    time_left = deadline - now
    if time_left > 30.0:
        eg_deadline = now + min(300.0, 0.55 * time_left)   # cap 5 min
        seq_eg, val_eg = _exp_gradient_opt(seq_perm, n, eg_deadline,
                                           max_iters=2_000_000, eta0=5.0)
    else:
        seq_eg, val_eg = seq_perm, val_perm

    # ----------- 3. Optional L‑BFGS smooth optimisation -----------------
    now = time.time()
    if time.time() < deadline - 5.0 and _opt is not None and (deadline - now) > 5.0:
        seq_lb = _lbfgs_smooth_optimize(seq_eg, n, deadline,
                                         max_iter_per_beta=4000)
        val_lb = _objective_surrogate(seq_lb)
        if val_lb < val_eg:
            best_seq, best_val = seq_lb, val_lb
        else:
            best_seq, best_val = seq_eg, val_eg
    else:
        best_seq, best_val = seq_eg, val_eg

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The strategy combines a diverse family of analytical initial guesses with
    a focused optimisation pipeline (permutation annealing → EG → L‑BFGS).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # 0. Problem dimension – reuse previous length if available.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            n = int(len(prev))
        except Exception:
            n = 4000
    else:
        n = 4000

    # --------------------------------------------------------------
    # 1. Build a modest set of analytical initial candidates.
    # --------------------------------------------------------------
    candidates = []

    # 1a) existing solution (if any)
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                s = arr.sum()
                if s > 0.0:
                    arr = arr / s
                    candidates.append(arr)
        except Exception:
            pass

    # 1b) pure uniform (baseline)
    candidates.append(_shape_uniform(n))

    # 1c) sinusoid (zero at the ends)
    candidates.append(_shape_sinusoid(n))

    # 1d) random exponential decays
    for _ in range(8):
        rate = 10 ** rng.uniform(-5.0, -2.0)          # 1e‑5 … 1e‑2
        candidates.append(_shape_exp_decay(n, rate))

    # 1e) random power‑law decays
    for _ in range(8):
        alpha = rng.uniform(0.2, 2.0)
        candidates.append(_shape_power_law(n, alpha))

    # 1f) random centred Gaussians
    for _ in range(8):
        sigma = rng.uniform(n / 20.0, n / 2.0)
        candidates.append(_shape_gaussian(n, sigma))

    # 1g) semi‑circular profiles
    for _ in range(4):
        R = rng.uniform(n / 4.0, n / 2.0)
        candidates.append(_shape_semicircle(n, R))

    # 1h) symmetric exponential (peak in the centre)
    for _ in range(4):
        rate = 10 ** rng.uniform(-5.0, -2.0)
        candidates.append(_shape_sym_exponential(n, rate))

    # 1i) two‑step (high plateau + low background)
    for _ in range(4):
        p = rng.uniform(0.1, 0.9)          # fraction of high‑weight entries
        ratio = rng.uniform(2.0, 10.0)     # how much higher they are
        candidates.append(_shape_two_step(n, p, ratio))

    # --------------------------------------------------------------
    # 2. Score all candidates with the exact surrogate (ℓ₁‑norm = 1).
    # --------------------------------------------------------------
    scored = []
    for idx, seq in enumerate(candidates):
        val = _objective_surrogate(seq)
        scored.append((val, seq))
        print(f"[Init] candidate {idx:02d} surrogate = {val:.6f}")

    scored.sort(key=lambda x: x[0])   # ascending (lower is better)

    # --------------------------------------------------------------
    # 3. Take the best (lowest surrogate) candidate as the start point.
    # --------------------------------------------------------------
    best_start_val, best_start_seq = scored[0]
    print(f"[BestInit] lowest initial surrogate = {best_start_val:.6f}")

    # --------------------------------------------------------------
    # 4. Refine the best start using the full optimisation pipeline.
    # --------------------------------------------------------------
    refined_seq, refined_val = _refine_sequence(best_start_seq, n, deadline, rng)
    print(f"[Refine] surrogate after full pipeline = {refined_val:.6f}")

    # --------------------------------------------------------------
    # 5. If time permits, attempt a second refinement on the next best
    #    initial candidate (often yields a different local basin).
    # --------------------------------------------------------------
    now = time.time()
    if now < deadline - 10.0 and len(scored) > 1:
        sec_val, sec_seq = scored[1]
        print(f"[Secondary] second init surrogate = {sec_val:.6f}")
        sec_ref_seq, sec_ref_val = _refine_sequence(sec_seq, n, deadline, rng)
        print(f"[Secondary] surrogate after refinement = {sec_ref_val:.6f}")
        if sec_ref_val < refined_val:
            refined_seq, refined_val = sec_ref_seq, sec_ref_val

    # --------------------------------------------------------------
    # 6. Final evaluation via the provided function (for logging only).
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(refined_seq.tolist())
        print(f"[Result] final surrogate = {refined_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # --------------------------------------------------------------
    # 7. Store for the next run (facilitates incremental improvement).
    # --------------------------------------------------------------
    globals()["height_sequence_1"] = refined_seq.tolist()
    return refined_seq.tolist()
