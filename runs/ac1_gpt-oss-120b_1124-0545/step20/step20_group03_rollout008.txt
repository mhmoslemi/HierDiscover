def solve_lp_for_g(f, M):

"""
propose_candidate – multi‑restart momentum exponentiated‑gradient + L‑BFGS polishing

The optimiser works on the scale‑invariant formulation
    Φ(a) = 2·n·max(conv(a,a)),   a ≥ 0,   Σ a = 1 .
Because the true objective of ``evaluate_sequence`` is Φ(a) (the ℓ₁‑norm
is fixed to 1), we minimise Φ directly.

Steps
-------
1. Load the previous best sequence (global ``height_sequence_1``).  If it is
   missing or has a wrong length, fall back to a simple exponential decay.
2. Normalise the start vector to ℓ₁‑norm = 1 and compute the surrogate value.
3. Run a *momentum* exponentiated‑gradient (mirror‑descent) optimiser on the
   exact non‑smooth objective.  The gradient is the sub‑gradient of the
   active convolution entry.  Momentum (β=0.9 by default) and a
   1/√t step‑size give faster convergence than the plain scheme used
   previously.
4. Polish the result with a highly‑smoothed L‑BFGS optimiser that
   approximates the max by a log‑sum‑exp with a schedule of increasing
   β (up to 10⁶) and many internal iterations.
5. Perform many cheap random restarts (different families of initial
   distributions) while time permits.  Each restart receives a short
   EG polish (≈ 3·10⁵ updates) and a quick L‑BFGS pass; the best sequence
   found replaces the current global best.
6. After the restart phase, run a *long* EG polishing (≈ 2·10⁶ updates)
   followed by a final, thorough L‑BFGS optimisation.
7. Evaluate the final sequence with the provided ``evaluate_sequence``
   and store it in ``height_sequence_1`` for the next run.

All heavy work respects the 1000 s budget; the routine stops as soon as
the deadline is reached and returns the best sequence discovered.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:       # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact max‑convolution, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Φ(seq) = 2·n·max(conv(seq,seq)) for Σseq=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax utilities (mirror‑descent on the simplex)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Momentum exponentiated‑gradient on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_grad_momentum(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 2_000_000,
                       eta0: float = 10.0,
                       momentum: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with momentum on Φ(seq) = 2·n·max(conv(seq,seq)).
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # logits (always feasible)
    vel = np.zeros_like(a)            # momentum accumulator
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current convolution and its maximal index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry: g_j = 2·a_{max_idx‑j}
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[rev_idx]

        # momentum update (exponential moving average)
        vel = momentum * vel + (1.0 - momentum) * g

        # step‑size schedule η_t = η₀ / √t
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * vel
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Fast, simple EG without momentum (used for cheap random restarts)
# ----------------------------------------------------------------------
def _exp_grad_plain(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 300_000,
                    eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Plain exponentiated‑gradient (no momentum) – cheap polishing."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g = np.zeros_like(a)
        g[idx] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    Optimise the log‑sum‑exp smoothed version of Φ using L‑BFGS‑B.
    Operates in log‑softmax space to keep feasibility.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0, 500_000.0,
            1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (before the simplex projection)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # project the gradient onto the tangent space of the simplex
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Families for random initialisations (all returned already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _init_exponential(rng: np.random.Generator, n: int) -> np.ndarray:
    rate = rng.uniform(0.001, 0.02)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _init_powerlaw(rng: np.random.Generator, n: int) -> np.ndarray:
    p = rng.uniform(0.5, 2.5)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    return a / a.sum()


def _init_random_sorted(rng: np.random.Generator, n: int) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    a.sort()
    a = a[::-1]                     # descending order
    return a / a.sum()


def _init_random_uniform(rng: np.random.Generator, n: int) -> np.ndarray:
    a = rng.random(size=n)
    return a / a.sum()
