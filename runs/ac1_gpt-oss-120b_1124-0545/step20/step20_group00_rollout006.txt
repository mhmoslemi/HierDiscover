"""
Propose a candidate step‑function that beats the previous bound (~1.5085).

Algorithm
---------

1. Start from the best known sequence (the global variable
   ``height_sequence_1``) or from a uniform distribution.
   The sequence is normalised to have sum = 1 (the objective is
   scale‑invariant).

2. Repeatedly solve the *linear‑program* that for the current
   function f maximises  Σ g_j  subject to  (f∗g)_k ≤ M,
   where  M = max(f∗f).  The LP is solved with CVXPY/ECOS.
   The optimal g has a larger ℓ₁‑norm; after scaling it to the
   original ℓ₁‑norm we obtain a new candidate f′.
   If f′ improves the surrogate objective we accept it,
   otherwise we take a convex combination (1‑t) f + t f′.

3. After each successful LP step a short *exponentiated‑gradient*
   (mirror‑descent) polishing is performed to further reduce the
   true max‑convolution.

4. When the time budget is almost exhausted a final L‑BFGS pass
   on a smoothed “log‑sum‑exp” max‑convolution objective cleans up
   any remaining bias.

All steps respect the global 1000 s budget.  The best sequence found
is stored in ``height_sequence_1`` and returned as a plain list.
"""

import time
import math
import numpy as np

# optional imports
try:
    import cvxpy as cp            # LP solver
except Exception:                # pragma: no cover
    cp = None

try:
    import scipy.optimize as _opt
except Exception:                # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact because we keep sum(seq)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the true max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space.
    """
    if _opt is None:
        # SciPy not available – return the input unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
            2000.0, 5000.0, 20000.0, 50000.0, 100000.0,
            200000.0, 500000.0, 1000000.0,
        ]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient w.r.t a
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Linear programme: maximise Σ g   s.t.  (f∗g) ≤ M
# ----------------------------------------------------------------------
def _lp_max_sum(f: np.ndarray, M: float):
    """
    Solve  max Σ g_j  subject to g≥0 and conv(f,g) ≤ M (elementwise).

    Returns the optimal g as a NumPy array, or None if the problem
    is infeasible/unbounded.
    """
    if cp is None:                # pragma: no cover
        raise RuntimeError("CVXPY is not available")

    n = f.shape[0]
    g = cp.Variable(n)
    f_param = cp.Parameter(n, value=f, constant=True)
    conv_fg = cp.conv(f_param, g)          # length 2n‑1

    constraints = [g >= 0, conv_fg <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS tolerances are generous enough for our purpose.
    prob.solve(solver=cp.ECOS,
               warm_start=True,
               max_iters=200_000,
               abstol=1e-8,
               reltol=1e-8,
               feastol=1e-9)

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    return np.asarray(g.value, dtype=np.float64)

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found (list of floats)
    and stores it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    n = 4000                                        # length used previously

    # ------------------------------------------------------------------
    # Load previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.shape[0] == n:
            f = prev_arr.copy()
        else:
            f = None
    else:
        f = None

    if f is None:
        # fallback to uniform distribution
        f = np.full(n, 1.0 / n, dtype=np.float64)

    # Ensure strict positivity (helps the LP) and normalise sum to 1
    eps = 1e-12
    f = np.maximum(f, eps)
    f = f / f.sum()

    best_seq = f.copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] surrogate = {best_val:.6f}")

    iteration = 0
    no_improve_cnt = 0

    while time.time() < deadline and iteration < 10:
        iteration += 1
        # --------------------------------------------------------------
        # 1) compute current bound M = max(conv(f,f))
        # --------------------------------------------------------------
        conv_f = np.convolve(best_seq, best_seq)
        M = float(conv_f.max())
        print(f"\n--- Iteration {iteration} ---")
        print(f"Current max‑conv M = {M:.6e}")

        # --------------------------------------------------------------
        # 2) linear programme to find g with larger ℓ₁‑norm
        # --------------------------------------------------------------
        try:
            g_opt = _lp_max_sum(best_seq, M)
        except Exception as e:  # pragma: no cover
            print(f"[LP] exception: {e}")
            break

        if g_opt is None:
            print("[LP] No feasible solution (or unbounded). Stopping.")
            break

        sum_f = best_seq.sum()
        sum_g = g_opt.sum()
        if sum_g <= eps:
            print("[LP] Obtained g has near‑zero mass – abandoning.")
            break

        scaling = sum_f / sum_g          # < 1 if LP succeeded
        g_scaled = g_opt * scaling
        # enforce strict positivity and renormalise (numerics)
        g_scaled = np.maximum(g_scaled, eps)
        g_scaled = g_scaled / g_scaled.sum() * sum_f

        # --------------------------------------------------------------
        # 3) evaluate direct replacement
        # --------------------------------------------------------------
        val_direct = _objective_surrogate(g_scaled)
        print(f"LP gave sum(g) = {sum_g:.6f}, scaling = {scaling:.6f}")
        print(f"Direct‑replace surrogate = {val_direct:.6f}")

        improved = False
        if val_direct < best_val * 0.9999:
            best_seq = g_scaled
            best_val = val_direct
            improved = True
            print(f">>> Direct replacement improved to {best_val:.6f}")
        else:
            # try a convex combination (t chosen heuristically)
            t = 0.35
            combo = (1.0 - t) * best_seq + t * g_scaled
            combo = np.maximum(combo, eps)
            combo = combo / combo.sum() * sum_f
            val_combo = _objective_surrogate(combo)
            print(f"Convex‑combo surrogate = {val_combo:.6f}")

            if val_combo < best_val:
                best_seq = combo
                best_val = val_combo
                improved = True
                print(f">>> Combo improved to {best_val:.6f}")

        # --------------------------------------------------------------
        # 4) short exponentiated‑gradient polishing after any improvement
        # --------------------------------------------------------------
        if improved:
            no_improve_cnt = 0
            remaining = deadline - time.time()
            # allocate at most a few hundred‑thousand iterations
            max_iters = int(min(300_000, 80_000 * remaining / (budget_s)))
            seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                               deadline,
                                               max_iters=max_iters,
                                               eta0=5.0)
            if val_eg < best_val:
                best_seq = seq_eg
                best_val = val_eg
                print(f"[EG] polished improved to {best_val:.6f}")
        else:
            no_improve_cnt += 1
            if no_improve_cnt >= 3:
                print("No improvement for several iterations – terminating.")
                break

    # ------------------------------------------------------------------
    # 5) Final L‑BFGS smoothing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("\n--- Final L‑BFGS smoothing ---")
        final_deadline = deadline
        seq_final = _lbfgs_smooth_optimize(best_seq, n,
                                            final_deadline,
                                            beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                           200.0, 500.0, 1000.0,
                                                           2000.0, 5000.0],
                                            max_iter_per_beta=4000)
        val_final = _objective_surrogate(seq_final)
        print(f"[Final L‑BFGS] surrogate = {val_final:.6f}")
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f">>> Updated after final L‑BFGS to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Official evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
