def _local_transfer_hill(seq, n, deadline, rng, max_iters):
    cur_seq = seq.copy()
    cur_val = _objective_surrogate(cur_seq)

def _adam_smooth_opt(start_seq, n, deadline, betas_schedule, lr=0.1, max_iter=500000):
    a = start_seq.copy()
    v = np.log(a + 1e-30)
    m = np.zeros_like(v)
    v_t = np.zeros_like(v)
    beta1=0.9; beta2=0.999; epsilon=1e-8
    t=0
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for beta in betas_schedule:
        while time.time()<deadline:
            # compute conv, smoothed grad
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_b = np.exp(beta * (conv - shift))
            sum_exp = exp_b.sum()
            # weight distribution
            w = exp_b / sum_exp   # shape 2n-1
            # gradient w.r.t a: 2 * convolution of a reversed with w (same as L-BFGS)
            rev_a = a[::-1]
            full_grad = 2.0 * _conv_fft(rev_a, w)  # length 3n-2
            grad_a = full_grad[n-1:n-1+n]   # n elements aligned
            # map to log domain:
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            # Adam update
            t += 1
            m = beta1 * m + (1-beta1) * grad_v
            v_t = beta2 * v_t + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1**t)
            v_hat = v_t / (1 - beta2**t)
            v = v - lr * m_hat / (np.sqrt(v_hat) + epsilon)
            a = _softmax(v)

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
            # maybe break out after some iterations per beta
            if t % 5000 == 0:
                if time.time() > deadline: break
    return best_seq, best_val

    beta2=0.999
    eps=1e-8

"""
Proposes a non‑negative step‑function that (approximately) minimises
``evaluate_sequence``.  The method works in three stages:

1. **Initialisation** – start from the previously found best sequence
   (global variable ``height_sequence_1``) or from a uniform distribution.
   The vector is normalised to sum = 1.

2. **Smooth‑max Adam optimisation** – the exact max‑convolution is
   approximated by a log‑sum‑exp (soft‑max) with a sequence of increasing
   β‑values.  The optimisation runs in log‑softmax space with the Adam
   optimiser, keeping the best iterate seen.

3. **Local transfer hill‑climbing** – a simple coordinate‑wise move
   transfers a small amount of mass from one index to another; moves that
   improve the surrogate objective are kept.  A few short random restarts
   (perturb → Adam → hill‑climb) are performed while time permits.

The whole procedure respects the overall time budget (default 1000 s) and
stores the best sequence in the global variable ``height_sequence_1`` for
future continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ → simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (assumes the vector already sums to 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) for sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed (log‑sum‑exp) max‑convolution.
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     betas=None,
                     lr: float = 0.05,
                     max_iters_per_beta: int = 20000):
    """
    Adam optimiser on the smoothed max‑convolution.
    Works in log‑softmax space (variables = log a_i) and returns the best
    sequence seen together with its surrogate value.
    """
    if betas is None:
        # a modest schedule – starts smooth, ends very sharp
        betas = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                 1000.0, 2000.0, 5000.0]

    a = start_seq.copy()
    # log‑softmax variables (always keep them finite)
    v = np.log(a + 1e-30)

    m = np.zeros_like(v)          # first moment
    v_sq = np.zeros_like(v)       # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for beta in betas:
        # run a fixed number of iterations for each β, but stop early
        for _ in range(max_iters_per_beta):
            if time.time() > deadline:
                return best_seq, best_val

            # ----- smoothed max (log‑sum‑exp) -----
            conv = _conv_fft(a, a)               # length = 2n‑1
            max_c = np.max(conv)                 # shift for stability
            exp_beta = np.exp(beta * (conv - max_c))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                # weight vector on conv entries

            # ----- gradient wrt a (exact for the smoothed objective) -----
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # n‑dimensional gradient

            # map to gradient in log‑softmax space
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ----- Adam update -----
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_sq = beta2 * v_sq + (1.0 - beta2) * (grad_v * grad_v)

            m_hat = m / (1.0 - beta1 ** t)
            v_sq_hat = v_sq / (1.0 - beta2 ** t)

            v = v - lr * m_hat / (np.sqrt(v_sq_hat) + eps)
            a = _softmax(v)                      # stays on the simplex

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple transfer‑based hill climbing (mass moved from i → j)
# ----------------------------------------------------------------------
def _transfer_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly transfers a random fraction of the mass at
    one position to another position.  Only improving moves are accepted.
    A few short random restarts are performed later in ``propose_candidate``.
    """
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)

    best = cur.copy()
    best_val = cur_val

    for it in range(max_iters):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if cur[i] <= 1e-12:
            continue

        # move a random fraction (up to 50 %) of the mass at i to j
        frac = rng.random() * 0.5
        delta = frac * cur[i]
        if delta <= 0.0:
            continue

        ns = cur.copy()
        ns[i] -= delta
        ns[j] += delta

        val = _objective_surrogate(ns)
        if val < cur_val:
            cur = ns
            cur_val = val
            if val < best_val:
                best = ns.copy()
                best_val = val

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine:
      * loads the previously found best sequence (if any) or falls back to
        a uniform distribution,
      * refines it with a smooth‑max Adam optimiser,
      * applies a transfer‑based hill‑climber,
      * performs a few short random‑restart cycles (perturb → Adam →
        hill‑climb) while time permits,
      * stores the best sequence globally as ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    t_start = time.time()
    # give a small safety margin so we never exceed the time budget
    deadline = t_start + budget_s - 2.0

    # --------------------------------------------------------------
    # Problem dimension – keep the same size used in earlier runs
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous best if it exists; otherwise start from uniform
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        seq = np.asarray(prev, dtype=np.float64)
    else:
        seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ensure feasibility
    seq = np.maximum(seq, 0.0)
    tot = seq.sum()
    if tot <= 0.0:
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq = seq / tot

    print("\n[Init] surrogate = {:.6f}".format(_objective_surrogate(seq)))

    # --------------------------------------------------------------
    # 1. Adam optimisation on smoothed max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _adam_smooth_opt(seq, n, deadline,
                                    betas=[1.0, 5.0, 20.0, 80.0, 200.0,
                                           500.0, 1000.0, 2000.0, 5000.0],
                                    lr=0.05,
                                    max_iters_per_beta=20000)
        print("[Adam] best surrogate = {:.6f}".format(val))

    # --------------------------------------------------------------
    # 2. Transfer‑based hill climbing
    # --------------------------------------------------------------
    if time.time() < deadline:
        seq, val = _transfer_hill_climb(seq, n, deadline, rng,
                                        max_iters=200_000)
        print("[HillClimb] best surrogate = {:.6f}".format(val))

    # --------------------------------------------------------------
    # 3. A few short random‑restart cycles (perturb → Adam → hill‑climb)
    # --------------------------------------------------------------
    restart_budget = max(0.0, deadline - time.time())
    # allocate at most 5 restarts or until time runs out
    for restart_idx in range(5):
        if time.time() >= deadline:
            break
        # give each restart a small slice of the remaining budget
        sub_deadline = time.time() + min(30.0, restart_budget / (5 - restart_idx))
        # random perturbation of the current best
        pert = seq + 0.02 * rng.normal(size=seq.shape)
        pert = np.maximum(pert, 0.0)
        s = pert.sum()
        if s == 0.0:
            continue
        pert = pert / s
        # short Adam run
        pert, val_pert = _adam_smooth_opt(pert, n, sub_deadline,
                                          betas=[200.0, 500.0, 1000.0],
                                          lr=0.03,
                                          max_iters_per_beta=5000)
        # short hill‑climb
        pert, val_pert = _transfer_hill_climb(pert, n, sub_deadline, rng,
                                              max_iters=50_000)
        if val_pert < _objective_surrogate(seq):
            seq = pert
            print("[Restart {}] improved surrogate = {:.6f}".format(restart_idx + 1,
                                                                    val_pert))

    # --------------------------------------------------------------
    # 4. Final official evaluation
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(seq.tolist())
        print("\n=== Finished ===\nSurrogate ≈ {:.6f}, true evaluate_sequence = {:.6f}"
              .format(_objective_surrogate(seq), true_val))
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible continuation runs
    globals()["height_sequence_1"] = seq.tolist()
    return seq.tolist()
