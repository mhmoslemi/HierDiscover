    v = np.log(a0 + 1e-30)
    best_a = a0.copy()
    best_val = _objective_surrogate(a0)
    t = 0
    for beta in betas:
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            t += 1
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = conv.max()
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

            # compute grad_a as described
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]

            # gradient w.r.t v
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # step
            eta = step0 / np.sqrt(t)
            v = v - eta * grad_v

            # evaluate surrogate (max conv)
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
    return best_a, best_val

"""
Propose candidate step function that minimizes evaluate_sequence.
The algorithm:
- Build a set of seed distributions (uniform, upscaled previous solution,
  beta-shaped families, Dirichlet, perturbed forms).
- Optionally run a crude global optimisation over Gaussian mixtures using
  differential evolution (if SciPy is available) or random sampling.
- Refine using a smooth max approximation with a schedule of increasing
  β. Gradient descent (in log‑softmax space) runs many iterations,
  employing an Adam‑style step‑size schedule for stability.
- Perform an exact exponentiated‑gradient (mirror‑descent) stage using the
  true max‑convolution sub‑gradient, with a very large number of iterations
  (bounded only by the remaining time budget).
- Finally apply a simple stochastic local‑search that moves a small amount of
  probability mass between random coordinates; this often removes the last
  remaining plateau.
The best sequence seen is stored in ``height_sequence_1`` and returned.
"""

import time
import math
import numpy as np

try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# --- basic utilities -------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# --- surrogate objective (depends only on shape, sum = 1)-------------
def _objective_surrogate(p: np.ndarray) -> float:
    """Return 2·n·max(conv(p,p)) for a probability vector p."""
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))

# --- seed families ---------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α+1,α+1) density on {0,…,n‑1}, normalised."""
    # use shifted index to avoid zero
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n  # in (0,1)
    # avoid division by zero for very negative alpha by adding a tiny epsilon
    eps = 1e-12
    shape = (np.maximum(x, eps) ** alpha) * (np.maximum(1.0 - x, eps) ** alpha)
    s = shape.sum()
    if s <= 0.0:
        return _uniform(n)
    return shape / s

def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing probability vector to length new_n."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    s = new_seq.sum()
    if s <= 0.0:
        return _uniform(new_n)
    return new_seq / s

# --- optimisation helpers --------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000):
    """
    Optimise the smoothed max‑convolution objective with L‑BFGS‑B.
    Variables live in log‑softmax space.
    """
    if _opt is None:
        # SciPy not available – skip this phase.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0,
                         2000000.0, 5000000.0, 10000000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # ---- gradient ----
            w = exp_beta / sum_exp                     # softmax over conv entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated gradient) optimiser for the exact max.
    Operates on probability vectors (sum = 1). Returns best seen vector.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

def _local_search(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  max_trials: int = 500_000,
                  eps: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing that moves a tiny amount of probability mass
    from a randomly chosen index to another one.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for t in range(max_trials):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Move a fraction eps of mass from i to j (if possible)
        delta = eps * a[i]
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # renormalise is not needed because we transferred mass
        val = _objective_surrogate(a_new)
        if val < best_val:
            best_val = val
            best_seq = a_new
            a = a_new
    return best_seq, best_val

"""
Propose a step‑function that (hopefully) beats the current best bound.
Algorithm
---------

1. **Seed pool** – uniform, the previous solution (if any), several Dirichlet
   draws, a handful of “U‑shaped’’ β‑distribution seeds (negative exponent)
   and tiny perturbations of the uniform vector.

2. **Random Gaussian‑mixture search** – generate many random mixtures of a
   few Gaussian windows (the parameters are drawn uniformly).  The best
   mixture from this cheap search is added to the pool.

3. **Smooth L‑BFGS refinement** – minimise a smooth approximation of the
   max‑convolution (log‑sum‑exp) in log‑softmax space.  The schedule of β
   values reaches up to 1e7, and each β is given a few thousand L‑BFGS‑B
   iterations.  This quickly drives the surrogate objective very close to
   the true optimum.

4. **Exact exponentiated‑gradient polishing** – a mirror‑descent (EG)
   optimiser works directly on the true max‑convolution.  It runs for as
   many iterations as the remaining time budget allows, using a decreasing
   step size η = η₀/√t.

5. **Local stochastic refinement** – tiny amounts of probability mass are
   moved between random coordinates; improvements are kept.  This simple
   hill‑climbing often removes the last remaining plateau.

All stages respect the overall 1000 s budget.  The best sequence seen is
stored in the global variable ``height_sequence_1`` and returned as a plain
list of floats.
"""

    L = len(x) + len(y) - 1

    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (the quantity we actually minimise)
# ----------------------------------------------------------------------
def _objective_surrogate(p: np.ndarray) -> float:
    """
    For a probability vector p (sum = 1) return
        2·n·max(conv(p,p))
    which is exactly the value of ``evaluate_sequence`` after the
    normalisation step.
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α+1,α+1)‐like shape on a discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5               # avoid exact 0/1
    x = i / n                                             # in (0,1)
    eps = 1e-12
    shape = (np.maximum(x, eps) ** alpha) * (np.maximum(1.0 - x, eps) ** alpha)
    s = shape.sum()
    if s <= 0.0:
        return _uniform(n)
    return shape / s


def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing probability vector to length new_n."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    s = new_seq.sum()
    if s <= 0.0:
        return _uniform(new_n)
    return new_seq / s


# ----------------------------------------------------------------------
# Gaussian mixture builder (used in the random global search)
# ----------------------------------------------------------------------
def _mixture_gaussian(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a probability vector from a set of 5 Gaussian components.
    params layout (K = 5):
        w_0 … w_4   – non‑negative weights (renormalised internally)
        μ_0 … μ_4   – means in [0,1]
        σ_0 … σ_4   – standard deviations in [0.001,0.5]
    """
    K = 5
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    w_sum = w.sum()
    if w_sum > 0.0:
        w = w / w_sum
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)          # points in [0,1]
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Smooth L‑BFGS optimisation (log‑softmax variables)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    Optimise a log‑sum‑exp smooth approximation of the max‑convolution.
    The optimisation runs in log‑softmax space; the returned vector is a
    probability distribution (sum = 1).
    """
    if _opt is None:          # SciPy missing – skip this phase
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
            2000.0, 5000.0, 20000.0, 50000.0, 100000.0,
            200000.0, 500000.0, 1000000.0, 2000000.0,
            5000000.0, 10000000.0
        ]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # abort if we ran out of time
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (mirror‑descent) optimiser on the true max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best vector seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny stochastic local search (mass shuffling)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  rng: np.random.Generator,
                  max_trials: int = 800_000,
                  eps: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Randomly move a small amount of probability mass from one index to another.
    Accepts the move only if it improves the surrogate objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for t in range(max_trials):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = eps * a[i]                # tiny chunk taken from i
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        val = _objective_surrogate(a_new)
        if val < best_val:
            best_val = val
            best_seq = a_new
            a = a_new
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (list of floats) and stores it in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 points is a good compromise between resolution
    # and speed.  The algorithm automatically short‑circuits if the time
    # budget is exhausted at any stage.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load any previously stored solution and resize it to the current n.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = _resize_sequence(prev_arr, n)
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1️⃣  Build a diverse pool of seed vectors (all ℓ₁‑normalised)
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if prev_arr is not None and prev_arr.sum() > 0.0:
        seeds.append(prev_arr / prev_arr.sum())

    # Dirichlet draws with a few different concentration parameters
    for conc in (0.05, 0.2, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Small perturbations of the uniform vector
    for _ in range(3):
        seeds.append(_perturb(_uniform(n), rng, sigma=0.004))

    # β‑distribution (U‑shaped) seeds with negative exponents
    for alpha in (-0.2, -0.4, -0.6, -0.8):
        seeds.append(_beta_shape(n, alpha))

    # ------------------------------------------------------------------
    # 2️⃣  Evaluate the seed pool and keep the best one.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    print("\n--- Evaluating initial seeds ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        print(f"seed #{idx + 1:2d} surrogate = {val:.8f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.8f}")

    # ------------------------------------------------------------------
    # 3️⃣  Random Gaussian‑mixture search (cheap global exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Random Gaussian‑mixture sampling ---")
        trials = 5000
        K = 5
        for t in range(trials):
            if time.time() > deadline:
                break
            rnd = np.concatenate([
                rng.random(K),                     # weights
                rng.random(K),                     # means
                rng.uniform(0.001, 0.5, K)         # sigmas
            ])
            seq = _mixture_gaussian(rnd, n)
            val = _objective_surrogate(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Mix] new best surrogate = {best_val:.8f} (iter {t})")

    # ------------------------------------------------------------------
    # 4️⃣  Smooth L‑BFGS refinement (log‑sum‑exp smoothing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- L‑BFGS smoothing of current best ---")
        # allocate about 35 % of the remaining budget to this phase
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.35

        seq_lbfgs = _lbfgs_smooth_optimize(best_seq, n, sub_deadline,
                                           beta_schedule=[
                                               1.0, 5.0, 20.0, 80.0,
                                               200.0, 500.0, 1000.0,
                                               2000.0, 5000.0,
                                               20000.0, 50000.0,
                                               100000.0, 200000.0,
                                               500000.0, 1000000.0,
                                               2000000.0, 5000000.0,
                                               10000000.0],
                                           max_iter_per_beta=8000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.8f}")
        if val_lbfgs < best_val:
            best_val = val_lbfgs
            best_seq = seq_lbfgs.copy()
            print(f"  >>> Updated global best = {best_val:.8f}")

    # ------------------------------------------------------------------
    # 5️⃣  Exact exponentiated‑gradient polishing (sub‑gradient on true max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing ---")
        polishing_deadline = deadline - 5.0
        # Rough estimate: one EG iteration costs ≈0.8 ms
        time_per_iter = 0.0008
        remaining = polishing_deadline - time.time()
        max_iters = max(200_000, int(remaining / time_per_iter))
        max_iters = min(max_iters, 5_000_000)

        seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                           polishing_deadline,
                                           max_iters=max_iters,
                                           eta0=5.0)
        print(f"[EG] surrogate = {val_eg:.8f}")
        if val_eg < best_val:
            best_val = val_eg
            best_seq = seq_eg.copy()
            print(f"  >>> Updated global best = {best_val:.8f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final tiny stochastic local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Local stochastic refinement ---")
        local_deadline = deadline - 2.0
        seq_loc, val_loc = _local_search(best_seq, n,
                                         local_deadline,
                                         rng,
                                         max_trials=800_000,
                                         eps=1e-5)
        print(f"[Local] surrogate = {val_loc:.8f}")
        if val_loc < best_val:
            best_val = val_loc
            best_seq = seq_loc.copy()
            print(f"  >>> Updated global best = {best_val:.8f}")

    # ------------------------------------------------------------------
    # 7️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.8f}, evaluate_sequence = {true_val:.8f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store the result for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
