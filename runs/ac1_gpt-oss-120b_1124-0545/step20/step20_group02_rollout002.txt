"""
Propose a step‑function (height sequence) that minimises
    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

Algorithm
---------
1.  Build a diverse set of *seed* sequences using several analytic families
    (uniform, sine‑window, Gaussian, exponential decay, Beta‑shape, Dirichlet,
    random exponential, low‑frequency random curves).  The best few (by the
    exact surrogate 2·n·max(conv) with sum = 1) are kept.

2.  For each kept seed run a **mirror‑descent / exponentiated‑gradient**
    optimisation on the *exact* max‑convolution objective.  The update
    works in log‑probability space, guaranteeing non‑negativity and
    ℓ₁‑normalisation automatically.  A sub‑gradient is assembled from all
    (near‑maximal) active convolution indices.

3.  The best polished sequence is refined once more with a longer
    exponentiated‑gradient run.  If time permits a short simulated‑annealing
    fine‑tune is performed.

4.  The final sequence is evaluated with the provided ``evaluate_sequence``
    function, stored in the global variable ``height_sequence_1`` and
    returned.

The whole routine respects the global time‑budget (default 1000 s) and
adapts the number of optimisation iterations to the remaining time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate objective (sum = 1 ⇒ surrogate = true objective)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) for a vector with sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space; this automatically enforces positivity & ℓ₁‑norm
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        # time check – break early if budget exhausted
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # active set: indices whose value is within a tiny relative tolerance
        eps = max_val * 1e-6
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient accumulates contributions from every active index
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (optional)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 150_000,
                         start_sigma: float = 0.001,
                         T_start: float = 5e-3,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Lightweight SA: Gaussian perturbations with linearly decreasing temperature.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

    # deterministic baselines
    _add_candidate(_uniform(n), "uniform")
    _add_candidate(_sine_window(n, gamma=1.2), "sine_window")
    _add_candidate(_gaussian(n, sigma_frac=0.12), "gaussian")
    _add_candidate(_exp_decay(n, rate=0.003), "exp_decay")
    _add_candidate(_sym_exp_decay(n, rate=0.003), "sym_exp_decay")
    _add_candidate(_beta_shape(n, alpha=0.6, beta=2.0), "beta(alpha=0.6,beta=2.0)")
    _add_candidate(_random_dirichlet(n, conc=0.8, rng=rng), "dirichlet(conc=0.8)")
    _add_candidate(_random_exponential(n, rng), "rand_exponential")

    if prev_arr is not None:
        _add_candidate(prev_arr, "previous")

    # ------------------------------------------------------------------
    # Randomised exploration of the analytic families
    # ------------------------------------------------------------------
    NUM_RANDOM = 250   # total random candidates (adjustable)
    for _ in range(NUM_RANDOM):
        family = rng.integers(0, 7)   # seven families defined below
        if family == 0:   # sine window with random gamma
            gamma = rng.uniform(0.5, 2.5)
            seq = _sine_window(n, gamma=gamma)
        elif family == 1:  # Gaussian with random sigma_frac
            sigma_frac = rng.uniform(0.04, 0.25)
            seq = _gaussian(n, sigma_frac=sigma_frac)
        elif family == 2:  # exponential decay
            rate = rng.uniform(0.0008, 0.008)
            seq = _exp_decay(n, rate=rate)
        elif family == 3:  # symmetric exponential decay
            rate = rng.uniform(0.0008, 0.008)
            seq = _sym_exp_decay(n, rate=rate)
        elif family == 4:  # beta shape
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            seq = _beta_shape(n, alpha=alpha, beta=beta)
        elif family == 5:  # Dirichlet
            conc = rng.uniform(0.1, 2.5)
            seq = _random_dirichlet(n, conc=conc, rng=rng)
        else:              # random exponential mixture
            seq = _random_exponential(n, rng)

        _add_candidate(seq, f"rand_{family}")

    # ------------------------------------------------------------------
    # Keep the best few seeds for polishing
    # ------------------------------------------------------------------
    candidates.sort(key=lambda x: x[0])
    TOP_K = 8
    top_seeds = [(seq, label) for (_, seq, label) in candidates[:TOP_K]]
    print(f"[Info] Selected {len(top_seeds)} seeds for polishing (best surrogate ≈ {candidates[0][0]:.6f})")

    # ------------------------------------------------------------------
    # Polishing each seed with exponentiated‑gradient mirror descent
    # ------------------------------------------------------------------
    best_seq = None
    best_val = math.inf

    for idx, (seed_seq, label) in enumerate(top_seeds, 1):
        if time.time() > deadline - 10.0:
            print("[Info] Approaching time limit – stopping polishing of further seeds.")
            break
        print(f"\n[Polish] Seed {idx}/{len(top_seeds)} ({label})")
        # allocate a modest number of iterations; the routine will stop early if time runs out
        seq_polished, val_polished = _exp_gradient_opt_multi(
            seed_seq,
            n,
            deadline,
            max_iters=600_000,
            eta0=4.5
        )
        print(f"  Polished surrogate = {val_polished:.6f}")

        if val_polished < best_val:
            best_val = val_polished
            best_seq = seq_polished.copy()
            print(f"  *** New best after polishing seed {idx} ***")

    if best_seq is None:
        # fallback – use the very best raw seed
        best_seq = top_seeds[0][0].copy()
        best_val = _objective_surrogate(best_seq)
        print("[Info] No polishing performed; using raw best seed.")

    # ------------------------------------------------------------------
    # Final high‑precision polishing (large iteration budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print("\n[Final] High‑precision EG polishing on the current best")
        seq_final, val_final = _exp_gradient_opt_multi(
            best_seq,
            n,
            deadline,
            max_iters=2_500_000,
            eta0=4.5
        )
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Final] Improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional simulated‑annealing fine‑tuning (uses remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("\n[Final] Simulated annealing fine‑tuning")
        seq_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            deadline,
            rng,
            max_attempts=120_000,
            start_sigma=0.001,
            T_start=5e-3,
            T_end=1e-9
        )
        if val_sa < best_val:
            best_seq = seq_sa
            best_val = val_sa
            print(f"[Final] SA improvement: surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Evaluate with the provided black‑box evaluator (real objective)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Error] evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
