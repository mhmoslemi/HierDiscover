def _smooth_objective_and_grad(v, n, beta):
    a = _softmax(v)  # shape n, sum=1
    conv = _conv_fft(a, a)  # length 2n-1
    max_c = np.max(conv)
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    return obj, grad_v

                   beta_schedule=None,

"""
Improved step‑function optimisation.

The algorithm proceeds in three stages.

1️⃣  **Initial population** – a handful of analytically‑defined shapes
    (uniform, triangular, linear ramps) together with a few
    random Dirichlet draws are normalised (ℓ₁‑norm = 1) and evaluated
    with the cheap surrogate 2·n·max(conv(a,a)).  The best few are kept
    as seeds.

2️⃣  **Continuation Adam optimisation** – each seed is refined by a
    smooth‑max formulation.  For a temperature β we optimise

        Fβ(a)=2·n·(max(conv)+log‑sum‑exp(β·(conv‑max))/β)

    using Adam on the log‑softmax parametrisation (guaranteeing
    non‑negativity and unit sum).  β is increased gradually
    (continuation) so the optimiser first sees a smooth landscape and
    finally concentrates on the true max.  Multiple restarts are run
    until the time budget is exhausted.

3️⃣  **Local stochastic polishing** – a cheap hill‑climber moves a small
    amount of mass between two random coordinates, accepting improvements
    and slowly cooling the step size.  This often squeezes a little more
    out of the Adam result.

The whole pipeline respects the 1000 s wall‑time limit.  The best
sequence found is stored in the global ``height_sequence_1`` variable
and returned as a plain Python list.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – only used for the L‑BFGS fallback (not required)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt  # noqa: F401
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – exact objective when sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth‑max objective and its gradient (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_obj_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (smooth objective, gradient w.r.t. log‑softmax parameters v).
    The optimisation variables are v = log a (a = softmax(v) has unit sum).
    """
    a = _softmax(v)                           # ℓ₁‑normalised, non‑negative
    conv = _conv_fft(a, a)                    # length 2n‑1
    max_c = np.max(conv)
    shift = max_c                              # numeric stability
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    w = exp_beta / sum_exp                    # weights for smooth max

    # gradient of 2·n·max_c part w.r.t. a is 2·n·grad_a where grad_a = 2·rev_a * w
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]      # shape (n,)

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)               # ∂F/∂v

    # smooth objective value
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser with continuation on β
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   steps_per_beta: int = 1500,
                   lr: float = 0.7):
    """
    Continuation Adam on the smooth‑max surrogate.
    Returns the best vector (ℓ₁‑normalised) seen and its smooth‑max value.
    """
    if beta_schedule is None:
        beta_schedule = [
            1, 5, 10, 20, 50, 100, 200, 500, 1000,
            2000, 5000, 10000, 20000, 50000,
            100000, 200000, 500000, 1000000,
        ]

    eps = 1e-30
    v = np.log(start + eps)          # log‑softmax parameters
    m = np.zeros_like(v)             # first moment
    v2 = np.zeros_like(v)            # second moment
    beta1, beta2 = 0.9, 0.999
    t = 0
    best_seq = None
    best_obj = float('inf')

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break
            t += 1
            obj, grad = _smooth_obj_grad(v, n, beta)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

            if obj < best_obj:
                best_obj = obj
                best_seq = _softmax(v).copy()

    if best_seq is None:                # fallback (should never happen)
        best_seq = _softmax(v)
    return best_seq, best_obj

        init_vectors.append(_norm(a))

    # previous best, if any
    if isinstance(prev, (list, tuple)) and len(prev) == n:
        prev_arr = np.asarray(prev, dtype=np.float64)
        init_vectors.append(_norm(prev_arr))

    # keep the best few according to the surrogate
    scored = [(_objective_surrogate(v), v) for v in init_vectors]
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:8]]
    best_val, best_seq = scored[0]
    print(f"Initial best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Continuation Adam optimisation (multiple restarts)
    # --------------------------------------------------------------
    print("\n--- Continuation Adam optimisation ---")
    restart = 0
    while time.time() < deadline and restart < 20:
        # choose a start vector: cycle through the top seeds, then random
        if restart < len(top_candidates):
            start_vec = top_candidates[restart]
        else:
            # fresh random Dirichlet (concentration drawn uniformly)
            conc = rng.uniform(0.05, 5.0)
            start_vec = rng.dirichlet(np.full(n, conc)).astype(np.float64)

        start_vec = _norm(start_vec)

        opt_seq, _ = _adam_optimize(start_vec, n, deadline,
                                    beta_schedule=[
                                        1, 5, 10, 20, 50, 100, 200, 500,
                                        1000, 2000, 5000, 10000, 20000,
                                        50000, 100000
                                    ],
                                    steps_per_beta=1500,
                                    lr=0.7)

        val = _objective_surrogate(opt_seq)
        if val < best_val:
            best_val, best_seq = val, opt_seq.copy()
            print(f"Restart {restart}: new best surrogate = {best_val:.6f}")

        restart += 1

    # --------------------------------------------------------------
    # 3️⃣  Local stochastic polishing (tiny mass moves)
    # --------------------------------------------------------------
    print("\n--- Local stochastic polishing ---")
    cur_seq = best_seq.copy()
    cur_val = best_val
    step_frac = 0.05
    it = 0
    while time.time() < deadline - 1.0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = step_frac * cur_seq[i] * rng.uniform(0.5, 1.5)
        delta = min(delta, cur_seq[i])
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        val_new = _objective_surrogate(new_seq)
        if val_new < cur_val:
            cur_seq = new_seq
            cur_val = val_new
            if cur_val < best_val:
                best_val, best_seq = cur_val, cur_seq.copy()
                print(f"Polish iter {it:6d}: new best surrogate = {best_val:.6f}")
            step_frac = min(0.2, step_frac * 1.02)
        else:
            step_frac = max(1e-4, step_frac * 0.9995)
        it += 1
        if it % 5000 == 0:
            print(f"Polish iter {it:6d}: current surrogate = {cur_val:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final true evaluation and bookkeeping
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
          f"true evaluate_sequence = {true_val}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
