        if max_iters is not None and it >= max_iters:

"""
Gradient‑based search for a step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm

1.  Generates a diverse set of analytic and random seeds (uniform,
    triangular, sine‑windows, exponential / logistic decays, Dirichlet
    draws, low‑frequency cosine blends and perturbations of a previously
    found solution).

2.  Evaluates the fast surrogate 2·n·max(conv) for all seeds and keeps the
    best few.

3.  Refines the current best seed with a short L‑BFGS run on a smooth
    log‑sum‑exp approximation of the maximum.

4.  Performs a long‑running **Adam** optimiser directly on the true
    max‑convolution.  The gradient is obtained by averaging the exact
    sub‑gradients of all convolution entries whose value lies within a
    small relative tolerance of the maximum (active‑set method).  The
    optimisation lives in the log‑softmax space so the simplex constraint
    is satisfied automatically.

5.  If time permits a final exact EG (exponentiated‑gradient) polishing
    pass is executed.

All work respects the global 1000 s budget; the best sequence found is
stored as ``height_sequence_1`` for possible warm‑starts on the next run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (for the L‑BFGS stage)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of *v* onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors (sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-γ·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    """Logistic decreasing shape: 1/(1+exp(steep·(x‑shift)))"""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sample from a symmetric Dirichlet."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves; clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Mixing utilities
# ----------------------------------------------------------------------
def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


def _best_mixed_pair(a: np.ndarray, b: np.ndarray,
                     deadline: float, steps: int = 30):
    """Ternary search for the best convex combination of a and b."""
    lo, hi = 0.0, 1.0
    best_val = None
    best_seq = None
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)
        if val1 < val2:
            hi = m2
            if best_val is None or val1 < best_val:
                best_val, best_seq = val1, seq1
        else:
            lo = m1
            if best_val is None or val2 < best_val:
                best_val, best_seq = val2, seq2
    if best_seq is None:
        best_seq = a.copy()
        best_val = _objective_surrogate(best_seq)
    else:
        mid = (lo + hi) / 2.0
        cand = _mix_shapes(a, b, mid)
        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_seq, best_val = cand, cand_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 6000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space so we stay on the probability simplex.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0,
                         2000000.0, 5000000.0,
                         10000000.0, 20000000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (smooth max)
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] Exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact EG optimiser (mirror‑descent) on max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry (index k)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (active‑set gradient)
# ----------------------------------------------------------------------
def _adam_opt_max(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  lr: float = 0.05,
                  delta: float = 0.01,
                  max_iters: int = None):
    """
    Adam optimiser on the true max‑convolution.
    Uses an averaged sub‑gradient over all convolution entries whose value
    lies within (1‑delta)·max.  Variables live in log‑softmax space.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    it = 0
    while True:
        if time.time() > deadline:
            break
        if max_iters is not None and it >= max_iters:
            break
        it += 1

        # current probabilities
        a = _softmax(log_a)

        # convolution and active set
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        thr = max_c * (1.0 - delta)
        active = np.where(conv >= thr)[0]
        if active.size == 0:               # fallback – only the argmax
            active = np.array([np.argmax(conv)], dtype=int)

        grad_a = np.zeros_like(a)

            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            grad_a[idx_i] += 2.0 * a[rev_idx]
        grad_a = grad_a / float(active.size)

        # mirror‑descent (log‑space) gradient
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep log_a in a reasonable range (prevents overflow)
        log_a = np.clip(log_a, -30.0, 30.0)

        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found (list of floats)
    and stores it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # problem size – the original run used n = 4000
    n = 4000

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, (list, tuple)):
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation (rich family)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Basic analytic windows
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 6.0, 20):
        seeds.append(_exp_decay(n, gamma))
    for steep in np.linspace(0.5, 30.0, 12):
        for shift in np.linspace(0.1, 0.9, 5):
            seeds.append(_logistic_decay(n, steep, shift))
    for gamma in np.linspace(0.5, 3.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Low‑frequency random cosine curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Perturbations of the previous best (if any)
    if prev_arr is not None:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(4):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Pairwise mixing of the top seeds
    # ------------------------------------------------------------------
    print("\n--- Mixing top seeds ---")
    mix_deadline = deadline - 2.0
    mixed_candidates = []

    for i in range(len(top_seeds)):
        for j in range(i + 1, len(top_seeds)):
            if time.time() > mix_deadline:
                break
            seq_mix, val_mix = _best_mixed_pair(top_seeds[i],
                                                top_seeds[j],
                                                mix_deadline,
                                                steps=25)
            mixed_candidates.append((seq_mix, val_mix))

    # Combine raw top seeds and mixtures
    all_candidates = evaluated[:top_k] + mixed_candidates
    all_candidates.sort(key=lambda x: x[1])
    best_seq = all_candidates[0][0].copy()
    best_val = all_candidates[0][1]
    print(f"\n[Init] best surrogate after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Short L‑BFGS smoothing to move away from poor local minima
    # ------------------------------------------------------------------
    if _opt is not None and time.time() < deadline:
        print("\n--- Quick L‑BFGS smoothing ---")
        seq_lbfgs = _lbfgs_smooth_optimize(best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[1.0, 10.0, 100.0,
                                                          1000.0, 1e4, 1e5,
                                                          1e6, 1e7, 1e8],
                                           max_iter_per_beta=3000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < best_val:
            best_seq, best_val = seq_lbfgs, val_lbfgs
            print(f"[L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Long Adam optimisation on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Adam optimisation on exact max ---")
        seq_adam, val_adam = _adam_opt_max(best_seq,
                                           n,
                                           deadline,
                                           lr=0.05,
                                           delta=0.01,
                                           max_iters=None)
        if val_adam < best_val:
            best_seq, best_val = seq_adam, val_adam
            print(f"[Adam] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Optional final EG polishing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq,
                                           n,
                                           deadline,
                                           max_iters=300_000,
                                           eta0=6.0)
        if val_eg < best_val:
            best_seq, best_val = seq_eg, val_eg
            print(f"[EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation (exact objective)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
