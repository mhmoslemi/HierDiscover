Key challenge: The evaluate_sequence function normalizes after converting to float and truncates negative numbers (clamp to 0), and caps at max 1000. This function expects sequence to be a list of non-negative floats. It then calculates:

- n = len(sequence)
- b_sequence = convolve(sequence, sequence): full linear convolution (length = 2*n-1)
- max_b = max(b_sequence)
- sum_a = sum(sequence)
- If sum_a < 0.01, returns inf.
- Return 2 * n * max_b / (sum_a ** 2)

Thus objective is 2 * n * max(conv(seq, seq)) / (sum(seq)**2).

If we normalize a constant factor: For any scaling factor c > 0, max(conv(c * seq, c * seq)) = c^2 * max(conv(seq, seq)), sum(c*seq) = c * sum(seq). So the ratio is 2 * n * (c^2 * maxconv) / ((c * sum)^2) = 2 * n * maxconv / (sum^2): scaling invariant. So we can think of it as minimizing 2*n * max(conv(seq, seq)) given fixed sum (or normalized sum=1). The "surrogate" used in code is exactly that: with sum=1, evaluating 2 * n * max(conv(seq, seq)). So the objective is homogenous scale-invariant of the vector, as expected.

"""
Propose a step‑function sequence that (approximately) minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three stages.

1. **Diverse deterministic seeding** – a large collection of analytic families
   (uniform, triangular, sine‐window, beta‐skewed, exponential, power‑law,
   mixtures and random Dirichlet draws).  All candidates are ℓ₁‑normalised and
   evaluated with a fast FFT‑based surrogate objective
   ``_objective_surrogate``.  The best ``top_k`` seeds are retained.

2. **Mirror‑descent on the exact max‑convolution** – for each seed an Adam‑style
   optimizer works in the log‑softmax (mirror) space.  The sub‑gradient of the
   max‑convolution is obtained from the (usually single) active index; the
   gradient is transformed to log‑variables and updated with Adam
   (bias‑corrected first‑ and second‑moments).  The best vector encountered
   during the run is stored.

3. **Polishing with simulated annealing** – a short low‑temperature SA pass
   (Gaussian perturbations of the log‑variables) explores the neighbourhood of
   the current global best.  The final candidate is evaluated with the provided
   ``evaluate_sequence`` function.

All time‑consuming operations respect the global deadline (default 1000 s)
so that a valid sequence is always returned.  The resulting sequence is stored
in the global variable ``height_sequence_1`` for possible reuse in later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N).real[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective – exact max‑convolution for ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq, seq)) for a sequence with sum == 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑distribution shape on 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a symmetric Dirichlet (conc > 0)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (log‑softmax space)
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 500_000,
                   lr: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Adam) on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    # work in log‑space (log‑softmax) – ensures positivity & ℓ₁‑norm
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # ----- active set -------------------------------------------------
        # tolerance generous enough to capture a flat peak
        eps_active = max_val * 1e-12
        active = np.where(conv >= max_val - eps_active)[0]

        # ----- sub‑gradient of max(conv) ---------------------------------
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            subgrad_maxconv = 2.0 * g_tot / active.size   # sub‑gradient of maxconv
        else:
            subgrad_maxconv = np.zeros_like(a)

        # Gradient of the surrogate (2·n·maxconv)
        grad = 2.0 * n * subgrad_maxconv

        # Mirror‑descent transformation (log‑softmax variables)
        grad_log = a * (grad - np.dot(grad, a))

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        log_a = log_a - step * m_hat
        a = _softmax(log_a)                # ensures sum == 1, non‑negative

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing polishing (optimises the surrogate)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 120_000,
                         sigma_start: float = 0.003,
                         T_start: float = 0.02,
                         T_end: float = 1e-7) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbations in log‑space with a linear cooling schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)

        # perturb in log‑space (more stable than direct additive)
        log_a = np.log(a + 1e-30)
        log_n = log_a + sigma * rng.normal(size=n)
        ns = _softmax(log_n)

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()

        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a slightly larger n gives more flexibility without
    # overwhelming the time budget.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previously found solution if it exists (continuity)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Deterministic seeding – a wide family of candidate shapes
    # ------------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # Basic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))

    # Asymmetric beta families (left‑skewed / right‑skewed)
    alphas = np.linspace(0.1, 12.0, 30)
    betas = np.linspace(0.1, 12.0, 30)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Exponential decays (covering very slow to fairly steep)
    rates = np.logspace(-4, -0.1, 45)   # from 1e‑4 to ~0.8
    for r in rates:
        seeds.append(_exp_decay(n, r))

    # Power‑law families
    ps = np.linspace(0.0, 5.0, 30)
    for p in ps:
        seeds.append(_power(n, p))

    # Mixtures: uniform + exponential (asymmetric bias)
    for r in rates[::6]:
        exp_seq = _exp_decay(n, r)
        for a in (0.05, 0.15, 0.30, 0.45):
            seeds.append(_mix_shapes(_uniform(n), exp_seq, a))

    # Random Dirichlet draws (including very sparse and very flat)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Perturbations of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        base = prev_arr / prev_arr.sum()
        seeds.append(base)
        for _ in range(8):
            pert = base + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start Adam optimisation on the top seeds
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget
        remaining = deadline - time.time()
        # give a slightly larger share to earlier seeds
        share = remaining * (0.20 / max(1, (top_k - si)))
        sub_deadline = time.time() + share

        seq_adam, val_adam = _adam_optimize(seed_vec.copy(),
                                            n,
                                            sub_deadline,
                                            max_iters=300_000,
                                            lr=0.6)

        # short SA polishing
        hill_deadline = sub_deadline - 0.5
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_adam,
                                                 n,
                                                 hill_deadline,
                                                 rng,
                                                 max_attempts=80_000,
                                                 sigma_start=0.002,
                                                 T_start=0.015,
                                                 T_end=5e-8)
        else:
            seq_sa, val_sa = seq_adam, val_adam

        # pick best among the three candidates (seed, Adam, SA)
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_adam, val_adam),
            (seq_sa, val_sa)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing – longer Adam run + final SA
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long Adam polishing ---")
        final_seq, final_val = _adam_optimize(global_best_seq,
                                               n,
                                               deadline,
                                               max_iters=2_500_000,
                                               lr=0.5)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final Adam] improved to {global_best_val:.6f}")

        if time.time() < deadline:
            print("\n--- Final simulated‑annealing polishing ---")
            final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                              n,
                                                              deadline,
                                                              rng,
                                                              max_attempts=100_000,
                                                              sigma_start=0.001,
                                                              T_start=5e-3,
                                                              T_end=1e-9)
            if final_sa_val < global_best_val:
                global_best_seq = final_sa_seq
                global_best_val = final_sa_val
                print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
