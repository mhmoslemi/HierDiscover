def _adam_exp_gradient(start_seq, n, deadline, max_iters=200000, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8, single_active=False):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for t in range(1, max_iters+1):
        if time.time()>deadline: break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set
        if single_active:
            k = conv.argmax()
            active = np.array([k], dtype=int)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        # compute gradient
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k-(n-1))
            i_end = min(n-1, k)
            idx = np.arange(i_start, i_end+1, dtype=int)
            rev = k - idx
            g_tot[idx] += a[rev]
        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # g is gradient of max conv in a-space. To compute gradient wrt log_a, chain rule:
        # gradient_v = a * (g - dot(g,a))  # as earlier for log_softmax.
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v = beta2 * v + (1 - beta2) * (grad_v**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val
```

"""
Propose a high‑quality step‑function sequence.

Algorithm
---------

1. **Massive stochastic seeding** – draw many candidate shapes from
   families that are known to produce low convolution peaks:
   uniform, triangular, sine‑window, beta‑distribution, exponential decay,
   power‑law, mixtures and very sparse Dirichlet draws.  All candidates are
   ℓ₁‑normalised (sum = 1).

2. **Surrogate evaluation** – for a normalised vector the true objective is
   ``2·n·max(conv(a,a))``.  The best ``top_k`` seeds are kept.

3. **Mirror‑descent with Adam** – optimisation is performed in the log‑soft‑max
   (exponentiated‑gradient) space, guaranteeing non‑negativity and unit sum.
   The exact sub‑gradient of the max‑convolution is obtained by an FFT‑based
   convolution with a uniform weight on the currently active indices.
   Adam’s adaptive learning‑rate speeds up convergence.

4. **Continuation L‑BFGS** – a smooth log‑sum‑exp surrogate of the max is
   optimised with L‑BFGS‑B (if SciPy is available) for a few β‑values.

5. **Final polishing** – a longer Adam run (single active index) followed by
   a low‑temperature simulated‑annealing pass.

All work respects the global 1000‑second budget.  The best sequence is stored
in the global variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import math
import numpy as np

# optional SciPy for L‑BFGS
try:
    import scipy.optimize as _opt
except Exception:      # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Adam mirror‑descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_exp_gradient(start_seq: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters: int = 300_000,
                       lr: float = 4.0,
                       beta1: float = 0.9,
                       beta2: float = 0.999,
                       eps: float = 1e-8,
                       single_active: bool = False):
    """
    Mirror‑descent (Adam) on the exact maximum of a*a.
    ``single_active`` – if True only the most active index is used;
    otherwise the whole active set (within 1e‑12 relative tolerance) is used.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    # initialise log‑softmax variables
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # full convolution a*a
        conv = _conv_fft(a, a)
        max_val = conv.max()

        # ----- active set -----
        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            eps_rel = max_val * 1e-12
            active = np.where(conv >= max_val - eps_rel)[0]

        # ----- gradient of max(conv) w.r.t. a (vectorised) -----
        if active.size == 0:
            grad_a = np.zeros_like(a)
        else:
            # uniform weights on the active indices
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            # gradient of weighted sum of conv entries:
            # g = 2 * (rev_a * w) sliced to length n
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- chain‑rule to log‑softmax space -----
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)                # dF/dlog_a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing feasibility.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # smooth max gradient
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.02,
                         T_end: float = 1e-7):
    """
    Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val
