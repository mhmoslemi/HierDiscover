"""
Step‑function optimizer for ``evaluate_sequence``.

The algorithm works in three phases:

1. **Seed generation** – a compact pool of analytic families
   (uniform, triangular, sine window, Gaussian, power‑law, beta,
   exponential) and a few random Dirichlet draws is created.
   The best seed (smallest surrogate 2·n·max(conv)) is kept.

2. **Smooth max optimisation** – the surrogate objective
   ``max(conv)`` is replaced by a log‑sum‑exp (smooth max) with a
   schedule of increasing β values.  An Adam optimiser works on the
   logits of the probability vector (exponentiated‑gradient / mirror
   descent).  At each iteration the exact unsmoothed objective is
   evaluated, and the best distribution seen is remembered.

3. **Exact‑max polishing** – a final exponentiated‑gradient run on the
   true max‑convolution (sub‑gradient on the active index) refines the
   solution.

All steps respect the overall 1000 s budget.  The final sequence is
stored in the global variable ``height_sequence_1`` for possible
continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:       # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numerical utilities (FFT‑based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient on the exact max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution (sub‑gradient on the
    active index)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max (log‑sum‑exp) objective
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule,
                     iters_per_beta: int = 3000,
                     lr: float = 0.02,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8):
    """Adam optimisation of the log‑sum‑exp surrogate of max(conv)."""
    # initialise logits
    v = np.log(start_seq + 1e-30)
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    best_seq = None
    best_val = np.inf
    it_total = 0

    for beta_val in beta_schedule:
        beta = float(beta_val)
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                # time out – return the best we have
                if best_seq is None:
                    best_seq = _softmax(v)
                return best_seq, best_val

            it_total += 1

            # probability vector
            a = _softmax(v)

            # full self‑convolution
            conv = _conv_fft(a, a)
            max_conv = float(np.max(conv))

            # smooth max via log‑sum‑exp
            shift = max_conv
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp

            # gradient of the smooth max w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # shape (n,)

            # translate to gradient w.r.t. logits (softmax Jacobian)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam bias‑corrected updates
            m1 = beta1 * m1 + (1.0 - beta1) * grad_v
            m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)
            m1_hat = m1 / (1.0 - beta1 ** it_total)
            m2_hat = m2 / (1.0 - beta2 ** it_total)

            v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

            # keep track of the *exact* (unsmoothed) objective
            cur_val = 2.0 * n * max_conv
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    # final fall‑back – if nothing was stored yet
    if best_seq is None:
        best_seq = _softmax(v)
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The method builds a pool of analytic / random
    seeds, picks the best surrogate, refines it with a smooth‑max Adam
    optimiser (log‑sum‑exp), finishes with an exact‑max exponentiated‑gradient
    polish, and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  4000 is a proven compromise.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previously‑found solution (if any) for continuity.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Seed generation (lightweight but diverse)
    # ------------------------------------------------------------------
    seeds = []

    # a few hand‑crafted analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_power(n, p=1.0))
    seeds.append(_exp_decay(n, rate=0.005))
    seeds.append(_sym_exp_decay(n, rate=0.005))
    seeds.append(_beta_shape(n, alpha=2.0, beta=2.0))

    # random Dirichlet draws with a spread of concentration values
    for conc in [0.01, 0.05, 0.1, 0.5, 1.0, 2.0]:
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # include the previous best if present
    if prev_arr is not None:
        seeds.append(prev_arr)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best seed
    # ------------------------------------------------------------------
    best_seed = None
    best_surrogate = np.inf
    for s in seeds:
        if time.time() > deadline:
            break
        s = np.maximum(s, 0.0)
        total = s.sum()
        if total == 0.0:
            continue
        s = s / total
        val = _objective_surrogate(s)
        if val < best_surrogate:
            best_surrogate = val
            best_seed = s.copy()

    if best_seed is None:                     # fallback – uniform
        best_seed = _uniform(n)
        best_surrogate = _objective_surrogate(best_seed)

    print(f"[Init] best surrogate from seeds = {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # 3. Smooth‑max Adam optimisation
    # ------------------------------------------------------------------
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0, 1_000_000.0
    ]
    # allocate roughly 70 % of remaining time to this phase
    remaining = deadline - time.time()
    # a generous per‑beta budget (will be trimmed by the time check)
    iters_per_beta = 3000

    print("\n--- Smooth‑max Adam optimisation ---")
    seq_adam, val_adam = _adam_smooth_opt(best_seed,
                                          n,
                                          deadline,
                                          beta_schedule,
                                          iters_per_beta=iters_per_beta,
                                          lr=0.02)

    print(f"[Adam] best surrogate after Adam = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4. Exact‑max polishing with exponentiated‑gradient (sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exact‑max exponentiated‑gradient polishing ---")
        seq_final, val_final = _exp_gradient_opt(seq_adam,
                                                 n,
                                                 deadline,
                                                 max_iters=400_000,
                                                 eta0=5.0)
        if val_final < val_adam:
            seq_adam, val_adam = seq_final, val_final
            print(f"[EG] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(seq_adam.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {val_adam:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible continuation runs
    globals()["height_sequence_1"] = seq_adam.tolist()
    return seq_adam.tolist()
