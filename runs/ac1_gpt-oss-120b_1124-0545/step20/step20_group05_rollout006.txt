def _adam_opt(start_seq, n, deadline, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8, max_iters=500000):

   log_a = np.log(start_seq + 1e-30)
   m = np.zeros_like(log_a)
   v = np.zeros_like(log_a)
   t = 0
   best_seq = start_seq; best_val = surrogate

   while time.time() < deadline and t < max_iters:
       t+=1
       a = _softmax(log_a)
       conv = _conv_fft(a, a)
       k = int(np.argmax(conv))
       
       # gradient w.r.t a (subgradient)
       g = np.zeros_like(a)
       i_start = max(0, k-(n-1))
       i_end = min(n-1, k)
       idx_i = np.arange(i_start, i_end+1, dtype=int)
       rev_idx = k-idx_i
       g[idx_i] = 2.0 * a[rev_idx]
       # gradient w.r.t log_a: chain rule of softmax
       # d f/d log_a = a * (g - dot(g, a))
       # Actually gradient w.r.t objective: partial derivative w.r.t a is g but effective gradient w.r.t log_a = a*(g - (a·g))
       dot = np.dot(g, a)
       grad_log = a * (g - dot)
       
       # Adam update
       m = beta1*m + (1-beta1)*grad_log
       v = beta2*v + (1-beta2)*(grad_log**2)
       m_hat = m / (1-beta1**t)
       v_hat = v / (1-beta2**t)
       log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)
       
       # evaluate
       cur_val = _objective_surrogate(_softmax(log_a))
       # Actually compute a = softmax(log_a)
       a = _softmax(log_a)
       cur_val = _objective_surrogate(a)
       if cur_val < best_val:
           best_val = cur_val
           best_seq = a.copy()
   return best_seq, best_val

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that minimizes
``evaluate_sequence``.  The algorithm consists of:

1. **Broad seeding** – generate a very large pool of analytic families
   (beta‑shapes with tiny α,β, sine windows with high γ, Kaiser windows,
   generalized Gaussians, double‑Gaussian, etc.) together with many random
   Dirichlet / exponential / low‑frequency curves.  Each candidate is
   ℓ₁‑normalised, the cheap surrogate objective `2·n·max(conv)` is computed
   and the best few are retained.

2. **Intensive mirror‑descent** – starting from the best seed we run a
   long exponentiated‑gradient (EG) optimisation in log‑softmax space.
   The sub‑gradient of the active convolution entry is used; the step‑size
   decays as η₀/√t.  The EG run consumes the majority of the time‑budget
   and yields the current best sequence.

3. **Continuation L‑BFGS** – the nonsmooth max is approximated by a
   log‑sum‑exp surrogate with a schedule of increasing β (smoothness
   parameter).  SciPy’s L‑BFGS‑B optimiser is invoked for each β.

4. **Local stochastic refinement** – a cheap hill‑climbing move transfers a
   small amount of probability mass between two random indices; any
   improvement is kept.  This cleans up remaining tiny irregularities.

5. **Uniform mixing** – a ternary search for the best convex blend of the
   current best sequence with the uniform distribution further reduces the
   peak.

The whole routine respects a 1000 s wall‑time limit (a 2.5 s safety margin).
The final sequence is returned as a plain ``list[float]`` and stored in the
global variable ``height_sequence_1`` for possible reuse.

# Low‑level numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector ℓ₁‑normalised to 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(np.maximum(0.0, 1 - y * y))
    a = np.i0(t)          # Modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _generalized_gaussian(n: int, sigma_frac: float, p: float) -> np.ndarray:
    """a_i ∝ exp( -|i-μ|^p / (2σ^p) )   (p>0)."""
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-np.abs(i - mu) ** p / (2.0 * sigma ** p))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random families
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


def _ternary_mix(a: np.ndarray, b: np.ndarray, deadline: float,
                 steps: int = 30):
    """Best convex combination of a and b by ternary search on α∈[0,1]."""
    lo, hi = 0.0, 1.0
    best_seq, best_val = None, float('inf')
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)
        if val1 < val2:
            hi = m2
            if val1 < best_val:
                best_val, best_seq = val1, seq1
        else:
            lo = m1
            if val2 < best_val:
                best_val, best_seq = val2, seq2
    mid = (lo + hi) / 2.0
    seq_mid = _mix_shapes(a, b, mid)
    val_mid = _objective_surrogate(seq_mid)
    if val_mid < best_val:
        best_seq, best_val = seq_mid, val_mid
    return best_seq, best_val

# Mirror‑descent (Exponentiated Gradient) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 5.0):
    """Mirror‑descent on exact max‑convolution.  Returns best sequence seen."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp   # kept for compatibility
except Exception:  # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space (v) guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0, 2000.0, 5000.0,
                         20000.0, 50000.0, 100000.0, 200000.0,
                         500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)          # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Simple stochastic local search
# ----------------------------------------------------------------------
def _local_search(seq: np.ndarray,
                  deadline: float,
                  moves: int = 30000,
                  step_scale: float = 0.015):
    """Random mass‑transfer hill‑climbing."""
    rng = np.random.default_rng()
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    n = len(seq)

    for _ in range(moves):
        if time.time() > deadline:
            break
        eps = step_scale * rng.random()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = min(eps, best_seq[i])
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq = new_seq / new_seq.sum()
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq
    return best_seq, best_val

        for sigma in np.linspace(0.03, 0.35, 14):

    # Low‑frequency curves

        for _ in range(12):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Previous best (if any)

    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                arr = arr / arr.sum()
                seeds.append(arr)
                for _ in range(5):
                    seeds.append(_perturb(arr, rng, sigma=0.004))
        except Exception:
            pass

    # --------------------------------------------------------------
    # 2.  Surrogate evaluation of all seeds (cheap)
    # --------------------------------------------------------------
    print("\n--- Evaluating seeds (surrogate) ---")
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop surrogate values after seeding:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3.  Intensive refinement
    # --------------------------------------------------------------
    best_overall_seq = top_seeds[0].copy()
    best_overall_val = _objective_surrogate(best_overall_seq)

    # 3a – long EG on the absolute best seed
    remaining = deadline - time.time()
    eg_deadline = time.time() + max(10.0, 0.55 * remaining)
    print("\n--- Long EG optimisation on best seed ---")
    seq_eg, val_eg = _exp_gradient_opt(best_overall_seq.copy(),
                                       n,
                                       eg_deadline,
                                       max_iters=4_000_000,
                                       eta0=7.0)
    if val_eg < best_overall_val:
        best_overall_val, best_overall_seq = val_eg, seq_eg.copy()
        print(f"  EG improvement: {val_eg:.6f}")

    # 3b – shorter EG on the remaining top seeds (quick boosts)
    for idx, seed_seq in enumerate(top_seeds[1:]):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        short_deadline = time.time() + max(5.0, 0.15 * remaining)
        seq_tmp, val_tmp = _exp_gradient_opt(seed_seq.copy(),
                                             n,
                                             short_deadline,

            print(f"  EG boost from seed #{idx+2}: {val_tmp:.6f}")

    # 3c – continuation L‑BFGS smoothing (if SciPy is available)
    if time.time() < deadline:
        print("\n--- L‑BFGS smoothing ---")
        lbfgs_deadline = deadline - 2.0
        seq_lb = _lbfgs_smooth_optimize(best_overall_seq,
                                         n,
                                         lbfgs_deadline,
                                         beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                        200.0, 500.0, 1000.0,
                                                        2000.0, 5000.0,
                                                        20000.0, 50000.0,
                                                        100000.0, 200000.0,
                                                        500000.0, 1000000.0,
                                                        2000000.0],
                                         max_iter_per_beta=4000)
        val_lb = _objective_surrogate(seq_lb)
        if val_lb < best_overall_val:
            best_overall_val, best_overall_seq = val_lb, seq_lb.copy()
            print(f"  L‑BFGS improvement: {val_lb:.6f}")

    # 3d – ternary mixing with uniform (sometimes lowers the peak)
    if time.time() < deadline:
        try:
            mixed_seq, mixed_val = _ternary_mix(best_overall_seq,
                                                _uniform(n),
                                                deadline,
                                                steps=35)
            if mixed_val < best_overall_val:
                best_overall_val, best_overall_seq = mixed_val, mixed_seq.copy()
                print(f"  Uniform‑mix improvement: {mixed_val:.6f}")
        except Exception:
            pass

    # 3e – final stochastic local search
    if time.time() < deadline:
        print("\n--- Final local search ---")
        local_deadline = deadline - 1.0
        seq_loc, val_loc = _local_search(best_overall_seq,
                                         local_deadline,
                                         moves=25000,
                                         step_scale=0.012)
        if val_loc < best_overall_val:
            best_overall_val, best_overall_seq = val_loc, seq_loc.copy()
            print(f"  Local‑search improvement: {val_loc:.6f}")

    # --------------------------------------------------------------
    # 4.  Official evaluation
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised", exc)

    print(f"\n=== Finished ===\nBest surrogate ≈ {best_overall_val:.6f}"
          f", true evaluate_sequence = {true_val}\nLength = {len(best_overall_seq)}")

    # Store for possible reuse
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
