    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (used when the ℓ₁‑norm equals 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax utilities (used by exponentiated‑gradient)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic starter families (used to generate initial candidates)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(i, p)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _linear_ramp(n: int, m_factor: float) -> np.ndarray:
    """Truncated linear ramp: a_i = max(0, 1 - i/m) with m = m_factor·n."""
    m = max(1, int(m_factor * n))
    a = np.maximum(0.0, 1.0 - np.arange(n, dtype=np.float64) / m)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution (sub‑gradient of the active entry).
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see analysis notes)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space to stay feasible.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Helper to keep a fixed‑size heap of the best candidates.
# ----------------------------------------------------------------------
def _push_candidate(heap: List[Tuple[float, np.ndarray]],
                    val: float,
                    seq: np.ndarray,
                    top_k: int) -> None:
    """
    Keep a min‑heap of size at most `top_k` with the smallest objective values.
    """
    if len(heap) < top_k:
        heapq.heappush(heap, (-val, seq))   # store negative to turn min‑heap into max‑heap
    else:
        # heap[0] holds the *largest* (worst) value because of negative sign
        if -heap[0][0] > val:
            heapq.heapreplace(heap, (-val, seq))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine builds a diverse set of start‑points, refines them with
    mirror‑descent and a final L‑BFGS smoothing, and returns the best sequence.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin for finalisation

    # --------------------------------------------------------------
    # Problem size – reuse previous length if possible.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list):
        try:
            n = len(prev)
            if n < 2:
                raise ValueError
        except Exception:
            n = 4000
    else:
        n = 4000

    # --------------------------------------------------------------
    # 1. Load previous solution (if any) and initialise a fallback.
    # --------------------------------------------------------------
    if prev is not None and len(prev) == n:
        seq0 = np.asarray(prev, dtype=np.float64)
        seq0 = np.maximum(seq0, 0.0)
        total = seq0.sum()
        if total == 0.0:
            seq0 = _uniform(n)
        else:
            seq0 = seq0 / total
    else:
        # Fallback exponential tail
        seq0 = _exp_decay(n, rate=0.003)

    best_surrogate = _objective_surrogate(seq0)
    print(f"[Init] surrogate of starting sequence = {best_surrogate:.6f}")

    # --------------------------------------------------------------
    # 2. Populate a pool of diverse candidates (keep the top_k best).
    # --------------------------------------------------------------
    top_k = 15
    heap: List[Tuple[float, np.ndarray]] = []
    _push_candidate(heap, best_surrogate, seq0, top_k)

    # exponential rates
    for rate in np.linspace(0.0005, 0.0105, 25):
        if time.time() > deadline:
            break
        seq = _exp_decay(n, rate)
        val = _objective_surrogate(seq)
        _push_candidate(heap, val, seq, top_k)

    # power‑law exponents
    for p in np.linspace(0.1, 3.5, 25):
        if time.time() > deadline:
            break
        seq = _power_law(n, p)
        val = _objective_surrogate(seq)
        _push_candidate(heap, val, seq, top_k)

    # truncated linear ramps
    for mf in np.linspace(0.05, 1.0, 20):
        if time.time() > deadline:
            break
        seq = _linear_ramp(n, mf)
        val = _objective_surrogate(seq)
        _push_candidate(heap, val, seq, top_k)

    # random Dirichlet draws with several concentration parameters
    alphas = [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]
    # keep generating while we have a comfortable amount of time left
    while time.time() < deadline - 120.0:  # reserve ~2 min for polishing
        alpha = float(rng.choice(alphas))
        g = rng.gamma(shape=alpha, scale=1.0, size=n)
        seq = g / g.sum()
        val = _objective_surrogate(seq)
        _push_candidate(heap, val, seq, top_k)

    # retrieve the best candidates (as a list sorted by surrogate value)
    candidates = sorted([(-v, s) for (v, s) in heap], key=lambda x: x[0])

    best_val, best_seq = candidates[0]
    print(f"[Pool] best surrogate after candidate generation = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Polishing phase – mirror‑descent on each top candidate.
    # --------------------------------------------------------------
    # Decide on a total budget for polishing (≈ 70 % of remaining time)
    remaining = deadline - time.time()
    polishing_budget = remaining * 0.70
    # Divide roughly equally among the candidates (but ensure a minimum)
    max_iters_per = max(20_000, int(1_200_000 / len(candidates)))

    for cur_val, seq in candidates:
        if time.time() > deadline:
            break
        # Run exponentiated‑gradient with a time‑aware cut‑off
        seq_pol, val_pol = _exp_gradient_opt(seq, n,
                                             deadline,
                                             max_iters=max_iters_per,
                                             eta0=5.0)
        print(f"[EG] polished candidate surrogate = {val_pol:.6f}")
        if val_pol < best_val:
            best_val, best_seq = val_pol, seq_pol

    # --------------------------------------------------------------
    # 4. Final L‑BFGS smooth optimisation (if time permits).
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0 and _opt is not None:
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline,
                                        max_iter_per_beta=3000)
        val_lb = _objective_surrogate(seq_lb)
        print(f"[L‑BFGS] surrogate after smoothing = {val_lb:.6f}")
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb

    # --------------------------------------------------------------
    # 5. Final evaluation via the provided function.
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    # Store for the next run (facilitates incremental improvement)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
