"""
Step‑function optimiser (mirrored Adam on the exact max‑convolution).

The programme builds a few cheap analytic seeds, keeps the best one,
and then spends the remaining budget on a large‑scale Adam optimiser
working directly on the non‑smoothed objective
      2·n·max(conv(a,a))   (a is constrained to the probability simplex).
The optimisation lives in the log‑domain (mirror‑descent) which guarantees
non‑negativity and unit sum.  Adam’s adaptive step‑size together with a
√t decay usually yields a better final value than the simple
η ∝ 1/√t schedule used in the reference code.  A tiny random Gaussian
perturbation is injected every 200 k iterations to help escape plateaus.

The function ``propose_candidate`` follows the required signature,
stores the best sequence in the global ``height_sequence_1`` for
continuations and finally returns the list of heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (sums to 1, all entries >0)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:                        # should never happen
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for a probability vector)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Returns 2·n·max(conv(seq,seq)) – the quantity that evaluate_sequence
    returns when the sequence sums to 1."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) , i = 0 .. n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·|i-(n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator,
                   init_lr: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Adam) on L = 2·n·max(conv(a,a)).
    ``start_seq`` must already sum to one.
    Returns the best sequence seen and its surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)

    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq, n)

    t = 0
    while time.time() < deadline:
        t += 1

        # ------- a from log‑space (softmax) ----------
        max_log = np.max(log_a)
        a = np.exp(log_a - max_log)
        a /= a.sum()

        # ------- exact max‑convolution & sub‑gradient ----------
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # gradient w.r.t. log‑a (mirror‑descent formula)
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # ------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # a simple √t decay for the learning‑rate (more aggressive early)
        lr = init_lr / (np.sqrt(t) + 1.0)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------- keep the best ----------
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            print(f"[Adam] iter {t:,d} – new best surrogate = {best_val:.6f}")

        # ------- occasional random jitter to shake off plateaus ----------
        if t % 200_000 == 0:
            log_a += 0.01 * rng.normal(size=n)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The method:
        1. generate a handful of analytic / random seeds,
        2. pick the best seed (by the exact surrogate),
        3. run a long Adam‑based mirror‑descent on the true max‑convolution.
    The best vector is stored in the global variable ``height_sequence_1``
    for possible continuation runs.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # a small safety margin

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load a previous solution if available (helps continuation runs)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) == n:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.ndim == 1 and prev_arr.size == n:
                prev_arr = np.maximum(prev_arr, 0.0)
                total = prev_arr.sum()
                if total > 0.0:
                    prev_arr = prev_arr / total
                else:
                    prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # --------------------------------------------------------------
    # 1. Build a modest pool of analytic / random seeds
    # --------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    # a few exponential decays (dense sweep on a log‑scale)
    for rate in np.logspace(-5, -0.2, 30):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Gaussian windows
    for sigma in np.linspace(0.08, 0.30, 15):
        seeds.append(_gaussian(n, sigma))

    # Beta‑shapes – focus on the region around (0.5,0.5) and more extreme values
    for a in (0.2, 0.5, 1.0, 2.0, 4.0):
        for b in (0.2, 0.5, 1.0, 2.0, 4.0):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (various concentrations)
    conc_values = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for conc in conc_values:
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # include the previous best (if any)
    if prev_arr is not None:
        seeds.append(prev_arr)

    # --------------------------------------------------------------
    # 2. Quickly evaluate all seeds and keep the best one
    # --------------------------------------------------------------
    best_seed = None
    best_val = float('inf')
    for vec in seeds:
        if time.time() > deadline:
            break
        # safety – clip & renormalise (should already be fine)
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec, n)
        if val < best_val:
            best_val = val
            best_seed = vec.copy()
    if best_seed is None:
        # fallback – uniform
        best_seed = _uniform(n)
        best_val = _objective_surrogate(best_seed, n)

    print(f"\n[Init] chosen seed surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 3. Main optimisation – Adam mirror descent on the exact max‑conv.
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    print(f"[Main] allocating roughly {remaining:.1f}s to Adam optimisation")
    opt_seq, opt_val = _adam_optimize(best_seed, n, deadline, rng, init_lr=5.0)

    print(f"\n[Adam] final surrogate = {opt_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final exact evaluation (the quantity we really care about)
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(opt_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {opt_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning – evaluate_sequence raised:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = opt_seq.tolist()
    return opt_seq.tolist()
