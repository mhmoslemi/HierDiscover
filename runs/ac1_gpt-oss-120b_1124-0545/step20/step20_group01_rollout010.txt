"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Strategy
--------
1. Build a large pool of cheap analytic / random seeds (uniform, power‑law,
   beta‑shapes, Dirichlet draws, etc.).
2. Normalise each seed to ℓ₁‑norm = 1 and keep the best few according to the
   surrogate objective 2·n·max(conv(a,a)).
3. Starting from each promising seed run a long entropy‑mirror‑descent
   (exponentiated‑gradient) optimiser that directly minimizes the exact
   peak of the self‑convolution.  The optimiser uses an AdaGrad‑style
   per‑coordinate stepsize which converges much faster than a plain
   1/√t schedule.
4. After the gradient phase perform a short simulated‑annealing style
   permutation hill‑climb (swaps of two positions) to possibly lower the
   max‑convolution further.
5. If SciPy is available run a brief L‑BFGS smoothing phase on the
   log‑softmax parametrisation (smooth max via log‑sum‑exp).
6. Return the best sequence found; store it in the global variable
   ``height_sequence_1`` for later continuation runs.

The whole procedure respects the 1000 s budget and always produces a
valid (non‑negative, ℓ₁‑normalised) sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the optional L‑BFGS polishing.
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:       # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numerical utilities (FFT‑based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when the vector is ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) – assumes Σ seq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p   (i = 0..n‑1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(α,…,α) with α = conc."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser with AdaGrad steps.
# ----------------------------------------------------------------------
def _exp_gradient_opt_adagrad(start_seq: np.ndarray,
                              n: int,
                              deadline: float,
                              max_iters: int = 3_000_000,
                              eta0: float = 6.0):
    """
    Mirror‑descent on the exact max‑convolution.
    Uses per‑coordinate AdaGrad steps in the log‑domain.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)               # stay in the interior of the simplex
    grad_sq_acc = np.zeros_like(a)         # Σ g_i²
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active diagonal (ordered pairs i+j = max_idx)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # AdaGrad accumulation
        grad_sq_acc += g * g
        step = eta0 / (np.sqrt(grad_sq_acc) + 1e-12)   # per‑coordinate step size
        log_a = log_a - step * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation hill‑climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Random swaps with a very mild annealing schedule.
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 0.01 * cur_val
    T_end = 1e-5 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or rng.random() < math.exp(-(val - cur_val) / T):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS smoothing on the smooth‑max (log‑sum‑exp) surrogate.
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000):
    """
    Optimise a smooth approximation of max(conv) using L‑BFGS‑B.
    Operates in log‑softmax space.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1_000.0,
                         2_000.0, 5_000.0,
                         20_000.0, 50_000.0,
                         100_000.0, 200_000.0,
                         500_000.0, 1_000_000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            m = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - m))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (m + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # softmax weights of the smooth‑max
            rev_a = a[::-1]
            grad_a = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

    seeds.append(_beta_shape(n, 0.8, 0.8))
    seeds.append(_beta_shape(n, 0.3, 0.3))
