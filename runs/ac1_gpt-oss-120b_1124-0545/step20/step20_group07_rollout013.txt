provided ``evaluate_sequence`` function.

Algorithm Overview
------------------
1. **Extensive parametric sweep** – generate a few hundred analytic and random
   candidate vectors (uniform, power‑law, beta‑shaped, sine, Gaussian,
   Kaiser, Tukey, arcsine, low‑frequency sinusoids, Dirichlet, exponential …).
   All candidates are ℓ₁‑normalised (sum = 1) and ranked by the cheap surrogate

        ✱ 2·n·max(conv(a,a))    (exact for normalised vectors).

   The best ``top_k`` candidates are kept.

2. **Mirror‑descent (exponentiated‑gradient) refinement** – the best
   candidate is refined for a long time using the exact sub‑gradient of the
   max‑convolution.  The step‑size is η₀/√t and the parameters live in
   log‑softmax space, guaranteeing non‑negativity and unit sum.

3. **Adam optimisation on a smooth max** – starting from the EG result we
   minimise a smooth approximation of the max (log‑sum‑exp) with Adam.
   The smoothing parameter β is increased gradually (continuation) so that
   the optimisation first sees a very smooth landscape and then focuses on the
   true max.

4. **Targeted stochastic polishing** – the current vector is repeatedly
   altered by moving a small amount of mass from the index participating most
   in the current maximal convolution entry to a low‑mass index.  Accept only
   improving moves, slowly decay the step‑size and keep the best ever seen.

The whole search respects a 1000 s wall‑time limit.  The final vector is stored

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a unit‑ℓ₁ vector: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)               # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine shape: a_i ∝ 1/√((i+0.5)*(n+0.5-i))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray,
             rng: np.random.Generator,
             sigma: float = 0.001) -> np.ndarray:
    """Add small Gaussian noise, clip, renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b, both already ℓ₁‑norm‑1."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s

def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 600_000,
                      eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active shift k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()

    return best_seq, best_val

# ----------------------------------------------------------------------
def _adam_smooth_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          beta_schedule=None,
                          steps_per_beta: int = 2500,
                          lr: float = 0.3):
    """
    Adam optimiser for the smooth approximation of max(conv(a,a)).
    Parameters live in log‑softmax space, guaranteeing non‑negativity
    and unit sum.
    """
    if beta_schedule is None:
        beta_schedule = [1, 5, 20, 80, 200, 500, 1000,
                         2000, 5000, 20000, 50000, 100000,
                         200000, 500000, 1000000]

    v = np.log(start_seq + 1e-30)                     # log‑softmax params
    m = np.zeros_like(v)
    u = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break
            t += 1
            a = _softmax(v)                # ℓ₁‑normalised probabilities
            conv = _conv_fft(a, a)
            shift = np.max(conv)           # for numerical stability
            exp_term = np.exp(beta * (conv - shift))
            sum_exp = exp_term.sum()
            # smoothed objective (not used directly, only for logging)
            # obj = shift + np.log(sum_exp) / beta

            # gradient w.r.t a (as derived in the original code)
            w = exp_term / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            # full convolution of rev_a with w (length = n + (2n-1) -1 = 3n-2)
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # transform gradient to log‑softmax space
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update on v
            m = beta1 * m + (1 - beta1) * grad_v
            u = beta2 * u + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t)
            u_hat = u / (1 - beta2 ** t)
            v = v - lr * m_hat / (np.sqrt(u_hat) + eps)

        # occasional logging
        cur_val = _objective_surrogate(_softmax(v))
        print(f"  [Adam] β={beta:.0e} → surrogate = {cur_val:.6f}")

    return _softmax(v)

def _targeted_polish(seq: np.ndarray,
                     n: int,
                     deadline: float,
                     init_step_frac: float = 0.04) -> np.ndarray:
    """
    Repeatedly move a small fraction of mass from the index that
    contributes most to the current maximal convolution entry to a
    low‑mass index.  Accept only improving moves.
    """
    a = seq.copy()
    cur_val = _objective_surrogate(a)
    step_frac = init_step_frac
    rng = np.random.default_rng(0xC0FFEE)

    it = 0
    while time.time() < deadline:
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # compute contributions a[i] * a[k-i] for the active lag
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        contrib = a[idx_i] * a[rev_idx]

        i_max = idx_i[np.argmax(contrib)]

        # pick a destination index with relatively little mass
        j = rng.integers(0, n)
        if j == i_max:
            continue

        # propose moving a fraction of a[i_max] to a[j]
        delta = step_frac * a[i_max] * rng.uniform(0.5, 1.5)
        delta = min(delta, a[i_max])

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[j] += delta

        new_val = _objective_surrogate(a_new)
        if new_val < cur_val:
            a = a_new
            cur_val = new_val
            step_frac = min(0.20, step_frac * 1.02)
        else:
            step_frac = max(1e-5, step_frac * 0.999)

        it += 1
        if it % 4000 == 0:
            print(f"  [Polish] iter {it:6d}: surrogate = {cur_val:.6f}")

    return a

        _maybe_insert(_gaussian(n, sigma_frac))

    for beta in np.linspace(0.5, 30.0, 30):
        _maybe_insert(_kaiser_window(n, beta))

    for alpha in np.linspace(0.0, 1.0, 15):
        _maybe_insert(_tukey_window(n, alpha))

    # beta‑shape – finer grid on both parameters
    alphas = np.geomspace(0.05, 10.0, 12)
    betas = np.geomspace(0.05, 10.0, 12)
    for a_ in alphas:
        for b_ in betas:
            _maybe_insert(_beta_shape(n, a_, b_))

    # power‑law – focus on the region up to p=0.6 (the sweet spot)
    for p in np.linspace(0.0, 0.65, 25):
        _maybe_insert(_power(n, p))

    # arcsine shape (two‑singular‑edge candidate)
    _maybe_insert(_arcsine(n))

    # Low‑frequency sinusoids
    for nf in (2, 3, 4, 5):
        for _ in range(6):
            _maybe_insert(_lowfreq_curve(n, nf, rng))

    # Dirichlet draws – many concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(4):
            _maybe_insert(_random_dirichlet(n, conc, rng))

    # Exponential draws
    for _ in range(12):
        _maybe_insert(_random_exponential(n, rng))

    # Perturb previous best if it exists
    if isinstance(prev, (list, tuple)):
        arr = np.asarray(prev, dtype=np.float64)
        if arr.shape == (n,) and arr.sum() > 0.0:
            base = _norm(arr)
            _maybe_insert(base)
            for sigma in (0.002, 0.004, 0.008):
                for _ in range(4):
                    _maybe_insert(_perturb(base, rng, sigma=sigma))

    best_heap.sort(key=lambda x: x[0])
    best_seq = best_heap[0][1].copy()
    best_val = best_heap[0][0]
    print("\nTop candidates after sweep (surrogate):")
    for i, (v, _) in enumerate(best_heap[:top_k], 1):
        print(f"  #{i}: surrogate = {v:.6f}")

    # ------------------------------------------------------------------
    # 2. Mirror‑descent (exponentiated gradient) on the best candidate
    # ------------------------------------------------------------------
    eg_deadline = time.time() + max(0.0, deadline - time.time() - 120.0)  # leave ~2 min
    print(f"\n--- Exponentiated‑gradient refinement (until {eg_deadline - start_time:.1f}s) ---")
    best_seq, best_val = _exp_gradient_opt(best_seq, n, eg_deadline,
                                           max_iters=800_000, eta0=5.0)
    print(f"  EG best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam optimisation on smooth max (continuation in β)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("\n--- Adam continuation on smoothed max ---")
        start_vec = np.maximum(best_seq, 1e-12)
        start_vec = start_vec / start_vec.sum()
        seq_adam = _adam_smooth_optimize(start_vec, n, deadline,
                                         beta_schedule=[1, 5, 20, 80,
                                                        200, 500, 1000,
                                                        2000, 5000, 20000,
                                                        50000, 100000,
                                                        200000, 500000,
                                                        1000000],
                                         steps_per_beta=3000,
                                         lr=0.3)
        val_adam = _objective_surrogate(seq_adam)
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print(f"  Adam improvement: surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Targeted stochastic polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n--- Targeted stochastic polishing ---")
        best_seq = _targeted_polish(best_seq, n, deadline, init_step_frac=0.04)
        best_val = _objective_surrogate(best_seq)
        print(f"  Polishing result surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final true evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
          f"true evaluate_sequence = {true_val}")

    # store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
