    return a

def _random_multi_gaussian(n, rng):
    # random number of components 2-5
    K = rng.integers(2,6)
    a = np.zeros(n, dtype=np.float64)
    for _ in range(K):

        center_frac = rng.uniform(0.2,0.8)
        sigma = sigma_frac * n
        mu = center_frac * (n-1)
        i = np.arange(n, dtype=np.float64)
        a += np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s

"""
Optimiser for the step‑function evaluation problem.

The objective for a non‑negative vector `a` with Σa = 1 is

    J(a) = 2·n·max(conv(a,a))

where `conv` is the full linear convolution.
`J` is scale‑invariant, so we work with ℓ₁‑normalised vectors.
The search consists of three stages

1️⃣  Rich seed generation – analytic windows (uniform, triangular,
    sine, Gaussian, logistic, exponential, Kaiser, Tukey, etc.),
    random Dirichlet draws and randomly placed Gaussian mixtures.

2️⃣  Fast polishing – for the `k` best seeds we run a long
    exponentiated‑gradient (mirror‑descent) optimiser on the *exact*
    max‑convolution objective.  The update is

        a ← softmax(log a – η·g)          (g is the sub‑gradient)

    with a √t‑decay stepsize η₀/√t.

3️⃣  Discrete “hill‑climbing” – after polishing we perform a limited
    number of random swaps of two entries (the ℓ₁‑norm stays 1) and
    keep any swap that reduces the surrogate.  This re‑orders the
    masses and often lowers the peak of the anti‑diagonal sums.

All work respects a global time budget (default 1000 s).  The best
sequence found is stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with Σ = 1:
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-γ·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    """Logistic decreasing shape: 1/(1+exp(steep·(x‑shift)))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sample from a symmetric Dirichlet."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_multi_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random mixture of 2‑5 Gaussians with random centres / widths."""
    k = rng.integers(2, 6)               # number of components
    a = np.zeros(n, dtype=np.float64)
    for _ in range(k):
        sigma_frac = rng.uniform(0.04, 0.12)
        centre_frac = rng.uniform(0.15, 0.85)
        sigma = sigma_frac * n
        mu = centre_frac * (n - 1)
        i = np.arange(n, dtype=np.float64)
        a += np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on  J(a)=2·n·max(conv(a,a)).
    Returns the best vector seen and its surrogate value.
    """
    a = start_seq.copy()
    # initialise log‑probabilities
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation hill‑climbing (re‑order the mass)
# ----------------------------------------------------------------------
def _perm_hill_climb(seq: np.ndarray,
                     n: int,
                     deadline: float,
                     steps: int = 2000,
                     rng: np.random.Generator = None) -> tuple[np.ndarray, float]:
    """
    Randomly swap two entries; keep the swap if the surrogate improves.
    Very cheap – each evaluation is an FFT‑based convolution.
    """
    if rng is None:
        rng = np.random.default_rng()
    a = seq.copy()
    best_val = _objective_surrogate(a)
    best_seq = a.copy()

    for _ in range(steps):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # swap
        a[i], a[j] = a[j], a[i]
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        else:
            # revert
            a[i], a[j] = a[j], a[i]

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (length 4000) that minimises
    `evaluate_sequence`.  Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    n = 4000

    # ------------------------------------------------------------------
    # 1. Seed generation (rich analytic & stochastic families)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for g in np.linspace(0.5, 5.0, 9):
        seeds.append(_sine_window(n, gamma=g))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for s in np.linspace(0.04, 0.30, 9):
        seeds.append(_gaussian(n, sigma_frac=s))

    for steep in np.linspace(0.5, 20.0, 8):
        for shift in np.linspace(0.1, 0.9, 5):
            seeds.append(_logistic_decay(n, steep, shift))

    for beta in np.linspace(1.0, 15.0, 7):
        seeds.append(_kaiser_window(n, beta))

    for a in np.linspace(0.15, 0.85, 7):
        seeds.append(_tukey_window(n, a))

    # random Dirichlet draws (different concentrations)
    for conc in (0.01, 0.03, 0.07, 0.15, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random mixtures of Gaussians
    for _ in range(12):
        seeds.append(_random_multi_gaussian(n, rng))

    # include the previous best if present
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape == (n,):
                seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Refinement of the best seeds (long EG + permutation hill‑climb)
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # allocate a share of the remaining budget
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.45 / (top_k - idx)

        print(f"\n[Refine #{idx+1}] starting from seed (surrogate {_objective_surrogate(seed_vec):.6f})")

        # ---- EG polishing (most work) ----
        seq_eg, val_eg = _exp_gradient_opt(seed_vec,
                                            n,
                                            sub_deadline,
                                            max_iters=500_000,
                                            eta0=6.0)

        # ---- permutation hill‑climbing (cheap) ----
        if time.time() < sub_deadline:
            seq_hc, val_hc = _perm_hill_climb(seq_eg, n,
                                              sub_deadline,
                                              steps=2000,
                                              rng=rng)
        else:
            seq_hc, val_hc = seq_eg, val_eg

        # pick the best among raw seed, EG, HC
        candidates = [(seed_vec, _objective_surrogate(seed_vec)),
                      (seq_eg, val_eg),
                      (seq_hc, val_hc)]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Refine #{idx+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < best_val:
            best_val = local_best_val
            best_seq = local_best_seq.copy()
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long polishing on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        fin_deadline = time.time() + (deadline - time.time()) * 0.9
        seq_final, val_final = _exp_gradient_opt(best_seq,
                                                 n,
                                                 fin_deadline,
                                                 max_iters=1_000_000,
                                                 eta0=5.0)
        if val_final < best_val:
            best_val = val_final
            best_seq = seq_final
            print(f"[Final EG] improved to {best_val:.6f}")

        # final permutation hill‑climb
        if time.time() < deadline:
            print("\n--- Final permutation hill‑climb ---")
            seq_final2, val_final2 = _perm_hill_climb(best_seq,
                                                      n,
                                                      deadline,
                                                      steps=5000,
                                                      rng=rng)
            if val_final2 < best_val:
                best_val = val_final2
                best_seq = seq_final2
                print(f"[Final HC] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Exact evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for next run (if any)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
