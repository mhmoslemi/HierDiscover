"""
propose_candidate
-----------------

A faster, more exhaustive search for a non‑negative step‑function (probability
vector) that minimises the evaluation function

    2·n·max(conv(p,p))   (equivalently ``evaluate_sequence``)

The algorithm works in four stages:

1. **Very broad seeding** – deterministic families (uniform, various windows,
   beta‑shapes with tiny α,β, symmetric power‑law, …) plus many random
   Dirichlet / exponential samples, all normalised to sum‑1.  This gives a
   few thousand diverse candidates.

2. **Surrogate evaluation** – each seed is scored with the exact surrogate
   objective (no approximation).  The best few are kept.

3. **Refinement** – the current best seed undergoes three optimisation
   phases, all respecting the remaining time budget:
      * Exponentiated‑gradient (mirror descent) on the exact max‑convolution,
        with an added symmetry projection after each iteration.
      * Continuation L‑BFGS on a log‑sum‑exp smooth approximation of the max
        (if SciPy is available).
      * Simple stochastic mass‑transfer local search.

   After each phase the best‑seen vector is kept.

4. **Finalisation** – the result is symmetrised, evaluated with the provided
   ``evaluate_sequence`` and stored in the global ``height_sequence_1`` for
   later reuse.

All vectors are kept on the simplex (non‑negative, sum = 1), therefore the
scaling enforced by ``evaluate_sequence`` has no effect on the objective.
The search uses only NumPy (and SciPy when present) and respects a total
wall‑time of roughly ``budget_s`` seconds (default 1000 s).
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Basic numeric utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Make a vector symmetric w.r.t. the centre and re‑normalise."""
    b = (a + a[::-1]) * 0.5
    s = b.sum()
    if s == 0.0:
        return a.copy()
    return b / s


# ------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector ℓ₁‑normalised to 1:
       2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    return a / a.sum()


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape on a discrete grid."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _symmetric_power(n: int, p: float) -> np.ndarray:
    """Symmetric mixture of two power‑law tails."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (1.0 / (i ** p)) + (1.0 / ((n - i + 1) ** p))
    return a / a.sum()


# ------------------------------------------------------------
# Random families
# ------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    return a / a.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    return a / a.sum()


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


# ------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on exact max
# ------------------------------------------------------------
def _exp_gradient_opt_sym(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 1_500_000,
                         eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution.
    After each soft‑max update the vector is symmetrised and re‑normalised.
    Returns the best sequence seen together with its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space to keep the simplex constraint automatically
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        # index where convolution attains its maximum
        k = int(np.argmax(conv))

        # sub‑gradient of the selected entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        # enforce symmetry
        a = _symmetrize(a)

        # recompute log_a for the next iteration
        log_a = np.log(a + 1e-30)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space, guaranteeing non‑negativity
    and the simplex constraint.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (softmax space)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)          # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Very cheap stochastic local search
# ------------------------------------------------------------
def _local_search(seq: np.ndarray,
                  deadline: float,
                  moves: int = 20000,
                  step_scale: float = 0.02):
    """
    Simple random mass‑transfer hill‑climbing.
    At each move a small amount of probability is moved from one index to another.
    """
    rng = np.random.default_rng()
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    n = len(seq)

    for _ in range(moves):
        if time.time() > deadline:
            break
        eps = step_scale * rng.random()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = min(eps, best_seq[i])
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq = new_seq / new_seq.sum()
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq
    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  Returns the best list found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.5          # safety margin

    n = 4000

    # ------------------------------------------------------------------
    # 1. Very broad seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Generating seeds (n={n}) ---")
    seeds = []

    # deterministic / classic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    seeds.append(_alternating_dirichlet(n, rng))

    # beta‑shapes – include tiny α,β for U‑shaped densities
    alphas = np.concatenate([
        np.geomspace(0.02, 0.2, 5),   # very small to moderate
        np.linspace(0.2, 5.0, 25)    # broader sweep
    ])
    betas = alphas.copy()
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # symmetric power‑law mixtures (U‑shape)
    for p in np.linspace(0.0, 2.5, 15):
        seeds.append(_power(n, p))
        seeds.append(_symmetric_power(n, p))

    # random Dirichlet with very low concentration (spiky at ends)
    for conc in (0.02, 0.01, 0.005, 0.002, 0.001):
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(30):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixture of random shapes with uniform (adds extra diversity)
    for mix_alpha in (0.2, 0.5, 0.8):
        for _ in range(6):
            base = _random_dirichlet(n, 0.1, rng)
            seeds.append(_mix_shapes(_uniform(n), base, mix_alpha))

    # previous best, if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                arr = arr / arr.sum()
                seeds.append(arr)
                for _ in range(5):
                    seeds.append(_perturb(arr, rng, sigma=0.003))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Evaluate all seeds (cheap surrogate)
    # ------------------------------------------------------------------
    print("\n--- Evaluating seeds (surrogate) ---")
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seqs = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop surrogate values after seeding:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Refinement of the best seed
    # ------------------------------------------------------------------
    best_seq = top_seqs[0].copy()
    best_val = _objective_surrogate(best_seq)

    # 3a – Exponentiated‑gradient with symmetry
    print("\n--- EG refinement (symmetrised) ---")
    eg_deadline = deadline - 30.0   # keep some time for later phases
    seq_eg, val_eg = _exp_gradient_opt_sym(best_seq, n, eg_deadline,
                                           max_iters=2_000_000,
                                           eta0=5.0)
    if val_eg < best_val:
        best_val, best_seq = val_eg, seq_eg.copy()
        print(f"EG improvement: {val_eg:.6f}")

    # 3b – Continuation L‑BFGS smoothing (if SciPy is present)
    if time.time() < deadline - 20.0:
        print("\n--- L‑BFGS smoothing ---")
        seq_lb = _lbfgs_smooth_optimize(best_seq, n, deadline)
        seq_lb = _symmetrize(seq_lb)
        val_lb = _objective_surrogate(seq_lb)
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()
            print(f"L‑BFGS improvement: {val_lb:.6f}")

    # 3c – Simple stochastic local search
    if time.time() < deadline - 5.0:
        print("\n--- Local search ---")
        seq_loc, val_loc = _local_search(best_seq, deadline,
                                         moves=50000,
                                         step_scale=0.02)
        seq_loc = _symmetrize(seq_loc)
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"Local‑search improvement: {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 4. Final evaluation
    # ------------------------------------------------------------------
    best_seq = _symmetrize(best_seq)   # final sanity symmetrisation
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised", exc)

    print(f"\n=== Finished ===\nBest surrogate ≈ {best_val:.6f}, "
          f"true evaluate_sequence = {true_val}\nLength = {len(best_seq)}")

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
