"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Deterministic seeding** – a modest set of analytically‑defined
   families (uniform, triangular, beta‑skewed, exponential decay,
   power‑law and a few Dirichlet draws).  All candidates are ℓ₁‑normalised
   and evaluated with the exact objective (the “surrogate” is the true
   value when the sum equals 1).

2. **Focused refinement** – the best few seeds are each refined by
   exponentiated‑gradient mirror descent (with both single‑active and
   full‑active sub‑gradient versions), followed by a continuation L‑BFGS
   optimisation of a smooth log‑sum‑exp surrogate and a short simulated
   annealing pass.  The current global best is updated whenever a
   improvement is found.

3. **Long polishing** – the final global best is run through a very long
   single‑active exponentiated‑gradient phase and a final simulated‑annealing
   sweep.  All work respects the 1000 s wall‑clock budget (with a 2 s
   safety margin).  The resulting list is stored in the global variable
   ``height_sequence_1`` and returned.

The implementation uses only NumPy and (optionally) SciPy.  All
operations stay in double precision; no file or network I/O is performed.
"""

import time
import math
import numpy as np

# Optional SciPy import – used for the L‑BFGS smoothing stage.
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation for a vector with sum = 1:
        2·n·max(conv(seq, seq))
    This equals `evaluate_sequence` when the sum is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all returned ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta distribution on indices 1…n (ℓ₁‑normalised)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), i = 0…n‑1, then ℓ₁‑normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p, ℓ₁‑normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet distribution (ℓ₁‑normalised by construction)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture (1‑α)·a + α·b, renormalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 5.0,
                      single_active: bool = False) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    `single_active` uses only the most active index for the sub‑gradient,
    otherwise all near‑max indices are used.
    Returns the best sequence seen and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # work in log‑space; the soft‑max guarantees positivity & ℓ₁‑norm
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # ----- active set -------------------------------------------------
        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        # ----- sub‑gradient ------------------------------------------------
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            # exact sub‑gradient of the max (average over active indices)
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # ----- learning‑rate (1/√t decay) ---------------------------------
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max yielding a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 1200) -> np.ndarray:
    """
    Optimises a smooth approximation of the max‑convolution using L‑BFGS‑B.
    Works in log‑softmax variables to keep feasibility.
    """
    if _opt is None:
        # SciPy unavailable – return unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)  # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

            # Gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp                     # weights over convolution indices
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # slice back to length n

            # Chain‑rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 120_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.02,
                         T_end: float = 1e-7) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

        if time.time() >= deadline:
            break

        # Allocate a modest slice of the remaining budget for this seed
        remaining = deadline - time.time()
        # Give at most 12 % of the remaining time to a single seed
        sub_deadline = time.time() + min(remaining * 0.12, 30.0)

        # (a) Exponentiated‑gradient with full active set
        seq_eg_full, val_eg_full = _exp_gradient_opt(
            seed_vec.copy(),
            n,
            sub_deadline,
            max_iters=180_000,
            eta0=8.0,
            single_active=False,
        )

        # (b) Short L‑BFGS smoothing (smooth max with increasing β)
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(
                seq_eg_full,
                n,
                sub_deadline,
                beta_schedule=[1.0, 10.0, 50.0, 200.0, 800.0, 3_000.0, 12_000.0],
                max_iter_per_beta=800,
            )
            val_lbfgs = _objective(seq_lbfgs)
        else:
            seq_lbfgs = seq_eg_full
            val_lbfgs = val_eg_full

        # (c) Single‑active EG for a finer descent
        if time.time() < sub_deadline:
            seq_eg_sa, val_eg_sa = _exp_gradient_opt(
                seq_lbfgs.copy(),
                n,
                sub_deadline,
                max_iters=120_000,
                eta0=6.0,
                single_active=True,
            )
        else:
            seq_eg_sa, val_eg_sa = seq_lbfgs, val_lbfgs

        # (d) Brief simulated‑annealing polishing
        hill_deadline = sub_deadline - 0.3
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(
                seq_eg_sa,
                n,
                hill_deadline,
                rng,
                max_attempts=80_000,
                start_sigma=0.002,
                T_start=0.015,
                T_end=5e-8,
            )
        else:
            seq_sa, val_sa = seq_eg_sa, val_eg_sa

        # Pick the best among the candidates generated for this seed
        candidates = [
            (seed_vec, _objective(seed_vec)),
            (seq_eg_full, val_eg_full),
            (seq_lbfgs, val_lbfgs),
            (seq_eg_sa, val_eg_sa),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long polishing – heavy single‑active EG + SA
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (single‑active) ---")
        final_seq, final_val = _exp_gradient_opt(
            global_best_seq,
            n,
            deadline,
            max_iters=2_500_000,
            eta0=6.0,
            single_active=True,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final SA polishing ---")
        final_sa_seq, final_sa_val = _simulated_annealing(
            global_best_seq,
            n,
            deadline,
            rng,
            max_attempts=100_000,
            start_sigma=0.001,
            T_start=5e-3,
            T_end=1e-9,
        )
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Store and return the best sequence
    # --------------------------------------------------------------
    globals()["height_sequence_1"] = global_best_seq.tolist()
    print(f"\n=== Finished ===\nBest objective = {global_best_val:.6f}")
    return global_best_seq.tolist()
