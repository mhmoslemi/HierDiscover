"""
Propose a non‑negative step function that minimizes
    2 * n * max(conv(a, a)) / (sum(a)**2)

The implementation follows a three‑stage pipeline:

1. **Seeding** – a rich collection of analytic families (uniform,
   triangular, sine‑window, beta‑shapes, exponential decays,
   symmetric exponentials, power‑laws, random Dirichlet draws and mixes)
   is generated, ℓ₁‑normalised and evaluated with a cheap FFT‑based surrogate.
   The best few seeds are kept.

2. **Refinement** – the best seed is improved by
   * exponentiated‑gradient mirror descent (full‑active then single‑active)
   * a continuation L‑BFGS optimiser on a smooth log‑sum‑exp surrogate
   * a low‑temperature simulated‑annealing pass.

   All optimisation steps respect the global 1000 s deadline and are
   stopped early if time runs out.

3. **Final evaluation** – the best sequence found is passed to the
   provided ``evaluate_sequence`` and stored globally as ``height_sequence_1``.

The algorithm is deliberately different from the reference implementation:
it concentrates optimisation effort on a *single* promising seed,
uses both full‑active and single‑active sub‑gradient steps,
adds extra analytic families (symmetric exponentials and beta‑mixtures),
and allocates a longer polishing phase.  The search should therefore
push the bound below the previous 1.5085 value.
"""

import time
import math
import numpy as np

# ---------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ---------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover

# ---------------------------------------------------------------
# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector, sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ---------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ---------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ---------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ---------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_mixture(n: int, alpha: float, beta: float) -> np.ndarray:
    """Convex mixture of a left‑skewed and its mirrored counterpart."""
    a = _beta_shape(n, alpha, beta)
    b = _beta_shape(n, beta, alpha)
    mixed = (a + b) * 0.5
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i), i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """Symmetric sum of a left‑ and a right‑exponential decay."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ---------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ---------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 5.0,
                      single_active: bool = False):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    *single_active*: if True only the index attaining the maximal
    convolution value is used for the sub‑gradient.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space; the soft‑max guarantees positivity & ℓ₁‑norm
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # ----- active set -------------------------------------------------
        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        # ----- sub‑gradient accumulator ----------------------------------
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            # exact sub‑gradient of the max is the *average* of the active ones
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # ----- learning‑rate (1/√t decay) -------------------------------
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ---------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ---------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 1200):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Operates in the log‑softmax space, guaranteeing feasibility.
    """
    if _opt is None:
        return start_vec  # SciPy unavailable – return unchanged

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp                     # weights over each conv index
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # back‑to‑size‑n slice

            # chain rule for log‑softmax (mirror‑descent)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ---------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ---------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.02,
                         T_end: float = 1e-7):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ---------------------------------------------------------------
# Main entry point
# ---------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under a 1000 s budget.
    Returns a plain ``list[float]`` and stores the result in the
    global variable ``height_sequence_1`` for possible reuse.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a slightly larger n gives the optimiser more freedom
    # while still keeping per‑iteration cost modest.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Load a previous solution if available (helps continuity across runs)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Deterministic seeding – systematic grids over a few families
    # ------------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # simple analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (0.5, 0.8, 1.0, 1.3, 1.6, 2.0):
        seeds.append(_sine_window(n, gamma))

    # beta shapes (including symmetric mixtures)
    alphas = np.linspace(0.2, 8.0, 20)
    betas = np.linspace(0.2, 8.0, 20)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))
            seeds.append(_beta_mixture(n, a, b))

    # exponential decays (left‑hand and symmetric)
    rates = np.logspace(-4, -0.2, 30)
    for r in rates:
        seeds.append(_exp_decay(n, r))
        seeds.append(_sym_exp_decay(n, r))

    # power‑law family
    ps = np.linspace(0.0, 5.0, 25)
    for p in ps:
        seeds.append(_power(n, p))

    # random Dirichlet draws (including very sparse priors)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # perturbations of a previously known best (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        base = prev_arr / prev_arr.sum()
        seeds.append(base)
        for _ in range(6):
            pert = base + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.clip(vec, 0.0, None)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(7, len(evaluated))   # keep a modest number of promising seeds
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Focused refinement on the *single* best seed
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # ---- a) Exponentiated‑gradient, full active set -----------------
        seq_fg, val_fg = _exp_gradient_opt(global_best_seq.copy(),
                                           n,
                                           deadline,
                                           max_iters=400_000,
                                           eta0=7.0,
                                           single_active=False)
        if val_fg < global_best_val:
            global_best_seq, global_best_val = seq_fg, val_fg
            print(f"[Full‑active EG] improved to {global_best_val:.6f}")

        # ---- b) Continuation L‑BFGS smoothing -------------------------
        if time.time() < deadline:
            seq_lb = _lbfgs_smooth_optimize(global_best_seq,
                                            n,
                                            deadline,
                                            max_iter_per_beta=2000)
            val_lb = _objective_surrogate(seq_lb)
            if val_lb < global_best_val:
                global_best_seq, global_best_val = seq_lb, val_lb
                print(f"[L‑BFGS] improved to {global_best_val:.6f}")

        # ---- c) Exponentiated‑gradient, single active (fine tune) -----
        if time.time() < deadline:
            seq_sa, val_sa = _exp_gradient_opt(global_best_seq.copy(),
                                               n,
                                               deadline,
                                               max_iters=1_200_000,
                                               eta0=5.0,
                                               single_active=True)
            if val_sa < global_best_val:
                global_best_seq, global_best_val = seq_sa, val_sa
                print(f"[Single‑active EG] improved to {global_best_val:.6f}")

        # ---- d) Final simulated‑annealing polish ----------------------
        if time.time() < deadline:
            final_seq, final_val = _simulated_annealing(global_best_seq,
                                                        n,
                                                        deadline,
                                                        rng,
                                                        max_attempts=120_000,
                                                        start_sigma=0.001,
                                                        T_start=0.01,
                                                        T_end=1e-9)
            if final_val < global_best_val:
                global_best_seq, global_best_val = final_seq, final_val
                print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
