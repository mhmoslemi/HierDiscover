"""
Optimisation of a non‑negative step‑function for the evaluation metric
``evaluate_sequence``.

The algorithm proceeds as follows

1. **Very rich seed pool** – analytic windows (uniform, triangular,
   sine‑, cosine‑, Gaussian, exponential, power‑law, beta‑shape),
   random Dirichlet vectors (various concentrations) and a few
   random piecewise‑linear curves.  All vectors are ℓ₁‑normalised.

2. **Select the most promising seeds** – keep the best `top_k`
   according to the exact surrogate
   ``2·n·max(conv(a,a))`` (this is equal to ``evaluate_sequence``
   for ℓ₁‑normalised vectors).

3. **Intensive mirror‑descent (exponentiated‑gradient) polishing**
   of each selected seed.  The update is performed on the log‑domain
   (soft‑max) using the sub‑gradient of the active convolution entry.
   A relatively large initial step‑size (`eta0`) and a `1/√t`
   decay are used.

4. **Optional L‑BFGS smoothing** – a smooth max‑convolution
   (log‑sum‑exp) is minimised with SciPy’s L‑BFGS‑B if SciPy is
   available.  The β‑schedule is fairly long, each stage is given
   several thousand iterations.

5. **Final long mirror‑descent** – the current global best is
   refined for the rest of the time budget.

6. **Random‑restart polishing** – several short restarts consisting of
   a small random perturbation followed by a short mirror‑descent run.

All heavy work respects the overall `budget_s` seconds limit.
The best sequence is stored in the global variable ``height_sequence_1``
so that later runs can continue from it.
"""

import time
import math
import numpy as np

# ---------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS phase.
# ---------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:       # pragma: no cover

# ---------------------------------------------------------
# Numerical utilities (FFT‑based full convolution)
# ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ---------------------------------------------------------
# Exact surrogate for ℓ₁‑normalised vectors:
#   2·n·max(conv(a,a))
# ---------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ---------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ---------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(np.pi * i / (n - 1))) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.dirichlet(np.full(n, conc))
    thresh = np.percentile(vec, 80)
    vec = np.where(vec >= thresh, vec, 0.0)
    total = vec.sum()
    if total == 0.0:
        return _uniform(n)
    return vec / total


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _perturb(seq: np.ndarray,
             rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add small Gaussian noise and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ---------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑conv.
# ---------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ---------------------------------------------------------
# L‑BFGS smoothing (log‑sum‑exp approximation of max)
# ---------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """Minimise a smoothed max‑convolution via L‑BFGS‑B."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ---------------------------------------------------------
# Main entry point
# ---------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best ``list[float]`` found before the timeout.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Problem dimension – a compromise between quality and speed.
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if any) – useful for continuation runs.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Massive seed generation (analytic families + random draws)
    # ------------------------------------------------------------------
    seeds = []

    # Analytic families – fairly dense grids
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # Sine and cosine windows – sweep exponents
    for gamma in np.linspace(0.1, 4.0, 30):
        seeds.append(_sine_window(n, gamma=gamma))
        seeds.append(_cosine_window(n, gamma=gamma))

    # Gaussian windows
    for sigma_frac in np.linspace(0.05, 0.35, 20):
        seeds.append(_gaussian(n, sigma_frac))

    # Power‑law tails
    for p in np.linspace(0.0, 4.0, 20):
        seeds.append(_power(n, p))

    # Exponential decay – fine log‑spaced sweep
    for rate in np.logspace(-5, -0.3, 50):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Beta‑shape – many random draws
    for _ in range(150):
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws – diverse concentration values
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(20):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Very sparse Dirichlet – extreme sparsity
    for conc in [0.001, 0.005, 0.01]:
        for _ in range(10):
            seeds.append(_random_sparse_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(60):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(15):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Piecewise‑linear random curves
    for breaks in (3, 5, 8):
        for _ in range(15):
            seeds.append(_piecewise_linear(n, breaks, rng))

    # Mixtures of uniform with a few shapes
    mix_alphas = (0.2, 0.5, 0.8)
    for alpha in mix_alphas:
        for rate in (0.001, 0.005, 0.02):
            curve = _exp_decay(n, rate)
            seeds.append(_mix_shapes(_uniform(n), curve, alpha))

    # Random permutations of a few base shapes – break symmetry aggressively
    base_shapes = [_exp_decay(n, 0.003), _sym_exp_decay(n, 0.004),
                   _beta_shape(n, 2.0, 2.0), _gaussian(n, 0.18)]
    for base in base_shapes:
        for _ in range(15):
            seeds.append(np.random.permutation(base))

    # Include the previous best (if any) and a few perturbed copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(12):
            seeds.append(_perturb(prev_arr, rng, sigma=0.004))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the top‑k seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best from the very best seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refinement of the top seeds (heavy EG + optional L‑BFGS)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # Allocate a larger share to the absolute best seed
        if si == 0:
            share = remaining * 0.55
        else:
            share = remaining * 0.45 / max(1, (top_k - 1))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] time budget ≈ {share:.1f}s")

        # ---- (a) exponentiated‑gradient polishing ----
        seq_eg, val_eg = _exp_gradient_opt(seed_vec,
                                           n,
                                           sub_deadline,
                                           max_iters=400_000,
                                           eta0=8.0)

        # ---- (b) optional L‑BFGS smoothing ----
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               max_iter_per_beta=4000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_eg, val_eg

        # Keep the best variant for this seed
        local_best_seq, local_best_val = min(
            [(seed_vec, _objective_surrogate(seed_vec)),
             (seq_eg, val_eg),
             (seq_lbfgs, val_lbfgs)],
            key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long EG polishing (as much time as left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_800_000,
                                                 eta0=6.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Random‑restart polishing (few short runs)
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # small random perturbation followed by a short EG run
        pert_seq = _perturb(global_best_seq, rng, sigma=0.01)
        seq2, val2 = _exp_gradient_opt(pert_seq,
                                       n,
                                       deadline,
                                       max_iters=150_000,
                                       eta0=7.0)
        if val2 < global_best_val:
            global_best_seq = seq2
            global_best_val = val2
            print(f"[Restart] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation via evaluate_sequence
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
