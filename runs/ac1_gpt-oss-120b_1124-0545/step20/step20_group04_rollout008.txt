    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2

    n = 4000

    # 1. Generate seed pool from analytic families with fine grids
    seeds = []
    # Uniform
    seeds.append(_uniform(n))
    # Triangular not needed maybe

    for gamma in np.linspace(0.05, 10.0, 200):
        seeds.append(_exp_decay(n, gamma))
    # Logistic: steep (logspace)
    for steep in np.logspace(np.log10(0.5), np.log10(200), 100):
        for shift in np.linspace(0.1, 0.9, 9):
            seeds.append(_logistic_decay(n, steep, shift))
    # Power
    for p in np.linspace(0.0, 5.0, 101):
        seeds.append(_power(n, p))
    # Gaussian sigma
    for sigma_frac in np.linspace(0.05, 0.3, 51):
        seeds.append(_gaussian(n, sigma_frac))
    # Beta shape
    for a_ in np.linspace(0.05, 10.0, 41):
        for b_ in np.linspace(0.05, 10.0, 41):
            seeds.append(_beta_shape(n, a_, b_))
    # Tukey
    for alpha in np.linspace(0.0, 1.0, 51):
        seeds.append(_tukey_window(n, alpha))

    # Maybe also add random Dirichlet draws.
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Remove duplicates (if shape may produce same). Not needed.

    # Evaluate all seeds quickly with surrogate (objective without scaling). Keep top 20
    scored = []
    for s in seeds:
        if time.time()>deadline:
            break
        val = _objective_surrogate(s)
        scored.append((s,val))
    scored.sort(key=lambda x:x[1])
    top_k = 20
    best_seq, best_val = scored[0][0], scored[0][1]

    print(f"Initial best surrogate: {best_val}")

    # 2. Hill climbing using random perturbations and exponential gradient.

    # Use exponentiated gradient (EG) longer
    # We'll implement EG with step size schedule derived from iteration.

    a = best_seq.copy()
    log_a = np.log(a + 1e-30)
    best_surrogate = best_val
    best_seq = a.copy()
    iteration = 0
    # approximate runtime: each iteration conv ~0.001 sec --> 500k iterations ~500 sec.

    while time.time() < deadline:
        # compute gradient subgradient at active index
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        g = np.zeros_like(a)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        iteration += 1
        eta = 5.0 / math.sqrt(iteration)
        log_a -= eta * g
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_surrogate:
            best_surrogate = cur_val
            best_seq = a.copy()
            print(f"Iter {iteration}: improved surrogate to {best_surrogate}")
        # Additionally periodically random perturbation to escape local minima:
        if iteration % 20000 == 0:
            # jitter

            log_a = np.log(a + 1e-30)
    # After loop, compute true objective
    true_val = evaluate_sequence(best_seq.tolist())
    print(f"Finished: surrogate {best_surrogate}, true {true_val}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
Improved step‑function search for `evaluate_sequence`.

Algorithm
---------

1️⃣ **Rich seed pool** – Analytic windows (exponential, logistic,
   beta, power, Gaussian, Tukey), dense parameter sweeps and a few
   Dirichlet draws. Every seed is ℓ₁‑normalised (sum = 1) so that the
   surrogate `2·n·max(conv)` equals the true objective.

2️⃣ **Quick surrogate ranking** – All seeds are evaluated with the fast
   FFT‑based convolution. The best *k* seeds are retained.

3️⃣ **Local refinement** – For each of the retained seeds we perform
   *a* a short L‑BFGS run on a smooth log‑sum‑exp approximation of the
   max‑convolution (β is increased gradually) and *b* an
   exponentiated‑gradient (EG) run on the exact max‑convolution.
   The best result across all refinements is kept.

4️⃣ **Final polishing** – A long EG run (as many iterations as the
   remaining budget allows) followed by a final high‑β L‑BFGS pass.

All heavy work respects a global deadline (default 1000 s).  The
resulting list of heights is stored in the global variable
`height_sequence_1` and returned.

The code uses only NumPy and SciPy (optional) and runs on a single
CPU.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)

try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-γ·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    """Logistic decreasing shape: 1/(1+exp(steep·(x‑shift)))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sample from a symmetric Dirichlet."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 6000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space so we stay on the probability simplex.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1_000.0,
                         2_000.0, 5_000.0, 10_000.0,
                         20_000.0, 50_000.0, 100_000.0,
                         200_000.0, 500_000.0, 1_000_000.0,
                         2_000_000.0, 5_000_000.0, 1e7, 2e7, 5e7, 1e8]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # log‑sum‑exp smoothing (stable)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # Guard against under‑flow (all zeros)
            if sum_exp == 0.0:
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
            else:
                w = exp_beta / sum_exp
            # gradient w.r.t. a (see derivation in the analysis)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] Exception {e} at β={beta_val}")
            break

    return _softmax(v)


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution.
    Returns the best sequence seen.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 50000 == 0:
                print(f"[EG] iteration {it:,} – new best surrogate {best_val:.6f}")

    return best_seq, best_val

            seeds.append(_logistic_decay(n, steep, shift))

    # power law – fine grid up to p=5
    for p in np.linspace(0.0, 5.0, 101):
        seeds.append(_power(n, p))

    # beta‑shape – moderate grid on (α,β)
    for a_ in np.linspace(0.05, 5.0, 21):
        for b_ in np.linspace(0.05, 5.0, 21):
            seeds.append(_beta_shape(n, a_, b_))

    # Tukey window – very fine grid
    for alpha in np.linspace(0.0, 1.0, 41):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws with very low concentration (spiky candidates)
    for conc in (0.001, 0.003, 0.005, 0.01, 0.02, 0.05):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Perturbations of the previous best (if any) – these often sit close to a local optimum
    if prev_arr is not None and prev_arr.sum() > 0.0:
        prev_norm = prev_arr / prev_arr.sum()
        seeds.append(prev_norm)
        for _ in range(6):
            seeds.append(_perturb(prev_norm, rng, sigma=0.005))

    # ------------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation ({len(seeds)} candidates) ---")
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(30, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds after surrogate evaluation:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:02d}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Local refinement of the best few seeds
    # ------------------------------------------------------------------
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # time earmarked for refinement
    mix_deadline = deadline - 1.0
    num_refine = min(5, len(top_seeds))

    for idx in range(num_refine):
        if time.time() >= deadline:
            break
        seed_vec = top_seeds[idx]
        # allocate a slice of the remaining budget
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.4 / (num_refine - idx)

        print(f"\n[Refine #{idx+1}] start surrogate = {_objective_surrogate(seed_vec):.6f}")

        # ---- L‑BFGS smoothing (more thorough) ----
        seq_lbfgs = _lbfgs_smooth_optimize(seed_vec,
                                           n,
                                           sub_deadline,
                                           beta_schedule=None,
                                           max_iter_per_beta=8000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)

        # ---- EG polishing (as many iterations as time permits) ----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs,
                                               n,
                                               sub_deadline,
                                               max_iters=250_000,
                                               eta0=8.0)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # pick the best among the three
        candidates = [(seed_vec, _objective_surrogate(seed_vec)),
                      (seq_lbfgs, val_lbfgs),
                      (seq_eg, val_eg)]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Refine #{idx+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < best_val:
            best_seq = local_best_seq
            best_val = local_best_val
            print(f"  >>> Updated global best → {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long EG polishing on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        seq_final, val_final = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_000_000,
                                                 eta0=10.0)
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Final EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final high‑β L‑BFGS smoothing (tight approximation of the max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final high‑β L‑BFGS smoothing ---")
        seq_final_lbfgs = _lbfgs_smooth_optimize(best_seq,
                                                n,
                                                deadline,
                                                beta_schedule=None,
                                                max_iter_per_beta=12_000)
        val_final_lbfgs = _objective_surrogate(seq_final_lbfgs)
        if val_final_lbfgs < best_val:
            best_seq = seq_final_lbfgs
            best_val = val_final_lbfgs
            print(f"[Final L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the official function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
