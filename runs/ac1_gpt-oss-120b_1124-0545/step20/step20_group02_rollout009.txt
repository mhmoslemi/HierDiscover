"""
Propose a non‑negative step‑function (height sequence) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The algorithm works in three phases:

1. **Exploration** – generate a diverse set of analytic / random
   initial shapes (sine windows, Gaussians, exponential decays,
   Beta‑shapes, low‑frequency random curves, Dirichlet samples) and run a
   short *projected gradient descent* (PGD) on each of them.  PGD uses the
   exact max‑convolution sub‑gradient, projects onto the probability
   simplex and monotonically decreases the surrogate objective
   ``2·n·max(conv)``.  The best sequence found becomes the seed for the
   next phase.

2. **Intensive polishing** – a long PGD run with a decaying step size.
   The gradient is taken w.r.t. the *full* active set (all shifts whose
   convolution value is within a tiny relative tolerance of the current
   maximum).  The step size follows ``α = α₀ / sqrt(t)``; the projection
   onto the simplex guarantees feasibility (non‑negative, sum = 1).

3. **Fine‑tuning** – a lightweight simulated‑annealing (SA) pass that
   perturbs the best sequence with a temperature‑controlled Gaussian
   noise and accepts occasional worsenings.  This helps escape any tiny
   local plateaus left by PGD.

All work respects the global time budget (default 1000 s).  The final
sequence is stored in the global variable ``height_sequence_1`` and
returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray, n: int) -> float:
    """Exact surrogate 2·n·max(conv(seq, seq)) for a sequence with sum = 1."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x ≥ 0, sum x = 1}.  Implements the algorithm of Duchi et al.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    # quick exit for already feasible vectors
    if v.min() >= 0 and abs(v.sum() - 1.0) <= 1e-12:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise to exactly 1
    w_sum = w.sum()
    if w_sum <= 0:
        # fallback to uniform
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors, i.e. sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Projected Gradient Descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _pgd_opt(start_seq: np.ndarray,
             n: int,
             deadline: float,
             max_iters: int = 250_000,
             alpha0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simple projected gradient descent on the exact max‑convolution.
    The update rule is:
        a ← Π_simplex(a - α·g)
    where g is the sub‑gradient of max(conv(a,a)) taking *all*
    active shifts (within a tiny relative tolerance of the current
    maximum) into account.  The step size α decays like α₀/√t.
    Returns the best sequence and its surrogate value.
    """
    a = start_seq.copy()
    cur_obj = _objective_surrogate(a, n)
    best_seq = a.copy()
    best_obj = cur_obj

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # 1) compute convolution and the active set
        conv = _conv_fft(a, a)
        max_val = conv.max()
        eps = max_val * 1e-7
        active = np.where(conv >= max_val - eps)[0]

        # 2) sub‑gradient for the full active set (no averaging)
        active_mask = np.zeros_like(conv)
        active_mask[active] = 1.0
        rev_a = a[::-1]
        grad_full = _conv_fft(rev_a, active_mask)          # length = 3n‑2
        g = 2.0 * grad_full[n - 1 : n - 1 + n]              # shape (n,)

        # 3) step size schedule
        alpha = alpha0 / math.sqrt(it)

        # 4) project onto the simplex
        candidate = a - alpha * g
        candidate = _project_simplex(candidate)

        cand_obj = _objective_surrogate(candidate, n)

        # 5) accept only if we improve (monotone descent)
        if cand_obj < cur_obj:
            a = candidate
            cur_obj = cand_obj
            if cand_obj < best_obj:
                best_obj = cand_obj
                best_seq = candidate.copy()

        # occasional early stop if improvement stalls
        if it % 20_000 == 0 and best_obj < cur_obj * 0.9999:
            pass   # placeholder for possible logging

    return best_seq, best_obj


# ----------------------------------------------------------------------
# Simple simulated annealing polishing (same idea as the reference code)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 120_000,
                         start_sigma: float = 0.001,
                         T_start: float = 5e-3,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbations with a linearly decreasing temperature.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a, n)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns, n)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best sequence in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # spare a margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0.0:
                    prev_arr /= s
                else:
                    prev_arr = None
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # ------------------------------------------------------------------
    # Initialise best sequence
    # ------------------------------------------------------------------
    best_seq = None
    best_obj = np.inf

    if prev_arr is not None:
        best_seq = prev_arr.copy()
        best_obj = _objective_surrogate(best_seq, n)
        print("[Info] Loaded previous best sequence (obj = {:.6f})".format(best_obj))
    else:
        # start from a sane analytic shape
        best_seq = _sine_window(n, gamma=1.2)
        best_obj = _objective_surrogate(best_seq, n)
        print("[Info] No previous solution – starting from sine window (obj = {:.6f})".format(best_obj))

    # ------------------------------------------------------------------
    # Phase 1 – exploration with many seeds + short PGD
    # ------------------------------------------------------------------
    exploration_funcs = []

    # deterministic families with a few parameter choices
    for g in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0):
        exploration_funcs.append(lambda g=g: _sine_window(n, gamma=g))
    for s in (0.05, 0.10, 0.20, 0.35):
        exploration_funcs.append(lambda s=s: _gaussian(n, sigma_frac=s))
    for r in (0.001, 0.003, 0.006, 0.010):
        exploration_funcs.append(lambda r=r: _exp_decay(n, rate=r))
    for r in (0.001, 0.003, 0.006, 0.010):
        exploration_funcs.append(lambda r=r: _sym_exp_decay(n, rate=r))
    for a, b in [(0.6, 0.6), (1.0, 1.0), (2.0, 2.0), (0.5, 2.0), (2.0, 0.5)]:
        exploration_funcs.append(lambda a=a, b=b: _beta_shape(n, alpha=a, beta=b))

    # random Dirichlet families
    for conc in (0.1, 0.3, 0.5, 1.0, 2.0):
        for _ in range(3):
            exploration_funcs.append(lambda conc=conc: _random_dirichlet(n, conc, rng))

    # low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            exploration_funcs.append(lambda nf=nf: _lowfreq_curve(n, num_freq=nf, rng=rng))

    # Run the short PGD runs
    short_iters = 8_000
    for idx, maker in enumerate(exploration_funcs, 1):
        if time.time() > deadline:
            break
        try:
            seq = maker()
        except Exception:
            continue
        seq = np.asarray(seq, dtype=np.float64)
        seq = np.maximum(seq, 0.0)
        seq_sum = seq.sum()
        if seq_sum == 0.0:
            continue
        seq /= seq_sum

        cur_seq, cur_obj = _pgd_opt(seq, n, deadline,
                                    max_iters=short_iters,
                                    alpha0=0.5)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = cur_seq.copy()
            print(f"[Exploration] New best after seed {idx}: obj = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – intensive polishing (long PGD)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # a rough estimate: 0.001 s per iteration → allocate up to 200 k iters
        long_iters = min(200_000, int(remaining / 0.001))
        print(f"\n--- Long PGD polishing (≈{long_iters} iters) ---")
        best_seq, best_obj = _pgd_opt(best_seq, n, deadline,
                                      max_iters=long_iters,
                                      alpha0=0.5)
        print(f"[Long PGD] Best surrogate after polishing: {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated Annealing fine‑tuning ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline, rng,
                                             max_attempts=150_000,
                                             start_sigma=0.001,
                                             T_start=5e-3,
                                             T_end=1e-9)
        if val_sa < best_obj:
            best_seq = seq_sa
            best_obj = val_sa
            print(f"[SA] Improved surrogate to {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the supplied evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_obj:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Error] evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
