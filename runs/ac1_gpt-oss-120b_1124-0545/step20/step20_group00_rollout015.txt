def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    n = 4000

    best_seq = None
    best_val = np.inf

    def maybe_update(seq):
        nonlocal best_seq, best_val
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"New global best: {best_val:.6f}")
        return val

    # seed list:
    candidates = []
    candidates.append(_uniform(n))
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.size == n and arr.sum()>0:
            candidates.append(arr/arr.sum())
    # add random Dirichlet with random concentration
    for _ in range(4):
        conc = 10**rng.uniform(-3, 1)  # range 0.001 to 10
        vec = rng.dirichlet(np.full(n, conc))
        candidates.append(vec)
    # exponential draws
    for _ in range(3):
        vec = rng.exponential(scale=1.0, size=n)
        vec = vec / vec.sum()
        candidates.append(vec)
    # power-law draws
    for _ in range(3):
        # sample exponents
        alpha = rng.uniform(0.5, 3.0)
        i = np.arange(1, n+1)
        vec = (i**-alpha)
        vec = vec / vec.sum()
        # random permutation
        rng.shuffle(vec)
        candidates.append(vec)

    # evaluate seeds
    print("--- Evaluating seed candidates ---")
    for seq in candidates:
        if time.time() > deadline: break
        maybe_update(seq)

    # Search loop
    iteration = 0
    while time.time() < deadline - 0.5:
        iteration += 1
        # generate a new random candidate
        # Choose a generation method random
        method = rng.choice(['dirichlet','exp','power','mixgauss'])
        if method == 'dirichlet':
            conc = 10**rng.uniform(-4, 0)  # low concentration
            seq = rng.dirichlet(np.full(n, conc))
        elif method == 'exp':
            seq = rng.exponential(scale=1.0, size=n)
            seq = seq/np.sum(seq)
        elif method == 'power':
            alpha = rng.uniform(0.5, 4.0)
            i = np.arange(1, n+1)
            seq = (i**-alpha)
            seq = seq/seq.sum()
            rng.shuffle(seq)
        else: # mix Gaussian mixture
            K = 5
            params = np.concatenate([
                rng.random(K),
                rng.random(K),
                rng.uniform(0.001, 0.5, K)
            ])
            seq = _mixture_gaussian(params, n)

        # optionally perturb best_seq a bit to avoid local minima
        if best_seq is not None and rng.random() < 0.2:
            # generate from best with small Gaussian noise
            seq = _perturb(best_seq, rng, sigma=0.001)

        # perform short polishing using exponentiated gradient
        sub_deadline = time.time() + min(4.0, deadline - time.time() - 0.5)
        seq_opt, val_opt = _exp_gradient_opt(seq, n,
                                             sub_deadline,
                                             max_iters=300_000,
                                             eta0=3.0)
        # smooth L-BFGS a couple steps
        seq_opt = _lbfgs_smooth_optimize(seq_opt, n, deadline - 0.2,
                                         beta_schedule=[1.,5.,20.,80.,200.,500.,1000.],
                                         max_iter_per_beta=2000)
        val_opt = _objective_surrogate(seq_opt)
        maybe_update(seq_opt)

    # final thorough polishing on best_seq
    if best_seq is not None and time.time() < deadline - 1.0:
        print("--- Final polishing of best sequence ---")
        seq_final, val_final = _exp_gradient_opt(best_seq, n,
                                                 deadline,
                                                 max_iters=2_000_000,
                                                 eta0=5.0)
        val_final = _objective_surrogate(seq_final)
        if val_final < best_val:
            best_val = val_final
            best_seq = seq_final.copy()
            print(f"Polish gave new best: {best_val:.6f}")

    # final evaluation using provided evaluate_sequence
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Result ===\nSurrogate {best_val:.6f}, evaluate {true_val:.6f}")
    except Exception as e:
        print("Evaluation error:", e)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
Propose a candidate step‑function that improves the current best bound.
---------------------------------------------------------------------

The approach combines several cheap, highly diversified constructions
(e.g. Dirichlet draws, exponential draws, and systematic permutations)
with a fast “exponentiated‑gradient” (mirror‑descent) polishing step.
All candidates are kept on the probability simplex (sum = 1) so that the
surrogate objective `2·n·max(conv(a,a))` equals the true
`evaluate_sequence` value.

Main ideas
~~~~~~~~~~
* **Diverse seeds** – Dirichlet samples with many concentration values,
  pure exponential draws and a uniform vector.  For each seed we also try
  a random shuffle, an interleaved “large‑small” placement and a
  modular permutation (step coprime to *n*).  This quickly explores the
  space of highly non‑smooth step‑functions.

* **Fast surrogate** – `np.fft.rfft` based convolution (`_conv_fft`) lets us
  evaluate the objective in ≈0.5 ms for n = 4000.

* **Mirror‑descent polishing** – an exact sub‑gradient of the
  max‑convolution is used inside an exponentiated‑gradient iteration
  (`_exp_gradient_opt`).  The routine respects the simplex constraint by
  working in log‑softmax space.

* **Iterative search** – while time remains we repeatedly generate a new
  random base vector, apply a cheap permutation, keep the best one,
  and optionally polish promising candidates.  At the end a long polish
  of the best sequence is performed.

All helper functions are defined at top level, no closures or file I/O
are used, and the total runtime never exceeds the provided budget
(‑2 s safety margin).  The final sequence is stored in the global
`height_sequence_1` variable for reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used only for an optional final L‑BFGS polishing)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a probability vector: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _dirichlet_vector(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `conc` (all entries ≥ 0, sum = 1)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _exponential_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Exponential(1) draw, then normalised."""
    vals = rng.exponential(scale=1.0, size=n)
    vals /= vals.sum()
    return vals


def _interleaved_permutation(vals: np.ndarray) -> np.ndarray:
    """
    Place the largest, second‑largest, … alternately at the leftmost
    and rightmost free positions.  This spreads big numbers far apart.
    """
    n = len(vals)
    order = np.argsort(-vals)          # indices of descending values
    out = np.empty(n, dtype=vals.dtype)
    left, right = 0, n - 1
    for i, idx in enumerate(order):
        if i % 2 == 0:
            out[left] = vals[idx]
            left += 1
        else:
            out[right] = vals[idx]
            right -= 1
    return out


def _modular_permutation(vals: np.ndarray, step: int) -> np.ndarray:
    """
    Permutes `vals` via (i * step) mod n (step must be coprime to n).
    This yields a pseudo‑random but evenly spaced ordering.
    """
    n = len(vals)
    perm = (np.arange(n, dtype=np.int64) * step) % n
    out = np.empty_like(vals)
    out[perm] = vals
    return out


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated gradient) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the true max‑convolution.
    Variables live in log‑softmax space, guaranteeing non‑negativity
    and unit sum after every iteration.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (the shift attaining the max)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# (Optional) L‑BFGS refinement of the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000):
    """
    Optimize a log‑sum‑exp smoothed version of the objective using L‑BFGS‑B.
    Variables live in log‑softmax space.  If SciPy is unavailable this
    function returns the input unchanged.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
            2000.0, 5000.0, 20000.0, 50000.0, 100000.0,
            200000.0, 500000.0, 1000000.0,
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient w.r.t a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)  # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:  # pragma: no cover
            break

    return _softmax(v)

    if best_seq is not None and time.time() < deadline - 1.0:
        print("\n--- Final polishing of best sequence (mirror‑descent) ---")
        final_deadline = deadline - 0.5
        best_seq, best_val = _exp_gradient_opt(best_seq, n,
                                               final_deadline,
                                               max_iters=2_500_000,
                                               eta0=5.0)
        best_val = _objective_surrogate(best_seq)
        print(f"[Final EG] surrogate = {best_val:.6f}")

        # optional L‑BFGS smoothing pass (if SciPy available)
        if _opt is not None and time.time() < deadline - 0.5:
            print("--- Optional final L‑BFGS smoothing ---")
            lbgd_deadline = deadline - 0.2
            best_seq = _lbfgs_smooth_optimize(best_seq, n,
                                              lbgd_deadline,
                                              beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                             200.0, 500.0, 1000.0,
                                                             2000.0, 5000.0],
                                              max_iter_per_beta=4000)
            best_val = _objective_surrogate(best_seq)
            print(f"[Final L‑BFGS] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Official evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible later invocations
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
