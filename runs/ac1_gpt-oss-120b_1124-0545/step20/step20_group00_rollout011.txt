def _random_local_search(seq, deadline, attempts=50000, sigma=0.01):
    best_seq=seq.copy()
    best_val=_objective_surrogate(best_seq)
    n=len(seq)
    rng = np.random.default_rng()
    while attempts>0 and time.time()<deadline:
        attempts-=1
        # propose by moving small delta from i to j
        i,j = rng.integers(0,n,2)
        if i==j:
            continue
        delta = best_seq[i] * sigma * (rng.random())
        # ensure delta <= best_seq[i] and not exceed limit 1 (but clipping later)
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # ensure non-neg
        if new_seq[i] < 0:
            continue
        val=_objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq,best_val

combining a richer pool of seeds, an extended differential‑evolution
search, aggressive L‑BFGS smoothing and a custom “soft‑max
mirror‑descent’’ optimiser that uses the smooth approximation of the

exact exponentiated‑gradient polish plus a lightweight random local
search.

import numpy as np

# Try to import SciPy, but continue if it is not present.

except Exception:  # pragma: no cover

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full result)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective – exact surrogate (the factor 2·n is the only thing that
# changes when we evaluate the real function).
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(seq, seq))  (seq is assumed to sum to 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on *n* points."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√(i·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) pdf on the discrete grid {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular (rise‑and‑fall) distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add zero‑mean Gaussian noise, clip and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Gaussian mixture (the parametric family used by the DE optimiser)
# ----------------------------------------------------------------------
def _mixture_gaussian(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a probability vector from ``params``.
    The layout for K components (K = 5) is
        w_0 … w_{K‑1}      (non‑negative)
        μ_0 … μ_{K‑1}      (in [0,1])
        σ_0 … σ_{K‑1}      (positive)
    The mixture is normalised to sum to 1.
    """
    K = 5
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    w_sum = w.sum()
    if w_sum > 0.0:
        w = w / w_sum
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# L‑BFGS smoothing (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000):
    """
    Optimise the smooth max‑convolution objective using L‑BFGS‑B.
    Variables live in the log‑softmax space.
    """
    if _opt is None:
        # SciPy not present – give up early.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the weighted sum part
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]          # d/d a of sum w·conv
            grad_a = grad_a * (2.0 * n)                   # factor from outer 2·n

            # chain rule through the softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (sub‑gradient of the true max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the true max‑convolution.
    Uses the sub‑gradient at the (single) active index.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Soft‑max mirror descent (smooth max) – our own implementation
# ----------------------------------------------------------------------
def _smooth_mirror_descent(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           iters_per_beta: int = 20_000):
    """
    Gradient descent in the log‑softmax space using the smooth
    max‑approximation (log‑sum‑exp).  Returns a (probability) vector and
    its exact surrogate value.
    """
    if beta_schedule is None:
        # A generous schedule that ends with a very sharp approximation.
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0,
                         2000000.0, 5000000.0, 10000000.0]

    # initialise log‑probabilities
    log_a = np.log(start_seq + 1e-30)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    it_total = 0
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break

            a = _softmax(log_a)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # compute soft‑max weights over the full convolution vector
            # (shift by max to avoid overflow)
            exp_beta = np.exp(beta_val * (conv - max_c))
            w = exp_beta / np.sum(exp_beta)   # shape (2n‑1,)

            # gradient of the weighted sum of convolutions
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of sum w·conv

            # full objective gradient (the outer factor 2·n)
            grad_a = grad_weighted * (2.0 * n)

            # chain rule through soft‑max
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # simple step‑size schedule (decreasing like 1/√t)
            it_total += 1
            eta = 0.30 / math.sqrt(it_total)   # tuned constant
            log_a = log_a - eta * grad_log

            # occasionally evaluate the true (non‑smoothed) objective
            if it_total % 5000 == 0:
                cur_val = _objective_surrogate(_softmax(log_a))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(log_a).copy()

    # final cleanup
    final_seq = _softmax(log_a)
    final_val = _objective_surrogate(final_seq)
    if final_val < best_val:
        best_seq, best_val = final_seq, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local improvement
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 60_000,
                         sigma: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Perform a simple random walk on the probability simplex:
    repeatedly move a small random amount from one coordinate to another.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a fraction of mass from i to j
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

            return time.time() > deadline

            )

        except Exception as e:  # pragma: no cover

        trials = 6_000
        for t in range(trials):
            if time.time() > deadline:
                break
            rand_params = np.concatenate([
                rng.random(K),                  # weights
                rng.random(K),                  # means
                rng.uniform(0.001, 0.5, K)      # sigmas

            seq_rand = _mixture_gaussian(rand_params, n)
            val_rand = _objective_surrogate(seq_rand)

                print(f"[Rnd] new best surrogate = {best_val:.6f} (iter {t})")

"""
Propose a candidate step‑function that improves the current bound.

The algorithm mixes several tactics:
  * a rich pool of deterministic and random seeds (uniform, arcsine,
    beta‑shaped, triangular, Dirichlet draws, perturbations);
  * a global optimisation in a low‑dimensional Gaussian‑mixture family
    using differential evolution (or a random fallback);
  * aggressive L‑BFGS smoothing of the log‑sum‑exp surrogate;
  * a custom “soft‑max mirror descent’’ that directly minimises the
    exact objective while using a smooth approximation of the max‑convolution;
  * an exact exponentiated‑gradient polish on the true max‑convolution;
  * a lightweight random local search for the last few percent of the
    time budget.

All stages respect the overall 1000 s budget; the best sequence seen so
far is stored in the global variable ``height_sequence_1`` and returned as
a plain list of floats.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for the global DE optimiser)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numerical utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on the grid {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Gaussian mixture (parametric family for DE)
# ----------------------------------------------------------------------
def _mixture_gaussian(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a probability vector from ``params``.
    Layout (K = 5):
        w_0 … w_{K‑1}      (non‑negative)
        μ_0 … μ_{K‑1}      (in [0,1])
        σ_0 … σ_{K‑1}      (positive)
    The resulting vector is normalised to sum to 1.
    """
    K = 5
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    w_sum = w.sum()
    if w_sum > 0.0:
        w = w / w_sum
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# L‑BFGS smoothing (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000):
    """
    Optimise the smooth max‑convolution objective using L‑BFGS‑B.
    Variables live in the log‑softmax space.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]      # d/d a of weighted sum
            grad_a = grad_a * (2.0 * n)               # outer factor 2·n

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (sub‑gradient of the true max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the true max‑convolution.
    Uses the sub‑gradient at the (single) active index.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Soft‑max mirror descent (smooth max) – custom implementation
# ----------------------------------------------------------------------
def _smooth_mirror_descent(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           iters_per_beta: int = 15_000):
    """
    Gradient descent in log‑softmax space using the smooth
    log‑sum‑exp approximation of the max‑convolution.
    Returns the best exact surrogate value seen.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0,
                         2000000.0, 5000000.0, 10000000.0]

    log_a = np.log(start_seq + 1e-30)
    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    it_total = 0
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break

            a = _softmax(log_a)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            exp_beta = np.exp(beta_val * (conv - max_c))
            w = exp_beta / np.sum(exp_beta)                     # weights over 2n‑1 entries

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]          # d/d a of Σ w·conv

            grad_a = grad_weighted * (2.0 * n)                  # outer factor 2·n

            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            it_total += 1
            eta = 0.30 / math.sqrt(it_total)                    # tuned constant
            log_a = log_a - eta * grad_log

            if it_total % 5000 == 0:
                cur_val = _objective_surrogate(_softmax(log_a))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(log_a).copy()

    final_seq = _softmax(log_a)
    final_val = _objective_surrogate(final_seq)
    if final_val < best_val:
        best_seq, best_val = final_seq, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local search
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 80_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: repeatedly move a small
    random amount from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val
