"""
Search for a non‑negative step‑function (height sequence) that minimises
evaluate_sequence(seq) = 2·n·max(conv(seq, seq)) / (sum(seq)**2).

Approach
--------
1.  Build a pool of cheap analytic candidates (sine‑window, Gaussian,

    Adam optimiser on the L2 loss between self‑convolution and the uniform
    distribution (length 2n‑1).  Returns the best sequence found and its
    surrogate value.

    target = np.full(target_len, 1.0 / target_len, dtype=np.float64)

    v = np.log(initial_seq + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta2_pow = 1.0

    best_val = _objective_surrogate(best_seq)

        grad_a = 4.0 * full[n - 1:n - 1 + n]              # size n

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1_pow)

    # Load previous result if it exists
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        prev_arr = np.asarray(prev, dtype=np.float64)
        prev_arr = np.maximum(prev_arr, 0.0)
        s = prev_arr.sum()
        if s > 0.0:
            prev_arr = prev_arr / s
        else:
            prev_arr = None
    else:
        prev_arr = None

    # ------------------------------------------------------------------
    # Build a pool of cheap candidates
    # ------------------------------------------------------------------
    candidates = []

    if prev_arr is not None:
        candidates.append(prev_arr)

    # deterministic analytic families
    candidates.append(_sine_window(n, gamma=1.0))
    candidates.append(_sine_window(n, gamma=1.2))
    candidates.append(_gaussian(n, sigma_frac=0.15))
    candidates.append(_gaussian(n, sigma_frac=0.25))
    candidates.append(_exp_decay(n, rate=0.0015))
    candidates.append(_sym_exp_decay(n, rate=0.0012))

    # beta‑shape grid (few points)
    for a in (0.3, 0.6, 1.0, 2.0):
        for b in (0.3, 0.6, 1.0, 2.0):
            candidates.append(_beta_shape(n, alpha=a, beta=b))

    # random Dirichlet draws (different concentrations)
    for _ in range(6):
        candidates.append(_random_dirichlet(n, conc=0.5, rng=rng))
        candidates.append(_random_dirichlet(n, conc=0.2, rng=rng))

    # ------------------------------------------------------------------
    # Pick the best candidate from the pool (surrogate)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')
    for seq in candidates:
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[Init] best surrogate from pool = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Adam optimiser on L² loss (flatten convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stage 1: Adam on L² (flatten convolution) ---")
        seq_l2, val_l2 = _adam_optimize_l2(best_seq, n, deadline,
                                          max_iters=1_500_000,
                                          lr=0.22)
        if val_l2 < best_val:
            best_val = val_l2
            best_seq = seq_l2
            print(f"[Stage 1] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimiser on exact max (sub‑gradient polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stage 2: Adam on max‑convolution (polish) ---")
        seq_max, val_max = _adam_optimize_max(best_seq, n, deadline,
                                              max_iters=2_000_000,
                                              lr=0.14)
        if val_max < best_val:
            best_val = val_max
            best_seq = seq_max
            print(f"[Stage 2] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stage 3: Simulated annealing fine‑tuning ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                              rng,
                                              max_attempts=250_000,
                                              start_sigma=0.0008,
                                              T_start=4e-3,
                                              T_end=5e-10)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa
            print(f"[SA] improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the supplied function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Error] evaluate_sequence raised an exception:", exc)

    # Store for subsequent runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
