"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.  The algorithm is a hybrid of:

1. quick seed evaluation (uniform, Dirichlet draws, tiny perturbations);
2. a low‑dimensional global optimisation of a Gaussian‑mixture model
   (DE or random search);
3. a *linear‑programming improvement* – the classic “LP‑step’’ from the
   reference paper: maximise the ℓ₁‑mass of a vector g under the
   constraint (f * g) ≤ M (where M = max(f*f)).  The solution is mixed
   with the current iterate.
4. smooth L‑BFGS optimisation of a log‑sum‑exp surrogate (continuation on β);
5. exponentiated‑gradient (mirror‑descent) polishing on the exact max‑convolution;
6. a final short L‑BFGS pass and evaluation with ``evaluate_sequence``.

All stages respect the 1000 s overall budget and constantly keep the best
sequence seen so far (stored in the global ``height_sequence_1`` variable).
"""

# Optional SciPy import (used for DE, L‑BFGS and linear programming)

# ----------------------------------------------------------------------
# Basic utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective (exact surrogate when the vector sums to 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)).  The vector is assumed to sum to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, project to the simplex."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser on the true max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

                         20_000.0, 50_000.0,
                         100_000.0, 200_000.0, 500_000.0, 1_000_000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient w.r.t a
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Parametric mixture (Gaussian components) – used for the global DE phase
# ----------------------------------------------------------------------
def _mixture_gaussian(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a probability vector from ``params``.
    params layout (K components = 5):
        w_0 … w_{K-1}      (non‑negative, will be normalised)
        μ_0 … μ_{K-1}      (in [0,1])
        σ_0 … σ_{K-1}      (positive)
    The resulting vector is normalised to sum to 1.
    """
    K = 5
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)

    w_sum = w.sum()
    if w_sum > 0:
        w = w / w_sum
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Linear‑programming step (the “LP‑step’’ from the paper)
# ----------------------------------------------------------------------
def _build_conv_constraint_matrix(seq: np.ndarray, n: int):
    """
    Return a sparse matrix A (shape (2n‑1, n)) such that
    ``A @ x`` equals ``conv(seq, x)`` for any non‑negative vector x.
    """
    if _sp is None:               # pragma: no cover
        raise RuntimeError("scipy.sparse is required for the LP step")
    offsets = np.arange(n, dtype=int)          # 0 … n‑1
    data = [np.full(n, seq[i], dtype=np.float64) for i in range(n)]
    A = _sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    return A


def _lp_maximize_mass(seq: np.ndarray, n: int, M: float):
    """
    Solve the LP:
        maximise   Σ x_j
        subject to conv(seq, x) ≤ M    (element‑wise)
                   x ≥ 0
    Returns the optimal x (or None on failure) and the attained sum.
    """
    if _opt is None:                # pragma: no cover
        return None, None
    try:
        A = _build_conv_constraint_matrix(seq, n)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # minimise -Σ x → maximise Σ x
        bounds = [(0, None)] * n
        res = _opt.linprog(c=c, A_ub=A, b_ub=b,
                           bounds=bounds, method='highs',
                           options={'presolve': True})
        if not res.success:
            print("[LP] linprog failed:", res.message)
            return None, None
        x = res.x
        total = float(x.sum())
        return x, total
    except Exception as e:  # pragma: no cover
        print("[LP] exception:", e)
        return None, None


def _lp_improvement_phase(seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_steps: int = 8):
    """
    Repeatedly perform the LP‑step and mix the solution back into the
    current iterate.  Stops when no improvement is possible or time runs out.
    """
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)

    step = 0
    while time.time() < deadline and step < max_steps:
        conv_f = _conv_fft(best_seq, best_seq)
        M = float(np.max(conv_f))

        x, total = _lp_maximize_mass(best_seq, n, M)
        if x is None or total is None:
            break

        print(f"[LP] step {step+1} – sum(g) = {total:.6f},  M = {M:.6f}")

        if total <= 1.0001:          # LP cannot increase ℓ₁‑mass → stop
            break

        g_norm = x / total            # normalise to have sum = 1

        # try a few mixing coefficients and keep the best improvement
        improved = False
        best_local_seq = best_seq
        best_local_val = best_val

        for t in (0.1, 0.3, 0.5, 0.7, 1.0):
            cand = (1.0 - t) * best_seq + t * g_norm
            cand_val = _objective_surrogate(cand)
            if cand_val < best_local_val:
                best_local_val = cand_val
                best_local_seq = cand
                improved = True

        if improved:
            best_seq = best_local_seq
            best_val = best_local_val
            print(f"[LP]  → improved surrogate = {best_val:.6f}")
            step += 1
        else:
            # no mixing improved the objective → stop the LP loop
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the value returned by
    ``evaluate_sequence``.  The best sequence found (as a list of floats) is
    stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin for final evaluation
    deadline = start_time + budget_s - 2.0

    n = 4000                               # problem size used in the reference run

    # ------------------------------------------------------------------
    # Load previous solution if it exists and matches the required length
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr.copy()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1️⃣  Build a small pool of cheap seeds
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]
    if prev_arr is not None and prev_arr.sum() > 0:
        seeds.append(prev_arr / prev_arr.sum())

    for conc in (0.05, 0.2, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(3):
        seeds.append(_perturb(_uniform(n), rng, sigma=0.004))

    # ------------------------------------------------------------------
    # Evaluate seeds and keep the best (surrogate = 2·n·max(conv))
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    print("\n--- Evaluating initial seeds ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        print(f"  seed #{idx+1:2d} surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Global optimisation in a low‑dimensional Gaussian‑mixture family
    # ------------------------------------------------------------------
    def de_obj(params):
        seq = _mixture_gaussian(params, n)
        return _objective_surrogate(seq)

    K = 5
    bounds = []
    for _ in range(K):
        bounds.append((0.0, 1.0))            # weights (clipped inside the builder)
    for _ in range(K):
        bounds.append((0.0, 1.0))            # means
    for _ in range(K):
        bounds.append((0.001, 0.5))          # sigmas

    if _opt is not None and time.time() < deadline:
        print("\n--- Differential Evolution on Gaussian mixture ---")
        popsize = 12
        maxiter = 35

        def de_cb(xk, convergence):
            return time.time() > deadline

        try:
            result = _opt.differential_evolution(
                de_obj,
                bounds,
                seed=int(seed),
                popsize=popsize,
                maxiter=maxiter,
                polish=False,
                updating='deferred',
                callback=de_cb,
                workers=1,
                init='latinhypercube',
                disp=False,
            )
            seq_de = _mixture_gaussian(result.x, n)
            val_de = _objective_surrogate(seq_de)
            print(f"[DE] best surrogate = {val_de:.6f}")
            if val_de < best_val:
                best_val, best_seq = val_de, seq_de.copy()
        except Exception as e:  # pragma: no cover
            print("[DE] failed:", e)
    else:
        # Simple random search as a fallback
        print("\n--- Random Gaussian‑mixture sampling (fallback) ---")
        trials = 5000
        for t in range(trials):
            if time.time() > deadline:
                break
            rand_params = np.concatenate([
                rng.random(K),                # weights
                rng.random(K),                # mus
                rng.uniform(0.001, 0.5, K)    # sigmas
            ])
            seq_rand = _mixture_gaussian(rand_params, n)
            val_rand = _objective_surrogate(seq_rand)
            if val_rand < best_val:
                best_val, best_seq = val_rand, seq_rand.copy()
                print(f"[Rnd] new best surrogate = {best_val:.6f} (iter {t})")

    # ------------------------------------------------------------------
    # 3️⃣  LP improvement phase (the “state‑of‑the‑art’’ LP iteration)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- LP improvement phase ---")
        remaining = deadline - time.time()
        lp_deadline = time.time() + remaining * 0.4   # about 40 % of what’s left
        best_seq, best_val = _lp_improvement_phase(best_seq, n, lp_deadline)
        print(f"[LP phase] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Smooth L‑BFGS refinement (log‑sum‑exp smoothing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- L‑BFGS smoothing of current best ---")
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.45
        seq_lbfgs = _lbfgs_smooth_optimize(best_seq, n,
                                           sub_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0,
                                                          80.0, 200.0, 500.0,
                                                          1_000.0, 2_000.0,
                                                          5_000.0, 20_000.0,
                                                          50_000.0, 100_000.0,
                                                          200_000.0, 500_000.0,
                                                          1_000_000.0],
                                           max_iter_per_beta=8_000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        print(f"[L‑BFGS] surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Exact mirror‑descent (exponentiated‑gradient) polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing ---")
        polishing_deadline = deadline - 5.0
        # scale number of iterations to the remaining time slice
        max_iters = int(2_500_000 *
                        (polishing_deadline - time.time()) /
                        (deadline - start_time))
        max_iters = max(200_000, min(max_iters, 3_000_000))
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                           polishing_deadline,
                                           max_iters=max_iters,
                                           eta0=5.0)
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Optional final L‑BFGS pass (to clean any remaining smoothing bias)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Final short L‑BFGS pass ---")
        final_deadline = deadline - 2.0
        seq_final = _lbfgs_smooth_optimize(best_seq, n,
                                            final_deadline,
                                            beta_schedule=[1.0, 5.0, 20.0,
                                                           80.0, 200.0,
                                                           500.0, 1_000.0,
                                                           2_000.0, 5_000.0],
                                            max_iter_per_beta=4_000)
        val_final = _objective_surrogate(seq_final)
        print(f"[Final L‑BFGS] surrogate = {val_final:.6f}")
        if val_final < best_val:
            best_val, best_seq = val_final, seq_final.copy()
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
