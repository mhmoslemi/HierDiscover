"""
Propose a step‑function (non‑negative vector with sum 1) that minimizes
evaluate_sequence.  The optimizer combines three complementary phases
that fit into the 1000 s budget:

1. **Mirror‑descent (exponentiated‑gradient)** on the exact max‑convolution
   objective – fast convergence from a good initial shape.
2. **Mass‑transfer simulated annealing** – random moves that transfer a
   small amount of probability mass between two positions (or swap values);
   this explores directions that gradient‑based methods cannot reach.
3. **Optional polishing** – a short second EG pass and, if SciPy is present,
   an L‑BFGS optimisation of a smoothed max‑convolution.

All phases keep the vector on the probability simplex (non‑negative,
ℓ₁‑norm = 1).  The best sequence found is stored in the global variable
`height_sequence_1` for use by subsequent runs.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase.
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:       # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact when ℓ₁‑norm = 1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple helper families for initialisation
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on n entries."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-rate·i), i=0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Numerically stable softmax (used by exponentiated‑gradient)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Projection onto the probability simplex (ℓ₁‑norm = 1, non‑negative)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the simplex {x≥0, sum x = 1} (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero; return uniform
        return _uniform(n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # work in log‑space to keep positivity and unit sum automatically
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        inds = np.arange(i_start, i_end + 1, dtype=int)
        g[inds] = 2.0 * a[k_max - inds]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing with mass‑transfer moves (keeps the ℓ₁‑norm = 1)
# ----------------------------------------------------------------------
def _mass_transfer_sa(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Randomly moves a small amount of probability mass between two positions
    (or swaps two values).  Uses a linear temperature schedule.
    Returns the best sequence observed and its surrogate value.
    """
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # temperature schedule – a small fraction of the current value
    T_start = cur_val * 1e-2
    T_end = cur_val * 1e-8

    # max transfer starts relatively large and decays linearly
    init_step = 0.02  # up to 2 % of the total mass per move at the start

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T_start + (T_end - T_start) * (it / max_iters)
        step = init_step * (1.0 - it / max_iters)

        # decide move type
        if rng.random() < 0.7:                     # mass‑transfer move
            i = rng.integers(0, n)
            # avoid transferring from an almost‑zero bucket
            if cur[i] < 1e-12:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            max_transfer = min(cur[i], step)
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer
            ns = cur.copy()
            ns[i] -= delta
            ns[j] += delta
        else:                                      # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur.copy()
            ns[i], ns[j] = ns[j], ns[i]

        val = _objective(ns)

        # accept if improvement, otherwise Metropolis criterion
        if val < cur_val:
            cur, cur_val = ns, val
            if val < best_val:
                best, best_val = ns.copy(), val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur, cur_val = ns, val

    return best, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """
    Optimise a smoothed max‑convolution objective using L‑BFGS‑B.
    Operates in log‑softmax space to stay feasible.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c                     # stabiliser for exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    Strategy (fits into the 1000 s budget):
      1. Load a previous solution (global ``height_sequence_1``) or fall back
         to an exponential‑decay shape.
      2. Run a long exponentiated‑gradient (mirror‑descent) pass.
      3. Refine the result with mass‑transfer simulated annealing.
      4. If time permits, a short second EG pass followed by an optional
         L‑BFGS smoothing phase.
      5. Return the best sequence and store it globally.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the historic dimension if available.
    # ------------------------------------------------------------------
    n = 4000
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        seq0 = np.asarray(prev, dtype=np.float64)
    else:
        seq0 = None

    if seq0 is None:
        # Fallback: smooth exponential decay.
        seq0 = _exp_decay(n, rate=0.003)

    # Normalise to ℓ₁‑norm = 1 and ensure non‑negativity.
    seq0 = np.maximum(seq0, 0.0)
    tot = seq0.sum()
    if tot == 0.0:
        seq0 = _uniform(n)
    else:
        seq0 = seq0 / tot

    print(f"[Init] surrogate = {_objective(seq0):.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Mirror‑descent (exponentiated gradient)
    # Allocate roughly 70 % of the remaining budget.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    eg_time = max(0.0, remaining * 0.70)
    eg_deadline = time.time() + eg_time
    seq_eg, _ = _exp_gradient_opt(seq0, n, eg_deadline,
                                  max_iters=10_000_000, eta0=5.0)
    cur_best_seq = seq_eg
    cur_best_val = _objective(cur_best_seq)
    print(f"[EG] surrogate after phase 1 = {cur_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Mass‑transfer Simulated Annealing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        sa_deadline = deadline - 5.0
        seq_sa, _ = _mass_transfer_sa(cur_best_seq, n, sa_deadline, rng,
                                      max_iters=800_000)
        val_sa = _objective(seq_sa)
        print(f"[SA] surrogate after annealing = {val_sa:.6f}")
        if val_sa < cur_best_val:
            cur_best_seq, cur_best_val = seq_sa, val_sa

    # ------------------------------------------------------------------
    # Phase 3 – Optional short second EG pass (refinement)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        eg2_deadline = deadline - 5.0
        seq_eg2, _ = _exp_gradient_opt(cur_best_seq, n, eg2_deadline,
                                       max_iters=4_000_000, eta0=3.0)
        val_eg2 = _objective(seq_eg2)
        print(f"[EG2] surrogate after refinement = {val_eg2:.6f}")
        if val_eg2 < cur_best_val:
            cur_best_seq, cur_best_val = seq_eg2, val_eg2

    # ------------------------------------------------------------------
    # Phase 4 – Optional L‑BFGS smoothing (if SciPy is available)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0 and _opt is not None:
        seq_lb = _lbfgs_smooth_optimize(cur_best_seq, n, deadline,
                                        max_iter_per_beta=4000)
        val_lb = _objective(seq_lb)
        print(f"[L‑BFGS] surrogate after smoothing = {val_lb:.6f}")
        if val_lb < cur_best_val:
            cur_best_seq, cur_best_val = seq_lb, val_lb

    # ------------------------------------------------------------------
    # Final evaluation via the provided black‑box function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(cur_best_seq.tolist())
        print(f"[Result] surrogate = {cur_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next run (facilitates incremental improvement)
    globals()["height_sequence_1"] = cur_best_seq.tolist()
    return cur_best_seq.tolist()
