"""
Search for a non‑negative step‑function (length 4000) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm combines:

1. A rich collection of analytic “seed’’ families (windows, decays, beta‑shapes,
   low‑frequency cosine sums, Dirichlet draws, …).  All seeds are ℓ₁‑normalised.
2. Fast surrogate evaluation (the ℓ₁‑normalised objective is simply
   `2·n·max(conv)`), using an FFT based convolution.
3. A long stochastic hill‑climbing phase: repeatedly take a convex combination
   of the current best with a fresh random Dirichlet vector.  The mixing
   coefficient α is annealed from ~0.3 down to 1e‑5, allowing both large jumps
   and fine‑grained refinements.
4. A projected sub‑gradient descent phase on the exact max‑convolution.
   The sub‑gradient is taken at the active index k=argmax(conv), the update is
   Euclidean (a←a−η g) and a Euclidean projection onto the probability simplex
   (non‑negative, sum = 1) is performed after each step.
5. The final sequence is evaluated with the provided `evaluate_sequence`
   function and stored in the global variable ``height_sequence_1`` for the
   next run.

The whole search respects a 1000 s wall‑clock budget (minus a small safety
margin).  The code is fully self‑contained, uses only NumPy and the standard
library, and does not rely on any closures or lambdas.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of *v* onto the probability simplex
    {w≥0, sum(w)=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for an ℓ₁‑normalised vector: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1.0 - y * y)
    a = np.i0(t)                     # Modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-γ·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    """Logistic decreasing shape: 1/(1+exp(steep·(x‑shift)))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves; clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sample from a symmetric Dirichlet (parameters = conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser on exact max‑convolution
# ----------------------------------------------------------------------
def _projected_subgradient_opt(start_seq: np.ndarray,
                               n: int,
                               deadline: float,
                               max_iters: int = 400_000,
                               eta0: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Euclidean projected sub‑gradient descent on `max(conv(a,a))`.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient at the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)
        a = a - eta * g
        a = _proj_simplex(a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Subgrad] iter {it:,} → {best_val:.6f}")

    return best_seq, best_val

            best_val = val
            best_seq = vec.copy()
            print(f"[Seed] improved to {best_val:.6f}")

    print(f"[After seeds] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    mix_deadline = deadline - 40.0          # reserve ~40 s for sub‑gradient phase
    max_alpha = 0.30
    min_alpha = 1e-5

        cand = (1.0 - alpha) * best_seq + alpha * rand_seq
        cand = cand / cand.sum()

        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Mix] iter {it_mix:,} → {best_val:.6f}")

        it_mix += 1

        # every few thousand steps try a random permutation (may discover
        # a better ordering)
        if it_mix % 5000 == 0:
            perm = best_seq.copy()
            rng.shuffle(perm)
            val_perm = _objective_surrogate(perm)
            if val_perm < best_val:
                best_val = val_perm
                best_seq = perm.copy()
                print(f"[Perm] iter {it_mix:,} → {best_val:.6f}")

    print(f"[Mixing finished] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = deadline
        seq_refined, val_refined = _projected_subgradient_opt(
            best_seq, n, subgrad_deadline,
            max_iters=400_000, eta0=1.5
        )
        if val_refined < best_val:
            best_val = val_refined
            best_seq = seq_refined.copy()
            print(f"[Subgrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final exact evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
