def _adam_optimize(start_vec, n, deadline, beta_schedule=None, steps_per_beta=2000):
    if beta_schedule is None: same list.
    v = np.log(start_vec + 1e-30)
    best_seq = softmax(v)
    best_val = _objective_surrogate(best_seq)
    m = np.zeros_like(v)
    v_mom2 = np.zeros_like(v)
    t_iter = 0
    eps = 1e-8
    lr = 0.1 # maybe adjust.

    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return best_seq
            a = _softmax(v)
            # compute convolution
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth approximated value
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
            # gradient w.r.t a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]
            # gradient v
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            # Adam update
            t_iter += 1
            m = 0.9 * m + 0.1 * grad_v
            v_mom2 = 0.999 * v_mom2 + 0.001 * (grad_v * grad_v)
            m_hat = m / (1 - 0.9 ** t_iter)
            v_hat = v_mom2 / (1 - 0.999 ** t_iter)
            # step size heuristics: maybe lr = 0.1 / (beta**0.5) or constant
            v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

            # evaluate current a
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    return best_seq

"""

        2 * n * max(conv(a, a)) / (sum(a)**2)

The optimisation proceeds in three stages:

1. **Rich seed generation** – a wide collection of analytic windows,
   exponential / logistic / V‑shaped families and random Dirichlet draws.
   All seeds are ℓ₁‑normalised (sum = 1).  The best few seeds are kept.

   Variables live in log‑softmax space, so the probability simplex
   constraints are satisfied automatically.  Adam (with momentum) carries

3. **Exact sub‑gradient polishing (Exponentiated Gradient)** – after the
   smooth phase we run a long mirror‑descent (EG) on the true max‑convolution
   to obtain a final refinement.

The whole routine respects a global 1000 s budget (a 5 s safety margin).

# --------------------------------------------------------------
# Helper utilities (FFT, softmax, projection)
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# --------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """For a vector with sum = 1 returns 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# --------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-γ·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    """Logistic decreasing shape: 1/(1+exp(steep·(x‑shift)))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _v_shape_exponential(n: int, gamma: float) -> np.ndarray:
    """V‑shaped (U‑shaped) exponential: a_i ∝ exp(-γ·min(i,n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i)
    a = np.exp(-gamma * dist / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to ≥0."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# --------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp surrogate
# --------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   steps_per_beta: int = 8000,
                   lr: float = 0.15) -> np.ndarray:
    """
    Adam in log‑softmax space for the smoothed max‑convolution.
    Returns the best probability vector encountered.
    """
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
                         2000.0, 5000.0, 10000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0,
                         2000000.0, 5000000.0, 10000000.0]

    # log‑softmax variables
    v = np.log(start_vec + 1e-30)

    # Adam state
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t_iter = 0

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                break

            a = _softmax(v)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            # log‑sum‑exp smooth max
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # objective (for monitoring – not used in the update)
            _obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (derived from smooth max)
            w = exp_beta / sum_exp               # weight for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t. log‑softmax variables (chain rule)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v * grad_v)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat_corr = v_hat / (1.0 - beta2 ** t_iter)

            v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

            # Keep best (evaluated on the exact surrogate, not the smoothed one)
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq


# --------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# --------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution.
    Returns the best vector found and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max‑entry k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns the best sequence (list of floats)
    and stores it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # 5‑second safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    # ----------------------------------------------------------
    # Problem size (keep the same size as the previous best)
    # ----------------------------------------------------------
    n = 4000

    # ----------------------------------------------------------
    # Load previous solution, if any, to seed the search
    # ----------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    # ----------------------------------------------------------
    # 1. Seed generation (a broad family of analytic shapes)
    # ----------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Basic windows
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))
    for sigma in np.linspace(0.09, 0.26, 14):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 2.0, 13):
        seeds.append(_power(n, p))

    # Beta‑shapes (including U‑shaped cases)
    alphas = np.linspace(0.2, 5.0, 7)
    betas = np.linspace(0.2, 5.0, 7)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # Exponential / logistic families
    for gamma in np.linspace(0.1, 8.0, 16):
        seeds.append(_exp_decay(n, gamma))
    for steep in np.linspace(0.5, 30.0, 10):
        for shift in np.linspace(0.1, 0.9, 9):
            seeds.append(_logistic_decay(n, steep, shift))

    # New V‑shaped exponential family (mass concentrated at the ends)
    for gamma in np.linspace(0.5, 8.0, 14):
        seeds.append(_v_shape_exponential(n, gamma))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Perturbations of the previous best (if available)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(4):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ----------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ----------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ----------------------------------------------------------
    # 3. Adam optimisation from the best few seeds
    # ----------------------------------------------------------
    print("\n--- Adam optimisation (smooth max) ---")
    refined = []
    # we run Adam from the single best seed and from the second best
    for idx, seed_vec in enumerate(top_seeds[:2]):
        if time.time() >= deadline:
            break
        sub_deadline = deadline - 2.0  # a tiny safety buffer
        print(f"[Adam #{idx+1}] start (surrogate {_objective_surrogate(seed_vec):.6f})")
        seq_adam = _adam_optimize(seed_vec, n, sub_deadline,
                                  steps_per_beta=12000,
                                  lr=0.12)
        val_adam = _objective_surrogate(seq_adam)
        print(f"[Adam #{idx+1}] finished surrogate = {val_adam:.6f}")
        refined.append((seq_adam, val_adam))

    # ----------------------------------------------------------
    # 4. Keep the best candidate from seeds + Adam runs
    # ----------------------------------------------------------
    candidates = evaluated[:top_k] + refined
    candidates.sort(key=lambda x: x[1])
    best_seq = candidates[0][0].copy()
    best_val = candidates[0][1]
    print(f"\n[Init] best surrogate after Adam = {best_val:.6f}")

    # ----------------------------------------------------------
    # 5. Final long EG polishing on the current best
    # ----------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing (exact max) ---")
        seq_final, val_final = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=900_000,
                                                 eta0=5.5)
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Final EG] improved to {best_val:.6f}")

    # ----------------------------------------------------------
    # 6. Exact evaluation and reporting
    # ----------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nsurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
