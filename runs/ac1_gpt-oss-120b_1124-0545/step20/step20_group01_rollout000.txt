"""
Propose a step‑function sequence (non‑negative heights) that minimises
``evaluate_sequence``.  The strategy is:

1. **Rich seed generation** – a collection of analytic families (uniform,
   power‑law, exponential, symmetric exponential, Gaussian, beta‑shaped)
   plus random Dirichlet draws and random piecewise‑linear curves.
2. **Surrogate pre‑selection** – evaluate the cheap surrogate
   `2·n·max(conv(p,p))` (with `sum(p)=1`) for all seeds and keep the
   best few.
3. **Mirror‑descent polishing** – for each top seed run an exponentiated‑gradient
   (EG) / mirror‑descent loop that directly reduces the peak of the exact
   self‑convolution.  The gradient is formed from *all* maximal convolution
   indices (not only a single one) for a more stable sub‑gradient.
4. **Local random mass‑transfer search** – a cheap simulated‑annealing style
   hill‑climber that moves small amounts of mass between two positions.
   This can escape tiny plateaus after EG.
5. **Final long EG run** – the current global best receives a long EG
   polishing using the remaining time budget.
6. The best sequence found is stored in ``height_sequence_1`` for possible
   continuation runs and returned as a plain ``list[float]``.

The whole search respects a 1000 s wall‑time limit; all heavy work stops
as soon as the deadline is reached.
"""

import time
import math
import numpy as np

# -------------------------- FFT utilities -------------------------- #
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# -------------------------- Surrogate -------------------------- #
def _objective_surrogate(p: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(p, p)).
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))

# ==================== Analytic families ==================== #
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - c))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _random_sparse_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    vec = rng.dirichlet(np.full(n, conc))
    thresh = np.percentile(vec, 80)   # keep top 20 %
    vec = np.where(vec >= thresh, vec, 0.0)
    tot = vec.sum()
    if tot == 0.0:
        return _uniform(n)
    return vec / tot

def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()

def _perturb(p: np.ndarray, rng: np.random.Generator, sigma: float = 0.003) -> np.ndarray:
    noisy = p + sigma * rng.normal(size=p.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s == 0.0:
        return p.copy()
    return noisy / s

# --------------------- Mirror‑descent (EG) --------------------- #
def _exp_gradient_opt(start_p: np.ndarray,
                      deadline: float,
                      max_iters: int = 5_000_000,
                      eta0: float = 8.0,
                      tol_active: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses all indices whose convolution value is within `tol_active` of the maximum.
    """
    n = len(start_p)
    a = start_p.copy()
    log_a = np.log(a + 1e-30)
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute self‑convolution
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        # active set (allow a tiny tolerance to capture plateaus)
        act_idx = np.where(conv >= max_c - tol_active)[0]
        # build sub‑gradient: average over active indices
        g = np.zeros_like(a)
        for k in act_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - i_range
            g[i_range] += 2.0 * a[rev_idx]
        g /= max(1, len(act_idx))

        # step‑size schedule (sqrt decay)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
        # optional early stop if improvement stalls (not enforced)
    return best_a, best_val

# -------------------- Random mass‑transfer hill‑climb -------------------- #
def _mass_transfer_hillclimb(start_p: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator,
                             max_iters: int = 300_000,
                             init_step: float = 0.01,
                             decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style local search.
    At each step transfers a small amount of probability mass from i to j.
    The step size decays exponentially.
    """
    p = start_p.copy()
    cur_val = _objective_surrogate(p)
    best_p = p.copy()
    best_val = cur_val
    step = init_step

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # pick two distinct indices
        i, j = rng.integers(0, len(p), size=2)
        if i == j:
            continue
        # propose moving delta from i to j (delta ≤ p[i])
        delta = step * rng.random()
        delta = min(delta, p[i])
        if delta <= 0.0:
            continue
        ns = p.copy()
        ns[i] -= delta
        ns[j] += delta
        # normalisation is already preserved
        val = _objective_surrogate(ns)
        if val < cur_val:
            p = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_p = ns.copy()
        else:
            # simple downhill acceptance with tiny probability
            if rng.random() < 0.001:
                p = ns
                cur_val = val
        # decay step size slowly
        step *= decay
    return best_p, best_val

    # Include previous best and a few perturbations if available

        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:

        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

        globals()["height_sequence_1"] = best_seq.tolist()
        return best_seq.tolist()

    # --------------------------------------------------------------
    # 3. Initialise global best from the very best seed
    # --------------------------------------------------------------

    global_best_val = _objective_surrogate(global_best_seq)

    # --------------------------------------------------------------

    # --------------------------------------------------------------

                                                      sub_deadline,
                                                      rng,
                                                      max_iters=150_000,
                                                      init_step=0.02,
                                                      decay=0.9997)
        else:
            seq_mt, val_mt = seq_eg, val_eg

        # pick best of this seed's candidates
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_mt, val_mt)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f" >>> New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long EG polishing on the global best (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=12_000_000,
                                                 eta0=8.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Very short final mass‑transfer tweak (few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 20.0
        seq_last, val_last = _mass_transfer_hillclimb(global_best_seq.copy(),
                                                      short_deadline,
                                                      rng,
                                                      max_iters=80_000,
                                                      init_step=0.01,
                                                      decay=0.9998)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final tweak] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Evaluate with the provided evaluate_sequence function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
