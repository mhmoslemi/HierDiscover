def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector normalised to sum = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:

    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)  # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:

def _exp_decay(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay a_i ∝ exp(-γ·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    """Logistic decreasing shape: 1/(1+exp(steep·(x‑shift)))."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    s = a.sum()

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:

def _sparse_random(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Create a sparse vector with k random positive entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    a[idx] = rng.random(k) + 0.1  # avoid exact zeros at chosen places

def _beta_exp_shape(n: int, alpha: float, beta: float, gamma: float) -> np.ndarray:
    """
    Generalised beta‑exponential shape:
        a_i ∝ (i+1)^α · (n‑i)^β · exp(‑γ·i/(n‑1))
    All parameters may be positive or negative; the expression is kept
    non‑negative by taking max with 0.
    """
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** alpha) * ((n - i) ** beta) * np.exp(-gamma * i / (n - 1))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


def _random_convex_combination(seeds: list[np.ndarray],
                               max_terms: int,
                               rng: np.random.Generator) -> np.ndarray:
    """
    Random convex combination of up to `max_terms` distinct seeds.
    """
    k = rng.integers(2, max_terms + 1)
    chosen = rng.choice(seeds, size=k, replace=False)
    weights = rng.dirichlet(np.ones(k))
    mix = np.zeros_like(seeds[0])
    for w, s in zip(weights, chosen):
        mix += w * s
    s = mix.sum()
    if s <= 0.0:
        return chosen[0].copy()
    return mix / s

# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space so we stay on the probability simplex.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0,
            2000.0, 5000.0, 20000.0, 50000.0, 100000.0,
            200000.0, 500000.0, 1_000_000.0, 2_000_000.0,
            5_000_000.0, 10_000_000.0, 20_000_000.0,
            50_000_000.0, 100_000_000.0, 200_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # smooth max via log‑sum‑exp (stable)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient w.r.t a
            w = exp_beta / sum_exp                     # weights for each conv entry
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # slice length n

            # Chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] Exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 6.0,
                     patience: int = 30_000):
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry (index k)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val - 1e-14:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                break

    return best_seq, best_val

    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Classic analytic windows (same set as previous run)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in np.linspace(1.0, 15.0, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 14):
        seeds.append(_gaussian(n, sigma))
    for p in np.linspace(0.0, 2.0, 13):
        seeds.append(_power(n, p))
    # Beta‑shape grid
    alphas = np.linspace(0.2, 5.0, 7)
    betas = np.linspace(0.2, 5.0, 7)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))
    for alpha in np.linspace(0.1, 0.9, 7):
        seeds.append(_tukey_window(n, alpha))

    # Exponential / logistic families (new)
    for gamma in np.linspace(0.1, 8.0, 16):
        seeds.append(_exp_decay(n, gamma))
    for steep in np.linspace(0.5, 30.0, 10):
        for shift in np.linspace(0.1, 0.9, 9):
            seeds.append(_logistic_decay(n, steep, shift))

    # Generalised beta‑exponential shapes (new)
    for _ in range(80):
        alpha = rng.uniform(-1.0, 3.0)
        beta = rng.uniform(-1.0, 3.0)
        gamma = rng.uniform(0.0, 5.0)
        seeds.append(_beta_exp_shape(n, alpha, beta, gamma))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Sparse random vectors (few spikes)
    for k in (30, 60, 120, 240):
        for _ in range(3):
            seeds.append(_sparse_random(n, k, rng))

    # Perturbations of the previous best (if any)
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(10):
            noisy = prev_arr + 0.005 * rng.normal(size=n)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s > 0:
                seeds.append(noisy / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(60, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Random convex mixing of top seeds
    # ------------------------------------------------------------------
    print("\n--- Random mixing of top seeds ---")
    mixed = []
    mix_deadline = deadline - 2.0
    for _ in range(3000):
        if time.time() > mix_deadline:
            break
        mix = _random_convex_combination(top_seeds, max_terms=5, rng=rng)
        mixed.append((mix, _objective_surrogate(mix)))

    # Keep best mixes
    candidates = evaluated[:top_k] + mixed
    candidates.sort(key=lambda x: x[1])
    best_seq = candidates[0][0].copy()
    best_val = candidates[0][1]
    print(f"\n[Init] best surrogate after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine a few elite candidates (L‑BFGS + EG)
    # ------------------------------------------------------------------
    refined = []
    num_refine = min(7, len(candidates))
    for idx in range(num_refine):
        if time.time() >= deadline:
            break
        seed_vec = candidates[idx][0]
        # allocate time proportionally
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.4 / (num_refine - idx)

        print(f"\n[Refine #{idx+1}] seed surrogate = {_objective_surrogate(seed_vec):.6f}")

        # L‑BFGS continuation (more generous iteration budget)
        seq_lbfgs = _lbfgs_smooth_optimize(seed_vec,
                                           n,
                                           sub_deadline,
                                           beta_schedule=None,
                                           max_iter_per_beta=12000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)

        # EG polishing (longer)
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs,
                                               n,
                                               sub_deadline,
                                               max_iters=250_000,
                                               eta0=8.0,
                                               patience=40_000)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # Pick the best among the three versions
        local_best_seq, local_best_val = min(
            [(seed_vec, _objective_surrogate(seed_vec)),
             (seq_lbfgs, val_lbfgs),
             (seq_eg, val_eg)],
            key=lambda x: x[1]
        )
        print(f"[Refine #{idx+1}] local best surrogate = {local_best_val:.6f}")

        refined.append((local_best_seq, local_best_val))
        if local_best_val < best_val:
            best_seq = local_best_seq
            best_val = local_best_val
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long EG polishing on the current best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        seq_final, val_final = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=500_000,
                                                 eta0=7.0,
                                                 patience=50_000)
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Final EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Optional final L‑BFGS smoothing (tight β‑schedule)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS smoothing ---")
        seq_final_lbfgs = _lbfgs_smooth_optimize(best_seq,
                                                n,
                                                deadline,
                                                beta_schedule=None,
                                                max_iter_per_beta=12000)
        val_final_lbfgs = _objective_surrogate(seq_final_lbfgs)
        if val_final_lbfgs < best_val:
            best_seq = seq_final_lbfgs
            best_val = val_final_lbfgs
            print(f"[Final L‑BFGS] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation (exact objective)
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
