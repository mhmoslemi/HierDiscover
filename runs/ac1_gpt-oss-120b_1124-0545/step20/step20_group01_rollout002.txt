"""
Propose a step‑function sequence (non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm (≈ 1000 s budget)

1. **Generate a huge seed pool** – analytic families (uniform, sine‑window,
   Gaussian, power‑law, exponential, symmetric‑exponential, beta‑shape,
   random Dirichlet, sparse Dirichlet, random exponential, low‑frequency
   cosine, piecewise‑linear) and random permutations of some shapes.
   Every vector is clipped to ≥0 and ℓ₁‑normalised.

2. **Select the most promising seeds** – keep the top‑k (by the exact
   surrogate 2·n·max(conv(a,a))) for intensive refinement.

3. **Continuous simulated‑annealing** – for each selected seed we perform a
   long SA run that moves a small amount of mass from one index to another
   (or swaps two entries).  The temperature is linearly cooled; the move is
   accepted with the Metropolis rule.  The best vector seen is kept.

4. **Exponentiated‑gradient polishing** – starting from the SA result we
   run mirror‑descent (log‑space EG) on the exact max‑convolution.
   The step‑size ηₜ = η₀/√t.  The run continues until the time budget for
   the seed is exhausted.

5. **Optional L‑BFGS smoothing** – if SciPy is available, we minimise a
   smooth log‑sum‑exp approximation of the max‑convolution for a few β
   values.

6. **Global best selection** – the best refined seed becomes the current
   global optimum.

7. **Final long EG polishing** – the global best is polished for the
   remaining time (many millions of EG iterations).

8. **Evaluation** – the final candidate is evaluated with the provided
   ``evaluate_sequence`` and stored in the global variable
   ``height_sequence_1`` for possible continuation runs.

All heavy loops check the time limit frequently, guaranteeing termination
before the 1000 s deadline.
"""

# Numerical utilities (FFT‑based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * |i - (n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_dirichlet(n: int, conc: float,
                             rng: np.random.Generator) -> np.ndarray:
    """Keep only the largest 20 % of entries of a Dirichlet draw."""
    vec = rng.dirichlet(np.full(n, conc))
    thresh = np.percentile(vec, 80)
    vec = np.where(vec >= thresh, vec, 0.0)
    total = vec.sum()
    if total == 0.0:
        return _uniform(n)
    return vec / total


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int,
                   rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine components, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _piecewise_linear(n: int, num_breaks: int,
                      rng: np.random.Generator) -> np.ndarray:
    """Random piecewise‑linear curve with ``num_breaks`` interior break‑points."""
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Continuous simulated‑annealing (mass‑transfer moves)
# ----------------------------------------------------------------------
def _continuous_sa(start_seq: np.ndarray, n: int, deadline: float,
                   rng: np.random.Generator,
                   max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """SA that moves a small fraction of mass from one index to another."""
    seq = start_seq.copy()
    cur_val = _objective_surrogate(seq)
    best_seq = seq.copy()
    best_val = cur_val

    # temperature schedule – from 20 % of current value down to 1e‑6
    T_start = max(cur_val * 0.2, 1e-8)
    T_end = max(cur_val * 1e-6, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T_start + (T_end - T_start) * (it / max_iters)

        # decide move type
        if rng.random() < 0.6:               # mass transfer (most common)
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or seq[i] == 0.0:
                continue
            frac = rng.uniform(0.0, 0.08)    # up to 8 % of the source mass
            delta = seq[i] * frac
            ns = seq.copy()
            ns[i] -= delta
            ns[j] += delta
        else:                                 # pure swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        # surrogate evaluation
        val = _objective_surrogate(ns)

        if val < cur_val:
            seq = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        else:
            # Metropolis acceptance
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray, n: int,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (sub‑gradient on active index)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    check_interval = 1000   # update best occasionally

    while time.time() < deadline and it < max_iters:
        it += 1

        # full convolution & locate active index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max_idx - (n - 1)
        if i_start < 0:
            i_start = 0
        i_end = max_idx
        if i_end > n - 1:
            i_end = n - 1
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % check_interval == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS smoothing (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray, n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500):
    """Smoothed max‑convolution minimisation using L‑BFGS‑B (if SciPy is present)."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)       # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

            break
        vec = np.maximum(vec, 0.0)
        total = vec.sum()
        if total == 0.0:
            continue
        vec = vec / total
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        raise RuntimeError("No valid seeds were generated.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start refinement (SA → EG → optional L‑BFGS)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # share roughly 12 % of the remaining time per remaining seed
        share = remaining * 0.12 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # ---- (a) continuous simulated annealing ----
        seq_sa, val_sa = _continuous_sa(seed_vec.copy(),
                                         n,
                                         sub_deadline,
                                         rng,
                                         max_iters=250_000)

        # ---- (b) exponentiated‑gradient polishing ----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_sa,
                                               n,
                                               sub_deadline,
                                               max_iters=800_000,
                                               eta0=6.0)
        else:
            seq_eg, val_eg = seq_sa, val_sa

        # ---- (c) optional L‑BFGS smoothing ----
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               beta_schedule=[1.0, 5.0, 20.0,
                                                              80.0, 200.0, 500.0,
                                                              1_000.0, 2_000.0,
                                                              5_000.0, 20_000.0,
                                                              50_000.0, 100_000.0],
                                               max_iter_per_beta=1800)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_eg, val_eg

        # ---- (d) short second SA (helps escape local traps) ----
        if time.time() < sub_deadline:
            seq_sa2, val_sa2 = _continuous_sa(seq_lbfgs.copy(),
                                              n,
                                              sub_deadline,
                                              rng,
                                              max_iters=80_000)
        else:
            seq_sa2, val_sa2 = seq_lbfgs, val_lbfgs

        # Pick best refinement of this seed
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_sa, val_sa),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_sa2, val_sa2)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long EG polishing (as much time as left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=5_000_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Tiny final permutation tweak (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 30.0
        seq_last, val_last = _continuous_sa(global_best_seq.copy(),
                                            n,
                                            short_deadline,
                                            rng,
                                            max_iters=100_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final tweak] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
