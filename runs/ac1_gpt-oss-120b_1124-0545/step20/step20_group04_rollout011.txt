    all_candidates.sort(key=lambda x: x[1])
    best_seq = all_candidates[0][0].copy()
    best_val = all_candidates[0][1]
    print(f"Initial best surrogate {best_val}")

    # refinement
    num_refine = min(5, len(all_candidates))
    for idx in range(num_refine):
        if time.time() >= deadline: break
        seed_vec = all_candidates[idx][0]
        # allocate time share
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.4 / (num_refine - idx)
        # L-BFGS
        seq_lbfgs = _lbfgs_smooth_optimize(seed_vec, n, sub_deadline)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        # EG
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs, n, sub_deadline, max_iters=500_000, eta0=7.0)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # pick best
        candidates = [(seed_vec, _objective_surrogate(seed_vec)), (seq_lbfgs, val_lbfgs), (seq_eg, val_eg)]
        seq_local, val_local = min(candidates, key=lambda x: x[1])
        if val_local < best_val:
            best_seq, best_val = seq_local, val_local
            print(f"New global best after ref {idx+1}: {best_val}")

    # final EG polishing
    if time.time() < deadline:
        seq_final, val_final = _exp_gradient_opt(best_seq, n, deadline, max_iters=2_000_000, eta0=6.0)
        if val_final < best_val:
            best_seq, best_val = seq_final, val_final
            print(f"Final EG improved to {best_val}")

    # final L-BFGS smoothing (optional)
    if time.time() < deadline:
        seq_final_lbfgs = _lbfgs_smooth_optimize(best_seq, n, deadline)
        val_final_lbfgs = _objective_surrogate(seq_final_lbfgs)
        if val_final_lbfgs < best_val:
            best_seq, best_val = seq_final_lbfgs, val_final_lbfgs
            print(f"Final L-BFGS improved to {best_val}")

    # evaluate exact
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"Exact evaluation: {true_val}")
    except Exception as e:
        print(f"Evaluate error: {e}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

Propose a step‑function (non‑negative vector) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds in several stages:

1.  Rich seed generation: analytic windows (Hamming, Kaiser, etc.),
    new families (inverse‑sqrt, inverse‑quadratic, Chebyshev, flat‑top,
    DPSS), plus many random multi‑Gaussian and sparse seeds.

2.  Fast surrogate evaluation (the objective for ℓ₁‑normalised vectors
    equals ``2·n·max(conv)``) using an FFT‑based convolution.

3.  Pairwise ternary‑search mixing of the top seeds and a few random
    triple mixes.

4.  Refinement of the best few candidates:
    * continuation L‑BFGS on a smooth log‑sum‑exp approximation of the
      max‑convolution,
    * exponentiated‑gradient (EG) polishing on the exact max‑convolution.

5.  Final long EG polishing and a simple hill‑climbing loop that repeatedly

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _objective_surrogate(seq: np.ndarray) -> float:
    """Surrogate objective = 2·n·max(conv(seq, seq)) for sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0

                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _double_gaussian(n: int, sigma_frac: float = 0.15,

def _lowfreq_curve(n: int, num_freq: int,
                   rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves; clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _multi_gaussian(n: int, rng: np.random.Generator,
                    max_components: int = 6,

    """Random sum of a few Gaussian bumps."""

def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, cut negative entries, renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b, already ℓ₁‑normalised."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


def _best_mixed_pair(a: np.ndarray, b: np.ndarray,
                     deadline: float, steps: int = 30):
    """Ternary search for the best convex combination of a and b."""
    lo, hi = 0.0, 1.0
    best_val = None
    best_seq = None
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)
        if val1 < val2:
            hi = m2
            if best_val is None or val1 < best_val:
                best_val = val1
                best_seq = seq1
        else:
            lo = m1
            if best_val is None or val2 < best_val:
                best_val = val2
                best_seq = seq2
    if best_seq is None:
        best_seq = a.copy()
        best_val = _objective_surrogate(best_seq)
    else:
        mid = (lo + hi) / 2.0
        cand = _mix_shapes(a, b, mid)
        cand_val = _objective_surrogate(cand)
        if cand_val < best_val:
            best_seq = cand
            best_val = cand_val
    return best_seq, best_val


# ---------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ---------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 8000):
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space to stay on the probability simplex.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0,
                         2000000.0, 5000000.0,
                         10000000.0, 20000000.0]

    v = np.log(start_vec + 1e-30)   # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # Gradient wrt a
            w = exp_beta / sum_exp                     # weights over conv entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # Chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] Exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ---------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on exact max
# ---------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry (index k)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ---------------------------------------------------------
# Main entry point
# ---------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence found (list of floats) and stores it in the global
    variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for final evaluation / printing
    deadline = start_time + budget_s - 5.0

    # -------------------------------------------------
    # Problem size
    # -------------------------------------------------
    n = 4000

    # -------------------------------------------------
    # Load previous solution if present
    # -------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, list) and len(previous) == n:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    # -------------------------------------------------
    # 1. Seed generation
    # -------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = []

    # Analytic windows (existing)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        seeds.append(_sine_window(n, gamma=gamma))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in np.linspace(1.0, 15.0, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.09, 0.26, 14):
        seeds.append(_gaussian(n, sigma))
    for sigma in (0.10, 0.12):
        for shift in (0.20, 0.25, 0.30):
            seeds.append(_double_gaussian(n, sigma_frac=sigma, shift_frac=shift))
    for p in np.linspace(0.0, 2.0, 13):
        seeds.append(_power(n, p))

    # Beta‑shape grid
    alphas = np.linspace(0.2, 5.0, 7)
    betas = np.linspace(0.2, 5.0, 7)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    for alpha in np.linspace(0.1, 0.9, 7):
        seeds.append(_tukey_window(n, alpha))

    # Exponential / logistic families (new carefully sampled)
    for gamma in np.linspace(0.1, 8.0, 16):
        seeds.append(_exp_decay(n, gamma))
    for steep in np.linspace(0.5, 30.0, 12):
        for shift in np.linspace(0.1, 0.9, 11):
            seeds.append(_logistic_decay(n, steep, shift))

    # New analytic families
    seeds.append(_inverse_sqrt(n))
    seeds.append(_inverse_quad(n))
    if _sci_windows is not None:
        # flattop, Chebyshev, DPSS (if SciPy is available)
        seeds.append(_flattop_window(n))
        seeds.append(_chebwin_window(n, att=100.0))
        seeds.append(_dpss_window(n, NW=3.0))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Alternating Dirichlet pattern
    seeds.append(_alternating_dirichlet(n, rng))

    # Random multi‑Gaussian mixtures
    for _ in range(80):
        seeds.append(_multi_gaussian(n, rng))

    # Random sparse vectors
    for _ in range(40):
        seeds.append(_random_sparse(n, rng, target_nonzero=80))

    # Perturbations of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(6):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # -------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # -------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(15, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # -------------------------------------------------
    # 3. Pairwise and triple mixing of top seeds
    # -------------------------------------------------
    print("\n--- Mixing top seeds ---")
    mix_deadline = deadline - 2.0
    mixed_candidates = []

    # Pairwise ternary search mixing
    for i in range(len(top_seeds)):
        for j in range(i + 1, len(top_seeds)):
            if time.time() > mix_deadline:
                break
            seq_mix, val_mix = _best_mixed_pair(top_seeds[i],
                                                top_seeds[j],
                                                mix_deadline,
                                                steps=30)
            mixed_candidates.append((seq_mix, val_mix))

    # Random triple mixes
    for _ in range(8):
        if time.time() > mix_deadline:
            break
        a, b, c = rng.choice(top_seeds, size=3, replace=False)
        weights = rng.dirichlet([1.0, 1.0, 1.0])
        mix = weights[0] * a + weights[1] * b + weights[2] * c
        mix /= mix.sum()
        mixed_candidates.append((mix, _objective_surrogate(mix)))

    all_candidates = evaluated[:top_k] + mixed_candidates
    all_candidates.sort(key=lambda x: x[1])
    best_seq = all_candidates[0][0].copy()
    best_val = all_candidates[0][1]
    print(f"\n[Init] best surrogate after mixing = {best_val:.6f}")

    # -------------------------------------------------
    # 4. Refinement of the best few candidates (L‑BFGS + EG)
    # -------------------------------------------------
    refined_candidates = []
    num_refine = min(5, len(all_candidates))

    for idx in range(num_refine):
        if time.time() >= deadline:
            break
        seed_vec = all_candidates[idx][0]
        remaining = deadline - time.time()
        # allocate a share of the remaining time to this refinement
        sub_deadline = time.time() + remaining * 0.45 / (num_refine - idx)

        print(f"\n[Refine #{idx+1}] start surrogate = {_objective_surrogate(seed_vec):.6f}")

        # L‑BFGS smoothing (smooth max)
        seq_lbfgs = _lbfgs_smooth_optimize(seed_vec,
                                           n,
                                           sub_deadline,
                                           beta_schedule=None,
                                           max_iter_per_beta=10000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)

        # EG polishing (exact max); short if time is low
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs,
                                               n,
                                               sub_deadline,
                                               max_iters=300_000,
                                               eta0=6.5)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # Choose the best among seed, LBFGS, EG
        cand_list = [(seed_vec, _objective_surrogate(seed_vec)),
                     (seq_lbfgs, val_lbfgs),
                     (seq_eg, val_eg)]
        local_best_seq, local_best_val = min(cand_list, key=lambda x: x[1])

        print(f"[Refine #{idx+1}] local best surrogate = {local_best_val:.6f}")

        refined_candidates.append((local_best_seq, local_best_val))

        if local_best_val < best_val:
            best_seq, best_val = local_best_seq, local_best_val
            print(f"  >>> Updated global best = {best_val:.6f}")

    # -------------------------------------------------
    # 5. Final long EG polishing (full remaining budget)
    # -------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        seq_final, val_final = _exp_gradient_opt(best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=2_000_000,
                                                 eta0=5.5)
        if val_final < best_val:
            best_seq, best_val = seq_final, val_final
            print(f"[Final EG] improved to {best_val:.6f}")

    # -------------------------------------------------
    # 6. Additional hill‑climbing loop (perturb + short EG)
    # -------------------------------------------------
    while time.time() < deadline - 0.5:
        cand = _perturb(best_seq, rng, sigma=0.0015)
        # give each perturbation a modest budget (≈2 % of remaining time)
        sub_deadline = time.time() + 0.02 * (deadline - time.time())
        cand_seq, cand_val = _exp_gradient_opt(cand,
                                               n,
                                               sub_deadline,
                                               max_iters=150_000,
                                               eta0=5.0)
        if cand_val < best_val:
            best_seq, best_val = cand_seq, cand_val
            print(f"[Hill‑climb] improved to {best_val:.6f}")

    # -------------------------------------------------
    # 7. Optional final L‑BFGS smoothing
    # -------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final L‑BFGS smoothing ---")
        seq_final_lbfgs = _lbfgs_smooth_optimize(best_seq,
                                                n,
                                                deadline,
                                                beta_schedule=None,
                                                max_iter_per_beta=10000)
        val_final_lbfgs = _objective_surrogate(seq_final_lbfgs)
        if val_final_lbfgs < best_val:
            best_seq, best_val = seq_final_lbfgs, val_final_lbfgs
            print(f"[Final L‑BFGS] improved to {best_val:.6f}")

    # -------------------------------------------------
    # 8. Official evaluation with the provided function
    # -------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
