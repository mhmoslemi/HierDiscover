"""
Propose a non‑negative step‑function (height sequence) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The method below works only with the public ``evaluate_sequence`` function.
It starts from the previously best sequence (if any) or from a random
Dirichlet draw, then performs a long simulated‑annealing / mirror‑descent
search that moves mass between positions and swaps entries.  The objective
``_objective_surrogate`` (2·n·max(convolution)) is exactly the value returned
by ``evaluate_sequence`` for any ℓ₁‑normalised vector, so we can optimise it
directly.  The algorithm respects the whole time budget (default 1000 s) and
stores the best sequence found in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a vector with sum = 1:
        2·n·max(conv(seq, seq))
    The returned value coincides with ``evaluate_sequence`` for a
    ℓ₁‑normalised sequence.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random ℓ₁‑normalised vector from a Dirichlet distribution."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    if s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


# ----------------------------------------------------------------------
# Simulated‑annealing / mirror‑descent search
# ----------------------------------------------------------------------
def _anneal(
    cur_seq: np.ndarray,
    cur_val: float,
    n: int,
    start_time: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Long annealing loop that moves mass between entries and swaps positions.
    Returns the best sequence seen and its surrogate value.
    """
    best_seq = cur_seq.copy()
    best_val = cur_val

    # Annealing schedule parameters
    total_time = deadline - start_time
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        # linear cooling from 1e-2 to 1e-6
        T = 1e-2 * (1.0 - elapsed / total_time) + 1e-6 * (elapsed / total_time)
        # step size for mass transfer – shrinks from 0.02 to 1e-4
        eps = 0.02 * (1.0 - elapsed / total_time) + 1e-4 * (elapsed / total_time)

        # ------------------------------------------------------------------
        # propose a new candidate
        # ------------------------------------------------------------------
        if rng.random() < 0.5:                         # ----- swap move -----
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                          # -- transfer move --
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            # transfer a small fraction of mass from i to j
            delta = rng.uniform(0.0, eps) * cur_seq[i]
            if delta == 0.0:
                continue
            ns = cur_seq.copy()
            ns[i] -= delta
            ns[j] += delta

        # ------------------------------------------------------------------
        # evaluate
        # ------------------------------------------------------------------
        nval = _objective_surrogate(ns)

        # acceptance rule (Metropolis)
        if nval < cur_val or rng.random() < math.exp(-(nval - cur_val) / (T + 1e-12)):
            cur_seq = ns
            cur_val = nval
            if nval < best_val:
                best_val = nval
                best_seq = ns.copy()
                print(f"[{elapsed:6.1f}s] New best surrogate = {best_val:.6f}")

        # occasional random restart to escape possible plateaus
        if it % 100_000 == 0 and it > 0:
            # use a fresh Dirichlet draw (small concentration -> many tiny pieces)
            cand = _random_dirichlet(n, conc=0.03, rng=rng)
            cand_val = _objective_surrogate(cand)
            # keep the restart only if it looks promising
            if cand_val < best_val:
                best_seq, best_val = cand, cand_val
                cur_seq, cur_val = cand, cand_val
                print(f"[{elapsed:6.1f}s] Restart improved best to {best_val:.6f}")
            else:
                # otherwise replace current point (gives a chance to explore a new basin)
                cur_seq, cur_val = cand, cand_val

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget; after finishing it writes the best
    sequence to the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # small safety margin

    # --------------------------------------------------------------
    # Problem size (fixed – the challenge uses 4000 entries)
    # --------------------------------------------------------------
    n = 4000

    # Load previous solution if it exists and has the correct size
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) == n:
        try:
            cur_seq = np.asarray(prev, dtype=np.float64)
            cur_seq = np.maximum(cur_seq, 0.0)
            s = cur_seq.sum()
            if s > 0.0:

                cur_val = _objective_surrogate(cur_seq)
                print("[Info] Loaded previous best sequence.")
            else:
                raise ValueError
        except Exception:
            cur_seq = None
    else:
        cur_seq = None

    # --------------------------------------------------------------
    # If no valid previous solution, start from a random Dirichlet draw
    # --------------------------------------------------------------
    if cur_seq is None:
        # small concentration → many tiny components, which is a good

        cur_seq = _random_dirichlet(n, conc=0.03, rng=rng)
        cur_val = _objective_surrogate(cur_seq)
        print("[Info] No previous sequence – starting from random Dirichlet.")

    # --------------------------------------------------------------
    # Heavy simulated‑annealing / mirror‑descent refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Annealing & mass‑transfer search ---")
        best_seq, best_val = _anneal(cur_seq, cur_val, n, start_time, deadline, rng)
    else:
        best_seq, best_val = cur_seq, cur_val

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    true_val = None
    try:

        print(
            f"\n=== Finished ===\nSurrogate = {best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("[Error] evaluate_sequence raised an exception:", exc)

    # Store for future runs

    return best_seq.tolist()
