import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0/v.shape[0])
    return e / s

def _objective_surrogate(seq):
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _random_dirichlet(n, conc, rng):
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _perturb(seq, rng, sigma=0.002):
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

def _mixture_gaussian(params, n):
    K = 5
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2*K], 0.0, 1.0)
    sigma = np.clip(params[2*K:3*K], 0.001, 0.5)
    w_sum = w.sum()
    if w_sum > 0:
        w = w / w_sum
    else:
        w = np.full(K, 1.0 / K)
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k])**2) / (2.0 * sigma[k]**2))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s

def _exp_gradient_opt(start_seq, n, deadline, max_iters=300000, eta0=5.0):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # build subgradient
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _lbfgs_smooth_optimize(start_vec, n, deadline, beta_schedule=None, max_iter_per_beta=4000):
    try:
        import scipy.optimize as _opt
    except Exception:
        return start_vec
    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                         1000.0, 2000.0, 5000.0, 20000.0, 50000.0,
                         100000.0, 200000.0, 500000.0, 1000000.0]
    v = np.log(start_vec + 1e-30)
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)
            # gradient w.r.t a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)   # length = 2n-1
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()
        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break
    return _softmax(v)

def _beta_distribution_shape(n, alpha, beta_param):
    # a_i ∝ (i+1)^{alpha-1} * (n-i)^{beta_param-1}
    i = np.arange(n) + 1
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (beta_param - 1))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Multi‑restart search for a step‑function that minimises ``evaluate_sequence``.
    The algorithm:
      1. Build a diverse pool of random seeds (uniform, Dirichlet, Gaussian mixtures,
         Beta‑distribution shapes).
      2. For each seed run a short L‑BFGS smoothing (log‑sum‑exp max) and a longer
         exponentiated‑gradient polish.
      3. Keep the best sequence seen (lowest surrogate value) and finally evaluate
         it with ``evaluate_sequence``.
    All stages respect the overall time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a safety margin
    n = 4000

    # Load previous solution if any
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, list) and len(prev) == n:
        prev_arr = np.asarray(prev, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1. Build seed pool
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]
    if prev_arr is not None and prev_arr.size == n:
        seeds.append(prev_arr / prev_arr.sum())

    # Dirichlet draws with a range of concentrations
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Small perturbations of uniform
    for _ in range(3):
        seeds.append(_perturb(_uniform(n), rng, sigma=0.004))

    # Random Gaussian mixture seeds
    K = 5
    for _ in range(5):
        params = np.concatenate([
            rng.random(K),                         # weights
            rng.random(K),                         # means
            rng.uniform(0.001, 0.5, K)             # sigmas
        ])
        seeds.append(_mixture_gaussian(params, n))

    # Beta‑distribution shaped seeds (alpha,beta) randomised
    for _ in range(5):
        a_param = rng.uniform(0.2, 5.0)
        b_param = rng.uniform(0.2, 5.0)
        seeds.append(_beta_distribution_shape(n, a_param, b_param))

    # ------------------------------------------------------------------
    # 2. Evaluate seeds and keep the best
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    print("\n--- Evaluating initial seeds (surrogate) ---")
    for idx, s in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(s)
        print(f"seed #{idx+1:02d} surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = s.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop – run a number of independent restarts
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        # decide whether to start from a fresh random seed or from current best
        if restart == 0 and best_seq is not None:
            start_vec = best_seq.copy()
        else:
            # generate a new random Dirichlet seed (moderate concentration)
            conc = rng.uniform(0.1, 1.5)
            start_vec = _random_dirichlet(n, conc, rng)

        # allocate a fraction of remaining time for this restart
        now = time.time()
        remaining = deadline - now
        # we aim at at most 2 min per restart
        per_restart = min(120.0, remaining * 0.6)   # keep some budget for later
        if per_restart < 5.0:
            break
        restart_deadline = now + per_restart

        # --- L‑BFGS smoothing (moderate budget) ---
        seq_lbfgs = _lbfgs_smooth_optimize(start_vec, n,
                                            restart_deadline,
                                            beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                           200.0, 500.0, 1000.0,
                                                           2000.0, 5000.0,
                                                           20000.0, 50000.0],
                                            max_iter_per_beta=3000)
        val_l = _objective_surrogate(seq_lbfgs)
        # --- Exponentiated‑gradient polishing ---
        # give most time to EG
        eg_deadline = restart_deadline
        seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs, n, eg_deadline,
                                           max_iters=500000, eta0=5.0)
        print(f"[restart {restart}] EG surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print(f"  >>> New global best = {best_val:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # 4. Final polishing (extra L‑BFGS after best found)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        final_deadline = deadline
        seq_final = _lbfgs_smooth_optimize(best_seq, n,
                                           final_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1000.0,
                                                          2000.0, 5000.0],
                                           max_iter_per_beta=5000)
        val_final = _objective_surrogate(seq_final)
        print(f"[Final L‑BFGS] surrogate = {val_final:.6f}")
        if val_final < best_val:
            best_val, best_seq = val_final, seq_final.copy()
            print(f"  >>> Updated global best after final L‑BFGS = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
Multi‑restart optimisation for the “min‑max convolution” problem.

The objective (after normalising the sequence to sum = 1) is
    F(a) = 2·n·max_k (a * a)[k] .
A low‑dimensional parametric family (Gaussian mixtures, beta‑shapes,
periodic patterns, Dirichlet draws) is sampled to obtain diverse seeds.
Each seed is refined in two stages:

1. L‑BFGS on a smooth approximation of max (log‑sum‑exp) – this gives a
   fast, gradient‑based push toward low‑peak convolutions.
2. Exact exponentiated‑gradient (mirror descent) on the true max‑convolution
   using the sub‑gradient that corresponds to the active index.

The whole process runs many independent restarts (limited by the global
time budget) and keeps the best sequence seen.  Finally the best candidate
is evaluated with the provided ``evaluate_sequence`` function and stored in
the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# ----------------------------------------------------------------------
# Objective (surrogate = exact when sum(seq)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) for a normalised probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip, and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mixture_gaussian(params: np.ndarray, n: int) -> np.ndarray:
    """5‑component Gaussian mixture on [0,1] (weights are normalised internally)."""
    K = 5
    w = np.maximum(params[0:K], 0.0)
    mu = np.clip(params[K:2 * K], 0.0, 1.0)
    sigma = np.clip(params[2 * K:3 * K], 0.001, 0.5)
    w_sum = w.sum()
    if w_sum > 0:
        w = w / w_sum
    else:
        w = np.full(K, 1.0 / K)

    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    a = np.zeros(n, dtype=np.float64)
    for k in range(K):
        a += w[k] * np.exp(-((i - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))

    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _beta_distribution_shape(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Discrete Beta distribution on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (beta_param - 1))
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _periodic_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sinusoidal pattern with random period and phase, shifted to non‑negative."""
    period = rng.integers(20, 200)
    phase = rng.uniform(0.0, 2.0 * np.pi)
    x = np.arange(n, dtype=np.float64)
    a = np.sin(2.0 * np.pi * x / period + phase)
    a = a - a.min()                    # make all values ≥0
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Exact mirror‑descent (exponentiated‑gradient) optimiser on the true max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Sub‑gradient mirror descent (log‑space) on max(conv(a,a))."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (active index = k)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth max optimisation with L‑BFGS (log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _lbfgs_obj(v: np.ndarray, n: int, beta: float):
    """Smoothed max‑convolution objective and its gradient (log‑space)."""
    a = _softmax(v)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (weights from the smooth max)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)               # length = 2n‑1
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # push the gradient through the softmax (log‑space)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000):
    """
    L‑BFGS‑B on the smoothed objective.
    If SciPy is unavailable the input vector is returned unchanged.
    """
    try:
        import scipy.optimize as _opt
    except Exception:                     # pragma: no cover
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0,
                         20000.0, 50000.0,
                         100000.0, 200000.0,
                         500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        try:
            res = _opt.minimize(fun=_lbfgs_obj,
                                x0=v,
                                args=(n, beta),
                                method='L-BFGS-B',
                                jac=True,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except Exception:                 # pragma: no cover
            break
    return _softmax(v)

    """

    start_time = time.time()

    n = 4000                                          # problem size

    # --------------------------------------------------------------
    # Load previous solution if it exists
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(previous, list) and len(previous) == n:

    # --------------------------------------------------------------

            rng.random(K),                         # means

        seeds.append(_mixture_gaussian(params, n))

    # Random Beta‑distribution shapes
    for _ in range(5):
        a_param = rng.uniform(0.2, 5.0)
        b_param = rng.uniform(0.2, 5.0)

    for _ in range(5):

    # --------------------------------------------------------------
    # 2️⃣ Evaluate the seed pool and keep the current best
    # --------------------------------------------------------------

    best_val = np.inf

    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break

            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # 3️⃣ Multi‑restart optimisation (L‑BFGS → EG)

    restart = 0
    while time.time() < deadline:
        # Choose a fresh starting point.  First restart re‑uses the
        # current best, later ones use random Dirichlet draws.

        else:

            start_vec = _random_dirichlet(n, conc, rng)

        now = time.time()
        remaining = deadline - now
        # Allocate at most 120 s (or a generous share of the remaining budget)
        per_restart = min(120.0, remaining * 0.6)
        if per_restart < 5.0:
            break
        restart_deadline = now + per_restart

        seq_lbfgs = _lbfgs_smooth_optimize(start_vec, n,
                                           restart_deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1000.0,
                                                          2000.0, 5000.0,
                                                          20000.0, 50000.0],
                                           max_iter_per_beta=3000)

        # --- Exact EG polishing (takes the bulk of the allocated time) ---
        seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs, n,
                                           restart_deadline,
                                           max_iters=800_000,
                                           eta0=5.0)

        if val_eg < best_val:

            print(f"  >>> New global best = {best_val:.6f}")

    # 4️⃣ Final short L‑BFGS pass (clean‑up)

    if best_seq is not None and time.time() < deadline:
        print("\n--- Final short L‑BFGS pass ---")
        final_seq = _lbfgs_smooth_optimize(best_seq, n,
                                           deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1000.0,
                                                          2000.0, 5000.0],
                                           max_iter_per_beta=5000)
        final_val = _objective_surrogate(final_seq)
        print(f"[Final L‑BFGS] surrogate = {final_val:.6f}")
        if final_val < best_val:
            best_val, best_seq = final_val, final_seq.copy()
            print(f"  >>> Updated global best after final L‑BFGS = {best_val:.6f}")

    # --------------------------------------------------------------
    # 5️⃣ Official evaluation with the supplied `evaluate_sequence`
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:               # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible later calls
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
