def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _continuation_lbfgs(start_vec, n, deadline, beta_schedule=None, max_iter_per_beta=1500):
    if _opt is None:
        return start_vec
    if beta_schedule is None:
        beta_schedule = [...]
    v = np.log(start_vec + 1e-30)
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()
        try:
            res = _opt.minimize(fun=fun, x0=v, method='L-BFGS-B', jac=True, callback=callback,
                                options={'maxiter': max_iter_per_beta, 'disp': False, 'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:
            print(f"[L-BFGS] Exception {e} at beta {beta}")
            break
    return _softmax(v)

def _exp_gradient_opt(start_seq, n, deadline, max_iters=500000, eta0=5.0, single_active=False):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set
        if single_active:
            k = int(conv.argmax())
            active = np.array([k])
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]
        # compute subgradient
        g_tot = np.zeros_like(a)
        for idx in active:
            i_start = max(0, idx - (n-1))
            i_end = min(n-1, idx)
            i_range = np.arange(i_start, i_end+1, dtype=np.int64)
            rev_idx = idx - i_range
            g_tot[i_range] += a[rev_idx]
        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _simulated_annealing(start_seq, n, deadline, rng, max_attempts=150000, start_sigma=0.001, T_start=0.01, T_end=1e-8):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    cur_val = best_val
    attempts = 0
    while attempts < max_attempts and time.time() < deadline:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)
        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns = ns / s
        val = _objective(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1
    return best_seq, best_val

    The search uses a large set of analytic seed families, a continuation
    L‑BFGS smoothing of the max‑convolution (log‑sum‑exp), followed by
    exponentiated‑gradient descent and a brief simulated‑annealing polish.
    """
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    n = 6000  # dimension – larger gives more freedom but modest cost

    # ------------------------------------------------------------------
    # 1. Generate candidate seeds from various parametric families
    # ------------------------------------------------------------------
    seeds = []

    # basic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))

    # beta shapes (a few combos)
    for a in (0.2, 0.5, 1.0, 2.0, 5.0):

            seeds.append(_beta_shape(n, a, b))

        for mix in (0.1, 0.3, 0.5, 0.7, 0.9):
            seeds.append(_mix_shapes(_uniform(n), exp_seq, mix))

        for _ in range(3):

    # mirror versions (right‑skew)
    mirrored = [s[::-1].copy() for s in list(seeds)]
    seeds.extend(mirrored)

    # ------------------------------------------------------------------
    # 2. Normalise and evaluate all seeds (surrogate)
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:

"""
Propose a step‑function sequence that minimises

    evaluate_sequence(sequence) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The search proceeds in several phases:

1️⃣  **Seeding** – a rich palette of analytic families (uniform,
    triangular, sine‑window, beta‑shapes, exponentials, power‑laws,
    mixtures) plus random Dirichlet draws and their mirrored versions.
    All candidates are ℓ₁‑normalised and evaluated with the exact (surrogate)
    objective 2·n·max(conv).

2️⃣  **Continuation L‑BFGS** – the nonsmooth max‑convolution is replaced by
    a smooth log‑sum‑exp surrogate.
    For a schedule of increasing ‘β’ values we minimise the surrogate with
    SciPy’s L‑BFGS‑B, always keeping the variables in the simplex via a
    log‑softmax parametrisation.  Very few hundred L‑BFGS iterations per β
    already give a high‑quality refinement.

3️⃣  **Global polishing** – the current best is fed to a long
    exponentiated‑gradient (mirror‑descent) optimiser that uses the exact
    max‑convolution sub‑gradient (single‑active index).  The step‑size decays
    as η₀/√t.

4️⃣  **Final simulated‑annealing** – a short low‑temperature SA run
    (Gaussian perturbations) to escape tiny local bumps.

All phases respect the global 1000 s budget.  The final sequence is stored
in the global variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – yields a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors of length n
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i), i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _continuation_lbfgs(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       beta_schedule=None,
                       max_iter_per_beta: int = 1500):
    """Smoothly minimise 2·n·max(conv) by log‑sum‑exp continuation."""
    if _opt is None:
        # SciPy unavailable – return unchanged
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0, 5_000_000.0,
            20_000_000.0, 100_000_000.0
        ]

    # work in log‑softmax space (guarantees feasibility)
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()                           # numerical stability
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth objective (log‑sum‑exp)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                  # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule for log‑softmax (mirror‑descent)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            # abort if we are out of the time budget
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      eta0: float = 5.0,
                      single_active: bool = False):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    If ``single_active`` is True only the most active index is taken for
    the sub‑gradient.  Returns the best sequence encountered and its value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # log‑space variables – guarantees positivity & ℓ₁‑norm
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # ----- active set -------------------------------------------------
        if single_active:
            k = int(conv.argmax())
            active = np.array([k], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        # ----- sub‑gradient (exact for the active indices) ---------------
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            # average over the active set
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # ----- learning‑rate (1/√t decay) -------------------------------
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 150_000,
                         start_sigma: float = 0.001,
                         T_start: float = 0.01,
                         T_end: float = 1e-9):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val
