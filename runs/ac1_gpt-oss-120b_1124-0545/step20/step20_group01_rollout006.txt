"""
step‑function optimiser – fast multi‑seed L‑BFGS + exponentiated‑gradient polishing

* generate a compact pool of analytically‑inspired and random Dirichlet seeds
* keep the few best (by the exact surrogate 2·n·max(conv(a,a)) )
* for each seed (plus a random permutation) run a long L‑BFGS optimisation on a
  smooth approximation of the max‑convolution (log‑sum‑exp with a growing β)
* finish with a high‑precision exponentiated‑gradient (mirror‑descent) run on the
  true max‑convolution, and a final short permutation hill‑climb
* the best sequence is stored in the global ``height_sequence_1`` for continuation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – used for the L‑BFGS phase.
# ----------------------------------------------------------------------
try:                                          # pragma: no cover
    import scipy.optimize as _opt
except Exception:                            # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Numerical utilities (FFT‑based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a probability vector (sum = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·|i-(n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosines, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution (sub‑gradient on the active index)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based hill climbing (simulated‑annealing style)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on the surrogate objective."""
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                     # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                       # Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.

    Strategy:
      1. build a modest pool of analytic / random seeds,
      2. keep the top‑few by the exact surrogate,
      3. for each seed (and a random permutation) run an extended L‑BFGS
         optimisation on a smoothed max‑convolution (log‑sum‑exp),
      4. finish with a long exponentiated‑gradient run on the true max,
      5. a short permutation anneal for the final touch,
      6. return the best sequence and store it globally.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0   # keep a small safety margin

    # --------------------------------------------------------------
    # problem dimension – a few thousand entries give a good trade‑off
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 1. seed generation
    # --------------------------------------------------------------
    seeds = []

    # deterministic families – a concise grid
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for sigma_frac in (0.08, 0.12, 0.18, 0.24):
        seeds.append(_gaussian(n, sigma_frac))

    for rate in (0.001, 0.003, 0.007, 0.015):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    for conc in (0.1, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    for nf in (3, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # previous best (if present) – gives continuity between runs
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # --------------------------------------------------------------
    # 2. quick surrogate evaluation – keep the very best few
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(4, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seed surrogate values ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. refinement of each top seed
    # --------------------------------------------------------------
    beta_schedule = [
        0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0,
        200.0, 500.0, 1_000.0, 2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0, 100_000.0,
        200_000.0, 500_000.0, 1_000_000.0, 2_000_000.0,
        5_000_000.0, 10_000_000.0
    ]

    for si, base_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # we will treat the original vector and one random permutation as two
        # separate starting points – this helps to escape ordering traps.
        start_variants = [base_vec, np.random.permutation(base_vec)]

        for variant_idx, start_vec in enumerate(start_variants):
            sub_deadline = deadline
            print(f"\n[Seed {si+1}/{top_k} – variant {variant_idx}] L‑BFGS start")

            # -----------------------------------------------------------------
            # L‑BFGS on smoothed max‑convolution (if SciPy is available)
            # -----------------------------------------------------------------
            if _opt is not None:
                v = np.log(start_vec + 1e-30)

                for beta in beta_schedule:
                    if time.time() > sub_deadline:
                        break
                    beta_val = float(beta)

                    def fun(v_local):
                        a = _softmax(v_local)
                        conv = _conv_fft(a, a)
                        max_c = np.max(conv)

                        # log‑sum‑exp with shift for stability
                        exp_beta = np.exp(beta_val * (conv - max_c))
                        sum_exp = exp_beta.sum()
                        obj = 2.0 * n * (max_c + np.log(sum_exp) / beta_val)

                        w = exp_beta / sum_exp
                        rev_a = a[::-1]
                        full = _conv_fft(rev_a, w)
                        grad_a = 2.0 * full[n - 1:n - 1 + n]

                        # gradient w.r.t. logits
                        dot = np.dot(grad_a, a)
                        grad_v = a * (grad_a - dot)
                        return obj, grad_v

                    # abort quickly if we are about to exceed the deadline
                    def callback(_):
                        if time.time() > sub_deadline:
                            raise KeyboardInterrupt()

                    try:
                        res = _opt.minimize(fun=fun,
                                            x0=v,
                                            method='L-BFGS-B',
                                            jac=True,
                                            callback=callback,
                                            options={'maxiter': 8000,
                                                     'disp': False,
                                                     'gtol': 1e-12})
                        v = res.x
                        a = _softmax(v)
                        val = _objective_surrogate(a)

                        if val < global_best_val:
                            global_best_seq = a.copy()
                            global_best_val = val
                            print(f"    * new global best (L‑BFGS) = {global_best_val:.6f}")
                    except KeyboardInterrupt:
                        break
                    except Exception as e:   # pragma: no cover
                        print(f"[L‑BFGS] exception {e} at β={beta_val}")
                        break
                # after L‑BFGS, take the resulting probability vector as start for EG
                start_vec = _softmax(v)
            else:
                # SciPy unavailable – fall back to the raw start vector
                start_vec = start_vec.copy()

            # -----------------------------------------------------------------
            # final exponentiated‑gradient polishing (exact max)
            # -----------------------------------------------------------------
            if time.time() < sub_deadline:
                print("[EG] final polishing")
                seq_eg, val_eg = _exp_gradient_opt(start_vec,
                                                  n,
                                                  sub_deadline,
                                                  max_iters=500_000,
                                                  eta0=5.0)
                if val_eg < global_best_val:
                    global_best_seq = seq_eg.copy()
                    global_best_val = val_eg
                    print(f"    * new global best (EG) = {global_best_val:.6f}")

    # ----------------------------------------------------------------------
    # 5. Tiny final permutation/annealing tweak (if any time left)
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final permutation tweak ---")
        short_deadline = time.time() + 30.0
        seq_last, val_last = _perm_hill_climb(global_best_seq.copy(),
                                              n,
                                              short_deadline,
                                              rng,
                                              max_iters=120_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final tweak] improved to {global_best_val:.6f}")

    # ----------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ----------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:               # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
