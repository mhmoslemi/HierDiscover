def compute_obj_and_grad(v, n, beta):
    a = softmax(v)
    conv = conv_fft(a, a)
    shift = np.max(conv)  # for stability
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    smooth_max = shift + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_max

    w = exp_beta / sum_exp  # shape (2n-1)
    rev_a = a[::-1]
    # compute conv_rev_a_w using FFT (len = 2n-1)
    grad = 2.0 * conv_fft(rev_a, w)  # returns length 2n-1
    # we need gradient for a corresponding to positions n-1:n-1+n (center)
    grad_a = grad[n-1:n-1+n]
    # gradient wrt v (logits):
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v

def adam_optimize(start_logit, n, deadline, beta_schedule, iters_per_beta):
    v = start_logit.copy()
    best_seq = softmax(v)
    best_obj = _objective_surrogate(best_seq)  # compute actual max conv.
    best_v = v.copy()
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2  # will tune maybe.
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            t += 1
            obj_smooth, grad_v = compute_obj_and_grad(v, n, beta)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_v
            v2 = beta2 * v2 + (1 - beta2) * (grad_v * grad_v)
            m_hat = m / (1 - beta1 ** t)
            v2_hat = v2 / (1 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v2_hat) + eps)

            # occasionally evaluate true surrogate
            if _ % 200 == 0:
                a = softmax(v)
                val = _objective_surrogate(a)
                if val < best_obj:
                    best_obj = val
                    best_seq = a.copy()
                    best_v = v.copy()

def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0 # safe.

    prev_seq = globals().get('height_sequence_1')
    if isinstance(prev_seq, (list, tuple)) and len(prev_seq) > 0:
        n = len(prev_seq)
        base_seq = np.array(prev_seq, dtype=np.float64)
    else:
        n = 4000
        base_seq = np.full(n, 1.0/n)

    # Normalize to sum 1
    base_seq = np.maximum(base_seq, 0)
    base_seq = base_seq / base_seq.sum()

    # Prepare list of start vectors for Adam restarts
    start_logit_list = []

    # Add previous solution (logits)
    start_logit_list.append(np.log(base_seq + 1e-30))

    # Add a few random Dirichlet draws
    for conc in (0.1, 0.3, 1.0):
        if time.time() > deadline: break
        rand_seq = rng.dirichlet(np.full(n, conc))
        start_logit_list.append(np.log(rand_seq + 1e-30))

    # Add some analytic shapes (like Gaussian with various sigma) (converted to logits)
    def add_shape(vec):
        if time.time() > deadline: return
        start_logit_list.append(np.log(vec + 1e-30))

    # Use Gaussian shapes
    for sigma_frac in (0.1, 0.15, 0.2):
        add_shape(_gaussian(n, sigma_frac))

    # Use Tukey shapes
    for alpha in (0.2, 0.5, 0.8):
        add_shape(_tukey_window(n, alpha))

    # Up to some number, limit to maybe 15 start vectors
    max_starts = 15
    start_logit_list = start_logit_list[:max_starts]

    # Optimization parameters
    beta_schedule = [1, 5, 20, 80, 200, 500, 1000, 2000, 5000, 20000, 50000, 100000, 200000, 500000, 1000000]
    iters_per_beta = 1000  # moderate

    best_seq = None

    for idx, v0 in enumerate(start_logit_list):
        if time.time() > deadline:
            break
        print(f"Restart {idx+1}/{len(start_logit_list)}")
        seq, val, v_opt = adam_optimize(v0, n, deadline, beta_schedule, iters_per_beta)
        if val < best_val:

    if time.time() < deadline - 2.0 and best_seq is not None:
        print("Local polishing...")
        seq = best_seq.copy()
        cur_val = best_val
        step_frac = 0.05
        it = 0
        while time.time() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j:

                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = seq.copy()
                    print(f"  iter {it}: improved best = {best_val:.6f}")

    # Final evaluation

    print(f"\nFinished. Surrogate: {best_val:.6f}, true: {true_val}")

    globals()["height_sequence_1"] = best_seq.tolist()

   yielding a smooth approximation whose gradient can be written in closed
   form.  For each seed we run an Adam optimiser in the *log‑softmax*
   (log‑probability) space, gradually increasing β (continuation).  The
   optimisation respects the simplex constraints automatically.

3. **Local stochastic polishing** – after the Adam phase we perform a cheap
   hill‑climber that moves a small fraction of mass between two random
   coordinates; any improvement is kept.  This typically removes the last
   few bits of sub‑optimality.

The whole routine respects the 1000 s wall‑time budget.  The best sequence
found is stored in the global ``height_sequence_1`` variable and returned as
a plain Python list.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors via FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# ----------------------------------------------------------------------
# Surrogate (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Exact objective for a vector with ℓ₁‑norm = 1:
       2·n·max(conv(a,a)).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian on {0,…,n‑1}.
       sigma_frac = σ / n (fraction of the length)."""
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Cosine‑tapered Tukey window."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed max / gradient utilities
# ----------------------------------------------------------------------
def _smooth_max_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Given log‑softmax parameters ``v`` (i.e. a = softmax(v)),
    compute the smooth approximation of the max‑convolution
    (log‑sum‑exp with temperature β) and its gradient w.r.t. ``v``.
    Returns (smooth_obj, grad_v).
    """
    a = _softmax(v)                     # probability vector, sum = 1
    conv = _conv_fft(a, a)              # length 2n‑1
    shift = np.max(conv)                # for numerical stability

    # log‑sum‑exp smoothing of the max
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    smooth_max = shift + np.log(sum_exp) / beta
    smooth_obj = 2.0 * n * smooth_max    # smooth surrogate of the true objective

    # gradient wrt a (see derivation in analysis)
    w = exp_beta / sum_exp              # weights, sum to 1
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length 2n‑1
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    return smooth_obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser (log‑softmax space) with continuation on β
# ----------------------------------------------------------------------
def _adam_optimize(start_logits: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule,
                   iters_per_beta: int = 2000,
                   lr: float = 0.2):
    """
    Adam optimisation of the smooth max‑conv objective.
    Returns best (sequence, surrogate value, logits) seen.
    """
    v = start_logits.copy()
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)          # second‑moment accumulator
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)
    best_v = v.copy()

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            t += 1
            smooth_obj, grad_v = _smooth_max_and_grad(v, n, beta)

            # Adam update (minimisation → move opposite to gradient)
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v * grad_v)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            step = lr * m_hat / (np.sqrt(v2_hat) + eps)
            v -= step

            # occasional exact surrogate check (every 200 steps)
            if (_ % 200) == 0:
                a = _softmax(v)
                val = _objective_surrogate(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    best_v = v.copy()
    return best_seq, best_val, best_v


# ----------------------------------------------------------------------
# Local stochastic polishing (tiny mass moves)
# ----------------------------------------------------------------------
def _local_polish(seq: np.ndarray,
                  n: int,
                  deadline: float,
                  rng: np.random.Generator):
    """
    Simple hill‑climber that repeatedly moves a fraction of mass from one
    coordinate to another.  Accepts only improving moves.
    """
    cur_val = _objective_surrogate(seq)
    best_seq = seq.copy()
    best_val = cur_val
    step_frac = 0.05
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = step_frac * seq[i] * rng.uniform(0.5, 1.5)
        delta = min(delta, seq[i])
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        val_new = _objective_surrogate(new_seq)
        if val_new < cur_val:
            seq = new_seq
            cur_val = val_new
            if cur_val < best_val:
                best_val = cur_val
                best_seq = seq.copy()
                print(f"  polishing iter {it}: new best surrogate = {best_val:.6f}")
            step_frac = min(0.2, step_frac * 1.02)
        else:
            step_frac = max(1e-5, step_frac * 0.9995)
        it += 1
        if it % 5000 == 0:
            print(f"  polishing iter {it}: current surrogate = {cur_val:.6f}")
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine combines several random restarts,
    Adam optimisation on a smooth max‑convolution surrogate, and a final
    stochastic polishing stage.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size (reuse previous length if available)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple)) and len(prev) > 0:
        n = len(prev)
        base = np.asarray(prev, dtype=np.float64)
    else:
        n = 4000
        base = _uniform(n)

    # Normalise to ℓ₁‑norm = 1
    base = np.maximum(base, 0.0)
    if base.sum() == 0.0:
        base = _uniform(n)
    base = base / base.sum()

    # ------------------------------------------------------------------
    # Build a collection of starting points (log‑softmax vectors)
    # ------------------------------------------------------------------
    start_logits = []

    # 1) previous best (if any)
    start_logits.append(np.log(base + 1e-30))

    # 2) a few random Dirichlet draws
    for conc in (0.05, 0.2, 0.8, 2.0):
        if time.time() > deadline:
            break
        rand_vec = rng.dirichlet(np.full(n, conc)).astype(np.float64)
        start_logits.append(np.log(rand_vec + 1e-30))

    # 3) simple analytic shapes
    def _add_shape(vec):
        if time.time() > deadline:
            return
        start_logits.append(np.log(vec + 1e-30))

    _add_shape(_uniform(n))
    _add_shape(_gaussian(n, 0.12))
    _add_shape(_gaussian(n, 0.18))
    _add_shape(_tukey_window(n, 0.3))
    _add_shape(_tukey_window(n, 0.7))

    # ------------------------------------------------------------------
    # Optimisation schedule
    # ------------------------------------------------------------------
    beta_schedule = [1, 5, 20, 80, 200, 500, 1000,
                     2000, 5000, 20000, 50000, 100000,
                     200000, 500000, 1000000]

    # Number of Adam steps per β – we can afford a few thousand
    iters_per_beta = 2500

    best_seq = None
    best_val = np.inf

    print(f"\n--- Adam optimisation (n={n}) ---")
    for idx, logit in enumerate(start_logits):
        if time.time() > deadline:
            break
        print(f" Restart {idx + 1}/{len(start_logits)}")
        seq, val, _ = _adam_optimize(logit, n, deadline,
                                      beta_schedule,
                                      iters_per_beta=iters_per_beta,
                                      lr=0.2)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"  New best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Local stochastic polishing (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("\n--- Local stochastic polishing ---")
        best_seq, best_val = _local_polish(best_seq, n, deadline, rng)

    # ------------------------------------------------------------------
    # Final true evaluation and bookkeeping
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
          f"true evaluate_sequence = {true_val}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
