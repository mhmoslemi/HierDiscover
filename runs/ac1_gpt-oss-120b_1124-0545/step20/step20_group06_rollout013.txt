def _adam_optimize(seed_a, n, deadline, beta_schedule, lr=0.1, max_iters_per_beta=2000):
    logits = np.log(seed_a + 1e-30)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0
    best_a = seed_a.copy()
    best_val = _objective_surrogate(seed_a)   # note: using surrogate (max conv) not smooth.
    for beta in beta_schedule:
        while time.time() < deadline:
            # compute gradient
            a = _softmax(logits)
            # compute smooth obj + gradient
            conv = _conv_fft(a, a)
            max_val = conv.max()
            # compute smooth version using current beta to compute w
            shift = max_val
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            obj_smooth = 2.0 * n * (shift + np.log(sum_exp) / beta)

            # compute w
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # gradient wrt logits
            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)

            # Adam update
            t += 1
            m = 0.9 * m + (1 - 0.9) * grad_logits
            v = 0.999 * v + (1 - 0.999) * (grad_logits ** 2)
            m_hat = m / (1 - 0.9 ** t)
            v_hat = v / (1 - 0.999 ** t)
            eps = 1e-8
            lr_t = lr / (np.sqrt(t) + 1)   # optionally decay
            logits = logits - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            # maintain best surrogate (actual max conv)
            cur_a = _softmax(logits)
            cur_val = _objective_surrogate(cur_a)
            if cur_val < best_val:
                best_val = cur_val
                best_a = cur_a.copy()
            # termination criteria maybe if t > max_iters or time.
            if t >= max_iters_per_beta:
                break
        # after finishing beta, maybe reset m/v? we could continue; we keep momentum.
    return best_a, best_val

"""
A fast, highly‑parallel optimiser for the step‑function bound.

The goal is to minimise

    2 * n * max(conv(a, a)) / (sum(a)**2)

which is invariant under scaling, therefore all candidates are ℓ₁‑normalised
(sum = 1) and constrained to be non‑negative.  The optimiser works in three
phases:

1. **Seeding** – a large deterministic / random pool of analytic families
   (uniform, beta, exponential, power‑law, anti‑symmetric sine/linear,
   mixtures …).  All seeds are ℓ₁‑normalised and evaluated with a cheap
   FFT‑based surrogate that returns 2·n·max(conv).  The best few are kept.

2. **Adam optimisation** – smooth approximation of the max‑convolution via
   log‑sum‑exp.  Adam (adaptive‑momentum) updates the log‑probabilities
   (mirror‑descent) while a schedule of increasing β makes the smooth
   maximum converge to the true maximum.  Each seed receives a short
   budget; the best result becomes the global incumbent.

3. **Polishing** – a long Adam run on the global incumbent, followed by an
   optional L‑BFGS refinement (if SciPy is present) and a final call to the
   official `evaluate_sequence`.  The best‑found sequence is stored in the
   global variable ``height_sequence_2`` and returned as a plain list.

The algorithm respects a 1000 s budget (2 s safety margin) and uses only
numpy, math and the optional scipy optimiser.  No file or network I/O is
performed.
"""

import time
import math
import numpy as np

# optional SciPy for the final L‑BFGS refinement
try:
    import scipy.optimize as _opt
except Exception:      # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Anti‑symmetric families (new – aim to reduce the reverse inner‑product)
# ----------------------------------------------------------------------
def _anti_linear(n: int, factor: float) -> np.ndarray:
    """
    Linear decreasing shape with clipping:
        a_i ∝ max(0, 1/n + factor * (mid - i))
    The factor is a scalar that controls the steepness.
    """
    mid = (n - 1) / 2.0
    base = 1.0 / n
    a = base + factor * (mid - np.arange(n, dtype=np.float64))
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _anti_sine(n: int, amp: float) -> np.ndarray:
    """
    Uniform base + an anti‑symmetric sine perturbation.
    a_i ∝ max(0, 1/n + amp * sin(π*(i/(n‑1)‑0.5))).
    """
    i = np.arange(n, dtype=np.float64)
    phi = np.pi * (i / (n - 1) - 0.5)
    a = 1.0 / n + amp * np.sin(phi)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule,
                   max_iters_per_beta: int = 1500,
                   lr0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed max‑convolution.
    Returns the best ℓ₁‑normalised sequence encountered and its *exact*
    surrogate value (using the true max, not the smoothed version).
    """
    # work in log‑probability space
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_seq = None
    best_val = np.inf
    it_tot = 0

    for beta in beta_schedule:
        for _ in range(max_iters_per_beta):
            if time.time() > deadline:
                break

            # probability vector
            a = _softmax(log_a)

            # smoothed max via log‑sum‑exp
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            obj_smooth = 2.0 * n * (shift + np.log(sum_exp) / beta)

            # gradient w.r.t a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule for log‑softmax (mirror‑descent)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it_tot += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)
            m_hat = m / (1.0 - beta1 ** it_tot)
            v_hat = v / (1.0 - beta2 ** it_tot)
            lr = lr0 / (math.sqrt(it_tot) + 1.0)

            log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # exact surrogate (true max) for bookkeeping
            cur_a = _softmax(log_a)
            cur_val = _objective_surrogate(cur_a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_a.copy()

        if time.time() > deadline:
            break

    if best_seq is None:
        best_seq = _softmax(log_a)
        best_val = _objective_surrogate(best_seq)
    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS refinement (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000) -> np.ndarray:
    """Continuation L‑BFGS on the smoothed max (requires SciPy)."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1, 5, 20, 80, 200, 500, 1_000,
                         2_000, 5_000, 20_000, 50_000, 100_000]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:   # pragma: no cover
            break
    return _softmax(v)

    # ------------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # basic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))

    # anti‑symmetric families (new)
    for factor in np.linspace(0.0, 5.0, 12):
        seeds.append(_anti_linear(n, factor))
    for amp in np.linspace(0.0, 0.4, 12):
        seeds.append(_anti_sine(n, amp))

    # beta‑skewed families – a finer grid covering extreme values
    alphas = np.concatenate((np.linspace(0.01, 0.5, 7),
                             np.linspace(0.5, 5.0, 7),
                             np.linspace(5.0, 12.0, 5)))
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # exponential decays (include very shallow rates)
    rates = np.logspace(-5, -0.2, 30)
    for r in rates:
        seeds.append(_exp_decay(n, r))

    # power‑law families
    for p in np.linspace(0.0, 5.0, 15):
        seeds.append(_power(n, p))

    # mixtures of uniform and exponential
    for r in rates[::5]:
        exp_seq = _exp_decay(n, r)
        for a in (0.1, 0.3, 0.5):
            seeds.append(_mix_shapes(_uniform(n), exp_seq, a))

    # random Dirichlet draws (including very sparse vectors)
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        for _ in range(4):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # previous best (if any) – bring it in as a seed
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Normalise and evaluate all seeds (cheap surrogate)
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Optimise each top seed (short Adam runs)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # split remaining budget roughly equally among remaining seeds
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.12 / max(1, (top_k - idx))

        seq_adam, val_adam = _adam_optimize(seed_vec,
                                            n,
                                            sub_deadline,
                                            beta_schedule=[1, 5, 20, 80, 200, 500,
                                                           1_000, 2_000, 5_000,
                                                           20_000, 50_000, 100_000],
                                            max_iters_per_beta=1200,
                                            lr0=0.6)

        print(f"[Seed {idx+1}/{top_k}] Adam surrogate = {val_adam:.6f}")

        if val_adam < global_best_val:
            global_best_val = val_adam
            global_best_seq = seq_adam
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long‑run polishing (Adam + optional L‑BFGS)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long Adam polishing ---")
        final_seq, final_val = _adam_optimize(global_best_seq,
                                               n,
                                               deadline,
                                               beta_schedule=[1, 5, 20, 80, 200, 500,
                                                              1_000, 2_000, 5_000,
                                                              20_000, 50_000, 100_000,
                                                              200_000],
                                               max_iters_per_beta=2500,
                                               lr0=0.5)
        if final_val < global_best_val:
            global_best_val = final_val
            global_best_seq = final_seq
            print(f"[Final Adam] improved to {global_best_val:.6f}")

        # optional L‑BFGS refinement with a high β
        if time.time() < deadline:
            print("\n--- Optional L‑BFGS refinement ---")
            lbfgs_seq = _lbfgs_smooth_optimize(global_best_seq,
                                               n,
                                               deadline,
                                               beta_schedule=[1_000_000, 2_000_000, 5_000_000],
                                               max_iter_per_beta=800)
            lbfgs_val = _objective_surrogate(lbfgs_seq)
            if lbfgs_val < global_best_val:
                global_best_val = lbfgs_val
                global_best_seq = lbfgs_seq
                print(f"[L‑BFGS] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_2"] = global_best_seq.tolist()
    return global_best_seq.tolist()
