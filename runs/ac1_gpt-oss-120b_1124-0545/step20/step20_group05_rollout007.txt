def _anneal(seq, deadline, init_temp=0.1, final_temp=1e-5, steps_per_temp=5000):

   T = init_temp
   current = seq.copy()
   cur_val = _objective_surrogate(current)
   rng = np.random.default_rng()
   while time.time() < deadline and T > final_temp:
        for _ in range(steps_per_temp):
            if time.time() > deadline:
                break
            # propose move: choose i, j and delta
            i, j = rng.integers(0, len(seq), size=2)
            if i == j: continue
            delta = rng.exponential(scale=0.01)  # small
            delta = min(delta, current[i])
            new = current.copy()
            new[i] -= delta
            new[j] += delta
            new = new / new.sum()
            new_val = _objective_surrogate(new)
            if new_val < cur_val or rng.random() < np.exp((cur_val - new_val)/T):
                current, cur_val = new, new_val
        T *= 0.9  # cooling
   return current, cur_val

"""
Improved search for a non‑negative step‑function that minimises the
evaluation function ``evaluate_sequence`` (i.e. 2·n·max(conv) after normalisation).

The algorithm
--------------
1. **Broad seeding** – generate a large, diverse collection of candidates:
   * classic windows (uniform, triangular, Hamming, …)
   * extreme beta‑shapes with tiny parameters (strong edge‑mass)
   * edge‑power laws and an arcsine‑like shape
   * random Dirichlet / exponential / low‑frequency curves
   * a few mixtures / perturbations of the previous best.

2. **Surrogate ranking** – compute the cheap exact surrogate
   ``2·n·max(conv)`` for every seed, keep the best few.

3. **Heavy refinement** – for each top seed
   * mirror‑descent / exponentiated‑gradient on the exact max‑convolution
     (many iterations, time‑controlled)
   * a continuation L‑BFGS optimiser on a smooth log‑sum‑exp surrogate
     (if SciPy is available)
   * a long stochastic local‑search (hill‑climbing).

4. **Optional simulated‑annealing** – use any remaining time to explore
   further with a simple Metropolis scheme.

5. **Final evaluation** – the best vector is scored with the official
   ``evaluate_sequence`` and stored in the global variable
   ``height_sequence_1`` for reuse.

All steps respect the overall wall‑time budget (default 1000 s) and
never rely on file or network I/O.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Basic numeric utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ------------------------------------------------------------
# Surrogate objective (exact for l1‑normalised vectors)
# ------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector ℓ₁‑normalised to 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic families (all return ℓ₁‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped density on {1,…,n} (strictly positive for α,β>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _double_gaussian(n: int, sigma_frac: float = 0.15,
                    shift_frac: float = 0.25) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = shift_frac * (n - 1)
    mu2 = (1 - shift_frac) * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Extra families that proved useful in practice

def _edge_power(n: int, p: float) -> np.ndarray:
    """Weight ∝ 1/((i+1)^p + (n-i)^p) – strong mass at the two ends."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / ((i + 1) ** p + (n - i) ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density ∝ 1/√[x(1−x)]."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Random families
# ------------------------------------------------------------
def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _alternating_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    raw = rng.dirichlet(np.full(n, 0.3)).astype(np.float64)
    sorted_desc = np.sort(raw)[::-1]
    a = np.empty_like(sorted_desc)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_desc):
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add small Gaussian jitter, clip & renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s

# ------------------------------------------------------------
def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (both already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


# ------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimisation
# ------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution.
    Returns the best sequence seen and its surrogate value."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑softmax space to keep the simplex
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 3000):
    """Optimise the smooth max‑convolution objective with L‑BFGS‑B."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1000.0,
                         2000.0, 5000.0, 20000.0,
                         50000.0, 100000.0, 200000.0,
                         500000.0, 1000000.0]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient wrt a (softmax space)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)          # chain rule through softmax
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ------------------------------------------------------------
# Simple stochastic local search (hill‑climbing)
# ------------------------------------------------------------
def _local_search(seq: np.ndarray,
                  deadline: float,
                  moves: int = 20_000,
                  step_scale: float = 0.02):
    """Random mass‑transfer hill‑climbing."""
    rng = np.random.default_rng()
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    n = len(seq)

    for _ in range(moves):
        if time.time() > deadline:
            break
        eps = step_scale * rng.random()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = min(eps, best_seq[i])
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq = new_seq / new_seq.sum()
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq
    return best_seq, best_val


# ------------------------------------------------------------
# Simulated annealing (optional final polishing)
# ------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.1,
                         final_temp: float = 1e-5,
                         steps_per_temp: int = 2000):
    """Very simple Metropolis scheme with a geometric temperature schedule."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    n = len(cur)
    T = init_temp

    while time.time() < deadline and T > final_temp:
        for _ in range(steps_per_temp):
            if time.time() > deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # propose moving a modest amount of probability
            delta = 0.02 * rng.random()
            delta = min(delta, cur[i])
            if delta == 0.0:
                continue
            new = cur.copy()
            new[i] -= delta
            new[j] += delta
            new = new / new.sum()
            new_val = _objective_surrogate(new)
            # Metropolis acceptance
            if new_val < cur_val or rng.random() < math.exp((cur_val - new_val) / T):
                cur, cur_val = new, new_val
        T *= 0.9
    return cur, cur_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # Length of the vector – 4000 works well and matches the previous run.
    n = 4000

    # ------------------------------------------------------------------
    # 1.  Rich seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # classic deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    seeds.append(_alternating_dirichlet(n, rng))

    # extreme beta‑shapes (α,β down to 0.02)
    alphas = np.concatenate((np.geomspace(0.02, 0.5, 12),
                             np.linspace(0.5, 3.0, 9)))
    betas = alphas
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # edge‑power families – concentrate mass at both ends
    for p in np.linspace(0.8, 1.4, 9):
        seeds.append(_edge_power(n, p))

    # arcsine‑like shape
    seeds.append(_arcsine_shape(n))

    # sine window sweep
    for gamma in np.linspace(0.1, 5.0, 70):
        seeds.append(_sine_window(n, gamma))

    # Kaiser window sweep
    for beta_param in np.linspace(0.1, 30.0, 70):
        seeds.append(_kaiser_window(n, beta_param))

    # Gaussian sweep
    for sigma in np.linspace(0.04, 0.35, 40):
        seeds.append(_gaussian(n, sigma))

    # Power‑law sweep
    for p in np.linspace(0.0, 2.5, 60):
        seeds.append(_power(n, p))

    # Random Dirichlet (various concentrations)
    for conc in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(4):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Small perturbations of the previous best (if present)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape[0] == n and arr.sum() > 0:
                arr = arr / arr.sum()
                seeds.append(arr)
                for _ in range(4):
                    seeds.append(_perturb(arr, rng, sigma=0.003))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2.  Surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    print("\n--- Evaluating seeds (surrogate) ---")
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop surrogate values after seeding:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3.  Heavy refinement of the best seeds
    # ------------------------------------------------------------------
    best_overall_seq = top_seeds[0].copy()
    best_overall_val = _objective_surrogate(best_overall_seq)
    print(f"\nInitial best surrogate = {best_overall_val:.6f}")

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Refinement for seed #{idx+1} ---")
        # Exponentiated‑gradient – allocate up to 40 % of the remaining budget
        eg_deadline = time.time() + max(5.0, 0.4 * (deadline - time.time()))
        seq_eg, val_eg = _exp_gradient_opt(seed_seq.copy(),
                                           n,
                                           eg_deadline,
                                           max_iters=2_500_000,
                                           eta0=6.0)
        if val_eg < best_overall_val:
            best_overall_val, best_overall_seq = val_eg, seq_eg.copy()
            print(f"  EG improvement: {val_eg:.6f}")

        if time.time() > deadline:
            break

        # L‑BFGS smoothing – only on the current overall best
        lbfgs_deadline = deadline - 1.5
        seq_lb = _lbfgs_smooth_optimize(best_overall_seq,
                                         n,
                                         lbfgs_deadline,
                                         beta_schedule=[1, 5, 20, 80,
                                                        200, 500, 1000,
                                                        2000, 5000, 20000,
                                                        50000, 100000, 200000,
                                                        500000, 1000000],
                                         max_iter_per_beta=4000)
        val_lb = _objective_surrogate(seq_lb)
        if val_lb < best_overall_val:
            best_overall_val, best_overall_seq = val_lb, seq_lb.copy()
            print(f"  L‑BFGS improvement: {val_lb:.6f}")

        if time.time() > deadline:
            break

        # Local stochastic hill‑climbing
        local_deadline = deadline - 0.5
        seq_loc, val_loc = _local_search(best_overall_seq,
                                         local_deadline,
                                         moves=50_000,
                                         step_scale=0.02)
        if val_loc < best_overall_val:
            best_overall_val, best_overall_seq = val_loc, seq_loc.copy()
            print(f"  Local‑search improvement: {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final polishing with simulated annealing (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        ann_seq, ann_val = _simulated_annealing(best_overall_seq,
                                                deadline,
                                                init_temp=0.1,
                                                final_temp=1e-5,
                                                steps_per_temp=3000)
        if ann_val < best_overall_val:
            best_overall_val, best_overall_seq = ann_val, ann_seq.copy()
            print(f"  Simulated‑annealing improvement: {ann_val:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final official evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised", exc)
        true_val = None

    print(f"\n=== Finished ===\nBest surrogate ≈ {best_overall_val:.6f}"
          f", true evaluate_sequence = {true_val}")
    print(f"Length of the construction: {len(best_overall_seq)}")

    # Store for possible reuse in a later run
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
