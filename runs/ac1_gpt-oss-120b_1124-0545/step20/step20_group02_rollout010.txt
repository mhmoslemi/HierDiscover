"""
Propose a step function (height sequence) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The algorithm combines a cheap large‑scale random / analytic search with a
mirror‑descent (exponentiated‑gradient) polishing stage and a light simulated
annealing finish.  All work respects a 1000 s budget; the best sequence is
stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Basic utilities (FFT convolution, softmax,‐etc.)
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s if s > 0.0 else np.full_like(v, 1.0 / len(v))


# --------------------------------------------------------------
# Surrogate evaluation (exact max, ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a probability vector (sum = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# --------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - c))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# --------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# --------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)            # work in log space (automatically ℓ₁‑normalised)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # active set: indices whose convolution value is within 1e‑6 relative tolerance
        eps = max_val * 1e-6
        active = np.where(conv >= max_val - eps)[0]

        # accumulate sub‑gradient contributions from all active shifts
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            # true sub‑gradient of the max is the *average* of the active gradients;
            # scaling the step by the size of the active set yields a more stable update.
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)       # classic 1/√t decay
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# --------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# --------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.005,
                         T_start: float = 0.01,
                         T_end: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Light‑weight SA: Gaussian perturbations with a linearly decreasing temperature.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linearly cool temperature
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best sequence in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Fixed problem size
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load any previously found solution
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0.0:
                    prev_arr = prev_arr / s
                else:
                    prev_arr = None
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # --------------------------------------------------------------
    # Initial best candidate (either previous or a sensible analytic shape)
    # --------------------------------------------------------------
    if prev_arr is not None:
        best_seq = prev_arr.copy()
        best_val = _objective_surrogate(best_seq)
        print("[Info] Loaded previous best sequence.")
    else:
        best_seq = _sine_window(n, gamma=1.2)
        best_val = _objective_surrogate(best_seq)
        print("[Info] No previous data – starting from a sine window.")

    # --------------------------------------------------------------
    # Phase 1 – rapid exploration over analytic families & random draws
    # --------------------------------------------------------------
    def _try_candidate(seq: np.ndarray, label: str = ""):
        nonlocal best_seq, best_val, deadline
        if time.time() > deadline:
            return
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Phase 1] New best ({label}) : {best_val:.6f}")

    # a few deterministic families
    for gamma in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0):
        _try_candidate(_sine_window(n, gamma=gamma), f"sine(γ={gamma})")

    for sigma in (0.05, 0.10, 0.20, 0.30, 0.45):
        _try_candidate(_gaussian(n, sigma_frac=sigma), f"gauss(σ={sigma})")

    for rate in (0.001, 0.003, 0.006, 0.012, 0.025):
        _try_candidate(_exp_decay(n, rate=rate), f"exp(rate={rate})")
        _try_candidate(_sym_exp_decay(n, rate=rate), f"sym‑exp(rate={rate})")

    # beta‑shape grid (moderately sized)
    alphas = (0.5, 1.0, 2.0, 3.0, 5.0)
    betas = (0.5, 1.0, 2.0, 3.0, 5.0)
    for a in alphas:
        for b in betas:
            _try_candidate(_beta_shape(n, alpha=a, beta=b), f"beta(α={a},β={b})")

    # --------------------------------------------------------------
    # Phase 2 – random Dirichlet draws (fast, many)
    # --------------------------------------------------------------
    rnd_draws = 3000                      # adjustable; cheap thanks to FFT
    for i in range(rnd_draws):
        if time.time() > deadline:
            break
        conc = rng.uniform(0.4, 4.0)      # concentration controls “spikiness”
        cand = _random_dirichlet(n, conc=conc, rng=rng)
        _try_candidate(cand, f"dirichlet({conc:.2f})")

    # --------------------------------------------------------------
    # Phase 3 – short EG polishing on the current best (quick improvement)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Short EG polishing (Phase 3) ---")
        seq_eg, val_eg = _exp_gradient_opt_multi(best_seq,
                                                n,
                                                deadline,
                                                max_iters=200_000,
                                                eta0=6.0)
        if val_eg < best_val:
            best_seq = seq_eg
            best_val = val_eg
            print(f"[Phase 3] EG improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – heavy EG polishing (more iterations, up to remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        # allocate the bulk of the remaining time to this stage
        remaining = deadline - time.time()
        # estimate safe iteration count (≈ 0.0002 s per iteration on typical hardware)
        safe_iters = int(min(2_000_000, remaining / 0.00025))
        print("\n--- Heavy EG polishing (Phase 4) ---")
        print(f"    → allocating up to {safe_iters} iterations")
        seq_eg, val_eg = _exp_gradient_opt_multi(best_seq,
                                                n,
                                                deadline,
                                                max_iters=safe_iters,
                                                eta0=5.5)
        if val_eg < best_val:
            best_seq = seq_eg
            best_val = val_eg
            print(f"[Phase 4] EG improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – final simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final Simulated‑Annealing (Phase 5) ---")
        seq_sa, val_sa = _simulated_annealing(best_seq,
                                             n,
                                             deadline,
                                             rng,
                                             max_attempts=180_000,
                                             start_sigma=0.001,
                                             T_start=4e-3,
                                             T_end=1e-9)
        if val_sa < best_val:
            best_seq = seq_sa
            best_val = val_sa
            print(f"[Phase 5] SA improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Official evaluation via the provided evaluate_sequence()
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[Error] evaluate_sequence raised an exception:", exc)

    # Store for future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
