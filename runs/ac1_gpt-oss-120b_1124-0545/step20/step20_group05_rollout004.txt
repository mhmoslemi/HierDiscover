def _lp_step(f, M, deadline):
    from scipy.optimize import linprog
    n = len(f)
    # Build sparse matrix A for constraints f * g <= M
    rows = np.empty(n*n, dtype=np.int32)
    cols = np.empty(n*n, dtype=np.int32)
    data = np.empty(n*n, dtype=np.float64)
    for i in range(n):
        start = i*n
        rows[start:start+n] = i + np.arange(n)
        cols[start:start+n] = np.arange(n)
        data[start:start+n] = f[i]
    A = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'time_limit': max(0, deadline - time.time())})
    if not res.success:
        print("LP failed", res.message)
        return None
    g = res.x
    return g

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of heights) that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq, seq)) / (sum(seq)²)

The algorithm proceeds in three phases:

1. **Seeding & exponentiated‑gradient (EG) refinement** – a modest
   collection of analytic families (uniform, triangular, sine‑window,
   Kaiser, Gaussian, power‑law, beta‑shapes, low‑frequency random curves)
   is generated, normalised to ℓ₁ = 1 and the cheap surrogate
   ``2·n·max(conv)`` is evaluated.  The best seed is further refined by a
   long EG run (mirror‑descent in log‑softmax space) that directly uses
   the exact maximal convolution entry.

2. **Linear‑programming “mass‑transfer’’ step** – with the current best
   vector `f` we compute `M = max(conv(f,f))`.  We then solve the LP

        maximise   Σ g_j
        subject to (f * g)_k ≤ M   for all k
                   g_j ≥ 0

   The LP has `n` variables and `2·n‑1` linear constraints and is solved
   with SciPy’s HiGHS interior‑point method (sparse matrix).  The optimal
   vector `g` typically has a larger ℓ₁‑norm than `f`.

3. **Convex mixing & final polishing** – a ternary‑search over the mixing
   weight α ∈ [0,1] is performed on the convex combination  
   `h(α) = α·f + (1‑α)·g_norm` (where `g_norm = g / Σg`).  The mixture is
   re‑normalised to sum = 1 after each trial, and the surrogate is
   evaluated.  The best mixture replaces `f`.  Afterwards a short EG run
   and a very cheap stochastic local search are applied.

The whole pipeline respects the global time‑budget (default 1000 s).  The
best sequence found is stored in the global variable `height_sequence_1`
and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for the LP step)
# ----------------------------------------------------------------------
try:
    import scipy.sparse as _sp
    import scipy.optimize as _opt
    _has_scipy = True
except Exception:                         # pragma: no cover
    _has_scipy = False

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(seq, seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Softmax utilities (used by EG)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)          # Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Mixing helpers
# ----------------------------------------------------------------------
def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (both already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s


def _ternary_mix(a: np.ndarray, b: np.ndarray, deadline: float, steps: int = 30):
    """Best convex combination of a and b by ternary search on α∈[0,1]."""
    lo, hi = 0.0, 1.0
    best_seq, best_val = None, float('inf')
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)
        if val1 < val2:
            hi = m2
            if val1 < best_val:
                best_val, best_seq = val1, seq1.copy()
        else:
            lo = m1
            if val2 < best_val:
                best_val, best_seq = val2, seq2.copy()
    mid = (lo + hi) / 2.0
    seq_mid = _mix_shapes(a, b, mid)
    val_mid = _objective_surrogate(seq_mid)
    if val_mid < best_val:
        best_seq, best_val = seq_mid, val_mid
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     eta0: float = 5.0):
    """Mirror‑descent on exact max‑convolution. Returns best sequence seen."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap stochastic local search
# ----------------------------------------------------------------------
def _local_search(seq: np.ndarray,
                  deadline: float,
                  moves: int = 20000,
                  step_scale: float = 0.02):
    """
    Random mass‑transfer hill‑climbing.
    A small amount of probability is moved from one index to another.
    """
    rng = np.random.default_rng()
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    n = len(seq)

    for _ in range(moves):
        if time.time() > deadline:
            break
        eps = step_scale * rng.random()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = min(eps, best_seq[i])
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq = new_seq / new_seq.sum()
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq
    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming step (maximise ℓ₁‑norm under convolution bound)
# ----------------------------------------------------------------------
def _lp_max_sum(f: np.ndarray, M: float, deadline: float):
    """
    Solve
        maximise Σ g_j
        s.t.   (f * g)_k ≤ M    ∀k∈[0,2n‑2]
               g_j ≥ 0
    Returns the optimal g (not normalised) or None on failure.
    Requires SciPy.
    """
    if not _has_scipy:                     # pragma: no cover
        return None

    n = f.shape[0]

    # static index arrays (shared across calls)
    if not hasattr(_lp_max_sum, "row_idx") or _lp_max_sum.n != n:
        i_arr = np.repeat(np.arange(n, dtype=np.int32), n)
        j_arr = np.tile(np.arange(n, dtype=np.int32), n)
        _lp_max_sum.row_idx = i_arr + j_arr                # shape (n*n,)
        _lp_max_sum.col_idx = j_arr
        _lp_max_sum.n = n

    data = np.repeat(f.astype(np.float64), n)               # length n*n
    A = _sp.csr_matrix((data,
                        (_lp_max_sum.row_idx, _lp_max_sum.col_idx)),
                       shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                       # maximise sum → minimise -sum
    bounds = [(0, None)] * n

    # HiGHS interior‑point method (default for method='highs')
    try:
        res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                           method='highs',
                           options={'presolve': True})
    except Exception as exc:                               # pragma: no cover
        print("LP exception:", exc)
        return None

    if not res.success:
        print("LP failed:", res.message)
        return None
    return res.x
