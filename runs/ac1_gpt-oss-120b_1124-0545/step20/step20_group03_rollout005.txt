def _smooth_obj_and_grad(v, n, beta):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = np.max(conv)   # shift for numerical stability
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1 : n-1 + n]
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v

"""
propose_candidate – multi‑restart Adam + exact sub‑gradient polishing

The algorithm works on the ℓ₁‑normalised simplex (∑ aᵢ = 1) because the
objective 2·n·max(conv(a,a)) is invariant under scaling.  Each restart
generates a distinct starting distribution (Dirichlet or exponential decay)
and runs a continuation scheme on the smooth approximation
    f_β(a) = 2·n·( max(conv) + (1/β)·log Σₖ e^{β·(convₖ−max)} )
using Adam.  The β‑schedule goes from very smooth (β≈1) to almost the true
max (β≈10⁶).  After the smoothing phase a short exponentiated‑gradient (mirror–
descent) phase on the exact max‑convolution refines the solution.  The best
sequence (according to the original `evaluate_sequence`) is stored in the
global variable ``height_sequence_1`` for the next run.

The whole search respects the 1000 s budget; the number of restarts and the
iteration limits are adjusted dynamically based on the remaining time.
"""

import math
import time
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for a fallback L‑BFGS optimiser.
# ----------------------------------------------------------------------
try:                     # pragma: no cover
    import scipy.optimize as _opt
except Exception:       # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Low‑level FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Soft‑max (probability simplex) utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Simple analytic starters
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """aᵢ ∝ exp(-rate·i), i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Exact surrogate (for ∑ aᵢ = 1): 2·n·max(conv(a,a))
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth max‑convolution objective + gradient (log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (obj, grad_v) where
        obj = 2·n·(shift + log Σ exp(β·(conv−shift))/β)
    with shift = max(conv) for numerical stability,
    and grad_v is the gradient w.r.t. the logit vector v.
    """
    a = _softmax(v)                       # probability simplex, sum = 1
    conv = _conv_fft(a, a)                # length = 2·n‑1
    shift = np.max(conv)                  # stability shift
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # Gradient w.r.t. a of the smoothed max:
    w = exp_beta / sum_exp                # weights (sum to 1)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length = 3·n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]  # slice of length n

    # Convert gradient to logit space (soft‑max Jacobian)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the smooth objective (continuation in β)
# ----------------------------------------------------------------------
def _adam_optimize(v0: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule,
                   steps_per_beta: int = 2000,
                   lr0: float = 0.3) -> np.ndarray:
    """
    Adam optimiser that walks through a list of β values.
    Returns the final logit vector.
    """
    v = v0.copy()
    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # moments are re‑initialised for each β – this works well in practice
        m = np.zeros_like(v)
        v_m = np.zeros_like(v)

        # step‑size scaled roughly as 1/√β (larger β → smaller steps)
        lr = lr0 / (np.sqrt(beta) + 1e-12)

        for it in range(1, steps_per_beta + 1):
            if time.time() > deadline:
                return v
            obj, grad = _smooth_obj_and_grad(v, n, beta)

            # Adam updates
            m = beta1 * m + (1.0 - beta1) * grad
            v_m = beta2 * v_m + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v_m / (1.0 - beta2 ** it)

            v = v - lr * m_hat / (np.sqrt(v_hat) + eps)

    return v


# ----------------------------------------------------------------------
# Exact max‑convolution polishing (mirror‑descent on the true max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     eta0: float = 5.0):
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (all contributing indices)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point: search for the best step‑function
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Strategy:
      1. Choose a problem size n (≈5 000) and several random starts.
      2. For each start run a continuation Adam optimiser on a smooth max‑convolution.
      3. Polish the result with exact sub‑gradient mirror‑descent.
      4. Keep the best sequence (according to the true evaluation) and store it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a modest safety margin
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem dimension – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    n = 5000
    print(f"[Info] problem size n = {n}")

    # --------------------------------------------------------------
    # β‑schedule for the smooth approximation (log‑sum‑exp)
    # --------------------------------------------------------------
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0, 1_000_000.0
    ]

    # --------------------------------------------------------------
    # Global bookkeeping
    # --------------------------------------------------------------
    best_seq = None
    best_true = float('inf')
    best_surr = float('inf')

    # --------------------------------------------------------------
    # Number of independent restarts – limited by remaining time.
    # --------------------------------------------------------------
    max_restarts = 6
    for restart in range(max_restarts):
        if time.time() > deadline:
            break

        # ---------- initialise a fresh probability vector ----------
        if restart == 0 and "height_sequence_1" in globals():
            # reuse the previous best (interpolate if length differs)
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.size != n:
                # simple linear interpolation to the new size
                x_old = np.linspace(0, 1, prev.size)
                x_new = np.linspace(0, 1, n)
                seq0 = np.interp(x_new, x_old, prev)
            else:
                seq0 = prev.copy()
        else:
            # alternate between a Dirichlet draw and a simple exponential decay
            if restart % 2 == 0:
                # Dirichlet with concentration <1 → a diverse shape
                seq0 = rng.dirichlet(alpha=np.full(n, 0.8))
            else:
                # exponential decay (rate chosen randomly in a small interval)
                rate = rng.uniform(0.0025, 0.0040)
                seq0 = _exp_decay(n, rate=rate)

        # Normalise to the simplex (sum = 1) and protect against round‑off.
        seq0 = np.maximum(seq0, 0.0)
        total = seq0.sum()
        if total == 0.0:
            seq0 = _uniform(n)
        else:
            seq0 = seq0 / total

        # Convert to logit space, add a touch of noise to escape symmetries.
        v0 = np.log(seq0 + 1e-30)
        v0 += rng.normal(scale=0.2, size=n)

        # ---------- Adam‑based continuation on the smooth objective ----------
        print(f"[Restart {restart}] start Adam optimisation")
        v_opt = _adam_optimize(v0, n, deadline,
                               beta_schedule=beta_schedule,
                               steps_per_beta=2000,
                               lr0=0.3)
        seq_opt = _softmax(v_opt)
        surr_val = _objective_surrogate(seq_opt)
        print(f"[Restart {restart}] surrogate after Adam = {surr_val:.6f}")

        # ---------- Exact‑max polishing (exponentiated‑gradient) ----------
        # Allocate a modest slice of the remaining budget for polishing.
        remaining = deadline - time.time()
        if remaining > 30.0:
            # Rough estimate: ~2000 iterations per second (depends on hardware)
            max_iters = int(min(2_000_000, remaining * 1500))
            seq_polished, surr_polished = _exp_gradient_opt(seq_opt, n,
                                                            deadline,
                                                            max_iters=max_iters,
                                                            eta0=5.0)
            print(f"[Restart {restart}] surrogate after EG polish = {surr_polished:.6f}")
        else:
            seq_polished = seq_opt
            surr_polished = surr_val

        # ---------- True evaluation ----------
        try:
            true_val = evaluate_sequence(seq_polished.tolist())
        except Exception as exc:          # pragma: no cover
            print(f"[Restart {restart}] evaluate_sequence raised {exc}")
            true_val = float('inf')
        print(f"[Restart {restart}] true evaluate_sequence = {true_val:.6f}")

        # ---------- Keep the best overall ----------
        if true_val < best_true:
            best_true = true_val
            best_seq = seq_polished.copy()
            best_surr = surr_polished

        # Optional early exit if we have already reached the target bound.
        if best_true <= 1.503:
            print("[Info] target bound reached – terminating early")
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but guarantees a valid return.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _exp_decay(n, rate=0.003)

    # Store for the next round (enables incremental improvement).
    globals()["height_sequence_1"] = best_seq.tolist()
    print(f"[Result] best true = {best_true:.6f}, best surrogate = {best_surr:.6f}")
    return best_seq.tolist()
