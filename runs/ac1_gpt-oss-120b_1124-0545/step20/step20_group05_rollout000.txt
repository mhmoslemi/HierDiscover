def propose_candidate(seed=123,budget_s=1000,**kwargs):

    # Add beta shape finer seed: sample random alphas,beta
    for _ in range(200):
        al = rng.uniform(0.1,8.0)

        seeds.append(_beta_shape(n, al, be))

    # random Dirichlet with many conc values
    for conc in [0.005,0.015,0.025,0.05,0.1,0.3,0.7,1.5]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random lowfreq combos
    for nf in range(2,8):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # previous best
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0:
                arr = arr/arr.sum()
                seeds.append(arr)
                for _ in range(3):
                    seeds.append(_perturb(arr, rng, sigma=0.004))
        except Exception:
            pass

    # Evaluate surrogates quickly
    evaluated = [(vec,_objective_surrogate(vec)) for vec in seeds if time.time() < deadline]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec,_ in evaluated[:top_k]]
    print("Top seeds after surrogate:")
    for i,(vec,val) in enumerate(evaluated[:top_k]):
        print(f"#{i+1} {val:.6f}")

    # For each top seed, run ADAM
    best_seq = None
    best_val = float('inf')
    for idx, seed_seq in enumerate(top_seeds):
        if time.time() > deadline:
            break
        # ensure symmetry
        seed_seq = (seed_seq + seed_seq[::-1]) / 2
        # reference start time

        seq_opt, val_opt = _adam_optimize(seed_seq, n, seed_deadline)
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that
minimises ``evaluate_sequence``.

The algorithm consists of three phases:

1. **Broad seeding** – generate a few thousand candidate vectors from many
   analytic families (Gaussian, beta‑shaped, Kaiser, random Dirichlet, low‑freq
   cosine curves, …).  All candidates are ℓ₁‑normalised, symmetric and their
   cheap surrogate objective ``2·n·max(conv)`` is evaluated.  The best few
   seeds are kept.

2. **ADAM optimisation** – starting from each top seed we run a custom
   Adam optimiser in the log‑softmax space.  The objective is a smooth
   approximation of the true max: ``f_β(a)=2·n·log∑exp(β·conv(a))/β``.
   β is increased geometrically during optimisation, so the algorithm first
   follows a smooth landscape and then focuses on the true maximum.  The
   optimisation respects the simplex constraint and a symmetry constraint
   (a[i]=a[n‑1‑i]) by symmetrising the log‑weights each iteration.

3. **Local fine‑tuning** – a cheap random mass‑transfer hill‑climber is run
   for a few seconds on the best vector found by Adam.

The best vector is returned (and stored in the global ``height_sequence_1``
for possible reuse).  The whole search fits into the 1000 s budget and
normally improves the bound below the target 1.503.

"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector ℓ₁‑normalised to 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised, symmetric vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)               # modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Simple stochastic local search (mass‑transfer)
# ----------------------------------------------------------------------
def _local_search(seq: np.ndarray,
                  deadline: float,
                  moves: int = 20000,
                  step_scale: float = 0.02) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climbing."""
    rng = np.random.default_rng()
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    n = len(seq)

    for _ in range(moves):
        if time.time() > deadline:
            break
        eps = step_scale * rng.random()
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = min(eps, best_seq[i])
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq = new_seq / new_seq.sum()
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq
    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimisation of a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   init_lr: float = 0.12,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Optimises a smooth surrogate
        f_β(a) = 2·n·log(∑exp(β·conv(a,a))) / β
    using Adam in the log‑softmax space.  The vector a is kept symmetric
    (a[i] = a[n-1-i]) by symmetrising the log‑weights each iteration.
    """
    # initialise log‑weights
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    t = 0
    beta = 10.0               # start with a smooth objective
    next_beta_inc = 4000      # increase β every this many iterations

    while time.time() < deadline:
        t += 1

        # Adam step – gradients are computed w.r.t. v
        a = _softmax(v)                     # still symmetric because v is symmetric
        # ------------------------------------------------------------
        # compute smoothed gradient
        # ------------------------------------------------------------
        conv = _conv_fft(a, a)               # length 2n‑1
        maxc = np.max(conv)
        shifted = conv - maxc                # for numerical stability
        exp_beta = np.exp(beta * shifted)
        w = exp_beta / exp_beta.sum()        # softmax weights over convolution indices

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # shape (n,)

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)          # ∂f/∂v (chain rule through softmax)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v2 / (1.0 - beta2 ** t)

        # update
        v = v - init_lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce symmetry of log‑weights (keeps a symmetric)
        v = (v + v[::-1]) * 0.5

        # occasionally evaluate the *exact* surrogate objective
        if t % 500 == 0:
            cur_val = _objective_surrogate(_softmax(v))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(v).copy()

        # schedule for β (increase to focus on true max)
        if t % next_beta_inc == 0 and beta < 1e6:
            beta *= 2.0
            # reduce learning rate a bit after β grows
            init_lr *= 0.7

    # final evaluation with the latest v
    final_a = _softmax(v)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_val, best_seq = final_val, final_a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for final evaluation & prints
    deadline = start_time + budget_s - 2.5

    n = 4000                                 # length of the step function

    # --------------------------------------------------------------
    # 1️⃣  Broad seeding
    # --------------------------------------------------------------
    print(f"\n--- Generating seeds (n={n}) ---")
    seeds = []

    # deterministic families (already symmetric)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_gaussian(n, 0.10))
    seeds.append(_gaussian(n, 0.15))
    seeds.append(_beta_shape(n, 1.0, 1.0))    # simple symmetric beta
    seeds.append(_kaiser_window(n, 5.0))
    seeds.append(_kaiser_window(n, 12.0))

    # random Dirichlet (various concentrations)
    for conc in (0.01, 0.03, 0.07, 0.15, 0.3, 0.7, 1.2):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponentials
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency cosine curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # systematic beta‑shape sweep (coarse – for diversity)
    for a_ in np.linspace(0.2, 4.0, 9):
        for b_ in np.linspace(0.2, 4.0, 9):
            seeds.append(_beta_shape(n, a_, b_))

    # a few random beta‑shape draws (continuous)
    for _ in range(120):
        al = rng.uniform(0.05, 8.0)
        be = rng.uniform(0.05, 8.0)
        seeds.append(_beta_shape(n, al, be))

    # previous best (if any) + perturbations
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and np.isfinite(arr).all():
                arr = np.maximum(arr, 0.0)
                arr = arr / arr.sum()
                seeds.append(arr)
                for _ in range(5):
                    seeds.append(_perturb(arr, rng, sigma=0.004))
        except Exception:
            pass

    # --------------------------------------------------------------
    # 2️⃣  Evaluate surrogate for seeds (cheap)
    # --------------------------------------------------------------
    print("\n--- Evaluating seeds (surrogate) ---")
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:

        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop surrogate values after seeding:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Adam optimisation from the top seeds
    # --------------------------------------------------------------
    best_overall_seq = top_seeds[0].copy()
    best_overall_val = _objective_surrogate(best_overall_seq)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Adam optimisation for seed #{idx+1} ---")
        # enforce symmetry before optimisation
        seed_seq = (seed_seq + seed_seq[::-1]) * 0.5
        seed_seq = seed_seq / seed_seq.sum()

        seq_opt, val_opt = _adam_optimize(seed_seq, n, deadline)
        print(f"   Adam surrogate result: {val_opt:.6f}")
        if val_opt < best_overall_val:
            best_overall_val, best_overall_seq = val_opt, seq_opt.copy()
            print(f"   ** New best after Adam: {val_opt:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final local stochastic search
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final local search ---")
        seq_loc, val_loc = _local_search(best_overall_seq,
                                         deadline,
                                         moves=25000,
                                         step_scale=0.015)
        if val_loc < best_overall_val:
            best_overall_val, best_overall_seq = val_loc, seq_loc.copy()
            print(f"   Local‑search improvement: {val_loc:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  True evaluation and output
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised", exc)

    print(f"\n=== Finished ===\nBest surrogate ≈ {best_overall_val:.6f}"
          f", true evaluate_sequence = {true_val}\nLength = {len(best_overall_seq)}")

    # store globally for possible reuse in a later run
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
