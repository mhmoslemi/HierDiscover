        conv_new[i:i+n] -= 2.0 * delta * a
        conv_new[2*i] += delta * delta
        conv_new[2*j] += delta * delta
        new_max = conv_new.max()
        new_val = 2.0 * n * new_max
        if new_val + tol < best_val:
            # accept
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            best_val = new_val
            best_seq = a.copy()
        moves += 1
    return best_seq, best_val

"""
Propose a candidate step‑function that (hopefully) beats the current best
upper bound (~1.5085).  The optimisation consists of three stages:

1. **Seeding** – start from the previous best (if any) together with a few
   random Dirichlet draws.
2. **Smooth Adam optimisation** – minimise a smooth approximation of
   ``max(conv(a,a))`` in the log‑softmax (probability) parametrisation.
   The smoothness is controlled by a schedule of β values (1 → 1e6) and
   Adam is used to accelerate convergence.  The best vector seen during
   the run is retained.
3. **Local mass‑transfer hill‑climbing** – starting from the best vector,
   repeatedly move a small amount of probability mass from a random source
   index to a random target index.  The convolution is updated exactly by a
   cheap O(n) slice operation; the move is kept only if it reduces the
   true objective ``2·n·max(conv)``.  This refines the solution further.

All stages respect the overall 1000 s budget; the final sequence is stored
in the global variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a probability vector:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution objective
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule,
                   max_iters_per_beta: int = 10000):
    """
    Optimise the log‑sum‑exp smoothed max‑convolution using Adam.
    Variables live in the log‑softmax space (v → a = softmax(v)).
    """
    v = np.log(start_vec + 1e-30)   # initialise log‑softmax variables
    m = np.zeros_like(v)           # first moment estimate
    v2 = np.zeros_like(v)          # second moment estimate
    t = 0                           # iteration counter
    best_seq = None
    best_val = np.inf

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # run a fixed number of inner Adam steps (or stop early for time)
        for _ in range(max_iters_per_beta):
            if time.time() > deadline:
                break

            a = _softmax(v)                     # current probability vector
            conv = _conv_fft(a, a)              # a*a
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp               # weights for each conv entry

            # gradient wrt a (exact for the smoothed objective)
            rev_a = a[::-1]
            grad_a_full = _conv_fft(rev_a, w)   # length 3n‑2
            grad_a = 2.0 * grad_a_full[n - 1:n - 1 + n]

            # chain rule through the softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            t += 1
            m = 0.9 * m + 0.1 * grad_v
            v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)
            m_hat = m / (1.0 - 0.9 ** t)
            v_hat = v2 / (1.0 - 0.999 ** t)

            # learning‑rate schedule decreasing with sqrt(beta)
            lr = 0.2 / (np.sqrt(beta) + 1e-12)
            v = v - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            cur_val = 2.0 * n * max_c
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    if best_seq is None:
        best_seq = _softmax(v)
    return best_seq, best_val


# ----------------------------------------------------------------------
# Local hill‑climbing by moving a tiny amount of mass between two indices
# ----------------------------------------------------------------------
def _local_move_optimize(a: np.ndarray,
                         n: int,
                         deadline: float,
                         max_moves: int = 2_000_000):
    """
    Simple stochastic hill‑climbing: repeatedly move a small amount δ of
    probability mass from a source index i to a target index j.  The
    convolution is updated exactly by slice operations
    (O(n) per trial).  Moves that lower the true surrogate objective
    2·n·max(conv) are kept.
    """
    a = a.copy()
    conv = _conv_fft(a, a)                     # exact convolution of the start
    best_val = 2.0 * n * np.max(conv)
    best_seq = a.copy()
    rng = np.random.default_rng()

    # pre‑compute the length of the full convolution
    full_len = 2 * n - 1
    tol = 1e-12
    moves = 0

    while time.time() < deadline and moves < max_moves:
        # pick a source index with non‑negligible mass
        i = rng.integers(n)
        if a[i] < 1e-14:
            continue

        # pick a distinct target index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # fraction of the source mass to move (up to ~30 %)
        max_delta = a[i]
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random() * 0.3
        if delta <= 0.0:
            continue

        # --- exact update of the convolution ---------------------------------
        # copy only for the trial (size ≈ 8 KB → cheap)
        conv_new = conv.copy()

        # the shift vectors a_{k‑j} and a_{k‑i} are simply padded copies of a
        conv_new[j:j + n] += 2.0 * delta * a
        conv_new[i:i + n] -= 2.0 * delta * a

        # the δ² self‑product terms
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # accept only improving moves
        if new_val + tol < best_val:
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            best_val = new_val
            best_seq = a.copy()
        moves += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point

    previous = globals().get("height_sequence_1")

            n = prev_arr.size
