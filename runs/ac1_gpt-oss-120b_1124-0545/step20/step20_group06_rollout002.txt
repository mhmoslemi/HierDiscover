"""
Propose a step‑function that minimizes

    2 * n * max(conv(a,a)) / (sum(a)**2)

using a multi‑stage pipeline:

1. Generate many deterministic and random seeds (uniform, beta‑skewed,
   exponential, power‑law, mixtures, Dirichlet draws, perturbations of the
   previous best).

2. Keep the best few seeds according to a cheap ℓ₁‑normalised surrogate
   (the same objective with sum(a)=1, i.e. 2·n·max(conv)).

3. Refine each top seed with an exponentiated‑gradient mirror‑descent on the
   exact max‑convolution (full active set → single active → more iterations).

4. Perform a continuation L‑BFGS optimisation on a smoothed max (log‑sum‑exp)
   with an aggressive β‑schedule that finally reaches β≈1e12, driving the
   smooth approximation to the true max.

5. A short stochastic hill‑climb (Gaussian perturbations) finishes the search.

All steps respect the global 1000 s budget (a 2 s safety margin).  The best
sequence found is stored in the global variable ``height_sequence_1`` and also
returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:

except Exception:  # pragma: no cover

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with sum = 1:
       2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i) , i = 0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p  (p≥0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 5.0,
                      single_active: bool = False):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space; the soft‑max guarantees positivity & ℓ₁‑norm
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # ----- active set -------------------------------------------------
        if single_active:
            active = np.array([int(conv.argmax())], dtype=np.int64)
        else:
            eps = max_val * 1e-12
            active = np.where(conv >= max_val - eps)[0]

        # ----- sub‑gradient accumulator ----------------------------------
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        # ----- learning‑rate (1/√t decay) -------------------------------
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 1500):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing feasibility.
    """
    if _opt is None:
        return start_vec  # SciPy unavailable – return unchanged

    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
            1_000.0, 2_000.0, 5_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0, 5_000_000.0,
            1e7, 5e7, 1e8, 5e8, 1e9, 5e9, 1e10
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()                     # stability shift
            # compute smooth max = shift + log( sum exp(beta*(conv - shift)) ) / beta
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t. a (smooth max)
            w = exp_beta / sum_exp                 # weights over each convolution index
            rev_a = a[::-1]
            # full convolution rev_a * w → length = 3n‑2, slice central part
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]   # slice back to length n

            # chain rule for log‑softmax (mirror‑descent)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        # callback that aborts if we run out of time
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Stochastic hill‑climb (Gaussian perturbations) on the surrogate
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_attempts: int = 250_000,
                sigma_start: float = 0.003,
                sigma_end: float = 5e-5):
    """
    Simple hill‑climb: Gaussian perturbations with a linear decay of the
    perturbation magnitude. Accept only improvements (no uphill moves).
    Returns the best sequence and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linearly decay sigma
        sigma = sigma_start + (sigma_end - sigma_start) * (attempts / max_attempts)
        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s
        val = _objective_surrogate(ns)
        if val < best_val:
            best_val = val
            best_seq = ns.copy()
            a = ns
        attempts += 1
    return best_seq, best_val

        sub_deadline = time.time() + remaining_time * 0.15 / max(1, (top_k - si))

        seq_eg, val_eg = _exp_gradient_opt(seed_vec.copy(),
                                           n,
                                           sub_deadline,
                                           max_iters=120_000,
                                           eta0=8.0,
                                           single_active=False)

        # (b) Switch to single‑active for a sharper descent
        if time.time() < sub_deadline:
            seq_eg2, val_eg2 = _exp_gradient_opt(seq_eg.copy(),
                                                n,
                                                sub_deadline,
                                                max_iters=80_000,
                                                eta0=6.0,
                                                single_active=True)
        else:
            seq_eg2, val_eg2 = seq_eg, val_eg

        # (c) L‑BFGS smoothing (moderate β‑schedule)
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg2,
                                               n,
                                               sub_deadline,
                                               beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                              200.0, 500.0, 1_000.0,
                                                              2_000.0, 5_000.0,
                                                              20_000.0, 50_000.0,
                                                              200_000.0, 500_000.0],
                                               max_iter_per_beta=1000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_eg2, val_eg2

        # ------------------------------------------------------------------
        # Pick local best among the refinements
        # ------------------------------------------------------------------
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_eg2, val_eg2),
            (seq_lbfgs, val_lbfgs)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing – long EG (single active) + aggressive L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (single‑active) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 n,
                                                 deadline,
                                                 max_iters=800_000,
                                                 eta0=6.0,
                                                 single_active=True)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final high‑β L‑BFGS polishing ---")
        final_seq = _lbfgs_smooth_optimize(global_best_seq,
                                           n,
                                           deadline,
                                           beta_schedule=[1.0, 5.0, 20.0, 80.0,
                                                          200.0, 500.0, 1_000.0,
                                                          2_000.0, 5_000.0,
                                                          20_000.0, 50_000.0,
                                                          200_000.0, 500_000.0,
                                                          1_000_000.0, 5_000_000.0,
                                                          1e7, 5e7, 1e8, 5e8,
                                                          1e9, 5e9, 1e10],
                                           max_iter_per_beta=1200)
        final_val = _objective_surrogate(final_seq)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final L‑BFGS] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Very short hill‑climb for any last tweaks
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final hill‑climb polishing ---")
        final_seq, final_val = _hill_climb(global_best_seq,
                                           n,
                                           deadline,
                                           rng,
                                           max_attempts=120_000,
                                           sigma_start=0.002,
                                           sigma_end=2e-5)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final hill‑climb] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
