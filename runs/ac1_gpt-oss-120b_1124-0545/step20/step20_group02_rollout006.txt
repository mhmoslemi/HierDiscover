"""
Propose a 4000‑dimensional non‑negative step‑function that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The search consists of three complementary phases:

1. **Smooth‑max L‑BFGS refinement** – a sequence of increasingly sharp
   log‑sum‑exp surrogates is minimised in the log‑softmax (probability‑simplex)
   parametrisation.  This quickly drives the max‑convolution towards a local
   optimum.

2. **Exact sub‑gradient mirror descent (exponentiated‑gradient)** – a high‑precision
   EG pass (up to a few million iterations) reduces the surrogate further.

3. **Linear‑programming “mass‑inflation” step** – from the current iterate we solve

       max   Σ g_i
       s.t.  (f * g)_k ≤ M   for all k
             g_i ≥ 0

   where `M = max(conv(f,f))`.  The optimal `g` can carry *more* ℓ₁‑mass than `f`.
   After normalising `g` we mix it with the current vector (including the pure
   normalised `g`) and keep the best mixture.  This step often yields a
   substantial drop of the max‑convolution because the convolution bound
   `M` is divided by the attained mass.

Finally a short simulated‑annealing polishing is performed and the best
sequence is stored in the global variable ``height_sequence_1`` for reuse.
The whole routine respects a time budget (default 1000 s) and prints
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional scientific imports
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
except Exception:          # pragma: no cover
    _opt = None
    _sp = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = v.max()
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) assuming Σ seq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families – useful for initial guesses
# ----------------------------------------------------------------------
def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Smooth surrogate for L‑BFGS (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------
class _SmoothMax:
    """
    Callable surrogate: 2·n·(shift + log( Σ exp(β·(conv−shift)) ) / β)
    The gradient is supplied in the log‑softmax (v) space.
    """
    __slots__ = ("n", "beta")

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, v: np.ndarray) -> tuple[float, np.ndarray]:
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_c = conv.max()
        shift = max_c
        exp_beta = np.exp(self.beta * (conv - shift))
        sum_exp = exp_beta.sum()
        obj = 2.0 * self.n * (shift + np.log(sum_exp) / self.beta)

        # gradient w.r.t. a (smooth max derivative)
        w = exp_beta / sum_exp                         # shape (2n‑1,)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length = 3n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # chain rule to v‑space (log‑softmax)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return obj, grad_v


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 3_000_000,
                            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) directly on the max‑convolution.
    Returns the best sequence encountered and its exact surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space – automatically respects positivity and ℓ₁‑norm
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()

        # active set (indices within relative tolerance of the max)
        eps = max_val * 1e-6
        active = np.where(conv >= max_val - eps)[0]

        # sub‑gradient: sum of contributions from all active indices
        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming mass‑inflation refinement
# ----------------------------------------------------------------------
def _lp_refine(seq: np.ndarray,
               deadline: float,
               max_mixing: bool = True) -> tuple[np.ndarray, float]:
    """
    Solve   max Σ g_i   s.t.   (seq * g)_k ≤ M   ∀k ,  g_i ≥ 0
    where M = max(conv(seq,seq)).  The LP is built with a sparse Toeplitz
    convolution matrix.  Returned sequence is the best mixture of the
    original vector and the normalised LP solution (including optionally the
    pure LP solution) that improves the exact objective.
    """
    if _opt is None or _sp is None:
        # SciPy not available – skip the LP step
        return seq, _objective_surrogate(seq)

    n = len(seq)
    M = _conv_fft(seq, seq).max()
    L = 2 * n - 1

    # Build convolution matrix A where A @ g = conv(seq, g)
    # A has constant diagonals: diagonal offset d (0≤d<n) is filled with seq[d].
    diags = [np.full(n, seq[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=int)                 # 0 … n‑1
    A = _sp.diags(diagonals=diags, offsets=offsets,
                  shape=(L, n), format='csc')

    # Linear program data
    c = -np.ones(n, dtype=np.float64)                # maximise Σ g_i  → minimise −Σ g_i
    b = np.full(L, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Time limit for the solver (fallback if not supported)
    time_left = deadline - time.time()
    lp_options = {"presolve": True}
    if hasattr(_opt, "highs") and time_left > 0:
        # new SciPy versions understand a `time_limit` option (seconds)
        lp_options["time_limit"] = max(time_left, 0.0)

    try:
        res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                           method="highs",
                           options=lp_options)
    except Exception as e:  # pragma: no cover
        print("[LP] Linprog exception:", e)
        return seq, _objective_surrogate(seq)

    if not res.success:
        # LP did not converge – fall back to the input sequence
        print("[LP] No feasible improvement:", res.message)
        return seq, _objective_surrogate(seq)

    g = res.x
    S = g.sum()
    if S <= 1e-12:
        return seq, _objective_surrogate(seq)

    g_norm = g / S                                    # ℓ₁‑normalised LP solution

    # Try a few convex mixtures of the original and the LP solution.
    # The pure LP solution (t=1) is also tested.
    candidates = [seq]                                 # always keep the original
    if max_mixing:
        for t in (0.2, 0.5, 0.8, 1.0):
            cand = (1.0 - t) * seq + t * g_norm
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
            candidates.append(cand)

    best_seq = seq
    best_val = _objective_surrogate(seq)

    for cand in candidates:
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing polishing (helps escape tiny local minima)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 150_000,
                         start_sigma: float = 0.003,
                         T_start: float = 5e-3,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1
    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the supplied
    ``evaluate_sequence`` function.  The routine respects the supplied time
    budget and stores its best result in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin for final I/O

    # ------------------------------------------------------------------
    # Problem size (fixed for the whole run)
    # ------------------------------------------------------------------
    n = kwargs.get("n", 4000)

    # ------------------------------------------------------------------
    # Load previous solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = np.maximum(prev_arr, 0.0)
                s = prev_arr.sum()
                if s > 0.0:
                    prev_arr = prev_arr / s
                else:
                    prev_arr = None
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # ------------------------------------------------------------------
    # Initial seed
    # ------------------------------------------------------------------
    if prev_arr is not None:
        init_seq = prev_arr.copy()
        print("[Info] Loaded previous best sequence.")
    else:
        init_seq = _sine_window(n, gamma=1.2)
        print("[Info] No previous sequence – starting from sine window.")

    best_seq = init_seq.copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – smooth‑max L‑BFGS refinement (more aggressive schedule)
    # ------------------------------------------------------------------
    if _opt is not None and time.time() < deadline:
        print("\n--- Stage 1: L‑BFGS on smooth max surrogate ---")
        v = np.log(best_seq + 1e-30)               # log‑softmax parameters

        # extended schedule of β values (up to 1e9)
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0,
            200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0,
            1e7, 2e7, 5e7, 1e8
        ]

        for beta in beta_schedule:
            if time.time() > deadline:
                break
            print(f"  * beta = {beta:.0e}")
            smooth_obj = _SmoothMax(n, beta)

            def _callback(_):
                if time.time() > deadline:
                    raise KeyboardInterrupt()

            try:
                res = _opt.minimize(fun=smooth_obj,
                                    x0=v,
                                    method="L-BFGS-B",
                                    jac=True,
                                    callback=_callback,
                                    options={"maxiter": 4000,
                                             "ftol": 1e-15,
                                             "disp": False})
                v = res.x
            except KeyboardInterrupt:
                print("[L‑BFGS] Time budget exhausted – exiting early.")
                break
            except Exception as e:   # pragma: no cover
                print("[L‑BFGS] Exception:", e)
                break

            cur_seq = _softmax(v)
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"    > new best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – high‑precision exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stage 2: Exponentiated‑gradient polishing ---")
        seq_eg, val_eg = _exp_gradient_opt_multi(best_seq,
                                                n,
                                                deadline,
                                                max_iters=2_500_000,
                                                eta0=4.5)
        if val_eg < best_val:
            best_seq = seq_eg
            best_val = val_eg
            print(f"[EG] Improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – linear‑programming mass‑inflation refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stage 3: LP mass‑inflation refinement ---")
        seq_lp, val_lp = _lp_refine(best_seq, deadline, max_mixing=True)
        if val_lp < best_val:
            best_seq = seq_lp
            best_val = val_lp
            print(f"[LP] Improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – optional extra EG pass after LP (quick)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stage 4: Post‑LP EG polishing (short) ---")
        seq_eg2, val_eg2 = _exp_gradient_opt_multi(best_seq,
                                                   n,
                                                   deadline,
                                                   max_iters=600_000,
                                                   eta0=4.0)
        if val_eg2 < best_val:
            best_seq = seq_eg2
            best_val = val_eg2
            print(f"[EG2] Improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Stage 5: Simulated‑annealing fine‑tuning ---")
        seq_sa, val_sa = _simulated_annealing(best_seq,
                                              n,
                                              deadline,
                                              rng,
                                              max_attempts=130_000,
                                              start_sigma=0.0015,
                                              T_start=4e-3,
                                              T_end=1e-9)
        if val_sa < best_val:
            best_seq = seq_sa
            best_val = val_sa
            print(f"[SA] Improved surrogate to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (scale‑invariant – we keep ℓ₁‑norm = 1)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("[Error] evaluate_sequence raised:", exc)

    print("\n=== Finished ===")
    print(f"Surrogate (ℓ₁‑norm = 1) ≈ {best_val:.6f}")
    if true_val is not None:
        print(f"True evaluate_sequence   = {true_val:.6f}")

    # Store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
