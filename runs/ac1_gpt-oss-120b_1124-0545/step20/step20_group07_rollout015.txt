   Fourier curves.  After ℓ₁‑normalisation each candidate is scored with a
   cheap surrogate ``2·n·max(conv(a,a))`` (exact for a sum‑normalised vector).

2. **Adam‑optimisation on a smoothed max** – the best candidate from the
   sweep is refined by minimising a log‑sum‑exp smoothed version of the
   max‑convolution using Adam in the log‑softmax (simplex) space.
   The smoothing parameter β is increased gradually (continuation).

3. **Simulated‑annealing polishing** – a cheap stochastic hill‑climber
   performs mass swaps between random indices; worsening moves are
   accepted with a temperature dependent probability, allowing the
   search to escape shallow local minima.

The whole routine respects the 1000 s wall‑time limit; the best sequence
found is stored in the global ``height_sequence_1`` variable and also
returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used only for L‑BFGS and DPSS windows)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

try:
    import scipy.signal as _signal
except Exception:  # pragma: no cover
    _signal = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Return ``2·n·max(conv(a,a))`` for a vector with ℓ₁‑norm = 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic window families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)                # Modified Bessel I0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Cosine‑tapered window (α=0 ⇒ uniform, α=1 ⇒ Hann)."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Superposition of a few low‑frequency cosine terms."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_window(n: int, scale: float = 0.05) -> np.ndarray:
    """Shifted logistic (monotone) window."""
    i = np.arange(n, dtype=np.float64)
    x = (i - (n - 1) / 2.0) / (scale * n)
    a = 1.0 / (1.0 + np.exp(-x))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) sequence, first eigenvector."""
    if _signal is None:
        return _uniform(n)
    a = _signal.windows.dpss(n, nw, Kmax=1, sym=False)[0]
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_rev(a: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·a[::-1] (both already normalised)."""
    b = a[::-1]
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_vec: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule=None,
                     eta: float = 0.2,
                     eps: float = 1e-12,
                     max_iter_per_beta: int = 8000):
    """
    Minimises the smooth max (log‑sum‑exp) of the auto‑convolution using
    Adam in log‑softmax space.
    """
    if beta_schedule is None:
        beta_schedule = [1, 5, 20, 80, 200, 500, 1000,
                         2000, 5000, 20000, 50000,
                         100000, 200000, 500000, 1000000]

    # work in log‑softmax to enforce simplex automatically
    v = np.log(start_vec + 1e-30)
    m = np.zeros_like(v)
    vhat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999

    for t_idx, beta in enumerate(beta_schedule, 1):
        if time.time() > deadline:
            break
        beta_f = float(beta)

        for it in range(1, max_iter_per_beta + 1):
            if time.time() > deadline:
                break

            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth max via log‑sum‑exp
            shift = max_c
            exp_beta = np.exp(beta_f * (conv - shift))
            sum_exp = exp_beta.sum()
            smooth_obj = 2.0 * n * (shift + np.log(sum_exp) / beta_f)

            # gradient wrt a (see derivation in the original code)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule through softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update in log‑space
            m = beta1 * m + (1 - beta1) * grad_v
            vhat = beta2 * vhat + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t_idx)
            vhat_hat = vhat / (1 - beta2 ** t_idx)
            v -= eta * m_hat / (np.sqrt(vhat_hat) + eps)

        # small re‑normalisation to guard against numerical drift
        v = np.log(_softmax(v))

    return _softmax(v)


# ----------------------------------------------------------------------
# Simple simulated‑annealing polishing
# ----------------------------------------------------------------------
def _simulated_anneal(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      init_temp: float = 5e-4,
                      final_temp: float = 1e-6,
                      steps: int = 150_000) -> np.ndarray:
    """
    Stochastic hill‑climber with temperature schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    # linear temperature schedule
    for step in range(1, steps + 1):
        if time.time() > deadline:
            break
        T = init_temp + (final_temp - init_temp) * (step / steps)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a random fraction (scaled by temperature) of mass from i → j
        delta = T * cur[i] * rng.uniform(0.5, 1.5)
        delta = min(delta, cur[i])
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        val_new = _objective_surrogate(new)
        accept = False
        if val_new < cur_val:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(val_new - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = new
            cur_val = val_new
            if cur_val < best_val:
                best_val, best = cur_val, cur.copy()

    return best


# ----------------------------------------------------------------------
# Main entry point – the function that will be called by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    ``evaluate_sequence``.  Returns the best list found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse previous length if available
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple)) and len(prev) > 0:
        n = len(prev)
    else:
        n = 4000

    # ------------------------------------------------------------------
    # Helper to normalise a vector to ℓ₁‑norm = 1
    # ------------------------------------------------------------------
    def _norm(v: np.ndarray) -> np.ndarray:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return _uniform(n)
        return v / s

    # ------------------------------------------------------------------
    # 1. Very wide parametric sweep (cheap surrogate)
    # ------------------------------------------------------------------
    print(f"\n--- Parametric sweep (n={n}) ---")
    top_k = 20
    best_heap: list[tuple[float, np.ndarray]] = []   # (surrogate, vec)

    def _maybe_insert(vec: np.ndarray):
        val = _objective_surrogate(vec)
        if len(best_heap) < top_k:
            best_heap.append((val, vec))
            best_heap.sort(key=lambda x: x[0])
        else:
            if val < best_heap[-1][0]:
                best_heap[-1] = (val, vec)
                best_heap.sort(key=lambda x: x[0])

    # baseline
    _maybe_insert(_uniform(n))

    # ----- Analytic families -------------------------------------------
    for gamma in np.linspace(0.01, 1.5, 30):
        _maybe_insert(_sine_window(n, gamma))

    for sigma in np.linspace(0.05, 0.30, 20):
        _maybe_insert(_gaussian(n, sigma))

    for beta in np.linspace(0.5, 30.0, 30):
        _maybe_insert(_kaiser_window(n, beta))

    for alpha in np.linspace(0.0, 1.0, 15):
        _maybe_insert(_tukey_window(n, alpha))

    # Bartlett, Blackman, Hann (via Tukey)
    _maybe_insert(_tukey_window(n, 0.0))   # uniform again (kept for completeness)
    _maybe_insert(_tukey_window(n, 1.0))   # Hann

    # DPSS windows (if SciPy present)
    if _signal is not None:
        for nw in (2, 3, 4, 5, 6):
            _maybe_insert(_dpss_window(n, nw))

    # Beta shapes (grid)
    alist = [0.2, 0.5, 0.8, 1.0, 2.0, 4.0, 8.0]
    for a_ in alist:
        for b_ in alist:
            _maybe_insert(_beta_shape(n, a_, b_))

    # Power‑law family (very fine grid)
    for p in np.linspace(0.0, 2.5, 101):
        _maybe_insert(_power(n, p))

    # Logistic (monotone) windows
    for sc in (0.02, 0.05, 0.10, 0.20):
        _maybe_insert(_logistic_window(n, sc))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(12):
            _maybe_insert(_lowfreq_curve(n, nf, rng))

    # Random Dirichlet (wide concentration range)
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 5.0):
        for _ in range(8):
            _maybe_insert(_random_dirichlet(n, conc, rng))

    # Random exponential draws
    for _ in range(12):
        _maybe_insert(_random_exponential(n, rng))

    # Mixed with reverse – can reduce the central correlation
    for _ in range(20):
        base = _random_dirichlet(n, 0.3, rng)
        _maybe_insert(_mix_rev(base, rng.uniform(0.3, 0.7)))

    # Include the previous best if it exists
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.size == n and arr.sum() > 0.0:
            _maybe_insert(_norm(arr))

    # Sort heap (ascending surrogate)
    best_heap.sort(key=lambda x: x[0])
    top_seqs = [vec for _, vec in best_heap[:top_k]]
    best_val, best_seq = best_heap[0]
    print("\nTop candidates after sweep (surrogate):")
    for i, (val, _) in enumerate(best_heap[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam‑based continuation on a smoothed max‑convolution
    # ------------------------------------------------------------------
    if _opt is not None and time.time() < deadline - 20.0:
        print("\n--- Adam continuation smoothing ---")
        # ensure all entries are strictly positive for log‑softmax
        start_vec = np.maximum(best_seq, 1e-14)
        start_vec = start_vec / start_vec.sum()
        seq_adam = _adam_smooth_opt(start_vec,
                                    n,
                                    deadline,
                                    beta_schedule=[1, 5, 20, 80, 200,
                                                   500, 1000, 2000, 5000,
                                                   20000, 50000, 100000,
                                                   200000, 500000, 1000000],
                                    max_iter_per_beta=12000)
        val_adam = _objective_surrogate(seq_adam)
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print(f"  Adam improvement: surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing polishing (large moves)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("\n--- Simulated‑annealing polishing ---")
        seq = best_seq.copy()
        seq = _simulated_anneal(seq, n, deadline,
                                init_temp=5e-4,
                                final_temp=1e-7,
                                steps=int(3e5))
        val_new = _objective_surrogate(seq)
        if val_new < best_val:
            best_val, best_seq = val_new, seq.copy()
            print(f"  SA polishing improvement: surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final true evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
          f"true evaluate_sequence = {true_val}")

    # store for next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
