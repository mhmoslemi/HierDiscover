def _lbfgs_opt(v0, n, deadline, beta_schedule=None):
    if _opt is None:
        return v0
    if beta_schedule is None:
        beta_schedule = [...]
    v = v0.copy()
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()
        try:
            res = _opt.minimize(fun, x0=v, method='L-BFGS-B', jac=True, callback=callback,
                                options={'maxiter': 2000, 'ftol':1e-12, 'disp':False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break
    return v

"""
Propose a step‑function (non‑negative list) that minimizes the
``evaluate_sequence`` objective.

Algorithmic outline
-------------------
1. Choose a problem dimension *n* (default 4000) and generate a diverse
   collection of *seed* probability vectors (exponential decay,
   uniform, binary patterns, random Dirichlet draws and the previous best
   if it exists).
2. Represent a probability vector `a` by its log‑softmax variables
   `v = log(a)`.  In this space the simplex constraint is always satisfied.
3. For each seed run a *smooth* L‑BFGS‑B optimisation of the
   log‑sum‑exp approximation of the max‑convolution:
        fβ(a) = 2·n·(  max(conv(a,a))  )
                ≈ 2·n·( shift + log∑exp(β·(conv‑shift))/β )
   The schedule of β values grows from 1 up to 10⁶, gradually turning the
   smooth surrogate into the true max.  The gradient w.r.t. `v` is obtained
   analytically (see ``_smooth_obj_grad``).  The optimizer is stopped as soon
   as the global time budget (≈1000 s) is exhausted.
4. Keep the best vector found (lowest surrogate value).  As a final polish,
   run a few thousand steps of exponentiated‑gradient mirror‑descent on the
   exact max‑convolution (sub‑gradient on the active index).  This often
   yields a small extra drop.
5. Return the best vector as a plain ``list[float]`` and store it in a global
   variable ``height_sequence_2`` for use by later runs.

The combination of many random restarts, a smooth surrogate optimisation,
and a final exact sub‑gradient pass typically improves the previous bound
(≈1.5085) and can reach the target ≈1.503 within the allocated time.

import time

# ----------------------------------------------------------------------

# ----------------------------------------------------------------------
# FFT‑based convolution helpers (real FFT, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Softmax utilities (numerically stable)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact max‑convolution when Σa = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for a probability vector (sum=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic starter families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) , i=0..n‑1, then normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _binary_pattern(n: int, k: int) -> np.ndarray:
    """k equally‑spaced unit masses (sum = 1)."""
    seq = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return seq
    indices = np.linspace(0, n - 1, k, dtype=int)
    seq[indices] = 1.0 / k
    return seq


# ----------------------------------------------------------------------
# Smooth max‑convolution objective and gradient (log‑sum‑exp)
# ----------------------------------------------------------------------
def _smooth_obj_grad(v: np.ndarray, n: int, beta: float):
    """
    Given log‑softmax variables v, return (f, g) where
    f = 2·n·(shift + log∑exp(beta·(conv‑shift))/beta)
    g = ∂f/∂v (same shape as v).
    """
    a = _softmax(v)                     # probability vector, Σa = 1
    conv = _conv_fft(a, a)
    shift = np.max(conv)                # numerical stabiliser
    # (conv - shift) ≤ 0, safe for exp
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    # smooth approximation of max
    f = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # weight vector for the smoothed max (probability distribution)
    w = exp_beta / sum_exp

    # Gradient wrt a : 2 * conv_rev_a_w  (see derivation in analysis)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                     # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # Adjust for softmax parametrisation: ∂f/∂v = a ⊙ ( grad_a - <grad_a,a> )
    dot = np.dot(grad_a, a)
    g = a * (grad_a - dot)
    return f, g


# ----------------------------------------------------------------------
# L‑BFGS‑B optimisation of the smooth surrogate (multi‑β schedule)
# ----------------------------------------------------------------------
def _lbfgs_opt(v0: np.ndarray,
               n: int,
               deadline: float,
               beta_schedule=None,
               max_iter_per_beta: int = 1500):
    """
    Run a sequence of L‑BFGS‑B minimisations, each with a larger β.
    Returns the final log‑softmax vector v.
    """
    if _opt is None:
        # No scipy – fall back to the starting point.
        return v0

    if beta_schedule is None:
        beta_schedule = [
            1.0,
            5.0,
            20.0,
            80.0,
            200.0,
            500.0,
            1_000.0,
            2_000.0,
            5_000.0,
            20_000.0,
            50_000.0,
            100_000.0,
            200_000.0,
            500_000.0,
            1_000_000.0,
        ]

    v = v0.copy()
    for beta in beta_schedule:
        if time.time() > deadline:
            break

        beta = float(beta)

        def fun(v_local):
            f, g = _smooth_obj_grad(v_local, n, beta)
            return f, g

        def callback(_):
            # Early‑exit when we run out of time.
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'ftol': 1e-12,
                                         'disp': False})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta}")
            break
    return v


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      eta0: float = 5.0):
    """
    Sub‑gradient mirror‑descent (exponentiated gradient) that directly
    minimises the true max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry of the convolution
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size schedule
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point expected by the test harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine runs under a 1000 s wall‑time budget,
    uses multiple random restarts, smooth L‑BFGS optimisation, and a final
    exponentiated‑gradient polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem size (keep it modest for speed).  Feel free to change.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Gather seed probability vectors (all sum to 1, non‑negative).
    # ------------------------------------------------------------------
    seed_vecs = []

    # 1. Previous best, if available.
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        arr = np.asarray(prev, dtype=np.float64)
        arr = np.maximum(arr, 0.0)
        s = arr.sum()
        if s > 0.0:
            seed_vecs.append(arr / s)

    # 2. Analytic families.
    seed_vecs.append(_exp_decay(n, rate=0.001))
    seed_vecs.append(_exp_decay(n, rate=0.003))
    seed_vecs.append(_exp_decay(n, rate=0.006))
    seed_vecs.append(_uniform(n))
    seed_vecs.append(_binary_pattern(n, k=80))
    seed_vecs.append(_binary_pattern(n, k=160))

    # 3. A few random Dirichlet draws.
    for _ in range(4):
        rnd = rng.exponential(scale=1.0, size=n)
        seed_vecs.append(rnd / rnd.sum())

    # ------------------------------------------------------------------
    # Global trackers.
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Main optimisation loop over all seeds (respect time budget).
    # ------------------------------------------------------------------
    for idx, seq0 in enumerate(seed_vecs):
        if time.time() > deadline - 5.0:
            break

        seq0 = np.maximum(seq0, 0.0)
        s = seq0.sum()
        if s == 0.0:
            seq0 = _uniform(n)
        else:
            seq0 = seq0 / s

        # log‑softmax start point
        v0 = np.log(seq0 + 1e-30)

        # ---------- L‑BFGS on smooth surrogate ----------
        v_opt = _lbfgs_opt(v0, n, deadline)
        a_opt = _softmax(v_opt)
        val_opt = _objective_surrogate(a_opt)

        print(f"[Restart {idx+1}/{len(seed_vecs)}] "
              f"surrogate after L‑BFGS = {val_opt:.6f}")

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()

    # ------------------------------------------------------------------
    # Final polishing with exponentiated‑gradient mirror‑descent.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3.0:
        seq_polished, val_polished = _exp_gradient_opt(best_seq, n, deadline,
                                                      max_iters=300_000,
                                                      eta0=5.0)
        print(f"[Polish] surrogate after EG = {val_polished:.6f}")
        if val_polished < best_val:
            best_val = val_polished
            best_seq = seq_polished.copy()

    # ------------------------------------------------------------------
    # Final (true) evaluation via the supplied function.
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"[Result] surrogate = {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised:", exc)

    # Store for potential future runs.
    globals()["height_sequence_2"] = best_seq.tolist()
    return best_seq.tolist()
