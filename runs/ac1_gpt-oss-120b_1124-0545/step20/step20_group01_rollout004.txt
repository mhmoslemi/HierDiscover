    beta1 = 0.9; beta2 = 0.999; eps = 1e-8
    t = 0
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # assign iterations (maybe constant)
        max_iter = iters_per_beta if isinstance(iters_per_beta, int) else iters_per_beta.get(beta, 2000)
        for i in range(max_iter):
            t += 1
            if time.time() > deadline:
                break
            a = _softmax(v)  # returns probs sum=1
            # compute conv
            conv = _conv_fft(a, a)
            max_conv = conv.max()
            # compute smooth weights
            shift = max_conv
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp  # shape conv length

            rev_a = a[::-1]

            m_hat = m / (1 - beta1 ** t)

                    best_seq = a.copy()
                    # Print improvement

"""
Deterministic‑annealing optimiser for the max‑self‑convolution bound.

The algorithm works in three stages:

1.  **Rich seed pool** – a collection of analytic families (uniform,
    triangular, sine‑window, Gaussian, power‑law, exponential, beta‑shape,
    random Dirichlet, sparse Dirichlet, piecewise‑linear) together with a few
    random permutations.  All candidates are ℓ₁‑normalised and evaluated with
    the cheap surrogate `2·n·max(conv(a,a))`.  The best `top_k` seeds are kept.

2.  **Annealed Adam optimisation** – starting from each seed we minimise the
    *smooth* max‑convolution  `F_β(a)= 2·n·(shift+log Σ exp(β·(conv‑shift))/β)`
    with an Adam‑style mirror‑descent in the log‑softmax space.
    The temperature `β` is increased geometrically (deterministic annealing);
    at each temperature a few thousand Adam steps are performed.
    This refines the shape far better than a plain sub‑gradient step.

3.  **Final permutation hill‑climb** – a short simulated‑annealing phase
    (swap moves and tiny Gaussian perturbations) is applied to the best
    vector found in step 2 to polish the ordering of the heights.

All work respects the overall 1000 s budget.  The best sequence is stored in
the global variable ``height_sequence_1`` for possible continuation runs.
"""  # noqa: E501

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – always sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – the quantity we try to minimise."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·|i-(n‑1)/2|)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Keep only the largest 20 % of a Dirichlet draw."""
    vec = rng.dirichlet(np.full(n, conc))
    thresh = np.percentile(vec, 80)
    vec = np.where(vec >= thresh, vec, 0.0)
    tot = vec.sum()
    if tot == 0.0:
        return _uniform(n)
    return vec / tot


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    """Random piecewise‑linear curve with a given number of interior break‑points."""
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (Adam) optimiser on the *smooth* max‑convolution.
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule,
                   iters_per_beta: int = 2000,
                   lr0: float = 0.5):
    """
    Mirror‑descent in log‑softmax space using Adam.
    The objective at temperature β is a smooth approximation of max(conv).
    """
    # Initialise log‑weights
    v = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(v)
    u = np.zeros_like(v)          # second moment (named 'u' to avoid clash with builtin)
    b1, b2 = 0.9, 0.999
    eps = 1e-12

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    t = 0
    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        # Simple linear allocation of inner iterations
        max_it = iters_per_beta
        for _ in range(max_it):
            t += 1
            if time.time() >= deadline:
                break

            # ----- a = softmax(v) -----
            a = _softmax(v)

            # ----- convolution a*a -----
            conv = _conv_fft(a, a)                     # length 2n‑1
            max_conv = np.max(conv)

            # ----- smooth weights w (softmax over β·conv) -----
            shift = max_conv
            exp_beta = np.exp(beta * (conv - shift))   # shift prevents overflow
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                      # length 2n‑1

            # ----- gradient wrt a (exact) -----
            # grad_a = 2 * (a_rev * w)   – central n entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                  # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # ----- gradient wrt v (mirror‑descent) -----
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ----- Adam update on v -----
            m = b1 * m + (1.0 - b1) * grad_v
            u = b2 * u + (1.0 - b2) * (grad_v ** 2)

            m_hat = m / (1.0 - b1 ** t)
            u_hat = u / (1.0 - b2 ** t)

            # learning rate scales roughly as 1/(β+1)
            lr = lr0 / (beta + 1.0)

            v = v - lr * m_hat / (np.sqrt(u_hat) + eps)

            # ----- keep the best (exact surrogate) -----
            cur_val = 2.0 * n * max_conv
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing permutation optimiser (used for a final polish)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000):
    """
    Swaps two positions or applies a tiny Gaussian perturbation.
    Accepts worsening moves with a decreasing temperature.
    Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    conv = _conv_fft(cur_seq, cur_seq)
    cur_val = 2.0 * n * np.max(conv)

    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule (linear in iteration count)
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:       # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                         # Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        conv = _conv_fft(ns, ns)
        val = 2.0 * n * np.max(conv)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            # Metropolis acceptance
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the provided ``evaluate_sequence`` objective.

    The routine builds a large pool of analytic and random seeds, keeps the
    best few, refines each of them with a deterministic‑annealing Adam optimiser,
    and finishes with a short permutation hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previously saved solution (if any) for warm‑start.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                # Normalise just in case the stored vector is not perfectly ℓ₁‑normalised.
                arr = arr / max(arr.sum(), 1e-30)
                prev_seed = arr
            else:
                prev_seed = None
        except Exception:
            prev_seed = None
    else:
        prev_seed = None

    # ------------------------------------------------------------------
    # 1. Build a generous seed pool (analytic + random families).
    # ------------------------------------------------------------------
    seeds = []

    # always keep the previous best (if available)
    if prev_seed is not None:
        seeds.append(prev_seed)

    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    for gamma in np.linspace(0.5, 3.0, 10):
        seeds.append(_sine_window(n, gamma=gamma))

    for sigma_frac in np.linspace(0.06, 0.30, 10):
        seeds.append(_gaussian(n, sigma_frac))

    for p in np.linspace(0.0, 4.0, 10):
        seeds.append(_power(n, p))

    for rate in np.logspace(-5, -0.4, 12):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # Dirichlet draws of various concentrations
    for conc in [0.05, 0.1, 0.5, 1.0, 2.0]:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))
    # Sparse Dirichlet draws (very few non‑zeros)
    for conc in [0.001, 0.005]:
        for _ in range(2):
            seeds.append(_random_sparse_dirichlet(n, conc, rng))

    # Random piecewise‑linear curves
    for br in (3, 5):
        for _ in range(4):
            seeds.append(_piecewise_linear(n, br, rng))

    # Random permutations of a few “interesting” bases – break symmetry
    base_shapes = [_exp_decay(n, 0.003), _sym_exp_decay(n, 0.004),
                   _beta_shape(n, 2.0, 2.0), _gaussian(n, 0.18)]
    for base in base_shapes:
        for _ in range(4):
            seeds.append(np.random.permutation(base))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of every seed – keep the top‑k.
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        raise RuntimeError("No valid seed generated within time budget.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise the global best with the absolute best seed.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine each top seed with deterministic‑annealing Adam.
    # ------------------------------------------------------------------
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
        2000.0, 8000.0, 32000.0, 128000.0, 512000.0
    ]

    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # split remaining time equally among the seeds that are still to be processed
        remaining = deadline - time.time()
        share = remaining * 0.85 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si + 1}] annealing – time budget ≈ {share:.1f}s")

        seq_opt, val_opt = _adam_optimize(
            start_seq=seed_vec,
            n=n,
            deadline=sub_deadline,
            beta_schedule=beta_schedule,
            iters_per_beta=2500,
            lr0=0.5,
        )

        print(f"  -> annealed surrogate = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"    *** New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final short permutation hill‑climb to fine‑tune ordering.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + remaining * 0.90
        print("\n--- Final permutation hill‑climb ---")
        seq_final, val_final = _perm_hill_climb(
            start_seq=global_best_seq.copy(),
            n=n,
            deadline=sub_deadline,
            rng=rng,
            max_iters=150_000,
        )
        if val_final < global_best_val:
            global_best_seq = seq_final
            global_best_val = val_final
            print(f"[Final hill‑climb] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate the true objective via the provided function.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store the best sequence for possible continuation runs.
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
