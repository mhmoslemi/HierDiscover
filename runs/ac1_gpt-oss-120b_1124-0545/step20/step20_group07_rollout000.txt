def build_basis(n, K):
    i = np.arange(n, dtype=np.float64)
    cols = 1 + 2 * K
    B = np.empty((n, cols), dtype=np.float64)
    B[:,0] = 1.0
    col = 1
    for k in range(1, K+1):
        B[:,col] = np.cos(2.0*np.pi*k*i/n)
        col += 1
        B[:,col] = np.sin(2.0*np.pi*k*i/n)
        col += 1
    return B

"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm (≈ 100 s on a single core):
1.  Choose a moderate length (reuse the previous length or 4000).
2.  Parametrise the shape with a low‑frequency Fourier basis:
        v_i = θ_0 + Σ_{k=1}^K (θ_{2k-1}·cos(2π k i/n) + θ_{2k}·sin(2π k i/n))
    and obtain the probability vector a_i = softmax(v_i) (automatically
    non‑negative and ℓ₁‑normalised).
3.  Optimise the parameters θ with Adam on a *smooth* version of the
    max‑convolution objective:
        f(θ) = 2·n·(max(conv(a,a)) + log‑sum‑exp_β(conv(a,a))/β )
    The smoothing parameter β is increased gradually (continuation).
    The gradient is obtained analytically using the convolution theorem,
    so each iteration costs only a few FFTs.
4.  Run several random restarts (one warm‑started from the previous
    solution, the others random) and keep the best shape found.
5.  Polish the best shape with a cheap stochastic “move‑mass”
    hill‑climber.
6.  Store the result in the global ``height_sequence_1`` variable and
    return it as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for ℓ₁‑normalised vectors: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _build_basis(n: int, K: int) -> np.ndarray:
    """
    Fourier basis matrix B (n × (1+2K)):
        column 0      → constant term
        columns 2k‑1 → cos(2πk·i/n)
        columns 2k   → sin(2πk·i/n)
    """
    i = np.arange(n, dtype=np.float64)
    cols = 1 + 2 * K
    B = np.empty((n, cols), dtype=np.float64)
    B[:, 0] = 1.0
    c = 1
    for k in range(1, K + 1):
        B[:, c] = np.cos(2.0 * np.pi * k * i / n)
        c += 1
        B[:, c] = np.sin(2.0 * np.pi * k * i / n)
        c += 1
    return B


def _smooth_obj_and_grad(theta: np.ndarray,
                         B: np.ndarray,
                         n: int,
                         beta: float) -> tuple[float, np.ndarray]:
    """
    smoothed objective (β‑log‑sum‑exp of the convolution) and its gradient
    w.r.t. the Fourier coefficients ``theta``.
    """
    v = B @ theta                     # (n,)
    a = _softmax(v)                   # probability vector, sum=1
    conv = _conv_fft(a, a)            # length 2n‑1
    m = np.max(conv)                  # exact max entry
    shift = m
    # weights for the smooth max (log‑sum‑exp)
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    smooth_max = shift + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_max

    w = exp_beta / sum_exp            # shape (2n‑1,)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)        # length 3n‑2, we need the middle n entries
    grad_a = 2.0 * full[n - 1:n - 1 + n]   # (n,)

    # gradient through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)       # (n,)
    grad_theta = B.T @ grad_v         # (d,)

    return obj, grad_theta

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple)) and len(prev) > 0:
        n = len(prev)
    else:
        n = 4000

    # Fourier basis (low frequencies only)
    K = 12                      # number of sinusoidal pairs → 1+2K coefficients
    B = _build_basis(n, K)
    d = B.shape[1]

    # --------------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------------
    beta_schedule = [
        0.5, 2.0, 5.0, 20.0, 80.0, 200.0, 500.0,
        2000.0, 5000.0, 20000.0, 50000.0, 100000.0,
        200000.0, 500000.0, 1000000.0
    ]
    steps_per_beta = 3000          # Adam steps for each smoothing level
    max_restarts = 3               # 1 warm‑start + 2 random

    best_val = np.inf
    best_seq = None

    print(f"\n=== Fourier‑Adam optimisation (n={n}, K={K}) ===")

    for restart in range(max_restarts):
        if time.time() > deadline:
            break

        # ----- initialise θ -------------------------------------------------
        if restart == 0 and isinstance(prev, (list, tuple)):
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape[0] == n and prev_arr.sum() > 0:
                v0 = np.log(np.maximum(prev_arr, 1e-30))
                theta0, *_ = np.linalg.lstsq(B, v0, rcond=None)
                theta = theta0.astype(np.float64)
            else:
                theta = rng.normal(scale=0.1, size=d)
        else:
            theta = rng.normal(scale=0.1, size=d)

        # Adam moments
        m = np.zeros_like(theta)
        v_adam = np.zeros_like(theta)
        t_counter = 0

        # ----- continuation over β -----------------------------------------
        for beta in beta_schedule:
            if time.time() > deadline:
                break
            for step in range(steps_per_beta):
                if time.time() > deadline:
                    break

                # forward pass
                v_raw = B @ theta
                a = _softmax(v_raw)
                conv = _conv_fft(a, a)
                maxc = np.max(conv)

                # smoothed gradient (log‑sum‑exp)
                shift = maxc
                exp_beta = np.exp(beta * (conv - shift))
                sum_exp = exp_beta.sum()
                w = exp_beta / sum_exp

                rev_a = a[::-1]
                full = _conv_fft(rev_a, w)
                grad_a = 2.0 * full[n - 1:n - 1 + n]

                dot = np.dot(grad_a, a)
                grad_v = a * (grad_a - dot)
                grad_theta = B.T @ grad_v

                # Adam update
                t_counter += 1
                m = 0.9 * m + 0.1 * grad_theta
                v_adam = 0.999 * v_adam + 0.001 * (grad_theta ** 2)
                m_hat = m / (1.0 - 0.9 ** t_counter)
                v_hat = v_adam / (1.0 - 0.999 ** t_counter)
                step_size = 0.1 / (np.sqrt(v_hat) + 1e-8)
                theta = theta - step_size * m_hat

                # periodic exact evaluation (every 200 steps)
                if step % 200 == 0:
                    exact_val = 2.0 * n * maxc
                    if exact_val < best_val:
                        best_val = exact_val
                        best_seq = a.copy()
                        print(f"[restart {restart+1}] β={beta:.1e} step={step:04d} "
                              f"new best surrogate={best_val:.6f}")

        # final check for this restart
        v_raw = B @ theta
        a = _softmax(v_raw)
        val = _objective_surrogate(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[restart {restart+1}] final surrogate={best_val:.6f}")

    # --------------------------------------------------------------
    # If everything failed, fall back to uniform
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective_surrogate(best_seq)

    # --------------------------------------------------------------
    # Local stochastic polishing (tiny mass transfers)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        print("\n--- Local stochastic polishing ---")
        seq = best_seq.copy()
        cur_val = best_val
        step_frac = 0.05
        it = 0
        while time.time() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = step_frac * seq[i] * rng.uniform(0.5, 1.5)
            delta = min(delta, seq[i])
            new_seq = seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            val_new = _objective_surrogate(new_seq)
            if val_new < cur_val:
                seq = new_seq
                cur_val = val_new
                if cur_val < best_val:
                    best_val, best_seq = cur_val, seq.copy()
                    print(f"  it {it:06d}: new best surrogate={best_val:.6f}")
                step_frac = min(0.2, step_frac * 1.02)
            else:
                step_frac = max(1e-4, step_frac * 0.9995)
            it += 1
            if it % 5000 == 0:
                print(f"  it {it:06d}: cur surrogate={cur_val:.6f}")

    # --------------------------------------------------------------
    # Final true evaluation and bookkeeping
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised", exc)

    print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
          f"true evaluate_sequence = {true_val}")

    # store for the next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
