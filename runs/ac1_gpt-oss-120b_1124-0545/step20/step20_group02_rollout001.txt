"""
Propose a non‑negative step‑function (height sequence) that minimises

    evaluate_sequence(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

Algorithm (≈ 1000 s budget)

1.  Choose a small to moderate problem size n (500 – 2500).  Smaller n lowers the
    factor 2·n in the objective, while a well‑shaped distribution can keep the
    maximal self‑convolution low.

2.  Generate a large pool of random candidate vectors for each n using a
    variety of parametric families (Dirichlet, Beta, Gaussian, exponential,
    low‑frequency cosine mixtures, etc.).  The candidates are ℓ₁‑normalised,
    non‑negative and evaluated *exactly* with a fast FFT‑based convolution.

3.  Keep the best few candidates overall (by the surrogate objective, which is
    identical to the true value when the sum = 1).  These become the seeds for
    a local optimisation phase.

4.  Local optimisation – mirror‑descent (exponentiated‑gradient) on the
    exact max‑convolution.  The algorithm works in log‑softmax space, thus
    maintaining positivity and the simplex constraint automatically.  The
    sub‑gradient is built from all (near‑maximal) active convolution indices.

5.  The top refined seed is polished with a brief simulated‑annealing pass.

6.  The best vector found is evaluated with the supplied ``evaluate_sequence``
    function and stored in the global variable ``height_sequence_1``.

All heavy work respects the global time budget; the routine stops well before
the deadline and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate (exact) objective for ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq, seq)) for a vector with sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - center))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and re‑normalize."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 500_000,
                            eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns the best sequence found and its exact objective value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set: indices whose value is within 1e-6 relative tolerance
        eps = max_val * 1e-6
        active = np.where(conv >= max_val - eps)[0]

        g_tot = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_tot[idx_i] += a[rev_idx]

        if active.size:
            # average sub‑gradient over active indices
            g = 2.0 * g_tot / active.size
        else:
            g = np.zeros_like(a)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (lightweight)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 120_000,
                         start_sigma: float = 0.001,
                         T_start: float = 5e-3,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbations with a linear cooling schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear temperature schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 1 – broad random search (fast, many candidates)
    # ------------------------------------------------------------------
    # We'll collect the top K candidates (by surrogate value) across all n.
    TOP_K = 12
    candidate_pool: list[tuple[np.ndarray, float, int]] = []

    print("\n--- Stage 1 : random search ---")
    for n in n_values:
        if time.time() > deadline:
            break
        print(f"[Stage 1] Exploring n = {n}")

        # deterministic seeds
        deterministic = [
            _uniform(n),
            _sine_window(n, gamma=1.2),
            _gaussian(n, sigma_frac=0.15),
            _exp_decay(n, rate=0.005),
            _sym_exp_decay(n, rate=0.008),
            _beta_shape(n, alpha=2.0, beta=2.0)
        ]

        for seq in deterministic:
            v = _objective_surrogate(seq)
            candidate_pool.append((seq, v, n))

        # random families
        n_random = 120  # number of random draws per n (adjustable)
        for _ in range(n_random):
            if time.time() > deadline:
                break
            seq = _random_candidate(n, rng)
            v = _objective_surrogate(seq)
            candidate_pool.append((seq, v, n))

        # keep only the best TOP_K candidates found so far
        candidate_pool.sort(key=lambda x: x[1])
        candidate_pool = candidate_pool[:TOP_K]

    # ------------------------------------------------------------------
    # Stage 2 – local refinement with exponentiated‑gradient
    # ------------------------------------------------------------------
    print("\n--- Stage 2 : EG polishing ---")
    # Process candidates from best to worst, allocating remaining time equally.
    for idx, (seq, val, n) in enumerate(candidate_pool):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # Allocate at least 30 s per polishing run; if not enough time, skip.
        if remaining < 30.0:
            break
        max_iters = 600_000           # generous but still affordable for modest n
        per_run_deadline = time.time() + min(remaining, 60.0)  # cap at 60 s per candidate
        print(f"[EG] Polishing candidate #{idx + 1} (n={n}, start val={val:.6f})")
        seq_opt, val_opt = _exp_gradient_opt_multi(seq, n, per_run_deadline,
                                                   max_iters=max_iters,
                                                   eta0=5.0)
        print(f"     → polished val = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq, best_n = val_opt, seq_opt.copy(), n
            print(f"[Info] New global best: value = {best_val:.6f} (n={best_n})")

    # ------------------------------------------------------------------
    # Stage 3 – final simulated‑annealing fine‑tuning (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n--- Stage 3 : Simulated annealing fine‑tuning ---")
        per_sa_deadline = deadline
        seq_sa, val_sa = _simulated_annealing(best_seq, best_n,
                                              per_sa_deadline, rng,
                                              max_attempts=150_000,
                                              start_sigma=0.001,
                                              T_start=5e-3,
                                              T_end=1e-9)
        if val_sa < best_val:
            best_val, best_seq, best_n = val_sa, seq_sa.copy(), best_n
            print(f"[Info] SA improved global best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation and store result
    # ------------------------------------------------------------------
    if best_seq is None:
        raise RuntimeError("No candidate was generated within the time budget.")

    # The surrogate already equals the true value when sum = 1.
    true_val = evaluate_sequence(best_seq.tolist())
    print("\n=== Final result ===")
    print(f"Best n = {best_n}")
    print(f"Surrogate (≈ true) = {best_val:.6f}")
    print(f"evaluate_sequence    = {true_val:.6f}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
