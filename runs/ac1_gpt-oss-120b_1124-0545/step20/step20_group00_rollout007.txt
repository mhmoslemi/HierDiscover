"""
Propose a candidate step‑function that improves the current best bound.
Algorithm
---------

1.  Load the previous best vector (if any) and build a small pool of
    simple seeds: uniform, Dirichlet draws, a few parametric families
    (power‑law edges, linear edge‑boosted shapes).

2.  Evaluate all seeds with the exact surrogate  ``2·n·max(conv(a,a))``
    and keep the best one.

3.  Refine the best seed by **Adam** optimisation on a *smooth*
    approximation of the max‑convolution.  The smoothness is controlled
    by a temperature parameter β that is increased (β←β·2) every few
    thousand iterations, ending at β≈10⁶.  Variables are kept in the
    log‑softmax space, guaranteeing non‑negativity and unit L¹‑norm.

4.  Polishing: a long run of exponentiated‑gradient (mirror descent)
    on the true non‑smoothed objective.  The step size follows the
    classic η_t = η₀/√t schedule.

5.  Final evaluation with the provided ``evaluate_sequence`` and store
    the result in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via a real FFT (O(n log n)).
    Lengths of x and y are assumed to be ≤ 4000, so a fixed-size FFT
    (next power of two ≥ 2·4000‑1 = 8192) is fast and convenient.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)          # 8192 for n≈4000
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for a probability vector:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple parametric families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on n points."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _dirichlet_draw(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _shape_power_gamma(gamma: float, n: int) -> np.ndarray:
    """
    Edge‑heavy shape: a_i ∝ (i+1)^(-γ) + (n-i)^(-γ).
    γ = 0 gives the uniform distribution.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    j = np.arange(n, 0, -1, dtype=np.float64)
    raw = np.power(i, -gamma) + np.power(j, -gamma)
    s = raw.sum()
    if s == 0.0:
        return _uniform(n)
    return raw / s


def _shape_linear_alpha(alpha: float, n: int) -> np.ndarray:
    """
    Linear edge boost: a_i ∝ 1 + α·|i−center|.
    For α>0 the ends are higher than the centre.
    """
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    raw = 1.0 + alpha * np.abs(idx - centre)
    raw = np.maximum(raw, 0.0)          # safety
    s = raw.sum()
    if s == 0.0:
        return _uniform(n)
    return raw / s

# Adam optimiser on the smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 200_000,
              beta_start: float = 1.0,
              beta_factor: float = 2.0,
              beta_max: float = 1_000_000.0,
              beta_step: int = 2_000,
              lr: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth approximation of the max‑convolution.
    Variables live in log‑softmax space (``v``) guaranteeing the simplex
    constraints.  The temperature β is increased geometrically every
    `beta_step` iterations.
    Returns the best probability vector seen and its true surrogate value.
    """
    # Initialise log‑variables
    eps = 1e-30
    log_a = np.log(start_seq + eps)

    # Adam moments
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    beta = beta_start
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    # Adam hyper‑parameters
    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Increase β according to schedule
        if (it % beta_step) == 0 and beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # Current probability vector
        a = _softmax(log_a)

        # Convolution and smoothed weights
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        # exponential weighting (log‑sum‑exp trick)
        exp_beta = np.exp(beta * (conv - max_c))
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp                       # size 2·n‑1

        # Gradient w.r.t. a (see derivation in the problem statement)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                   # length 3·n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on log‑variables
        m = b1 * m + (1.0 - b1) * grad_v
        v = b2 * v + (1.0 - b2) * (grad_v ** 2)
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Track the best (true) surrogate value
        cur_val = float(2.0 * n * max_c)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # Light progress output (once per 5000 it)
        if (it % 5000) == 0:
            print(f"[Adam] it={it:6d}  β={beta:8.1f}  surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated gradient) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if (it % 100_000) == 0:
            print(f"[EG] it={it:7d}  surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found (list of floats)
    and stores it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    n = 4000                                        # fixed problem size

    # ------------------------------------------------------------------
    # Load any previously stored solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size != n:
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1️⃣  Build a modest pool of seeds (fast evaluation)
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    if prev_arr is not None and prev_arr.sum() > 0.0:
        # Normalise the previous solution to the simplex
        prev_norm = prev_arr / prev_arr.sum()
        seeds.append(prev_norm)

    # Dirichlet draws with several concentration values
    for conc in (0.05, 0.2, 1.0):
        for _ in range(2):
            seeds.append(_dirichlet_draw(n, conc, rng))

    # Power‑law edge shapes
    for gamma in (0.0, 0.2, 0.5, 1.0, 2.0, 3.0, 4.0):
        seeds.append(_shape_power_gamma(gamma, n))

    # Linear edge‑boosted shapes
    for alpha in (0.5, 1.0, 2.0, 4.0):
        seeds.append(_shape_linear_alpha(alpha, n))

    # ------------------------------------------------------------------
    # Evaluate the seed pool and keep the best
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Evaluating initial seeds ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        print(f"  seed #{idx+1:2d} surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam optimisation on a smooth approximation of max‑conv
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        rem = deadline - time.time()
        # Allocate roughly 55 % of the remaining budget to Adam
        adam_deadline = time.time() + rem * 0.55
        max_iters = 300_000                        # generous upper bound
        seq_adam, val_adam = _adam_opt(best_seq, n,
                                       adam_deadline,
                                       max_iters=max_iters,
                                       beta_start=1.0,
                                       beta_factor=2.0,
                                       beta_max=1_000_000.0,
                                       beta_step=2_000,
                                       lr=0.15)
        print(f"[Adam] best surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Exhaustive mirror‑descent (exponentiated gradient) polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing ---")
        # Allocate the rest of the budget (minus a tiny buffer)
        polishing_deadline = deadline - 5.0
        remaining_iters = int(2_000_000 *
                              (polishing_deadline - time.time()) /
                              (deadline - start_time))
        remaining_iters = max(200_000, min(remaining_iters, 5_000_000))

        seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                           polishing_deadline,
                                           max_iters=remaining_iters,
                                           eta0=5.0)
        print(f"[EG] surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print(f"  >>> Updated global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Official evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
