def _adam_optimize(start_seq, n, deadline, beta_schedule, iters_per_beta=5000, lr=0.05):
    # convert start_seq to numpy array and ensure sum=1.
    a = start_seq.copy()
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / a_sum

    # initialize logits: log_a = np.log(a + eps)
    v = np.log(a + 1e-30)

    # init Adam states
    m = np.zeros_like(v)
    v_m = np.zeros_like(v)
    t = 0

    best_seq = a.copy()
    best_val = _objective_surrogate(a) # compute exact max conv

    for beta in beta_schedule:
        if time.time() >= deadline - 0.5: break
        for i in range(iters_per_beta):
            t += 1
            if time.time() >= deadline - 0.5:
                break

            # compute softmax a
            a = _softmax(v)

            # compute conv
            conv = _conv_fft(a, a)

            # compute smoothed weights
            # using shift = max(conv) for numerical stability
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp

            # gradient wrt a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]  # length n

            # gradient wrt logits
            # The gradient of softmax: grad_v = a * (grad_a - dot(grad_a, a))
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            m = 0.9 * m + 0.1 * grad_v
            v_m = 0.999 * v_m + 0.001 * (grad_v ** 2)
            m_hat = m / (1 - 0.9 ** t)
            v_hat = v_m / (1 - 0.999 ** t)
            # Learning rate may scale: lr / (sqrt(v_hat) + eps)
            v = v - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # compute exact surrogate (max conv) after update
            if (i % 10) == 0:  # occasional evaluate
                a_check = _softmax(v)
                cur_val = _objective_surrogate(a_check)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a_check.copy()

    return best_seq, best_val

"""
Propose a step‑function (non‑negative sequence) that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The algorithm explores several simple analytical families (uniform,
exponential decay, V‑shaped, Beta‑distribution, random Dirichlet) as
starting points, then refines each candidate with a continuation
method:

    • A smoothed max‑convolution loss   Lβ(a)= 2·n·(1/β)·log Σ exp(β·conv)
    • Adam optimisation on the logits of a softmax
      (so the ℓ₁‑norm = 1 is always satisfied)
    • β is increased gradually (continuation) to approach the true
      max‑convolution.

After the continuation phases the best sequence found across all
restarts is polished a little with the exact sub‑gradient mirror‑descent
used in the previous baseline.  The whole search respects the 1000 s
budget and stores the best sequence in the global variable
``height_sequence_1`` for the next run.
"""
import math
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helper utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback – should never happen
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate (2·n·max(conv(seq,seq))) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – used as seeds for the optimiser
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _v_shape(n: int) -> np.ndarray:
    """V‑shaped (low in the centre, high at the borders)."""
    half = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.where(i <= half, i, n - 1 - i)  # linear ramp up then down
    a = a + 1e-12   # avoid exact zeros
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density sampled on the mid‑points of n bins."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # pdf proportional to x^{α-1} (1-x)^{β-1}
    # Use log‑space to avoid underflow for extreme parameters
    log_pdf = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    # Normalisation constant (log‑Beta) via gamma functions
    from math import lgamma

    log_norm = lgamma(alpha) + lgamma(beta) - lgamma(alpha + beta)
    pdf = np.exp(log_pdf - log_norm)
    pdf = np.maximum(pdf, 0.0)
    s = pdf.sum()
    if s == 0.0:
        return _uniform(n)
    return pdf / s


def _random_dirichlet(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet(α,…,α) vector (α = concentration)."""
    g = rng.gamma(concentration, 1.0, size=n)
    if g.sum() == 0.0:
        return _uniform(n)
    return g / g.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution loss
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule,
                   inner_iters: int = 3000,
                   lr: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Continuation Adam optimiser.
    Returns (best_sequence, best_surrogate_value) seen during the run.
    """
    # initialise logits (log‑softmax parameters)
    a = np.maximum(start_seq, 0.0)
    s = a.sum()
    if s == 0.0:
        a = _uniform(n)
    else:
        a = a / s
    logits = np.log(a + 1e-30)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for beta in beta_schedule:
        if time.time() > deadline - 0.5:
            break
        beta = float(beta)
        for _ in range(inner_iters):
            if time.time() > deadline - 0.5:
                break

            # ----------------------------------------------------------
            # a = softmax(logits)  – guarantees sum(a)=1, a≥0
            # ----------------------------------------------------------
            a = _softmax(logits)

            # ----------------------------------------------------------
            # Convolution and smoothed max weights
            # ----------------------------------------------------------
            conv = _conv_fft(a, a)                     # length 2n‑1
            shift = np.max(conv)                        # for numerical stability
            # w_k ∝ exp(beta·(conv_k−shift)) ; Σ w_k = 1
            exp_term = np.exp(beta * (conv - shift))
            sum_exp = exp_term.sum()
            w = exp_term / sum_exp

            # ----------------------------------------------------------
            # Gradient w.r.t. a (derivative of Σ w_k·conv_k)
            # ----------------------------------------------------------
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]        # slice to length n

            # ----------------------------------------------------------
            # Gradient w.r.t. logits (softmax chain rule)
            # ----------------------------------------------------------
            dot = np.dot(grad_a, a)
            grad_logits = a * (grad_a - dot)

            # ----------------------------------------------------------
            # Adam update
            # ----------------------------------------------------------
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_logits
            v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            logits = logits - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----------------------------------------------------------
            # Occasionally evaluate the true (non‑smoothed) surrogate
            # ----------------------------------------------------------
            if (_ & 15) == 0:            # every 16 iterations
                cur_val = _objective_surrogate(_softmax(logits))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(logits).copy()

    # final check
    final_seq = _softmax(logits)
    final_val = _objective_surrogate(final_seq)
    if final_val < best_val:
        best_val = final_val
        best_seq = final_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact max‑convolution mirror‑descent (sub‑gradient) used for polishing
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution (sub‑gradient on the
    active index).  Returns (best_sequence, best_surrogate).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline - 0.5:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point (required by the platform)
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs several random restarts, each refined by a continuation
    Adam optimiser on a smoothed max‑convolution loss, followed by a short
    exact mirror‑descent polishing phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size (the previous run used 4000 – we keep the same)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Gather initial seeds
    # ------------------------------------------------------------------
    seeds = []

    # 1) previous best, if it matches the required length
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        arr = np.asarray(prev, dtype=np.float64)
        arr = np.maximum(arr, 0.0)
        if arr.sum() == 0.0:
            arr = _uniform(n)
        else:
            arr = arr / arr.sum()
        seeds.append(arr)

    # 2) a few deterministic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.003))

    # 3) random exponential decays
    for _ in range(2):
        rate = rng.uniform(0.001, 0.010)
        seeds.append(_exp_decay(n, rate=rate))

    # 4) V‑shape
    seeds.append(_v_shape(n))

    # 5) a few Beta‑distributions with random parameters
    for _ in range(3):
        a = rng.uniform(0.2, 5.0)
        b = rng.uniform(0.2, 5.0)
        seeds.append(_beta_distribution(n, alpha=a, beta=b))

    # 6) random Dirichlet draws (concentration < 1 gives sparse shapes)
    for conc in (0.2, 0.5, 1.0):
        seeds.append(_random_dirichlet(n, concentration=conc, rng=rng))

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    beta_sched = [
        1, 5, 20, 80, 200, 500, 1000,
        2000, 5000, 20000, 50000, 100000,
        200000, 500000, 1_000_000
    ]
    inner_iters = 3000          # iterations per β value (adjusted later if time runs out)

    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation loop over all seeds
    # ------------------------------------------------------------------
    for idx, init_seq in enumerate(seeds):
        if time.time() > deadline - 5.0:
            break   # not enough time for another full run

        print(f"[Restart {idx + 1}/{len(seeds)}] Starting Adam continuation...")
        seq_adam, val_adam = _adam_optimize(
            start_seq=init_seq,
            n=n,
            deadline=deadline,
            beta_schedule=beta_sched,
            inner_iters=inner_iters,
            lr=0.05,
        )
        print(f"    [Adam] surrogate = {val_adam:.6f}")

        # Polish the result with a short exact sub‑gradient phase
        seq_polished, val_polished = _exp_gradient_opt(
            start_seq=seq_adam,
            n=n,
            deadline=deadline,
            max_iters=150_000,
            eta0=5.0,
        )
        print(f"    [Polish] surrogate = {val_polished:.6f}")

        if val_polished < global_best_val:
            global_best_val = val_polished
            global_best_seq = seq_polished.copy()
            print(f"    ** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation via the provided function (for safety)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # Should never happen – fall back to a uniform distribution
        global_best_seq = _uniform(n)

    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"[Result] surrogate = {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next run (facilitates incremental improvement)
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
