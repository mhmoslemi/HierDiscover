def _adam_smoothing(start_seq, n, deadline, beta_schedule, iter_per_beta=2000, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):

    v = np.log(start_seq + 1e-30)

    best_seq = start_seq
    best_val = _objective_surrogate(start_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # compute steps as many as possible but limited.
        for t in range(iter_per_beta):
            if time.time() > deadline:
                break
            # compute gradient
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            # compute w for smoothing
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp

            # gradient w.r.t a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]
            # gradient w.r.t v
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_v   # if we maintain m, etc.
            v = v - adam_lr * m_corr etc.

def _objective_surrogate(seq):   # surrogate = 2 * n * max(conv)
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2 * n * np.max(conv))

def _smooth_obj_grad(v, n, beta):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_beta)
    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    # smooth objective.
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    return obj, grad_v

def _adam_opt(start_seq, n, deadline,
              beta_schedule=None,
              iter_per_beta_base=2000,
              lr=0.5,

    if beta_schedule is None:
        beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0,

                         500000.0, 1_000_000.0, 5_000_000.0,

    best_seq = start_seq.copy()

def _exp_gradient_opt(start_seq, n, deadline, max_iters=500000, eta0=5.0):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute convolution
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # compute subgradient (active index)
        g = np.zeros_like(a)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        idx = np.arange(i_start, i_end+1, dtype=int)
        rev_idx = k - idx
        g[idx] = 2.0 * a[rev_idx]
        # update log_a with diminishing steps
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    return best_seq, best_val

    sin2 = np.sin(np.pi * (np.arange(n-half) + 0.5) / (n-half))
    cos_seq = np.concatenate([sin1, sin2])
    seeds.append(("double_sin", cos_seq / cos_seq.sum()))
    # Random Gaussian mixture seeds
    for _ in range(5):
        K = rng.integers(2, 8)
        w = rng.dirichlet(np.ones(K))
        mu = rng.random(K)
        sigma = rng.uniform(0.01, 0.2, K)

"""
Propose a step‑function (non‑negative vector with sum = 1) that minimises
the evaluate_sequence() objective.

Algorithm
---------

1. **Seed generation** – a diverse pool of initial probability vectors:
   uniform, the previous best, many Dirichlet draws, Beta‑shaped vectors,
   a sinusoid, a double‑sinusoid and several random mixtures of Gaussians.

2. **Smooth optimisation** – each seed is refined by Adam on the
   *smooth‐max* surrogate

        F_β(v) = 2·n·[ max(conv(a,a))  (β→∞) ]

   where `v` are log‑softmax variables (a = softmax(v)).
   A continuation on β (1 → 5 → 20 → … → 1e6) gradually tightens the
   approximation to the true max while Adam’s adaptive steps keep the
   optimisation stable.

3. **Exact polishing** – after the smooth phase a few hundred‑thousand
   exponentiated‑gradient (mirror‑descent) iterations are performed on the
   true max‑convolution objective.  The sub‑gradient is taken from the
   currently maximal convolution entry.

4. **Best‑of‑all** – the best vector seen (lowest surrogate value) is kept,
   finally evaluated with the provided ``evaluate_sequence`` and stored in
   the global ``height_sequence_1`` for the next round.

All stages respect a global 1000 s time budget; the routine returns the
best sequence found before the deadline.
"""

import time
import math
import numpy as np

# optional SciPy – not required for the core algorithm
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for a probability vector:
    2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _smooth_obj_and_grad(v: np.ndarray, n: int, beta: float):
    """
    Returns (smooth objective, gradient w.r.t. log‑softmax variables v)
    for the smoothing parameter beta.
    """
    a = _softmax(v)                         # probabilities
    conv = _conv_fft(a, a)                  # length 2n-1
    max_c = np.max(conv)
    shift = max_c

    # log‑sum‑exp smoothing of the max
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = np.sum(exp_beta)
    w = exp_beta / sum_exp                  # weights, sum to 1

    # gradient w.r.t. a (see derivation in the description)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)              # length 2n-1
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    # smooth objective (2·n·smooth_max)
    smooth_max = shift + np.log(sum_exp) / beta
    obj = 2.0 * n * smooth_max
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the smooth objective
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              beta_schedule=None,
              iter_per_beta_base: int = 800,
              lr: float = 0.6,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8):
    """Adam optimiser in log‑softmax space on the smooth‑max surrogate."""
    if beta_schedule is None:
        beta_schedule = [
            1.0, 5.0, 20.0, 80.0, 200.0,
            500.0, 1000.0, 2000.0, 5000.0,
            20000.0, 50000.0,
            100000.0, 200000.0,
            500000.0, 1_000_000.0,
        ]

    v = np.log(start_seq + 1e-30)
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(start_seq)

    m = np.zeros_like(v)          # first moment
    v2 = np.zeros_like(v)         # second moment
    t = 0                         # iteration counter for bias‑correction

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # increase the number of inner steps for larger beta
        iters = int(iter_per_beta_base * (math.log10(beta + 1) + 1))

        for _ in range(iters):
            if time.time() > deadline:
                break

            t += 1
            obj, grad = _smooth_obj_and_grad(v, n, beta)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v2 / (1.0 - beta2 ** t)

            step = lr / (np.sqrt(v_hat) + eps)
            v = v - step * m_hat

            # occasional exact evaluation
            if t % 1000 == 0:
                a = _softmax(v)
                val = _objective_surrogate(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()

    # final evaluation
    a = _softmax(v)
    val = _objective_surrogate(a)
    if val < best_val:
        best_val, best_seq = val, a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) polishing on the true max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient from the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple seed families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = (i ** (a - 1.0)) * ((n - 1 - i) ** (b - 1.0))
    shape = np.maximum(shape, 0.0)
    if shape.sum() == 0.0:
        return _uniform(n)
    return shape / shape.sum()


def _sinusoid(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    seq = np.sin(np.pi * (i + 0.5) / n)
    seq = np.maximum(seq, 0.0)
    return seq / seq.sum()


def _double_sin(n: int) -> np.ndarray:
    half = n // 2
    i1 = np.arange(half, dtype=np.float64)
    i2 = np.arange(n - half, dtype=np.float64)
    s1 = np.sin(np.pi * (i1 + 0.5) / half)
    s2 = np.sin(np.pi * (i2 + 0.5) / (n - half))
    seq = np.concatenate([s1, s2])
    seq = np.maximum(seq, 0.0)
    return seq / seq.sum()


def _gaussian_mixture(n: int, rng: np.random.Generator,
                      min_comp: int = 2, max_comp: int = 8) -> np.ndarray:
    K = rng.integers(min_comp, max_comp + 1)
    w = rng.dirichlet(np.ones(K))
    mu = rng.random(K)                     # in [0,1]
    sigma = rng.uniform(0.01, 0.2, K)       # reasonable widths
    x = np.arange(n, dtype=np.float64) / (n - 1.0)
    seq = np.zeros(n, dtype=np.float64)
    for k in range(K):
        seq += w[k] * np.exp(-((x - mu[k]) ** 2) / (2.0 * sigma[k] ** 2))
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s == 0.0:
        return _uniform(n)
    return seq / s


def _generate_seeds(n: int, rng: np.random.Generator):
    """Create a diverse pool of initial probability vectors."""
    seeds = []

    # 1. simple shapes
    seeds.append(("uniform", _uniform(n)))
    seeds.append(("sinusoid", _sinusoid(n)))
    seeds.append(("double_sin", _double_sin(n)))

    # 2. previous best, if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(("previous", arr / arr.sum()))

    # 3. Dirichlet draws with a wide range of concentrations
    for conc in (0.001, 0.005, 0.01, 0.05, 0.2, 1.0, 5.0):
        for _ in range(2):
            seeds.append((f"dirichlet_{conc:.3f}", _random_dirichlet(n, conc, rng)))

    # 4. random Beta‑shaped vectors
    for _ in range(4):
        a = rng.exponential(1.0) + 0.1
        b = rng.exponential(1.0) + 0.1
        seeds.append((f"beta_{a:.2f}_{b:.2f}", _beta_shape(n, a, b)))

    # 5. random Gaussian mixtures
    for _ in range(5):
        seeds.append(("gauss_mix", _gaussian_mixture(n, rng)))

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found (list of floats)
    and stores it in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    n = 4000                                         # problem size

    # ------------------------------------------------------------------
    # 1️⃣  Build a pool of seeds
    # ------------------------------------------------------------------
    seed_pool = _generate_seeds(n, rng)

    best_seq = None
    best_val = np.inf

    print("\n--- Evaluating raw seeds (surrogate) ---")
    for name, seq in seed_pool:
        if time.time() > deadline:
            break
        val = _objective_surrogate(seq)
        print(f"  seed {name:16s} surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Optimisation passes – each seed gets a share of the remaining time
    # ------------------------------------------------------------------
    for name, seed_vec in seed_pool:
        if time.time() > deadline:
            break

        # give each seed at most 30 s total (adjust if we are close to deadline)
        remain = deadline - time.time()
        if remain < 8.0:
            break
        allowed = min(30.0, remain * 0.6)          # 60 % of what is left for this seed
        sub_deadline = time.time() + allowed

        print(f"\n--- Optimising seed {name} (budget {allowed:.1f}s) ---")

        # ---- Phase A: Adam on smooth max ----
        seq_smooth, val_smooth = _adam_opt(
            seed_vec,
            n,
            sub_deadline,
            beta_schedule=[
                1.0, 5.0, 20.0, 80.0, 200.0,
                500.0, 1000.0, 2000.0, 5000.0,
                20000.0, 50000.0,
                100000.0, 200000.0, 500000.0, 1_000_000.0
            ],
            iter_per_beta_base=600,
            lr=0.6,
        )
        print(f"[{name}] after Adam surrogate = {val_smooth:.6f}")

        # ---- Phase B: Exact max polishing (exponentiated gradient) ----
        if time.time() < deadline:
            polishing_deadline = deadline - 1.0
            seq_polish, val_polish = _exp_gradient_opt(
                seq_smooth,
                n,
                polishing_deadline,
                max_iters=800_000,
                eta0=5.0,
            )
            print(f"[{name}] after EG surrogate = {val_polish:.6f}")
        else:
            seq_polish, val_polish = seq_smooth, val_smooth

        # ---- Keep the global best ----
        if val_polish < best_val:
            best_val = val_polish
            best_seq = seq_polish.copy()
            print(f"  >>> GLOBAL BEST updated to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print("\n=== Finished ===")
        print(f"Best surrogate  = {best_val:.6f}")
        print(f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
