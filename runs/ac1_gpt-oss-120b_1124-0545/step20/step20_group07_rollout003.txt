Pseudo-code for _adam_optimize(start_seq, n, deadline, beta_schedule, iters_per_beta):

    v = log(start_seq + eps)

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(start_seq)

    for beta in beta_schedule:
        max_iters = iters_per_beta
        m = np.zeros_like(v)
        v2 = np.zeros_like(v)
        t = 0
        while t < max_iters:
            if time.time() > deadline:
                break outer loops
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

            # gradient
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full_grad = _conv_fft(rev_a, w)   # len 3n-2
            grad_a = 2.0 * full_grad[n-1:n-1+n]
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            t += 1
            m = beta1*m + (1-beta1)*grad_v
            v2 = beta2*v2 + (1-beta2)*(grad_v**2)
            m_hat = m / (1 - beta1**t)
            v2_hat = v2 / (1 - beta2**t)
            lr = lr0 / (1 + 0.0001 * t)   # optional decay
            v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-8)

            cur_seq = _softmax(v)
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        if time.time() > deadline:
            break

    return best_seq, best_val

"""
Propose a non‑negative step‑function (list of heights) that minimises the
`evaluate_sequence` objective.

Algorithm
---------

1. **Initialisation** – start from the previous best sequence if available,
   otherwise use a uniform distribution (ℓ₁‑norm = 1).

2. **Continuation Adam optimisation** – minimise a smooth approximation of
   the max‑convolution using Adam in log‑softmax space.
   The smoothness is increased through a schedule of β‑values, each
   performing a few thousand Adam updates.  The exact surrogate
   `2·n·max(conv(a,a))` is evaluated after every iteration and the best
   iterate is kept.

3. **Random restarts** – a handful of Dirichlet‑drawn seeds are optimised
   with the same continuation Adam routine; any improvement replaces the
   current best.

4. **Local stochastic polishing** – tiny mass‑transfer moves between
   random coordinates are performed (hill‑climbing with an adaptive step
   size).  This stage exploits any remaining time budget.

5. **Final evaluation** – the best vector is passed to `evaluate_sequence`,
   saved globally as ``height_sequence_1`` and returned as a plain list.

All numbers are kept non‑negative, summed to 1, and the routine respects a
user‑specified wall‑time budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (inputs are real)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _uniform(n: int) -> np.ndarray:
    """Uniform probability vector of length n (ℓ₁‑norm = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate for a vector with ℓ₁‑norm = 1:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule,
                   max_iters_per_beta: int,
                   rng: np.random.Generator):
    """
    Minimise the smooth max‑convolution using Adam in log‑softmax space.
    Returns the best probability vector (ℓ₁‑norm = 1) encountered and its
    exact surrogate value.
    """
    # initialise in log‑probability space
    eps = 1e-30
    v = np.log(start_seq + eps)
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    beta1, beta2 = 0.9, 0.999
    lr0 = 0.5           # base learning rate
    eps_adam = 1e-8

    for idx, beta in enumerate(beta_schedule):
        if time.time() > deadline:
            break

        m = np.zeros_like(v)
        v2 = np.zeros_like(v)
        t = 0

        # allocate a modest number of iterations; break if time runs out
        for _ in range(max_iters_per_beta):
            if time.time() > deadline:
                break

            # current probability vector and its auto‑convolution
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max via log‑sum‑exp (used only for the gradient)
            shift = max_c
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp                      # weights for each conv entry

            # gradient w.r.t. a (see discussion in the analysis)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)                # chain‑rule through softmax

            # optional clipping to keep updates stable
            grad_v = np.clip(grad_v, -10.0, 10.0)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            lr = lr0 / (1.0 + 0.001 * t)               # mild decay
            v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

            # evaluate exact surrogate and keep the best
            cur_seq = _softmax(v)
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        # status update after each β
        print(f"  β={beta:.0e}, best surrogate = {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic polishing – tiny mass swaps between coordinates
# ----------------------------------------------------------------------
def _stochastic_polish(seq: np.ndarray,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator,
                       cur_val: float):
    """
    Hill‑climbing moves that transfer a small fraction of mass from one
    coordinate to another.  The step size adapts to recent successes.
    Returns an improved (or unchanged) vector and its surrogate value.
    """
    best_seq = seq.copy()
    best_val = cur_val
    step_frac = 0.02          # initial fraction of the source mass to move
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose moving a random fraction of a[i] towards a[j]
        delta = step_frac * seq[i] * rng.uniform(0.5, 1.5)
        if delta > seq[i]:
            delta = seq[i]

        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective_surrogate(new_seq)

        if new_val < cur_val:
            seq = new_seq
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = seq.copy()
                print(f"  polish iter {it:6d}: new best surrogate = {best_val:.6f}")

            # slightly increase aggressiveness after a success
            step_frac = min(0.2, step_frac * 1.02)
        else:
            # slowly decay step size on failure
            step_frac = max(1e-5, step_frac * 0.9995)

        it += 1
        if it % 5000 == 0:
            print(f"  polish iter {it:6d}: current surrogate = {cur_val:.6f}")

    return best_seq, best_val
