"""
Improved step‑function search.

The algorithm consists of four stages:

1. **Rich parametric sweep** – a large collection of analytic windows
   (sine, Gaussian, Kaiser, Tukey, beta, power‑law, double‑exponential,
   random Dirichlet / low‑frequency curves) is sampled with fine grids.
   Every candidate is ℓ₁‑normalised (∑a = 1) and scored by the cheap surrogate

        2·n·max(conv(a,a))

   (exact when the sum equals 1).  The best CANDIDATES are kept.

2. **Exponentiated‑gradient (mirror descent) refinement** – for each of the
   top candidates we run many EG steps on the exact non‑smooth objective.
   The update lives in log‑softmax space, thus the simplex and non‑negativity
   constraints are automatically satisfied.

3. **Continuation L‑BFGS on a smooth max** – starting from the best EG
   result we minimise a smooth approximation of the max via the log‑sum‑exp
   trick, increasing the smoothing parameter β gradually.  To avoid Python
   closures the objective and a time‑limit callback are defined at module
   scope and read global variables set just before each optimisation stage.

4. **Local stochastic polishing** – a cheap hill‑climber repeatedly moves a
   small fraction of mass between two random coordinates, accepting improving
   moves and slowly annealing the step size.

The whole pipeline respects a 1000 s wall‑time limit; the best sequence is
stored in the global ``height_sequence_1`` variable and also returned as a
plain list.
"""

import time

import operator
import numpy as np

# Optional SciPy import (used only for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Global variables used by the L‑BFGS callbacks (to avoid closures)
# ----------------------------------------------------------------------
_lbfgs_n = None          # length of the sequence
_lbfgs_beta = None       # current smoothing parameter
_lbfgs_deadline = None   # time limit for L‑BFGS

def _lbfgs_fun(v):
    """Objective and gradient for smooth max convolution (global variables used)."""
    a = _softmax(v)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    shift = max_c
    beta = _lbfgs_beta
    # smooth max via log‑sum‑exp
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * _lbfgs_n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                     # weights for each conv entry
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length 3n‑2
    grad_a = 2.0 * full[_lbfgs_n - 1:_lbfgs_n - 1 + _lbfgs_n]

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_callback(_):
    """Interrupt L‑BFGS when the global deadline is reached."""
    if time.time() > _lbfgs_deadline:
        raise KeyboardInterrupt()


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a vector with ℓ₁‑norm = 1:
    2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = _uniform(n)
    return a


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = _uniform(n)
    return a


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = _uniform(n)
    return a


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    t = beta * np.sqrt(1 - y * y)
    a = np.i0(t)               # Modified Bessel function I0
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = _uniform(n)
    return a


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Cosine‑tapered window.  α=0 ⇒ uniform, α=1 ⇒ Hann."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        a /= a.sum()
        return a
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1))
                                     - 2 / alpha + 1)))
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = _uniform(n)
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution (with support on {1,…,n})."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = _uniform(n)
    return a


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = _uniform(n)
    return a


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    a /= s
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    return a


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = _uniform(n)
    return a


def _double_exponential(n: int, lam: float) -> np.ndarray:
    """Symmetric sum of two exponentials decaying from both ends."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = _uniform(n)
    return a


def _perturb(seq: np.ndarray,
             rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add small Gaussian noise, clip, re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·a + (1‑α)·b (vectors already ℓ₁‑normalised)."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s <= 0.0:
        return a.copy()
    return mixed / s

def _maybe_insert(heap: list, val: float, vec: np.ndarray, max_size: int) -> None:
    """Insert (val, vec) into ``heap`` if it belongs to the best ``max_size``."""
    if len(heap) < max_size:
        heap.append((val, vec))
        heap.sort(key=operator.itemgetter(0))
    else:
        if val < heap[-1][0]:
            heap[-1] = (val, vec)
            heap.sort(key=operator.itemgetter(0))


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimisation on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 5000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Variables live in log‑softmax space so the simplex constraint is respected.
    """
    global _lbfgs_n, _lbfgs_beta, _lbfgs_deadline
    if _opt is None:
        # SciPy not available – fall back to the input vector unchanged.
        return start_vec

    if beta_schedule is None:
        beta_schedule = [1, 5, 20, 80, 200, 500, 1000,
                         2000, 5000, 10000, 20000, 50000,
                         100000, 200000, 500000, 1000000]

    _lbfgs_n = n
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline - 2.0:
            break
        _lbfgs_beta = float(beta)
        _lbfgs_deadline = deadline

        try:
            res = _opt.minimize(fun=_lbfgs_fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=_lbfgs_callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta}")
            break

    return _softmax(v)

            _maybe_insert(best_heap, _objective_surrogate(vec), vec, top_k)
    # random beta pairs
    for _ in range(30):
        a_ = 10 ** rng.uniform(-2, 1)   # between 0.01 and 10
        b_ = 10 ** rng.uniform(-2, 1)
        vec = _beta_shape(n, a_, b_)
        _maybe_insert(best_heap, _objective_surrogate(vec), vec, top_k)

    # Power law
    for p in np.linspace(0.0, 4.0, 30):
        vec = _power(n, p)
        _maybe_insert(best_heap, _objective_surrogate(vec), vec, top_k)

    # Double exponential (new family)
    for lam in np.logspace(-3, 0.7, 30):   # about 0.001 .. 5
        vec = _double_exponential(n, lam)
        _maybe_insert(best_heap, _objective_surrogate(vec), vec, top_k)

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5, 6):
        for _ in range(6):
            vec = _lowfreq_curve(n, nf, rng)
            _maybe_insert(best_heap, _objective_surrogate(vec), vec, top_k)

    # Random Dirichlet draws
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            vec = _random_dirichlet(n, conc, rng)
            _maybe_insert(best_heap, _objective_surrogate(vec), vec, top_k)

    # Random exponential draws
    for _ in range(12):
        vec = _random_exponential(n, rng)
        _maybe_insert(best_heap, _objective_surrogate(vec), vec, top_k)

    # Perturbations of the previous best (if any)
    if isinstance(prev, (list, tuple)) and len(prev) == n:
        prev_arr = np.asarray(prev, dtype=np.float64)
        if prev_arr.sum() > 0.0:
            base = prev_arr.copy()
            base = base / base.sum()
            _maybe_insert(best_heap, _objective_surrogate(base), base, top_k)
            for sigma in (0.002, 0.004, 0.006):
                for _ in range(3):
                    pert = _perturb(base, rng, sigma=sigma)
                    _maybe_insert(best_heap, _objective_surrogate(pert), pert, top_k)

    # ------------------------------------------------------------------
    # Keep the top candidates
    # ------------------------------------------------------------------
    best_heap.sort(key=operator.itemgetter(0))
    top_seqs = [vec for _, vec in best_heap[:top_k]]
    best_val, best_seq = best_heap[0]
    print("\nTop candidates after sweep (surrogate):")
    for i, (val, _) in enumerate(best_heap[:top_k], 1):
        print(f"  #{i}: {val:.6f}")

    # ------------------------------------------------------------------
    # 2. Exponentiated‑gradient refinement (lots of steps)
    # ------------------------------------------------------------------
    remaining_time = deadline - time.time()
    # Reserve ~40 % of the remaining budget for EG, 30 % for L‑BFGS,
    # the rest for polishing.
    eg_budget = 0.40 * remaining_time
    eg_deadline = time.time() + eg_budget
    print(f"\n--- EG refinement (budget ≈ {eg_budget:.1f}s) ---")
    # Only a handful of the absolute best candidates are worth a long run.
    for idx, seq in enumerate(top_seqs[:5]):
        if time.time() > eg_deadline:
            break
        # Determine a generous iteration bound respecting the remaining time.
        time_left = eg_deadline - time.time()
        # Rough estimate: one EG step ≈ 0.00025 s for n=4000 (empirical).
        # Guard against division by zero.
        est_step = 0.0003
        max_iters = max(1, int(time_left / est_step))
        # Cap at an absolute maximum to avoid runaway loops.
        max_iters = min(max_iters, 1_200_000)

        seq_opt, val_opt = _exp_gradient_opt(seq.copy(),
                                             n,
                                             eg_deadline,
                                             max_iters=max_iters,
                                             eta0=5.0)
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print(f"  Candidate {idx+1}: new best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS smoothing (smooth max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0 and _opt is not None:
        print("\n--- L‑BFGS continuation smoothing ---")
        start_vec = np.maximum(best_seq, 1e-12)
        start_vec = start_vec / start_vec.sum()
        # The schedule is deliberately long but we respect the remaining time.
        seq_lbfgs = _lbfgs_smooth_optimize(start_vec,
                                           n,
                                           deadline,
                                           beta_schedule=[1, 5, 20, 80,
                                                          200, 500, 1000,
                                                          2000, 5000, 10000,
                                                          20000, 50000,
                                                          100000, 200000,
                                                          500000, 1000000],
                                           max_iter_per_beta=8000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print(f"  L‑BFGS improvement: surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Local stochastic polishing (tiny mass moves)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        print("\n--- Local stochastic polishing ---")
        seq = best_seq.copy()
        cur_val = best_val
        step_frac = 0.05
        it = 0
        while time.time() < deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = step_frac * seq[i] * rng.uniform(0.5, 1.5)
            if delta > seq[i]:
                delta = seq[i]
            new_seq = seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            val_new = _objective_surrogate(new_seq)
            if val_new < cur_val:
                seq = new_seq
                cur_val = val_new
                if cur_val < best_val:
                    best_val, best_seq = cur_val, seq.copy()
                    print(f"  iter {it:6d}: new best surrogate = {best_val:.6f}")
                step_frac = min(0.2, step_frac * 1.02)
            else:
                step_frac = max(1e-4, step_frac * 0.9995)
            it += 1
            if it % 5000 == 0:
                print(f"  iter {it:6d}: cur surrogate = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final true evaluation and bookkeeping
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
          f"true evaluate_sequence = {true_val}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
