def exp_gradient_opt(start_seq, n, deadline, max_iters=500000, eta0=5.0):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while it < max_iters and time.time() < deadline:
        it +=1
        conv = _conv_fft(a,a)
        k = np.argmax(conv)
        # compute subgradient g
        g = np.zeros_like(a)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        idx = np.arange(i_start, i_end+1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val
```

"""
Propose a step‑function that minimises

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2)

The algorithm is a lightweight evolution of the previous pipeline
with a few new ideas:

*   a richer seed pool – we add edge‑heavy analytic windows and many
    random low‑frequency cosine curves.
*   a quick surrogate objective 2·n·max(conv) (the denominator cancels
    for any normalisation) evaluated via an FFT based convolution.
*   aggressive convex mixing of the top‑k seeds using ternary search.
*   a long exponentiated‑gradient (mirror‑descent) polish on the exact
    max‑convolution, with a square‑root decay stepsize.
*   a final ternary line‑search that mixes the current best with a
    uniform distribution and with its mirror image – this often reduces
    the central peak further.
*   a final short EG polish after the line‑search.

All work is done under the global time budget (default 1000 s).  The
best sequence found is stored in ``height_sequence_1`` for the next
run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective – exact for any normalisation
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)^2)."""
    n = len(seq)
    sum_a = np.sum(seq)
    if sum_a == 0.0:
        return float("inf")
    conv = _conv_fft(seq, seq)
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simple normalisation helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex blend α·a + (1‑α)·b and renormalise to unit sum."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s == 0.0:
        return a.copy()
    return mixed / s


def _best_mixed_pair(a: np.ndarray, b: np.ndarray,
                     deadline: float, steps: int = 30):
    """Ternary search for the best convex combination of a and b."""
    lo, hi = 0.0, 1.0
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)
    for _ in range(steps):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        seq1 = _mix_shapes(a, b, m1)
        seq2 = _mix_shapes(a, b, m2)
        val1 = _objective_surrogate(seq1)
        val2 = _objective_surrogate(seq2)
        if val1 < val2:
            hi = m2
            if val1 < best_val:
                best_val, best_seq = val1, seq1
        else:
            lo = m1
            if val2 < best_val:
                best_val, best_seq = val2, seq2
    # final refinement at the middle point
    mid = (lo + hi) / 2.0
    cand = _mix_shapes(a, b, mid)
    cand_val = _objective_surrogate(cand)
    if cand_val < best_val:
        best_val, best_seq = cand_val, cand
    return best_seq, best_val


# ----------------------------------------------------------------------
# Analytic families (all return a non‑negative vector, not normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _exp_decay(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-gamma * i)
    return a / a.sum()


def _logistic_decay(n: int, steep: float, shift: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(steep * (x - shift)))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of a few low‑frequency cosine waves, clipped to non‑negative."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _edge_heavy(n: int, power: float = 1.0) -> np.ndarray:
    """Mass concentrated at the two edges, controlled by a power law."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** (-power) + (n - i) ** (-power))
    return a / a.sum()


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.003) -> np.ndarray:
    """Add Gaussian noise, clip to non‑negative and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    if noisy.sum() == 0.0:
        return seq.copy()
    return noisy / noisy.sum()


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient optimiser (mirror descent on the true max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx
        g[idx] = 2.0 * a[rev_idx]

        # decaying stepsize
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

    stores the best found sequence in the global variable `height_sequence_1`,
    and returns it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Determine problem size – reuse the size of a previously stored solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            n = prev_arr.size
        except Exception:
            prev_arr = None
            n = 4000
    else:
        prev_arr = None
        n = 4000

    # ------------------------------------------------------------------
    # 1️⃣  Build a diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # classic analytic windows (already present in the historic run)
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in np.linspace(0.2, 3.0, 12):
        seeds.append(_sine_window(n, gamma))
    for sigma in np.linspace(0.09, 0.26, 14):
        seeds.append(_gaussian(n, sigma))
    for gamma in np.linspace(0.1, 8.0, 16):
        seeds.append(_exp_decay(n, gamma))
    for steep in np.linspace(0.5, 30.0, 10):
        for shift in np.linspace(0.1, 0.9, 9):
            seeds.append(_logistic_decay(n, steep, shift))
    alphas = np.linspace(0.2, 5.0, 7)
    betas = np.linspace(0.2, 5.0, 7)
    for a_ in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a_, b_))

    # new families that concentrate on the edges
    for power in (0.5, 1.0, 1.5, 2.0):
        seeds.append(_edge_heavy(n, power))

    # random Dirichlet draws with many concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random exponential draws
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # low‑frequency random cosine curves
    for nf in (2, 3, 4, 5):
        for _ in range(8):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # perturbations of a previously found best (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        seeds.append(prev_arr / prev_arr.sum())
        for _ in range(5):
            seeds.append(_perturb(prev_arr, rng, sigma=0.003))

    # ------------------------------------------------------------------
    # 2️⃣  Fast surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(15, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seed surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Pairwise ternary mixing of the elite seeds
    # ------------------------------------------------------------------
    mix_deadline = deadline - 2.0
    mixed = []

    for i in range(len(top_seeds)):
        for j in range(i + 1, len(top_seeds)):
            if time.time() > mix_deadline:
                break
            seq_mix, val_mix = _best_mixed_pair(
                top_seeds[i], top_seeds[j], mix_deadline, steps=25
            )
            mixed.append((seq_mix, val_mix))

    # include a few random triples (very cheap)
    for _ in range(6):
        if time.time() > mix_deadline:
            break
        a, b, c = rng.choice(top_seeds, size=3, replace=False)
        w = rng.dirichlet([1.0, 1.0, 1.0])
        mix = w[0] * a + w[1] * b + w[2] * c
        mix /= mix.sum()
        mixed.append((mix, _objective_surrogate(mix)))

    # ------------------------------------------------------------------
    # 4️⃣  Assemble all candidates and pick the current best
    # ------------------------------------------------------------------
    all_cands = evaluated[:top_k] + mixed
    all_cands.sort(key=lambda x: x[1])
    best_seq = all_cands[0][0].copy()
    best_val = all_cands[0][1]
    print(f"\n[Init] best surrogate after mixing = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Long exponentiated‑gradient polish (exact max)
    # ------------------------------------------------------------------
    # we give the remaining time a generous slice for this step
    refine_deadline = deadline - 2.0
    if time.time() < refine_deadline:
        print("\n--- EG polishing (main) ---")
        seq_eg, val_eg = _exp_gradient_opt(
            best_seq, n, refine_deadline,
            max_iters=2_000_000,
            eta0=5.0
        )
        if val_eg < best_val:
            best_seq, best_val = seq_eg, val_eg
            print(f"[EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final line‑search mixing with uniform and with the mirror image
    # ------------------------------------------------------------------
    def _line_search_mix(base: np.ndarray, other: np.ndarray,
                         deadline_local: float) -> tuple[np.ndarray, float]:
        """Ternary search on α ∈ [0,1] for the best blend α·base+(1-α)·other."""
        lo, hi = 0.0, 1.0
        best_seq_local = base.copy()
        best_val_local = _objective_surrogate(base)
        for _ in range(30):
            if time.time() > deadline_local:
                break
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            s1 = _mix_shapes(base, other, m1)
            v1 = _objective_surrogate(s1)
            s2 = _mix_shapes(base, other, m2)
            v2 = _objective_surrogate(s2)
            if v1 < v2:
                hi = m2
                if v1 < best_val_local:
                    best_val_local, best_seq_local = v1, s1
            else:
                lo = m1
                if v2 < best_val_local:
                    best_val_local, best_seq_local = v2, s2
        # final refinement at midpoint
        mid = (lo + hi) / 2.0
        s_mid = _mix_shapes(base, other, mid)
        v_mid = _objective_surrogate(s_mid)
        if v_mid < best_val_local:
            best_val_local, best_seq_local = v_mid, s_mid
        return best_seq_local, best_val_local

    # mix with uniform
    if time.time() < deadline:
        print("\n--- Mixing with uniform ---")
        mix_dead = deadline - 1.5
        cand, cand_val = _line_search_mix(best_seq, _uniform(n), mix_dead)
        if cand_val < best_val:
            best_seq, best_val = cand, cand_val
            print(f"[Mix‑Uniform] improved to {best_val:.6f}")

    # mix with mirror image (reverse order)
    if time.time() < deadline:
        print("\n--- Mixing with mirror image ---")
        rev = best_seq[::-1].copy()
        mix_dead = deadline - 1.5
        cand, cand_val = _line_search_mix(best_seq, rev, mix_dead)
        if cand_val < best_val:
            best_seq, best_val = cand, cand_val
            print(f"[Mix‑Mirror] improved to {best_val:.6f}")

    # one last short EG polish after the mix‑steps
    if time.time() < deadline:
        print("\n--- Final short EG polish ---")
        final_dead = deadline - 0.5
        seq_final, val_final = _exp_gradient_opt(
            best_seq, n, final_dead,
            max_iters=300_000,
            eta0=4.0
        )
        if val_final < best_val:
            best_seq, best_val = seq_final, val_final
            print(f"[Final EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Exact evaluation (real score) and store the answer
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                 # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
