"""
Propose a candidate step‐function with a low self‑convolution peak.

The algorithm consists of three stages:

1. **Parametric seeding** –  a symmetric power‑law family
       a_i ∝ (i+1+δ)^{-α} + (n‑i+δ)^{-α}
   is sampled (grid + random) and the best seed is kept.
   The objective 2·n·max(conv(a,a)) (the value returned by
   ``evaluate_sequence`` for a normalised sequence) is computed with an
   FFT‑based convolution.

2. **Exact projected sub‑gradient descent** – the gradient of the
   max‑entry of the self‑convolution (a true sub‑gradient) is used.
   After each step the vector is projected onto the probability simplex
   {x≥0, Σx=1}.  An adaptive step size with back‑tracking is employed,
   together with occasional random mass swaps to escape plateaus.

3. **Short random restarts** – while time permits, new seeds from the
   power‑law family are generated, a brief gradient run is performed and
   the best result is retained.

All work is done with NumPy only; the total running time is limited to
the supplied ``budget_s`` (default 1000 s).  The returned sequence is
non‑negative, sums to 1 and therefore has the same objective value as
``evaluate_sequence`` would compute.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """
    Full linear self‑convolution using a real FFT.
    Returns a vector of length 2·len(seq)‑1.
    """
    n = seq.shape[0]
    size = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    # eliminate tiny negatives caused by round‑off
    conv[conv < 0.0] = 0.0
    return conv


def _score_fft(seq: np.ndarray) -> float:
    """
    Objective value 2·n·max(conv(seq, seq)) assuming Σseq = 1.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1} (Duchi et al., 2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    return np.maximum(v - theta, 0.0)


def _sym_power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Symmetric power‑law sequence:
    a_i ∝ (i+1+δ)^(-α) + (n-i+δ)^(-α)   ,   i = 0,…,n‑1.
    The result is normalised to Σa = 1.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset          # i+1+δ
    left = idx ** (-alpha)
    right = (n - np.arange(0, n, dtype=float) + offset) ** (-alpha)
    a = left + right
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a


def _exact_subgradient(seq: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑entry of the self‑convolution.
    Only the index attaining the maximum is used (exact sub‑gradient).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    k = int(conv.argmax())                     # index of the current max
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    i_range = np.arange(lo, hi + 1, dtype=int)
    # contribution of a_i to the max entry = 2 * a_{k-i}
    grad = np.zeros_like(seq)
    grad[i_range] = 2.0 * seq[k - i_range]
    return grad


def _run_exact_gradient(
    seq: np.ndarray,
    time_budget: float,
    *,
    step0: float = 0.5,
    decay: float = 0.9999,
    min_step: float = 1e-10,
    max_iters: int = 10 ** 9,
) -> tuple[np.ndarray, float]:
    """
    Projected (exact) sub‑gradient descent with simple back‑tracking.

    Returns the best sequence encountered and its score.
    """
    start = time.time()
    n = seq.shape[0]

    cur_score = _score_fft(seq)
    best_seq = seq.copy()
    best_score = cur_score

    step = step0
    it = 0

    while it < max_iters and (time.time() - start) < time_budget:
        it += 1

        grad = _exact_subgradient(seq)

        # try a step; if it does not improve, halve the step until it does
        accepted = False
        for _ in range(10):          # at most 10 halvings
            new_seq = _project_simplex(seq - step * grad)
            new_score = _score_fft(new_seq)
            if new_score < cur_score - 1e-14:
                accepted = True
                break
            step *= 0.5
            if step < min_step:
                break

        if accepted:
            seq = new_seq
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_seq = seq.copy()
                best_score = cur_score
                print(f"[improve] it={it}, score={best_score:.6f}")

            # mild increase after a successful step
            step = max(step / decay, min_step)
        else:
            # no improvement – perform a tiny random exchange to escape
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(seq[i] * 0.1, 1e-4)
            seq[i] -= eps
            seq[j] += eps
            seq = _project_simplex(seq)
            cur_score = _score_fft(seq)
            step = max(step * 0.5, min_step)

        # occasional random mass swap (helps with flat regions)
        if it % 400 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(seq[i] * 0.2, 1e-3)
            seq[i] -= eps
            seq[j] += eps
            seq = _project_simplex(seq)

    return best_seq, best_score
