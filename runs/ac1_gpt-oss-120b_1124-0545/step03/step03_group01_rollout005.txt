"""
Search for step function minimizing 2*n*max(conv(a,a))/(sum(a)^2).

Algorithm:
1. Generate diverse seeds (uniform, geometric, power‑law, sine, Hann, Tukey, random).
2. For each length (4000, 8000, …) run a short Adam optimisation on a
   smoothed version of the max‑convolution.
3. Refine the best result with exact sub‑gradient mirror descent.
4. Perform a cheap stochastic local search: repeatedly draw
   Dirichlet samples centred at the current best (parameter λ controls
   concentration) and keep any improvement.
All phases stop as soon as the global time budget is exhausted.
The best sequence (scaled to sum = 1) is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT (full length).
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Seed families (all normalised to sum 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float = 0.5) -> np.ndarray:
    """Tukey window (cosine‑tapered). 0 ≤ alpha ≤ 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.empty_like(i)
    edge = alpha * (n - 1) / 2.0
    for k in range(n):
        if i[k] < edge:
            w[k] = 0.5 * (1 + np.cos(np.pi * ( (2 * i[k]) / (alpha * (n - 1)) - 1 )))
        elif i[k] <= (n - 1) - edge:
            w[k] = 1.0
        else:
            w[k] = 0.5 * (1 + np.cos(np.pi * ( (2 * i[k]) / (alpha * (n - 1)) - (2 / alpha) + 1 )))
    return w / w.sum()


def _random_softmax(rng: np.random.Generator, n: int) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------


def _objective_from_a(a: np.ndarray) -> float:
    """
    Compute 2 * n * max(conv(a,a)) for a already normalised to sum=1.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed max
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smoothed max of the auto‑convolution.
    Returns the best sequence (as a numpy array) encountered and its objective.
    """
    v = np.log(seed + 1e-30)               # initialise log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0               # smoothing parameter
    beta_mul = 1.07          # accelerate smoothing

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)

        # exact objective (fast) – sum(a) == 1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:4d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient w.r.t a (∇_a Σ w_k·conv_k) -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- Jacobian of softmax (grad w.r.t. v) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent on simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 15000) -> tuple[np.ndarray, float]:
    """
    Refine a sequence with exact sub‑gradient of max(conv) using an
    exponentiated‑gradient (mirror‑descent) update.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # Sub‑gradient of max(conv) w.r.t a (see derivation)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Exponentiated‑gradient (mirror‑descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)               # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic local search around a current best (Dirichlet perturbation)
# ----------------------------------------------------------------------


def _local_dirichlet_search(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_outer_iters: int = 200,
                            samples_per_iter: int = 32,
                            lam_init: float = 5.0,
                            lam_max: float = 200.0,
                            lam_factor: float = 1.15) -> tuple[np.ndarray, float]:
    """
    Perform cheap stochastic hill‑climbing by drawing Dirichlet samples
    centred at the current best.  The concentration parameter λ is
    gradually increased so the search becomes more local over time.
    """
    best_seq = start_seq.copy()
    best_val = _objective_from_a(best_seq)

    lam = lam_init
    eps = 1e-6
    outer = 0

    while time.time() < deadline and outer < max_outer_iters:
        outer += 1
        for _ in range(samples_per_iter):
            # Dirichlet parameters centred on best_seq
            alpha = lam * best_seq + eps
            cand = np.random.default_rng().dirichlet(alpha)
            val = _objective_from_a(cand)
            if val < best_val:
                best_val = val
                best_seq = cand
                print(f"[Dirichlet] n={n} outer={outer:3d} new best={val:.6f}")

        lam = min(lam * lam_factor, lam_max)

    return best_seq, best_val

def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # uniform baseline
    seeds.append(_uniform(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n

    # smooth sine window

    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float = 0.5) -> np.ndarray:
    """Tukey (cosine‑tapered) window, 0 ≤ α ≤ 1."""
    i = np.arange(n, dtype=np.float64)
    w = np.empty_like(i)
    edge = alpha * (n - 1) / 2.0
    for k in range(n):
        if i[k] < edge:
            w[k] = 0.5 * (1 + np.cos(np.pi * ((2 * i[k]) / (alpha * (n - 1)) - 1)))
        elif i[k] <= (n - 1) - edge:
            w[k] = 1.0
        else:
            w[k] = 0.5 * (1 + np.cos(np.pi * ((2 * i[k]) / (alpha * (n - 1))
                                            - (2 / alpha) + 1)))
    return w / w.sum()


def _random_softmax(rng: np.random.Generator, n: int) -> np.ndarray:
    logits = rng.standard_normal(n)
    return _softmax(logits)


# ----------------------------------------------------------------------
# Objective helper
# ----------------------------------------------------------------------


def _objective_from_a(a: np.ndarray) -> float:
    """Exact objective for a simplex‑normalised vector a."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Adam optimisation on smoothed max
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smoothed max of the auto‑convolution.
    Returns the best sequence encountered and its exact objective.
    """
    v = np.log(seed + 1e-30)               # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0          # smoothing parameter
    beta_mul = 1.07     # growth factor

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)

        # exact objective (fast, sum(a)=1)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:4d} best={cur_val:.6f}")

        # smoothed‑max weights
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient w.r.t. a of Σ w_k·conv_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Jacobian of softmax (grad w.r.t. v)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent on simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 12000) -> tuple[np.ndarray, float]:
    """
    Refine a sequence using the true sub‑gradient of max(conv) with an
    exponentiated‑gradient / mirror‑descent update.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_from_a(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic Dirichlet local search
# ----------------------------------------------------------------------


def _local_dirichlet_search(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            rng: np.random.Generator,
                            max_outer_iters: int = 150,
                            samples_per_iter: int = 48,
                            lam_init: float = 5.0,
                            lam_max: float = 300.0,
                            lam_factor: float = 1.20) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing by drawing Dirichlet samples centred at the
    current best.  The concentration λ grows, making the search more
    local over time.
    """
    best_seq = start_seq.copy()
    best_val = _objective_from_a(best_seq)

    lam = lam_init
    eps = 1e-6
    outer = 0

    while time.time() < deadline and outer < max_outer_iters:
        outer += 1
        for _ in range(samples_per_iter):
            alpha = lam * best_seq + eps
            cand = rng.dirichlet(alpha)
            val = _objective_from_a(cand)
            if val < best_val:
                best_val = val
                best_seq = cand
                print(f"[Dirichlet] n={n} outer={outer:3d} new best={val:.6f}")

        lam = min(lam * lam_factor, lam_max)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generator (includes extra windows)
# ----------------------------------------------------------------------


def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # uniform baseline
    seeds.append(_uniform(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # geometric × power mix
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic smooth windows
    seeds.append(_sine_window(n))
    seeds.append(_hann_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))

    # random soft‑max vectors
    for _ in range(7):
        seeds.append(_random_softmax(rng, n))

    # reuse previously discovered sequence if length matches
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the allotted time budget.
    Returns the best sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # Previous construction (if any) is made available through a global.
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Lengths to explore: start with 4000, then 8000, finally 12000 if time permits.
    lengths = [4000, 8000, 12000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous, rng)

        # --------------------------------------------------------------
        # Quick scan of raw seeds
        # --------------------------------------------------------------
        for s in seeds:
            val = _objective_from_a(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n} new global best={val:.6f}")

        # --------------------------------------------------------------
        # Phase 1 – Adam on smoothed max (moderate number of iterations)
        # --------------------------------------------------------------
        for idx, seed_vec in enumerate(seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=1800)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall] n={n} new best after Adam={val:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient refinement
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_ref, val_ref = _subgradient_refine(global_best_seq, n, deadline,
                                                   max_iters=12000)
            if seq_ref is not None and val_ref < global_best_val:
                global_best_val = val_ref
                global_best_seq = seq_ref.copy()
                print(f"[overall] after sub‑grad n={n} best={val_ref:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – Stochastic Dirichlet local search (cheap hill‑climb)
        # --------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_loc, val_loc = _local_dirichlet_search(
                global_best_seq, n, deadline, rng,
                max_outer_iters=150, samples_per_iter=48,
                lam_init=5.0, lam_max=300.0, lam_factor=1.20)
            if seq_loc is not None and val_loc < global_best_val:
                global_best_val = val_loc
                global_best_seq = seq_loc.copy()
                print(f"[overall] after Dirichlet search n={n} best={val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be taken)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective_from_a(global_best_seq)
        print("[fallback] using uniform distribution")

    # Store for possible reuse by subsequent runs.
    globals()["height_sequence_1"] = global_best_seq.copy()

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")
    return global_best_seq.tolist()
