def _smooth_max_and_grad(a, epsilon, fft_len):
    # Returns max_conv, gradient (length n)
    # Compute convolution via FFT
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*len(a)-1]
    max_val = float(np.max(conv))
    # Softmax weights
    # shift by max_val for numerical stability
    exp_weights = np.exp((conv - max_val) / epsilon)
    w = exp_weights / np.sum(exp_weights)
    # Compute gradient via correlation a ⋆ w
    # need FFT of w to same length
    fw = np.fft.rfft(w, n=fft_len)
    # correlation via a cross w: corr = irfft(fa * conj(fw))
    corr = np.fft.irfft(fa * np.conj(fw), n=fft_len)
    # Extract correlation shift indices from (len(w)-1) to (len(w)-1 + n -1)
    offset = len(w) - 1  # = 2*n -2
    grad = 2.0 * corr[offset: offset + len(a)]
    return max_val, grad
```

"""
Propose a candidate step‑function sequence that minimises
``evaluate_sequence``.  The search consists of two stages:

1. **Projected Adam optimisation** of a smooth approximation of
   `max(conv(a,a))`.  The smooth max is the soft‑max of the
   convolution entries with a temperature ε that is decreased
   exponentially from the current maximum value down to a tiny
   floor (≈1e‑4).  The gradient of the smooth max is
   `2·(a ⋆ w)` where `w` are the soft‑max weights and `⋆` denotes the
   full correlation.  Adam’s adaptive step‑size together with a
   projection onto the ℓ₁‑simplex (∑a = √(2·n), a_i ≥ 0) yields fast
   descent and escapes from many shallow local minima.

2. **Short simulated‑annealing refinement** starting from the best
   vector found by Adam.  Random mass transfers between two
   positions are proposed; downhill moves are always accepted,
   uphill moves with probability `exp(-(Δ)/T)`.  The temperature T
   decays exponentially from `0.3·current_best` to `1e‑5·current_best`.

Both phases respect the global time budget (≈ 1000 s).  The algorithm
tries a few problem sizes (the length of the historic construction
if available, plus 4000 and 8000) and keeps the globally best
sequence.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , ∑ w = s }."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (covers both convolution and correlation)."""
    needed = 3 * n - 2
    return 1 << (needed - 1).bit_length()


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Return (max value, index) of the full self‑convolution a*a."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    time_limit: float,
    rng: random.Random,
    init_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of `max(conv(a,a))` for a fixed length n.
    Returns the best sequence found and its (non‑smoothed) max‑convolution.
    """
    start = time.time()
    deadline = start + time_limit
    target_sum = math.sqrt(2.0 * n)
    fft_len = _fft_len_for_n(n)

    # initialise a (use supplied init if possible, otherwise random)
    if init_seq is not None:
        if init_seq.size >= n:
            # random subsample without replacement
            idx = rng.sample(range(init_seq.size), n)
            a = init_seq[idx].astype(float).copy()
        else:
            a = np.concatenate(
                [init_seq.astype(float), np.random.exponential(scale=1.0, size=n - init_seq.size)]
            )
    else:
        a = np.random.exponential(scale=1.0, size=n)

    a = a / a.sum() * target_sum

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = target_sum * 0.02  # about 2 % of the ℓ₁‑mass as a base step
    t = 0

    # bookkeeping
    best_seq = a.copy()
    best_val, _ = _max_convolution(a, fft_len)

    # smoothing parameters
    epsilon_min = 1e-4  # final temperature for the soft‑max

    while time.time() < deadline:
        t += 1

        # ----- compute convolution and its maximum -----------------------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        max_conv = float(conv.max())

        # keep best (non‑smoothed) value seen so far
        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            print(f"[Adam n={n} iter={t}] new best max(conv)={best_val:.6f}")

        # ----- temperature schedule for the soft‑max --------------------
        # exponential decay from max_conv to epsilon_min
        frac = (time.time() - start) / max(time_limit, 1e-12)
        epsilon = max_conv * ((epsilon_min / max_conv) ** frac)

        # ----- compute soft‑max weights ---------------------------------
        # shift by max_conv for numerical stability
        exp_weights = np.exp((conv - max_conv) / epsilon)
        w = exp_weights / exp_weights.sum()

        # ----- gradient of the smooth max (2 * correlation) ------------
        fw = np.fft.rfft(w, n=fft_len)
        corr = np.fft.irfft(fa * np.conj(fw), n=fft_len)
        offset = 2 * n - 2                     # index that corresponds to shift 0
        grad = 2.0 * corr[offset : offset + n]  # length‑n gradient vector

        # ----- Adam update + projection onto the simplex ----------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = base_lr / math.sqrt(t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_candidate = a - step
        a_candidate = _simplex_projection(a_candidate, target_sum)

        a = a_candidate

    # final check (in case the loop ended exactly after an improvement)
    final_val, _ = _max_convolution(a, fft_len)
    if final_val < best_val:
        best_val = final_val
        best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(
    seq: np.ndarray,
    target_sum: float,
    time_limit: float,
    rng: random.Random,
) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing hill‑climber starting from *seq*.
    Returns the best sequence discovered and its max‑convolution.
    """
    n = seq.size
    fft_len = _fft_len_for_n(n)
    a = seq.copy()
    cur_val, _ = _max_convolution(a, fft_len)
    best_seq = a.copy()
    best_val = cur_val

    # temperature schedule (exponential)
    T0 = cur_val * 0.30
    Tmin = cur_val * 1e-5
    start = time.time()
    deadline = start + time_limit

    step_counter = 0
    while time.time() < deadline:
        # linear progress within SA phase
        elapsed = time.time() - start
        frac = elapsed / max(time_limit, 1e-12)
        T = T0 * ((Tmin / T0) ** frac)

        # propose a random mass transfer
        i = rng.randrange(n)
        if a[i] <= 1e-12:
            continue
        j = rng.randrange(n)
        if i == j:
            continue

        delta = a[i] * 0.10 * rng.random()  # move at most 10 % of source mass
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta
        # ℓ₁‑sum is preserved, non‑negativity holds by construction

        cand_val, _ = _max_convolution(a_cand, fft_len)

        # acceptance rule
        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / max(T, 1e-12)):
            a = a_cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = a_cand.copy()
                print(f"[SA step={step_counter}] new best max(conv)={best_val:.6f}")

        step_counter += 1

    return best_seq, best_val

    # final projection (protects against tiny numerical drift)
    n_final = best_overall_seq.size
    target_sum = math.sqrt(2.0 * n_final)
    best_overall_seq = _simplex_projection(best_overall_seq, target_sum)

    print(f"[DONE] best max(conv) = {best_overall_val:.6f} (length = {n_final})")
    return best_overall_seq.tolist()
