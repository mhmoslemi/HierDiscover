"""
propose_candidate
=================

Search for a non‑negative step‑function (probability vector) that minimizes

    2·n·max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear convolution (numpy.convolve).  The algorithm
consists of three stages:

1. **Broad parametric sweep** – a large collection of analytically
   motivated families (beta‑shaped, Tukey, sine‑power, cosine‑power,
   Gaussian, geometric+power, Dirichlet, random soft‑max) is sampled.
   Each candidate is normalised to sum‑to‑one and evaluated with an
   FFT‑based convolution.  The best seed is retained.

2. **Refinement** – starting from the best seed, an Adam optimiser on the
   log‑softmax parametrisation is run.  The loss is a smooth (log‑sum‑exp)
   approximation of the maximum convolution coefficient.  ``beta`` is
   increased geometrically, turning the smooth loss into the true max.
   Progressively better sequences are recorded.

3. **Perturbed restarts** – while time permits, the current best solution
   is perturbed (small Gaussian noise in logit space) and refined again.
   The overall best sequence is returned as a plain Python list of floats.

The routine respects a user‑supplied time budget (default 1000 s) and
always returns a valid non‑negative sequence; the evaluation function
`evaluate_sequence` can be called afterwards to obtain the exact score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real input)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a simplex vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Parametric families (all return a normalized probability vector)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _sin_power(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _cos_power(n: int, gamma: float) -> np.ndarray:
    # shifted cosine to stay non‑negative on [0,n‑1]
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(np.pi * (i + 0.5) / n)) ** gamma
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    # a_i ∝ (i+1)^{alpha} * (n-i)^{beta}
    i = np.arange(n, dtype=np.float64) + 1.0
    a = (i ** alpha) * ((n - i + 1) ** beta)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _tukey_window(n: int, taper: float) -> np.ndarray:
    """
    Tukey window with taper ∈ [0,1].  taper=0 → rectangular,
    taper=1 → Hann window.
    """
    if taper <= 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    if taper >= 1:
        return _sine_window(n)

    a = np.empty(n, dtype=np.float64)
    per = taper / 2.0
    for i in range(n):
        x = i / (n - 1)
        if x < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (x / per - 1)))
        elif x > 1 - per:
            a[i] = 0.5 * (1 + np.cos(np.pi * ((x - 1) / per + 1)))
        else:
            a[i] = 1.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _gaussian_window(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian, sigma expressed in absolute index units."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _vonmises_window(n: int, kappa: float) -> np.ndarray:
    """von‑Mises (circular normal) window."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(kappa * np.cos(2.0 * np.pi * i / n))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Fast evaluation (scale‑invariant – sum assumed to be 1)
# ----------------------------------------------------------------------


def _fast_objective(seq: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(seq,seq))  assuming sum(seq)=1."""
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Optimisation on the simplex (log‑softmax + Adam)
# ----------------------------------------------------------------------


def _refine_sequence(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 8.0,
    beta_mul: float = 1.03,
    lr0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.
    Returns the best sequence (as np.ndarray) and the associated fast objective.
    """
    # logits (stable start)
    v = np.log(start_seq + 1e-30)

    m = np.zeros_like(v)          # first moment
    v2 = np.zeros_like(v)         # second moment

    best_seq = start_seq.copy()
    best_val = _fast_objective(best_seq, n)

    beta = start_beta
    it = 0

    while True:
        it += 1
        if time.time() > deadline:
            break

        # current simplex point
        a = _softmax(v)

        # conv & objective
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ---- smoothed max (log‑sum‑exp) weights ----
        beta *= beta_mul
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient w.r.t. a (smooth loss) ----
        rev_a = a[::-1]                     # reversed a
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # ---- gradient w.r.t. logits (softmax) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam update ----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence minimising ``evaluate_sequence``.
    The routine respects the supplied time budget (default 1000 s) and
    returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # small safety margin

    # Target length – try 8000 first, fall back to 4000 if time is tight
    lengths = [8000, 4000]
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"[main] Optimising length = {n}")

        # --------------------------------------------------------------
        # 1. Build a rich pool of seed sequences (all on the simplex)
        # --------------------------------------------------------------
        seed_seqs = []

        # geometric families (r = 1 - c/n)
        for c in np.linspace(0.5, 6.0, 14):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_geom(n, r))

        # power‑law families
        for p in np.linspace(0.0, 3.0, 13):
            seed_seqs.append(_power(n, p))

        # combined geometric × power families
        for c in np.linspace(0.5, 5.0, 9):
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in np.linspace(0.0, 3.0, 9):
                seed_seqs.append(_geom_power(n, r, p))

        # symmetric geometric
        for c in np.linspace(0.5, 5.0, 9):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_sym_geom(n, r))

        # sine‑power window
        for gamma in np.linspace(0.5, 5.0, 12):
            seed_seqs.append(_sin_power(n, gamma))

        # cosine‑power window
        for gamma in np.linspace(0.5, 5.0, 12):
            seed_seqs.append(_cos_power(n, gamma))

        # beta‑shaped (symmetric and asymmetric)
        for a in [0.2, 0.5, 1.0, 2.0, 4.0]:
            for b in [0.2, 0.5, 1.0, 2.0, 4.0]:
                seed_seqs.append(_beta_shape(n, a, b))

        # Tukey windows (different taper fractions)
        for taper in np.linspace(0.0, 1.0, 11):
            seed_seqs.append(_tukey_window(n, taper))

        # Gaussian windows (different sigma)
        for sigma_frac in np.linspace(0.08, 0.30, 10):
            sigma = sigma_frac * n
            seed_seqs.append(_gaussian_window(n, sigma))

        # von‑Mises windows (circular concentration)
        for kappa in np.linspace(0.0, 8.0, 9):
            seed_seqs.append(_vonmises_window(n, kappa))

        # Dirichlet samples (various concentrations)
        for alpha in [0.1, 0.3, 0.7, 1.5, 3.0]:
            a = np.random.gamma(alpha, 1.0, size=n)
            if a.sum() > 0:
                seed_seqs.append(a / a.sum())

        # Random soft‑max seeds
        for _ in range(7):
            v_rand = np.random.randn(n)
            seed_seqs.append(_softmax(v_rand))

        # Re‑use a previously found construction if available
        prev = globals().get("height_sequence_1")
        if prev is not None:
            try:
                a_prev = np.asarray(prev, dtype=np.float64)
                if a_prev.ndim == 1:
                    if a_prev.shape[0] != n:
                        # Resize by truncation / zero‑padding
                        a_adj = np.zeros(n, dtype=np.float64)
                        m = min(a_prev.shape[0], n)
                        a_adj[:m] = a_prev[:m]
                        a_prev = a_adj
                    if a_prev.sum() > 0:
                        seed_seqs.append(a_prev / a_prev.sum())
            except Exception:
                pass

        # --------------------------------------------------------------
        # 2. Fast scan of all seeds (objective = 2·n·max(conv))
        # --------------------------------------------------------------
        for idx, a in enumerate(seed_seqs):
            cur_val = _fast_objective(a, n)
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[scan] new global best from seed {idx} = {cur_val:.6f}")

        # --------------------------------------------------------------
        # 3. Refinement: Adam on log‑softmax for the current best seed
        # --------------------------------------------------------------
        if global_best_seq is None:
            global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
            print("[fallback] using uniform start")

        refined_seq, refined_val = _refine_sequence(
            global_best_seq,
            n,
            deadline,
            start_beta=8.0,
            beta_mul=1.03,
            lr0=0.9,
        )
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[overall] new global best after refinement (len={n}) = {refined_val:.6f}")

        # --------------------------------------------------------------
        # 4. Perturbed restarts while time permits
        # --------------------------------------------------------------
        while time.time() < deadline:
            # Small Gaussian noise on logits (log‑softmax parameters)
            noisy_logits = np.log(global_best_seq + 1e-30) + 0.025 * np.random.randn(n)
            perturbed_seq = _softmax(noisy_logits)

            seq2, val2 = _refine_sequence(
                perturbed_seq,
                n,
                deadline,
                start_beta=8.0,
                beta_mul=1.03,
                lr0=0.9,
            )
            if val2 < global_best_val:
                global_best_val = val2
                global_best_seq = seq2.copy()
                print(f"[perturb] improved global best to {val2:.6f}")

            # Early exit if we already beat the target
            if global_best_val <= 1.503:
                print("[target] reached target bound, terminating early")
                break

        # --------------------------------------------------------------
        # End of this length – move on if time remains
        # --------------------------------------------------------------

    # ------------------------------------------------------------------
    # Final sanity checks and return
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # Very unlikely, but fall back to a uniform vector of length 4000
        n = 4000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("[fallback] final uniform sequence")

    # Ensure exact normalisation (scale‑invariant, but keep sum=1)
    s = global_best_seq.sum()
    if s > 0:
        global_best_seq = global_best_seq / s

    result = global_best_seq.tolist()
    print(f"[final] best objective (fast) = {global_best_val:.6f}")
    return result
