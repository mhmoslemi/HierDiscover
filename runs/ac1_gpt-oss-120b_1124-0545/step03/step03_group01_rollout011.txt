def project_onto_simplex(v):
    # Like algorithm from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose candidate step heights using multi-scale smoothed max optimisation,
mirror‑descent sub‑gradient refinement and stochastic hill‑climbing.
The search starts from many diverse seeds, runs a fast Adam optimisation
on a small problem (n≈2k), interpolates the best solution to the target
length (4000 or 8000) and then refines it with exact sub‑gradient updates
and a simple random local search.  The routine respects a global time
budget (default 1000 s) and returns the best sequence found.
"""

import time, numpy as np
# helpers: next_pow2, conv_fft, softmax (maybe not needed), project_onto_simplex

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_fft(a,b):
    L=len(a)+len(b)-1
    N=_next_pow2(L)
    fa=np.fft.rfft(a, n=N)
    fb=np.fft.rfft(b, n=N)
    conv=np.fft.irfft(fa*fb, n=N)[:L]
    return conv

def _project_onto_simplex(v):
    # Duchi et al.
    n=v.shape[0]
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u) - 1
    ind=np.arange(1, n+1)
    cond = u - cssv/ind > 0
    if not np.any(cond):
        # all zeros
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w=np.maximum(v - theta, 0)
    return w

def _softmax(v):
    m=v.max()
    e=np.exp(v-m)
    return e/e.sum()
```

def _smooth_optimize(a0, n, deadline, max_iters=5000, beta0=1.0, beta_mul=1.05, eta0=0.5):
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_fft(a,a)
    best_val = 2.0*n*np.max(conv)
    beta = beta0
    for t in range(1,max_iters+1):
        if time.time()>deadline:
            break
        # conv
        conv = _conv_fft(a,a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum==0:
            w = np.ones_like(conv)/len(conv)
        else:
            w = w / w_sum
        # gradient
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n-1:n-1+n]
        # step size
        eta = eta0 / np.sqrt(t)
        a = a - eta * grad
        a = _project_onto_simplex(a)
        # evalu
        conv = _conv_fft(a,a)
        val = 2.0 * n * np.max(conv)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[smooth] n={n} it={t} best={val:.6f}")
        beta *= beta_mul
    return best_a, best_val

def _exact_subgrad_refine(a0, n, deadline, max_iters=20000):
    a = a0.copy()
    best_a = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a,a))
    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        conv = _conv_fft(a,a)
        max_idx = int(np.argmax(conv))
        val = 2.0 * n * conv[max_idx]
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[subgrad] n={n} it={it} best={val:.6f}")
        # subgradient g
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # exponentiated gradient step
        eta = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - eta * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()
    return best_a, best_val

"""
Optimisation of a non‑negative step–function for the objective

    2·n·max(conv(a, a)) / (sum(a)²)

The search works in three stages:

1. **Diverse seeding** – a large collection of normalised seed vectors is
   generated (uniform, sine, triangle, arcsine, geometric, power‑law,
   geometric × power, symmetric geometric, and many random Dirichlet draws).
   The best few seeds (by the fast FFT objective) are kept for deeper work.

2. **Smoothed‑max Adam optimisation** – each retained seed is optimised
   on the smooth approximation *log‑sum‑exp* of the maximum convolution.
   The parameters are kept on the probability simplex via a log‑softmax
   representation and Adam updates.  A slowly increasing smoothing factor
   concentrates the loss onto the true max.

3. **Exact sub‑gradient refinement + stochastic hill climbing** – the
   best sequence from the Adam phase is further polished with a mirror‑
   descent sub‑gradient step (the exact max is used) and finally with a
   simple random local‑search that moves a tiny amount of mass between
   two random positions and accepts improvements.

All stages respect a global time budget (default 1000 s).  The routine
returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level primitives
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (vector → simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """Fast objective for a normalised (sum == 1) vector."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed families (all returned already normalised to sum 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _triangle(n: int) -> np.ndarray:
    """Triangular (tent) shape, peaks in the centre."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(i - mid) / mid
    a[a < 0] = 0
    return a / a.sum()


def _arcsine(n: int) -> np.ndarray:
    """Arcsine‑like distribution (large values near the borders)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n + 1 - i))
    return a / a.sum()


def _random_dirichlet(rng: np.random.Generator, n: int, alpha: float = 0.5) -> np.ndarray:
    a = rng.gamma(alpha, 1.0, size=n)
    if a.sum() == 0.0:
        a = np.ones(n)
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None = None) -> List[np.ndarray]:
    """Collect a diverse set of starting points (already normalised)."""
    rng = np.random.default_rng(123456)
    seeds: List[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_triangle(n))
    seeds.append(_arcsine(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # geometric × power mixtures
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # a bunch of random Dirichlet draws
    for _ in range(30):
        alpha = rng.uniform(0.2, 2.0)
        seeds.append(_random_dirichlet(rng, n, alpha))

    # reuse a previously discovered construction if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

# Adam optimiser on the smoothed‑max loss
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 6000) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max of the autocorrelation.
    Returns the best sequence (as a numpy array) encountered and its exact
    objective value (sum == 1 ⇒ 2·n·max(conv)).
    """
    # initialise log‑softmax parameters
    v = np.log(seed + 1e-30)

    # Adam moments
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # hyper‑parameters
    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # smoothing schedule
    beta_s = 1.0
    beta_mul = 1.03

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                       # stays on the simplex

        # fast objective (sum(a) == 1)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:4d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta_s *= beta_mul
        shift = np.max(conv)                   # safe shift for the exponent
        w = np.exp(beta_s * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t a (∇ₐ Σ w_k·conv_k) -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # ----- convert to gradient w.r.t. v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val

# Exact sub‑gradient refinement (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 20000) -> Tuple[np.ndarray, float]:
    """
    Refine a sequence using the true sub‑gradient of the maximum
    autocorrelation entry.  An exponentiated‑gradient (mirror‑descent)
    update keeps the vector on the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # Sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Mirror‑descent (exponentiated‑gradient) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)               # numerical stability
        a = np.exp(log_a)
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing
# ----------------------------------------------------------------------


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 20000) -> Tuple[np.ndarray, float]:
    """
    Very cheap random local search: move a random fraction of the mass from
    a random index i to a different index j, accept the move if it improves
    the exact objective.  Occasionally the step size is reduced.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    val = 2.0 * n * np.max(conv)

    best_seq = a.copy()
    best_val = val

    # adaptive magnitude (starts at 5 % of the source weight)
    mag_factor = 0.05

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.choice(n, size=2, replace=False)

        if a[i] == 0.0:
            continue
        # random fraction of the source weight (bounded by mag_factor)
        delta = a[i] * mag_factor * rng.random()
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv_new = _conv_fft(a_new, a_new)
        val_new = 2.0 * n * np.max(conv_new)

        if val_new < val:
            a = a_new
            conv = conv_new
            val = val_new
            if val_new < best_val:
                best_val = val_new
                best_seq = a_new.copy()
                print(f"[Hill] n={n} it={it:5d} best={best_val:.6f}")

        # every few thousand steps, shrink the magnitude a bit
        if it % 3000 == 0:
            mag_factor = max(0.001, mag_factor * 0.7)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget and returns the best sequence found as a plain list of
    floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # Re‑use a previously found construction if available
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Work on the length that gave the best results historically (4000)
    # ------------------------------------------------------------------
    n_target = 4000

    # ------------------------------------------------------------------
    # 1) generate seeds and keep the most promising few
    # ------------------------------------------------------------------
    seeds = _make_seeds(n_target, previous)
    print(f"[Info] Generated {len(seeds)} seeds for n={n_target}")

    # quick scan
    seed_vals = []
    for s in seeds:
        val = _objective(s)
        seed_vals.append((val, s))
    seed_vals.sort(key=lambda x: x[0])
    top_seeds = [s for _, s in seed_vals[:5]]   # keep five best seeds

    # ------------------------------------------------------------------
    # 2) Adam optimisation on the smoothed max (more iterations than before)
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        seq, val = _adam_optimize(seed_vec, n_target, deadline, max_iters=8000)
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Overall] after Adam on seed {idx} best={val:.6f}")

    # If nothing succeeded (unlikely), fall back to the very best raw seed
    if global_best_seq is None:
        best_raw_seq = seed_vals[0][1]
        global_best_seq = best_raw_seq.copy()
        global_best_val = _objective(best_raw_seq)
        print("[Fallback] using best raw seed")

    # ------------------------------------------------------------------
    # 3) Exact sub‑gradient refinement
    # ------------------------------------------------------------------
    seq_ref, val_ref = _subgradient_refine(global_best_seq, n_target, deadline,
                                           max_iters=18000)
    if val_ref < global_best_val:
        global_best_val = val_ref
        global_best_seq = seq_ref.copy()
        print(f"[Overall] after sub‑grad refinement best={val_ref:.6f}")

    # ------------------------------------------------------------------
    # 4) Random hill‑climbing local search
    # ------------------------------------------------------------------
    seq_hc, val_hc = _hill_climb(global_best_seq, n_target, deadline,
                                 max_iters=25000)
    if val_hc < global_best_val:
        global_best_val = val_hc
        global_best_seq = seq_hc.copy()
        print(f"[Overall] after hill‑climb best={val_hc:.6f}")

    # ------------------------------------------------------------------
    # 5) (Optional) Try the larger length 8000 if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_big = 8000
        print("\n=== Trying larger length 8000 ===")
        # Interpolate the current best to the larger grid
        x_small = np.linspace(0, 1, n_target)
        x_big = np.linspace(0, 1, n_big)
        interpolated = np.interp(x_big, x_small, global_best_seq)
        interpolated = interpolated / interpolated.sum()
        # a quick Adam pass
        seq_big, val_big = _adam_optimize(interpolated, n_big, deadline,
                                          max_iters=5000)
        if val_big < global_best_val:
            global_best_val = val_big
            global_best_seq = seq_big.copy()
            print(f"[Overall] after 8000‑Adam best={val_big:.6f}")
        # sub‑gradient refinement on the large vector
        seq_big_ref, val_big_ref = _subgradient_refine(global_best_seq, n_big,
                                                      deadline, max_iters=12000)
        if val_big_ref < global_best_val:
            global_best_val = val_big_ref
            global_best_seq = seq_big_ref.copy()
            print(f"[Overall] after 8000‑sub‑grad best={val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – evaluate via the provided function (optional)
    # ------------------------------------------------------------------
    try:
        # The external evaluator caps values, but the vector already respects
        # the required constraints (non‑negative, finite, reasonably sized).
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
