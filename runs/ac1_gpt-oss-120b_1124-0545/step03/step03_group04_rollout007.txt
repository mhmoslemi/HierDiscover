"""
Propose a step‑function (non‑negative sequence) that minimizes the
``evaluate_sequence`` objective.

The strategy has three parts:

1. **Broad parametric sweep** – explore simple families
   (power‑law, exponential, linear) with several lengths, offsets and
   exponents.  The best candidate from this cheap sweep is taken as a
   starting point.

2. **Projected sub‑gradient descent on the exact max‑convolution**
   objective.  For a vector a (∑a=1) the objective is
   ``f(a)=max_k (a∗a)[k]``.  The sub‑gradient of the max at an index
   ``k*`` (where the maximum is attained) is
   ``g_j = 2·a_{k*‑j}`` (zero when the index is out of range).  After a
   gradient step we project back onto the probability simplex
   (non‑negative, sum 1).

3. **Escaping plateaus** – occasional random mass swaps and a very
   small amount of momentum are used to avoid getting stuck.

The loop respects the global time budget (default 1000 s) and returns
the best sequence found (as a plain list of floats).  The code uses only
NumPy and the standard library, runs on a single CPU and is fully
self‑contained.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i ≥ 0 , Σ x_i = 1 }.
    Implements the algorithm of Duchi et al. (ICML 2008).

    Parameters
    ----------
    v : np.ndarray
        Input vector (any real values).

    Returns
    -------
    np.ndarray
        Projected vector (non‑negative, sum‑1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT (fast for lengths ≥ ~1e3).

    Parameters
    ----------
    x, y : np.ndarray
        Real 1‑D input arrays.

    Returns
    -------
    np.ndarray
        Full convolution (length = len(x)+len(y)-1).
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())  # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float, offset: int = 0) -> np.ndarray:
    """
    Normalised power‑law: a_i ∝ (i+1+offset)^(-α).

    Parameters
    ----------
    n      : int
        Length of the sequence.
    alpha  : float
        Exponent (α>0).
    offset : int
        Number of leading zeros (the power‑law starts at index ``offset``).

    Returns
    -------
    np.ndarray
        Length‑`n` array with sum = 1 (all entries ≥ 0).
    """
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(1, n - offset + 1, dtype=float) + offset
        a[offset:] = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, r: float, offset: int = 0) -> np.ndarray:
    """
    Normalised exponential decay: a_i ∝ r^{i-offset} for i≥offset,
    zeros before ``offset``.  0<r<1.

    Parameters
    ----------
    n      : int
    r      : float
    offset : int

    Returns
    -------
    np.ndarray
    """
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(0, n - offset, dtype=float)
        a[offset:] = r ** idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _linear_sequence(n: int, offset: int = 0) -> np.ndarray:
    """
    A simple decreasing linear sequence (triangular shape) with optional
    leading zeros.

    a_i = (n-i) for i≥offset, else 0.  Normalised to sum = 1.
    """
    a = np.zeros(n, dtype=float)
    if offset < n:
        idx = np.arange(offset, n, dtype=float)
        a[offset:] = n - idx
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when the vector sums to 1:
    2·n·max(conv(a,a)).
    """
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimizes ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence discovered.

    Parameters
    ----------
    seed : int
        Random seed (reproducibility).
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Ignored – present for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence (non‑negative, sum > 0.01) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Broad parametric sweep (cheap but often very effective)
    # ------------------------------------------------------------------
    best_score = np.inf
    best_seq = None

    # candidate lengths – we stay below ~10000 to keep convolutions cheap
    length_candidates = [1500, 2000, 3000, 4000, 5000, 6000, 8000]

    # families and their parameter grids
    power_alphas = [0.2, 0.35, 0.5, 0.65, 0.8]
    power_offsets = [0, 5, 10, 30, 100]          # leading zeros
    exp_rs = [0.90, 0.94, 0.97, 0.99]
    exp_offsets = [0, 5, 20, 50]

    linear_offsets = [0, 10, 30, 100]

    for n in length_candidates:
        if time.time() > deadline:
            break

        # --- Power‑law family -------------------------------------------------
        for offset in power_offsets:
            if offset >= n:
                continue
            for alpha in power_alphas:
                a = _power_sequence(n, alpha, offset)
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[sweep] n={n} power α={alpha:.2f} off={offset} "
                          f"score={score:.6f}")

        # --- Exponential family -----------------------------------------------
        for offset in exp_offsets:
            if offset >= n:
                continue
            for r in exp_rs:
                a = _exp_sequence(n, r, offset)
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[sweep] n={n} exp r={r:.2f} off={offset} "
                          f"score={score:.6f}")

        # --- Linear decreasing family -----------------------------------------
        for offset in linear_offsets:
            if offset >= n:
                continue
            a = _linear_sequence(n, offset)
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[sweep] n={n} linear off={offset} score={score:.6f}")

        # --- Random Dirichlet seeds -------------------------------------------
        for _ in range(2):
            a = np.random.rand(n)
            a /= a.sum()
            score = _objective(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[sweep] n={n} random dirichlet score={score:.6f}")

        # --- Use the global ``height_sequence_1`` if available -----------------
        try:
            init = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
            if init.size == n:
                a = np.maximum(init, 0.0)
                if a.sum() == 0.0:
                    a = np.full(n, 1.0 / n)
                else:
                    a /= a.sum()
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[sweep] n={n} global init score={score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Refine the current best by sub‑gradient descent on the max‑conv
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a modest uniform sequence
        n = 2000
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    a = best_seq.copy()
    n = a.size

    # Momentum variables (optional but helpful)
    momentum = np.zeros_like(a)

    # Learning‑rate schedule
    lr = 0.8               # initial step size
    lr_decay = 0.9999
    lr_min = 1e-6

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        conv = _conv_fft(a, a)
        max_val = conv.max()
        k_star = int(conv.argmax())          # index where the max is attained

        # ----- sub‑gradient of the max‑convolution -------------------------
        idx = k_star - np.arange(n)          # desired partner index for each entry
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]       # g_j = 2·a_{k*-j}

        # Optional momentum (exponential moving average of gradients)
        momentum = 0.9 * momentum + grad

        # Gradient step + projection onto the simplex
        a = a - lr * momentum
        a = _project_onto_simplex(a)

        # Decay learning rate
        lr = max(lr * lr_decay, lr_min)

        # ----- occasional random mass‑exchange to help escape plateaus -----
        if iteration % 70 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            # move at most 30 % of the source mass (but never more than the step size)
            delta = min(a[i] * 0.3, lr * 0.5)
            if delta > 1e-12:
                a[i] -= delta
                a[j] += delta
                a = _project_onto_simplex(a)

        # ----- periodic exact evaluation ------------------------------------
        if iteration % 30 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[improve] iter={iteration} n={n} score={best_score:.6f}")

            # early exit if we have already beaten the target bound
            if best_score < 1.503:
                print("[target reached]")
                break

    # ------------------------------------------------------------------
    # 3. Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f} (n={best_seq.size}) "
          f"iterations={iteration}")
    return best_seq.tolist()
