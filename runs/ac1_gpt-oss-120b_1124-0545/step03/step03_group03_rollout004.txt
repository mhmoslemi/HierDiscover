"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm proceeds in three stages:

1. **Extended analytic search** – a dense grid + cheap random sampling over the
   “product” family  a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}.  This family is known to
   produce very low convolution peaks.  The best point found is kept as a seed.

2. **Hard‑max Adam optimisation** – we minimise the exact (non‑smooth) objective
   using the sub‑gradient of the max‑convolution.  The gradient of
   Φ(a)=max_k (a*a)_k is
        g = 2·conv_rev(w, a)   with w = soft‑max_{λ→∞} (a*a)
   i.e. w is the (normalised) indicator of the maximal convolution entries.
   An Adam optimiser (first‑ and second‑moment adaptation) together with a
   simplex projection yields fast convergence.

3. **Escape‑mechanisms** – every few seconds of stagnation a small random jitter
   is added, and occasional random “mass‑transfer’’ moves are performed.
   The best vector observed before the 1000 s budget expires is returned.

All helper routines are top‑level, no closures are used and the code respects
the required time budget.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a with itself (zero‑padded) via FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """Exact objective (scale invariant) for a non‑negative vector a.
    The vector is normalised to sum 1 before evaluation.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    return float(2.0 * a.shape[0] * np.max(conv))   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {w≥0, Σw=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """Generate a normalised analytic candidate.
    mode = 'product' → a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}
    mode = 'sum'     → a_i ∝ (i+off)^{-p} + (n‑1‑i+off)^{-p}
    """
    i = np.arange(n, dtype=np.float64)
    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def gradient_hard_max(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of Φ(a)=max_k (a*a)_k.
    Uses a hard mask on the maximal convolution entries.
    """
    n = a.shape[0]
    conv = conv_fft(a)
    m = np.max(conv)
    # entries that are (almost) maximal – relative tolerance 1e-12
    mask = conv >= m - max(1e-12 * m, 1e-14)
    w = mask.astype(float) / mask.sum()          # normalised weights
    # convolution w_rev * a  (same pattern as in the smoothed version)
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    # extract the n entries that correspond to the gradient
    start = n - 1
    end = 2 * n - 1
    slice_part = conv_wa[start:end]               # length n
    g = 2.0 * slice_part[::-1]
    return g


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) with a low
    value of `evaluate_sequence`.  The function respects the 1000 s
    time budget, prints progress information and finally returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5            # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # Phase 1 – dense analytic grid + cheap random sampling (product family)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    print("[init] analytic grid search (n={})".format(n))

    # coarse grid (identical to the previous baseline)
    p_vals = np.arange(0.30, 0.95 + 1e-9, 0.05)
    offset_vals = np.arange(0.0, 1.6, 0.2)
    for p in p_vals:
        for off in offset_vals:
            cand = analytic_candidate(n, p, off if off > 0 else 1e-6, "product")
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
    print(f"[init] best after coarse grid: {best_score:.6f}")

    # additional random sampling over a wider domain (product mode)
    # allocate ~20 % of the total budget for this stage
    sampling_deadline = start_time + 0.20 * budget_s
    while time.time() < sampling_deadline:
        # draw parameters on a log‑scale for p, linear for offset
        p = 10 ** np.random.uniform(-1, 1)           # p ∈ [0.1, 10]
        off = np.random.uniform(0.001, 5.0)          # offset ∈ [0.001,5]
        cand = analytic_candidate(n, p, off, "product")
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
        # occasional sanity check print
        if np.random.rand() < 0.001:
            print(f"[sample] p={p:.3f}, off={off:.3f} → {sc:.6f}")

    # if a global reference is present, evaluate it as well
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            print(f"[init] global reference score = {sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = ref.copy()

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation using the hard‑max sub‑gradient
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall back to a uniform vector (should never happen)
        best_seq = np.ones(n) / n
        best_score = compute_score(best_seq)

    a = best_seq.copy()
    cur_score = compute_score(a)

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedules (linear in time)
    step_start, step_end = 0.15, 5e-5          # learning‑rate like parameter
    temp_start, temp_end = 0.8, 1e-4          # Metropolis temperature
    restart_cooldown = 25.0                   # seconds without improvement

    last_improve = start_time
    iteration = 0

    print("[opt] starting Adam optimisation (hard‑max gradient)")

    while time.time() < deadline:
        elapsed = time.time() - start_time
        # linear schedules
        progress = elapsed / (budget_s - 20.0)          # keep a small final margin
        lr = step_start + progress * (step_end - step_start)
        temp = temp_start + progress * (temp_end - temp_start)

        # ------------------------------------------------------------------
        # Gradient step (hard‑max)
        # ------------------------------------------------------------------
        g = gradient_hard_max(a)               # shape (n,)
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        cand = a - step_vec
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)

        cand_score = compute_score(cand)

        # ---------------------------------------------------------------
        # Metropolis‑style acceptance
        # ---------------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # downhill moves occasionally accepted
            if temp > 1e-12:
                prob = math.exp(-(cand_score - cur_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:07d}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # ---------------------------------------------------------------
        # Occasional random “mass‑transfer’’ move to inject diversity
        # ---------------------------------------------------------------
        if np.random.rand() < 0.04:      # 4 % of iterations
            cand = random_move(a, lr * 0.5)
            cand_score = compute_score(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    last_improve = time.time()
                    print(f"[{iteration:07d}] best after random move = {best_score:.6f}")

        # ---------------------------------------------------------------
        # Stagnation handling – jitter restart
        # ---------------------------------------------------------------
        if time.time() - last_improve > restart_cooldown:
            jitter = np.random.rand(n) * 1e-4
            a = best_seq + jitter
            a = project_simplex(a)
            cur_score = compute_score(a)
            last_improve = time.time()
            print(f"[{iteration:07d}] jitter restart (stagnation) – cur_score={cur_score:.6f}")

        # early exit if we already beat the target
        if best_score <= 1.503:
            print("[done] target reached, exiting early")
            break

    total_time = time.time() - start_time
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
