"""
A two‑phase optimizer for the step‑function bound.

Phase 1 – Smoothed‑max gradient descent.
    * The objective max(conv(a,a)) is replaced by a smooth
      log‑sum‑exp surrogate:
          f_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k])
      For β→∞ this converges to the true max.
    * Gradient of f_β can be written in closed form:
          ∇f_β(a) = 2·(w ⋆ a) ,
      where w is the soft‑max weight vector over the convolution
      entries and “⋆” denotes the (linear) cross‑correlation.
      Both the convolution and the correlation are evaluated
      by FFTs of length a power‑of‑two ≥ 2·n−1.
    * A Nesterov‑style momentum update with a diminishing step‑size
      (α_t = α₀/√t) is applied, followed by projection onto the
      ℓ₁‑simplex {a≥0, Σa = √(2n)}.

Phase 2 – Simulated annealing.
    * Starts from the best vector found in phase 1.
    * Randomly transfers a small fraction of mass between two
      positions (the ℓ₁‑norm is preserved, non‑negativity holds).
    * Metropolis acceptance with an exponential cooling schedule.

Both phases respect a global time budget (default 1000 s) and
continually record the best sequence.  The returned list is ready
for the external ``evaluate_sequence`` call.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex { w ≥ 0 , Σ w = s }."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (FFT length for linear convolution)."""
    return 1 << ((2 * n - 1).bit_length())


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Return (max(conv), argmax) of the linear self‑convolution a*a."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence minimising ``evaluate_sequence``.
    The method combines a smoothed‑max gradient descent with a
    simulated‑annealing final refinement.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # leave a small safety margin for the evaluator wrapper
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries are fast enough for many FFTs.
    # If a previous construction is available we reuse its length.
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=float)
        n = base.size
        a = base.copy()
    else:
        n = 3000                     # default size
        a = np.random.exponential(scale=1.0, size=n)

    target_sum = math.sqrt(2.0 * n)   # ℓ₁‑norm fixed by the evaluator

    # Normalise to the required ℓ₁‑norm
    a = a.astype(float)
    a = a / a.sum() * target_sum

    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Phase 1 – smoothed‑max gradient descent with momentum
    best_seq = a.copy()
    cur_max, _ = _max_convolution(a, fft_len)
    best_val = cur_max
    print(f"[INIT] n={n}, target L1={target_sum:.3f}, max(conv)={cur_max:.6f}")

    # schedule of β (smoothness) values
    beta_schedule = [1.0, 3.0, 10.0, 30.0, 80.0]
    # momentum parameter
    mu = 0.9
    # velocity for Nesterov update
    velocity = np.zeros_like(a)

    it = 0
    last_improve = 0

    # allocate roughly 45 % of the total budget for this phase
    phase1_deadline = start_time + budget_s * 0.45

    for beta in beta_schedule:
        # step‑size tuned to the current β (larger β → steeper landscape)
        alpha0 = target_sum * 0.02 / max(beta, 1.0)
        while time.time() < phase1_deadline:
            # ----------- forward FFT of a
            A_fft = np.fft.rfft(a, n=fft_len)

            # ----------- full self‑convolution (needed for the soft‑max)
            conv_full = np.fft.irfft(A_fft * A_fft, n=fft_len)[: 2 * n - 1]

            # ----------- soft‑max weights over the convolution entries
            max_c = conv_full.max()
            # subtract max to avoid overflow; exponentials are ≤1
            exp_shift = np.exp(beta * (conv_full - max_c))
            w = exp_shift / exp_shift.sum()          # shape (2n‑1,)

            # ----------- gradient = 2·(w ⋆ a)
            w_fft = np.fft.rfft(w, n=fft_len)
            corr_full = np.fft.irfft(np.conj(w_fft) * A_fft, n=fft_len)
            grad = 2.0 * corr_full[:n]               # real‑valued gradient

            # ----------- Nesterov‑style momentum update
            step = alpha0 / math.sqrt(it + 1.0)
            velocity = mu * velocity - step * grad
            a_candidate = a + velocity
            a_candidate = _simplex_projection(a_candidate, target_sum)

            a = a_candidate
            it += 1

            # record best objective on the exact (non‑smoothed) max
            cur_max, _ = _max_convolution(a, fft_len)
            if cur_max < best_val:
                best_val = cur_max
                best_seq = a.copy()
                last_improve = it
                print(f"[{it:7d}] β={beta:.1f} new best max(conv)={best_val:.6f}")

            # simple stagnation restart
            if it - last_improve > 8000:
                a = np.random.exponential(scale=1.0, size=n)
                a = a / a.sum() * target_sum
                velocity.fill(0.0)
                last_improve = it
                print(f"[{it:7d}] restart due to stagnation (β={beta})")

            if time.time() >= phase1_deadline:
                break

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing starting from the best vector found
    a = best_seq.copy()
    cur_max, _ = _max_convolution(a, fft_len)
    best_val = cur_max
    best_seq = a.copy()
    print(f"[SA‑START] max(conv)={best_val:.6f}")

    # exponential cooling from 20 % of current best to 1e‑5 · best
    T0 = best_val * 0.20
    Tmin = best_val * 1e-5
    sa_start = time.time()
    sa_deadline = deadline
    step_counter = 0

    while time.time() < sa_deadline:
        elapsed = time.time() - sa_start
        frac = elapsed / (sa_deadline - sa_start + 1e-12)
        T = T0 * ((Tmin / T0) ** frac)   # monotone exponential decay

        # propose a random mass transfer between two distinct positions
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # move up to 10 % of the source mass
        delta = a[i] * 0.10 * np.random.rand()
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta
        # ℓ₁‑norm unchanged, non‑negativity preserved

        cand_max, _ = _max_convolution(a_cand, fft_len)

        # Metropolis acceptance
        if cand_max < cur_max:
            a = a_cand
            cur_max = cand_max
            if cand_max < best_val:
                best_val = cand_max
                best_seq = a_cand.copy()
                print(f"[SA {step_counter:7d}] new best max(conv)={best_val:.6f}")
        else:
            prob = math.exp(-(cand_max - cur_max) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = a_cand
                cur_max = cand_max

        step_counter += 1

    print(f"[DONE] best max(conv) = {best_val:.6f} after {it} grad iters "
          f"and {step_counter} SA steps")
    return best_seq.tolist()
