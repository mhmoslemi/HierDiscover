def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time

    start = time.time()
    deadline = start + max(0, budget_s - 0.5)  # small margin

    n = int(kwargs.get('n', 2000))

    best_global_a = None
    best_global_score = float('inf')

    num_attempts = 5
    for attempt in range(num_attempts):
        remaining_time = deadline - time.time()
        if remaining_time < 5:
            break
        # allocate portion of remaining time: part = remaining_time * 0.9 maybe
        alloc = remaining_time * 0.9 / (num_attempts - attempt)  # flexible; or equal share

        # initialize a
        if attempt == 0 and 'height_sequence_1' in globals():
            a0 = np.array(globals()['height_sequence_1'], dtype=float)
            if a0.shape[0] != n:
                a0 = a0[:n] if a0.shape[0] > n else np.pad(a0, (0,n-a0.shape[0]))
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:
            rng = np.random.default_rng(seed + attempt)
            # sample Dirichlet with concentration alpha = 0.5
            a0 = rng.dirichlet(alpha=np.full(n, 0.5))
        # call optimizer
        a_opt, score_opt = _optimize_a(a0, n, alloc, seed + attempt)
        if score_opt < best_global_score:
            best_global_score = score_opt
            best_global_a = a_opt.copy()
            print(f"[attempt {attempt}] new global best score {best_global_score:.6f}")

        # optionally update seed etc.
    # finalize
    if best_global_a is None:
        # fallback to uniform
        best_global_a = np.ones(n) / n
    # ensure non-negative and float
    best_global_a = np.maximum(best_global_a, 0.0)

def _optimize_a(a_init, n, budget_s, seed):
    import numpy as np, time

    v = np.log(a_init + 1e-12)   # shape n
    # Adam parameters:
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    # schedule:
    beta_start, beta_end = 0.1, 5000.0
    lr_start, lr_end = 0.2, 0.001

    # precompute fft_len
    fft_len = 1 << ((3 * n - 2).bit_length())

    start = time.time()
    deadline = start + budget_s - 0.1

    best_a = a_init.copy()
    # compute best_score using conv
    fa = np.fft.fft(best_a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    best_score = 2.0 * n * conv.max()
    last_improve_time = start

    it = 0
    mask_phase_start = 0.7 # fraction of allocated time after which we use hard max

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = min(1.0, elapsed / (budget_s - 5.0))  # progress

        beta = beta_start * (1.0 - frac) + beta_end * frac
        lr = lr_start * (1.0 - frac) + lr_end * frac

        a = softmax(v)  # compute probabilities

        # FFT of a
        fa = np.fft.fft(a, fft_len)
        conv_aa = np.real(np.fft.ifft(fa * fa))[:2 * n - 1]

        # Determine weight vector w
        if frac < mask_phase_start:
            # softmax weighting (smooth)
            z = beta * conv_aa
            z_max = np.max(z)
            w = np.exp(z - z_max)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.size
            else:
                w = w / w_sum
        else:
            # hard max mask
            max_val = conv_aa.max()
            eps_mask = max_val * 1e-12 + 1e-14
            mask = (conv_aa >= max_val - eps_mask).astype(float)
            mask_sum = mask.sum()
            if mask_sum == 0:
                mask = np.zeros_like(conv_aa)
                mask[np.argmax(conv_aa)] = 1.0
                mask_sum = 1.0
            w = mask / mask_sum

        # compute gradient via FFT
        fw = np.fft.fft(w, fft_len)
        conv_wa = np.real(np.fft.ifft(fw * fa))[:3 * n - 2]
        g2 = conv_wa[:n]
        corr_full = np.real(np.fft.ifft(np.conj(fw) * fa))[:3 * n - 2]
        g1 = corr_full[:n]
        g = 2.0 * n * (g1 + g2)

        dot_ag = np.dot(a, g)
        grad_v = a * (g - dot_ag)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # exact score
        cur_score = 2.0 * n * conv_aa.max()
        # update best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve_time = time.time()
            # print intermediate
            if it % 5000 == 0:
                print(f"[{it}] best score {best_score:.6f}")

        # stagnation restart
        if time.time() - last_improve_time > 20.0:
            # jitter around best found
            jitter = np.random.default_rng(seed + it).normal(scale=0.01, size=n)
            v = np.log(best_a + 1e-12) + jitter
            m.fill(0)
            v_hat.fill(0)
            last_improve_time = time.time()
            print(f"restart after stagnation at iter {it}, time {elapsed:.1f}s")

    # Return best sequence
    return best_a, best_score

"""
Optimise a non‑negative step‑function (probability vector) for the
objective  E(a)=2·n·max(conv(a,a)) / (sum(a)**2).

The optimiser works in two phases:
  1.  A temperature‑annealed soft‑max is used to obtain a smooth
      approximation of the max‑operator.  Adam updates on the
      unconstrained log‑parameters (v) move the distribution towards
      lower convolution peaks.
  2.  In the final part of the run the true max is used (hard mask).
      This yields a genuine sub‑gradient of the original nonsmooth
      objective and finishes the refinement.

Multiple independent starts (including the historic ``height_sequence_1``
if present) are tried within the total time budget; the best sequence
found is returned as a plain ``list[float]``.  All arithmetic is performed
on the probability simplex (sum=1) via a soft‑max parametrisation, thus
the scale‑invariance of the objective is respected automatically.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max mapping ℝⁿ → simplex."""
    v_max = np.max(v)
    exp_v = np.exp(v - v_max)
    return exp_v / exp_v.sum()


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _optimize_a(
    a_init: np.ndarray,
    n: int,
    alloc_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Optimise a single start vector ``a_init`` for at most ``alloc_s`` seconds.
    Returns the best vector found and its exact (scale‑invariant) score.
    """
    # log‑parametrisation (soft‑max guarantees feasibility)
    v = np.log(a_init + 1e-12)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    # schedules (linear in normalised time p∈[0,1])
    beta_start, beta_end = 0.1, 5000.0         # soft‑max temperature
    lr_start, lr_end = 0.25, 0.001           # learning‑rate

    # FFT size (covers convolution of length 3·n‑2)
    fft_len = _next_pow2(3 * n - 2)

    # bookkeeping
    start = time.time()
    deadline = start + alloc_s - 0.05  # tiny safety margin
    best_a = a_init.copy()
    # exact score for a probability vector is 2·n·max(conv)
    fa = np.fft.fft(best_a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    best_score = 2.0 * n * conv.max()
    last_improve = start
    it = 0

    # after this fraction of the allocated time we switch to the hard‑max mask
    mask_phase_start = 0.70

    while time.time() < deadline:
        elapsed = time.time() - start
        # progress fraction p ∈ [0,1]
        p = min(1.0, elapsed / (alloc_s - 5.0))

        # temperature and learning‑rate schedules
        beta = beta_start * (1.0 - p) + beta_end * p
        lr = lr_start * (1.0 - p) + lr_end * p

        # ----------- current probabilities -----------------
        a = softmax(v)               # Σa = 1, a_i ≥ 0

        # ----------- convolution a * a -------------------
        fa = np.fft.fft(a, fft_len)
        conv_aa = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        # ----------- weight vector w  --------------------
        if p < mask_phase_start:
            # smooth soft‑max over the convolution entries
            z = beta * conv_aa
            z_max = np.max(z)
            w = np.exp(z - z_max)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / w_sum
        else:
            # hard mask on the true maximum (sub‑gradient of max)
            max_val = conv_aa.max()
            # tolerance – absolute but scaled to avoid missing ties
            eps_mask = max_val * 1e-12 + 1e-14
            mask = (conv_aa >= max_val - eps_mask).astype(float)
            mask_sum = mask.sum()
            if mask_sum == 0:
                mask = np.zeros_like(conv_aa)
                mask[np.argmax(conv_aa)] = 1.0
                mask_sum = 1.0
            w = mask / mask_sum

        # ----------- gradient wrt a (via FFT) -------------
        fw = np.fft.fft(w, fft_len)

        # (w * a)  – linear convolution, first n entries
        conv_wa = np.real(np.fft.ifft(fw * fa))[: 3 * n - 2]
        g2 = conv_wa[:n]

        # w ⋆ a  – linear cross‑correlation, first n entries
        corr_full = np.real(np.fft.ifft(np.conj(fw) * fa))[: 3 * n - 2]
        g1 = corr_full[:n]

        # gradient of the (smoothed) max‑objective
        g = 2.0 * n * (g1 + g2)

        # chain rule for the soft‑max parametrisation
        dot_ag = np.dot(a, g)
        grad_v = a * (g - dot_ag)

        # --------------- Adam update ----------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # --------------- exact score & bookkeeping -------
        cur_score = 2.0 * n * conv_aa.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            if it % 2000 == 0:
                print(
                    f"[{it:07d}] new best = {best_score:.6f}"
                    f" (t={elapsed:.1f}s, beta={beta:.1f})"
                )

        # --------------- restart on stagnation ----------
        if time.time() - last_improve > 20.0:
            # jitter the log‑parameters around the current best point
            jitter = rng.normal(scale=0.01, size=n)
            v = np.log(best_a + 1e-12) + jitter
            m.fill(0.0)
            v_hat.fill(0.0)
            last_improve = time.time()
            print(
                f"[{it:07d}] restart after {elapsed:.1f}s (no improvement)"
            )

    return best_a, best_score


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget (default
    1000 s) and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Base random seed – different runs use different seeds.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Optional ``n`` (length of the sequence).  Defaults to 2000.

    Returns
    -------
    list[float]
        The best coefficient list discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)  # tiny safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Prepare several independent starts (including the historic one)
    # ------------------------------------------------------------------
    attempts = 6
    global_best_a = None
    global_best_score = float("inf")
    rng_global = np.random.default_rng(seed)

    # pre‑compute remaining attempts to split the budget dynamically
    for attempt in range(attempts):
        time_left = deadline - time.time()
        if time_left < 10.0:
            break
        # allocate a fraction of the remaining time to this attempt
        remaining_attempts = attempts - attempt
        alloc = time_left * 0.90 / remaining_attempts

        # ---- initialise a ------------------------------------------------
        if attempt == 0 and "height_sequence_1" in globals():
            a0 = np.array(globals()["height_sequence_1"], dtype=float)
            if a0.size != n:
                # truncate or pad with zeros to match the requested length
                if a0.size > n:
                    a0 = a0[:n]
                else:
                    a0 = np.pad(a0, (0, n - a0.size))
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:
            # Dirichlet draw with a modest concentration → diverse starts
            alpha = np.full(n, 0.5)
            a0 = rng_global.dirichlet(alpha)

        # ---- run the optimiser for this start ---------------------------
        a_opt, score_opt = _optimize_a(
            a_init=a0,
            n=n,
            alloc_s=alloc,
            rng=np.random.default_rng(seed + attempt + 12345),
        )
        if score_opt < global_best_score:
            global_best_score = score_opt
            global_best_a = a_opt.copy()
            elapsed = time.time() - start_time
            print(
                f"[attempt {attempt}] new global best = {global_best_score:.6f}"
                f" (elapsed {elapsed:.1f}s)"
            )

    # ------------------------------------------------------------------
    # 2) Final safety: fall back to a uniform vector if nothing was found
    # ------------------------------------------------------------------
    if global_best_a is None:
        global_best_a = np.full(n, 1.0 / n)

    # ensure non‑negativity (already guaranteed) and return as plain list
    global_best_a = np.maximum(global_best_a, 0.0)

    total_time = time.time() - start_time
    print(f"[done] best score={global_best_score:.6f}, time={total_time:.1f}s")
    return global_best_a.tolist()
