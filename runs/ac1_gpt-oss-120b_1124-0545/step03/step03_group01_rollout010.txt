"""
Propose a step‑function (non‑negative vector) that minimizes
    2·n·max(conv(a,a)) / (sum(a)²)

Algorithm
---------
1. Generate diverse seed vectors (uniform, geometric, power‑law, symmetric,
   sine‑window, random soft‑max, and reuse the previous best if available).

2. Pick the best seed (by the exact objective).

3. Repeatedly improve the current vector using an *alternating LP + mixing*
   scheme inspired by the literature:
   * For the current vector f compute M = max(conv(f,f)).
   * Solve a linear program:
         maximise   Σ b_j
         subject to conv(f,b) ≤ M   (only the “active’’ constraints where
                                     conv(f,f) ≈ M are kept)
                     b_j ≥ 0
     This finds a vector b that can carry as much mass as possible while
     keeping the mixed convolution bounded by M.
   * Normalise g = b / Σb (so Σg = 1) and replace
         f ← (1‑t)·f + t·g   with a few candidate step‑sizes t.
     The mixture stays on the probability simplex and usually lowers the
     maximal auto‑convolution.

4. After each successful LP step perform a short *mirror‑descent* (exponentiated‑gradient)
   refinement that uses the exact sub‑gradient of the true max.  This
   quickly removes remaining bumps.

5. Continue the above loop until the global deadline (≈ 1000 s) is hit,
   trying lengths 4000 and, if time permits, 8000.  The best sequence
   found over all runs is returned as a plain list of floats.

The implementation relies only on NumPy and SciPy (linprog with the HiGHS
solver).  All helpers are defined at top‑level; no closures or file I/O
are used.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution via real FFT.
    Returns the full result of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Seed families (already normalised to sum 1)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest set of diverse starting points for length n."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))

    rng = np.random.default_rng(12345)
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Objective helper (sum of sequence is assumed to be 1)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv(a,a)) for a normalised to sum 1."""
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Refine a sequence using the exact sub‑gradient of max(conv).
    Returns (best_sequence, best_value) seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = _objective(a, n)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of the max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) update
        lr = 0.5 / math.sqrt(it)   # a little more aggressive than 0.6 before
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)     # numerical stability
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

    return best_seq, best_val


# ----------------------------------------------------------------------
# Alternating Linear‑Programming + mixing step
# ----------------------------------------------------------------------


def _lp_improve(cur_seq: np.ndarray,
                n: int,
                deadline: float,
                max_lp_iters: int = 15) -> tuple[np.ndarray, float]:
    """
    Perform up to `max_lp_iters` alternating LP‑mix‑refine steps.
    Returns (best_sequence, best_value) seen.
    """
    best_seq = cur_seq.copy()
    best_val = _objective(cur_seq, n)

    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        # current convolution and its maximum
        conv = _conv_fft(cur_seq, cur_seq)
        M = np.max(conv)

        # choose a modest active set (indices where conv is close to the max)
        thr = M * 0.97
        idxs = np.where(conv >= thr)[0]
        if idxs.size < 5:                     # fallback to a few central indices
            idxs = np.arange(max(0, n - 3), min(2 * n - 1, n + 3))

        # build the (limited) constraint matrix A @ b ≤ M
        A = np.zeros((len(idxs), n), dtype=np.float64)
        for row, k in enumerate(idxs):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            cols = np.arange(k - i_end, k - i_start + 1, dtype=int)
            A[row, cols] = cur_seq[i_start:i_end + 1]

        # solve LP: maximise Σ b_j  ⇔ minimise -Σ b_j
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(len(idxs), M),
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            print(f"[LP] iteration {lp_it}: linprog failed ({res.message})")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            print("[LP] zero mass returned – aborting LP loop")
            break

        # normalise to keep sum = 1
        g = b / sum_b

        # try a few mixing coefficients
        improved = False
        for t in (0.5, 0.2, 0.1):
            new_seq = (1.0 - t) * cur_seq + t * g
            new_seq = np.maximum(new_seq, 0.0)
            new_seq /= new_seq.sum()
            new_val = _objective(new_seq, n)

            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()
                cur_seq = new_seq
                improved = True
                print(f"[LP‑mix] n={n} lp_it={lp_it} t={t:.2f} val={new_val:.6f}")
                break

        if not improved:
            # no improvement – stop the LP loop early
            print(f"[LP] n={n} lp_it={lp_it} no improvement, terminating LP phase")
            break

        # short sub‑gradient refinement to polish the new point
        cur_seq, cur_val = _subgradient_refine(cur_seq, n, deadline, max_iters=2000)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()
            print(f"[LP‑post‑sub] n={n} lp_it={lp_it} refined val={cur_val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    `evaluate_sequence` objective.  The routine respects a global time
    budget and returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # retrieve a previously discovered construction if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # lengths we will attempt – start with 4000, then 8000 if time allows
    lengths = [4000, 8000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous)

        # quick evaluation of raw seeds
        for s in seeds:
            val = _objective(s, n)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n} new global best={val:.6f}")

        # start from the current best of this length
        cur_seq = global_best_seq if global_best_seq.shape[0] == n else seeds[0].copy()
        cur_val = _objective(cur_seq, n)

        # a brief sub‑gradient polishing before the LP phase
        cur_seq, cur_val = _subgradient_refine(cur_seq, n, deadline, max_iters=3000)
        if cur_val < global_best_val:
            global_best_val, global_best_seq = cur_val, cur_seq.copy()
            print(f"[init‑sub] n={n} improved to {cur_val:.6f}")

        # alternating LP + mixing + refinement
        cur_seq, cur_val = _lp_improve(cur_seq, n, deadline, max_lp_iters=12)
        if cur_val < global_best_val:
            global_best_val, global_best_seq = cur_val, cur_seq.copy()
            print(f"[overall] after LP phase n={n} best={cur_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be reached)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = _uniform(n)
        global_best_val = _objective(global_best_seq, n)
        print("[fallback] using uniform distribution")

    # expose the best sequence for a possible next call
    globals()["height_sequence_1"] = global_best_seq.copy()

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(global_best_seq)})")
    return global_best_seq.tolist()
