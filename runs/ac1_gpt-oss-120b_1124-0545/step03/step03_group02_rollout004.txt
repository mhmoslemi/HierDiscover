import numpy as np
import time

def project_simplex(v):
    """Euclidean projection of v onto probability simplex (sum=1, v>=0)."""
    # algorithm from Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def initialise_sequence(n, seed):
    """Return a starting probability vector of length n."""
    rng = np.random.default_rng(seed)
    # if height_sequence_1 defined, use it:
    if "height_sequence_1" in globals():
        seq = np.array(globals()["height_sequence_1"], dtype=float)
        if seq.size == n:
            seq = np.maximum(seq, 0.0)
            seq = seq / seq.sum()
            return seq
    # else try arcsine analytic shape:
    i = np.arange(n, dtype=float)
    seq = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    seq = seq / seq.sum()
    return seq

def compute_gradient_max(a, conv, n, tol=1e-12):
    """Return subgradient of 2*n*max(conv) at a (assuming sum a = 1)."""
    M = conv.max()
    # indices where conv is within tolerance of max
    # using absolute tolerance or relative?
    # Use absolute tolerance small relative to M
    mask = conv >= M - max(tol, M * 1e-12)
    ks = np.nonzero(mask)[0]
    # compute average gradient
    g = np.zeros_like(a)
    for k in ks:
        # compute shift of a by k
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        # j indexes to update
        j_idx = np.arange(low, high + 1)
        i_idx = k - j_idx
        # update g
        g[j_idx] += 4.0 * n * a[i_idx]
    if len(ks) > 1:
        g /= len(ks)
    return g

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Subgradient exploration on the exact max objective.
    Uses exponentiated updates with a decaying step size and random restarts.
    Returns the best non‑negative sequence found within the time budget.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    n = int(kwargs.get("n", 2000))

    # Pre‑compute FFT length (power of two)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # Helper to evaluate exact score (2*n*max(conv))
    def exact_score(a):
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        return float(2.0 * n * conv.max()), conv

    # Initialise a mix of candidates; start from arcsine / previous best / random
    best_a = initialise_sequence(n, seed)
    best_score, best_conv = exact_score(best_a)

    # Keep track of when we last improved
    last_improve = start

    # Parameters for the exponentiated subgradient update
    step_start = 0.5          # initial step size (multiplier of normalized gradient)
    step_end = 0.02           # final step size
    jitter_scale = 1e-3       # magnitude of random perturbations on restart
    stagnation_time = 30.0    # seconds without improvement before restart
    iter_count = 0

    # Current working sequence
    a = best_a.copy()

    while time.time() < deadline:
        elapsed = time.time() - start
        p = min(1.0, elapsed / (budget_s - 10.0))

        # current step size (linear schedule)
        step = step_start * (1.0 - p) + step_end * p

        # Compute convolution and max
        score, conv = exact_score(a)

        # Record best
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_conv = conv.copy()
            last_improve = time.time()
            print(f"[{iter_count:07d}] new best {best_score:.6f} (t={elapsed:.1f}s)")

        # Compute subgradient of the max term
        g = compute_gradient_max(a, conv, n)

        # Normalise gradient to range [-1,1] to keep exponent reasonable
        g_norm = g / (np.max(np.abs(g)) + 1e-16)

        # Exponentiated update (mirror descent with KL divergence)
        a = a * np.exp(-step * g_norm)

        # Renormalise to the probability simplex
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum) or np.isinf(a_sum):
            # fallback to uniform distribution
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # Occasionally inject a tiny jitter to avoid exact zeroes
        if iter_count % 1000 == 0 and iter_count > 0:
            jitter = np.random.randn(n) * 1e-12
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()

        # Restart if we have been stagnant for a while
        if time.time() - last_improve > stagnation_time:
            # Build a fresh start around the best known point
            jitter = np.random.randn(n) * jitter_scale
            a = np.maximum(best_a + jitter, 0.0)
            a = a / a.sum()
            last_improve = time.time()
            print(f"[{iter_count:07d}] restart after {stagnation_time:.0f}s (t={elapsed:.1f}s)")

        iter_count += 1

    total = time.time() - start
    print(f"[done] iterations={iter_count}, best score={best_score:.6f}, time={total:.1f}s")

    # Ensure output complies with evaluate_sequence expectations
    best_a = np.maximum(best_a, 0.0)
    # Clip to modest values (the optimizer never creates huge numbers)
    best_a = np.minimum(best_a, 1000.0)
    return best_a.tolist()

"""
Subgradient‐descent with exponentiated (mirror) updates.

The objective (for a probability vector a, Σa=1) is

    E(a) = 2·n·max(conv(a,a))

which is scale‑invariant, so we keep Σa=1 throughout.
At each iteration we

1. compute the full linear convolution via FFT,
2. locate the (almost) maximal entries,
3. build the sub‑gradient of the max‑function:
       ∇E = 4·n * average_{k∈K} shift(a,k) ,
   where shift(a,k)[j] = a[k‑j] (zero outside the range),
4. normalise the gradient to the interval [‑1,1],
5. perform an exponentiated (KL‑mirror) step
       a ← a·exp(‑η·∇E)   and renormalise to the simplex,
6. restart from the best‑so‑far (with tiny jitter) if
   no improvement has been seen for a while.

The schedule for the step size η is linear from `step_start`
to `step_end` over the allowed time.  The algorithm keeps the
best sequence found and returns it as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (Σv=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def initialise_sequence(n: int, seed: int) -> np.ndarray:
    """
    Produce a reasonable start vector of length n.
    Preference is given to a previously discovered good sequence
    (`height_sequence_1`).  If unavailable we fall back to the
    classic arcsine shape (the continuous optimum for the related
    problem).
    The returned vector is a probability vector (Σ=1, non‑negative).
    """
    rng = np.random.default_rng(seed)
    if "height_sequence_1" in globals():
        cand = np.array(globals()["height_sequence_1"], dtype=float)
        if cand.size == n:
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
            return cand

    # arcsine shape –  a_i ∝ 1/√[(i+½)(n-i-½)]
    i = np.arange(n, dtype=float)
    cand = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand = cand / cand.sum()
    return cand


def exact_score(a: np.ndarray, fft_len: int) -> (float, np.ndarray):
    """
    Compute the exact objective 2·n·max(conv(a,a))
    and return also the raw convolution vector.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max()), conv


def subgradient_of_max(a: np.ndarray, conv: np.ndarray, n: int) -> np.ndarray:
    """
    Return the sub‑gradient of 2·n·max(conv) at a.
    The sub‑gradient of the max is the average of the gradients of
    all convolution entries that are (almost) maximal.
    """
    M = conv.max()
    # treat indices whose value is within a tiny relative tolerance as maximal
    tol = max(1e-12, M * 1e-12)
    mask = conv >= M - tol
    ks = np.nonzero(mask)[0]

    g = np.zeros_like(a)
    for k in ks:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        j_idx = np.arange(low, high + 1, dtype=int)
        i_idx = k - j_idx
        # gradient of max entry w.r.t. a_j is 4·n·a_{k‑j}
        g[j_idx] += 4.0 * n * a[i_idx]

    if len(ks) > 1:
        g /= len(ks)
    return g


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises the evaluation function.  The routine respects a
    time budget (default 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    n = int(kwargs.get("n", 2000))

    # FFT length – next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 1).bit_length())

    # initialise with the best known (or arcsine) and evaluate it
    a = initialise_sequence(n, seed)
    best_score, best_conv = exact_score(a, fft_len)
    best_a = a.copy()
    last_improve = start_time

    # hyper‑parameters for the exponentiated subgradient updates
    step_start = 0.30      # larger values give faster early movement
    step_end   = 0.010     # fine‑grained adjustments at the end
    jitter_scale = 1e-3    # scale of random noise on a restart
    stagnation_time = 30.0 # seconds without improvement before a restart
    iter_cnt = 0

    while time.time() < deadline:
        # progress fraction (used for linear schedules)
        elapsed = time.time() - start_time
        p = min(1.0, elapsed / (budget_s - 10.0))

        # linear annealing of the step size η
        eta = step_start * (1.0 - p) + step_end * p

        # --- evaluate current objective ---------------------------------
        score, conv = exact_score(a, fft_len)

        # keep the best ever seen
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_conv = conv.copy()
            last_improve = time.time()
            print(f"[{iter_cnt:07d}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # --- compute sub‑gradient of the max ---------------------------
        g = subgradient_of_max(a, conv, n)

        # normalise gradient to the range [-1, 1] to keep exp(·) stable
        max_abs = np.max(np.abs(g)) + 1e-16
        g_norm = g / max_abs

        # --- exponentiated (mirror) update -----------------------------
        a = a * np.exp(-eta * g_norm)
        a_sum = a.sum()
        if a_sum == 0 or not np.isfinite(a_sum):
            # fall back to a uniform distribution if something went wrong
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # occasional tiny jitter to avoid exact zeroes (numerical hygiene)
        if iter_cnt % 1000 == 0 and iter_cnt > 0:
            jitter = np.random.randn(n) * 1e-12
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()

        # --- stagnation handling ---------------------------------------
        if time.time() - last_improve > stagnation_time:
            # restart around the current best with a small random kick
            jitter = np.random.randn(n) * jitter_scale
            a = np.maximum(best_a + jitter, 0.0)
            a = a / a.sum()
            last_improve = time.time()
            print(f"[{iter_cnt:07d}] restart after stagnation (t={elapsed:.1f}s)")

        iter_cnt += 1

    total = time.time() - start_time
    print(f"[done] iterations={iter_cnt}, best score={best_score:.6f}, time={total:.1f}s")

    # final safety – guarantee non‑negativity and reasonable magnitude
    best_a = np.maximum(best_a, 0.0)
    best_a = np.minimum(best_a, 1000.0)   # well below the evaluate_sequence cap

    return best_a.tolist()
