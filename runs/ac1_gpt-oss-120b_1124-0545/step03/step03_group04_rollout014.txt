def conv_max(a):
    n = a.shape[0]
    size = 2*n-1
    N = 1 << ((size-1).bit_length())
    A = np.fft.rfft(a, N)

"""
Propose a step‑function (non‑negative sequence) with a low
evaluate_sequence() value.

The algorithm consists of three stages:

1. **Randomised parametric search** –  a family of monotone
   sequences a_i ∝ (i+1)^(-α)·exp(-τ·i)·(1+β·(i/(n‑1)-½)) is explored.
   For each random triple (α,τ,β) a short evaluation (n≈2000) is
   performed using an FFT‑based convolution; the best parameters found
   are kept.

2. **Upscaling** –  the best parameters are used to build a longer
   candidate (n≈5000).  Because the shape depends only on the relative
   index i/n, the quality measured on the small problem transfers to the
   larger one.

3. **Fine‑grained refinement** –  a few thousand iterations of a
   smoothed “max‑gradient” descent (soft‑max approximation of the
   ℓ∞‑norm) are applied to the up‑scaled sequence, always projecting
   back onto the probability simplex.  The best sequence ever seen is
   returned.

The whole routine respects a 1000 s wall‑clock budget, uses only NumPy,
and returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_fft(a: np.ndarray) -> float:
    """Full convolution via FFT and return its maximum entry."""
    n = a.shape[0]
    size = 2 * n - 1
    N = 1 << ((size - 1).bit_length())          # next power of two
    Af = np.fft.rfft(a, N)
    conv = np.fft.irfft(Af * Af, N)[:size]
    return float(conv.max())


def _make_sequence(alpha: float, tau: float, beta: float, n: int) -> np.ndarray | None:
    """
    Build a normalised candidate sequence of length ``n``.
    The shape is a power‑law with optional exponential decay and a linear tilt.
    Returns None if the construction would create negative entries.
    """
    i = np.arange(n, dtype=float)
    base = (i + 1.0) ** (-alpha) * np.exp(-tau * i)

    if beta != 0.0:
        tilt = 1.0 + beta * (i / (n - 1.0) - 0.5)
        # tilt may become negative – reject such candidates
        if np.any(base * tilt <= 0.0):
            return None
        base = base * tilt

    total = base.sum()
    if total <= 0.0:
        return None
    return base / total


def _softmax_max_conv(a: np.ndarray, eps: float) -> tuple[np.ndarray, float]:
    """
    Compute the soft‑max approximation of max(conv) and the associated
    weight vector (used for the smoothed gradient).
    """
    n = a.shape[0]
    conv = _conv_max_fft(a)    # we need the whole convolution, not only the max
    # Actually we need the full convolution for the soft‑max.
    size = 2 * n - 1
    N = 1 << ((size - 1).bit_length())
    Af = np.fft.rfft(a, N)
    full_conv = np.fft.irfft(Af * Af, N)[:size]

    # soft‑max weights
    expo = np.clip(full_conv / eps, a_min=None, a_max=50.0)
    w = np.exp(expo)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    return w, full_conv.max()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s``
    seconds and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  A few seconds are kept as safety margin.
    kwargs : dict
        Unused – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence (non‑negative, sum > 0.01) discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Randomised parametric search on a smaller problem (n≈2000)
    # ------------------------------------------------------------------
    n_small = 2000
    best_score_small = np.inf
    best_params = None
    best_seq_small = None

    # pre‑computed index vector for the small problem
    i_small = np.arange(n_small, dtype=float)

    # number of random draws we will try – limited only by the time budget
    while time.time() < deadline:
        # sample three parameters:
        #   α ∈ [0.35,0.65]   (power‑law exponent)
        #   τ ∈ [0, 0.002]   (exponential decay)
        #   β ∈ [‑0.5,0.5]   (linear tilt, creates asymmetry)
        alpha = np.random.uniform(0.35, 0.65)
        tau   = np.random.uniform(0.0, 0.002)
        beta  = np.random.uniform(-0.5, 0.5)

        seq = _make_sequence(alpha, tau, beta, n_small)
        if seq is None:
            continue

        conv_max = _conv_max_fft(seq)
        score = 2.0 * n_small * conv_max   # same objective used by evaluate_sequence

        if score < best_score_small:
            best_score_small = score
            best_params = (alpha, tau, beta)
            best_seq_small = seq.copy()
            print(f"[search] new best score={best_score_small:.6f} "
                  f"α={alpha:.4f} τ={tau:.6f} β={beta:.4f}")

        # early exit if we have already beaten the target
        if best_score_small < 1.503:
            break

    # ------------------------------------------------------------------
    # 2. Upscale the best candidate to the target length (≈5000)
    # ------------------------------------------------------------------
    n_target = 5000
    if best_params is not None:
        alpha, tau, beta = best_params
        seq = _make_sequence(alpha, tau, beta, n_target)
        if seq is None:
            # fall back to a plain power‑law if something went wrong
            seq = np.arange(1, n_target + 1, dtype=float) ** (-0.5)
    else:
        # fallback – pure power‑law (the classic starter)
        seq = np.arange(1, n_target + 1, dtype=float) ** (-0.5)

    # normalise to unit sum (the optimisation works on the simplex)
    seq /= seq.sum()
    a = seq.copy()

    # keep the global best discovered sequence (objective on the target size)
    conv_max = _conv_max_fft(a)
    best_score = 2.0 * n_target * conv_max
    best_seq = a.copy()
    print(f"[init] n={n_target}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Fine‑grained smoothed max‑gradient descent (few thousand iters)
    # ------------------------------------------------------------------
    # Hyper‑parameters – chosen to be conservative given the short
    # remaining budget.
    lr = 0.15
    lr_decay = 0.9999
    lr_min = 1e-7

    sigma = 0.6          # smoothing factor multiplier of the current max
    sigma_decay = 0.9999
    sigma_min = 0.02

    momentum = np.zeros_like(a)
    iteration = 0

    # allocate a few seconds for the refinement (max 30 s or whatever is left)
    refine_deadline = min(deadline, time.time() + 28.0)

    while time.time() < refine_deadline:
        iteration += 1

        # (a) compute convolution and the current maximum
        conv_full = _conv_max_fft(a)   # we only need the max value here
        max_conv = conv_full

        # (b) update the smoothing parameter ε = σ·max(conv)
        sigma = max(sigma * sigma_decay, sigma_min)
        eps = max(sigma * max_conv, 1e-12)

        # (c) soft‑max weights (full convolution needed for the gradient)
        size = 2 * n_target - 1
        N = 1 << ((size - 1).bit_length())
        Af = np.fft.rfft(a, N)
        full_conv = np.fft.irfft(Af * Af, N)[:size]

        expo = np.clip(full_conv / eps, a_min=None, a_max=50.0)
        w = np.exp(expo)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # (d) gradient of the smoothed objective:
        #     ∇f = v + a ,   v_j = Σ_i w_{i+j} a_i
        # using a convolution (a * reversed_w)
        w_rev = w[::-1]
        v_full = np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(w_rev, N), N)[:3 * n_target - 2]
        # the needed slice (see original code)
        idx = (2 * n_target - 2) - np.arange(n_target)
        v = v_full[idx]

        grad = v + a

        # (e) momentum term
        momentum = 0.9 * momentum + grad

        # (f) gradient step & projection onto simplex
        a = a - lr * momentum
        a = _project_onto_simplex(a)

        # (g) learning‑rate decay
        lr = max(lr * lr_decay, lr_min)

        # (h) periodic tiny random mass‑swap to escape shallow plateaus
        if iteration % 120 == 0:
            i, j = np.random.choice(n_target, 2, replace=False)
            delta = min(a[i] * 0.3, a[i] * lr)
            if delta > 0.0:
                a[i] -= delta
                a[j] += delta
                a = _project_onto_simplex(a)

        # (i) evaluate exact objective
        conv_max_now = _conv_max_fft(a)
        cur_score = 2.0 * n_target * conv_max_now
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f}, iterations={iteration}")
    return best_seq.tolist()
