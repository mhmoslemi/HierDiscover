def propose_candidate(seed=42, budget_s=1000):
    # Setup random seed
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 1.0

    # <choose lengths>
    lengths = [2000, 4000, 8000]  # maybe order ascending, fallback
    best_val = float('inf')
    best_seq = None

    for n in lengths:
        if time.time() > deadline:
            break
        # Generate seeds
        seeds = []
        # uniform
        seeds.append(np.ones(n)/n)
        # sine window
        seeds.append(_sine_window(n))
        # geometric r = 1 - 2/n
        seeds.append(_geom(n, max(0.0, 1-2/n)))
        # power law p from 0.0 to 2
        for p in [0.0, 0.5, 1.0, 1.5, 2.0]:
            seeds.append(_power(n,p))
        # random 10 seeds
        for _ in range(10):
            logits = rng.standard_normal(n)
            seeds.append(_softmax(logits))
        # DFT sqrt uniform seed
        try:
            L = 2*n
            # uniform conv target = 1/(2*n-1) for each entry length=2*n-1 (except edges)
            # Let's approximate target conv vector t of length L-1:
            t = np.full(2*n-1, 1.0/(2*n-1))
            # Want a such that conv = t; a = idft(sqrt(fft(t))) but need real.
            spec = np.fft.rfft(t)
            sqrt_spec = np.sqrt(np.maximum(spec,0))
            a_complex = np.fft.irfft(sqrt_spec, n=L)
            a = a_complex[:n]  # take first n components
            a = np.maximum(a, 0)
            if a.sum() > 0:
                a /= a.sum()
                seeds.append(a)
        except Exception:
            pass
        # Add previous best sequence if length matches
        prev = globals().get("height_sequence_1")
        if prev is not None:
            try:
                prev_arr = np.asarray(prev, dtype=np.float64)
                if prev_arr.shape[0] == n:
                    seeds.append(prev_arr/prev_arr.sum())
            except Exception:
                pass

        # Evaluate seeds quick:
        for s in seeds:
            val = evaluate_sequence(s.tolist())
            if val < best_val:
                best_val, best_seq = val, s.copy()
                print(f"[seed] n={n} new best {val:.6f}")

        # Phase 1: Adam with smoothed max
        for seed_idx, s0 in enumerate(seeds):
            if time.time()>deadline:
                break
            # Run Adam
            a_opt, val_opt = _adam_optimize_v2(s0, n, deadline, max_iters=5000)
            if a_opt is not None and val_opt < best_val:
                best_val, best_seq = val_opt, a_opt.copy()
                print(f"[Adam] n={n} seed={seed_idx} improved to {val_opt:.6f}")

        # Phase 2: Subgradient refine
        if best_seq is not None and len(best_seq)==n:
            a_ref, val_ref = _subgrad_refine_v2(best_seq, n, deadline, max_iters=30000)
            if a_ref is not None and val_ref < best_val:
                best_val, best_seq = val_ref, a_ref.copy()
                print(f"[Subgrad] n={n} refined best {val_ref:.6f}")

        # Phase 3: Random local search (hill climbing / simulated annealing)
        if best_seq is not None and len(best_seq) == n:
            a_search, val_search = _random_local_search(best_seq, n, deadline,
                                                          max_iters=20000)
            if a_search is not None and val_search < best_val:
                best_val, best_seq = val_search, a_search.copy()
                print(f"[Random] n={n} local search best {val_search:.6f}")

    # Return best_seq as list
    if best_seq is None:
        # fallback
        best_seq = np.ones(4000)/4000
    print(f"=== Final best objective {best_val:.6f} length {len(best_seq)} ===")
    return best_seq.tolist()

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())
def _conv_fft(a,b):
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa*fb, n=N)[:L]
    return conv
def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    sum_e = e.sum()
    if sum_e == 0:
        return np.full_like(v, 1.0 / v.size)
    return e / sum_e
def _uniform(n):
    return np.ones(n, dtype=np.float64) / n
def _sine_window(n):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)
def _geom(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)
def _power(n, p):
    idx = np.arange(1, n+1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)
def _geom_power(n, r, p):
    idx = np.arange(1, n+1, dtype=np.float64)
    a = (r ** (idx-1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)
def _sym_geom(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    b = r ** np.arange(n-1, -1, -1, dtype=np.float64)
    a = a + b
    s = a.sum()
    return a / s if s != 0 else _uniform(n)

def _make_seeds(n, previous=None, rng=None):
    seeds = []
    # baseline uniform
    seeds.append(_uniform(n))
    # sine window
    seeds.append(_sine_window(n))
    # geometric with r in range
    for c in np.linspace(0.5, 4.0, 7):
        r = 1.0 - c / n
        if r > 0:
            seeds.append(_geom(n, r))
    # power law
    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))
    # geometric*power
    for c in np.linspace(0.5, 3.0, 5):
        r = 1.0 - c / n
        if r <= 0: continue
        for p in np.linspace(0.0, 1.5, 5):
            seeds.append(_geom_power(n, r, p))
    # symmetric geometric
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r > 0:
            seeds.append(_sym_geom(n, r))
    # random softmax seeds
    if rng is None:
        rng = np.random.default_rng()
    for _ in range(10):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))
    # DFT sqrt uniform seed
    try:
        # target uniform convolution distribution length = 2n-1
        target = np.full(2*n-1, 1.0/(2*n-1), dtype=np.float64)
        # compute sqrt of its Fourier transform
        spec = np.fft.rfft(target)
        sqrt_spec = np.sqrt(np.maximum(spec, 0.0))
        # inverse transform to get approx a (zero-padded to length 2n)
        a_full = np.fft.irfft(sqrt_spec, n=2*n)
        a = a_full[:n]
        a = np.maximum(a, 0.0)
        if a.sum() > 0:
            a = a / a.sum()
            seeds.append(a)
    except Exception:
        pass
    # previous best if matches length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())
    return seeds

def _adam_optimize(seed, n, deadline, max_iters=8000):
    """
    Adam on smoothed max (log-softmax parametrisation).
    Returns best sequence (numpy array) and best value (2*n*max(conv)).
    """
    v = np.log(seed + 1e-30)  # log-softmax params
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.8
    beta_smooth = 0.5  # start smoothing factor
    beta_mul = 1.02
    best_seq = None
    best_val = float('inf')
    t = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)  # sum to 1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:4d} best={cur_val:.6f}")
        # smoothed max
        beta_smooth *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0:
            continue
        w = w / w_sum
        # gradient of weighted sum conv wrt a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length 3n-2
        grad_a = 2.0 * full[n-1:n-1+n]
        # gradient wrt v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** t)
        v2_hat = v2 / (1 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)
    return best_seq, best_val
```

def _subgrad_refine(start_seq, n, deadline, max_iters=30000):
    """
    Mirror‑descent (exponentiated‑gradient) on the simplex using the exact
    sub‑gradient of max(conv). Returns best sequence and its objective.
    """
    a = start_seq.copy()
    # ensure nonnegativity and normalization
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = _uniform(n)
    else:
        a /= s
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")
        # compute subgradient g
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # exponentiated gradient update
        lr = 0.6 / np.sqrt(it)  # schedule
        # avoid zero
        log_a = np.log(a + 1e-30) - lr * g
        # stability shift
        log_a = log_a - np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0:
            a = _uniform(n)
        else:
            a = a / s
    return best_seq, best_val

def _random_local_search(start_seq, n, deadline, max_iters=5000, init_step=0.05, init_temp=1e-3):
    """
    Simple hill‑climbing / simulated‑annealing style local search.
    Moves a small amount of mass between two positions.
    Returns best sequence and its objective.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_val = 2.0 * n * np.max(conv)
    best_seq = a.copy()
    best_val = cur_val
    step = init_step
    temp = init_temp
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # decrease step and temperature gradually
        # exponential decay
        step_factor = 0.9995
        temp_factor = 0.9995
        step = max(step * step_factor, 1e-5)
        temp = max(temp * temp_factor, 1e-8)
        # choose i (source) with a_i > 0
        # we can sample from distribution proportional to a to increase chance
        # Use rng.choice with probabilities a
        i = rng.choice(n, p=a/a.sum())
        # Choose j different from i uniformly
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # amount to move: delta = step * a[i] * random
        max_move = a[i]
        if max_move <= 0:
            continue
        delta = step * max_move * rng.random()
        # apply move
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # ensure non-negativity (should hold)
        # evaluate candidate
        conv_cand = _conv_fft(a_candidate, a_candidate)
        cand_val = 2.0 * n * np.max(conv_cand)
        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            # probabilistic acceptance
            prob = np.exp(-(cand_val - cur_val) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            a = a_candidate
            conv = conv_cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = a.copy()
                print(f"[Random] n={n} it={it:5d} best={cand_val:.6f}")
    return best_seq, best_val
```

def _mix_with_candidates(base_seq, n, deadline, alphas=None):

    # also include previous best seeds generation?
    best_seq = base_seq.copy()
    best_val = 2.0 * n * np.max(_conv_fft(best_seq, best_seq))
    for cand in candidates:
        for alpha in alphas:

2. **Adam on a smoothed max** – parameters are the logits of a soft‑max
   (so the iterate stays on the probability simplex).  The loss is a
   smoothed version of the max (log‑sum‑exp) with a temperature that is
   increased gradually.  Adam updates keep the iterates on the simplex.

3. **Exact sub‑gradient mirror descent** – starting from the best Adam
   result we perform exponentiated‑gradient updates using the true
   sub‑gradient of the max‑convolution term.  The learning‑rate decays
   as 1/√t.

4. **Random local search** – a simple simulated‑annealing hill‑climbing
   that moves a small amount of mass between two coordinates and accepts
   worsening moves with a temperature schedule.

5. **Convex mixing** – the current best sequence is mixed with a few
   simple candidates (uniform, sine‑window, symmetric geometric) for a
   range of mixing weights; the best mixture is kept.

All stages stop as soon as the global deadline is reached.  The best
sequence found (as a plain Python list) is returned, and it is also
stored in the global variable ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT.
    Returns the full result of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


# ----------------------------------------------------------------------
# Seed families (all returned already normalised to sum 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    b = r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a = a + b
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _make_seeds(n: int, previous: np.ndarray | None = None, rng: np.random.Generator = None) -> list[np.ndarray]:
    """Create a modest collection of diverse starting points for length `n`."""
    if rng is None:
        rng = np.random.default_rng()
    seeds: list[np.ndarray] = []

    # baseline families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.0, 7):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))

    # geometric × power mix
    for c in np.linspace(0.5, 3.0, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.5, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # random soft‑max vectors
    for _ in range(10):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # DFT‑sqrt‑uniform (may contain small negatives → project)
    try:
        target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)
        spec = np.fft.rfft(target)
        sqrt_spec = np.sqrt(np.maximum(spec, 0.0))
        a_full = np.fft.irfft(sqrt_spec, n=2 * n)
        a = a_full[:n]
        a = np.maximum(a, 0.0)
        if a.sum() > 0.0:
            a = a / a.sum()
            seeds.append(a)
    except Exception:
        pass

    # reuse previously discovered sequence if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max loss
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation of the smoothed max of the auto‑convolution.
    Returns the best sequence (as a numpy array) encountered and its
    objective value (2·n·max(conv) because sum(a)=1).
    """
    v = np.log(seed + 1e-30)                     # log‑softmax parameters
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.8

    # smoothing schedule
    beta_smooth = 0.5
    beta_mul = 1.02

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                     # stays on the simplex

        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:4d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta_smooth *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta_smooth * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w·conv wrt a -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- gradient wrt logits v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Sub‑gradient refinement (exact max, mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgrad_refine(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 25000) -> tuple[np.ndarray | None, float]:
    """
    Refine a sequence using the exact sub‑gradient of max(conv).
    Exponentiated‑gradient (mirror‑descent) updates keep the vector on
    the probability simplex.
    """
    a = np.maximum(start_seq, 0.0)
    s = a.sum()
    if s == 0.0:
        a = _uniform(n)
    else:
        a = a / s

    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # Sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Mirror‑descent (exponentiated‑gradient) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)               # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random local search (simple simulated‑annealing)
# ----------------------------------------------------------------------


def _random_local_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 10000,
                         init_step: float = 0.05,
                         init_temp: float = 1e-3) -> tuple[np.ndarray | None, float]:
    """
    Hill‑climbing / simulated‑annealing style search that moves a tiny amount
    of mass between two positions.  Returns the best sequence found.
    """
    rng = np.random.default_rng()
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_val = 2.0 * n * np.max(conv)

    best_seq = a.copy()
    best_val = cur_val
    step = init_step
    temp = init_temp

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # decay of step size and temperature
        step = max(step * 0.9995, 1e-5)
        temp = max(temp * 0.9995, 1e-8)

        # pick source index i proportional to its mass (more mass → larger moves)
        i = rng.choice(n, p=a / a.sum())
        if a[i] <= 0.0:
            continue

        # pick a distinct target index uniformly
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # amount to transfer
        delta = step * a[i] * rng.random()
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta

        conv_cand = _conv_fft(a_cand, a_cand)
        cand_val = 2.0 * n * np.max(conv_cand)

        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(cand_val - cur_val) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_cand
            conv = conv_cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = a.copy()
                print(f"[Random] n={n} it={it:5d} best={cand_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mixing with simple candidates
# ----------------------------------------------------------------------


def _mix_with_candidates(base_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         alphas: list[float] | None = None) -> tuple[np.ndarray, float]:
    """
    Convexly mixes `base_seq` with a few simple reference sequences.
    Returns the best mixture found.
    """
    if alphas is None:
        alphas = [0.0, 0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]

    # reference candidates
    candidates = [_uniform(n), _sine_window(n), _sym_geom(n, 0.9)]

    best_seq = base_seq.copy()
    best_val = 2.0 * n * np.max(_conv_fft(best_seq, best_seq))

    for cand in candidates:
        for alpha in alphas:
            if time.time() > deadline:
                break
            mix = (1.0 - alpha) * base_seq + alpha * cand
            mix = np.maximum(mix, 0.0)
            s = mix.sum()
            if s == 0.0:
                continue
            mix /= s
            cur_val = 2.0 * n * np.max(_conv_fft(mix, mix))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = mix.copy()
                print(f"[Mix] alpha={alpha:.5f} updated best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    evaluation objective.  The routine respects the given time budget
    and returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # possible lengths (shorter lengths are cheaper to explore)
    lengths = [2000, 4000, 6000, 8000]

    best_seq = None
    best_val = float("inf")

    # previous global sequence, if any
    prev_global = globals().get("height_sequence_1")
    if prev_global is not None:
        try:
            prev_arr = np.asarray(prev_global, dtype=np.float64)
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # Seed generation
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous=prev_arr, rng=rng)

        # ------------------------------------------------------------------
        # Quick scan of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            val = evaluate_sequence(s.tolist())
            if val < best_val:
                best_val, best_seq = val, s.copy()
                print(f"[seed] n={n} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam on smoothed max
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=5000)
            if seq is not None and val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Adam] n={n} seed={idx} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Exact sub‑gradient mirror descent
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            seq_ref, val_ref = _subgrad_refine(best_seq, n, deadline, max_iters=25000)
            if seq_ref is not None and val_ref < best_val:
                best_val, best_seq = val_ref, seq_ref.copy()
                print(f"[SubGrad] n={n} refined best={val_ref:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Random local search (simulated annealing)
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            seq_rand, val_rand = _random_local_search(best_seq, n, deadline, max_iters=10000)
            if seq_rand is not None and val_rand < best_val:
                best_val, best_seq = val_rand, seq_rand.copy()
                print(f"[Random] n={n} local‑search best={val_rand:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – Convex mixing with simple candidates
        # ------------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            seq_mix, val_mix = _mix_with_candidates(best_seq, n, deadline)
            if seq_mix is not None and val_mix < best_val:
                best_val, best_seq = val_mix, seq_mix.copy()
                print(f"[Mix] n={n} mixing improved best={val_mix:.6f}")

    # ----------------------------------------------------------------------
    # Final sanity check and fallback
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = _uniform(n)
        best_val = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform distribution")
    else:
        # ensure the final value is computed with the official evaluator
        best_val = evaluate_sequence(best_seq.tolist())

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    # expose for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
