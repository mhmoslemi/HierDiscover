def solve_lp(seq, M):
    n = len(seq)
    # Build sparse matrix A where A @ b <= M
    from scipy.sparse import diags
    import numpy as np
    A = diags(seq, offsets=range(n), shape=(2*n-1, n), format='csc')
    # Setup bounds
    bounds = [(0, None)] * n
    # objective: maximize sum b => minimize -sum b
    c = -np.ones(n)
    # RHS
    b_ub = np.full(2*n-1, M, dtype=float)
    # Solve
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return current seq (no improvement)
        return seq.copy()
    return res.x
```

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# define helper functions _next_pow2, _conv_fft, _score_fft, _project_simplex, etc.

def _lp_optimize(seq, M):
    n = seq.shape[0]
    A = diags(seq, offsets=np.arange(n), shape=(2*n-1, n), format='csc')
    b_ub = np.full(2*n-1, M, dtype=float)
    bounds = [(0, None)] * n
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # return original seq as fallback
        return seq
    return res.x
```

Note: "offsets" argument must be array of ints; np.arange(n) yields array.

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

The approach consists of three stages:

1. **Coarse low‑dimensional optimisation** – a short linear‑program
   (LP) is solved for a modest length (≈2000).  With the current sequence
   f we maximise the total mass Σb_j under the linear constraints
   (f∗b)_k ≤ M, where M = max(f∗f).  The LP yields a direction that
   strictly increases the total mass while keeping the convolution
   bound unchanged.  Mixing f with the normalised LP‑solution reduces
   the peak M.

2. **Upsampling** – the improved short sequence is interpolated to a
   larger length (≈18000).  This preserves the shape that flattens the
   convolution while allowing a finer discretisation.

3. **Fine‑grained projected gradient descent** – the large sequence is
   refined with a projected gradient method that follows the true
   sub‑gradient of the max‑entry of the self‑convolution.  The gradient
   is computed from all indices attaining the current maximum (exact
   sub‑gradient).  The iterate is projected onto the probability simplex
   after each step.

All work respects the 1000 s budget, prints progress and finally returns
the best sequence found (non‑negative, Σ>0.01).  The routine uses only
NumPy, SciPy and the standard library.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Generate a normalised non‑negative power‑law sequence
    a_i ∝ (i+1+offset)^{‑α}.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Linear self‑convolution using a real FFT (full length 2n‑1)."""
    n = seq.shape[0]
    size = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    conv[conv < 0.0] = 0.0
    return conv


def _score_fft(seq: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a normalised sequence
    (Σseq == 1).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_step_exact(seq: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step that uses the true sub‑gradient
    (all indices attaining the current maximum of the convolution).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    M = conv.max()
    max_idx = np.where(conv >= M - 1e-12)[0]          # all active constraints
    grad = np.zeros_like(seq)

    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1, dtype=int)
        grad[i_range] += seq[k - i_range]

    grad *= 2.0
    # make the step orthogonal to the affine hyperplane Σx = 1
    grad -= grad.mean()
    new_seq = seq - step * grad
    new_seq = _project_simplex(new_seq)
    return new_seq


def _run_gradient(
    seq: np.ndarray,
    time_budget: float,
    *,
    max_iters: int = 10 ** 9,
    step0: float = 0.5,
    decay: float = 0.9999,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for a fixed length sequence.
    Returns the best sequence seen and its score.
    """
    start = time.time()
    best_seq = seq.copy()
    best_score = _score_fft(seq)

    step = step0
    it = 0
    while it < max_iters and (time.time() - start) < time_budget:
        it += 1
        seq = _gradient_step_exact(seq, step)

        cur_score = _score_fft(seq)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[grad] it={it}, score={best_score:.6f}")

        step = max(step * decay, 1e-9)

    return best_seq, best_score


def _lp_optimize(seq: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the linear program

        maximise   Σ b_i
        subject to  (seq * b)_k ≤ M   for all k,
                    b_i ≥ 0.

    Returns the optimal b (may be all‑zeros if the solver fails).
    """
    n = seq.shape[0]
    # Build the (2n‑1) × n constraint matrix A where
    # A[k,i] = seq[k‑i]  if 0≤k‑i<n else 0.
    # scipy.diags creates exactly this Toeplitz structure.
    A = diags(seq, offsets=np.arange(n), shape=(2 * n - 1, n), format='csc')
    rhs = np.full(2 * n - 1, M, dtype=float)
    bounds = [(0, None)] * n
    # maximise Σb  ⇔ minimise −Σb
    c = -np.ones(n, dtype=float)

    res = linprog(c, A_ub=A, b_ub=rhs, bounds=bounds, method='highs')
    if not res.success:
        # fall back to the current sequence (no improvement)
        return np.zeros_like(seq)
    return res.x


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  Uses a low‑dimensional LP to obtain a good
    starting shape, upsamples it, and finally refines with projected
    gradient descent.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Low‑dimensional LP‑based optimisation (n ≈ 2000)
    # ------------------------------------------------------------------
    n_small = 2000
    # initialise with a random power‑law exponent
    alpha0 = np.random.uniform(0.4, 0.7)
    seq = _power_sequence(n_small, alpha0)
    best_seq = seq.copy()
    best_score = _score_fft(best_seq)
    print(f"[init] n={n_small}, α={alpha0:.5f}, score={best_score:.6f}")

    # perform a few LP‑updates; each iteration reduces the convolution peak
    while time.time() < deadline - 30.0 and best_score > 1.55:
        conv = _conv_fft(best_seq)
        M = conv.max()
        b = _lp_optimize(best_seq, M)
        if b.sum() == 0.0:                           # LP failed – give up
            break
        # normalise the LP solution to sum = 1 (same scale as best_seq)
        g = b / b.sum()
        # convex combination (step size t)
        t = 0.4
        cand = (1 - t) * best_seq + t * g
        cand = _project_simplex(cand)
        cand_score = _score_fft(cand)
        print(f"[lp] t={t:.2f}, sum(b)={b.sum():.6f}, score={cand_score:.6f}")
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = cand.copy()
        else:
            break                                 # no further improvement

    # ------------------------------------------------------------------
    # 2. Upsample to a large length (≈18000) for final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        n_large = 18000
        # linear interpolation from the small optimum
        x_small = np.linspace(0.0, 1.0, n_small)
        x_large = np.linspace(0.0, 1.0, n_large)
        large_seq = np.interp(x_large, x_small, best_seq)
        large_seq = np.maximum(large_seq, 0.0)
        large_seq /= large_seq.sum()
        # ------------------------------------------------------------------
        # 3. Fine‑grained projected gradient descent on the large sequence
        # ------------------------------------------------------------------
        remaining = max(0.0, deadline - time.time())
        large_seq, final_score = _run_gradient(
            large_seq,
            time_budget=remaining,
            max_iters=10 ** 9,
            step0=0.4,
            decay=0.99995,
        )
        best_seq = large_seq
        best_score = final_score
        print(f"[final] score={best_score:.6f}, length={n_large}")

    else:
        # not enough time for up‑sampling; return the small‑scale optimum
        print("[skip] insufficient time for up‑sampling, returning small solution")
        final_score = best_score

    # final safety projection onto the simplex
    best_seq = _project_simplex(best_seq)

    print(f"[return] best_score={best_score:.6f}, length={best_seq.shape[0]}")
    return best_seq.tolist()
