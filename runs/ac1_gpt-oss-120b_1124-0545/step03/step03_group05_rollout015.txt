def _adam_hardmax_step(a, a_rev, size1, size2, m, v, t, lr, beta1, beta2, epsilon):
    conv_full = _full_convolution(a, size1)
    max_b = conv_full.max()
    # Determine mask: indices where conv is within tolerance of max_b.
    tol = max_b * 1e-12 + 1e-12
    mask = conv_full >= (max_b - tol)
    weight = mask.astype(np.float64) / mask.sum()
    g_full = _conv_fft(a_rev, weight, size2)
    n = a.shape[0]
    grad = 2.0 * g_full[n-1:n-1+n]   # gradient to ascend objective; for descent we subtract
    # Adam moments:
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad**2)
    # bias correction
    t_adj = t + 1  # iteration index start at 0
    m_hat = m / (1 - beta1**t_adj)
    v_hat = v / (1 - beta2**t_adj)
    # update
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
    # project onto nonnegative simplex
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a) / len(a)
    else:
        a_new /= s
    return a_new, m, v

"""
Propose a low‑value step–function.

The search consists of two stages within the 1000 s budget:

1️⃣  Adam‑guided sub‑gradient descent on a smooth approximation of
    the peak of the self‑convolution.  The sub‑gradient is taken from
    the (almost) maximal convolution entries (hard max), which gives a
    very accurate descent direction.  After each step the vector is
    projected onto the probability simplex.

2️⃣  A fast simulated‑annealing / mass‑transfer phase.  A move consists
    of transferring a small amount of probability mass from one index
    to another.  The convolution can be updated analytically using only
    three affected entries, so the new maximal value is obtained in
    O(1) (apart from occasional full scans when the current maximum is
    touched).  A Metropolis rule with a decaying temperature explores
    the neighbourhood and finishes the optimisation.

Both phases keep the best sequence seen and finally return it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution via FFT (real‑to‑real)."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Full linear convolution a * a (length 2·len(a)-1)."""
    return _conv_fft(a, a, size)


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a random fraction of mass from one entry to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # renormalise (should stay 1, but safe against round‑off)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= s
    return a_new


def _adam_hardmax_step(
    a: np.ndarray,
    a_rev: np.ndarray,
    size1: int,
    size2: int,
    m: np.ndarray,
    v: np.ndarray,
    it: int,
    lr: float,
    beta1: float,
    beta2: float,
    eps: float,
) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    One Adam step using a hard‑max (sub‑gradient) weight.

    Returns the updated a, and the new first‑ and second‑moment vectors.
    """
    # convolution a*a and mask of active (max) entries
    conv_full = _full_convolution(a, size1)
    max_b = conv_full.max()
    tol = max_b * 1e-12 + 1e-14
    mask = conv_full >= (max_b - tol)
    weight = mask.astype(np.float64)
    weight_sum = weight.sum()
    if weight_sum == 0.0:
        weight = np.ones_like(weight) / weight.size
    else:
        weight /= weight_sum

    # gradient of the smooth max (descent direction = -gradient)
    g_full = _conv_fft(a_rev, weight, size2)
    n = a.shape[0]
    grad = 2.0 * g_full[n - 1 : n - 1 + n]       # true gradient (ascent)

    # Adam moment updates
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    it1 = it + 1
    m_hat = m / (1.0 - beta1 ** it1)
    v_hat = v / (1.0 - beta2 ** it1)

    # descent step
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

    # project onto the simplex (non‑negative, sum = 1)
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= s

    return a_new, m, v

    m_vec = np.zeros(n, dtype=np.float64)

    beta1 = 0.9
    beta2 = 0.999

    rand_prob = 0.08               # probability of a cheap random transfer

        cur_score = 2.0 * n * max_b

            best_seq = a.copy()

            elapsed = last_improve - start

            a = _random_transfer(a, rand_scale)
            a_rev = a[::-1].copy()
        else:
            a, m_vec, v_vec = _adam_hardmax_step(
                a,
                a_rev,
                size1,
                size2,
                m_vec,
                v_vec,
                it,
                lr,
                beta1,
                beta2,
                eps,
            )
            a_rev = a[::-1].copy()
        it += 1

        # occasional restart if we have been stuck for a while
        if time.time() - last_improve > 45.0:
            a = np.random.rand(n)
            a /= a.sum()
            a_rev = a[::-1].copy()
            m_vec.fill(0.0)
            v_vec.fill(0.0)
            last_improve = time.time()
            print(f"[Adam {it:7d}] restart after stagnation")

    # --------------------------------------------------------------
    # Phase 2 – fast simulated annealing with explicit convolution update
    # --------------------------------------------------------------
    a = best_seq.copy()                     # start from the best Adam point
    b = _full_convolution(a, size1)         # full convolution a*a
    cur_max = b.max()
    cur_max_idx = int(np.argmax(b))
    cur_score = 2.0 * n * cur_max
    best_score = cur_score
    best_seq = a.copy()

    # annealing schedules
    T = 0.02                                 # initial temperature
    decay_T = 0.99995                         # exponential cooling
    transfer_scale = 0.25                     # start with relatively big transfers
    decay_ts = 0.9999                         # slowly shrink transfer size

    it2 = 0
    while time.time() < deadline:
        # pick a source index with probability proportional to its mass
        i = np.random.choice(n, p=a)
        if a[i] == 0.0:
            continue
        # pick a distinct target index uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        ai = a[i]
        aj = a[j]
        # transfer a fraction of the source mass
        delta = transfer_scale * ai * np.random.rand()
        if delta <= 0.0:
            continue

        # indices that change in the convolution
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j

        # new values of the three affected entries
        b2i_new = b[idx2i] - 2.0 * delta * ai + delta * delta
        b2j_new = b[idx2j] - 2.0 * delta * aj + delta * delta

        # compute the new maximal value
        if cur_max_idx in (idx2i, idx2j, idxij):
            # the current peak is touched → recompute full max
            b_tmp = b.copy()
            b_tmp[idx2i] = b2i_new
            b_tmp[idx2j] = b2j_new
            b_tmp[idxij] = bij_new
            new_max = b_tmp.max()
        else:
            # only the three updated entries can exceed the old max
            new_max = max(cur_max, b2i_new, b2j_new, bij_new)

        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        # Metropolis acceptance
        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            if np.random.rand() < np.exp(-delta_score / max(T, 1e-12)):
                accept = True

        if accept:
            # apply the move
            a[i] = ai - delta
            a[j] = aj + delta
            b[idx2i] = b2i_new
            b[idx2j] = b2j_new
            b[idxij] = bij_new
            cur_score = new_score
            cur_max = new_max
            cur_max_idx = int(np.argmax(b))

            # record improvement
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - start
                print(f"[Anneal {it2:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # schedule updates
        T *= decay_T
        transfer_scale = max(transfer_scale * decay_ts, 0.001)
        it2 += 1

        # occasional large random jump to escape deep plateaux
        if it2 % 200000 == 0:
            a = np.random.rand(n)
            a /= a.sum()
            b = _full_convolution(a, size1)
            cur_max = b.max()
            cur_max_idx = int(np.argmax(b))
            cur_score = 2.0 * n * cur_max
            print(f"[Anneal {it2:7d}] random restart")

    total = time.time() - start
    print(f"[done] iterations={it + it2}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
