"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three phases:

1️⃣  Analytic exploration – a dense grid (and some random sampling) of
    simple families:
        • product  a_i ∝ ((i+δ)*(n‑1‑i+δ))^{‑p}
        • sum      a_i ∝ (i+δ)^{‑p} + (n‑1‑i+δ)^{‑p}
        • exponential a_i ∝ exp(‑α·i)
        • mixtures of the above
    The best vector found (according to the exact objective) is kept.

2️⃣  Momentum‑based smoothed‑max optimisation.
    The smooth approximation
        Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)
    is minimised by a projected Adam‑style update.
    λ is annealed from a small value (≈0.5) to a large one (≈2000)
    so that the smooth max morphs into a true max.
    After each update the vector is projected onto the probability simplex
    (non‑negative, sum = 1).

3️⃣  Stagnation handling – if no improvement has been seen for a while,
    a small random jitter is added to the current best vector to escape
    plateaus.

The routine is deliberately written with no closures, no lambdas and
uses only NumPy and the standard library, so it can be executed in the
provided evaluation harness.  It respects a 1000 s budget (a 0.5 s safety
margin is kept) and returns the best sequence discovered.
"""

import time

# Simple utilities – no closures

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Zero‑padded linear convolution a * a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant) for a non‑negative vector a.
    The vector is projected onto the probability simplex (sum = 1)
    before evaluation – this matches the behaviour of the harness.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s                     # normalise to sum = 1
    conv = conv_fft(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def gradient_smooth(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k) w.r.t. a.
    Both the convolution a*a and the weighted convolution are performed
    via FFTs.
    """
    n = a.shape[0]

    conv = conv_fft(a)                         # length 2n‑1

    # soft‑max weights, stabilised by subtracting the maximum
    max_c = np.max(conv)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)            # length 2n‑1, sum = 1

    # weighted convolution w_rev * a  (see analysis in the prompt)
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)

    start = n - 1
    end = 2 * n - 1
    slice_part = conv_wa[start:end]                # length n
    g = 2.0 * slice_part[::-1]                     # reverse to get g_i
    return g

# Analytic families
# ----------------------------------------------------------------------


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum = 1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    mode = 'sum'      → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")

    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def exponential_candidate(n: int, alpha: float) -> np.ndarray:
    """Geometric (exponential) decay a_i ∝ exp(-alpha·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a = a / a.sum()
    return a


def mixed_candidate(n: int, p: float, offset: float, w: float) -> np.ndarray:
    """Mixture w·product + (1‑w)·sum, normalised to sum = 1."""
    prod = analytic_candidate(n, p, offset, "product")
    summ = analytic_candidate(n, p, offset, "sum")
    a = w * prod + (1.0 - w) * summ
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation function.
    The routine first explores several analytic families, then refines the
    best vector with a momentum Adam‑style smoothed‑max optimisation,
    applying occasional jitter restarts.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    #   0.  Parameter handling
    # ------------------------------------------------------------------
    # default length – we also try a larger length because the constant
    #  (limit of the objective) improves slowly with n.
    length_choices = [kwargs.get("n", 2000), 4000, 6000]

    best_seq = None
    best_score = np.inf
    best_n = None

    # ------------------------------------------------------------------
    #   1. Analytic families (grid + random exploration)
    # ------------------------------------------------------------------
    print("[init] start analytic exploration")
    for n in length_choices:
        # grid for product / sum families
        p_grid = np.linspace(0.30, 2.00, 18)            # 0.30 … 2.00
        offset_grid = np.array([1e-3, 0.01, 0.05, 0.10, 0.20, 0.50])
        for mode in ("product", "sum"):
            for p in p_grid:
                for off in offset_grid:
                    cand = analytic_candidate(n, p, off, mode)
                    sc = compute_score(cand)
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand.copy()
                        best_n = n
        # exponential family
        for alpha in np.linspace(0.0005, 0.0100, 12):   # crude sweep
            cand = exponential_candidate(n, alpha)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                best_n = n

        # random mixes of product/sum
        for _ in range(200):
            p = np.random.uniform(0.3, 2.5)
            off = np.random.uniform(1e-4, 0.7)
            w = np.random.rand()
            cand = mixed_candidate(n, p, off, w)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                best_n = n

        # random Dirichlet draws (full support)
        for _ in range(300):
            cand = np.random.dirichlet(np.ones(n))
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                best_n = n

        # incorporate a possible global reference (e.g. height_sequence_1)
        if "height_sequence_1" in globals():
            ref = np.asarray(globals()["height_sequence_1"], dtype=float)
            if ref.shape[0] == n:
                ref = np.maximum(ref, 0.0)
                ref = ref / ref.sum()
                sc = compute_score(ref)
                if sc < best_score:
                    best_score = sc
                    best_seq = ref.copy()
                    best_n = n

        print(f"[init] n={n} best score so far = {best_score:.6f}")

    # ------------------------------------------------------------------
    #   2. Smoothed‑max optimisation with Adam momentum
    # ------------------------------------------------------------------
    a = best_seq.copy()               # current working vector (already normalised)
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # annealing schedules
    lam_start, lam_end = 0.5, 2000.0
    step_start, step_end = 0.15, 2e-5
    temp_start, temp_end = 1.0, 0.0005

    iteration = 0
    last_improve = start_time
    stagnation_cooldown = 30.0      # seconds without improvement before a jitter

    while time.time() < deadline:
        elapsed = time.time() - start_time
        # progress ∈ [0,1] (we leave a small margin for the final printouts)
        progress = elapsed / (budget_s - 20.0)
        progress = min(1.0, max(0.0, progress))

        # linear schedule for λ, step size and temperature
        lam = lam_start + progress * (lam_end - lam_start)
        step = step_start + progress * (step_end - step_start)
        temp = temp_start + progress * (temp_end - temp_start)

        # gradient of the smooth‑max
        g = gradient_smooth(a, lam)

        # Adam update (pre‑projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        cand = a - update
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)

        cand_score = compute_score(cand)

        # Metropolis‑style acceptance
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            if temp > 1e-12:
                prob = math.exp(-(cand_score - best_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:07d}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # jitter restart if stuck
        if iteration % 20000 == 0:
            if time.time() - last_improve > stagnation_cooldown:
                jitter = np.random.rand(n) * 1e-3
                a = best_seq + jitter
                a = project_simplex(a)
                m = np.zeros_like(a)
                v = np.zeros_like(a)
                print(f"[{iteration:07d}] jitter restart (stagnation)")

        # early exit when we cross the target
        if best_score <= 1.503:
            print("[done] target reached – exiting early")
            break

        iteration += 1

    total_time = time.time() - start_time
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
