"""
Gradient‑free local search for the step‑function bound.

The objective (with sum a = 1) is  E(a)=2·n·max(conv(a,a)).
We start from a classic arcsine shape (or reuse the previously
found `height_sequence_1`).  At each iteration we identify the
indices that contribute most to the current maximum of the convolution,
compute a one‑dimensional linear model for the effect of moving a small
amount of mass from a “bad’’ index to a “good’’ one, and then solve a
tiny 1‑D convex optimisation (binary search) to find the optimal amount
δ that minimises the maximal convolution value.
The move is accepted if it decreases the true objective.  When no
progress is seen for a while a random jitter is applied to escape
plateaus.  The algorithm respects the time budget and returns the best
sequence encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------


def initialise_a(n: int, seed: int = 42) -> np.ndarray:
    """
    Return a feasible initial probability vector of length `n`.

    If a global variable `height_sequence_1` exists and matches the
    required length it is reused, otherwise the classic arcsine shape
    (inverse‑sqrt law) is returned and normalised to sum‑one.
    """
    rng = np.random.default_rng(seed)

    # reuse previous solution if available
    if "height_sequence_1" in globals():
        a0 = np.array(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.shape[0] == n and np.isfinite(a0).all():
            a0 = a0 / a0.sum()
            return a0

    # arcsine (inverse‑sqrt) initial guess
    i = np.arange(n, dtype=float)
    a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a0 = a0 / a0.sum()
    return a0


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution a*a using a real FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    # size = next power of two >= 2*n-1
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, L)
    conv_full = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv_full


def gradient_wrt_a(a: np.ndarray, max_inds: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient of max(conv(a,a)) w.r.t. a.
    For each index i, g_i = Σ_{k∈max_inds} a_{k-i} (ignoring out‑of‑range).
    """
    n = a.shape[0]
    g = np.zeros(n, dtype=float)
    for k in max_inds:
        # the range of i where k-i is a valid index
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # vector of a[k-i] for i in [i_start .. i_end]
        idx = k - np.arange(i_start, i_end + 1)
        g[i_start : i_end + 1] += a[idx]
    return g


def optimal_delta(conv: np.ndarray,
                  a: np.ndarray,
                  src: int,
                  dst: int,
                  delta_limit: float,
                  eps: float = 1e-14) -> float:
    """
    For a fixed source `src` and destination `dst` compute the δ∈[0,delta_limit]
    that minimises max(conv + s·δ) where s_k = 2·(a_{k-dst}−a_{k‑src}).
    The function is convex and piecewise linear; a short binary search finds
    the minimizing point.
    """
    n = a.shape[0]
    length = 2 * n - 1
    idx = np.arange(length)

    src_idx = idx - src
    dst_idx = idx - dst
    a_src = np.where((src_idx >= 0) & (src_idx < n), a[src_idx], 0.0)
    a_dst = np.where((dst_idx >= 0) & (dst_idx < n), a[dst_idx], 0.0)

    s = 2.0 * (a_dst - a_src)               # slope of each convolution entry

    # early exit: if every slope belonging to the current maximal entries
    # is non‑negative, moving mass cannot lower the max.
    max_val = conv.max()
    active = np.where(conv >= max_val - eps)[0]
    if np.all(s[active] >= 0):
        return 0.0

    lo, hi = 0.0, delta_limit
    # 20–25 iterations give sub‑machine‑epsilon precision
    for _ in range(22):
        mid = (lo + hi) * 0.5
        tmp = conv + s * mid
        cur = tmp.max()
        # slopes of entries that are currently maximal
        mask = tmp >= cur - eps
        if np.any(s[mask] < 0):
            lo = mid          # we can still go up, max will still go down
        else:
            hi = mid
    return lo

def propose_candidate(seed: int = 42,

    """

    # final safety: non‑negative values, no NaNs/Infs
    best_a = np.maximum(best_a, 0.0)
    best_a = np.where(np.isfinite(best_a), best_a, 0.0)

    # expose the best sequence as a global for possible reuse
    globals()["height_sequence_2"] = best_a.tolist()
    return best_a.tolist()
