"""
Propose a candidate step‑function (non‑negative heights) that minimises
evaluate_sequence().
The algorithm builds a diverse set of analytically‑motivated seeds,
selects the best one by a fast convolution, then refines it in two
stages:

1. Adam optimisation on a smooth approximation of the max‑convolution.
   The smoothing temperature β grows slowly up to a moderate cap (≈30)
   so that the gradient does not vanish.

2. Hard‑max sub‑gradient descent (projected onto the probability simplex)
   which directly minimises the true supremum of the convolution.

The two stages are repeated from several perturbed restarts until the
time budget (default 1000 s) is exhausted.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Output length = len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of [Wang & Carreira‑Perpiñán, 2013].
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Seed families (all return a probability vector of length n)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _mix_geometric(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    """Mixture α·geom(r1)+(1‑α)·geom(r2) (renormalised)."""
    a = alpha * _geom(n, r1) + (1.0 - alpha) * _geom(n, r2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _mix_power(n: int, p1: float, p2: float, alpha: float) -> np.ndarray:
    a = alpha * _power(n, p1) + (1.0 - alpha) * _power(n, p2)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Objective (fast version) – 2·n·max(conv) assuming sum=1
# ----------------------------------------------------------------------


def _fast_obj(a: np.ndarray, n: int) -> float:
    """Fast objective: 2·n·max(a * a).  a is assumed to sum to 1."""
    return 2.0 * n * np.max(_conv_fft(a, a))


# ----------------------------------------------------------------------
# Adam optimiser on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------


def _refine_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 2.0,
    max_beta: float = 30.0,
    beta_mul: float = 1.01,
    lr0: float = 0.8,
    max_iters: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.
    Returns (best_sequence, best_objective).
    """
    v = np.log(a0 + 1e-30)        # logits (real numbers)
    m = np.zeros_like(v)         # first moment
    v2 = np.zeros_like(v)        # second moment
    best_seq = a0.copy()
    best_val = _fast_obj(a0, n)

    beta = start_beta
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        # current probabilities
        a = _softmax(v)

        # evaluate
        cur_val = _fast_obj(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[adam] it={it:6d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta = min(beta * beta_mul, max_beta)
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)

        # numerical stable soft‑max over the convolution entries
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:               # on the unlikely event of underflow
            continue
        w /= w_sum

        # gradient w.r.t. a (smooth loss)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]    # central n entries

        # gradient w.r.t. logits (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient descent with simplex projection
# ----------------------------------------------------------------------


def _refine_subgrad(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 4000,
    lr0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns (best_sequence, best_objective).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _fast_obj(a, n)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, a)
        t_max = int(np.argmax(conv))
        cur_max = conv[t_max]
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[subgrad] it={it:5d} best={cur_val:.6f}")

        # gradient of the chosen lag (sub‑gradient of max)
        rev_a = a[::-1]
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[t_max] = 1.0
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad = 2.0 * full[n - 1 : n - 1 + n]  # central slice

        # step size schedule (decreasing)
        lr = lr0 / np.sqrt(it)

        a = a - lr * grad
        # project back onto the probability simplex
        a = _project_simplex(a)

    return best_a, best_val
