def simplex_projection(v):
    """
    Euclidean projection of a vector v onto the probability simplex {x>=0, sum x = 1}.
    Returns projected vector.
    """
    # reference: Duchi et al. 2008.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Projected Adam optimizer for the “max self‑convolution” problem.

The objective to minimise (up to a harmless scale factor) is

    F(a) = 2·n·max_k (a * a)[k]          (∑a = 1 , a_i ≥ 0)

where `*` denotes linear convolution.  Direct optimisation of the
non‑smooth max is difficult, therefore we minimise a smooth
log‑sum‑exp surrogate

    Sτ(a) = τ·log∑_k exp( (a*a)[k] / τ ).

Grad Sτ = 2·(a_rev ∗ w) with w_i ∝ exp((b_i‑max(b))/τ).
We apply Adam (first‑ and second‑moment adaptation) to the gradient,
project the iterate onto the probability simplex after each step,
and enforce symmetry a_i = a_{n‑1‑i} (the optimal solution is
symmetric).  An exponential schedule slowly reduces τ, turning the
smooth surrogate into the true max.  Random “mass‑transfer” moves
occasionally perturb the current point to escape plateaux, and if no
improvement is observed for a while the iterate is re‑initialised
around the best point found so far.

The routine stops automatically after the allotted wall‑clock budget
(≈ 1000 s) and returns the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT.  ``size`` must be ≥ len(u)+len(v)‑1.
    Returns the full convolution of length len(u)+len(v)‑1.
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size1: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length)."""
    return _conv_fft(a, a, size1)


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of
    Duchi et al., “Efficient Projections onto the ℓ1‑Ball for
    Learning in High Dimensions”, ICML 2008.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a random fraction of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    allowed = a[i]
    if allowed == 0.0:
        return a
    delta = scale * allowed * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # re‑normalise to stay on the simplex
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / n)
    else:
        a_new /= s
    return a_new


def _initial_distribution(n: int) -> np.ndarray:
    """
    Produce a non‑negative symmetric start vector of length ``n``,
    normalised to unit ℓ₁‑mass.
    Preference order:
      1. Global variable ``height_sequence_1`` (interpolated if needed).
      2. Symmetric triangular shape with a small random modulation.
    """
    # 1. try to use the previously discovered sequence
    seq0 = globals().get("height_sequence_1", None)
    if seq0 is not None:
        seq0 = np.asarray(seq0, dtype=float)
        if seq0.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0, 1, seq0.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            a = np.interp(x_new, x_old, seq0)
        else:
            a = seq0.copy()
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.random.rand(n)
        a /= a.sum()
        # enforce symmetry
        a = (a + a[::-1]) * 0.5
        a /= a.sum()
        return a

    # 2. triangular shape (peak at centre) with a tiny random jitter
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - centre) / centre          # triangle ∈[0,1]
    # multiplicative noise – keeps the overall shape
    tri *= (0.9 + 0.2 * np.random.rand(n))
    tri = np.clip(tri, 0.0, None)
    if tri.sum() == 0.0:
        tri = np.ones_like(tri)
    a = tri / tri.sum()
    # enforce exact symmetry (averaging)
    a = (a + a[::-1]) * 0.5
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step function within ``budget_s`` seconds.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the harness gives ≈1000 s).
    kwargs : dict
        Optional ``n`` – length of the sequence (default 6000).

    Returns
    -------
    list[float]
        The best sequence found (list of Python floats).
    """

    start = time.time()

    n = int(kwargs.get("n", 6000))

    size1 = _next_pow2(2 * n - 1)            # for a*a
    size2 = _next_pow2(3 * n - 2)            # for a_rev * weight

    # Adam hyper‑parameters ------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.2                           # base learning‑rate before Adam scaling

    # smoothing schedule --------------------------------------------------
    init_tau_factor = 0.5                    # τ = factor * max_b at start
    tau_decay = 0.9995                       # multiplicative decay per iteration

    # random‑transfer parameters -------------------------------------------
    rand_prob = 0.03                         # how often we perform a cheap move
    rand_scale = 0.04                        # max fraction of source mass to move

    # bookkeeping ---------------------------------------------------------
    best_seq = None
    best_score = np.inf
    total_iters = 0

    # because we keep the ℓ₁‑mass = 1)
    def _score_from_conv(conv_full, sum_a=1.0):
        return float(2.0 * n * conv_full.max() / (sum_a * sum_a))

    # --------------------------------------------------------------------
    # Main optimisation loop (multiple restarts until time runs out)
    # --------------------------------------------------------------------
    while time.time() < deadline:

        a = _initial_distribution(n)

        m = np.zeros_like(a)               # first moment
        v = np.zeros_like(a)               # second moment
        it = 0
        last_improve = time.time()
        tau_factor = init_tau_factor

        while True:

            max_b = conv_full.max()

                best_seq = a.copy()

                elapsed = now - start

            if np.random.rand() < rand_prob:
                a = _random_transfer(a, rand_scale)
                a_rev = a[::-1].copy()
                # no Adam state update – keep momentum consistent
                it += 1
                continue

            # 3) smooth‑max temperature (proportional to current max_b)
            tau = max_b * tau_factor
            tau_factor *= tau_decay
            if tau < 1e-12:               # avoid division by zero
                tau = 1e-12

            weight = np.exp((conv_full - max_b) / tau)
            w_sum = weight.sum()
            if w_sum == 0.0:

            else:

            g_full = _conv_fft(a_rev, weight, size2)

            # bias‑corrected moments

            v_hat = v / (1.0 - beta2 ** (it + 1))

            a = a - step

                a = (a + a[::-1]) * 0.5
                a_rev = a[::-1].copy()

                v.fill(0.0)
                tau_factor = init_tau_factor
                last_improve = time.time()
                print(f"[{total_iters + it:8d}] restart (stagnation)  time={now - start:.1f}s")

            it += 1

        total_iters += it

    # final reporting
    # --------------------------------------------------------------------
    total_time = time.time() - start
    print(f"[done] total_iters≈{total_iters}, best_score={best_score:.6f}, time={total_time:.1f}s")
    # safety: if nothing was found (should never happen) fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
    return best_seq.tolist()
