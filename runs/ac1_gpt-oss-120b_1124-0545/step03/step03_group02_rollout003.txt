    while time.time() < deadline:
        # choose source index i biased by a
        i = np.random.choice(n, p=a)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # step size factor decreasing over time
        elapsed = time.time() - start

        if eps <= 0:
            continue

        conv_new = conv.copy()

        conv_new[j:j+n] += 2.0 * eps * a_old

        idx_sum = i + j
        if idx_sum != 2*i and idx_sum != 2*j:
            conv_new[idx_sum] -= 2.0 * eps*eps
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        if new_score < best_score - 1e-12:
            # accept
            a = a.copy()
            a[i] -= eps
            a[j] += eps
            conv = conv_new
            best_score = new_score
            best_a = a.copy()
            print(f"[hill] iter time={elapsed:.1f}s new best={best_score:.6f}")
        # Optionally occasionally re-project onto simplex after many moves to avoid drift
        if np.random.rand() < 0.001:

            a = np.maximum(a + jitter, 0)
            a = a / a.sum()
            conv = convolve_fft(a)
    print(f"[done] best score={best_score:.6f}")

    best_a = np.maximum(best_a, 0.0)
    # ensure sum not zero
    if best_a.sum() == 0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()
    return best_a.tolist()

"""
A two‑stage optimizer for the step‑function bound.

Stage 1 – cheap parametric search:
    Search a 2‑parameter family a_i ∝ ((i+α)*(n‑1‑i+α))^{-γ}
    (γ≈0.5, α≈0.5 gives the classic arcsine shape).
    Randomly sample (γ,α), evaluate the exact objective
    (2·n·max(conv(a,a))) and keep the best vector.

Stage 2 – fast greedy hill‑climbing:
    Starting from the best vector, repeatedly move a small
    amount ε of mass from an index i (chosen proportionally to a_i)
    to a random index j.  The effect on the convolution
    is updated analytically using two slice operations
    (O(n) per move) and the quadratic correction ε².
    Only moves that improve the exact bound are accepted;
    occasional tiny random jitter (re‑normalisation) helps escape
    plateaus.  The whole procedure respects the 1000 s budget
    and returns a non‑negative sequence with sum = 1,
    which is optimal for the scale‑invariant objective.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, all top‑level)
# ----------------------------------------------------------------------


def convolve_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a using a real FFT."""
    n = a.shape[0]
    # length needed for full convolution = 2*n-1
    fft_len = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    # element‑wise product gives convolution in Fourier domain
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    # numerical noise may create tiny negative values – clip to zero
    conv[conv < 0] = 0.0
    return conv


def exact_score(a: np.ndarray) -> float:
    """Exact evaluation of the objective for a probability vector a (sum=1)."""
    n = a.shape[0]
    conv = convolve_fft(a)
    return float(2.0 * n * conv.max())


def parametric_shape(n: int, gamma: float, alpha: float) -> np.ndarray:
    """
    Generate a symmetric shape a_i ∝ ((i+α)*(n‑1‑i+α))^{-γ}.
    The vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    denom = (i + alpha) * (n - 1 - i + alpha)
    # avoid division by zero – α>0 guarantees positivity
    a = np.power(denom, -gamma)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hill_climb(initial_a: np.ndarray, start: float, deadline: float) -> np.ndarray:
    """
    Greedy local optimisation by transferring a tiny amount of mass.
    Returns the best vector seen before the deadline.
    """
    n = initial_a.shape[0]
    a = initial_a.copy()
    conv = convolve_fft(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()

    # Every iteration moves ε from a source index i to a destination j.
    # i is drawn with probability proportional to a (biased towards large entries).
    # The convolution update uses only O(n) vectorised operations.
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / (deadline - start)       # 0 → 1
        # step size slowly shrinks (initially ≈10 % of a[i])
        step_factor = 0.10 * (1.0 - progress) + 0.002

        # pick a source index weighted by the current mass distribution
        i = np.random.choice(n, p=a)
        # pick a destination uniformly (can be the same, in which case we skip)
        j = np.random.randint(0, n)
        if i == j:
            continue

        eps = step_factor * a[i]                     # amount to move
        if eps <= 0.0:
            continue

        # ---- compute the new convolution ----
        conv_new = conv.copy()
        # linear contribution: shift of the whole vector a (old values!)
        conv_new[i : i + n] -= 2.0 * eps * a
        conv_new[j : j + n] += 2.0 * eps * a

        # quadratic correction (ε² terms)
        conv_new[2 * i] += eps * eps
        conv_new[2 * j] += eps * eps
        idx_sum = i + j
        if idx_sum != 2 * i and idx_sum != 2 * j:
            conv_new[idx_sum] -= 2.0 * eps * eps

        # compute the new bound
        new_score = 2.0 * n * conv_new.max()

        # ----- acceptance test -----
        if new_score + 1e-12 < best_score:           # strict improvement
            # accept the move
            a[i] -= eps
            a[j] += eps
            conv = conv_new
            best_score = new_score
            best_a = a.copy()
            print(f"[hill] it={it:07d} time={elapsed:.1f}s new best={best_score:.6f}")
        else:
            # occasionally perform a tiny random jitter to shake stagnation
            if np.random.rand() < 0.001:
                jitter = np.random.randn(n) * 1e-4
                a = np.maximum(a + jitter, 0.0)
                a /= a.sum()
                conv = convolve_fft(a)

        it += 1

    return best_a


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  The routine respects the given time budget
    (default 1000 s) and returns the best sequence found.
    """
    # --------------------------------------------------------------
    # 1) initialisation
    # --------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # tiny safety margin

    n = int(kwargs.get("n", 2000))

    # Try to reuse a previously discovered sequence if it exists.
    if "height_sequence_1" in globals():
        seq = globals()["height_sequence_1"]
        if isinstance(seq, (list, tuple)) and len(seq) == n:
            a0 = np.array(seq, dtype=float)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = a0 / a0.sum()
        else:
            a0 = None
    else:
        a0 = None

    # Fallback: classic arcsine shape (close to a known optimum)
    if a0 is None:
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a0 = a0 / a0.sum()

    best_a = a0.copy()
    best_score = exact_score(best_a)
    print(f"[init] n={n} start score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Phase 1 – cheap parametric exploration
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.35 * budget_s
    while time.time() < phase1_deadline:
        # Sample around the known exponent 0.5 and offset 0.5
        gamma = np.random.uniform(0.35, 0.70)       # exponent
        alpha = np.random.uniform(0.1, 2.0)         # shift (both ends equal)
        cand = parametric_shape(n, gamma, alpha)
        cand_score = exact_score(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[param] new best={best_score:.6f} (γ={gamma:.4f}, α={alpha:.4f})")

    # --------------------------------------------------------------
    # 3) Phase 2 – greedy hill‑climbing refinement
    # --------------------------------------------------------------
    refined_a = hill_climb(best_a, start_time, deadline)

    # --------------------------------------------------------------
    # 4) final safety & return
    # --------------------------------------------------------------
    refined_a = np.maximum(refined_a, 0.0)
    total = refined_a.sum()
    if total == 0.0:
        refined_a = np.full(n, 1.0 / n)
    else:
        refined_a = refined_a / total

    final_score = exact_score(refined_a)
    print(f"[done] best score={final_score:.6f} (time={time.time() - start_time:.1f}s)")

    # Return a plain Python list as required by the evaluation harness
    return refined_a.tolist()
