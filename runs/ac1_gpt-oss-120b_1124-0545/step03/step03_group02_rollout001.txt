function proj_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function sequence via an accelerated smoothed‑max descent.

The algorithm works on the probability simplex (sum(a)=1) using a
soft‑max parametrisation a = softmax(v).  It minimises the exact objective
E = 2·n·max(conv(a,a)) by optimising a smooth approximation of the
max‑convolution (log‑sum‑exp) with an Adam optimiser.  An exponential
schedule raises the temperature β (tightening the soft‑max) and lowers the
learning‑rate.  The dimension n is taken larger than the previous 2000
(default 5000) to exploit the asymptotic limit of the bound.  Periodic
re‑initialisations (jitter around the current best) avoid stagnation.
After the main optimisation a short projected‑gradient refinement using the
exact max‑convolution is performed in any remaining time.

The function returns the best non‑negative sequence found within the
allocated budget (default 1000 s).  The returned list is normalised to sum
to 1, which is the natural scale for the evaluation function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    shift = x - np.max(x)
    exp_x = np.exp(shift)
    return exp_x / exp_x.sum()


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {w | w_i ≥ 0, Σ w_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def initialise_a(n: int, seed: int = 42) -> np.ndarray:
    """
    Initialise a probability vector of length n.

    If a global variable `height_sequence_1` exists, it is interpolated
    (or reused if the size matches).  Otherwise the classic arcsine‑shape
    a_i ∝ 1 / sqrt((i+0.5)(n-i-0.5)) is used.
    """
    rng = np.random.default_rng(seed)

    # Re‑use a previously found sequence if it is available.
    global_vars = globals()
    if "height_sequence_1" in global_vars:
        old = np.array(global_vars["height_sequence_1"], dtype=float)
        if old.size == n:
            a0 = np.maximum(old, 0.0)
            if a0.sum() > 0:
                return a0 / a0.sum()
        else:
            # Linear interpolation to the new size.
            x_old = (np.arange(old.size) + 0.5) / old.size
            x_new = (np.arange(n) + 0.5) / n
            interp = np.interp(x_new, x_old, old, left=0.0, right=0.0)
            a0 = np.maximum(interp, 0.0)
            if a0.sum() > 0:
                return a0 / a0.sum()

    # Default analytic start (inverse‑sqrt “arcsine” shape).
    i = np.arange(n, dtype=float)
    a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0:
        # Extremely unlikely, but fall back to uniform.
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0.sum()
    return a0


def exact_score(a: np.ndarray) -> float:
    """Exact evaluation of the objective for a probability vector."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max())


def refine_subgradient(a: np.ndarray,
                       time_budget: float) -> np.ndarray:
    """
    A short projected‑gradient refinement that uses the exact maximal
    convolution entry (sub‑gradient of the max).  Runs while time permits.
    """
    start = time.time()
    n = a.shape[0]
    # Initial step‑size – a modest value works well.
    step = 0.05
    # Pre‑allocate FFT length for the inner convolution evaluations.
    fft_len = 1 << ((3 * n - 2).bit_length())

    while time.time() - start < time_budget:
        # Compute convolution via FFT (faster than np.convolve for large n).
        fa = np.fft.fft(a, fft_len)
        conv_full = np.real(np.fft.ifft(fa * fa))[:2 * n - 1]
        pos = int(np.argmax(conv_full))

        # Sub‑gradient of max(conv) w.r.t. a:
        # g_i = 2 * a_{pos - i} for indices that lie inside [0, n-1].
        idx = pos - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # Gradient step and projection onto the simplex.
        a = a - step * g
        a = project_onto_simplex(a)

        # Decay step size slowly.
        step *= 0.9995
    return a


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation function.  The routine respects a time budget (default
    1000 s) and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    n : int, optional
        Length of the sequence (default 5000).  Larger n may improve the
        bound but increases per‑iteration cost.
    """
    # ------------------------------------------------------------------
    # Setup and hyper‑parameters
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    n = int(kwargs.get("n", 5000))

    # Initialise the probability vector and its soft‑max parametrisation.
    a = initialise_a(n, seed=seed)
    v = np.log(a + 1e-12)                     # log‑parametrisation (soft‑max)

    # Adam optimiser state.
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # Schedules (exponential in progress p ∈ [0,1]).
    lr_start, lr_end = 0.15, 0.001
    temp_start, temp_end = 5.0, 3000.0

    # Book‑keeping.
    best_a = a.copy()
    best_score = exact_score(a)
    last_improve_t = start_time
    it = 0

    # Pre‑allocate FFT length (next power of two ≥ 3·n‑2).
    fft_len = 1 << ((3 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # Main optimisation loop (smooth max via log‑sum‑exp)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1
        now = time.time()
        elapsed = now - start_time
        # Progress ratio, clamped to [0,1].
        p = min(1.0, elapsed / (budget_s - 10.0))

        # Exponential schedules.
        lr = lr_start * (lr_end / lr_start) ** p
        beta = temp_start * (temp_end / temp_start) ** p

        # Soft‑max parametrisation guarantees a ∈ simplex.
        a = softmax(v)

        # FFT of a (used repeatedly).
        fa = np.fft.fft(a, fft_len)

        # Convolution a * a (full length 2n‑1).
        conv_aa = np.real(np.fft.ifft(fa * fa))[:2 * n - 1]

        # Smoothed max: w_i ∝ exp(β·conv_i).
        z = beta * conv_aa
        z_max = np.max(z)
        w = np.exp(z - z_max)
        w /= w.sum()                              # shape (2n‑1,)

        # FFT of the weight vector.
        fw = np.fft.fft(w, fft_len)

        # ---------- g2 = (w * a) linear convolution ----------
        conv_wa = np.real(np.fft.ifft(fw * fa))[:3 * n - 2]
        g2 = conv_wa[:n]                          # first n entries

        # ---------- g1 = correlation w ⋆ a (valid part) ----------
        corr_full = np.real(np.fft.ifft(np.conj(fw) * fa))[:3 * n - 2]
        g1 = corr_full[:n]

        # Gradient of the smoothed objective.
        g = 2.0 * n * (g1 + g2)

        # Chain‑rule through the soft‑max parametrisation.
        dot_ag = np.dot(a, g)
        grad_v = a * (g - dot_ag)

        # Adam update (bias‑corrected).
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v -= lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # Exact current score (no smoothing).
        cur_score = 2.0 * n * conv_aa.max()

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve_t = time.time()
            print(f"[{it:07d}] new best score = {best_score:.6f} (t={elapsed:.1f}s)")

        # ------------------------------------------------------------------
        # Stagnation handling – occasional jittered restart.
        # ------------------------------------------------------------------
        if it % 3000 == 0:
            if time.time() - last_improve_t > 30.0:
                jitter = np.random.randn(n) * 1e-3
                v = np.log(best_a + 1e-12) + jitter
                m.fill(0.0)
                v_hat.fill(0.0)
                print(f"[{it:07d}] restart after stagnation (t={elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # Short exact‑max refinement in any remaining time.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        # Use a modest number of steps; the function respects the time budget.
        best_a = refine_subgradient(best_a, time_budget=remaining)

        # Re‑evaluate after refinement.
        refined_score = exact_score(best_a)
        if refined_score < best_score:
            best_score = refined_score
            print(f"[refine] improved score = {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] iterations={it}, best score={best_score:.6f}, total time={total_time:.1f}s")

    # Ensure non‑negativity, renormalise to sum = 1 (the natural scale).
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() > 0:
        best_a = best_a / best_a.sum()

    # Store the result for possible reuse in later runs.
    globals()["height_sequence_2"] = best_a.tolist()

    return best_a.tolist()
