"""
Propose a step‑function (non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)^2)

where n = len(a) and conv is the linear convolution.
The algorithm explores several lengths, builds many diverse seeds,
optimises each seed with an Adam optimiser on a smooth approximation of
the max, and then refines the best candidate for each length with an
exact sub‑gradient (exponentiated‑gradient) method.  A final
fine‑tuning pass is performed on the globally best sequence.

The implementation respects the 1000 s time budget and prints
progressive objective values.  The resulting list of floats is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT. Returns full length (len(x)+len(y)-1)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps ℝⁿ → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(rng: np.random.Generator, n: int, count: int = 5) -> list[np.ndarray]:
    seeds = []
    for _ in range(count):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))
    return seeds


def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to change length while preserving total mass."""
    if seq.size == new_n:
        return seq.copy()
    old_x = np.linspace(0, 1, seq.size)
    new_x = np.linspace(0, 1, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    s = new_seq.sum()
    return new_seq / s if s > 0 else np.full(new_n, 1.0 / new_n)


def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a diverse set of starting points for a given length."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    for c in np.linspace(0.5, 4.5, 7):
        r = 1.0 - c / n
        if r > 0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 3.5, 4):
        r = 1.0 - c / n
        if r <= 0:
            continue
        for p in np.linspace(0.0, 1.8, 4):
            seeds.append(_geom_power(n, r, p))

    for c in np.linspace(0.5, 4.0, 4):
        r = 1.0 - c / n
        if r > 0:
            seeds.append(_sym_geom(n, r))

    # random soft‑max vectors
    seeds.extend(_random_softmax(rng, n, count=8))

    # previous solution (if any) – both direct and interpolated versions
    if previous is not None:
        # direct (if same length) – already normalised
        if previous.shape[0] == n:
            seeds.append(previous / previous.sum())
        # interpolated copy
        seeds.append(_resize_sequence(previous, n))

        # perturbed copy of previous (add small noise)
        pert = previous + 0.02 * rng.normal(size=previous.shape)
        pert = np.clip(pert, 0, None)
        if pert.sum() > 0:
            seeds.append(_softmax(np.log(pert + 1e-30)))

    # ensure all seeds are on the simplex
    normalized = []
    for s in seeds:
        ss = s.sum()
        if ss == 0:
            continue
        normalized.append(s / ss)
    return normalized


# ----------------------------------------------------------------------
# Objective helpers
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact objective value (sum(a) is assumed positive)."""
    conv = _conv_fft(a, a)
    return 2.0 * len(a) * np.max(conv) / (np.sum(a) ** 2)


# ----------------------------------------------------------------------
# Adam optimisation on a smooth max (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 3000,
                   lr0: float = 0.8,
                   beta_start: float = 5.0,
                   beta_mul: float = 1.07) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max of the auto‑convolution.
    Returns best sequence (as numpy array) and its exact objective.
    """
    v = np.log(seed + 1e-30)          # log‑softmax parameters
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = beta_start

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)               # stays on simplex

        # evaluate exact objective for early stopping / logging
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:4d} best={cur_val:.6f}")

        # ----- smoothed‑max weighting (log‑sum‑exp) -----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)          # safe shift for exponentials
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient w.r.t. a (∇_a Σ w_k·conv_k) -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)    # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- chain rule through softmax -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on v -----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** t)
        v2_hat = v2 / (1.0 - 0.999 ** t)

        lr = lr0 / math.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    if best_seq is None:
        # fall back to seed normalized (should never happen)
        best_seq = _softmax(v)
        best_val = _objective(best_seq)
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent / exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 20000,
                        init_lr: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Refine a sequence using the exact sub‑gradient of the max convolution.
    Uses exponentiated‑gradient updates to stay on the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (np.sum(a) ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # exact sub‑gradient of the max‑conv entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step (mirror descent on simplex)
        lr = init_lr / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # for numerical stability
        a = np.exp(log_a)
        s = a.sum()
        a = a / s if s > 0 else _uniform(n)

    return best_seq, best_val

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget and returns the best sequence found (as a plain list of
    floats).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a safety margin for finalisation / printing
    deadline = start_time + budget_s - 5.0

    # retrieve previous construction if available
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size == 0:
                previous = None
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (increasing order).  The loop stops early if
    # we run out of time.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 16000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # ----- generate a modest but diverse pool of seeds -----
        seeds = _make_seeds(n, previous, rng)

        # quick evaluation of raw seeds (no optimisation)
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n:5d} new global best={val:.6f}")

        # ---- Adam phase for each seed ---------------------------------
        # iteration budget roughly scales as 2500 * 4000 / n
        max_adam_iters = max(500, int(2500 * 4000 / n))
        for idx, seed_vec in enumerate(seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline,
                                      max_iters=max_adam_iters,
                                      lr0=0.8,
                                      beta_start=5.0,
                                      beta_mul=1.07)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall Adam] n={n:5d} new global best={val:.6f}")

        # ---- Sub‑gradient refinement for the best sequence of this n ---
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            max_subgrad_iters = max(4000, int(20000 * 4000 / n))
            seq_ref, val_ref = _subgradient_refine(
                global_best_seq, n, deadline,
                max_iters=max_subgrad_iters,
                init_lr=0.9)
            if seq_ref is not None and val_ref < global_best_val:
                global_best_val = val_ref
                global_best_seq = seq_ref.copy()
                print(f"[overall SubGrad] n={n:5d} new global best={val_ref:.6f}")

    # ------------------------------------------------------------------
    # If time still left, perform one final refinement on the absolute
    # best sequence (irrespective of its length) to squeeze a few more
    # digits.
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        n = global_best_seq.shape[0]
        extra_iters = max(2000, int(5000 * 4000 / n))
        print(f"\n[Final] extra sub‑gradient refinement (iters={extra_iters})")
        seq_ref, val_ref = _subgradient_refine(
            global_best_seq, n, deadline,
            max_iters=extra_iters,
            init_lr=0.8)
        if seq_ref is not None and val_ref < global_best_val:
            global_best_val = val_ref
            global_best_seq = seq_ref.copy()
            print(f"[Final] improved global best={val_ref:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional) – evaluate via the provided function.
    # ------------------------------------------------------------------
    try:
        # In the execution environment evaluate_sequence is already defined.
        from __main__ import evaluate_sequence  # type: ignore
        final_obj = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest objective reported by evaluate_sequence = {final_obj:.6f}")
    except Exception:
        # If evaluate_sequence is not importable we fall back to our own compute.
        final_obj = _objective(global_best_seq)
        print(f"\n=== Finished ===\nBest objective (internal) = {final_obj:.6f}")

    # Return plain Python list as required.
    return global_best_seq.tolist()
