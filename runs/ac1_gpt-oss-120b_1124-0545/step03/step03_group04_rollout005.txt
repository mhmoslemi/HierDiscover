"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

The optimizer works as an accelerated smoothed‑max gradient descent on
the quantity  max_k (a * a)_k  (the self‑convolution peak).  The key
ingredients are:

*   A probability‑simplex projection that keeps the vector non‑negative
    and with total mass 1 (the objective is scale‑invariant).

*   A soft‑max (log‑sum‑exp) smoothing of the max; its exact gradient is
    g = 2·v where
        v_j = Σ_i w_{i+j}·a_i
    and w are the soft‑max weights of the current convolution.

*   An Adam‑style momentum update (adds robustness without extra
    hyper‑parameters).

*   Small random “mass‑swap’’ perturbations every few hundred steps to
    escape shallow plateaux.

*   Multiple restarts are performed automatically: after a
    configurable number of iterations the current best vector is
    disturbed by a tiny Gaussian noise and optimisation continues.
    The best vector seen over the whole 1000 s budget is returned.

The algorithm has no external dependencies except NumPy and uses only
FFT‑based convolutions, making a full run with n≈5 000–10 000 feasible
within the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the O(n) algorithm of
    Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sorted descending
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries would become negative → uniform vector
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using the FFT.
    Returns the full convolution (length = len(x) + len(y) – 1).
    """
    size = x.size + y.size - 1
    # next power of two for efficient FFT
    N = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _softmax_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_i ∝ exp(conv_i / eps).
    eps must be > 0.
    """
    # shift for numerical stability, then clip to avoid overflow
    shifted = conv / eps
    shifted = shifted - shifted.max()
    shifted = np.clip(shifted, a_min=None, a_max=50.0)
    expv = np.exp(shifted)
    total = expv.sum()
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return expv / total
