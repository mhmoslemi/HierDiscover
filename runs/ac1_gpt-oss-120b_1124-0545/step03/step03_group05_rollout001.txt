def _alternate_projection(a, fft_size, n):
    # compute F
    F = np.fft.rfft(a, n=fft_size)
    total_power = np.sum(np.abs(F)**2)
    # preserve DC magnitude as sum (should be 1)
    F0 = F[0]
    m = len(F) - 1
    # compute target magnitude for other components
    if m > 0:
        target_mag_sq = (total_power - np.abs(F0)**2) / m
        target_mag = np.sqrt(max(target_mag_sq, 0.0))
    else:
        target_mag = 0.0
    # adjust magnitude
    # preserve original phases
    # keep DC unchanged, others scaled to target magnitude
    new_F = np.empty_like(F)
    new_F[0] = F0  # keep DC
    # For rest
    # compute phases
    phases = np.exp(1j * np.angle(F[1:]))
    new_F[1:] = target_mag * phases
    # Inverse
    a_new = np.fft.irfft(new_F, n=fft_size)[:n]  # only first n entries
    # Clip to nonneg and renorm.
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0:
        a_new = np.ones_like(a_new) / n
    else:
        a_new /= s
    return a_new

def _alternating_projection(a: np.ndarray, fft_size: int, n: int) -> np.ndarray:
    """Perform one iteration of spectral magnitude flattening."""
    # Compute FFT of a (real)
    F = np.fft.rfft(a, n=fft_size)
    # Preserve DC amplitude (sum=1)
    F0 = F[0]
    # Compute total power
    total_power = np.sum(np.abs(F) ** 2)
    # Compute target magnitude for other frequencies (excluding DC)
    m = len(F) - 1  # number of non-DC frequencies
    if m > 0:
        # Ensure we keep total power constant
        target_mag_sq = max((total_power - np.abs(F0) ** 2) / m, 0.0)
        target_mag = np.sqrt(target_mag_sq)
    else:
        target_mag = 0.0
    # Build new spectrum with same phase but constant magnitude for non-DC
    # Use phase from original
    phase = np.exp(1j * np.angle(F[1:]))
    new_F = np.empty_like(F)
    new_F[0] = F0
    new_F[1:] = target_mag * phase
    # Inverse transform
    a_new = np.fft.irfft(new_F, n=fft_size)[:n]
    # Clip negative values and renormalize to sum 1
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a_new) / n
    else:
        a_new /= s
    return a_new

def _compute_softmax_gradient(a, conv_full, a_rev, size2, tau):

assistantfinal**Solution Overview**

The search consists of two complementary stages:

1. **Spectral flattening (alternating projections).**  
   The convolution `a*a` is the inverse Fourier transform of `|F|²` (`F` – the DFT of `a`).  
   If `|F|` is (almost) constant, `a*a` becomes flat and its maximum drops.  
   One iteration of the projection:
   * compute the FFT of the current `a`,
   * keep the DC component (it equals the ℓ₁–norm = 1),
   * replace the magnitudes of all other frequencies by the common value that preserves the total power,
   * inverse‑FFT, clip negatives, renormalise.

   This cheap operation dramatically reduces the peak of the auto‑convolution.

2. **Momentum soft‑max gradient descent.**  
   After the spectral stage we fine‑tune the sequence.  
   * A smooth approximation of the max is obtained with a soft‑max weighting  
     `w ∝ exp((b‑max(b))/τ)`.  
   * The gradient of the smoothed objective is `‑2·(a_rev ∗ w)`.  
   * An Adam‑style momentum update is applied, the vector is projected back to the
     probability simplex (clip ≥ 0 and renormalise).  
   * Occasionally we (i) perform a hard‑max gradient step (τ → 0) to get a
     true sub‑gradient, (ii) execute a tiny random mass‑transfer, or (iii)
     restart when stagnation is detected.

Both stages keep the total ℓ₁‑mass equal to 1, so the objective computed as

```
score = 2 * n * max(conv(a, a))
```

coincides with `evaluate_sequence` (the scaling factor cancels out).

The algorithm runs for the whole 1000 s budget, prints progress, and finally
returns the best sequence found.

```python
"""Step‑function optimizer.

The function `propose_candidate` searches for a non‑negative sequence a
(minimising 2·n·max(conv(a,a))) by

1. Spectral flattening – alternating projection onto a constant‑magnitude
   Fourier spectrum while keeping non‑negativity and ℓ₁‑norm = 1.
2. Momentum soft‑max gradient descent (Adam‑like) on a smooth approximation
   of the peak of the auto‑convolution, with occasional hard‑max steps,
   random mass‑transfer perturbations and restarts.

Both stages are performed inside the allotted time budget (default 1000 s)
and the best sequence encountered is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures, no lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution using FFT.
    `size` must be ≥ len(u)+len(v)-1.
    Returns the full convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size1: int) -> np.ndarray:
    """Full auto‑convolution a * a."""
    return _conv_fft(a, a, size1)


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    if a[i] == 0.0:
        return a
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= s
    return a_new


def _alternating_projection(a: np.ndarray, fft_size: int, n: int) -> np.ndarray:
    """
    One iteration of spectral magnitude flattening.
    Keeps the DC component (ℓ₁‑norm) and forces all other magnitudes to be equal
    while preserving total power.  After the inverse FFT the vector is clipped to
    non‑negative values and re‑normalised.
    """
    # FFT of the current sequence
    F = np.fft.rfft(a, n=fft_size)

    # DC component (must stay equal to 1 because sum(a)=1)
    F0 = F[0]

    # Total power = Σ|F|²  (Parseval)
    total_power = np.sum(np.abs(F) ** 2)

    # Number of non‑DC frequencies
    m = len(F) - 1
    if m > 0:
        # Distribute the remaining power uniformly
        target_mag_sq = max((total_power - np.abs(F0) ** 2) / m, 0.0)
        target_mag = np.sqrt(target_mag_sq)
    else:
        target_mag = 0.0

    # Build new spectrum: keep phase, replace magnitudes (except DC)
    phase = np.exp(1j * np.angle(F[1:]))
    new_F = np.empty_like(F)
    new_F[0] = F0
    new_F[1:] = target_mag * phase

    # Inverse FFT, keep only first n entries (original length)
    a_new = np.fft.irfft(new_F, n=fft_size)[:n]

    # Clip negative values and renormalise to ℓ₁‑norm = 1
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a_new) / n
    else:
        a_new /= s
    return a_new


def _compute_softmax_gradient(
    a: np.ndarray,
    conv_full: np.ndarray,
    a_rev: np.ndarray,
    size2: int,
    tau: float,
) -> np.ndarray:
    """
    Gradient of the smooth max (soft‑max) of the auto‑convolution.
    Returns a vector with zero sum (so that the ℓ₁‑norm is preserved after a
    simple projection).
    """
    max_b = conv_full.max()
    # Soft‑max weights (stable shift by max_b)
    weight = np.exp((conv_full - max_b) / tau)
    w_sum = weight.sum()
    if w_sum == 0.0:
        weight = np.ones_like(weight) / weight.size
    else:
        weight /= w_sum

    # Convolution of reversed a with the weight vector
    g_full = _conv_fft(a_rev, weight, size2)
    n = a.shape[0]
    g = g_full[n - 1 : n - 1 + n]  # central part, length n

    grad = -2.0 * g                # descent direction for the soft max
    grad -= grad.mean()            # enforce Σ grad_i = 0  (simple projection)
    return grad


def _hard_gradient(a: np.ndarray, n: int, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of the *hard* max (single index k_max).
    For each i with a valid partner j = k_max - i we have
        ∂b_{k_max} / ∂a_i = 2 * a_j,
    therefore the descent direction is -2 * a_j.
    """
    grad = np.zeros_like(a)
    start = max(0, k_max - (n - 1))
    end = min(k_max, n - 1)
    if start <= end:
        idx = np.arange(start, end + 1)
        grad[idx] = -2.0 * a[k_max - idx]
    grad -= grad.mean()   # keep Σ grad = 0
    return grad

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights with a low value of
    `evaluate_sequence`.  The routine respects a time budget (default 1000 s)
    and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Optional ``n`` – length of the sequence (default 6000).

    Returns
    -------
    list[float]
        The best sequence (Python floats) discovered.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # ------------------------------------------------------------------
    # Initialise a feasible probability vector (ℓ₁‑norm = 1)
    # ------------------------------------------------------------------
    try:
        seq0 = globals().get("height_sequence_1", None)
        if seq0 is not None:
            a = np.asarray(seq0, dtype=float)
            if a.size != n:
                # Linear interpolation to required length
                x_old = np.linspace(0, 1, a.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                a = np.interp(x_new, x_old, a)
            a = np.clip(a, 0.0, None)
            if a.sum() == 0.0:
                a = np.random.rand(n)
            a /= a.sum()
        else:
            raise KeyError
    except Exception:
        a = np.random.rand(n)
        a /= a.sum()

    # ------------------------------------------------------------------
    # Pre‑compute FFT sizes (powers of two)
    # ------------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)       # for a*a
    size2 = _next_pow2(3 * n - 2)       # for a_rev * soft‑max weight
    proj_fft_size = _next_pow2(2 * n)   # for spectral projection

    # ------------------------------------------------------------------
    # Bookkeeping
    # ------------------------------------------------------------------
    a_rev = a[::-1].copy()
    conv_full = _full_convolution(a, size1)
    best_score = 2.0 * n * conv_full.max()   # because sum(a)=1
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Spectral alternating projection (≈40 % of the budget)
    # ------------------------------------------------------------------
    stage1_deadline = start + 0.4 * (budget_s - 0.5)
    proj_iter = 0
    last_improve = start

    while time.time() < stage1_deadline:
        a = _alternating_projection(a, proj_fft_size, n)
        conv_full = _full_convolution(a, size1)
        cur_score = 2.0 * n * conv_full.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[proj {proj_iter:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")
        proj_iter += 1

        # Every few hundred iterations add a tiny random perturbation
        if proj_iter % 500 == 0:
            a = _random_transfer(a, 0.02)

        # Stagnation restart
        if time.time() - last_improve > 30.0:
            a = np.random.rand(n)
            a /= a.sum()
            a_rev = a[::-1].copy()
            last_improve = time.time()
            print(f"[proj {proj_iter}] restart due to stagnation")

        a_rev = a[::-1].copy()

    # ------------------------------------------------------------------
    # Stage 2 – Momentum soft‑max gradient descent with occasional hard steps
    # ------------------------------------------------------------------
    # Adam‑style moment estimates (actually simple momentum, but the code
    # follows the Adam pattern without the second‑moment term for speed)
    m = np.zeros_like(a)                     # momentum
    mu = 0.9                                 # momentum decay
    base_step = 0.6
    tau_factor = 0.4                         # τ = tau_factor * max_b
    tau_decay = 0.9995
    rand_prob = 0.06                         # probability of a random transfer
    rand_scale = 0.03
    stagnation_seconds = 25.0
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        conv_full = _full_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        # Record best solution
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # Random mass‑transfer perturbation
        if np.random.rand() < rand_prob:
            a = _random_transfer(a, rand_scale)
            a_rev = a[::-1].copy()
            it += 1
            continue

        # Temperature for the soft‑max (gradual annealing)
        tau = max(tau_factor * max_b, 1e-12)
        tau_factor *= tau_decay

        # Soft‑max gradient
        grad = _compute_softmax_gradient(a, conv_full, a_rev, size2, tau)

        # Every few hundred iterations blend in a true sub‑gradient
        if it % 200 == 0:
            k_max = int(np.argmax(conv_full))
            hard_grad = _hard_gradient(a, n, k_max)
            grad = 0.5 * grad + 0.5 * hard_grad

        # Diminishing step size
        step = base_step / np.sqrt(it + 1.0)

        # Momentum update (Adam‑style without second moment)
        m = mu * m - step * grad
        a_new = a + m
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.ones_like(a_new) / n
        else:
            a_new /= s

        a = a_new
        a_rev = a[::-1].copy()
        it += 1

        # Stagnation handling – restart from a fresh random point
        if time.time() - last_improve > stagnation_seconds:
            a = np.random.rand(n)
            a /= a.sum()
            a_rev = a[::-1].copy()
            m = np.zeros_like(a)
            tau_factor = 0.4
            last_improve = time.time()
            print(f"[{it}] restart due to stagnation")

    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
```
