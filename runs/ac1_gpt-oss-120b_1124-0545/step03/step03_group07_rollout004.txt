"""
Hybrid optimizer for the step‑function bound.

The algorithm consists of three stages within the allotted 1000 s:

1. **Soft‑max gradient descent** – we minimise a smooth approximation of
   `max(conv(a,a))`.  For a temperature λ the objective is
   `f_λ(a) = (1/λ)·log( Σ exp(λ·conv_i) )`.  Its gradient can be written
   analytically using a convolution of the soft‑max weights with the
   reversed vector.  An exponentiated‑gradient update (which keeps the
   iterates non‑negative) is performed and the iterate is projected onto
   the ℓ₁‑simplex `{a ≥ 0, Σ a = √(2n)}`.  λ is increased exponentially,
   turning the smooth problem into the original max‑problem.

2. **Projected sub‑gradient descent** on the exact max‑convolution.
   The sub‑gradient is `g_i = 2·a_{k‑i}` where `k` is the index attaining
   `max(conv)`.  After each step a simplex projection restores feasibility.
   Simple restarts break stagnation.

3. **Simulated annealing** – starting from the best vector found,
   random mass transfers are proposed.  Moves that lower the peak
   convolution are always accepted, otherwise they are accepted with
   Metropolis probability.  A geometric temperature schedule cools from
   `0.2·best` to `1e‑5·best`.

All three phases constantly record the best sequence encountered.  The
final return value is a plain Python list suitable for `evaluate_sequence`.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , Σ w = s }."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (used for self‑convolution)."""
    return 1 << ((2 * n - 1).bit_length())


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Return (max_convolution_value, argmax_index) for the self‑convolution
    a*a.  Computed with an FFT of length ``fft_len``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    idx = int(np.argmax(conv))
    val = float(conv[idx])
    return val, idx


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The search combines a smoothed‑max gradient
    descent, a projected sub‑gradient phase and a simulated‑annealing
    refinement.  The best sequence found within the time budget is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1) Choose a starting vector.
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=float)
    else:
        # a generous default length that still allows fast FFTs
        a = np.random.exponential(scale=1.0, size=5000)

    n = a.size
    target_sum = math.sqrt(2.0 * n)            # ℓ₁‑norm required for the bound
    a = a.astype(float)
    a = a / a.sum() * target_sum               # normalise to the required sum

    # ------------------------------------------------------------------
    # Pre‑compute FFT lengths (used repeatedly).
    fft_len1 = _fft_len_for_n(n)               # for self‑convolution a*a
    fft_len2 = 1 << ((3 * n - 2).bit_length())  # for gradient convolution (w * a_rev)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – soft‑max gradient descent (smooth approximation of max).
    phase1_deadline = start_time + budget_s * 0.35
    lam_start = 0.1
    lam_end = 30.0
    it = 0
    best_seq = a.copy()
    best_max, _ = _max_convolution(a, fft_len1)

    print(f"[INIT] n={n}, target_sum={target_sum:.3f}, init max(conv)={best_max:.6f}")

    while time.time() < phase1_deadline:
        # --- self‑convolution ---
        fa = np.fft.rfft(a, n=fft_len1)
        conv = np.fft.irfft(fa * fa, n=fft_len1)[: 2 * n - 1]

        # --- schedule for λ (exponential increase) ---
        prog = (time.time() - start_time) / (phase1_deadline - start_time)
        lam = lam_start * (lam_end / lam_start) ** prog

        # --- soft‑max weights (stable) ---
        maxc = conv.max()
        ex = np.exp(lam * (conv - maxc))        # subtract maxc to avoid overflow
        w = ex / ex.sum()                       # shape (2n‑1,)

        # --- gradient of the smooth objective ---
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=fft_len2)
        fa_rev = np.fft.rfft(a_rev, n=fft_len2)
        conv_rev_full = np.fft.irfft(fw * fa_rev, n=fft_len2)
        conv_rev_full = conv_rev_full[: (3 * n - 2)]   # linear convolution length

        # the segment that yields the gradient (indices n‑1 … 2n‑2)
        segment = conv_rev_full[n - 1 : 2 * n - 1]     # length n
        grad = 2.0 * segment[::-1]                    # see derivation above

        # --- exponentiated‑gradient update + simplex projection ---
        # a simple diminishing step size
        eta = 0.5 / (1.0 + it * 1e-4)
        factor = np.exp(-eta * grad)                 # element‑wise
        a = a * factor
        a = _simplex_projection(a, target_sum)

        # --- keep the best vector seen so far ---
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_seq = a.copy()
            print(f"[ Phase‑1 {it:7d} ] new best max(conv) = {best_max:.6f}")

        # occasional tiny random jitter to escape tiny plateaus
        if it % 5000 == 0 and it > 0:
            noise = np.random.normal(scale=0.005, size=n)
            a = a + noise
            a = np.clip(a, 0, None)
            a = _simplex_projection(a, target_sum)

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – exact projected sub‑gradient descent.
    a = best_seq.copy()
    best_seq = a.copy()
    best_max, best_k = _max_convolution(a, fft_len1)
    subgrad_deadline = start_time + budget_s * 0.65
    alpha0 = target_sum * 0.02
    it2 = 0
    last_improve = 0

    while time.time() < subgrad_deadline:
        cur_max, cur_k = _max_convolution(a, fft_len1)

        if cur_max < best_max:
            best_max = cur_max
            best_seq = a.copy()
            last_improve = it2
            print(f"[ Subgrad {it2:7d} ] improved max(conv) = {best_max:.6f}")

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, cur_k - (n - 1))
        i_end = min(cur_k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        g[idx] = 2.0 * a[cur_k - idx]

        # diminishing step size
        step = alpha0 / math.sqrt(it2 + 1.0)

        a_candidate = a - step * g
        a_candidate = _simplex_projection(a_candidate, target_sum)
        a = a_candidate
        it2 += 1

        # restart if nothing improved for a while
        if it2 - last_improve > 8000:
            a = np.random.exponential(scale=1.0, size=n)
            a = a / a.sum() * target_sum
            last_improve = it2
            print(f"[ Subgrad {it2:7d} ] restart due to stagnation")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing for final fine‑tuning.
    a = best_seq.copy()
    cur_max, _ = _max_convolution(a, fft_len1)
    best_max_sa = cur_max
    best_seq_sa = a.copy()
    sa_start = time.time()
    sa_deadline = deadline
    T0 = best_max_sa * 0.20          # initial temperature
    Tmin = best_max_sa * 1e-5
    step_counter = 0

    while time.time() < sa_deadline:
        # geometric cooling
        elapsed = time.time() - sa_start
        frac = elapsed / (sa_deadline - sa_start + 1e-12)
        T = T0 * ((Tmin / T0) ** frac)

        # propose a random mass transfer between two distinct positions
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = a[i] * 0.15 * np.random.rand()   # move up to 15 % of source mass
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta

        cand_max, _ = _max_convolution(a_cand, fft_len1)

        # Metropolis acceptance
        if cand_max < cur_max or np.random.rand() < math.exp(-(cand_max - cur_max) / max(T, 1e-12)):
            a = a_cand
            cur_max = cand_max
            if cand_max < best_max_sa:
                best_max_sa = cand_max
                best_seq_sa = a_cand.copy()
                print(f"[ SA {step_counter:7d} ] new best max(conv) = {best_max_sa:.6f}")

        step_counter += 1

    print(f"[DONE] best max(conv) = {best_max_sa:.6f} "
          f"after {it} soft‑max iters, {it2} sub‑grad iters, {step_counter} SA steps")
    return best_seq_sa.tolist()
