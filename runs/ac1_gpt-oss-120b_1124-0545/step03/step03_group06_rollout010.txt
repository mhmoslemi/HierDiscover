"""
Improved search for a non‑negative step function that minimises
``evaluate_sequence``.

The algorithm consists of three stages:

1. **Exploratory grid** – a short scan of power‑law families
   a_i ∝ (i+1+δ)^{-α} (α and offset δ) to obtain a decent starting
   point.

2. **Smoothed sub‑gradient descent** – the objective
   2·n·max(conv(a,a)) is approximated by the smooth log‑sum‑exp of the
   convolution entries.  The gradient of this smooth surrogate is a
   weighted sum of the exact sub‑gradients for all shifts of ``a`` and is
   computed efficiently with a single FFT‑based convolution.  A
   Polyak‑type step size ( (f‑f*) / ‖g‖² ) together with a momentum term
   guides the updates, after which the iterate is projected onto the
   probability simplex.  The smoothing parameter σ is increased
   gradually, so the method first follows a smooth landscape and then
   focuses on the true max‑norm.

3. **Random restarts & polishing** – a few short optimisation runs
   from random power‑law seeds are performed to escape possible local
   minima, and finally a longer run with a small step size is executed
   on the best sequence found.

All heavy work is performed with FFTs (O(n log n)), allowing the use of
a fairly large length (≈ 20 000) while staying well within the 1000 s
budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Normalised power‑law sequence a_i ∝ (i+1+offset)^{-α}.
    The result sums to 1.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_fft(x: np.ndarray) -> np.ndarray:
    """Full (linear) self‑convolution of a real vector using FFT."""
    n = x.shape[0]
    size = _next_pow2(2 * n - 1)
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[: 2 * n - 1]
    conv[conv < 0.0] = 0.0      # eliminate tiny negatives from round‑off
    return conv


def _score_fft(x: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a vector normalised to sum = 1."""
    n = x.shape[0]
    b = _conv_fft(x)
    return float(2.0 * n * b.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1} (algorithm of Duchi et al. 2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed sub‑gradient utilities
# ----------------------------------------------------------------------


def _smooth_subgrad(seq: np.ndarray, sigma: float):
    """
    Compute a smooth approximation of the sub‑gradient of
    f(seq) = max(conv(seq,seq)) using the log‑sum‑exp smoothing parameter
    ``sigma``.  Returns (grad, max_conv).

    The gradient is:
        g = 2 * Σ_k softmax_k(σ·conv) * shift(seq, k)
    where shift(seq,k) is the vector (seq_{k‑i})_{i}.
    """
    n = seq.shape[0]

    # 1) self‑convolution and soft‑max weights
    conv = _conv_fft(seq)
    maxb = conv.max()
    # (conv - maxb) ≤ 0, safe exponentiation
    exp_w = np.exp(sigma * (conv - maxb))
    w = exp_w / exp_w.sum()            # soft‑max (sums to 1)

    # 2) weighted sum of shifts via a single FFT convolution
    rev = seq[::-1]                    # reversed a_i
    size = _next_pow2(3 * n - 2)       # length of rev (n) + w (2n‑1) – 1
    f_rev = np.fft.rfft(rev, n=size)
    f_w = np.fft.rfft(w, n=size)
    grad_full = np.fft.irfft(f_rev * f_w, n=size)[: 3 * n - 2]
    # slice indices n‑1 … n‑1+n‑1 contain Σ_j a_j w_{i+j}
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]

    # numerical safety – the projection later will clip negatives anyway
    grad[np.isnan(grad)] = 0.0
    return grad, maxb


def _optimize(
    start_seq: np.ndarray,
    time_budget: float,
    *,
    sigma_start: float = 2.0,
    sigma_factor: float = 1.2,
    sigma_max: float = 80.0,
    momentum_beta: float = 0.9,
    max_step: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Smoothed sub‑gradient descent with Polyak step sizes and momentum.
    Returns the best sequence encountered and its exact score.
    """
    n = start_seq.shape[0]
    lower_score = 2.0 * n / (2 * n - 1)          # theoretical lower bound
    seq = start_seq.copy()
    best_seq = seq.copy()
    best_score = _score_fft(seq)

    momentum = np.zeros_like(seq)
    sigma = sigma_start
    it = 0
    start = time.time()

    while time.time() - start < time_budget:
        grad, maxb = _smooth_subgrad(seq, sigma)
        cur_score = 2.0 * n * maxb

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] σ={sigma:.2f} it={it} score={best_score:.6f}")

        # Polyak step size (f‑f*)/‖g‖², bounded for safety
        norm2 = np.dot(grad, grad)
        if norm2 > 0.0:
            step = (cur_score - lower_score) / (norm2 + 1e-16)
            step = min(step, max_step)
        else:
            step = 0.0

        # momentum update
        momentum = momentum_beta * momentum + (1.0 - momentum_beta) * grad

        # gradient step + simplex projection
        seq = seq - step * momentum
        seq = _project_simplex(seq)

        # occasional random mass exchange to avoid plateaus
        if it % 500 == 0 and it > 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(seq[i] * 0.2, 1e-5)
            seq[i] -= eps
            seq[j] += eps
            seq = _project_simplex(seq)

        # schedule for σ (more focus on the true max as optimisation proceeds)
        if it % 2000 == 0 and sigma < sigma_max:
            sigma = min(sigma * sigma_factor, sigma_max)

        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Find a non‑negative step sequence that minimises ``evaluate_sequence``.
    The routine runs a fast FFT based optimiser for roughly 20 000
    coefficients within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness’s final call
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Problem size – a few tens of thousands give a good balance.
    # ------------------------------------------------------------------
    n = 20000

    # ------------------------------------------------------------------
    # 2. Coarse exploration of power‑law families (α, offset).
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    alphas = np.linspace(0.44, 0.62, 25)          # covering the interesting region
    offsets = np.linspace(0.0, 5.0, 6)

    for alpha in alphas:
        for off in offsets:
            if time.time() >= deadline:
                break
            cand = _power_sequence(n, alpha, off)
            sc = _score_fft(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[grid] α={alpha:.4f} δ={off:.2f} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Short optimisation from the best grid point.
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = max(0.0, deadline - time.time())
        best_seq, best_score = _optimize(
            best_seq,
            time_budget=remaining,
            sigma_start=2.0,
            sigma_factor=1.2,
            sigma_max=30.0,
            momentum_beta=0.85,
            max_step=0.4,
        )
        print(f"[phase‑1] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Random restarts – cheap runs from random power‑law seeds.
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:   # reserve ~30 s for final polishing
        alpha = np.random.uniform(0.35, 0.75)
        off = np.random.uniform(0.0, 5.0)
        seq = _power_sequence(n, alpha, off)
        # tiny perturbation to increase diversity
        seq += np.random.randn(n) * 1e-6
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()

        remaining = max(0.0, deadline - time.time())
        seq, sc = _optimize(
            seq,
            time_budget=remaining,
            sigma_start=1.5,
            sigma_factor=1.3,
            sigma_max=40.0,
            momentum_beta=0.9,
            max_step=0.3,
        )
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[restart] improved to {best_score:.6f} (α={alpha:.4f},δ={off:.2f})")
        # early stop if we have already crossed the target bound
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 5. Final polish – a longer run with a small step size.
    # ------------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    best_seq, best_score = _optimize(
        best_seq,
        time_budget=remaining,
        sigma_start=5.0,
        sigma_factor=1.1,
        sigma_max=80.0,
        momentum_beta=0.9,
        max_step=0.2,
    )
    best_seq = _project_simplex(best_seq)   # final safety projection

    print(f"[final] best_score={best_score:.6f}, length={n}")
    return best_seq.tolist()
