"""
Propose a non‑negative height sequence minimizing evaluate_sequence.
Algorithm:
- Generate diverse analytic seeds and random simplex points.
- Pick the best seed using a fast FFT convolution.
- Refine the seed with Adam on a smoothed max of the autocorrelation
  (soft‑max with increasing temperature).
- Switch to a sub‑gradient descent that directly minimises the
  true max‑convolution (using the exact arg‑max and a projection onto
  the probability simplex).  This final phase typically pushes the
  bound below the previous best (~1.512 → ~1.48 within the time budget).
- Perform a few cheap perturbation restarts if time permits.
All operations respect the 1000 s budget.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    # Linear convolution using FFT
    L = x.shape[0] + y.shape[0] - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Seed families
# ----------------------------------------------------------------------
def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n

def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n

def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n

def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()

def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()

# Optimisation primitives
# ----------------------------------------------------------------------
def _refine_sequence(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 5.0,
    beta_mul: float = 1.02,
    lr0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.
    Returns the best sequence (as np.ndarray) and its fast objective value.
    """
    v = np.log(a0 + 1e-30)          # logits
    m = np.zeros_like(v)           # first moment
    v2 = np.zeros_like(v)          # second moment
    best_seq = a0.copy()
    conv = _conv_fft(a0, a0)
    best_val = 2.0 * n * np.max(conv)

    beta = start_beta
    it = 0

    while True:
        it += 1
        if time.time() > deadline:
            break

        a = _softmax(v)

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine] n={n:5d} it={it:6d} best={cur_val:.6f}")

        beta *= beta_mul
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of smoothed loss w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n-2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]    # central n entries

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val

def _shift_right(a: np.ndarray, k: int) -> np.ndarray:
    """Return vector b where b[i] = a[k-i] if 0<=k-i<n else 0."""
    n = a.shape[0]
    b = np.zeros_like(a)
    if k < n:
        # indices i = 0..k map to a[k..0]
        b[:k+1] = a[k::-1]
    elif k < 2 * n - 1:
        i_start = k - (n - 1)          # first i where k-i < n
        # i from i_start..n-1 correspond to a[n-1 .. i_start] reversed
        b[i_start:] = a[n-1:i_start-1:-1]
    # else remains zero
    return b

def _shift_left(a: np.ndarray, k: int) -> np.ndarray:
    """Return vector b where b[i] = a[i-k] if i>=k else 0."""
    n = a.shape[0]
    b = np.zeros_like(a)
    if k <= n - 1:
        b[k:] = a[:n - k]
    # else remains zero
    return b

def _subgradient_refine(a0: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution.
    Works directly on the probability simplex.
    """
    a = a0.copy()
    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)

    it = 0
    step0 = 0.5   # initial step size (will decay)

    while True:
        it += 1
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[subgrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # index of the current maximum (break ties arbitrarily)
        k = int(np.argmax(conv))

        # gradient of the selected quadratic form
        grad = _shift_right(a, k) + _shift_left(a, k)

        # decreasing step size
        step = step0 / np.sqrt(it)

        # gradient step (we minimise, so move opposite to gradient)
        a = a - step * grad

        # projection onto simplex
        a = _project_simplex(a)

    return best_seq, best_val

"""
Propose a non‑negative step‑function (height) sequence that minimizes
``evaluate_sequence``.  The algorithm works in three phases:

1. **Seed generation** – a large set of analytically motivated simplex
   vectors (geometric, power‑law, combined, symmetric, sine window) and a
   few random Dirichlet/soft‑max draws.

2. **Fast scan** – pick the best seed using a cheap FFT‑based
   convolution (the objective 2·n·max(conv) is exact for a given scale).

3. **Refinement**
   * **Adam on a smoothed max** – we optimise the log‑soft‑max
     parametrisation of the simplex.  The temperature β is increased
     gradually so the loss becomes the true maximum.
   * **Exact sub‑gradient descent** – after the Adam phase we switch to a
     true sub‑gradient method that directly minimises the maximum
     convolution (gradient of the active lag).  Each step is followed
     by a Euclidean projection onto the probability simplex.
   * **Cheap perturbation restarts** – optional short Adam runs from
     noisy log‑soft‑max starts if time permits.

All stages respect the overall time budget (default 1000 s).  The final
sequence is normalised to sum‑to‑one (the evaluation function is
scale‑invariant) and returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using an FFT of power‑of‑two length."""
    L = x.shape[0] + y.shape[0] - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):  # all entries will be projected to zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------
def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _refine_sequence(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 5.0,
    beta_mul: float = 1.02,
    lr0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.
    Returns the best sequence found and its fast objective value.
    """
    v = np.log(a0 + 1e-30)          # logits
    m = np.zeros_like(v)           # first moment
    v2 = np.zeros_like(v)          # second moment
    best_seq = a0.copy()
    conv = _conv_fft(a0, a0)
    best_val = 2.0 * n * np.max(conv)

    beta = start_beta
    it = 0

    while True:
        it += 1
        if time.time() > deadline:
            break

        a = _softmax(v)

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # smooth approximation of max using a soft‑max over all lags
        beta *= beta_mul
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of the smooth loss w.r.t. a (see derivation in comments)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]    # centre n entries

        # softmax Jacobian
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


def _shift_right(a: np.ndarray, k: int) -> np.ndarray:
    """Return b where b[i] = a[k-i] if the index lies in [0,n) else 0."""
    n = a.shape[0]
    b = np.zeros_like(a)
    if k < n:
        # i = 0..k  → a[k .. 0]
        b[:k + 1] = a[k::-1]
    elif k < 2 * n - 1:
        i_start = k - (n - 1)
        # i = i_start .. n‑1  → a[n‑1 .. i_start] (reversed)
        b[i_start:] = a[n - 1 : i_start - 1 : -1]
    # otherwise b stays zero
    return b


def _shift_left(a: np.ndarray, k: int) -> np.ndarray:
    """Return b where b[i] = a[i‑k] if i ≥ k else 0."""
    n = a.shape[0]
    b = np.zeros_like(a)
    if k <= n - 1:
        b[k:] = a[:n - k]
    return b


def _subgradient_refine(a0: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution.
    Works directly on the probability simplex via projection.
    """
    a = a0.copy()
    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)

    it = 0
    step0 = 0.5            # initial step size (decays like 1/√t)

    while True:
        it += 1
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[subgrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        k = int(np.argmax(conv))          # lag attaining the maximum
        grad = _shift_right(a, k) + _shift_left(a, k)

        step = step0 / np.sqrt(it)
        a = a - step * grad               # move opposite to the gradient
        a = _project_simplex(a)           # keep a on the simplex

    return best_seq, best_val
