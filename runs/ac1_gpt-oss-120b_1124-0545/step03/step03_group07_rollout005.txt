"""
Propose a candidate step‑function sequence using two complementary ideas:

1. **Linear‑programming refinement** – starting from a good seed we
   repeatedly solve the linear programme

        maximise   Σ b_j
        s.t.      (f * b)_k ≤ M   for every k                (1)
                  b_j ≥ 0

   where `M = max (f * f)` for the current sequence `f`.  The programme
   pushes mass to positions where `f` is small, thereby allowing a larger
   ℓ₁‑norm of `b`.  After solving (1) we rescale `b` to the required
   ℓ₁–norm `√(2n)` and mix it with `f`.  The mixing coefficient is kept
   modest (≈0.2) so that `M` steadily decreases.

2. **Fast simulated‑annealing finish** – after the LP phase we spend the
   remaining time performing a classic SA move set (transfer a random
   fraction of mass between two coordinates).  This stage is identical
   to the one used in the previous best solution but starts from a
   hopefully better point.

Both phases respect the global `budget_s` seconds.  All helper functions
are top‑level, the algorithm prints occasional progress messages and finally
returns the best sequence encountered as a plain list of floats.
"""

import time
import math
import random
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , ∑ w = s }."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be equal
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (used for linear convolution via FFT)."""
    return 1 << ((2 * n - 1).bit_length())


def _linear_convolution(a: np.ndarray) -> np.ndarray:
    """Return the (linear) convolution a * a as a 1‑D NumPy array."""
    n = a.shape[0]
    L = _fft_len_for_n(n)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


def _build_toeplitz_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix T such that (T @ b) = f * b
    (linear convolution).  Entry T[k, j] = f[k‑j] whenever the index is
    in range, otherwise zero.
    """
    n = f.shape[0]
    # each column j receives a copy of f shifted by +j
    data = []
    offsets = []
    for i in range(n):
        # diagonal with value f[i] appears at offset i
        data.append(np.full(n, f[i], dtype=float))
        offsets.append(i)
    T = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return T


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence that minimizes ``evaluate_sequence``.
    The search consists of a linear‑programming refinement phase followed
    by a simulated‑annealing polishing phase.
    """
    # ------------------------------------------------------------------
    # Initialise random generators
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Choose problem size and an initial vector.
    # Prefer the previously discovered construction if it exists.
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=float)
    else:
        # otherwise a generous random start
        base = np.random.exponential(scale=1.0, size=2500)

    n = base.size
    target_sum = math.sqrt(2.0 * n)          # the ℓ₁‑norm required by the objective

    # Normalise the start vector to the required ℓ₁‑norm
    a = base.astype(float)
    a = a / a.sum() * target_sum
    a = np.maximum(a, 0.0)                   # safety

    # ------------------------------------------------------------------
    # Global time handling
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # Keep track of the best solution seen so far
    conv_a = _linear_convolution(a)
    best_val = float(conv_a.max())
    best_seq = a.copy()
    print(f"[INIT] n={n}, target_sum={target_sum:.3f}, initial max(conv)={best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Linear‑programming refinement.
    # Allocate roughly 30 % of the total budget to this deterministic phase.
    lp_deadline = start_time + budget_s * 0.30
    lp_iter = 0
    # a modest mixing coefficient – large values could destabilise the process
    mix_coeff = 0.20

    while time.time() < lp_deadline:
        # current max‑convolution and the bound M
        conv_a = _linear_convolution(a)
        M = float(conv_a.max())

        # Build Toeplitz matrix for the convolution f * b
        T = _build_toeplitz_matrix(a)

        # Linear program: maximise Σ b   ⇔   minimise −Σ b
        c = -np.ones(n, dtype=float)               # objective coefficients
        ub = np.full(2 * n - 1, M, dtype=float)    # RHS of inequalities
        bounds = [(0.0, None)] * n                  # b_j ≥ 0

        # Solve with HiGHS (fast interior‑point/simplex hybrid)
        res = opt.linprog(
            c,
            A_ub=T,
            b_ub=ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 30.0},
        )

        if not res.success:
            print(f"[LP {lp_iter:3d}] linprog failed: {res.message}")
            break

        b_opt = np.maximum(res.x, 0.0)        # guard against tiny negatives

        # Scale b_opt so that its ℓ₁‑norm equals the target sum
        sum_b = b_opt.sum()
        if sum_b == 0.0:
            print(f"[LP {lp_iter:3d}] zero solution – terminating LP phase.")
            break
        scale = target_sum / sum_b
        b_scaled = b_opt * scale

        # Mix the new vector with the current one
        a = (1.0 - mix_coeff) * a + mix_coeff * b_scaled

        # Re‑project onto the simplex to kill any rounding drift
        a = _simplex_projection(a, target_sum)

        # Evaluate improvement
        conv_a = _linear_convolution(a)
        cur_val = float(conv_a.max())
        if cur_val < best_val - 1e-9:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP {lp_iter:3d}] improved max(conv) = {best_val:.6f}")

        # Heuristic stop: if we are already very close to the target bound
        if best_val <= 1.505:
            print("[LP] Reached near‑target value – exiting LP phase early.")
            break

        lp_iter += 1

    # ------------------------------------------------------------------
    # Phase 2 – Simulated annealing (fast hill‑climbing) on the best vector.
    # Allocate the remaining budget to this stochastic polishing phase.
    sa_start = time.time()
    cur_val = best_val
    a = best_seq.copy()
    step_counter = 0

    # speed‑up: pre‑compute the FFT length for the SA phase
    fft_len = _fft_len_for_n(n)

    # Temperature schedule (exponential decay)
    T0 = cur_val * 0.15          # initial temperature (15 % of current max)
    Tmin = cur_val * 1e-5

    while time.time() < deadline:
        # Exponential cooling based on elapsed SA time
        elapsed = time.time() - sa_start
        frac = elapsed / (deadline - sa_start + 1e-12)
        T = T0 * ((Tmin / T0) ** frac)

        # Propose a random move: transfer a fraction of mass from i to j
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = a[i] * 0.10 * np.random.rand()   # move at most 10 % of source
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta

        # Evaluate candidate via FFT (linear convolution)
        fa = np.fft.rfft(a_cand, n=fft_len)
        conv_cand = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        cand_val = float(conv_cand.max())

        # Metropolis acceptance rule
        if cand_val < cur_val:
            a = a_cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = a_cand.copy()
                print(f"[SA {step_counter:6d}] new best max(conv) = {best_val:.6f}")
        else:
            prob = math.exp(-(cand_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = a_cand
                cur_val = cand_val

        step_counter += 1

    # ------------------------------------------------------------------
    print(f"[DONE] best max(conv) = {best_val:.6f} "
          f"after {lp_iter} LP iters and {step_counter} SA steps")
    # Return the best sequence as a plain Python list
    return best_seq.tolist()
