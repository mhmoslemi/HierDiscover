"""
Hybrid top‑K sub‑gradient descent with momentum.
-----------------------------------------------------------------
We minimise  F(a)=2·n·max(conv(a,a))  (the value returned by
`evaluate_sequence` for a normalised sequence Σa_i=1)  by iteratively
reducing the largest convolution entries.

At each iteration:
* Compute the full linear convolution a*a.
* Select the K largest convolution entries (the active set).
* Build a sparse weight vector w that equally weights those K entries.
* The sub‑gradient of the (non‑smooth) max can be taken as the
  convex combination of the gradients of the active entries:
        ∇max ≈ 2·(a_rev ∗ w) .
  (a_rev is a reversed copy of a.)
* Apply a momentum update (β≈0.9) and a diminishing step size
  step=α/√(t+1) to obtain a new point.
* Project back onto the probability simplex (clip to ≥0 and renorm).

To avoid stagnation we randomly transfer a modest amount of mass
between two coordinates with a small probability, and we restart from
the current best solution (with a tiny random jitter) if no improvement
has been seen for a few seconds.

The routine respects the 1000 s budget, prints occasional progress
messages and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT (real‑to‑real).
    `size` must be ≥ len(u)+len(v)-1.
    Returns the first len(u)+len(v)-1 samples.
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1)."""
    return _conv_fft(a, a, size)


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a random fraction of the mass from one index to another.
    `scale` controls the maximum fraction of the source cell that may be moved.
    """
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # re‑normalise to stay on the simplex
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= s
    return a_new

# Main entry point
# ----------------------------------------------------------------------

    """
    Search for a low value of `evaluate_sequence` within the given time budget.

    Parameters

    kwargs : dict
        ``n`` – length of the step‑function (default 6000).

    Returns
    -------
    list[float]
        The best sequence discovered (Python floats).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # tiny safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # initialise a feasible distribution (sum = 1)
    # --------------------------------------------------------------
    try:

        if seq0 is not None:

                a = np.interp(x_new, x_old, a)
            a = np.clip(a, 0.0, None)
            if a.sum() == 0.0:

    a /= a.sum()                     # normalise to the simplex

    size1 = _next_pow2(2 * n - 1)       # for a*a

            # momentum update

            # adaptive step size (diminishing, 1/√t)
            step = base_step / np.sqrt(it + 1.0)

            momentum.fill(0.0)

        it += 1

    # --------------------------------------------------------------
    # final reporting
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    # Return the best sequence found (as plain Python floats)
    return best_seq.tolist()
