    i = np.arange(n, dtype=np.float64)

"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------

1. **Initialisation** – evaluate a few deterministic families
   (uniform, product/sum power‑laws, and the supplied reference) and keep
   the best.
2. **Randomised family search** – for the remaining initial budget
   generate candidates from flexible low‑dimensional families
   (asymmetric power‑laws, mixtures, Gaussians, triangles).  Each
   candidate is normalised to sum 1 and its exact objective
   `2·n·max(conv)` is computed (no extra normalisation is needed
   because the score is scale‑invariant).  The best of these seeds is
   kept.
3. **Hybrid descent** – starting from the best seed, run a long
   optimisation loop that alternates three move types:
   * sub‑gradient w.r.t. the currently maximal convolution entry,
   * a smoothed‑max gradient (soft‑max over the convolution entries),
   * a small random mass‑transfer move.
   After each move the vector is projected onto the probability simplex.
   An annealing schedule slowly reduces the step size, the smoothing
   parameter λ and a Metropolis temperature.
   If no improvement is seen for a while a jittered restart from the
   current best vector is performed.
4. The best vector discovered before the time budget expires is returned
   as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """Exact objective for a non‑negative vector (assumed Σa = 1)."""
    conv = conv_fft(a)
    return float(2.0 * a.shape[0] * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def gradient_smooth(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smoothed max Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k).
    Returns ∇Φ_λ.
    """
    n = a.shape[0]
    conv = conv_fft(a)                     # length 2n‑1

    max_c = np.max(conv)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)        # soft‑max weights, sum = 1

    w_rev = w[::-1]                        # reverse for convolution
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)

    start = n - 1
    end = 2 * n - 1
    slice_part = conv_wa[start:end]        # length n
    g = 2.0 * slice_part[::-1]
    return g


def subgradient_from_idx(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the (non‑smoothed) max‑convolution at index k.
    For the objective 2·n·max(conv) the gradient is
        g_i = 2·a_{k‑i}   (zero when k‑i is out of range)
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    return g


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum=1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+off)*(n‑1‑i+off))^{‑p}
    mode = 'sum'      → a_i ∝ (i+off)^{‑p} + (n‑1‑i+off)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")

    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def power_left(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+offset)^{-p}   (monotone decreasing from the left)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def power_right(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (n‑1‑i+offset)^{-p}   (monotone decreasing from the right)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def mixture_power(n: int, p_l: float, off_l: float,
                  p_r: float, off_r: float, w: float) -> np.ndarray:
    """
    Convex mixture of a left‑hand and a right‑hand power‑law.
    w ∈ [0,1] is the weight of the left component.
    """
    left = power_left(n, p_l, off_l)
    right = power_right(n, p_r, off_r)
    a = w * left + (1.0 - w) * right
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def product_general(n: int, p1: float, p2: float,
                    off1: float, off2: float) -> np.ndarray:
    """
    Asymmetric product family:
        a_i ∝ (i+off1)^{-p1} * (n‑1‑i+off2)^{-p2}
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + off1) ** (-p1) * (n - 1 - i + off2) ** (-p2)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def gaussian_candidate(n: int, mu: float, sigma: float) -> np.ndarray:
    """Discrete Gaussian (truncated) centred at mu with std sigma."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def triangle_candidate(n: int, mu: float, width: float) -> np.ndarray:
    """Triangular bump of total width 2·width centred at mu."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - np.abs(i - mu) / width)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation function.
    A combination of analytic grid search, randomised family search and
    a hybrid (sub‑gradient / smoothed) descent is employed.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin before the hard timeout
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Deterministic initial candidates
    # ------------------------------------------------------------------
    best_seq = np.ones(n, dtype=np.float64) / n          # uniform
    best_score = compute_score(best_seq)

    print(f"[init] uniform baseline score = {best_score:.6f}")

    # optional global reference
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            print(f"[init] global reference score = {sc:.6f}")
            if sc < best_score:
                best_score, best_seq = sc, ref.copy()

    # analytic families (product and sum power‑laws)
    p_vals = np.arange(0.30, 0.95 + 1e-9, 0.05)
    offset_vals = np.arange(0.0, 1.6, 0.2)
    modes = ["product", "sum"]
    for mode in modes:
        for p in p_vals:
            for off in offset_vals:
                off_eff = off if (mode == "sum" or off > 0.0) else 1e-3
                cand = analytic_candidate(n, p, off_eff, mode)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
    print(f"[init] after analytic grid best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Randomised family search (short initial phase)
    # ------------------------------------------------------------------
    random_phase_end = min(start_time + 200.0, deadline - 200.0)
    while time.time() < random_phase_end:
        fam = np.random.randint(0, 6)   # six families below
        if fam == 0:  # asymmetric product
            p1 = np.random.uniform(0.2, 1.2)
            p2 = np.random.uniform(0.2, 1.2)
            off1 = np.random.uniform(0.0, 5.0)
            off2 = np.random.uniform(0.0, 5.0)
            cand = product_general(n, p1, p2, off1, off2)
        elif fam == 1:  # left power‑law
            p = np.random.uniform(0.2, 1.5)
            off = np.random.uniform(0.0, 5.0)
            cand = power_left(n, p, off)
        elif fam == 2:  # right power‑law
            p = np.random.uniform(0.2, 1.5)
            off = np.random.uniform(0.0, 5.0)
            cand = power_right(n, p, off)
        elif fam == 3:  # mixture of left/right
            p_l = np.random.uniform(0.2, 1.5)
            off_l = np.random.uniform(0.0, 5.0)
            p_r = np.random.uniform(0.2, 1.5)
            off_r = np.random.uniform(0.0, 5.0)
            w = np.random.rand()
            cand = mixture_power(n, p_l, off_l, p_r, off_r, w)
        elif fam == 4:  # Gaussian bump
            mu = np.random.uniform(0.0, n - 1)
            sigma = np.random.uniform(n / 200.0, n / 5.0)
            cand = gaussian_candidate(n, mu, sigma)
        else:  # Triangular bump
            mu = np.random.uniform(0.0, n - 1)
            width = np.random.uniform(n / 30.0, n / 3.0)
            cand = triangle_candidate(n, mu, width)

        # evaluate
        conv = conv_fft(cand)
        sc = 2.0 * n * np.max(conv)    # exact objective (sum = 1)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
    print(f"[init] after random families best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Hybrid descent (main optimisation loop)
    # ------------------------------------------------------------------
    a = best_seq.copy()               # current iterate, already on the simplex
    cur_conv = conv_fft(a)
    cur_max = np.max(cur_conv)
    cur_score = 2.0 * n * cur_max
    iteration = 0
    last_improve = time.time()
    # schedules (linear)
    lam_start, lam_end = 0.5, 5000.0
    step_start, step_end = 0.25, 5e-6
    temp_start, temp_end = 0.8, 1e-4
    restart_cooldown = 20.0           # seconds without improvement before jitter

    while time.time() < deadline:
        elapsed = time.time() - start_time
        prog = elapsed / (budget_s - 30.0)   # leave a small margin for finalisation
        # linearly annealed parameters
        lam = lam_start + prog * (lam_end - lam_start)
        step = step_start + prog * (step_end - step_start)
        temp = temp_start + prog * (temp_end - temp_start)

        # --------------------------------------------------------------
        # Choose move type
        # --------------------------------------------------------------
        r = np.random.rand()
        if r < 0.55:                     # sub‑gradient of the current max
            max_idx = int(np.argmax(cur_conv))
            g = subgradient_from_idx(a, max_idx)
            cand = a - step * g
            cand = np.clip(cand, 0.0, None)
            cand = project_simplex(cand)
        elif r < 0.85:                   # smoothed‑max gradient
            g = gradient_smooth(a, lam)
            cand = a - step * g
            cand = np.clip(cand, 0.0, None)
            cand = project_simplex(cand)
        else:                            # occasional random mass‑transfer
            cand = random_move(a, step * 0.5)

        # --------------------------------------------------------------
        # Evaluate candidate (exact objective)
        # --------------------------------------------------------------
        cand_conv = conv_fft(cand)
        cand_max = np.max(cand_conv)
        cand_score = 2.0 * n * cand_max

        # --------------------------------------------------------------
        # Acceptance (Metropolis style)
        # --------------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            if temp > 1e-12:
                prob = math.exp(-(cand_score - cur_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = cand
            cur_conv = cand_conv
            cur_max = cand_max
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:06d}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # --------------------------------------------------------------
        # Restart / jitter if stagnated
        # --------------------------------------------------------------
        if (iteration % 5000 == 0) and (time.time() - last_improve > restart_cooldown):
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            cur_conv = conv_fft(a)
            cur_max = np.max(cur_conv)
            cur_score = 2.0 * n * cur_max
            print(f"[{iteration:06d}] jitter restart (stagnation)")

        # early exit if we already reach the target bound
        if best_score <= 1.503:
            print("[done] target reached, exiting early")
            break

        iteration += 1

    total_time = time.time() - start_time
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    # final safeguard – ensure non‑negativity and proper normalisation
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() == 0.0:
        best_seq = np.ones_like(best_seq) / n
    else:
        best_seq = best_seq / best_seq.sum()
    return best_seq.tolist()
