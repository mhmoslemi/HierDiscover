        # step
        a = a - eta * mom
        # project onto simplex
        a = _project_onto_simplex(a)

        # update step sizes
        eta = max(eta * eta_decay, min_eta)

"""
Propose a step‑function (non‑negative list) that minimizes the
``evaluate_sequence`` objective.

Algorithm
---------

1. **Length / starting shape** – Use the globally supplied
   ``height_sequence_1`` if it exists, otherwise search a small grid of
   power‑law shapes `a_i ∝ (i+1+offset)^{-α}` (α∈[0.3,0.7], offset∈[0,5]).
   The best shape (measured by the cheap bound `2·n·max(a*a)`) is taken
   as the initial distribution and projected onto the probability simplex
   (sum = 1).

2. **p‑norm smoothed descent** –  For a large exponent `p` the
   ℓₚ‑norm `‖a*a‖ₚ` approximates `‖a*a‖_∞`.  In each iteration we
   (i) compute the self‑convolution `c = a*a`, (ii) form a weight vector
   `w = (c / max(c))^{p‑1}` (so the largest entry gets weight 1 and the
   others are down‑weighted), (iii) obtain the gradient of
   `∑_k c_k^p` as `∇ = 2·p·(a * w_rev)` (full convolution of `a` with the
   reversed weight vector), (iv) take a momentum step and project back
   onto the simplex.  The exponent `p` is increased slowly (starting from
   4 up to a configurable maximum, e.g. 200) so that the optimisation
   gradually focuses on the true maximum.

3. **Exploration** – Every few hundred iterations a small random
   mass‑exchange between two coordinates is performed; if it improves
   the exact score it is kept.  This helps to escape shallow plateaus.

4. **Book‑keeping** – The exact objective (via ``evaluate_sequence``) is
   evaluated periodically (every 1000 iterations) and the best sequence
   seen so far is stored.  The routine stops a few seconds before the
   time budget expires and returns the best sequence.

The whole method runs only with NumPy, respects the 1000 s budget and
returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i ≥ 0 , Σ x_i = 1 }.
    Implements the algorithm of Duchi et al. (ICML 2008).

    Parameters
    ----------
    v : np.ndarray
        Input vector (any real values).

    Returns
    -------
    np.ndarray
        Projected vector (non‑negative, sum‑1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.

    Parameters
    ----------
    x, y : np.ndarray
        1‑D real arrays.

    Returns
    -------
    np.ndarray
        Length = len(x)+len(y)-1 convolution.
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Normalised power‑law sequence a_i ∝ (i+1+offset)^{-α}.

    Parameters
    ----------
    n      : int   – length of the sequence.
    alpha  : float – exponent (α>0).
    offset : float – optional index shift.

    Returns
    -------
    np.ndarray
        Length‑`n` probability vector (sum = 1, all entries ≥ 0).
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _fast_score(a: np.ndarray) -> float:
    """
    Cheap surrogate of the true objective when Σa = 1:
        score ≈ 2·n·max(a*a)
    This is exactly what ``evaluate_sequence`` returns under the sum‑= 1
    normalisation (apart from the clipping that never activates for our
    values).
    """
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The routine respects the time budget and
    returns the best sequence discovered.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  A few seconds are kept as safety margin.
    kwargs : dict
        Unused – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence (non‑negative, Σ a > 0.01) found.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin for final return

    # ---------------------------------------------------------------

        s = a.sum()
        if s > 0.0:

        else:
            raise ValueError
    except Exception:
        # fallback: pick a length (5000 works well) and search a small grid
        n = 5000
        best_a = _power_sequence(n, alpha=0.5)   # default
        best_score = _fast_score(best_a)

        alphas = np.linspace(0.35, 0.65, 13)      # 13 values

        for alpha in alphas:

                    best_a = cand

        a = best_a

    # ensure we are exactly on the simplex

    # ---------------------------------------------------------------

    cur_score = _fast_score(a)
    print(f"[init] n={n}, fast_score={cur_score:.6f}, exact={best_val:.6f}")

    # ---------------------------------------------------------------
    # 3. p‑norm smoothed gradient descent with momentum
    # ---------------------------------------------------------------
    # step‑size schedule
    lr = 0.15
    lr_min = 1e-7
    lr_decay = 0.99993

    p = 4.0
    p_max = 200.0

    beta = 0.85    # momentum coefficient

    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # ---- weights for p‑norm gradient ----

        # ---- compute gradient of Σ conv^p ----
        w_rev = w[::-1]
        v_full = _conv_fft(a, w_rev)           # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)       # slice that yields the true gradient
        v = v_full[idx]                        # shape (n,)
        grad = 2.0 * p * v                     # ∇(∑ conv^p)

        # ---- momentum & projection ----
        momentum = beta * momentum + grad
        a = a - lr * momentum
        a = _project_onto_simplex(a)

        # ---- schedule updates ----
        lr = max(lr * lr_decay, lr_min)
        if p < p_max:
            p = min(p_max, p + p_inc)

        # ---- occasional random mass‑exchange (exploration) ----

                best_seq = a.copy()

                      f"fast={_fast_score(a):.6f}, p={p:.1f}, lr={lr:.2e}")

            pert = np.random.randn(n) * 1e-3
            a = best_seq + pert
            a = _project_onto_simplex(a)
            lr = 0.15      # reset step‑size
            p = 4.0        # reset exponent schedule
            momentum.fill(0.0)
            last_improve_iter = iteration
            print(f"[restart] iter={iteration}")

    # ---------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.5 and best_seq.size > 0:

        n = a.size

                best_seq = cand.copy()
                a = cand

    # ---------------------------------------------------------------
    # 5. Return the best sequence discovered
    # ---------------------------------------------------------------
    print(f"[final] best_score={best_val:.6f}, total_iters={iteration}")
    return best_seq.tolist()
