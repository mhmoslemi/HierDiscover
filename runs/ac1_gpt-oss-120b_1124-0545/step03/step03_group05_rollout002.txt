Standard solution from Duchi (2008):

def simplex_projection(v):
    """Project v onto probability simplex (sum=1, v>=0). v is 1D np.ndarray."""
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho, the last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _spectral_flatten(a, size_fft):
    X = np.fft.rfft(a, n=size_fft)
    # keep DC component unchanged
    dc = X[0]
    mag = np.abs(X[1:])
    # compute average magnitude of non-DC components
    avg_mag = np.sqrt(np.mean(mag**2))
    # set magnitude to avg_mag, keep phase
    angles = np.angle(X[1:])
    X_new = np.empty_like(X)
    X_new[0] = dc
    X_new[1:] = avg_mag * np.exp(1j * angles)
    # reconstruct time domain
    a_new = np.fft.irfft(X_new, n=size_fft)[:a.shape[0]]
    # clip negative, renorm
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() == 0:
        a_new = np.random.rand(a.shape[0])
    a_new /= a_new.sum()
    return a_new

"""
Improved search for a low‑value step‑function sequence.

Algorithm Overview
------------------
* Initialise a feasible probability vector `a` (sum = 1, a_i ≥ 0).  If a
  global variable ``height_sequence_1`` exists it is interpolated,
  otherwise a random vector is used.
* Perform *Nesterov‑accelerated* projected gradient descent on a
  smooth approximation of the max‑convolution.
  – The smooth max is realised with a soft‑max weighting
    `w_k ∝ exp((b_k−max(b))/τ)`, where `b = a * a`.
  – Gradient of the smooth objective w.r.t. `a` is `‑2·(a_rev ∗ w)`.
  – After a gradient step the iterate is projected onto the probability
    simplex by the exact Euclidean projection (Duchi et al.).
* Occasionally (probability `rand_prob`) a cheap random “mass–transfer’’
  move is performed to escape flat plateaux.
* Every `spectral_interval` iterations the spectrum of `a` is flattened
  (all non‑DC Fourier magnitudes are replaced by their RMS value) and the
  vector is projected back – this tends to spread the mass and lowers the
  peak of the convolution.
* If no improvement has been seen for `stagnation_limit` seconds the
  algorithm restarts from a fresh random point.
* The best sequence seen during the entire 1000 s budget is returned.

The combination of Nesterov momentum, exact simplex projection,
spectral flattening and adaptive annealing of the soft‑max temperature
generally yields a lower bound than the plain sub‑gradient method.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – simple, no closures, no lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT. ``size`` must be ≥ len(u)+len(v)‑1.
    Returns the full convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size1: int) -> np.ndarray:
    """Linear convolution a*a (full length)."""
    return _conv_fft(a, a, size1)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # re‑normalise (keeps sum = 1)
    a_new /= a_new.sum()
    return a_new


def _spectral_flatten(a: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Replace all non‑DC Fourier magnitudes by their RMS value,
    keep the phases, invert and project back onto the simplex.
    """
    X = np.fft.rfft(a, n=size_fft)
    # preserve the DC component
    dc = X[0]
    # RMS magnitude of the remaining frequencies
    mags = np.abs(X[1:])
    rms = np.sqrt(np.mean(mags * mags))
    # build new spectrum
    angles = np.angle(X[1:])
    X_new = np.empty_like(X)
    X_new[0] = dc
    X_new[1:] = rms * np.exp(1j * angles)
    # inverse transform, keep first len(a) entries
    a_new = np.fft.irfft(X_new, n=size_fft)[: a.shape[0]]
    a_new = np.clip(a_new, 0.0, None)
    if a_new.sum() == 0.0:
        a_new = np.random.rand(a.shape[0])
    a_new /= a_new.sum()
    return a_new


def _initial_a(n: int) -> np.ndarray:
    """Return a feasible starting probability vector."""
    # try to reuse a globally supplied construction
    seq0 = globals().get("height_sequence_1", None)
    if seq0 is not None:
        a = np.asarray(seq0, dtype=float)
        if a.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0, 1, a.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            a = np.interp(x_new, x_old, a)
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.random.rand(n)
        a /= a.sum()
        return a
    # fallback: a smooth sine‑squared shape (often a good start)
    x = np.arange(n, dtype=float)
    a = np.sin(np.pi * (x + 0.5) / n) ** 2
    a += 1e-8                     # avoid exact zeros
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The implementation runs for at most ``budget_s``
    seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # problem size -------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # FFT sizes (powers of two) – keep them constant
    size1 = _next_pow2(2 * n - 1)       # for a*a
    size2 = _next_pow2(3 * n - 2)       # for a_rev * weight
    size_fft_spectral = _next_pow2(n)   # for the spectral flatten step

    # initialise ---------------------------------------------------------
    a = _initial_a(n)                  # probability vector (sum = 1)
    a_prev = a.copy()                  # for Nesterov momentum

    # bookkeeping
    conv_full = _full_convolution(a, size1)
    best_score = 2.0 * n * conv_full.max()
    best_seq = a.copy()
    it = 0
    last_improve = start

    # algorithmic hyper‑parameters ---------------------------------------
    base_step = 0.8                     # base step size for gradient descent
    rand_prob = 0.09                    # probability of a random transfer move
    rand_scale = 0.045                  # magnitude of random transfers
    tau_factor = 0.6                    # initial τ = τ_factor * max_b
    tau_decay = 0.9996                  # exponential decay of τ‑factor
    stagnation_limit = 20.0             # seconds without improvement → restart
    spectral_interval = 6000            # apply spectral flattening every N iters
    beta = 0.8                          # Nesterov momentum coefficient

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Nesterov extrapolation
        y = a + beta * (a - a_prev)

        # full convolution of the extrapolated point
        conv_y = _full_convolution(y, size1)
        max_b = conv_y.max()
        cur_score = 2.0 * n * max_b

        # record best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:8d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # decide on the move
        if np.random.rand() < rand_prob:
            # cheap random perturbation
            a = _random_transfer(a, rand_scale)
        else:
            # smooth‑max gradient step
            tau = max(tau_factor * max_b, 1e-14)
            tau_factor *= tau_decay                     # anneal temperature

            # soft‑max weights
            weight = np.exp((conv_y - max_b) / tau)
            w_sum = weight.sum()
            if w_sum == 0.0:
                weight = np.ones_like(weight) / weight.size
            else:
                weight /= w_sum

            # gradient = -2 * (y_rev ∗ weight)
            y_rev = y[::-1]
            g_full = _conv_fft(y_rev, weight, size2)
            g = g_full[n - 1 : n - 1 + n]               # central n entries
            grad = -2.0 * g

            # adaptive step size (diminishing)
            step = base_step / np.sqrt(it + 1.0)

            # gradient descent step on the extrapolated point
            y_new = y - step * grad

            # project back onto the simplex (exact Euclidean projection)
            a_new = _project_simplex(y_new)

            # update momentum variables
            a_prev = a
            a = a_new

        # occasional spectral flattening
        if it % spectral_interval == 0 and it > 0:
            a = _spectral_flatten(a, size_fft_spectral)

        # stagnation handling
        if time.time() - last_improve > stagnation_limit:
            a = np.random.rand(n)
            a /= a.sum()
            a_prev = a.copy()
            tau_factor = 0.6                         # reset temperature schedule
            last_improve = time.time()
            print(f"[{it:8d}] restart due to stagnation")

        it += 1

    # ------------------------------------------------------------------
    # final reporting
    # ------------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
