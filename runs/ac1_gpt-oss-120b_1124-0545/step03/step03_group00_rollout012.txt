def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    lengths = [8000, 4000]
    global_best_seq = None
    global_best_val = float('inf')

    for n in lengths:

        if time.time() > deadline:
            break

        # Build seed list
        seeds = []
        # add analytical families
        for c in np.linspace(0.5,5,20):
            r = np.exp(-c/n)
            seeds.append(_geom(n,r)) # geometric
        # power
        for p in np.linspace(0.1,5,20):
            seeds.append(_power(n,p))
        # combined
        for c in np.linspace(0.5,4,12):
            r = np.exp(-c/n)
            for p in np.linspace(0.1,3,12):
                seeds.append(_geom_power(n,r,p))
        # symmetric geometric
        for c in np.linspace(0.5,5,20):
            r = np.exp(-c/n)
            seeds.append(_sym_geom(n,r))
        # sine window
        seeds.append(_sine_window(n))
        # Hann, Hamming, Blackman
        seeds.append(np.hanning(n))
        seeds.append(np.hamming(n))
        seeds.append(np.blackman(n))
        # Kaiser
        for beta in np.linspace(2, 14, 10):
            try:
                w = np.kaiser(n, beta)
                w = w / w.sum()
                seeds.append(w)
            except Exception:
                pass
        # DPSS
        try:
            from scipy.signal import windows
            for NW in [2, 3, 4, 5, 6, 7]:
                w = windows.dpss(n, NW, Kmax=1, sym=True)
                w = w / w.sum()
                seeds.append(w)
        except Exception:
            pass

        # random Dirichlet samples
        for alpha in [0.2,0.5,0.8,2,5]:
            a = np.random.gamma(alpha,1.0,size=n)
            a /= a.sum()
            seeds.append(a)

        # random softmax seeds
        for _ in range(10):
            v = np.random.randn(n)
            seeds.append(_softmax(v))

        # Evaluate all seeds quickly
        for idx, a in enumerate(seeds):
            cur_val = 2.0*n*np.max(_conv_fft(a,a))
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
        # Parameter optimization for geometric shapes (refine c)
        def geom_obj(c):
            r = np.exp(-c/n)
            a = _geom(n, r)
            return 2.0*n*np.max(_conv_fft(a,a))
        # golden-section search for c in (0.01, 10)
        c_low, c_high = 0.01, 10.0
        phi = (1 + np.sqrt(5))/2
        for _ in range(25):
            c1 = c_high - (c_high - c_low) / phi
            c2 = c_low + (c_high - c_low) / phi
            f1 = geom_obj(c1)
            f2 = geom_obj(c2)
            if f1 < f2:
                c_high = c2
                if f1 < global_best_val:
                    global_best_val = f1
                    global_best_seq = _geom(n, np.exp(-c1/n)).copy()
            else:
                c_low = c1
                if f2 < global_best_val:
                    global_best_val = f2
                    global_best_seq = _geom(n, np.exp(-c2/n)).copy()
        # similarly for sym_geom
        def symgeom_obj(c):
            r = np.exp(-c/n)
            a = _sym_geom(n, r)
            return 2.0*n*np.max(_conv_fft(a,a))
        c_low, c_high = 0.01, 10.0
        for _ in range(25):
            c1 = c_high - (c_high - c_low)/phi
            c2 = c_low + (c_high - c_low)/phi
            f1 = symgeom_obj(c1)
            f2 = symgeom_obj(c2)
            if f1 < f2:
                c_high = c2
                if f1 < global_best_val:
                    global_best_val = f1
                    global_best_seq = _sym_geom(n, np.exp(-c1/n)).copy()
            else:
                c_low = c1
                if f2 < global_best_val:
                    global_best_val = f2
                    global_best_seq = _sym_geom(n, np.exp(-c2/n)).copy()
        # Parameter optimization for power law
        def power_obj(p):
            a = _power(n,p)
            return 2.0*n*np.max(_conv_fft(a,a))
        p_low, p_high = 0.01, 10.0
        for _ in range(25):
            p1 = p_high - (p_high - p_low)/phi
            p2 = p_low + (p_high - p_low)/phi
            f1 = power_obj(p1)
            f2 = power_obj(p2)
            if f1 < f2:
                p_high = p2
                if f1 < global_best_val:
                    global_best_val = f1
                    global_best_seq = _power(n,p1).copy()
            else:
                p_low = p1
                if f2 < global_best_val:
                    global_best_val = f2
                    global_best_seq = _power(n,p2).copy()
        # Now run refinement using Adam
        if global_best_seq is None:
            global_best_seq = np.ones(n) / n
        refined_seq, refined_val = _refine_sequence(global_best_seq, n, deadline, start_beta=2.0, beta_mul=1.015, lr0=0.5)
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()

        # Additional perturbation restarts while time left
        while time.time() < deadline:
            # add small noise in log space
            logits = np.log(global_best_seq + 1e-30) + 0.01 * np.random.randn(n)
            pert_seq = _softmax(logits)
            seq2, val2 = _refine_sequence(pert_seq, n, deadline, start_beta=2.0, beta_mul=1.015, lr0=0.5)
            if val2 < global_best_val:
                global_best_val = val2
                global_best_seq = seq2.copy()
    # final normalization, return list
    if global_best_seq is None:
        global_best_seq = np.ones(4000) / 4000
    s = global_best_seq.sum()
    if s > 0:
        global_best_seq = global_best_seq / s
    return global_best_seq.tolist()

"""
Propose a non‑negative step‑function (list of heights) that minimises the
objective used by ``evaluate_sequence``:

    obj = 2 * n * max(conv(a, a)) / (sum(a)**2)

The function works in three phases while respecting a 1000 s time budget.

1. **Seed generation** – a wide variety of analytically‑motivated probability
   vectors (geometric, power‑law, combined, symmetric, sine, Hann, Hamming,
   Blackman, Kaiser and DPSS windows) and a few random Dirichlet draws are
   built.  For each length we also keep the previously found best construction
   (if any) as an extra seed.

2. **Parameter refinement** – the most promising 1‑parameter families
   (geometric, symmetric‑geometric and pure power‑law) are optimised with a
   cheap golden‑section search on the single shaping parameter; the best
   vectors from this stage are added to the seed pool.

3. **Continuous optimisation** – the current best vector is refined by an
   Adam optimiser on the log‑softmax parametrisation of the simplex.
   The loss is a smoothed log‑sum‑exp approximation of the maximal
   autocorrelation; the smoothing parameter (``beta``) is increased
   geometrically so the optimiser eventually targets the true maximum.
   After the main run a few stochastic perturbations are launched as
   additional restarts.

The final vector is normalised to sum‑to‑one (the objective is scale‑invariant)
and returned as a plain Python list.  All numbers are non‑negative as required
by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using FFT (real inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple windows / families that generate probability vectors
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric distribution a_i ∝ r**i, normalised to sum‑one."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0.0:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined geometric × power‑law a_i ∝ r**i / (i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    if s == 0.0:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric: a_i ∝ r**i + r**(n-1-i)."""
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    return np.hanning(n).astype(np.float64) / np.hanning(n).sum()


def _hamming_window(n: int) -> np.ndarray:
    return np.hamming(n).astype(np.float64) / np.hamming(n).sum()


def _blackman_window(n: int) -> np.ndarray:
    return np.blackman(n).astype(np.float64) / np.blackman(n).sum()


def _kaiser_windows(n: int) -> list[np.ndarray]:
    """A small collection of Kaiser windows with various betas."""
    ws = []
    for beta in np.linspace(2.0, 14.0, 7):
        w = np.kaiser(n, beta).astype(np.float64)
        s = w.sum()
        if s > 0:
            ws.append(w / s)
    return ws


def _dpss_windows(n: int) -> list[np.ndarray]:
    """DPSS (Slepian) windows, if scipy is available."""
    ws = []
    try:
        from scipy.signal import windows
        for NW in range(2, 8):
            w = windows.dpss(n, NW, Kmax=1, sym=True).astype(np.float64)
            s = w.sum()
            if s > 0:
                ws.append(w / s)
    except Exception:
        pass
    return ws


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax simplex (smooth max loss)
# ----------------------------------------------------------------------


def _refine_sequence(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 2.0,
    beta_mul: float = 1.015,
    lr0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑softmax parametrisation.
    Returns the best sequence found (as np.ndarray) and its fast
    objective value ``2 * n * max(conv)``.
    """
    v = np.log(a0 + 1e-30)          # logits
    m = np.zeros_like(v)           # first moment
    v2 = np.zeros_like(v)          # second moment
    best_seq = a0.copy()
    cur_conv = _conv_fft(a0, a0)
    best_val = 2.0 * n * np.max(cur_conv)

    beta = start_beta
    it = 0

    while time.time() < deadline:
        it += 1

        # current distribution on the simplex
        a = _softmax(v)

        # convolution and (smooth) objective
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ---- smoothed‑max weights ----
        beta *= beta_mul
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:   # numerical safety
            continue
        w /= w_sum

        # ---- gradient w.r.t. a (smooth loss) ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length = 3n-2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]         # central n entries

        # ---- gradient w.r.t. logits v (softmax) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam update ----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Golden‑section search helpers for 1‑parameter families
# ----------------------------------------------------------------------


def _golden_section(func, lo: float, hi: float, iters: int = 25):
    """Return (x_opt, f_opt) after a golden‑section search on *func*."""
    phi = (1 + 5 ** 0.5) / 2.0
    x1 = hi - (hi - lo) / phi
    x2 = lo + (hi - lo) / phi
    f1 = func(x1)
    f2 = func(x2)
    for _ in range(iters - 1):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - (hi - lo) / phi
            f1 = func(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + (hi - lo) / phi
            f2 = func(x2)
    if f1 < f2:
        return x1, f1
    else:
        return x2, f2

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    """

    # length candidates – try the longer one first
    length_candidates = [8000, 4000]

    overall_best_seq = None
    overall_best_val = float("inf")

    for n in length_candidates:
        if time.time() > deadline:
            break

        print(f"[main] Optimising length = {n}")

        # ------------------------------------------------------------------
        # 1️⃣  Seed generation (all vectors are already normalised to sum=1)
        # ------------------------------------------------------------------
        seed_vecs = []

        for c in np.linspace(0.2, 6.0, 20):
            r = math.exp(-c / n)
            seed_vecs.append(_geom(n, r))

        # power‑law families
        for p in np.linspace(0.0, 4.0, 20):
            seed_vecs.append(_power(n, p))

        # combined geometric × power families (coarse grid)
        for c in np.linspace(0.5, 4.5, 12):
            r = math.exp(-c / n)
            for p in np.linspace(0.0, 2.5, 12):
                seed_vecs.append(_geom_power(n, r, p))

        # symmetric geometric
        for c in np.linspace(0.2, 6.0, 20):
            r = math.exp(-c / n)
            seed_vecs.append(_sym_geom(n, r))

        # classic smooth windows
        seed_vecs.append(_sine_window(n))
        seed_vecs.append(_hann_window(n))
        seed_vecs.append(_hamming_window(n))
        seed_vecs.append(_blackman_window(n))
        seed_vecs.extend(_kaiser_windows(n))
        seed_vecs.extend(_dpss_windows(n))

        # random Dirichlet draws
        for alpha in [0.2, 0.5, 0.8, 2.0, 5.0]:
            a = np.random.gamma(alpha, 1.0, size=n)
            if a.sum() > 0:
                seed_vecs.append(a / a.sum())

        # random soft‑max seeds
        for _ in range(8):

                        m = min(prev_arr.shape[0], n)

                    if prev_arr.sum() > 0:
                        seed_vecs.append(prev_arr / prev_arr.sum())
            except Exception:
                pass

        # ------------------------------------------------------------------

        best_val_n = float("inf")

            cur_val = 2.0 * n * np.max(_conv_fft(a, a))
            if cur_val < best_val_n:
                best_val_n = cur_val
                best_seq_n = a.copy()
                print(f"[scan] new best from seed {idx:3d} = {cur_val:.6f}")

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------

            r = math.exp(-c / n)
            a = _geom(n, r)
            return 2.0 * n * np.max(_conv_fft(a, a))

        if f_opt < best_val_n:
            best_val_n = f_opt
            best_seq_n = _geom(n, math.exp(-c_opt / n)).copy()
            print(f"[opt] geometric c={c_opt:.4f} obj={f_opt:.6f}")

        # symmetric geometric
        def symgeom_obj(c):
            r = math.exp(-c / n)
            a = _sym_geom(n, r)
            return 2.0 * n * np.max(_conv_fft(a, a))

        c_opt, f_opt = _golden_section(symgeom_obj, 0.01, 10.0, iters=25)
        if f_opt < best_val_n:
            best_val_n = f_opt
            best_seq_n = _sym_geom(n, math.exp(-c_opt / n)).copy()
            print(f"[opt] sym‑geom c={c_opt:.4f} obj={f_opt:.6f}")

        def power_obj(p):
            a = _power(n, p)
            return 2.0 * n * np.max(_conv_fft(a, a))

        p_opt, f_opt = _golden_section(power_obj, 0.01, 8.0, iters=25)
        if f_opt < best_val_n:
            best_val_n = f_opt
            best_seq_n = _power(n, p_opt).copy()
            print(f"[opt] power p={p_opt:.4f} obj={f_opt:.6f}")

        # 4️⃣  Refinement with Adam (log‑softmax) – runs until time runs out
        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        while time.time() < deadline:

                pert_seq,
                n,
                deadline,
                start_beta=2.0,
                beta_mul=1.015,
                lr0=0.5,
            )
            if val2 < best_val_n:

                best_seq_n = seq2.copy()
                print(f"[perturb] improved to {val2:.6f}")

        # --------------------------------------------------------------
        # Keep the globally best construction (across lengths)
        # --------------------------------------------------------------
        if best_val_n < overall_best_val:
            overall_best_val = best_val_n
            overall_best_seq = best_seq_n.copy()
            print(f"[overall] new global best (len={n}) = {best_val_n:.6f}")

    # ----------------------------------------------------------------------
    # Final sanity checks and return
    # ----------------------------------------------------------------------
    if overall_best_seq is None:
        # fall back to a uniform vector of length 4000
        overall_best_seq = np.ones(4000, dtype=np.float64) / 4000
        print("[fallback] using uniform construction")

    # Normalise to sum‑to‑one (objective is scale‑invariant)
    total = overall_best_seq.sum()
    if total > 0:
        overall_best_seq = overall_best_seq / total

    print(f"[final] best objective (fast) = {overall_best_val:.6f}")
    return overall_best_seq.tolist()
