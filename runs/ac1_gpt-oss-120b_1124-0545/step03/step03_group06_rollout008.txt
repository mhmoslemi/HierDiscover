"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Choose a relatively large length `n` (≈ 6·10⁴ users a good trade‑off
   between asymptotic quality and FFT speed).

2. Sample a low‑dimensional parametric family
      a_i ∝ (i+1+δ)^{-α}·exp(-λ·i)
   (α ≈ 0.4‑0.7, δ ≥ 0, λ ≥ 0) many times, keeping the best normalised
   candidate according to the exact objective (computed with an FFT‑based
   self‑convolution).

3. Refine the best candidate with a *projected sub‑gradient* method that
   uses the gradient of the *single* largest convolution entry.
   The update is
        a ← a – η·g ,   g_i = 2·a_{k_max‑i}   (zero outside the support)
   followed by Euclidean projection onto the probability simplex.
   Step size η decays slowly; every few hundred iterations a small random
   mass exchange is performed to escape plateaus.

4. A short series of random restarts (random perturbations of the current
   best sequence) are processed with a reduced step size.

5. Finally the sequence is symmetrised
        a ← (a + a[::-1]) / 2
   and re‑projected onto the simplex – this often lowers the peak of the
   self‑convolution for the same ℓ₁‑mass.

All heavy work (convolutions, gradients) uses NumPy’s FFT, which makes a
single iteration on n≈6·10⁴ cost only a few milliseconds, allowing many
thousands of iterations within the 1000 s budget.
The function returns a plain list of floats; the harness will call
``evaluate_sequence`` once on the result.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """
    Self‑convolution using a real FFT.  Returns the full length‑(2n‑1)
    vector.
    """
    n = seq.shape[0]
    size = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    conv[conv < 0] = 0.0            # clip tiny negatives from round‑off
    return conv


def _score_fft(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv) for a normalised sequence (Σseq == 1).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the O(n) algorithm of
    Duchi et al. (2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

# Parameterised families

def _make_seq_power_exp(
    n: int, alpha: float, offset: float = 0.0, lam: float = 0.0
) -> np.ndarray:
    """
    a_i ∝ (i+1+offset)^{-alpha} · exp(-lam·i), normalised to sum 1.
    All parameters are non‑negative; ``lam`` is a small exponential cut‑off.
    """
    idx = np.arange(n, dtype=float)
    # (i+1+offset) to avoid division by zero when offset=0
    base = (idx + 1.0 + offset) ** (-alpha)
    if lam > 0.0:
        base *= np.exp(-lam * idx)
    s = base.sum()
    if s == 0.0:
        base[:] = 1.0 / n
    else:
        base /= s
    return base

# Gradient descent using the *single* maximal convolution entry
# ----------------------------------------------------------------------


def _subgrad_max(seq: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at the current maximal index k_max.
    For each i the derivative is 2·seq[k_max‑i] when the index is valid.
    """
    n = seq.shape[0]
    grad = np.zeros_like(seq)
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    i_range = np.arange(lo, hi + 1, dtype=int)
    grad[i_range] = 2.0 * seq[k_max - i_range]
    return grad


def _run_gradient_max(
    seq: np.ndarray,
    time_budget: float,
    *,
    step0: float = 0.6,
    decay: float = 0.99995,
    max_iters: int = 10 ** 9,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that always moves opposite to the
    gradient of the *largest* entry of the self‑convolution.
    Returns the best sequence seen and its exact score.
    """
    start = time.time()
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = _score_fft(seq)

    step = step0
    it = 0

    while it < max_iters and (time.time() - start) < time_budget:
        it += 1

        conv = _conv_fft(seq)
        k_max = int(np.argmax(conv))

        grad = _subgrad_max(seq, k_max)

        # make the gradient tangent to the hyperplane Σx = 1
        grad -= grad.mean()

        new_seq = seq - step * grad
        new_seq = _project_simplex(new_seq)

        cur_score = _score_fft(new_seq)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = new_seq.copy()
            print(f"[improve] iter={it:,}, score={best_score:.6f}")

        # occasional random mass exchange (helps escape flat plateaus)
        if it % 300 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(step * new_seq[i], new_seq[i] * 0.25)
            new_seq[i] -= eps
            new_seq[j] += eps
            new_seq = _project_simplex(new_seq)

        seq = new_seq
        step = max(step * decay, 1e-9)

    return best_seq, best_score

            if cand.size > 0 and np.isfinite(cand).all():
                cand = np.maximum(cand, 0.0)
                if cand.sum() == 0.0:
                    raise ValueError
                cand /= cand.sum()
                if cand.size < n:
                    cand = np.concatenate([cand, np.zeros(n - cand.size)])
                elif cand.size > n:
                    cand = cand[:n]
                best_seq = cand.copy()
                best_score = _score_fft(best_seq)
                print("[init] seed from height_sequence_1, score={:.6f}".format(best_score))
        except Exception:
            best_seq = None

    # ------------------------------------------------------------------
    # 3. Random search over the low‑dimensional parametric family.
    #    Keep the best candidate found before the deadline.
    # ------------------------------------------------------------------
    # Rough budget for this phase (leave ~30 % for gradient polishing)
    param_phase_deadline = deadline - 180.0
    while time.time() < param_phase_deadline:
        alpha = np.random.uniform(0.35, 0.75)    # power‑law exponent
        offset = np.random.uniform(0.0, 5.0)    # small shift of the first term
        lam = np.random.uniform(0.0, 1.5e-3)    # exponential cut‑off
        seq = _make_seq_power_exp(n, alpha, offset, lam)
        score = _score_fft(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print("[param] new best {:.6f} (α={:.4f}, δ={:.3f}, λ={:.5f})".format(
                best_score, alpha, offset, lam))

    # ------------------------------------------------------------------
    # 4. Main projected gradient descent phase (long run).
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a plain uniform start
        best_seq = np.full(n, 1.0 / n, dtype=float)

    remaining = max(0.0, deadline - time.time())
    best_seq, best_score = _run_gradient_max(
        best_seq,
        time_budget=remaining,
        step0=0.7,
        decay=0.99997,
        max_iters=10 ** 9,
    )
    print("[grad‑phase‑1] best_score={:.6f}".format(best_score))

    # ------------------------------------------------------------------
    # 5. Short random restarts (cheap polishing).
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:
        # perturb the current best a little and run a brief descent
        seq = best_seq + np.random.randn(n) * 5e-6
        seq = np.maximum(seq, 0.0)
        if seq.sum() == 0.0:
            continue
        seq /= seq.sum()
        remaining = max(0.0, deadline - time.time())
        seq, sc = _run_gradient_max(
            seq,
            time_budget=remaining,
            step0=0.4,
            decay=0.9999,
            max_iters=3000,
        )
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("[restart] improved to {:.6f}".format(best_score))

    # ------------------------------------------------------------------
    # 6. Final symmetrisation – often reduces the peak marginally.
    # ------------------------------------------------------------------
    best_seq = (best_seq + best_seq[::-1]) * 0.5
    best_seq = _project_simplex(best_seq)

    final_score = _score_fft(best_seq)
    print("[final] best_score={:.6f}, length={}".format(final_score, n))

    # Return as a plain Python list.
    return best_seq.tolist()
