```
def _gradient_step_max(seq, step):
    n = seq.shape[0]
    conv = _conv_fft(seq)
    k = int(conv.argmax())
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi+1)
    grad = np.zeros_like(seq)
    grad[idx] = 2 * seq[k - idx]
    # shift to satisfy sum constraint
    grad -= grad.mean()
    new_seq = seq - step * grad
    # Clip small negatives after projection? We'll project.
    new_seq = _project_simplex(new_seq)
    return new_seq, conv.max()
```

```
def _power_exp_sequence(n, alpha, offset, lam):
    idx = np.arange(1, n+1, dtype=float) + offset
    a = idx ** (-alpha) * np.exp(-lam * (idx-1))
    s = a.sum()
    if s == 0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a
```

The submission implements a dedicated optimiser for the “step‑function” problem.
The algorithm is built around three ideas:

1. **Fast FFT based convolution** – all convolution operations are performed
   with a real‑FFT, i.e. `O(n log n)`.  This makes it possible to run
   many thousands of iterations inside the 1000 s budget.

2. **Projected sub‑gradient descent on the simplex** – the objective for a
   normalised vector `a` (`∑a = 1`) is  
   `F(a) = 2·n·max_k (a*a)[k]`.  
   The sub‑gradient of `max_k (a*a)[k]` is the sum of the gradients of all
   entries that are (almost) maximal.  The optimiser therefore
   * (a) computes the self‑convolution,  
   * (b) selects the indices whose value is within a tiny tolerance of the
        global maximum,  
   * (c) builds the exact sub‑gradient for those indices, and  
   * (d) takes a step and projects back onto the probability simplex
        `{x≥0, Σx=1}`.

   Using only the near‑maximal entries (instead of a fixed “top k” set) focuses the
   optimisation on the part of the objective that actually matters and yields a
   considerably faster reduction of the bound.

3. **Multiple restarts & polishing** – after an initial power‑law seed a long
   optimisation run is performed.  Afterwards a few short runs start from
   random power‑law seeds; the best result is kept and finally polished with a
   very small step size.

All helper functions are defined at module level (no closures) as required.
The code respects the 1000 s time limit, prints progress, and finally returns
the best sequence it discovered.

```python
"""
Fast projected sub‑gradient optimiser for the step‑function bound.

The algorithm works as follows:

1.  Initialise a candidate distribution on the simplex, mainly using a
    power‑law family a_i ∝ (i+1+δ)^‑α.

2.  Repeatedly minimise 2·n·max(conv(a,a)) by projected sub‑gradient steps.
    Only the indices whose convolution value is (almost) maximal are used to
    build the sub‑gradient; after each step the vector is projected onto the
    probability simplex.

3.  Perform several short random restarts and keep the best sequence.
    Finally run a long polishing phase with a very small step size.

All convolutions are computed with a real FFT (O(n log n)), which allows
working with n≈18000 comfortably inside the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Power‑law sequence a_i ∝ (i+1+offset)^{-α} normalised to sum = 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """
    Self‑convolution of ``seq`` using a real FFT.
    Returns the full length‑(2n‑1) vector.
    """
    n = seq.shape[0]
    size = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    # Clip tiny negative artefacts caused by rounding
    conv[conv < 0.0] = 0.0
    return conv


def _score_fft(seq: np.ndarray) -> float:
    """
    Exact objective value for a normalised sequence (∑seq = 1):
        2·n·max(conv(seq,seq))
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the O(n) algorithm of
    Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  Gradient step that focuses on the (near) maximal convolution entries
# ----------------------------------------------------------------------


def _gradient_step_nearmax(seq: np.ndarray, step: float) -> tuple[np.ndarray, float]:
    """
    One projected sub‑gradient step that reduces the largest values of the
    self‑convolution.

    Parameters
    ----------
    seq : np.ndarray
        Current probability vector (sum = 1, non‑negative).
    step : float
        Step size.

    Returns
    -------
    (np.ndarray, float)
        Updated probability vector (still on the simplex) and the current
        maximal convolution value.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)

    # indices whose convolution value is (almost) maximal
    max_val = conv.max()
    # absolute tolerance – small enough to capture ties caused by rounding
    eps = max_val * 1e-12 + 1e-15
    ks = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(seq)
    for k in ks:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        # contribution of a_{k-i}
        grad[i_range] += seq[k - i_range]

    # true sub‑gradient of max(conv) is 2·a_{k‑i}
    grad *= 2.0

    # make the step tangent to the affine hyperplane Σx = 1
    grad -= grad.mean()

    new_seq = seq - step * grad
    new_seq = _project_simplex(new_seq)
    return new_seq, max_val


def _run_gradient(seq: np.ndarray,
                  time_budget: float,
                  *,
                  step0: float = 0.5,
                  decay: float = 0.9999,
                  swap_interval: int = 500,
                  max_iters: int = 10 ** 9) -> tuple[np.ndarray, float]:
    """
    Run projected sub‑gradient descent for the given time budget.

    Returns the best sequence encountered and its corresponding score.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = _score_fft(seq)

    step = step0
    iteration = 0
    last_improve = 0
    start = time.time()

    while iteration < max_iters and (time.time() - start) < time_budget:
        iteration += 1

        seq, cur_max = _gradient_step_nearmax(seq, step)
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            last_improve = iteration
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # occasional random mass exchange to help escape plateaus
        if iteration % swap_interval == 0:
            i, j = np.random.choice(seq.size, 2, replace=False)
            # move a small fraction of the mass from i to j
            eps = min(step * seq[i], seq[i] * 0.3)
            if eps > 0.0:
                seq[i] -= eps
                seq[j] += eps
                seq = _project_simplex(seq)

        step = max(step * decay, 1e-9)

    return best_seq, best_score

def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,

    """

    The search consists of:

    * a power‑law initialisation (or reuse of ``height_sequence_1`` if present),
    * a long projected gradient phase with a sub‑gradient built from the

    * a final polishing run.

    The function returns before the allotted time expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = 18000

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_seq = None

                    raise ValueError
                cand /= s

                elif cand.size > n:
                    cand = cand[:n]
                best_seq = cand.copy()
                best_score = _score_fft(best_seq)
                print("[init] start from height_sequence_1")
        except Exception:
            best_seq = None

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_seq is None:

            score = _score_fft(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_alpha = alpha

        # fine refinement around the best exponent
        if best_seq is not None:

                                      best_alpha + 0.02, 81)
            for alpha in fine_alphas:
                if time.time() >= deadline:
                    break
                seq = _power_sequence(n, alpha)
                score = _score_fft(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_alpha = alpha

    # ------------------------------------------------------------------
    # 4. First (long) gradient descent phase on the current best candidate
    # ------------------------------------------------------------------
    if best_seq is not None:

            time_budget=remaining,
            step0=0.5,
            decay=0.99995,
            swap_interval=400,
        )

    # ------------------------------------------------------------------
    # 5. Random restarts – short gradient runs from random power‑law seeds
    # ------------------------------------------------------------------
    while time.time() < deadline - 30.0:   # keep ~30 s for a final polish
        alpha = np.random.uniform(0.35, 0.75)
        offset = np.random.uniform(0.0, 5.0)
        seq = _power_sequence(n, alpha, offset)

        # add a tiny Gaussian jitter to diversify the landscape
        seq += np.random.randn(n) * 1e-5
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()

        remaining = max(0.0, deadline - time.time())
        seq, score = _run_gradient(
            seq,
            time_budget=remaining,
            step0=0.4,
            decay=0.9999,
            swap_interval=300,
            max_iters=3000,          # short run
        )
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[restart] improved to {best_score:.6f} (α={alpha:.5f})")
        # early break if we have already reached the target constant
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 6. Final polishing – many iterations with a very small step size
    # ------------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    best_seq, best_score = _run_gradient(
        best_seq,
        time_budget=remaining,
        step0=0.2,
        decay=0.99998,
        swap_interval=200,
    )

    # Ensure exact feasibility
    best_seq = _project_simplex(best_seq)

    print(f"[final] best_score={best_score:.6f}, length={n}")
    return best_seq.tolist()
```
