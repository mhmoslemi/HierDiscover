        if w_sum == 0.0:

"""
Propose a good step‑function sequence for ``evaluate_sequence``.

The algorithm works in three phases:

1. **Initialisation** – start from a length‑``n`` power‑law
   ``a_i ∝ (i+1)^{-0.5}`` (the known near‑optimal shape).  If the harness
   provides a global ``height_sequence_1`` it is used instead.
2. **Entropic projected gradient descent** – minimise a smoothed version of
   ``max(conv)`` where ``conv = a * a``.  The smoothed objective is the
   log‑sum‑exp of the convolution entries; its gradient is
   ``∇f = 2·conv(a, w_rev)`` with ``w`` the soft‑max weights.
   An exponentiated‑gradient (mirror‑descent) update keeps the iterates on
   the probability simplex without an explicit Euclidean projection.
3. **Exploration** – every few hundred iterations a tiny random mass‑swap
   is performed to escape shallow plateaus.  The best sequence (lowest
   exact score) seen during the run is stored and finally returned.

All operations are performed with NumPy and FFT‑based convolutions,
so the routine easily handles sequences of several × 10⁴ elements well
within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using an FFT of size a power of two."""
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    return np.fft.irfft(Z, n=N)[:size]


def _power_sequence(n: int, alpha: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """Return ``a_i ∝ (i+1+offset)^{-alpha}`` normalised to sum‑1."""
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds as a safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be provided by the external harness
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.size
        a = np.maximum(init, 0.0)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            raise ValueError
        print("[init] using supplied height_sequence_1")
    except Exception:
        # fallback: a power‑law with exponent 0.5 – the classic near‑optimal shape
        n = 12000                      # a moderate size that still runs fast
        a = _power_sequence(n, alpha=0.5)
        print("[init] using power‑law start, n =", n)

    # make sure we are on the simplex
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    # keep the best sequence seen so far
    conv = _conv_fft(a, a)
    best_score = float(2 * n * conv.max())
    best_seq = a.copy()
    print(f"[status] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Entropic gradient descent --------------------------------------
    # ------------------------------------------------------------------
    lr = 0.35                # initial step size for the exponentiated update
    lr_decay = 0.9999
    lr_min = 1e-5

    sigma = 0.45             # smoothing factor (as a fraction of max(conv))
    sigma_decay = 0.99995
    sigma_min = 0.05         # never let epsilon become too tiny

    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # (a) current convolution and its maximum
        conv = _conv_fft(a, a)
        max_conv = conv.max()

        # (b) smoothing parameter ε = σ·max_conv
        sigma = max(sigma * sigma_decay, sigma_min)
        eps = max(sigma * max_conv, 1e-12)

        # (c) soft‑max weights w ∝ exp(conv / ε)
        expo = conv / eps
        # clip to avoid overflow in the exponential
        expo = np.clip(expo, a_min=None, a_max=50.0)
        # stable soft‑max
        shift = np.max(expo)
        w = np.exp(expo - shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # (d) gradient of the smoothed objective:
        #     ∇f = 2 * conv(a, w_rev)
        w_rev = w[::-1]
        v_full = _conv_fft(a, w_rev)           # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)       # slice that yields the gradient
        grad = 2.0 * v_full[idx]

        # (e) exponentiated‑gradient (mirror‑descent) update
        #     a_i ← a_i * exp(‑η * grad_i)  then renormalise
        mult = np.exp(-lr * grad)
        # guard against numerical extremes
        mult = np.clip(mult, 1e-12, 1e12)
        a = a * mult
        a = a / a.sum()                         # this stays on the simplex

        # (f) decay the learning rate
        lr = max(lr * lr_decay, lr_min)

        # (g) occasional tiny random mass exchange – helps to leave plateaus
        if iteration % 200 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            # move a modest fraction of the mass from i to j
            delta = min(a[i] * 0.3, a[i] * lr)
            if delta > 0.0:
                a[i] -= delta
                a[j] += delta
                a = a / a.sum()

        # (h) evaluate the exact (non‑smoothed) objective; keep the best
        conv_now = _conv_fft(a, a)
        cur_score = float(2 * n * conv_now.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] iter={iteration:6d}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Return best sequence discovered
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f} after {iteration} iterations")
    return best_seq.tolist()
