def generate_seeds(n, prev_seq=None):
    seeds = []
    # geometric families
    for c in np.linspace(0.5, 5.0, 15):
        r = 1.0 - c / n
        if r > 0:
            seeds.append(_geom(n, r))
    # power law
    for p in np.linspace(0.0, 3.0, 15):
        seeds.append(_power(n, p))
    # geom*power
    for c in np.linspace(0.5, 4.0, 10):
        r = 1.0 - c/n
        if r <= 0: continue
        for p in np.linspace(0.0, 2.5, 10):
            seeds.append(_geom_power(n, r, p))
    # sym geometric
    for c in np.linspace(0.5, 4.0, 10):
        r = 1.0 - c/n
        if r > 0:
            seeds.append(_sym_geom(n, r))
    # beta distribution shapes
    for a_ in [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]:
        for b_ in [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]:
            seeds.append(_beta_shape(n, a_, b_))
    # sine window
    seeds.append(_sine_window(n))
    # exponential shape
    for c in np.linspace(0.1, 5.0, 10):
        seeds.append(_exp_shape(n, c))
    # Dirichlet samples
    for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
        a = np.random.gamma(alpha, 1.0, size=n)
        a_sum = a.sum()
        if a_sum > 0:
            seeds.append(a / a_sum)
    # random softmax seeds
    for _ in range(5):
        v_rand = np.random.randn(n)
        seeds.append(_softmax(v_rand))
    # previous best?
    if prev_seq is not None:
        seeds.append(prev_seq)
    return seeds

def _beta_shape(n, alpha, beta):

    v = np.log(a0 + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    best_seq = a0.copy()
    best_val = eval_fast(best_seq)
    beta = start_beta
    it = 0
    while True:
        it += 1
        if time.time() > deadline:
            break
        # compute a from logits
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine] n={n:5d} it={it:6d} best={cur_val:.6f}")
        # increase beta
        beta *= beta_mul
        shift = cur_max
        # compute weights
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        # gradient wrt a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        # pick central n entries
        grad_a = 2.0 * full[n-1 : n-1 + n]
        # gradient wrt logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)
        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)
    return best_seq, best_val

"""
Propose a non‑negative step function (probability vector) with a low
self‑convolution peak.  The method builds a diverse set of analytical
seeds (geometric, power‑law, beta‑distribution, exponential, sine‑window,
random Dirichlet draws, etc.), evaluates them quickly with an FFT‑based
convolution, keeps the best few and refines them with an Adam optimiser
on a log‑softmax parametrisation.  The loss is a smoothed
log‑sum‑exp approximation of the maximum autocorrelation; the smoothing
parameter β is increased geometrically so the optimiser eventually
targets the true maximum.  After the gradient phase a short stochastic
hill‑climbing phase moves a small amount of probability mass between
positions to escape residual local minima.  The routine respects a
global time budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using FFT (real inputs)."""
    L = x.shape[0] + y.shape[0] - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Parametric seed families
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _sym_exp_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.minimum(idx, n - 1 - idx)
    a = np.exp(-c * dist)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete approximation of a Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _exp_shape(n: int, c: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-c * idx)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Fast objective (scale‑invariant)
# ----------------------------------------------------------------------


def _fast_objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) (sum(seq) is assumed = 1)."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Gradient‑based refinement (smooth max)
# ----------------------------------------------------------------------


def _refine_sequence(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 8.0,
    beta_mul: float = 1.02,
    lr0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.
    Returns the best sequence found (probability vector) and its fast objective.
    """
    a0 = a0 / a0.sum()
    v = np.log(a0 + 1e-30)          # logits
    m = np.zeros_like(v)           # first moment
    v2 = np.zeros_like(v)          # second moment
    best_seq = a0.copy()
    best_val = _fast_objective(best_seq)

    beta = start_beta
    it = 0

    while True:
        if time.time() > deadline:
            break
        it += 1

        # current probability vector
        a = _softmax(v)

        # compute convolution and current objective
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # increase smoothing parameter
        beta *= beta_mul
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient w.r.t. a (smooth loss)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]         # central n entries

        # gradient w.r.t. logits (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (optional final polish)
# ----------------------------------------------------------------------


def _hill_climb(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Move tiny probability packets between random coordinates and keep improvements."""
    n = seq.shape[0]
    cur_seq = seq.copy()
    cur_val = _fast_objective(cur_seq)
    while time.time() < deadline:
        eps = 1e-5  # absolute amount to relocate
        i = np.random.randint(n)
        if cur_seq[i] < eps:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        cand = cur_seq.copy()
        cand[i] -= eps
        cand[j] += eps
        # re‑normalise (numerical drift)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum
        val = _fast_objective(cand)
        if val < cur_val - 1e-9:
            cur_val = val
            cur_seq = cand
            print(f"[hill] improved to {val:.6f}")
    return cur_seq, cur_val

def _generate_seeds(n: int, prev_seq: np.ndarray | None = None) -> list[np.ndarray]:
    """Return a list of diverse candidate probability vectors of length n."""
    seeds: list[np.ndarray] = []

    # 1) geometric families (r = 1 - c/n)
    for c in np.linspace(0.5, 5.0, 13):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # 2) power‑law families
    for p in np.linspace(0.0, 3.0, 13):
        seeds.append(_power(n, p))

    # 3) combined geometric × power
    for c in np.linspace(0.5, 4.0, 8):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.5, 6):
            seeds.append(_geom_power(n, r, p))

    # 4) symmetric geometric
    for c in np.linspace(0.5, 4.0, 8):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # 5) symmetric exponential
    for c in np.linspace(0.1, 5.0, 8):
        seeds.append(_sym_exp_shape(n, c))

    # 6) beta‑distribution shapes
    for a_ in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for b_ in [0.2, 0.5, 1.0, 2.0, 5.0]:
            seeds.append(_beta_shape(n, a_, b_))

    # 7) mixed beta mixtures (few random draws)
    for _ in range(5):
        a1, b1 = np.random.uniform(0.2, 5.0, size=2)
        a2, b2 = np.random.uniform(0.2, 5.0, size=2)
        w = np.random.rand()
        mix = w * _beta_shape(n, a1, b1) + (1.0 - w) * _beta_shape(n, a2, b2)
        mix = np.maximum(mix, 0.0)
        mix /= mix.sum()
        seeds.append(mix)

    # 8) sine window (smooth baseline)
    seeds.append(_sine_window(n))

    # 9) exponential decay from one side
    for c in np.linspace(0.1, 5.0, 8):
        seeds.append(_exp_shape(n, c))

    # 10) Dirichlet draws (different concentrations)
    for alpha in [0.2, 0.5, 0.8, 2.0, 5.0]:
        a = np.random.gamma(alpha, 1.0, size=n)
        s = a.sum()
        if s > 0:
            seeds.append(a / s)

    # 11) random soft‑max seeds
    for _ in range(5):
        v_rand = np.random.randn(n)
        seeds.append(_softmax(v_rand))

    # 12) previous best (or up‑sampled version)
    if prev_seq is not None:
        if prev_seq.shape[0] == n:
            seeds.append(prev_seq.copy())
        else:
            # resample via linear interpolation
            old_n = prev_seq.shape[0]
            x_old = np.arange(old_n, dtype=np.float64)
            x_new = np.linspace(0.0, old_n - 1.0, n)
            seq_up = np.interp(x_new, x_old, prev_seq)
            seq_up = np.maximum(seq_up, 0.0)
            s = seq_up.sum()
            if s > 0:
                seq_up /= s
                seeds.append(seq_up)

    return seeds

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence minimising ``evaluate_sequence``.
    The routine respects a time budget (default 1000 s) and returns the best
    sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # small safety margin

    # ------------------------------------------------------------------
    # Load a previously discovered construction if it exists.
    # ------------------------------------------------------------------
    prev_global: np.ndarray | None = None
    try:
        prev = globals().get("height_sequence_1")
        if prev is not None:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0:
                arr = np.maximum(arr, 0.0)
                s = arr.sum()
                if s > 0:
                    prev_global = arr / s
    except Exception:
        prev_global = None

    global_best_seq: np.ndarray | None = None
    global_best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Length schedule – try larger sizes if time permits.
    # ------------------------------------------------------------------
    lengths = [8000, 12000, 16000]  # descending order of priority
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"[main] Optimising length = {n}")

        # ------------------------------------------------------------------
        # Build seed set (including up‑sampled previous best)
        # ------------------------------------------------------------------
        seeds = _generate_seeds(n, prev_seq=prev_global)

        # ------------------------------------------------------------------
        # Quick scan over seeds
        # ------------------------------------------------------------------
        best_local_seq: np.ndarray | None = None
        best_local_val: float = float("inf")
        for idx, a in enumerate(seeds):
            if time.time() > deadline:
                break
            # normalise (protect against zero sum)
            s = a.sum()
            if s <= 0.0:
                continue
            a = a / s
            val = _fast_objective(a)
            if val < best_local_val:
                best_local_val = val
                best_local_seq = a.copy()
            if val < global_best_val:
                global_best_val = val
                global_best_seq = a.copy()
                print(f"[scan] new global best from seed {idx} = {val:.6f}")

        if best_local_seq is None:
            # fallback: uniform distribution
            best_local_seq = np.full(n, 1.0 / n, dtype=np.float64)
            best_local_val = _fast_objective(best_local_seq)

        # ------------------------------------------------------------------
        # Refine the current best seed for this length
        # ------------------------------------------------------------------
        refined_seq, refined_val = _refine_sequence(
            best_local_seq,
            n,
            deadline,
            start_beta=8.0,
            beta_mul=1.02,
            lr0=0.9,
        )
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[overall] new global best after refinement (len={n}) = {refined_val:.6f}")

        # ------------------------------------------------------------------
        # Perturbed restarts (few quick re‑optimisations from noisy logits)
        # ------------------------------------------------------------------
        while time.time() < deadline:
            noisy = np.log(global_best_seq + 1e-30) + 0.02 * np.random.randn(n)
            pert_seq = _softmax(noisy)
            seq2, val2 = _refine_sequence(
                pert_seq,
                n,
                deadline,
                start_beta=8.0,
                beta_mul=1.02,
                lr0=0.9,
            )
            if val2 < global_best_val:
                global_best_val = val2
                global_best_seq = seq2.copy()
                print(f"[perturb] improved global best to {val2:.6f}")
            # perform a short hill‑climb after each perturbation
            seq3, val3 = _hill_climb(global_best_seq, deadline)
            if val3 < global_best_val:
                global_best_val = val3
                global_best_seq = seq3.copy()
                print(f"[perturb‑hill] improved global best to {val3:.6f}")
            # only a few restarts – break if we are already low on time
            if time.time() > deadline - 10.0:
                break

        # Keep the refined sequence as the new previous best for the next length
        prev_global = global_best_seq.copy()

    # ------------------------------------------------------------------
    # Final polishing (hill climbing) on the overall best sequence
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        seq_final, val_final = _hill_climb(global_best_seq, deadline)
        if val_final < global_best_val:
            global_best_val = val_final
            global_best_seq = seq_final
            print(f"[final‑hill] improved to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Sanity check & return
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # Very unlikely – fall back to a uniform vector of size 8000
        n = 8000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_val = _fast_objective(global_best_seq)

    # Ensure non‑negativity and unit sum (evaluation is scale‑invariant)
    global_best_seq = np.maximum(global_best_seq, 0.0)
    total = global_best_seq.sum()
    if total > 0.0:
        global_best_seq /= total

    print(f"[final] best objective (fast) = {global_best_val:.6f}")
    return global_best_seq.tolist()
