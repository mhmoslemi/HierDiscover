def hill_climb(a, max_steps, time_limit):
    k = int(a.sum())
    current = a.copy()
    current_score = compute_score(current)
    best_local = current_score
    best_seq_local = current.copy()
    for step in range(max_steps):
        if time.time() >= time_limit:
            break
        # propose swap: pick one index from ones, one from zeros
        ones = np.flatnonzero(current)
        zeros = np.flatnonzero(1-current)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = current.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_score = compute_score(cand)
        # accept if better
        if cand_score < current_score:
            current = cand
            current_score = cand_score
            if cand_score < best_local:
                best_local = cand_score
                best_seq_local = cand.copy()
        else:
            # maybe accept with probability
            if np.random.rand() < 0.001: # small probability
                current = cand
                current_score = cand_score
        # optionally break early if no improvement after many steps.
    return best_seq_local, best_local

def hill_climb(a, current_best_score, remaining_time):

    k = int(a.sum())
    best_local = compute_score(a)
    best_seq_local = a.copy()
    # schedule
    iteration = 0
    temp_start = 1.0
    temp_end = 0.001
    start = time.time()
    while time.time() - start < remaining_time and iteration < 20000:
        # pick random swap
        ones = np.flatnonzero(a)
        zeros = np.flatnonzero(1 - a)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = a.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_score = compute_score(cand)
        # temperature schedule linear
        elapsed = time.time() - start
        progress = elapsed / remaining_time
        temp = temp_start + progress * (temp_end - temp_start)
        accept = False
        if cand_score < best_local:
            accept = True
        else:
            if temp > 1e-12:
                prob = math.exp(-(cand_score - best_local) / temp)
                if np.random.rand() < prob:
                    accept = True
        if accept:
            a = cand
            if cand_score < best_local:
                best_local = cand_score
                best_seq_local = cand.copy()
        iteration += 1
    return best_seq_local, best_local

"""
Propose a non‑negative step function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The idea is to treat the vector as a binary (0/1) sequence.
For a binary vector the objective reduces to

    score = 2·n·M / k²

where k = number of 1‑entries and M is the maximum entry of the
ordered convolution a*a (the number of ordered pairs (i,j) with i+j = t).
Thus we can focus on *choosing the positions of the 1‑entries* such that
the maximal number of pairwise sums is small.

Algorithm
---------
1. Greedy construction.
   * Choose a target cardinality k.
   * Build the set S of positions one by one.
   * For each new element we try a few random candidates and pick the one
     that yields the smallest possible new maximum of the convolution.
   * The convolution is stored as an integer array `cnt[t] = |{(i,j)∈S×S:
     i+j=t}|`.  Updating it after adding a new index needs only O(|S|)
     operations.

2. Optional local improvement.
   * Starting from the greedy solution we perform a limited number of
     random swaps (remove a 1, insert a 0).  The convolution counts are
     updated in O(k) and a swap is accepted when it strictly reduces the
     current maximum M.

3. Re‑repeat the above for several values of k (80 … 500) and several
   restarts while the time budget (≈ 1000 s) permits.  Keep the best
   sequence found.

The routine respects the time limit, prints intermediate scores and
finally returns the best binary vector as a plain Python list of floats.
"""

import time
import math
import random
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a via FFT (real input)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def _score_binary(conv_counts: np.ndarray, k: int, n: int) -> float:
    """Exact objective for a binary vector (k ones)."""
    if k == 0:
        return np.inf
    max_m = int(conv_counts.max())
    return 2.0 * n * max_m / (k * k)


def _greedy_binary_set(n: int, k: int, attempts: int = 30) -> tuple[np.ndarray, np.ndarray]:
    """
    Build a binary vector of length n with exactly k ones.
    The set is built greedily: at each step a few random unused indices
    are examined and the one that yields the smallest possible new
    maximum of the ordered convolution is chosen.
    Returns (vector, convolution_counts) where convolution_counts[t]
    holds the number of ordered pairs (i,j)∈S×S with i+j=t.
    """
    conv = np.zeros(2 * n - 1, dtype=np.int32)
    chosen = np.zeros(n, dtype=bool)          # True ⇔ position already selected
    S = []                                     # list of selected indices

    for step in range(k):
        cur_max = int(conv.max())
        # random candidates among the still free indices
        free = np.where(~chosen)[0]
        if free.size == 0:
            break
        if free.size <= attempts:
            cand = free
        else:
            cand = np.random.choice(free, size=attempts, replace=False)

        S_arr = np.array(S, dtype=np.int32)   # may be empty
        best_i = -1
        best_max = None

        for i in cand:
            # candidate maximum after inserting i
            cand_max = cur_max
            if S_arr.size > 0:
                sums = i + S_arr                     # vectorised
                vals = conv[sums] + 2                # each unordered pair adds 2
                m = int(vals.max())
                if m > cand_max:
                    cand_max = m
            # self‑pair i+i contributes one more
            self_val = int(conv[2 * i] + 1)
            if self_val > cand_max:
                cand_max = self_val

            if (best_max is None) or (cand_max < best_max):
                best_max = cand_max
                best_i = int(i)

        # update structures with the chosen index
        if best_i < 0:      # should never happen
            break
        if S_arr.size > 0:
            sums = best_i + S_arr
            conv[sums] += 2
        conv[2 * best_i] += 1
        chosen[best_i] = True
        S.append(best_i)

    a = np.zeros(n, dtype=np.float64)
    a[S] = 1.0
    return a, conv


def _swap_improve(a: np.ndarray,
                  conv: np.ndarray,
                  max_iters: int,
                  deadline: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Simple local search: repeatedly try to swap a ‘1’ with a ‘0’.
    The convolution counts are updated in O(k) per swap.
    Only swaps that strictly reduce the current maximum are kept.
    """
    n = a.shape[0]
    k = int(a.sum())
    if k == 0:
        return a, conv

    # maintain lists of positions for fast random access
    ones = list(np.flatnonzero(a).astype(np.int32))
    zeros = list(np.flatnonzero(1 - a).astype(np.int32))
    cur_max = int(conv.max())
    cur_score = 2.0 * n * cur_max / (k * k)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # pick a random 1 and a random 0
        i = random.choice(ones)
        j = random.choice(zeros)

        # copy the convolution counts – size ≤ 4000, cheap
        new_conv = conv.copy()

        # remove contributions coming from i
        for p in ones:
            s = i + p
            if i == p:
                new_conv[s] -= 1
            else:
                new_conv[s] -= 2

        # add contributions of j with the *remaining* ones (i has been removed)
        for p in ones:
            if p == i:
                continue
            s = j + p
            new_conv[s] += 2
        new_conv[2 * j] += 1

        new_max = int(new_conv.max())
        new_score = 2.0 * n * new_max / (k * k)

        if new_score < cur_score:
            # accept the swap
            cur_score = new_score
            cur_max = new_max
            a[i] = 0.0
            a[j] = 1.0
            conv = new_conv
            # update the index containers
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)

    return a, conv


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function of length ``n`` (default 2000) that
    minimises 2·n·max(conv(a,a)) / (sum(a)**2).

    The search consists of many greedy constructions for various
    cardinalities k combined with a short local‑swap refinement.
    The best binary vector found before the time limit is returned.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = int(kwargs.get("n", 2000))

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) optional use of a provided reference construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            # evaluate using the same formula we use for binary vectors
            # (convert to binary because that is the cheapest form)
            # but keep the original values – they are a valid candidate.
            tmp = compute_score(ref) if 'compute_score' in globals() else np.inf
            if tmp < best_score:
                best_score = tmp
                best_seq = ref.copy()
                print(f"[init] reference sequence score = {tmp:.6f}")

    # ------------------------------------------------------------------
    # 2) main search loop – greedy + optional swap improvement
    # ------------------------------------------------------------------
    # candidate cardinalities (number of ones).  The range is chosen
    # empirically – larger k gives a stronger denominator, but the max
    # of the convolution grows slowly.
    k_values = list(range(80, 501, 20))

    # limit the total number of greedy constructions to stay well within
    # the budget; we stop when the deadline is near.
    greedy_runs = 0
    while time.time() < deadline:
        # pick a target cardinality – bias towards larger values
        k = int(np.random.choice(k_values))
        # ---- Greedy construction ------------------------------------
        a, conv = _greedy_binary_set(n, k, attempts=30)
        score = _score_binary(conv, k, n)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[greedy] k={k:3d}  max={int(conv.max()):3d}  score={score:.6f}  time={time.time()-start_time:.1f}s")

        # ---- Optional local swap refinement -------------------------
        # Give a small time budget to the refinement; stop early if we
        # are close to the overall deadline.
        if time.time() + 0.5 < deadline:
            a_ref, conv_ref = _swap_improve(a, conv,
                                            max_iters=5000,
                                            deadline=deadline)
            k_ref = int(a_ref.sum())
            score_ref = _score_binary(conv_ref, k_ref, n)
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[swap]   k={k_ref:3d}  max={int(conv_ref.max()):3d}  score={score_ref:.6f}")

        greedy_runs += 1
        # occasional progress report
        if greedy_runs % 20 == 0:
            print(f"[progress] runs={greedy_runs}  best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) final reporting and return
    # ------------------------------------------------------------------
    total_time = time.time() - start_time
    print(f"[done] performed {greedy_runs} greedy runs in {total_time:.1f}s")
    print(f"[best] final score = {best_score:.6f}")

    # Ensure the returned object is a plain Python list of floats.
    return best_seq.tolist()
