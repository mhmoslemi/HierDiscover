"""
Propose a step‑function by:

1. **Analytic families** –  a quick sweep over several parametrised
   non‑negative shapes (exponential, double‑exponential, a mixture with a
   uniform component and a cosine‐type edge‑heavy shape).  Each candidate
   is normalised to the required ℓ₁‑norm √(2·n) and its exact objective
   (2·n·max(conv)/sum²) is evaluated with an FFT‑based convolution.
   The best candidate of this phase is kept as a warm start.

2. **Smoothed gradient descent** –  the non‑smooth “max” in the objective
   is replaced by a soft‑max with a temperature p.  For a given p the
   gradient of the smoothed objective can be written analytically as a
   convolution of the current sequence with the soft‑max weights.  The
   update is performed with a decreasing step size and a projection onto
   the simplex (ℓ₁‑norm = √(2·n) and non‑negativity).  The temperature p
   is increased from 5 to 30 during the run to gradually focus the
   optimisation on the true maximum.  The best sequence encountered is
   returned.

Both phases respect the global time budget (≈ 1000 s) and constantly
print progress information.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex { w ≥ 0 , ∑ w = s }."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be equal
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (for a*a convolution)."""
    return 1 << ((2 * n - 1).bit_length())


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Return the maximum entry of the (linear) self‑convolution a*a.
    Uses an FFT of length fft_len (must be ≥ 2·len(a)‑1).
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return float(np.max(conv))


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, fft_len: int) -> float:
    """Exact objective 2·n·max(conv)/(sum(a)²) (no clipping needed)."""
    n = a.shape[0]
    max_conv = _max_convolution(a, fft_len)
    s = np.sum(a)
    # sum is guaranteed >0 because of construction
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine first scans a few low‑dimensional families (exponential,
    double‑exponential, uniform+exponential, cosine) and then refines the
    best candidate with a smoothed gradient descent on the max‑convolution.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # a tiny safety margin

    # ------------------------------------------------------------------
    # Problem size ------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 5000                     # generous default; still fast FFT‑wise
    target_sum = math.sqrt(2.0 * n)  # required ℓ₁‑norm

    # Pre‑compute FFT lengths that will be reused
    fft_len_conv = _fft_len_for_n(n)                     # for a*a
    fft_len_grad = 1 << ((3 * n - 2).bit_length())       # for gradient conv

    # ------------------------------------------------------------------
    # Phase 1 – cheap parametric search ----------------------------------
    best_seq = None
    best_val = float("inf")

    def update_best(a: np.ndarray) -> None:
        nonlocal best_seq, best_val
        val = _objective(a, fft_len_conv)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[PH1] new best obj = {best_val:.6f}")

    # uniform baseline
    a_uni = np.ones(n, dtype=float)
    a_uni = a_uni / a_uni.sum() * target_sum
    update_best(a_uni)

    # exponential family a_i = r**i
    r_vals = np.linspace(0.90, 0.9999, 300)
    # add a fine neighbourhood around the asymptotic optimum r≈1-2/n
    r_vals = np.concatenate([r_vals,
                             np.linspace(1 - 4.0 / n, 1 - 0.5 / n, 200)])
    for r in r_vals:
        a = np.power(r, np.arange(n, dtype=float))
        a = a / a.sum() * target_sum
        update_best(a)

    # double‑exponential (edges heavy)
    for r in np.linspace(0.90, 0.9999, 150):
        a = np.power(r, np.arange(n, dtype=float))
        a = a + a[::-1]
        a = a / a.sum() * target_sum
        update_best(a)

    # mixture: (1‑γ)*uniform + γ*exponential
    for gamma in (0.05, 0.10, 0.20, 0.35, 0.50):
        for r in np.linspace(0.90, 0.9999, 120):
            a = (1.0 - gamma) * np.ones(n) + gamma * np.power(r, np.arange(n, dtype=float))
            a = a / a.sum() * target_sum
            update_best(a)

    # cosine edge‑heavy shape: a_i = 1 + c*cos(2π i/(n‑1))
    for c in (0.2, 0.4, 0.6, 0.8):
        a = 1.0 + c * np.cos(2.0 * np.pi * np.arange(n, dtype=float) / (n - 1))
        a = np.maximum(a, 0.0)            # enforce non‑negativity
        a = a / a.sum() * target_sum
        update_best(a)

    # ------------------------------------------------------------------
    # Phase 2 – smoothed gradient descent on the exact objective -------
    a = best_seq.copy()
    best_obj = best_val
    best_seq = a.copy()

    # hyper‑parameters
    p_start, p_end = 5.0, 30.0                # soft‑max temperature schedule
    lr0 = target_sum * 0.08                  # initial step size (≈ 8 % of total mass)
    it = 0

    while time.time() < deadline:
        it += 1
        # Full convolution a*a (needed for the soft‑max weights)
        fa = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(fa * fa, n=fft_len_conv)[: 2 * n - 1]

        # soft‑max (smooth approximation of max)
        maxc = np.max(conv_full)
        # linear temperature increase
        elapsed = time.time() - start_time
        frac = min(elapsed / budget_s, 1.0)
        p = p_start + (p_end - p_start) * frac
        exps = np.exp(p * (conv_full - maxc))
        w = exps / np.sum(exps)               # shape (2·n‑1,)

        # gradient of the smoothed objective: 2·a ⊛ w_rev
        w_rev = w[::-1]
        af = np.fft.rfft(a, n=fft_len_grad)
        wf = np.fft.rfft(w_rev, n=fft_len_grad)
        grad_full = np.fft.irfft(af * wf, n=fft_len_grad)
        grad = 2.0 * grad_full[:n]

        # diminishing step size
        step = lr0 / math.sqrt(it)

        # gradient step + projection onto the simplex (ℓ₁‑norm = target_sum)
        a_candidate = a - step * grad
        a_candidate = _simplex_projection(a_candidate, target_sum)

        # evaluate candidate (cheap via FFT)
        cur_val = _objective(a_candidate, fft_len_conv)

        # greedy acceptance – always move to the candidate (helps escape plateaus)
        a = a_candidate

        # keep global best
        if cur_val < best_obj:
            best_obj = cur_val
            best_seq = a_candidate.copy()
            print(f"[GRAD {it:6d}] new best obj = {best_obj:.6f}")

        # occasional tiny random shuffle to break possible stagnation
        if it % 2000 == 0:
            i = np.random.randint(n)
            if a[i] > 1e-12:
                j = np.random.randint(n)
                if i != j:
                    delta = a[i] * 0.05 * np.random.rand()
                    tmp = a.copy()
                    tmp[i] -= delta
                    tmp[j] += delta
                    tmp = _simplex_projection(tmp, target_sum)
                    tmp_val = _objective(tmp, fft_len_conv)
                    if tmp_val < best_obj:
                        best_obj = tmp_val
                        best_seq = tmp.copy()
                        print(f"[SHUF {it:6d}] new best obj = {best_obj:.6f}")
                    a = tmp

    print(f"[DONE] final best obj = {best_obj:.6f} after {it} gradient steps")
    return best_seq.tolist()
