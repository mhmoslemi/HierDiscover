"""
Propose a high‑quality step‑function sequence.

Algorithm
---------
1. **Cantor‐measure initialization** – Build a histogram of the
   middle‑third Cantor measure on the interval [0,1] with
   `2**L` equally weighted atoms (L≈14).  After binning into `n` bins the
   vector is already a feasible (non‑negative, sum‑to‑1) candidate.
2. **Smooth‐max gradient descent** – Repeatedly minimise a
   differentiable approximation of `max(a*a)` using the soft‑max
   weighting `w ∝ exp((b‑max(b))/τ)`.  The gradient has the closed form
   `‑2·(a_rev ∗ w)` and is computed efficiently with FFTs.
   A Nesterov‐style momentum term accelerates convergence.
3. **Projection onto the simplex** – After each step the vector is
   clipped to non‑negative values and re‑normalised to total mass 1.
4. **Best‑ever tracking** – The lowest objective value seen
   (according to `evaluate_sequence`) is saved and finally returned.
   The whole search respects the 1000 s budget.

The Cantor initialization yields a distribution whose self‑convolution
is substantially flatter than that of a uniform sequence, providing a
strong starting point for the subsequent stochastic optimisation.
"""

import time
import numpy as np

# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT. ``size`` must be ≥ len(u)+len(v)‑1.
    Returns the full convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a (full length)."""
    return _conv_fft(a, a, size)


def _cantor_histogram(n: int, L: int = 14) -> np.ndarray:
    """
    Produce a length‑`n` histogram approximating the Cantor (1/3) measure.
    `L` determines the number of atoms (2**L).  The result sums to 1.
    """
    N = 1 << L                      # number of atoms
    # Powers of 3 used for the ternary digits
    powers = 3.0 ** np.arange(1, L + 1)          # (L,)
    # Bit matrix: shape (N, L), each row is the binary expansion of the integer
    idx = np.arange(N, dtype=np.uint32)[:, None]   # (N,1)
    bits = ((idx >> np.arange(L, dtype=np.uint32)) & 1).astype(np.float64)  # (N, L)
    # Position of each atom in [0,1)
    pos = (bits * (2.0 / powers)).sum(axis=1)      # (N,)
    # Histogram on the required grid
    hist, _ = np.histogram(pos, bins=n, range=(0.0, 1.0))
    a = hist.astype(np.float64) / float(N)        # normalise → sum == 1
    return a


def _smooth_gradient_step(
    a: np.ndarray,
    a_rev: np.ndarray,
    conv_full: np.ndarray,
    size2: int,
    step: float,
    tau: float,
) -> np.ndarray:
    """
    Perform one projected gradient descent step for the smooth max
    of the convolution `a*a`.  Uses the soft‑max weighting.
    """
    max_b = conv_full.max()
    # Soft‑max weights (shifted by max_b for stability)
    w = np.exp((conv_full - max_b) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum

    # Gradient = -2·(a_rev ∗ w)  → compute via FFT
    g_full = _conv_fft(a_rev, w, size2)
    n = a.shape[0]
    g = g_full[n - 1 : n - 1 + n]      # central n entries

    d = -2.0 * g                       # descent direction
    # Project to hyperplane Σd = 0 (keeps total mass unchanged)
    d -= d.mean()

    a_new = a + step * d
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= s
    return a_new
