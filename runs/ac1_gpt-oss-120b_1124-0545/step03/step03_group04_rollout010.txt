def propose_candidate(seed=..., budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    best_score = float('inf')
    best_seq = None

    # Try height_sequence_1 if exists
    try:
        init_seq = np.array(height_sequence_1, dtype=float)
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.sum() == 0:
            raise ValueError
        init_seq = init_seq / init_seq.sum()
        score = compute_score(init_seq)
        best_score = score
        best_seq = init_seq.copy()
        print(f"[init] using height_sequence_1: n={len(init_seq)}, score={score:.6f}")
    except Exception:
        pass

    # Parameter grid
    n_vals = [4000, 5000, 8000, 10000, 15000, 20000]
    alphas = np.linspace(0.42, 0.58, 17)  # to explore around 0.5
    offsets = np.linspace(0.0, 5.0, 21)

    for n in n_vals:
        if time.time() > deadline:
            break
        for alpha in alphas:
            if time.time() > deadline:
                break
            for offset in offsets:
                if time.time() > deadline:
                    break
                a = _power_sequence(n, alpha=alpha, offset=offset)
                # Already normalized to sum 1
                score = compute_score(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[grid] n={n}, alpha={alpha:.4f}, offset={offset:.2f}, score={score:.6f}")

    # Now refine using subgradient descent
    if best_seq is None:
        # fallback simple uniform maybe
        n = 5000
        best_seq = np.full(n, 1.0/n)
        best_score = compute_score(best_seq)

    a = best_seq.copy()
    n = a.shape[0]
    lr = 0.2
    lr_decay = 0.9999
    lr_min = 5e-5
    iteration = 0
    # We'll run until deadline

"""
Search for a non‑negative step‑function (a probability vector) that 
minimises the evaluate_sequence objective.

Strategy
--------
1. **Initial candidate pool** – generate sequences from two parametric families:
   * Power‑law: a_i ∝ (i+offset)^(-α)
   * Beta‑law:   a_i ∝ x_i^{a-1} (1−x_i)^{b-1},  x_i = (i+0.5)/n
   The parameters are sampled on a coarse grid and later randomly.
   Every candidate is normalised to sum = 1 and its exact score
   `2·n·max(conv)` (which is the value returned by evaluate_sequence when the
   sum is 1) is recorded.

2. **Refinement by sub‑gradient descent** – for the best candidate found so far
   we minimise the L∞‑norm of the self‑convolution.
   The sub‑gradient of `max(a*a)` at the index where the maximum occurs is
   `g_i = 2·a_{k−i}` (zero if the index is out of range).  After each step
   the vector is projected onto the probability simplex (Euclidean projection
   from Duchi et al., ICML 2008).  A simple learning‑rate schedule together with
   occasional Gaussian jitter helps to escape shallow plateaus.

3. **Time management** – the whole routine respects the supplied `budget_s`
   (default 1000 s).  The first 35 % of the budget is spent on the coarse
   search; the remainder is used for the iterative refinement.

The returned list is ready for `evaluate_sequence`.  All numbers are
non‑negative and the sum is strictly positive, so the evaluation never
hits the “invalid” branches.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries would become negative – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using FFT (real inputs)."""
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())  # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+offset)^{-α}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _beta_sequence(n: int, a: float = 0.5, b: float = 0.5) -> np.ndarray:
    """
    Discrete Beta(a,b) density on {0,…,n‑1}.
    x_i = (i+0.5)/n  (mid‑point rule) and
    a_i ∝ x_i^{a-1} (1-x_i)^{b-1}.
    """
    x = (np.arange(n) + 0.5) / n
    # compute the raw weights, guard against 0^neg
    with np.errstate(divide='ignore', invalid='ignore'):
        raw = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    raw = np.where(np.isfinite(raw), raw, 0.0)
    total = raw.sum()
    if total == 0.0:
        raw[:] = 1.0 / n
    else:
        raw /= total
    return raw


def _score_of_sequence(a: np.ndarray) -> float:
    """
    Exact score used by evaluate_sequence when the vector sums to 1:
    2 * n * max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * conv.max())
