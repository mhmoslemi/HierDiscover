def _conv_fft_two(x, y):
    size = _next_pow2(len(x) + len(y) - 1)
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: len(x)+len(y)-1]
    conv[conv < 0] = 0.0
    return conv

def _run_adam(seq, time_budget, max_iters=10**9, step0=0.5, beta_start=1.0, beta_factor=1.02, beta_max=100.0, eps=1e-8, beta1=0.9, beta2=0.999, debug=False):

    start = time.time()
    best_seq=seq.copy()
    best_score=_score_fft(seq)

    m=np.zeros_like(seq)
    v=np.zeros_like(seq)
    t=0
    beta=beta_start

    while t<max_iters and (time.time()-start)<time_budget:
        t+=1
        # compute conv and objective
        conv = _conv_fft(seq)
        # compute softmax weights
        # To avoid overflow: w = np.exp(beta * (conv - conv.max()))
        # adapt scaling
        maxc = conv.max()
        # shift
        exp_vals = np.exp(beta * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        w_rev = w[::-1]
        grad_full = _conv_fft_two(seq, w_rev)  # length 3n-2
        # compute grad_i = 2 * grad_full[(2n-2) - i] for i in range(n)
        idx_start = (2*len(seq)-2)
        grad = 2.0 * grad_full[idx_start - np.arange(len(seq))]
        # subtract mean to stay in sum=1 hyperplane.
        grad -= grad.mean()
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        seq_new = seq - step0 * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        seq = _project_simplex(seq_new)
        # evaluate
        cur_score = _score_fft(seq)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[adam improve] iter={t}, score={best_score:.6f}")

        # update beta (sharpness)
        beta = min(beta * beta_factor, beta_max)

    return best_seq, best_score

"""
Propose a non‑negative step‑function (probability vector) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Power‑law seeding** – a grid of power‑law sequences
   a_i ∝ (i+1+δ)^{-α} (α≈0.45‑0.60) is examined and the best
   candidate (Σa=1) is kept.

2. **Projected Adam on a smooth max** – the objective  max_k (a*a)_k
   is approximated by a soft‑max:
       φ_β(a)= (1/β)·log Σ_k exp(β·(a*a)_k).
   Its gradient is a weighted cross‑correlation which can be computed
   with two FFT‑based convolutions.  An Adam step is taken and the
   iterate is projected onto the probability simplex.  The “temperature”
   β is increased slowly (annealing) so the optimisation ends at the
   true max.

3. **Random restarts** – the whole Adam phase is repeated from a few
   random (α,δ) seeds with a short budget; the best overall sequence is
   retained.

4. **Final polishing** – a short projected sub‑gradient descent
   (max‑only) cleans up any remaining spikes.

All heavy work uses FFT (O(n log n)); the routine adapts to a 1000 s
budget and returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the O(n log n) algorithm of
    Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Produce a normalised non‑negative sequence proportional to
    (i+1+offset)^{-α} for i=0,…,n‑1.

    Parameters
    ----------
    n : int
        Length of the sequence.
    alpha : float
        Exponent (α>0).
    offset : float, optional
        Small shift that can be used to reduce the first coefficient.

    Returns
    -------
    np.ndarray
        Length‑`n` array with sum equal to 1.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Self‑convolution (linear) using real FFT. Returns length (2n‑1)."""
    n = seq.shape[0]
    size = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    conv[conv < 0.0] = 0.0          # eliminate tiny negatives
    return conv


def _conv_fft_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two arbitrary 1‑D arrays using FFT."""
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: len(a) + len(b) - 1]
    conv[conv < 0.0] = 0.0
    return conv


def _score_fft(seq: np.ndarray) -> float:
    """
    Exact value of the objective used by ``evaluate_sequence`` when
    Σseq == 1, i.e. 2·n·max(conv(seq,seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())

# Optimisation primitives
# ------------------------------------------------------------


def _run_adam(
    seq: np.ndarray,
    time_budget: float,
    *,
    max_iters: int = 10 ** 9,
    step0: float = 0.1,
    beta_start: float = 0.5,
    beta_factor: float = 1.02,
    beta_max: float = 30.0,
    adam_beta1: float = 0.9,
    adam_beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the smooth approximation of the max
    convolution.  Returns the best sequence seen together with its
    (exact) score.
    """
    start = time.time()
    n = seq.shape[0]

    best_seq = seq.copy()
    best_score = _score_fft(seq)

    m = np.zeros_like(seq)   # 1st moment
    v = np.zeros_like(seq)   # 2nd moment
    b1t = 1.0
    b2t = 1.0
    beta_cur = beta_start
    step = step0

    iteration = 0
    while iteration < max_iters and (time.time() - start) < time_budget:
        iteration += 1

        # 1) self‑convolution (used both for the loss and for weights)
        conv = _conv_fft(seq)                # length 2n‑1

        # 2) soft‑max weights (smooth max)
        maxc = conv.max()
        # shift for numerical stability
        exp_vals = np.exp(beta_cur * (conv - maxc))
        w = exp_vals / exp_vals.sum()        # sum to 1

        # 3) gradient = 2·cross‑correlation(seq, w)
        w_rev = w[::-1]
        grad_full = _conv_fft_two(seq, w_rev)          # length 3n‑2
        idx_start = 2 * n - 2
        grad = 2.0 * grad_full[idx_start - np.arange(n)]

        # remove component orthogonal to the simplex (keep Σgrad = 0)
        grad -= grad.mean()

        # 4) Adam update (projected)
        m = adam_beta1 * m + (1.0 - adam_beta1) * grad
        v = adam_beta2 * v + (1.0 - adam_beta2) * (grad * grad)
        b1t *= adam_beta1
        b2t *= adam_beta2
        m_hat = m / (1.0 - b1t)
        v_hat = v / (1.0 - b2t)

        seq_new = seq - step * m_hat / (np.sqrt(v_hat) + eps)
        seq = _project_simplex(seq_new)

        # 5) evaluate exact objective
        cur_score = _score_fft(seq)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[adam] iter={iteration}, score={best_score:.6f}")

        # 6) anneal β and (very gently) decay step size
        beta_cur = min(beta_cur * beta_factor, beta_max)
        step *= 0.99995

    return best_seq, best_score


def _run_gradient(
    seq: np.ndarray,
    time_budget: float,
    *,
    max_iters: int = 10 ** 9,
    top_k: int = 20,
    step0: float = 0.4,
    decay: float = 0.9999,
) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent that uses only the `top_k`
    largest entries of the self‑convolution to form the sub‑gradient.
    This is used only for a short final polishing stage.
    """
    start = time.time()
    best_seq = seq.copy()
    best_score = _score_fft(seq)

    step = step0
    iteration = 0
    n = seq.shape[0]

    while iteration < max_iters and (time.time() - start) < time_budget:
        iteration += 1

        conv = _conv_fft(seq)

        if top_k >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            top_idx = np.argpartition(-conv, top_k - 1)[:top_k]

        # sub‑gradient w.r.t the max‑entries (same formula as earlier)
        grad = np.zeros_like(seq)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1, dtype=int)
            grad[i_range] += seq[k - i_range]
        grad *= 2.0
        grad -= grad.mean()

        seq = seq - step * grad
        seq = _project_simplex(seq)

        # occasional random swap to avoid plateaus
        if iteration % 200 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps_swap = min(step * seq[i], seq[i] * 0.3)
            seq[i] -= eps_swap
            seq[j] += eps_swap
            seq = _project_simplex(seq)

        cur_score = _score_fft(seq)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[grad] iter={iteration}, score={best_score:.6f}")

        step = max(step * decay, 1e-8)

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:

    time budget (seconds) and returns a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for the harness's final call
    deadline = start_time + budget_s - 5.0

    # Choose a length – 20 000 gives a good trade‑off between
    # resolution and speed for the 1000 s budget.
    # --------------------------------------------------------
    n = 20000

    # --------------------------------------------------------
    best_seq = None
    best_score = np.inf
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=float)
            if cand.size > 0 and np.isfinite(cand).all():
                cand = np.maximum(cand, 0.0)
                if cand.sum() == 0.0:
                    raise ValueError
                cand /= cand.sum()
                if cand.size < n:
                    cand = np.concatenate([cand, np.zeros(n - cand.size)])
                elif cand.size > n:
                    cand = cand[:n]
                best_seq = cand.copy()
                best_score = _score_fft(best_seq)
                print("[init] using existing height_sequence_1")
        except Exception:
            best_seq = None

    # 1. Coarse + fine grid of power‑law seeds (if no better start)

    if best_seq is None:

        for alpha in coarse_alphas:
            if time.time() >= deadline:
                break
            seq = _power_sequence(n, alpha)
            score = _score_fft(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_alpha = alpha
                print(f"[coarse] α={alpha:.5f}, score={score:.6f}")

        if best_seq is not None:
            fine_alphas = np.linspace(best_alpha - 0.02, best_alpha + 0.02, 81)
            for alpha in fine_alphas:
                if time.time() >= deadline:
                    break
                seq = _power_sequence(n, alpha)
                score = _score_fft(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_alpha = alpha
                    print(f"[fine]   α={alpha:.5f}, score={score:.6f}")

    # --------------------------------------------------------
    # 2. Main Adam optimisation phase (most of the budget)
    # --------------------------------------------------------
    if best_seq is not None:
        remaining = max(0.0, deadline - time.time())
        best_seq, best_score = _run_adam(
            best_seq,
            time_budget=remaining,
            max_iters=10 ** 9,
            step0=0.12,
            beta_start=0.5,
            beta_factor=1.015,
            beta_max=32.0,
        )
        print(f"[adam‑phase‑1] best_score={best_score:.6f}")

    # --------------------------------------------------------
    # 3. Random restarts – short Adam runs from random seeds
    # --------------------------------------------------------
    while time.time() < deadline - 30.0:      # keep ~30 s for final polish
        alpha = np.random.uniform(0.35, 0.75)
        offset = np.random.uniform(0.0, 5.0)
        seq = _power_sequence(n, alpha, offset)

        # tiny Gaussian jitter to diversify the landscape
        seq += np.random.randn(n) * 1e-5
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()

        remaining = max(0.0, deadline - time.time())
        seq, score = _run_adam(
            seq,
            time_budget=remaining,
            max_iters=3000,
            step0=0.08,
            beta_start=0.5,
            beta_factor=1.02,
            beta_max=28.0,
        )
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[restart] improved to {best_score:.6f} (α={alpha:.5f})")
        if best_score <= 1.503:
            break

    # --------------------------------------------------------
    # 4. Final polishing – a few thousand sub‑gradient steps
    # --------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    if remaining > 0.0:
        best_seq, best_score = _run_gradient(
            best_seq,
            time_budget=remaining,
            max_iters=10 ** 9,
            top_k=15,
            step0=0.3,
            decay=0.99995,
        )
        print(f"[final‑grad] best_score={best_score:.6f}")

    # --------------------------------------------------------
    # 5. Clean‑up and return
    # --------------------------------------------------------
    best_seq = _project_simplex(best_seq)   # exact feasibility
    print(f"[result] final_score={best_score:.6f}, length={n}")
    return best_seq.tolist()
