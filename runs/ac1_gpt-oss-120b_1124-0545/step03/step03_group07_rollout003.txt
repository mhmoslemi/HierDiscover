"""
Gradient‑descent + smooth‑max + final simulated annealing.

We minimise   max_k (a * a)[k]   under the ℓ₁‑norm constraint
∑ a_i = √(2 n)  and a_i ≥ 0.  The infinity‑norm is smoothed by a
soft‑max (log‑sum‑exp) with a temperature β that is increased during the
run.  Adam is used for the descent step, after each update the vector is
projected onto the simplex (the feasible ℓ₁‑ball).  When the gradient
phase stalls we restart from a fresh random point.  The best vector
found is then refined by a classic simulated‑annealing (mass‑transfer)
phase.

The whole routine respects a user supplied time budget (default 1000 s)
and returns the best list of non‑negative heights found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , ∑ w = s }."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_len(min_len: int) -> int:
    """Smallest power‑of‑two ≥ min_len."""
    return 1 << ((min_len - 1).bit_length())


# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """
    Return the full convolution x*y using FFT of length L.
    Length of the result is len(x)+len(y)-1 (the tail is trimmed).
    """
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)
    return conv[: x.size + y.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Compute max(self‑convolution) and its index for a real vector a.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    idx = int(np.argmax(conv))
    val = float(conv[idx])
    return val, idx


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray,
                         beta: float,
                         fft_len_self: int,
                         fft_len_w: int) -> np.ndarray:
    """
    Compute the gradient of the smooth max
        f(a) = (1/β)·log ∑_k exp(β·(a*a)[k])
    with respect to a.
    """
    # 1) self‑convolution (full length 2n‑1)
    fa = np.fft.rfft(a, n=fft_len_self)
    conv_full = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * a.size - 1]

    # 2) soft‑max weights (stabilised)
    max_c = np.max(conv_full)
    exp_term = np.exp(beta * (conv_full - max_c))
    w = exp_term / np.sum(exp_term)          # size 2n‑1, sum to 1

    # 3) gradient = 2 · (w ⋆ a)   (⋆ = cross‑correlation without reversal)
    #    The cross‑correlation of w and a is the valid part of
    #    convolution(w[::-1], a).
    w_rev = w[::-1]
    conv_w_a = np.fft.irfft(
        np.fft.rfft(w_rev, n=fft_len_w) *
        np.fft.rfft(a, n=fft_len_w),
        n=fft_len_w
    )
    # valid part starts at index len(w)-1 and spans n entries
    start = (2 * a.size - 1) - 1          # len(w)-1 = 2n-2
    grad = 2.0 * conv_w_a[start:start + a.size]
    return grad

    deadline = start_time + budget_s
    # allocate 65 % of the budget to the smooth‑max descent
    descent_deadline = start_time + budget_s * 0.65

    # --------------------- Adam hyper‑parameters ----------------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = target_sum * 0.03                # initial learning‑rate
    # smooth‑max temperature schedule
    beta_start = 0.3
    beta_end = 25.0

    # --------------------- bookkeeping ---------------------------------
    best_seq = a.copy()
    best_val, _ = _max_convolution(a, fft_len_self)
    print(f"[INIT] n={n}, target_sum={target_sum:.3f}, init max(conv)={best_val:.6f}")

    # --------------------- gradient descent with restarts -------------
    it_total = 0
    restart_counter = 0
    while time.time() < descent_deadline:
        # random restart if we have been stuck for >3000 iterations
        if it_total == 0 or (it_total % 3000 == 0 and it_total > 0):
            restart_counter += 1
            a = np.random.exponential(scale=1.0, size=n)
            a = a / a.sum() * target_sum
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            # reset the smooth‑max temperature a little later
            beta_cur = beta_start
            print(f"[RESTART {restart_counter}] new random start")

        # increase smooth‑max temperature gradually (linear in elapsed descent time)
        elapsed = time.time() - start_time
        frac = elapsed / (descent_deadline - start_time + 1e-12)
        beta_cur = beta_start + frac * (beta_end - beta_start)

        # ---- gradient of smooth max ---------------------------------
        grad = _smooth_max_gradient(a, beta_cur, fft_len_self, fft_len_w)

        # ---- Adam update --------------------------------------------
        it_total += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it_total)
        v_hat = v / (1.0 - beta2 ** it_total)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        # projection on the simplex keeps the ℓ₁‑norm exactly target_sum
        a_candidate = _simplex_projection(a_candidate, target_sum)

        a = a_candidate

        # ---- evaluate current max(conv) ------------------------------
        cur_val, _ = _max_convolution(a, fft_len_self)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[{it_total:6d}] new best max(conv)={best_val:.6f}")

        # occasional tiny random perturbation to jump out of flat plateaus
        if it_total % 2500 == 0:
            i_src = np.random.randint(n)
            if a[i_src] > 1e-12:
                i_dst = np.random.randint(n)
                delta = a[i_src] * 0.05 * np.random.rand()
                a_pert = a.copy()
                a_pert[i_src] -= delta
                a_pert[i_dst] += delta
                a_pert = _simplex_projection(a_pert, target_sum)
                pert_val, _ = _max_convolution(a_pert, fft_len_self)
                if pert_val < best_val:
                    best_val = pert_val
                    best_seq = a_pert.copy()
                    print(f"[{it_total:6d}] perturbation improved best to {best_val:.6f}")
                a = a_pert

    # --------------------- simulated annealing phase -----------------
    # Start from the best sequence discovered so far
    a = best_seq.copy()
    cur_val, _ = _max_convolution(a, fft_len_self)
    best_val = cur_val
    best_seq = a.copy()
    print(f"[SA-START] max(conv)={best_val:.6f}")

    # Temperature schedule (exponential cooling)
    T_start = best_val * 0.25
    T_end = best_val * 1e-6
    sa_start = time.time()
    sa_deadline = deadline
    step_counter = 0

    while time.time() < sa_deadline:
        fraction = (time.time() - sa_start) / (sa_deadline - sa_start + 1e-12)
        T = T_start * ((T_end / T_start) ** fraction)

        # propose a mass transfer
        i = np.random.randint(n)
        if a[i] <= 1e-13:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = a[i] * 0.07 * np.random.rand()   # move up to 7 % of source
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta

        # fast evaluation (full FFT, still cheap for n≈2500)
        cand_val, _ = _max_convolution(a_cand, fft_len_self)

        # Metropolis rule
        if cand_val < cur_val:
            a = a_cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = a_cand.copy()
                print(f"[SA {step_counter:6d}] new best max(conv)={best_val:.6f}")
        else:
            prob = math.exp(-(cand_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = a_cand
                cur_val = cand_val

        step_counter += 1

    # --------------------- final report --------------------------------
    print(f"[DONE] best max(conv) = {best_val:.6f} "
          f"after {it_total} Adam steps, {step_counter} SA steps, "
          f"{restart_counter} restarts")
    # The evaluator clamps negatives and caps at 1000, but we keep a clean.
    return best_seq.tolist()
```
