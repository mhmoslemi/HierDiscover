def project_simplex(v):
    """Project v onto the probability simplex (sum=1, v>=0)."""
    # from https://github.com/indump/pytorch-simplex-projection
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # Find rho: last index where u_i > cssv_i / i
    rho = np.where(u - cssv/ind > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‐function that (empirically) improves the bound
E(a)=2·n·max(conv(a,a))/(sum(a)**2).

Algorithm
---------

1.  Initialise a symmetric Beta‑type shape
        a_i ∝ (x_i·(1−x_i))^α ,      x_i=(i+0.5)/n ,
    where α is searched on a small grid.  This family contains the
    uniform distribution (α=0), the classic arcsine shape (α=−0.5) and many
    intermediate profiles.  The best α on the grid is kept as a starting
    point.

2.  Refine the chosen vector with a **projected sub‑gradient descent**.
    For the current a we compute the convolution with itself,
    locate the index k* where the maximum is attained and use the exact
    sub‑gradient of the quantity max(conv) (which is 2·a_{k*−i} for the
    i‑th component, with a correction when i=k*/2).  After a step we
    project back onto the probability simplex (non‑negative, sum=1).

    The step size follows η_t = η₀/√t.  When progress stalls for a while a
    small random jitter is added (restart) to escape possible plateaus.

3.  Track the best vector seen (exact score recomputed with the true
    max‑convolution).  The routine respects a user supplied time budget
    (default 1000 s) and returns the best sequence found before the
    deadline.

The implementation uses only NumPy and the standard library, runs in
O(n log n) per iteration thanks to FFT‑based convolution, and works for
any reasonable length (default n=2000).  It can be easily adapted to
larger n – the evaluation function itself will automatically cap very
large entries.
"""

import time

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, v>=0)."""
    # Reference: Duchi et al. (2008) – Efficient Projections onto the L1‑Ball
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray, fft_len: int) -> tuple[np.ndarray, float, int]:
    """
    Compute a*a convolution using FFT and return the full vector,
    its maximum value and the index where the maximum occurs.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * len(a) - 1]
    k = int(np.argmax(conv))
    return conv, float(conv[k]), k


def _subgradient_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv(a,a)) w.r.t. a.
    The convolution is defined as np.convolve(a, a) (linear, not circular).
    """
    n = a.shape[0]
    i = np.arange(n)
    j = k_max - i
    mask = (j >= 0) & (j < n)

    g = np.zeros_like(a)
    # regular ordered‑pair contribution (double count)
    g[mask] = 2.0 * a[j[mask]]

    # correction for the diagonal term i=j (counted only once)
    if k_max % 2 == 0:
        i_eq = k_max // 2
        if 0 <= i_eq < n:
            g[i_eq] = a[i_eq]          # replace 2*a[i_eq] by a[i_eq]
    return g


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta‑type profile:
        a_i ∝ (x_i·(1−x_i))^α ,   x_i = (i+0.5)/n .
    The result is normalised to sum to 1.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # avoid 0**alpha for α<0 – x is never exactly 0 or 1
    a = (x * (1.0 - x)) ** alpha
    # ensure non‑negativity (numerical noise)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.ones_like(a) / n
    else:
        a /= a.sum()
    return a


def _initial_vector(n: int, seed: int = 42) -> np.ndarray:
    """

        2) a short grid search over Beta‑shapes (α in a modest range);
        3) fall back to the classic arcsine shape.
    """
    rng = np.random.default_rng(seed)

    # 1) reuse previous construction if it matches the required size
    if "height_sequence_1" in globals():
        cand = np.array(globals()["height_sequence_1"], dtype=np.float64)
        if cand.shape[0] == n:
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0:

    # 2) quick grid search over α ∈ [-0.6, 0.8] (step 0.05)
    best_a = None

    fft_len = 1 << ((2 * n - 1).bit_length())

            best_a = a_vec

    # 3) fallback to arcsine (α = -0.5)
    if best_a is None:

    return best_a

def propose_candidate(seed: int = 42,

    1000 s) and returns the best sequence found.

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # 0) initialise a candidate (grid‑search over a simple family)

    a = _initial_vector(n, seed=seed)       # sum(a) == 1
    best_a = a.copy()
    fft_len = 1 << ((2 * n - 1).bit_length())

    _, best_max_conv, _ = _conv_max(best_a, fft_len)
    best_score = 2.0 * n * best_max_conv

    # --------------------------------------------------------------
    eta0 = 0.5               # initial step‑size factor
    it = 0
    last_improve = start_time

        # progress – used only for occasional logging
        elapsed = time.time() - start_time

        cur_score = 2.0 * n * max_conv

        eta = eta0 / np.sqrt(it)

                last_improve = time.time()
                print(f"[{it:07d}] restart after stagnation (t={elapsed:.1f}s)")

    # --------------------------------------------------------------
    # 2) final sanity – ensure non‑negative finite values
    # --------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    # The evaluation routine caps values >1000, but we stay well below.
    total_time = time.time() - start_time
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_a.tolist()
