"""
Heuristic multi‑restart optimizer for the step‑function objective.

The search proceeds as follows:

1. A quick grid over two analytic families (product / sum of power‑law
   tails) is used to obtain a strong baseline.
2. Several random Dirichlet initialisations are generated.
3. For every start vector we run two optimisation phases:
   * a large‑scale L2‑loss minimisation that flattens the full
     convolution toward a constant target (target = 1/(2n−1));
   * a smoothed‑max gradient descent (log‑sum‑exp) that directly
     reduces the maximal convolution entry.
   Both phases use FFT‑based convolutions and project onto the
   probability simplex after each step.
4. The best sequence discovered before the 1000 s budget expires is
   returned.

The implementation is pure NumPy, uses no closures or lambdas and
conforms to the required `propose_candidate` signature.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions – all top level
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant) for a non‑negative vector a.
    The vector is assumed to lie on the simplex (sum = 1); the function
    normalises it for safety.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def gradient_smooth(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)
    with respect to a (assuming Σ a_i = 1).  The gradient is
        g_i = 2· Σ_k w_k·a_{k-i},
    where w_k are the soft‑max weights.  Both convolutions are
    performed via FFT.
    """
    n = a.shape[0]
    conv = conv_fft(a)                     # length 2n‑1
    max_c = np.max(conv)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)        # length 2n‑1, sum = 1
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    start = n - 1
    end = 2 * n - 1
    slice_part = conv_wa[start:end][::-1]
    g = 2.0 * slice_part
    return g


def gradient_L2(a: np.ndarray) -> np.ndarray:
    """
    Gradient of the quadratic loss
        L(a) = Σ_k ( (a*a)_k - μ )²,
    where μ = 1/(2n−1) is the uniform target (the total mass of the
    convolution is 1).  The gradient is
        g_i = 2· Σ_k ( (a*a)_k - μ )·a_{k-i},
    and is computed via two FFT‑based convolutions.
    """
    n = a.shape[0]
    conv = conv_fft(a)                     # length 2n‑1
    mu = 1.0 / (2 * n - 1)
    w = conv - mu                          # length 2n‑1
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    start = n - 1
    end = 2 * n - 1
    slice_part = conv_wa[start:end][::-1]
    g = 2.0 * slice_part
    return g


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum=1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    mode = 'sum'      → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def random_dirichlet(n: int, alpha: float = 0.5) -> np.ndarray:
    """Draw a random point from Dirichlet(alpha,…,alpha) and return as array."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        x = np.ones_like(x)
    return x / x.sum()


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def optimise_L2(start: np.ndarray, max_steps: int,
                step_start: float, step_end: float,
                deadline: float) -> np.ndarray:
    """
    L2‑flattening phase.  Simple projected gradient descent with a linearly
    decreasing step size.
    """
    a = start.copy()
    n = a.shape[0]
    for it in range(max_steps):
        if time.time() >= deadline:
            break
        # linear schedule
        step = step_start + (step_end - step_start) * (it / max_steps)
        g = gradient_L2(a)
        # basic gradient step
        a = a - step * g
        a = np.clip(a, 0.0, None)
        a = project_simplex(a)
    return a


def optimise_smooth(start: np.ndarray, max_steps: int,
                    lam_start: float, lam_end: float,
                    step_start: float, step_end: float,
                    temp_start: float, temp_end: float,
                    deadline: float) -> np.ndarray:
    """
    Smoothed‑max phase (log‑sum‑exp).  Uses a Metropolis acceptance rule
    and occasional random mass‑transfer moves to avoid plateaus.
    """
    a = start.copy()
    n = a.shape[0]
    best_local = a.copy()
    best_local_score = compute_score(a)
    last_improve = time.time()
    for it in range(max_steps):
        if time.time() >= deadline:
            break
        prog = it / max_steps
        lam = lam_start + prog * (lam_end - lam_start)
        step = step_start + prog * (step_end - step_start)
        temp = temp_start + prog * (temp_end - temp_start)

        # main move: gradient step with probability 0.85, else random mass swap
        if np.random.rand() < 0.85:
            g = gradient_smooth(a, lam)
            cand = a - step * g
            cand = np.clip(cand, 0.0, None)
            cand = project_simplex(cand)
        else:
            cand = random_move(a, step * 0.5)

        cand_score = compute_score(cand)

        accept = False
        if cand_score < best_local_score:
            accept = True
        elif temp > 1e-12:
            prob = math.exp(-(cand_score - best_local_score) / temp)
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = cand
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local = cand.copy()
                last_improve = time.time()
        # occasional jitter when stuck
        if it % 20000 == 0:
            if time.time() - last_improve > 30.0:
                jitter = np.random.rand(n) * 1e-3
                a = best_local + jitter
                a = project_simplex(a)
                last_improve = time.time()
    return best_local

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

    The routine builds a short analytic grid, generates several random
    Dirichlet starts and runs a two‑phase optimisation (L2 flattening
    followed by smoothed‑max descent) for each start.  The best
    sequence discovered before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = int(kwargs.get("n", 2000))

    # --------------------------------------------------------------
    # 1) Analytic grid search – produces a decent starting point
    # --------------------------------------------------------------
    p_vals = np.arange(0.30, 0.95 + 1e-9, 0.05)
    offset_vals = np.arange(0.0, 1.6, 0.2)
    modes = ["product", "sum"]

    best_seq = None
    best_score = np.inf

    print("[init] analytic grid search (n={})".format(n))
    for mode in modes:
        for p in p_vals:
            for off in offset_vals:
                off_eff = off if (mode == "sum" or off > 0.0) else 1e-3
                cand = analytic_candidate(n, p, off_eff, mode)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                # occasional progress output
                if p == p_vals[0] and off == offset_vals[0] and mode == "product":
                    print(f"[grid] mode={mode}, p={p:.2f}, off={off:.2f} → {sc:.6f}")

    # If a reference sequence is present in the global namespace, try it
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            print(f"[init] global reference score = {sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = ref.copy()

    print(f"[init] best analytic start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Build a pool of starting points (analytic + random Dirichlet)
    # --------------------------------------------------------------
    start_pool = [best_seq.copy()]
    # add a few Dirichlet starts with different concentration parameters
    for alpha in [0.3, 0.5, 0.8, 1.2]:
        for _ in range(2):
            a0 = random_dirichlet(n, alpha)
            start_pool.append(a0)

    # --------------------------------------------------------------
    # 3) Optimise each start – two phases: L2 flattening → smooth max
    # --------------------------------------------------------------
    global_best_seq = best_seq.copy()
    global_best_score = best_score

    # Phase budgets (tuned to fit into the ~1000 s total budget)
    # Roughly 150 k L2 steps + 150 k smooth steps per start works well.
    L2_steps = 150_000
    smooth_steps = 150_000

    # Schedules for the smooth phase (similar to the original code but a
    # slightly wider range for λ to approach the hard max more aggressively)
    lam_start, lam_end = 0.5, 5000.0
    step_start, step_end = 0.2, 1e-5
    temp_start, temp_end = 1.0, 0.001

    start_idx = 0
    while time.time() < deadline and start_idx < len(start_pool):
        a0 = start_pool[start_idx]
        start_idx += 1

        # ---- L2 flattening -------------------------------------------------
        a1 = optimise_L2(a0, max_steps=L2_steps,
                         step_start=0.3, step_end=1e-5,
                         deadline=deadline)
        # ---- smoothed‑max refinement ---------------------------------------
        a2 = optimise_smooth(a1, max_steps=smooth_steps,
                             lam_start=lam_start, lam_end=lam_end,
                             step_start=step_start, step_end=step_end,
                             temp_start=temp_start, temp_end=temp_end,
                             deadline=deadline)

        final_score = compute_score(a2)
        if final_score < global_best_score:
            global_best_score = final_score
            global_best_seq = a2.copy()
            elapsed = time.time() - start_time
            print(f"[global] new best = {global_best_score:.6f} after {elapsed:.1f}s (start #{start_idx})")

    total_time = time.time() - start_time
    print(f"[done] best score={global_best_score:.6f}, time={total_time:.1f}s")
    # Return plain Python list as required by the harness
    return global_best_seq.tolist()

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    n = int(kwargs.get("n", 2000))

    # --------------------------------------------------------------
    # 1) Analytic grid search – strong baseline
    # --------------------------------------------------------------
    p_vals = np.arange(0.30, 0.95 + 1e-9, 0.05)
    offset_vals = np.arange(0.0, 1.6, 0.2)
    modes = ["product", "sum"]

    best_seq = None
    best_score = np.inf

    print(f"[init] analytic grid search (n={n})")
    for mode in modes:
        for p in p_vals:
            for off in offset_vals:
                off_eff = off if (mode == "sum" or off > 0.0) else 1e-3
                cand = analytic_candidate(n, p, off_eff, mode)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                # occasional progress output
                if p == p_vals[0] and off == offset_vals[0] and mode == "product":
                    print(f"[grid] mode={mode}, p={p:.2f}, off={off:.2f} → {sc:.6f}")

    # Optional global reference
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            print(f"[init] global reference score = {sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = ref.copy()

    print(f"[init] best analytic start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Build a pool of starting points
    # --------------------------------------------------------------
    start_pool = [best_seq.copy()]
    # add a few Dirichlet starts with various concentration parameters
    for alpha in [0.3, 0.5, 0.8, 1.2]:
        for _ in range(2):
            start_pool.append(random_dirichlet(n, alpha))

    # --------------------------------------------------------------
    # 3) Optimise each start – L2 flattening → smoothed‑max refinement
    # --------------------------------------------------------------
    global_best_seq = best_seq.copy()
    global_best_score = best_score

    # Phase budgets – tuned to stay within the 1000 s total time limit
    L2_steps = 150_000
    smooth_steps = 150_000

    # Schedules for the smoothed‑max phase (slightly more aggressive λ)
    lam_start, lam_end = 0.5, 5000.0
    step_start, step_end = 0.2, 1e-5
    temp_start, temp_end = 1.0, 0.001

    start_idx = 0
    while time.time() < deadline and start_idx < len(start_pool):
        a0 = start_pool[start_idx]
        start_idx += 1

        # ----- L2 flattening ------------------------------------------------
        a1 = optimise_L2(a0, max_steps=L2_steps,
                         step_start=0.3, step_end=1e-5,
                         deadline=deadline)

        # ----- Smoothed‑max refinement --------------------------------------
        a2 = optimise_smooth(a1, max_steps=smooth_steps,
                             lam_start=lam_start, lam_end=lam_end,
                             step_start=step_start, step_end=step_end,
                             temp_start=temp_start, temp_end=temp_end,
                             deadline=deadline)

        final_score = compute_score(a2)
        if final_score < global_best_score:
            global_best_score = final_score
            global_best_seq = a2.copy()
            elapsed = time.time() - start_time
            print(f"[global] new best = {global_best_score:.6f} after {elapsed:.1f}s "
                  f"(start #{start_idx})")

    total_time = time.time() - start_time
    print(f"[done] best score={global_best_score:.6f}, time={total_time:.1f}s")
    return global_best_seq.tolist()
