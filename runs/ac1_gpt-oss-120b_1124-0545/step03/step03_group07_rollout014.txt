def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    random.seed(seed)
    n = ... # either size of height_sequence_1 else 2500
    target_sum = sqrt(2*n)
    a = init shape

    # Precompute conv and max
    def compute_conv(a):
        fv = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fv * fv, n=fft_len)[:2*n-1]
        return conv

    conv = compute_conv(a)
    max_val = conv.max()
    best_seq = a.copy()
    best_val = max_val

    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # SA parameters
    T0 = max_val * 0.1   # initial temperature
    Tmin = max_val * 1e-6
    total_sa_time = deadline - start_time
    step = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        # exponential cooling based on elapsed fraction
        frac = elapsed / total_sa_time
        T = T0 * ((Tmin / T0) ** frac)

        # compute active k (max)
        k = int(np.argmax(conv))
        # compute contributions c_i for all i that valid:
        c = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1)
        # contributions:
        c[idx] = a[idx] * a[k - idx]

        # build probability distribution for i:
        probs = c[idx]
        if probs.sum() == 0:
            # fallback random
            i = np.random.randint(n)
        else:
            probs = probs / probs.sum()
            i = np.random.choice(idx, p=probs)

        # pick j uniformly among indices not equal i
        j = np.random.randint(n-1)
        if j >= i:
            j += 1

        # propose delta: fraction of a[i] maybe up to 0.1 * a[i]
        delta = a[i] * 0.1 * np.random.rand()
        if delta <= 0:
            continue
        # ensure a[i] - delta >= 0
        # create candidate
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta

        # compute conv for candidate
        conv_cand = compute_conv(a_cand)
        cand_val = conv_cand.max()

        # accept if improvement or with probability
        if cand_val < max_val:
            a = a_cand
            conv = conv_cand
            max_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = a_cand.copy()
                print(f"[SA {step}] new best {best_val:.6f} time {elapsed:.1f}s")
        else:
            # Metropolis
            prob = math.exp(-(cand_val - max_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = a_cand
                conv = conv_cand
                max_val = cand_val
        step += 1

    print(f"[DONE] best max conv = {best_val:.6f} after {step} steps")
    return best_seq.tolist()

"""
Targeted simulated annealing for minimizing max self‑convolution.
First scans a few parametric families (exponential, polynomial) to get a good start,
then performs a large number of mass‑transfer moves that preferentially take mass
away from indices that contribute most to the current maximum convolution and
deposit it at indices that contribute least.  The sum of the sequence is kept
fixed (by moving mass between two positions) and non‑negativity is enforced.
An exponential cooling schedule drives a classic Metropolis acceptance rule.
The best sequence found before the time budget expires is returned.
"""

import time, math, random, numpy as np
from typing import List

def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project onto the simplex { w >= 0, sum w = s }."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Return the maximum of the self‑convolution a*a."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())

def _conv_full(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2n‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence`` (i.e. the maximum entry of its self‑convolution
    after normalising the ℓ₁‑norm to √(2·n)).  The algorithm consists of
    a short parametric sweep for a good initial guess followed by a
    targeted simulated‑annealing phase that preferentially moves mass
    away from the indices that dominate the current maximum convolution.
    """
    np.random.seed(seed)
    random.seed(seed)

    # --------------------------------------------------------------
    # Determine the problem size and initialise a feasible sequence.
    # ----------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=float)
        n = base.size
    else:
        n = 2500
        base = np.random.exponential(scale=1.0, size=n)

    target_sum = math.sqrt(2.0 * n)

    # Normalise the base sequence to the required ℓ₁‑norm.
    a = base / base.sum() * target_sum

    # --------------------------------------------------------------
    # Helper objects for fast convolution.
    fft_len = 1 << ((2 * n - 1).bit_length())   # smallest power of two ≥ 2n‑1

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep to obtain a stronger starter.
    # ----------------------------------------------------------------
    best_a = a.copy()
    best_val = _conv_max(best_a, fft_len)

    # exponential families a_i = r^i
    for r in np.linspace(0.99, 0.9999, 30):
        a_tmp = (r ** np.arange(n, dtype=float))
        a_tmp = a_tmp / a_tmp.sum() * target_sum
        val = _conv_max(a_tmp, fft_len)
        if val < best_val:
            best_val = val
            best_a = a_tmp.copy()
            print(f"[INIT] exponential r={r:.5f} → max={val:.6f}")

    # polynomial families a_i = (i+1)^p   (monotonically increasing)
    for p in np.linspace(0.2, 4.0, 30):
        a_tmp = (np.arange(1, n + 1, dtype=float) ** p)
        a_tmp = a_tmp / a_tmp.sum() * target_sum
        val = _conv_max(a_tmp, fft_len)
        if val < best_val:
            best_val = val
            best_a = a_tmp.copy()
            print(f"[INIT] polynomial p={p:.2f} → max={val:.6f}")

    # also try the reversed polynomial (decreasing)
    for p in np.linspace(0.2, 4.0, 30):
        a_tmp = (np.arange(n, 0, -1, dtype=float) ** p)
        a_tmp = a_tmp / a_tmp.sum() * target_sum
        val = _conv_max(a_tmp, fft_len)
        if val < best_val:
            best_val = val
            best_a = a_tmp.copy()
            print(f"[INIT] rev‑poly p={p:.2f} → max={val:.6f}")

    # Keep the best of the sweep as the starting point.
    a = best_a.copy()
    cur_conv = _conv_full(a, fft_len)
    cur_max = float(cur_conv.max())
    best_seq = a.copy()
    best_val = cur_max
    print(f"[START] n={n}, initial max‑conv={cur_max:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – targeted simulated annealing.
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # 0.5 s safety margin
    total_time = budget_s
    # Temperature schedule: start relatively high, cool exponentially.
    T0 = cur_max * 0.15               # enough to accept occasional uphill steps
    Tmin = cur_max * 1e-6
    step_counter = 0
    last_improve_step = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        # Exponential cooling based on fraction of total time used.
        frac = elapsed / total_time
        T = T0 * ((Tmin / T0) ** frac)

        # Identify the index where the self‑convolution is maximal.
        k_max = int(np.argmax(cur_conv))

        # --- contributions to the active max index ---------------------------------
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        contrib = a[idx] * a[k_max - idx]         # element‑wise products that form the max

        # Choose a source index i with probability proportional to its contribution.
        if contrib.sum() > 0:
            probs = contrib / contrib.sum()
            i = np.random.choice(idx, p=probs)
        else:
            i = np.random.randint(n)

        # Choose a destination index j that contributes the *least* to the active max.
        # (If multiple minima exist we pick one uniformly at random.)
        # Compute a_j * a_{k_max‑j} for all j where the term is defined.
        j_candidates = np.arange(n, dtype=int)
        valid = (k_max - j_candidates >= 0) & (k_max - j_candidates < n)
        dest_contrib = np.zeros(n, dtype=float)
        dest_contrib[valid] = a[j_candidates[valid]] * a[k_max - j_candidates[valid]]
        # We want the smallest contribution; ties are broken randomly.
        min_val = dest_contrib.min()
        low_idxs = np.flatnonzero(dest_contrib == min_val)
        # ensure we do not pick the same index as the source.
        low_idxs = low_idxs[low_idxs != i] if low_idxs.size > 1 else low_idxs
        if low_idxs.size == 0:
            # fallback: any index different from i
            j = (i + np.random.randint(1, n)) % n
        else:
            j = np.random.choice(low_idxs)

        # Propose moving a fraction of the source mass to the destination.
        if a[i] <= 0:
            continue
        # delta is at most 20 % of the source mass, chosen uniformly.
        delta = a[i] * 0.20 * np.random.rand()
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta

        # Compute the maximum convolution of the candidate.
        cand_conv = _conv_full(a_cand, fft_len)
        cand_max = float(cand_conv.max())

        # Metropolis acceptance.
        if cand_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(cand_max - cur_max) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = a_cand
            cur_conv = cand_conv
            cur_max = cand_max
            if cand_max < best_val - 1e-9:
                best_val = cand_max
                best_seq = a.copy()
                last_improve_step = step_counter
                print(f"[SA {step_counter:07d}] new best max={best_val:.6f} time={elapsed:.1f}s")
        # --- periodic large perturbation to escape possible stagnation ----------
        step_counter += 1
        if step_counter % 200000 == 0:
            # perform a random reshuffle of a fraction of the mass
            mask = np.random.rand(n) < 0.05
            mass = a[mask].sum()
            if mass > 0:
                a[mask] = 0.0
                # redistribute the collected mass uniformly (project onto simplex)
                a = _simplex_projection(a, target_sum)
                cur_conv = _conv_full(a, fft_len)
                cur_max = float(cur_conv.max())
                print(f"[RESHUFFLE {step_counter}] mass reshuffled, cur_max={cur_max:.6f}")

        # If we have gone a long time without improvement, do a tiny random tweak.
        if step_counter - last_improve_step > 500000:
            # tiny random redistribution of a few entries
            i1, i2 = np.random.choice(n, 2, replace=False)
            amt = a[i1] * 0.01 * np.random.rand()
            a[i1] -= amt
            a[i2] += amt
            cur_conv = _conv_full(a, fft_len)
            cur_max = float(cur_conv.max())
            last_improve_step = step_counter
            print(f"[TWEAK {step_counter}] small tweak, cur_max={cur_max:.6f}")

    # -----------------------------------------------------------------
    print(f"[DONE] best max‑conv = {best_val:.6f} after {step_counter} SA steps")
    return best_seq.tolist()
