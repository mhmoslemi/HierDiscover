def _refine_sequence_subgrad(a0, n, deadline, lr0=0.8):
    v = np.log(a0 + eps)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    best_seq = a0.copy()
    best_val = 2 * n * np.max(_conv_fft(a0, a0))
    it = 0
    while time.time() < deadline:
        it += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_max = conv.max()
        cur_val = 2 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine-sub] n={n} it={it} best={cur_val:.6f}")
        # active set
        active = np.where(conv >= cur_max - 1e-12)[0]
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
        # compute grad_a using convolution of reversed a and w as earlier
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]
        # gradient w.r.t logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)
        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)
        lr = lr0 / np.sqrt(it)
        v -= lr * m_hat / (np.sqrt(v2_hat) + 1e-12)
    return best_seq, best_val

"""
Optimisation of a non‑negative step function.

The optimiser builds a large set of analytically‑motivated seed
distributions (geometric, power‑law, Gaussian, Laplace, β‑distribution,
triangular, sine‑window, Dirichlet draws, random soft‑max).  The best
few seeds are refined by a two‑phase gradient descent on the simplex:

1.  A smoothed‑max (log‑sum‑exp) phase – Adam on the logits of a
    soft‑max parametrisation, with a slowly increasing “β’’ to gently
    concentrate the gradient on the current maximal convolution entry.

2.  An exact sub‑gradient phase – the active set where the
    convolution attains its maximum is taken, a uniform weight vector
    over that set is built and the exact sub‑gradient of the L∞‑norm
    is used (via a convolution with the reversed vector).  Adam is
    again applied.

After the two‑phase refinement a number of small random perturbations
(of the logits) are tried, each followed by a short sub‑gradient
refinement.  The best sequence seen before the global deadline is
returned (normalised to sum‑to‑one, which leaves the evaluation
function invariant).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → Δⁿ (simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _laplace(n: int, scale: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = np.exp(-np.abs(x) / scale)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _beta_dist(n: int, a: float, b: float) -> np.ndarray:
    # Discrete Beta on {1,…,n}
    i = np.arange(1, n + 1, dtype=np.float64)
    a_vals = (i ** (a - 1)) * ((n + 1 - i) ** (b - 1))
    s = a_vals.sum()
    return a_vals / s if s != 0 else np.full(n, 1.0 / n)


def _tent(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.minimum(i, n - 1 - i) + 1.0
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Two‑phase refinement procedures
# ----------------------------------------------------------------------


def _refine_sequence_smooth(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 1.0,
    beta_mul: float = 1.01,
    lr0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑soft‑max parametrisation using a
    smoothed max (log‑sum‑exp) as loss.
    Returns the best probability vector seen and its fast objective
    ``2·n·max(conv)``.
    """
    eps = 1e-30
    v = np.log(a0 + eps)          # logits
    m = np.zeros_like(v)          # first moment
    v2 = np.zeros_like(v)         # second moment

    best_seq = a0.copy()
    cur_conv = _conv_fft(a0, a0)
    best_val = 2.0 * n * np.max(cur_conv)

    beta = start_beta
    it = 0

    while True:
        it += 1
        if time.time() > deadline:
            break

        a = _softmax(v)

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[smooth] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # --- smoothed‑max weights ---------------------------------------
        beta *= beta_mul
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # --- gradient w.r.t. a (smooth loss) --------------------------
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]   # central n entries

        # --- gradient w.r.t. logits v (soft‑max chain rule) ----------
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # --- Adam update ----------------------------------------------
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v -= lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


def _refine_sequence_subgrad(
    a0: np.ndarray,
    n: int,
    deadline: float,
    lr0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑soft‑max parametrisation using the exact
    sub‑gradient of the L∞‑norm (max of the convolution).  The active
    set is the set of indices where the convolution attains its maximum.
    """
    eps = 1e-30
    v = np.log(a0 + eps)          # logits
    m = np.zeros_like(v)          # first moment
    v2 = np.zeros_like(v)         # second moment

    best_seq = a0.copy()
    cur_conv = _conv_fft(a0, a0)
    best_val = 2.0 * n * np.max(cur_conv)

    it = 0

    while True:
        it += 1
        if time.time() > deadline:
            break

        a = _softmax(v)

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[subgrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # active set (all indices attaining the maximum up to a tiny tolerance)
        tol = 1e-14
        active = np.where(conv >= cur_max - tol)[0]

        # uniform weights on the active set
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size

        # gradient of the max (sub‑gradient) via convolution with reversed a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # chain rule through the soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v -= lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence minimising ``evaluate_sequence``.
    The routine builds a rich pool of analytic seeds, evaluates them, then
    refines the best few using a two‑phase Adam optimiser (smooth → exact
    sub‑gradient).  Random perturbations of the final best are explored as
    well.  The function respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the calling wrapper
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a length – 8000 works well for the 1000 s budget.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Construct a diverse set of seed distributions (all sum to 1)
    # ------------------------------------------------------------------
    seeds = []

    # Geometric families
    for c in np.linspace(0.5, 5.0, 10):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # Power‑law families
    for p in np.linspace(0.0, 2.2, 10):
        seeds.append(_power(n, p))

    # Combined geometric × power families
    for c in np.linspace(0.5, 4.0, 7):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 7):
            seeds.append(_geom_power(n, r, p))

    # Symmetric geometric
    for c in np.linspace(0.5, 4.5, 7):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # Classical sine window
    seeds.append(_sine_window(n))

    # Gaussian family (different bandwidths)
    for sigma_factor in (0.12, 0.16, 0.20):
        sigma = sigma_factor * n
        seeds.append(_gaussian(n, sigma))

    # Laplace family
    for scale_factor in (0.12, 0.20):
        scale = scale_factor * n
        seeds.append(_laplace(n, scale))

    # Discrete Beta distributions
    for a, b in ((0.5, 0.5), (2, 2), (5, 1), (1, 5)):
        seeds.append(_beta_dist(n, a, b))

    # Simple tent (triangular) shape
    seeds.append(_tent(n))

    # Dirichlet draws (different concentration)
    for alpha in (0.2, 0.6, 0.9, 2.0, 5.0):
        a = np.random.gamma(alpha, 1.0, size=n)
        s = a.sum()
        if s > 0:
            seeds.append(a / s)

    # Random soft‑max seeds
    for _ in range(6):
        v_rand = np.random.randn(n)
        seeds.append(_softmax(v_rand))

    # Re‑use the previously discovered construction if present
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            a_prev = np.asarray(prev, dtype=np.float64)
            if a_prev.ndim == 1:
                if a_prev.shape[0] != n:
                    # adapt size by truncation / zero‑padding
                    a_adj = np.zeros(n, dtype=np.float64)
                    m = min(a_prev.shape[0], n)
                    a_adj[:m] = a_prev[:m]
                    a_prev = a_adj
                if a_prev.sum() > 0:
                    seeds.append(a_prev / a_prev.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Fast scan – keep the best seed and also the K best for deeper work
    # ------------------------------------------------------------------
    def fast_obj(a):
        return 2.0 * n * np.max(_conv_fft(a, a))

    best_val = float("inf")
    best_seq = None
    for i, a in enumerate(seeds):
        cur_val = fast_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        if i % 50 == 0:
            print(f"[scan] processed {i:4d}/{len(seeds):4d} seeds, current best = {best_val:.6f}")

    print(f"[scan] overall best seed value = {best_val:.6f}")

    # Keep the top‑K seeds for intensive refinement
    K = 8
    top_seeds = sorted(seeds, key=fast_obj)[:K]

    # ------------------------------------------------------------------
    # Two‑phase refinement on the top seeds
    # ------------------------------------------------------------------
    global_best_seq = best_seq
    global_best_val = best_val

    # Allocate a rough budget for each refinement round
    remaining = max(0.0, deadline - time.time())
    per_refine = max(5.0, remaining / (K * 2 + 5))

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() > deadline:
            break

        # Phase 1 – smoothed max
        sub_deadline = min(deadline, time.time() + per_refine)
        seq_s, val_s = _refine_sequence_smooth(
            seed_seq,
            n,
            sub_deadline,
            start_beta=1.0,
            beta_mul=1.01,
            lr0=0.6,
        )
        if val_s < global_best_val:
            global_best_val = val_s
            global_best_seq = seq_s.copy()
            print(f"[refine] smoother improved to {val_s:.6f}")

        if time.time() > deadline:
            break

        # Phase 2 – exact sub‑gradient
        sub_deadline = min(deadline, time.time() + per_refine)
        seq_sub, val_sub = _refine_sequence_subgrad(
            seq_s,
            n,
            sub_deadline,
            lr0=0.6,
        )
        if val_sub < global_best_val:
            global_best_val = val_sub
            global_best_seq = seq_sub.copy()
            print(f"[refine] sub‑gradient improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Random perturbation loop (keep improving until time runs out)
    # ------------------------------------------------------------------
    perturb_eps = 0.01
    while time.time() < deadline:
        noisy_logits = np.log(global_best_seq + 1e-30) + perturb_eps * np.random.randn(n)
        pert_seq = _softmax(noisy_logits)

        sub_deadline = min(deadline, time.time() + per_refine)
        seq_p, val_p = _refine_sequence_subgrad(
            pert_seq,
            n,
            sub_deadline,
            lr0=0.6,
        )
        if val_p < global_best_val:
            global_best_val = val_p
            global_best_seq = seq_p.copy()
            print(f"[perturb] improved to {val_p:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation (objective is scale‑invariant, but we keep sum=1)
    # ------------------------------------------------------------------
    total = global_best_seq.sum()
    if total > 0:
        global_best_seq = global_best_seq / total

    result = global_best_seq.tolist()
    print(f"[final] best objective (fast) = {global_best_val:.6f}")
    return result
