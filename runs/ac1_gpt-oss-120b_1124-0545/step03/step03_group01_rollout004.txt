"""
Search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of the following phases:

1. **Seed generation** – a rich collection of parameterised families
   (uniform, sine‑window, sine^p, beta‑shaped, geometric, power‑law,
   symmetric geometric, random soft‑max) together with an up‑sampled
   version of the best sequence found so far.

2. **Coarse evaluation** – the raw objective of each seed is computed
   and the top‑few seeds are kept for refinement.

3. **Smooth‑max Adam optimisation** – the probabilities a are
   represented by log‑softmax parameters v.  The loss is a smoothed
   maximum of the auto‑convolution obtained via the log‑sum‑exp trick.
   A slowly increasing temperature (β) together with Adam’s adaptive
   moments yields a fast descent on the simplex.

4. **Exact sub‑gradient refinement** – from the best Adam result a
   mirror‑descent (exponentiated‑gradient) update using the true
   sub‑gradient of the max‑convolution is performed for many iterations.

5. **Local random perturbation** – tiny mass transfers between random
   pairs of entries are accepted if they improve the true objective.
   This cheap stochastic hill‑climbing can squeeze out the last few
   percent.

All phases respect a global time budget (default 1000 s).  The best
sequence discovered is stored in the global variable ``height_sequence_1``
and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.ones(n, dtype=np.float64) / n


# ----------------------------------------------------------------------
# Parametric families (all returned already normalised to sum 1)
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _sine_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** p
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like polynomial shape: (i/n)^α * (1‑i/n)^β, indices 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64) / n
    a = (i ** alpha) * ((1.0 - i) ** beta)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _interpolate_to_length(seq: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of a shorter sequence to length n."""
    if len(seq) == n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, len(seq))
    x_new = np.linspace(0.0, 1.0, n)
    new = np.interp(x_new, x_old, seq)
    new[new < 0] = 0.0
    s = new.sum()
    if s == 0.0:
        return _uniform(n)
    return new / s


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a) == 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max loss
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed maximum of the auto‑convolution.
    Returns the best sequence (as a numpy array) encountered and its
    objective value (2·n·max(conv) because sum(a)=1).
    """
    v = np.log(seed + 1e-30)               # log‑softmax parameters
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # Adam hyper‑parameters
    lr0 = 0.6
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    # Smoothing temperature schedule (slow growth)
    beta = 1.0
    beta_mul = 1.015   # gentle increase

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                     # stays on the simplex

        # current true objective (fast)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta *= beta_mul
        shift = np.max(conv)                 # shift for numerical safety
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:                     # under‑flow -> fall back to true max
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ----- gradient w.r.t a (∇_a Σ w_k·conv_k) -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # ----- convert to gradient w.r.t. v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent refinement using the exact sub‑gradient of max(conv).
    Uses an exponentiated‑gradient (EG) update that keeps the vector on the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    rng = np.random.default_rng(12345)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # Sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Exponentiated‑gradient step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)               # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random local perturbation (cheap stochastic hill‑climbing)
# ----------------------------------------------------------------------


def _random_local_refine(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Perform a large number of tiny mass transfers between random indices.
    Accept a move only if it improves the true objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))
    rng = np.random.default_rng(98765)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.choice(n, size=2, replace=False)
        if a[i] == 0.0:
            continue
        # Transfer a small fraction of the mass at i to j
        eps = rng.uniform(0.0, min(a[i], 1e-2))
        if eps == 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # Compute new objective
        conv = _conv_fft(a_new, a_new)
        val = 2.0 * n * np.max(conv)

        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new                   # keep the improvement
            print(f"[Local] n={n} it={it:5d} best={val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (includes many parametric families)
# ----------------------------------------------------------------------


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse collection of starting points for length `n`.
    All seeds are already normalised to sum 1.
    """
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # sine^p family
    for p in np.linspace(0.5, 5.0, 9):
        seeds.append(_sine_power(n, p))

    # beta‑like shapes
    for alpha in np.linspace(0.0, 5.0, 6):
        for beta in np.linspace(0.0, 5.0, 6):
            seeds.append(_beta_shape(n, alpha, beta))

    # geometric families
    for c in np.linspace(0.5, 4.5, 7):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))

    # mixed geometric × power
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # random soft‑max vectors
    rng = np.random.default_rng(42)
    for _ in range(10):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # up‑sampled previous best (if any)
    if previous is not None:
        up = _interpolate_to_length(previous, n)
        seeds.append(up)

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given time
    budget and returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    # Try to reuse a previously discovered construction
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Lengths to explore – start modestly, enlarge if time permits
    lengths = [4000, 8000, 12000, 16000]

    global_best_seq = None
    global_best_val = float("inf")

    for idx_len, n in enumerate(lengths):
        if time.time() > deadline:
            break

        # Rough time share per remaining length
        remaining_lengths = len(lengths) - idx_len
        time_per_len = (deadline - time.time()) / remaining_lengths
        print(f"\n=== Optimising length {n} (approx. {time_per_len:.1f}s budget) ===")

        # --------------------------------------------------------------
        # 1. Seed generation and raw evaluation
        # --------------------------------------------------------------
        seeds = _make_seeds(n, previous)

        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n} new global best={val:.6f}")

        # Keep only the most promising seeds for the heavy optimisation
        seeds_sorted = sorted(seeds, key=_objective)
        top_seeds = seeds_sorted[:5]

        # --------------------------------------------------------------
        # 2. Adam optimisation (smooth max) – one run per top seed
        # --------------------------------------------------------------
        for seed_vec in top_seeds:
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline, max_iters=20000)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[Adam] n={n} new global best={val:.6f}")

        # --------------------------------------------------------------
        # 3. Exact sub‑gradient refinement (mirror descent)
        # --------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq_ref, val_ref = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=30000)
            if seq_ref is not None and val_ref < global_best_val:
                global_best_val = val_ref
                global_best_seq = seq_ref.copy()
                print(f"[SubGrad] n={n} new global best={val_ref:.6f}")

        # --------------------------------------------------------------
        # 4. Random local perturbation (cheap hill‑climbing)
        # --------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            seq_loc, val_loc = _random_local_refine(
                global_best_seq, n, deadline, max_iters=20000)
            if seq_loc is not None and val_loc < global_best_val:
                global_best_val = val_loc
                global_best_seq = seq_loc.copy()
                print(f"[Local] n={n} new global best={val_loc:.6f}")

        # Use the best sequence of this length as a warm‑start for the next
        previous = global_best_seq.copy() if global_best_seq is not None else None

    # ------------------------------------------------------------------
    # Final sanity check – make sure the returned list passes the verifier
    # ------------------------------------------------------------------
    if global_best_seq is None:
        global_best_seq = _uniform(4000)
        print("[fallback] using uniform distribution")

    # Clamp to the allowed range and ensure non‑negativity (the optimiser already respects this)
    best_list = np.clip(global_best_seq, 0.0, 1000.0).tolist()

    # Store globally for possible reuse in subsequent runs
    global height_sequence_1
    height_sequence_1 = best_list

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(best_list)})"
    )
    return best_list
