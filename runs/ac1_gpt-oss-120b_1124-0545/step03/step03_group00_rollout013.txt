def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT – returns length len(x)+len(y)-1."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric series a_i = r**i  (normalised to sum 1)."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i = 1/(i+1)^p  (normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    if s == 0.0:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined geometric × power‑law."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    if s == 0.0:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (mirrored about the centre)."""
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    if s == 0.0:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _sine_window(n: int) -> np.ndarray:
    """Smooth sine window, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _fast_objective(a: np.ndarray) -> float:
    """
    Fast (scale‑invariant) objective used during the search:
        2 * n * max(conv(a,a))
    The returned value equals evaluate_sequence(a) when sum(a)=1.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Adam optimiser that slowly sharpens the smoothed‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 5.0,
    beta_mul: float = 1.02,
    max_beta: float = 800.0,
    lr0: float = 0.8,
    t0: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑softmax representation.
    Returns the best sequence seen (as np.ndarray) and its fast objective.
    """
    # logits (log‑softmax parameters)
    v = np.log(a0 + 1e-30)

    m = np.zeros_like(v)      # first moment estimate
    v2 = np.zeros_like(v)     # second moment estimate

    best_seq = a0.copy()
    best_val = _fast_objective(a0)

    beta = start_beta
    it = 0

    while time.time() < deadline:
        it += 1

        # current probability vector (always sums to 1)
        a = _softmax(v)

        # convolution and fast objective
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[adam] n={n:5d} it={it:7d} best={cur_val:.6f}")

        # ------------------------------------------------------------------
        # Choose weight vector w.
        #   * for the first part of the run we use a smooth soft‑max;
        #   * once beta is large enough we switch to an exact one‑hot.
        # ------------------------------------------------------------------
        if beta < max_beta:
            # smooth approximation
            shift = cur_max
            w = np.exp(beta * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                # numerical glitch – skip this iteration
                beta *= beta_mul
                continue
            w /= w_sum
        else:
            # exact sub‑gradient: put all mass on the arg‑max lag(s)
            # in case of ties we spread uniformly over them
            max_idxs = np.flatnonzero(conv == cur_max)
            w = np.zeros_like(conv)
            w[max_idxs] = 1.0 / len(max_idxs)

        # gradient w.r.t. a of the smooth / exact loss
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]    # central n entries

        # gradient w.r.t. logits v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)                   # decreasing learning rate
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

        # increase smoothing parameter (geometric schedule)
        beta = min(beta * beta_mul, max_beta)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple hill‑climbing perturbation (mass transfer between two bins)
# ----------------------------------------------------------------------


def _local_perturb(a: np.ndarray, steps: int, radius: float = 0.02) -> np.ndarray:
    """
    Perform a quick coordinate‑wise hill‑climbing:
    repeatedly move a small ε of mass from a random index i to j.
    The move is kept only if it improves the fast objective.
    """
    n = len(a)
    cur = a.copy()
    cur_val = _fast_objective(cur)

    for _ in range(steps):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = radius * np.random.rand()
        # ensure we do not make a[i] negative
        if cur[i] < eps:
            continue
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        cand /= cand.sum()          # renormalise to stay on the simplex
        cand_val = _fast_objective(cand)
        if cand_val < cur_val:
            cur = cand
            cur_val = cand_val
    return cur


# ----------------------------------------------------------------------
# Main entry point – creates candidates, refines, returns best list
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises the
    supplied evaluate_sequence function.  The routine respects the
    supplied time budget (default 1000 s) and returns the best sequence
    found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # safety margin for final checks

    # ------------------------------------------------------------------
    # Lengths to try – we start with the largest that fits into the time.
    # ------------------------------------------------------------------
    lengths = [8000, 4000]                # fallback to 4000 if we run out of time
    global_best_seq = None
    global_best_val = float("inf")        # we minimise the fast objective

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n[main] ------- Optimising length = {n} -------")

        # --------------------------------------------------------------
        # 1) Build a diverse list of seeds (all are probability vectors)
        # --------------------------------------------------------------
        seed_seqs = []

        # Geometric families
        for c in np.linspace(0.5, 5.0, 12):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_geom(n, r))

        # Power‑law families
        for p in np.linspace(0.0, 2.2, 12):
            seed_seqs.append(_power(n, p))

        # Combined geometric × power families
        for c in np.linspace(0.5, 4.0, 8):
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in np.linspace(0.0, 2.0, 8):
                seed_seqs.append(_geom_power(n, r, p))

        # Symmetric geometric
        for c in np.linspace(0.5, 4.5, 8):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_sym_geom(n, r))

        # Sine window (good baseline)
        seed_seqs.append(_sine_window(n))

        # Dirichlet draws – give a few random sparse patterns
        for alpha in [0.2, 0.5, 0.8, 2.0, 5.0]:
            a = np.random.gamma(alpha, 1.0, size=n)
            if a.sum() > 0.0:
                seed_seqs.append(a / a.sum())

        # Random soft‑max seeds
        for _ in range(5):
            v_rand = np.random.randn(n)
            seed_seqs.append(_softmax(v_rand))

        # Use the previously found sequence if it exists
        prev = globals().get("height_sequence_1")
        if prev is not None:
            try:
                a_prev = np.asarray(prev, dtype=np.float64)
                if a_prev.ndim == 1:
                    if a_prev.shape[0] != n:
                        # truncate / zero‑pad to the required length
                        a_adj = np.zeros(n, dtype=np.float64)
                        m = min(a_prev.shape[0], n)
                        a_adj[:m] = a_prev[:m]
                        a_prev = a_adj
                    if a_prev.sum() > 0.0:
                        a_prev = a_prev / a_prev.sum()
                        seed_seqs.append(a_prev)
            except Exception:
                pass

        # --------------------------------------------------------------
        # 2) Quick scan – keep the top‑k seeds according to the fast objective
        # --------------------------------------------------------------
        top_k = 8
        scored = []
        for idx, s in enumerate(seed_seqs):
            val = _fast_objective(s)
            scored.append((val, s))
        scored.sort(key=lambda x: x[0])
        seed_pool = [s for (_, s) in scored[:top_k]]
        print(f"[scan] top‑{top_k} seed values: {[f'{v:.6f}' for v,_ in scored[:top_k]]}")

        # --------------------------------------------------------------
        # 3) Refine each seed with Adam + schedule, keep the best
        # --------------------------------------------------------------
        for seed_id, a_start in enumerate(seed_pool):
            if time.time() > deadline:
                break
            print(f"[refine] starting Adam on seed {seed_id+1}/{len(seed_pool)} (n={n})")
            refined_seq, refined_val = _adam_optimize(
                a_start,
                n,
                deadline,
                start_beta=5.0,
                beta_mul=1.03,
                max_beta=800.0,
                lr0=0.9,
                t0=0.02,
            )
            if refined_val < global_best_val:
                global_best_val = refined_val
                global_best_seq = refined_seq.copy()
                print(f"[overall] new global best (len={n}) = {refined_val:.6f}")

        # --------------------------------------------------------------
        # 4) Local hill‑climbing tweaks from the current global best
        # --------------------------------------------------------------
        if global_best_seq is not None:
            while time.time() < deadline - 1.0:
                perturbed = _local_perturb(global_best_seq, steps=2000, radius=0.01)
                val = _fast_objective(perturbed)
                if val < global_best_val:
                    global_best_val = val
                    global_best_seq = perturbed.copy()
                    print(f"[perturb] improved global best to {val:.6f}")

    # ------------------------------------------------------------------
    # Final safety checks and conversion to plain list
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # Fallback: uniform distribution (always feasible)
        n = 4000
        global_best_seq = np.ones(n, dtype=np.float64) / n
        print("[fallback] using uniform start as final answer")

    # Normalise to sum‑to‑one (evaluation function is scale‑invariant)
    s = global_best_seq.sum()
    if s > 0.0:
        global_best_seq = global_best_seq / s

    result = global_best_seq.tolist()
    print(f"[final] best fast objective = {global_best_val:.6f}")
    return result
