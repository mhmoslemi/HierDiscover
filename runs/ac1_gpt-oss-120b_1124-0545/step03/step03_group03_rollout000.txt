"""
Improved search for a non‑negative step function minimizing

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Broad analytic grid** – generates a few families of U‑shaped
   distributions (product, sum and distance‑based power laws) with
   many (p,offset) pairs and keeps the best normalised candidate.

2. **Exponentiated‑gradient descent** – works directly on the simplex
   using the smooth max approximation
       Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)
   Its gradient is obtained by FFTs.  An update of the form

       a ← a * exp(−η·∇Φ_λ)

   followed by renormalisation (mirror descent) respects non‑negativity
   and the probability‑simplex constraint.  The smoothing λ, the step
   size η and a Metropolis temperature are annealed exponentially.

3. **Escapes & jitter** – if no improvement is seen for a while the
   current solution is perturbed with a small random noise to leave
   plateaux.  Occasionally a pure random mass‑transfer move is tried.

The routine respects a 1000 s budget, prints intermediate scores and
returns the best sequence found.  The score printed is the exact
objective (the same that ``evaluate_sequence`` will compute).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact, scale‑invariant objective for a non‑negative vector.
    The vector is assumed to lie on the probability simplex (sum = 1);
    a small normalisation is performed for safety.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a small amount of mass from one index to another.
    The move respects the simplex by re‑normalisation.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = project_simplex(a_new)
    return a_new


def gradient_smooth(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smooth max approximation
        Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)

    The gradient formula is
        g_i = 2· Σ_k w_k·a_{k‑i},
    where w_k are soft‑max weights over the self‑convolution.
    Both convolutions are obtained via FFTs.
    """
    n = a.shape[0]

    # 1. self‑convolution
    conv = conv_fft(a)                 # length 2n‑1

    # 2. soft‑max weights, numerically stable
    max_c = np.max(conv)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)    # length 2n‑1, Σ w = 1

    # 3. weighted convolution (w reversed) * a
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)

    # extract the n entries that correspond to the gradient
    start = n - 1
    slice_part = conv_wa[start:start + n]    # length n
    g = 2.0 * slice_part[::-1]               # reverse to obtain g_i
    return g


def analytic_candidate_product(n: int, p: float, offset: float) -> np.ndarray:
    """
    a_i ∝ ((i+off)*(n‑1‑i+off))^{‑p}
    Symmetric, U‑shaped; offset>0 avoids division by zero.
    """
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_candidate_sum(n: int, p: float, offset: float) -> np.ndarray:
    """
    a_i ∝ (i+off)^{‑p} + (n‑1‑i+off)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_candidate_dist(n: int, p: float, offset: float) -> np.ndarray:
    """
    a_i ∝ (|i−mid| + offset)^{‑p},  mid = (n‑1)/2.
    Another symmetric U‑shape that can be tuned independently.
    """
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = (np.abs(i - mid) + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation metric.
    The routine runs for at most ``budget_s`` seconds and returns the
    best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))                  # length of the sequence

    # Analytic families to explore
    families = [
        ("product", analytic_candidate_product),
        ("sum", analytic_candidate_sum),
        ("dist", analytic_candidate_dist),
    ]

    # Parameter grids (coarse – the gradient phase will fine‑tune)
    p_vals = np.arange(0.15, 2.55, 0.15)            # ~16 values
    offset_vals = np.arange(0.0, 5.1, 0.25)          # ~21 values

    best_seq = None
    best_score = np.inf

    print("[init] Analytic grid search (n = {})".format(n))

    # ------------------------------------------------------------------
    # 1) Analytic grid search
    # ------------------------------------------------------------------
    for name, maker in families:
        for p in p_vals:
            for off in offset_vals:
                # Avoid exact zero offset for product family (division by zero)
                if name == "product" and off <= 0.0:
                    off_eff = 1e-3
                else:
                    off_eff = off
                cand = maker(n, p, off_eff)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[grid] new best {name} p={p:.2f} off={off:.2f} → {sc:.6f}")
    # If a reference sequence of the correct length is provided, consider it.
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            print(f"[init] reference score = {sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = ref.copy()
                print("[init] reference becomes current best")

    print(f"[init] best analytic start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Exponentiated‑gradient descent with annealing
    # ------------------------------------------------------------------
    a = best_seq.copy()          # current iterate (always on the simplex)
    iteration = 0
    last_improve = time.time()
    stagnation_cooldown = 20.0    # seconds without improvement before a jitter

    # Schedules – exponential for a smoother annealing
    lam_start, lam_end = 0.5, 1000.0
    step_start, step_end = 0.3, 5e-5
    temp_start, temp_end = 1.0, 0.001

    # main optimisation loop
    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 10.0)          # keep a tiny margin

        # exponential interpolation
        lam = lam_start * (lam_end / lam_start) ** progress
        step = step_start * (step_end / step_start) ** progress
        temp = temp_start * (temp_end / temp_start) ** progress

        # ------------------------------------------------------------------
        # Choose move type
        # ------------------------------------------------------------------
        if np.random.rand() < 0.85:                     # gradient‑based move
            grad = gradient_smooth(a, lam)
            # Exponentiated (mirror) update; keep numbers in a safe range
            update_factor = np.exp(-step * grad)
            # Avoid under‑flow to absolute zero – clip to a tiny floor
            update_factor = np.maximum(update_factor, 1e-16)
            cand = a * update_factor
            cand_sum = cand.sum()
            if cand_sum == 0.0:
                cand = np.ones_like(a) / n
            else:
                cand = cand / cand_sum
        else:                                            # random swap move
            cand = random_move(a, step * 0.5)

        cand_score = compute_score(cand)

        # Metropolis acceptance (helps escape shallow local minima)
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            if temp > 1e-12:
                prob = math.exp(-(cand_score - best_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:07d}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # ------------------------------------------------------------------
        # Stagnation handling – jitter restart
        # ------------------------------------------------------------------
        if time.time() - last_improve > stagnation_cooldown:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            last_improve = time.time()
            print(f"[{iteration:07d}] jitter restart (stagnation)")

        # Early exit if we already beat the target
        if best_score <= 1.503:
            print("[done] target reached, exiting early")
            break

        iteration += 1

    total_time = time.time() - start_time
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    # Return as ordinary Python list (the harness expects this)
    return best_seq.tolist()
