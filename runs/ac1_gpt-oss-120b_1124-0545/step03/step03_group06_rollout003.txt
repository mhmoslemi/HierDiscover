def _run_flatten(seq, time_budget, step0=0.5, decay=0.9999, log_interval=1000):
    start = time.time()
    n = len(seq)
    target_const = 1.0/(2*n-1)
    best_seq = seq.copy()
    conv = _conv_fft(seq)
    best_score = 2 * n * conv.max()
    step = step0
    iteration = 0
    while time.time() - start < time_budget:
        iteration += 1
        seq = _flatten_step(seq, step)
        step = max(step * decay, 1e-12)
        if iteration % log_interval == 0 or iteration == 1:
            conv = _conv_fft(seq)
            cur_score = 2 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[flatten improve] iter={iteration}, score={best_score:.6f}")
        # optionally randomness for jacks
        if iteration % 300 == 0:
            # small random perturbation
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(seq[i] * 0.05, step)
            seq[i] -= eps
            seq[j] += eps

"""
A two‑phase FFT‑accelerated optimizer for the “max self‑convolution” problem.

Phase 1 – L2‑flattening.
    minimise ½·∥a∗a – 1/(2n‑1)·1∥₂² by a projected gradient step.
    This drives the whole autocorrelation curve towards a flat shape,
    which typically reduces the peak far below the uniform‑distribution value.

Phase 2 – max‑subgradient polishing.
    after the flattening phase we switch to the classic sub‑gradient
    descent that directly reduces the largest convolution entry.

Both phases respect the simplex {a_i≥0, Σa_i=1}.  All convolutions are
performed with the FFT (O(n log n)) so we can afford lengths ≈ 2·10⁴
while staying well inside the 1000 s budget.  The routine returns a
Python list suitable for a single call to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Power‑law family a_i ∝ (i+1+offset)^{-α}, normalised to sum = 1.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full (linear) convolution of arbitrary length vectors using FFT.
    No clipping – signs are preserved.
    """
    size = _next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: len(a) + len(b) - 1]


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """
    Self‑convolution a∗a (full length 2·n‑1) with tiny negative values
    clipped to zero – the true convolution is non‑negative.
    """
    n = seq.shape[0]
    size = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    conv[conv < 0.0] = 0.0
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x≥0, Σx=1}.  O(n) implementation (Duchi et al., 2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase‑1: L2 flattening (make a∗a as flat as possible)
# ----------------------------------------------------------------------


def _flatten_step(seq: np.ndarray, step: float, const: float) -> np.ndarray:
    """
    One projected gradient step for the loss
        L = ½·∥a∗a – const·1∥₂²
    The analytical gradient is 4·( (a∗a – const)  *  a_rev ),
    where * denotes convolution and a_rev is the reversed vector.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)                     # a∗a
    e = conv - const                          # error term (can be negative)
    rev = seq[::-1]                           # a_rev
    conv_err = _conv_fft_general(e, rev)      # (a∗a – const) * a_rev, length = 3n‑2
    grad = 4.0 * conv_err[n - 1 : n - 1 + n]   # extract the n relevant components
    new_seq = seq - step * grad
    new_seq = _project_simplex(new_seq)
    return new_seq


def _run_flatten(seq: np.ndarray,
                 time_budget: float,
                 *,
                 step0: float = 0.5,
                 decay: float = 0.99997,
                 log_interval: int = 1000) -> tuple[np.ndarray, float]:
    """
    Run the L2‑flattening loop for at most ``time_budget`` seconds.
    Returns the best sequence (according to the true objective) and its score.
    """
    start = time.time()
    n = seq.shape[0]
    const = 1.0 / (2 * n - 1)          # the target value for each convolution entry
    best_seq = seq.copy()
    best_score = 2.0 * n * _conv_fft(seq).max()
    step = step0
    it = 0

    while time.time() - start < time_budget:
        it += 1
        seq = _flatten_step(seq, step, const)
        step = max(step * decay, 1e-12)

        # occasional evaluation (cheap compared to a full FFT)
        if it % log_interval == 0 or it == 1:
            conv = _conv_fft(seq)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[flatten] iter={it}, score={best_score:.6f}")

        # tiny random mass swap every few hundred steps (helps escape plateaus)
        if it % 300 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(step * seq[i], seq[i] * 0.05)
            seq[i] -= eps
            seq[j] += eps
            seq = _project_simplex(seq)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Phase‑2: Max‑subgradient descent (directly attacks the peak)
# ----------------------------------------------------------------------


def _max_subgradient_step(seq: np.ndarray, step: float) -> np.ndarray:
    """
    Sub‑gradient step that reduces the current maximum entry of a∗a.
    The gradient for a given peak index k is
        g_i = 2·a_{k‑i}  (zero if the index is out of range).
    After the step we project back onto the simplex.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    k = int(np.argmax(conv))                # index of the peak
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    i_range = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(seq)
    grad[i_range] = seq[k - i_range]        # a_{k‑i}
    grad *= 2.0
    grad -= grad.mean()                     # keep Σgrad = 0
    new_seq = seq - step * grad
    new_seq = _project_simplex(new_seq)
    return new_seq


def _run_max_subgradient(seq: np.ndarray,
                         time_budget: float,
                         *,
                         step0: float = 0.3,
                         decay: float = 0.99995,
                         log_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the original objective
    2·n·max(a∗a).  Stops after ``time_budget`` seconds.
    """
    start = time.time()
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = 2.0 * n * _conv_fft(seq).max()
    step = step0
    it = 0

    while time.time() - start < time_budget:
        it += 1
        seq = _max_subgradient_step(seq, step)
        step = max(step * decay, 1e-12)

        if it % log_interval == 0:
            cur_score = 2.0 * n * _conv_fft(seq).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[max‑sgd] iter={it}, score={best_score:.6f}")

        # occasional random swap (helps escape tiny flat regions)
        if it % 500 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(seq[i] * 0.1, step)
            seq[i] -= eps
            seq[j] += eps
            seq = _project_simplex(seq)

    return best_seq, best_score

def propose_candidate(seed: int = 0,

    """
    Search for a non‑negative step‑function that minimises

        1. a power‑law initialisation,
        2. an L2‑flattening stage (phase‑1),
        3. a max‑subgradient polishing stage (phase‑2).

    The routine respects the overall time budget and returns a list of
    floats (the best sequence found).
    """

    start_time = time.time()
    # keep a small safety margin for the final return
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------

    # final O(n²) evaluation cost.

    n = 20000
