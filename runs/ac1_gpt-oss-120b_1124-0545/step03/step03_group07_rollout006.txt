def _smooth_max_and_gradient(a, fft_len, beta):
    # compute full conv
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*a.size-1]
    max_val = conv.max()
    # compute softmax
    # To avoid overflow, subtract max
    exp_term = np.exp(beta * (conv - max_val))
    w = exp_term / exp_term.sum()
    # compute gradient: g = 2 * (w convolve rev(a)) at indices n-1..2n-2
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    conv_g = np.fft.irfft(fw * fa_rev, n=fft_len)
    n = a.size
    grad = 2.0 * conv_g[n-1:n-1+n]
    return max_val, grad, w
```

def _simulated_annealing(a_start, target_sum, fft_len, time_limit, initial_best_val):
    a = a_start.copy()
    cur_val,_ = _max_convolution(a, fft_len)
    best_val = cur_val
    best_seq = a.copy()
    start = time.time()
    T0 = best_val * 0.2
    Tmin = best_val * 1e-5
    step_counter = 0
    while time.time() < time_limit:
        # compute temperature based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / (time_limit - start + 1e-12)
        T = T0 * ((Tmin / T0) ** frac)
        # propose move
        i = np.random.randint(len(a))
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(len(a))
        if i == j:
            continue
        delta = a[i] * 0.1 * np.random.rand()
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta
        # compute max conv
        cand_val, _ = _max_convolution(a_cand, fft_len)
        # acceptance
        if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / max(T, 1e-12)):
            a = a_cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = a_cand.copy()
                print(f"[SA] step {step_counter}, new best {best_val:.6f}")
        step_counter += 1
    return best_seq, best_val

"""
search for a non‑negative step‑function sequence that minimizes the
evaluation function (which equals the maximum entry of the self‑convolution
when the ℓ₁‑norm is fixed to √(2·n)).

The algorithm consists of three stages:

1. **Smoothed projected gradient descent** – a differentiable soft‑max
   approximation of the max–convolution is used.  The gradient of the
   smooth objective is obtained with two FFT‑based convolutions, the step
   is taken in the opposite direction and the iterate is projected back
   onto the ℓ₁‑simplex {a≥0, Σa = √(2 n)}.

2. **Simulated annealing** – starting from the best vector found in
   stage 1, random mass transfers between two positions are accepted
   according to a Metropolis rule with an exponentially decaying temperature.
   This helps to escape shallow local minima.

3. **Fine‑tuning gradient descent** – a short high‑β (almost exact) gradient
   descent refines the final candidate.

All stages respect a total runtime of ~1000 s and continuously keep the
best sequence encountered.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , Σ w = s }."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 3·n‑2 (covers both conv‑lengths)."""
    return 1 << ((3 * n - 2).bit_length())


# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
def _smooth_max_and_gradient(
    a: np.ndarray, fft_len: int, beta: float
) -> tuple[float, np.ndarray]:
    """
    Return a smooth approximation of max(a*a) and its gradient.
    The smooth max is the soft‑max with parameter beta.
    Gradient is 2·(w ⊛ a_rev) where w are the soft‑max weights.
    """
    n = a.size
    # ----- self‑convolution (needed for both value and weights)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

    max_val = float(conv.max())

    # ----- soft‑max weights
    # stability: subtract max before exponentiation
    exp_terms = np.exp(beta * (conv - max_val))
    w = exp_terms / exp_terms.sum()

    # ----- gradient (2 * convolution of w with reversed a)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    grad_full = np.fft.irfft(fw * fa_rev, n=fft_len)
    # relevant slice: indices n‑1 … n‑1+n‑1
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return max_val, grad


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Exact max of the self‑convolution a*a and its index.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    idx = int(np.argmax(conv))
    val = float(conv[idx])
    return val, idx


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for the best step‑function sequence within ``budget_s`` seconds.
    Returns a plain Python list ready for ``evaluate_sequence``.
    """
    # ---- reproducibility -------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    # ---- problem size ----------------------------------------------------
    if "height_sequence_1" in globals():
        base_len = len(height_sequence_1)
    else:
        base_len = 2500

    # we aim for a slightly larger support (more degrees of freedom)
    n = max(base_len, 4000)            # keep FFT manageable (< 2³¹)
    target_sum = math.sqrt(2.0 * n)    # ℓ₁‑norm required by the objective

    # ---- FFT length -------------------------------------------------------
    fft_len = _fft_len_for_n(n)

    # ---- initial vector ---------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.zeros(n, dtype=float)
        a0[:base_len] = np.asarray(height_sequence_1, dtype=float)
        a0 = a0 / a0.sum() * target_sum
    else:
        a0 = np.random.exponential(scale=1.0, size=n)
        a0 = a0 / a0.sum() * target_sum

    best_seq = a0.copy()
    conv0 = _full_convolution(best_seq, fft_len)
    best_val = float(conv0.max())
    print(f"[INIT] n={n}, target_sum={target_sum:.3f}, init max(conv)={best_val:.6f}")

    # ==== Phase 1: smoothed projected gradient descent =====================
    start_time = time.time()
    phase1_deadline = start_time + budget_s * 0.55   # ~55 % of total budget
    a = best_seq.copy()
    it = 0
    last_improve = 0
    beta_start = 0.5
    beta_end = 20.0
    # step‑size scale (fraction of total mass moved each iteration)
    step_scale = 0.2

    while time.time() < phase1_deadline:
        # ---- progress fraction for beta schedule -------------------------
        now = time.time()
        frac = (now - start_time) / (phase1_deadline - start_time)
        beta = beta_start * ((beta_end / beta_start) ** frac)

        # ---- current convolution and smooth max -------------------------
        conv = _full_convolution(a, fft_len)
        max_val = float(conv.max())

        # record best exact max (conv.max() = objective value)
        if max_val < best_val:
            best_val = max_val
            best_seq = a.copy()
            last_improve = it
            print(f"[GD {it:6d}] new best max(conv) = {best_val:.6f}")

        # ---- gradient of the smooth max ---------------------------------
        # (re‑uses the just‑computed conv to avoid a second FFT)
        # compute soft‑max weights
        exp_terms = np.exp(beta * (conv - max_val))
        w = exp_terms / exp_terms.sum()

        # gradient = 2 * (w ⊛ rev(a))
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=fft_len)
        fa_rev = np.fft.rfft(a_rev, n=fft_len)
        grad_full = np.fft.irfft(fw * fa_rev, n=fft_len)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ---- step with diminishing size ---------------------------------
        norm = np.linalg.norm(grad)
        if norm > 0:
            step = (target_sum * step_scale) / (norm * math.sqrt(it + 1.0))
            a_candidate = a - step * grad
            a_candidate = _simplex_projection(a_candidate, target_sum)
            a = a_candidate
        else:
            # degenerate case: add a tiny random jitter
            a = _simplex_projection(a + np.random.normal(scale=1e-8, size=n), target_sum)

        it += 1

        # ---- simple stagnation handling ---------------------------------
        if it % 4000 == 0 and (it - last_improve) > 8000:
            # restart from a noisy perturbation of the current best
            a = best_seq.copy()
            a += np.random.normal(scale=0.02 * target_sum, size=n)
            a = np.maximum(a, 0.0)
            a = _simplex_projection(a, target_sum)
            last_improve = it
            print(f"[GD {it:6d}] restart due to stagnation")

    # ==== Phase 2: simulated annealing =====================================
    # Use the best vector obtained from Phase 1 as the starting point.
    phase2_deadline = start_time + budget_s * 0.95   # leave ~5 % for final refinement
    a = best_seq.copy()
    cur_val, _ = _max_convolution(a, fft_len)
    best_val = cur_val
    best_seq = a.copy()
    print(f"[SA START] initial max(conv) = {best_val:.6f}")

    # temperature schedule
    T0 = best_val * 0.20
    Tmin = best_val * 1e-5
    step_counter = 0
    while time.time() < phase2_deadline:
        # cooling based on elapsed fraction of the SA stage
        now = time.time()
        frac = (now - (phase1_deadline)) / (phase2_deadline - phase1_deadline + 1e-12)
        T = T0 * ((Tmin / T0) ** frac)

        # ----- propose a random mass transfer ----------------------------
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = a[i] * 0.10 * np.random.rand()   # move ≤10 % of source mass
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta

        cand_val, _ = _max_convolution(a_cand, fft_len)

        # ----- Metropolis acceptance ------------------------------------
        if cand_val < cur_val:
            a = a_cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = a_cand.copy()
                print(f"[SA {step_counter:7d}] new best max(conv) = {best_val:.6f}")
        else:
            prob = math.exp(-(cand_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = a_cand
                cur_val = cand_val

        step_counter += 1

    # ==== Phase 3: high‑β fine‑tuning ======================================
    phase3_deadline = start_time + budget_s - 0.2   # tiny safety margin
    a = best_seq.copy()
    it = 0
    while time.time() < phase3_deadline:
        # use a large beta to approximate the true max (near exact)
        beta = 80.0
        conv = _full_convolution(a, fft_len)
        max_val = float(conv.max())
        if max_val < best_val:
            best_val = max_val
            best_seq = a.copy()
            print(f"[FINETUNE {it:6d}] improved max(conv) = {best_val:.6f}")

        # gradient at high beta
        exp_terms = np.exp(beta * (conv - max_val))
        w = exp_terms / exp_terms.sum()
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=fft_len)
        fa_rev = np.fft.rfft(a_rev, n=fft_len)
        grad_full = np.fft.irfft(fw * fa_rev, n=fft_len)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # tiny step (the landscape is now steep)
        norm = np.linalg.norm(grad)
        if norm > 0:
            step = (target_sum * 0.02) / (norm * math.sqrt(it + 1.0))
            a = a - step * grad
            a = _simplex_projection(a, target_sum)
        else:
            a = _simplex_projection(a + np.random.normal(scale=1e-9, size=n), target_sum)

        it += 1
        if it % 2000 == 0:
            # occasional cheap check to avoid dead‑ends
            val_check, _ = _max_convolution(a, fft_len)
            if val_check < best_val:
                best_val = val_check
                best_seq = a.copy()
                print(f"[FINETUNE {it:6d}] checkpoint max(conv) = {best_val:.6f}")

    # ----------------------------------------------------------------------
    print(f"[DONE] final best max(conv) = {best_val:.6f}")
    return best_seq.tolist()
