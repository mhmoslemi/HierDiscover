def adam_update(a, conv_full, a_rev, size2, step, tau, m, v, t, beta1=0.9, beta2=0.999, eps=1e-8):
    max_b = conv_full.max()
    w = np.exp((conv_full - max_b) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        weight = np.full_like(w, 1./w.size)
    else:
        weight = w / w_sum

    g_full = conv_fft(a_rev, weight, size2)
    n = a.shape[0]
    g = g_full[n-1 : n-1 + n]
    grad = -2.0 * g
    # Project gradient to hyperplane sum=0 (since we will renorm later)
    grad = grad - grad.mean()
    # Adam updates
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a) / a.shape[0]
    else:
        a_new /= s
    return a_new, m, v
```

"""
Propose a low‑value step‑function sequence.

Algorithm
---------
* Initialise a probability vector (sum = 1, non‑negative).  Several
  deterministic candidates are tried first (uniform and a few symmetric
  Beta‑distributions).  The best among them seeds the main optimisation.
* A smooth‑max approximation of the peak of the self‑convolution
  `a*a` is minimised by Adam (adaptive‑momentum) gradient steps.
  The gradient of the smooth objective is
        -2·(a_rev ∗ w)               (central n entries)
  where `w` are soft‑max weights of the convolution values.
* After each update the vector is projected back onto the probability
  simplex (clip to ≥0 then renormalise).
* With a small probability a random “mass‑transfer’’ move is attempted
  (move a fraction of mass from one index to another).  If the move
  improves the true evaluation it is kept, otherwise it is discarded.
* If no improvement is observed for a while the algorithm restarts from
  a fresh random point to escape possible stagnation.
* The whole search respects the 1000 s budget and returns the best
  sequence found.
"""

import time
import numpy as np
from scipy import special  # for beta‑distribution normalisation (optional)


# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT (real‑valued, full length).
    ``size`` must be ≥ len(u) + len(v) - 1.
    Returns the full convolution (len = len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def full_conv(a: np.ndarray, size: int) -> np.ndarray:
    """Full self‑convolution a*a (linear)."""
    return conv_fft(a, a, size)


def random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a random fraction (≤scale) of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / n
    else:
        a_new /= s
    return a_new


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).  Simple clip‑and‑renorm works
    because all entries are already non‑negative after the gradient step.
    """
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        return np.ones_like(v) / v.shape[0]
    return v / s


def beta_symmetric(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density sampled at the centre of n equal bins.
    Returned vector is already normalised to sum‑to‑1.
    """
    x = (np.arange(n) + 0.5) / n
    # pdf ∝ x^{α-1} (1-x)^{α-1}
    pdf = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    # Guard against overflow/underflow
    pdf = np.nan_to_num(pdf, nan=0.0, posinf=0.0, neginf=0.0)
    total = pdf.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return pdf / total


def evaluate_fast(a: np.ndarray, n: int, size1: int) -> float:
    """
    Fast evaluation of the objective for a normalised distribution.
    The true `evaluate_sequence` function does clipping; here we keep
    all entries far below the clipping threshold, so the result matches.
    """
    conv = full_conv(a, size1)
    max_b = conv.max()
    return 2.0 * n * max_b


def adam_step(
    a: np.ndarray,
    conv_full: np.ndarray,
    a_rev: np.ndarray,
    size2: int,
    step: float,
    tau: float,
    m: np.ndarray,
    v: np.ndarray,
    beta1_pow: float,
    beta2_pow: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
):
    """
    One Adam update for the soft‑max approximated max‑convolution loss.

    Returns the updated vector and the updated Adam state.
    """
    max_b = conv_full.max()
    # soft‑max weights over the convolution entries
    w = np.exp((conv_full - max_b) / tau)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient = -2·(a_rev ∗ w)  (central n entries)
    g_full = conv_fft(a_rev, w, size2)
    n = a.shape[0]
    g = g_full[n - 1 : n - 1 + n]
    grad = -2.0 * g

    # remove the component that changes total mass (project to Σgrad=0)
    grad -= grad.mean()

    # Adam moments
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    beta1_pow *= beta1
    beta2_pow *= beta2

    m_hat = m / (1.0 - beta1_pow)
    v_hat = v / (1.0 - beta2_pow)

    a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
    a_new = project_simplex(a_new)

    return a_new, m, v, beta1_pow, beta2_pow

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step function within ``budget_s`` seconds.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Optional ``n`` – length of the sequence (default 6000).

    Returns
    -------
    list[float]
        The best sequence found (list of Python floats).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # initialise a feasible distribution
    # --------------------------------------------------------------
    # Try to use the global reference sequence if it exists.
    seq0 = globals().get("height_sequence_1", None)
    if seq0 is not None:
        seq0 = np.asarray(seq0, dtype=float)
        if seq0.size != n:
            # linear interpolation to required length
            old_x = np.linspace(0, 1, seq0.size, endpoint=False)
            new_x = np.linspace(0, 1, n, endpoint=False)
            a0 = np.interp(new_x, old_x, seq0)
        else:
            a0 = seq0.copy()
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() == 0.0:
            a0 = np.random.rand(n)
        a0 /= a0.sum()
    else:
        a0 = np.random.rand(n)
        a0 /= a0.sum()

    # --------------------------------------------------------------
    # generate a few parametric candidates (symmetric Beta)
    # --------------------------------------------------------------
    cand_alphas = [0.30, 0.40, 0.50, 0.60, 0.70, 0.80, 0.90, 1.00]
    candidates = [a0]  # start with whatever we have
    for alpha in cand_alphas:
        cand = beta_symmetric(n, alpha)
        candidates.append(cand)

    # also try simple mixtures of uniform and Beta(0.5) (U‑shaped)
    uniform = np.full(n, 1.0 / n)
    beta05 = beta_symmetric(n, 0.5)
    for lam in [0.2, 0.5, 0.8]:
        mix = lam * beta05 + (1.0 - lam) * uniform
        mix = project_simplex(mix)
        candidates.append(mix)

    # --------------------------------------------------------------
    # pick the best candidate according to the true objective
    # --------------------------------------------------------------
    size1 = next_pow2(2 * n - 1)   # for a*a
    size2 = next_pow2(3 * n - 2)   # for a_rev * weight

    best_seq = None
    best_score = np.inf
    for idx, cand in enumerate(candidates):
        # ensure values lie within the clipping window used by evaluate_sequence
        cand = np.clip(cand, 0.0, 1000.0)
        cand = project_simplex(cand)
        score = evaluate_fast(cand, n, size1)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
        # small progress report
        # (in a real run this prints at most a handful of lines)
        print(f"[init‑cand {idx:02d}] alpha≈{cand_alphas[idx-1] if idx>0 else 'mix'}  score={score:.6f}")

    # --------------------------------------------------------------
    # initialise Adam state
    # --------------------------------------------------------------
    a = best_seq.copy()
    a_rev = a[::-1].copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0

    # --------------------------------------------------------------
    # hyper‑parameters
    # --------------------------------------------------------------
    base_step = 0.6               # initial learning‑rate scale
    rand_prob = 0.12              # probability of a random transfer
    rand_scale = 0.04             # magnitude of random transfer
    tau_factor = 0.30             # τ = tau_factor * max_b  (starts relatively smooth)
    tau_decay = 0.9995            # geometric decay per iteration
    stagnation_seconds = 30.0     # restart if no improvement for this long
    last_improve = start
    it = 1

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # current convolution and score
        conv_full = full_conv(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        # update best if needed
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ------------------------------------------------------------------
        # stochastic move vs. gradient step
        # ------------------------------------------------------------------
        if np.random.rand() < rand_prob:
            # propose a local random mass transfer and keep it only if it improves
            a_trial = random_transfer(a, rand_scale)
            conv_trial = full_conv(a_trial, size1)
            trial_score = 2.0 * n * conv_trial.max()
            if trial_score < cur_score:
                a = a_trial
                a_rev = a[::-1].copy()
                # reset Adam moments – the direction may have changed drastically
                m.fill(0.0)
                v.fill(0.0)
                beta1_pow = 1.0
                beta2_pow = 1.0
        else:
            # Adam gradient descent on the smooth max
            tau = max(tau_factor * max_b, 1e-12)
            step = base_step / np.sqrt(it)      # diminishing step size
            a, m, v, beta1_pow, beta2_pow = adam_step(
                a,
                conv_full,
                a_rev,
                size2,
                step,
                tau,
                m,
                v,
                beta1_pow,
                beta2_pow,
                beta1,
                beta2,
            )
            a_rev = a[::-1].copy()
            # anneal the temperature factor
            tau_factor *= tau_decay

        # ------------------------------------------------------------------
        # occasional restart on stagnation
        # ------------------------------------------------------------------
        if it % 50000 == 0 and (time.time() - last_improve) > stagnation_seconds:
            a = np.random.rand(n)
            a /= a.sum()
            a_rev = a[::-1].copy()
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta2_pow = 1.0
            tau_factor = 0.30
            print(f"[{it:7d}] restart due to stagnation")

        it += 1

    # --------------------------------------------------------------
    # final refinement (short hill‑climbing phase if time permits)
    # --------------------------------------------------------------
    # Use the remaining time for a pure hill‑climbing on the true
    # evaluation function – this may capture tiny improvements that the
    # smooth surrogate missed.
    while time.time() < deadline - 0.1:
        # try a tiny random transfer
        a_trial = random_transfer(best_seq, 0.01)
        # evaluate using the exact function (avoid a second FFT for speed)
        conv_trial = full_conv(a_trial, size1)
        trial_score = 2.0 * n * conv_trial.max()
        if trial_score < best_score:
            best_score = trial_score
            best_seq = a_trial
            print(f"[refine] improved to {best_score:.6f}")
        # otherwise keep the current best

    total_time = time.time() - start
    print(f"[done] iterations={it:,} best score={best_score:.6f} time={total_time:.1f}s")
    return best_seq.tolist()
