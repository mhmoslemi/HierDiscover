"""
Propose a low‑value step function for ``evaluate_sequence``.

The algorithm mixes a thorough parametric search (power‑law, exponential,
arcsine) with a fast projected gradient descent that targets the
max‑entry of the self‑convolution.  The gradient uses only the current
maximum (top_k = 1) which gives a very directed sub‑gradient.
A sequence of random restarts, occasional mass‑swaps and a final
polishing stage are run inside the given time budget (default 1000 s).
All vectors are kept on the probability simplex (non‑negative, sum = 1)
so that ``evaluate_sequence`` sees a valid input.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """Generate a normalised power‑law sequence a_i ∝ (i+1+offset)^{-α}."""
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, lam: float) -> np.ndarray:
    """Generate a normalised exponential decay a_i ∝ exp(-lam·i)."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-lam * idx)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _arcsine_sequence(n: int) -> np.ndarray:
    """Generate a normalised arcsine‑type distribution a_i ∝ 1/√(i·(n+1‑i))."""
    i = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(i * (n + 1 - i))
    s = a.sum()
    a /= s
    return a


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Full self‑convolution (length 2·n‑1) using a real FFT."""
    n = seq.shape[0]
    size = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    conv[conv < 0.0] = 0.0
    return conv


def _score_fft(seq: np.ndarray) -> float:
    """Objective 2·n·max(conv) for a probability vector (sum = 1)."""
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1} (Duchi et al. 2008)."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_step(seq: np.ndarray, step: float, top_k: int) -> np.ndarray:
    """
    One projected (sub‑)gradient step that decreases the largest
    entries of the self‑convolution.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)

    # indices of the `top_k` largest convolution values
    if top_k >= conv.size:
        top_idx = np.arange(conv.size)
    else:
        top_idx = np.argpartition(-conv, top_k - 1)[:top_k]

    grad = np.zeros_like(seq)
    for k in top_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1, dtype=int)
        grad[i_range] += seq[k - i_range]

    grad *= 2.0                    # true sub‑gradient of max(conv)
    grad -= grad.mean()            # keep Σx = 1 unchanged after the step
    new_seq = seq - step * grad
    new_seq = _project_simplex(new_seq)
    return new_seq


def _run_gradient(
    seq: np.ndarray,
    time_budget: float,
    *,
    max_iters: int = 10 ** 9,
    top_k: int = 1,
    step0: float = 0.5,
    decay: float = 0.9999,
    print_interval: int = 1000,
) -> tuple[np.ndarray, float]:
    """
    Projected (sub‑)gradient descent on the L∞‑norm of the self‑convolution.
    Returns the best sequence seen and its score.
    """
    start = time.time()
    best_seq = seq.copy()
    best_score = _score_fft(seq)
    step = step0
    it = 0
    while it < max_iters and (time.time() - start) < time_budget:
        it += 1
        seq = _gradient_step(seq, step, top_k)

        # occasional random mass exchange to shake out plateaus
        if it % 250 == 0:
            i, j = np.random.choice(seq.size, 2, replace=False)
            eps = min(step * seq[i], seq[i] * 0.2)
            seq[i] -= eps
            seq[j] += eps
            seq = _project_simplex(seq)

        cur_score = _score_fft(seq)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            if it % print_interval == 0:
                print(f"[improve] iter={it}, score={best_score:.6f}")

        # simple step‑size adaptation
        if cur_score > best_score * 1.001:
            step *= 0.5
        else:
            step = max(step * decay, 1e-9)

    return best_seq, best_score
