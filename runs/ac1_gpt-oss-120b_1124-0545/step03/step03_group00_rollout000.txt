"""
Non‑negative step‑function optimizer.

The method combines a large pool of analytically motivated seeds,
quickly discards the worst ones, and then refines the survivors with
a *smooth log‑sum‑exp* surrogate of the max‑convolution.  The surrogate
is minimised using SciPy’s L‑BFGS‑B (gradient supplied) on the
log‑softmax parametrisation, which converges in a few hundred gradient
evaluations.  A multistage schedule of increasing smoothing (β) yields
a coarse‑to‑fine optimisation.  The best solution (symmetrised to
respect the inherent symmetry of the problem) is returned as a plain
Python list of floats.

The code respects the 1000 s budget, checks the remaining time
periodically and aborts cleanly.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Global configuration used by the loss/grad callbacks
# ----------------------------------------------------------------------
_curr_n = None          # length of the current vector
_curr_beta = None       # smoothing parameter for the current optimisation


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (output length = len(x)+len(y)-1)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Seed families (all return probability vectors that sum to one)
# ----------------------------------------------------------------------
def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    b = r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    a = a + b
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


# ----------------------------------------------------------------------
# Fast evaluation of the exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray) -> float:
    """2·n·max(conv) for a probability vector a (sum = 1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Surrogate loss (smooth max) and its gradient
# ----------------------------------------------------------------------
def _loss_and_grad(v: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns (loss, gradient) for the current smoothing parameter β.
    The loss is a smoothed approximation of max(conv) and the gradient
    is w.r.t. the logits v (log‑softmax parametrisation).
    """
    # Use global parameters
    beta = _curr_beta
    n = _curr_n

    # probabilities on the simplex
    a = _softmax(v)

    # convolution a*a
    conv = _conv_fft(a, a)                 # length 2n‑1
    shift = np.max(conv)                   # numeric stabiliser

    # smoothed weights (soft‑max over convolution values)
    w = np.exp(beta * (conv - shift))
    w_sum = np.sum(w)
    if w_sum == 0.0:                       # guard against under‑flow
        w = np.full_like(w, 1.0 / w.size)
        w_sum = 1.0
    w /= w_sum

    # surrogate loss: shift + (1/β)·log Σ exp(β·(conv‑shift))
    loss = shift + (1.0 / beta) * np.log(w_sum)

    # gradient w.r.t. a (see analysis notes)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)            # length 3n‑2
    grad_a = 2.0 * full[n - 1: n - 1 + n]  # central n entries

    # gradient w.r.t. logits v (softmax Jacobian)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)

    return float(loss), grad_v.astype(np.float64)


# ----------------------------------------------------------------------
# L‑BFGS‑B optimiser for a given β
# ----------------------------------------------------------------------
def _lbfgs_optimize(v0: np.ndarray, beta: float, maxiter: int,
                    deadline: float) -> np.ndarray:
    """
    Runs L‑BFGS‑B on the smooth surrogate with the supplied β.
    Stops early if the global deadline is reached.
    Returns the final logits (not the probability vector).
    """
    global _curr_n, _curr_beta

    _curr_n = v0.size
    _curr_beta = beta

    # SciPy does not support a direct “time‑out”.  We bound the number of
    # iterations and abort manually via a callback that raises a custom
    # exception when the time limit is hit.
    class TimeOutException(RuntimeError):
        pass

    def _callback(xk):
        if time.time() > deadline:
            raise TimeOutException()

    try:
        res = optimize.minimize(
            fun=_loss_and_grad,
            x0=v0,
            method="L-BFGS-B",
            jac=True,
            callback=_callback,
            options=dict(maxiter=maxiter, disp=False)
        )
    except TimeOutException:
        # Return the best point known so far (SciPy stores it in `x`).
        return v0
    return res.x


# ----------------------------------------------------------------------
# Main entry point required by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence minimising ``evaluate_sequence``.
    Uses a multi‑stage L‑BFGS‑B optimiser on a smooth log‑sum‑exp surrogate.
    Returns the best sequence found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size (larger sizes give more freedom but cost more)
    # ------------------------------------------------------------------
    n = 8000
    if time.time() > deadline:
        # Extremely low‑time situation – fall back to a tiny problem
        n = 2000

    print(f"[main] problem size n = {n}")

    # ------------------------------------------------------------------
    # 1) Build a diversified set of seed probability vectors
    # ------------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # Geometric families
    for c in np.linspace(0.5, 5.0, 12):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # Power‑law families
    for p in np.linspace(0.0, 2.2, 12):
        seeds.append(_power(n, p))

    # Combined geometric × power families
    for c in np.linspace(0.5, 4.0, 8):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 8):
            seeds.append(_geom_power(n, r, p))

    # Symmetric geometric
    for c in np.linspace(0.5, 4.5, 8):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # Smooth sine window – solid baseline
    seeds.append(_sine_window(n))

    # Dirichlet draws (different concentration levels)
    for alpha in [0.2, 0.5, 0.8, 2.0, 5.0]:
        a = np.random.gamma(alpha, 1.0, size=n)
        s = a.sum()
        if s > 0:
            seeds.append(a / s)

    # Random soft‑max starts
    for _ in range(5):
        v_rand = np.random.randn(n)
        seeds.append(_softmax(v_rand))

    # If a previous construction exists, reuse it (helps continuity)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            a_prev = np.asarray(prev, dtype=np.float64)
            if a_prev.ndim == 1:
                if a_prev.shape[0] != n:
                    adj = np.zeros(n, dtype=np.float64)
                    m = min(a_prev.shape[0], n)
                    adj[:m] = a_prev[:m]
                    a_prev = adj
                if a_prev.sum() > 0:
                    seeds.append(a_prev / a_prev.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2) Quick scan using the exact objective, keep only the top few seeds
    # ------------------------------------------------------------------
    fast_vals = [_fast_objective(s) for s in seeds]
    # sort indices by objective (lower is better)
    sorted_idx = np.argsort(fast_vals)[:7]   # keep the 7 most promising
    top_seeds = [seeds[i] for i in sorted_idx]

    # ------------------------------------------------------------------
    # 3) Multi‑stage L‑BFGS‑B refinement
    # ------------------------------------------------------------------
    # Stages with increasing smoothing (β).  The final stage uses a large β
    # so the surrogate tightly approximates the true max.
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0]

    # Keep the best solution seen so far
    best_seq = None
    best_val = float("inf")

    for seed_id, a0 in enumerate(top_seeds):
        if time.time() > deadline:
            break

        print(f"[seed {seed_id}] starting refinement, initial fast value = {fast_vals[sorted_idx[seed_id]]:.6f}")

        # Initialise logits from the seed probability vector
        v = np.log(a0 + 1e-30)

        # Run the schedule
        for stage, beta in enumerate(beta_schedule):
            if time.time() > deadline:
                break

            # Estimate a reasonable number of L‑BFGS iterations for this stage.
            # Early stages (small β) benefit from more iterations;
            # later stages (large β) converge quickly.
            max_iter_stage = 150 if beta < 20 else 80

            remaining = deadline - time.time()
            # Make sure we have at least a small slack before the deadline.
            if remaining < 0.5:
                break

            print(f"[seed {seed_id}] stage {stage+1}/{len(beta_schedule)} β={beta:.1f} maxiter={max_iter_stage}")

            v = _lbfgs_optimize(v, beta=beta,
                                maxiter=max_iter_stage,
                                deadline=deadline)

            # Evaluate the current probability vector
            a = _softmax(v)

            # Enforce symmetry (the problem is symmetric w.r.t. reversal)
            a = (a + a[::-1]) * 0.5
            a /= a.sum()          # renormalise after symmetrisation

            cur_val = _fast_objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[seed {seed_id}] new global best = {cur_val:.6f}")

            # Small random jitter before the next stage to avoid getting stuck
            jitter = 0.01 * np.random.randn(n)
            v = np.log(a + 1e-30) + jitter

        # ------------------------------------------------------------------
        # Optional perturb‑restart after all stages (if time permits)
        # ------------------------------------------------------------------
        while time.time() < deadline - 0.5:
            # Perturb the logits a bit and re‑optimise quickly with a modest β
            pert = np.log(best_seq + 1e-30) + 0.03 * np.random.randn(n)
            pert = _lbfgs_optimize(pert, beta=20.0, maxiter=60, deadline=deadline)
            a_pert = _softmax(pert)
            a_pert = (a_pert + a_pert[::-1]) * 0.5
            a_pert /= a_pert.sum()
            cur_val = _fast_objective(a_pert)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a_pert.copy()
                print(f"[perturb] improved global best = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Final sanity checks and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform distribution (guaranteed to be feasible)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _fast_objective(best_seq)

    # Normalise once more to be safe (evaluate_sequence clips negatives &
    # rescales, but we keep the invariant sum=1)
    best_seq /= best_seq.sum()

    # Convert to plain Python list
    result = best_seq.tolist()
    print(f"[final] best objective (fast) = {best_val:.6f}")
    return result
