"""
Hybrid optimisation for the step‑function problem.

The routine works in three stages:

1. **Parameter sweep** –  a 1‑parameter family a_i ∝ (i+1+offset)^{-α}
   (α≈0.5) is searched on a grid (α,offset).  The best candidate
   (lowest exact objective) is kept as a warm‑start.

2. **Guided hill‑climbing** –  The current convolution `c = a*a` is
   maintained explicitly.  At each iteration the index `k` where `c`
   attains its maximum is identified.  The contribution of each entry
   to that peak is `a_i·a_{k-i}`; we move a small amount of mass from a
   large‑contribution index to a small‑contribution index.  The update
   of the convolution is performed analytically (O(n) work) rather
   than by a full FFT, allowing tens of thousands of cheap iterations.
   The best sequence seen is recorded.

3. **Periodic re‑normalisation** –  to avoid drift caused by the
   incremental updates, the full convolution is recomputed with FFT
   every few thousand iterations.

The whole search respects the 1000 s budget and returns a list that can
be fed directly to `evaluate_sequence`.  The algorithm is deterministic
for a given `seed` but includes stochastic moves to escape local minima.
"""

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using FFT (real‑to‑real)."""
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:size]
    return conv


def _power_sequence_offset(n: int, alpha: float, offset: float) -> np.ndarray:
    """
    Power‑law sequence a_i ∝ (i+1+offset)^{-α}, normalised to sum = 1.
    All entries are non‑negative.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _objective(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (sum is 1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * conv.max())


def _update_convolution(conv: np.ndarray, a: np.ndarray,
                        i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a new convolution array after moving `delta` mass from index i to j.
    The update is performed analytically (O(n) work) rather than via FFT.
    """
    n = a.shape[0]
    Nconv = 2 * n - 1
    conv_new = conv.copy()

    # contribution of the change in entry i (a_i <- a_i-delta)
    start_i = i
    len_i = min(n, Nconv - start_i)
    conv_new[start_i:start_i + len_i] -= 2.0 * delta * a[:len_i]

    # contribution of the change in entry j (a_j <- a_j+delta)
    start_j = j
    len_j = min(n, Nconv - start_j)
    conv_new[start_j:start_j + len_j] += 2.0 * delta * a[:len_j]

    # quadratic term delta^2
    ii = 2 * i
    if ii < Nconv:
        conv_new[ii] += delta * delta
    jj = 2 * j
    if jj < Nconv:
        conv_new[jj] += delta * delta
    ij = i + j
    if ij < Nconv:
        conv_new[ij] -= 2.0 * delta * delta

    return conv_new

    except Exception:
        # length 5000 works well for the time budget
        n = 5000
        a = _power_sequence_offset(n, alpha=0.5, offset=0.0)

    # ensure exact simplex
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    # ------------------------------------------------------------------
    # 2. Coarse grid search over (alpha, offset) – cheap but often improves
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_obj = _objective(best_a)
    best_alpha = 0.5
    best_offset = 0.0

    alphas = np.linspace(0.45, 0.55, 21)   # 0.45 … 0.55
    offsets = np.linspace(0.0, 3.0, 31)    # 0 … 3

    print(f"[grid] start objective = {best_obj:.6f}")

    for alpha in alphas:
        for off in offsets:
            if time.time() >= deadline:
                break
            cand = _power_sequence_offset(n, alpha, off)
            obj = _objective(cand)
            if obj < best_obj:
                best_obj = obj
                best_a = cand.copy()
                best_alpha = alpha
                best_offset = off
                print(f"[grid] new best: α={alpha:.3f}, off={off:.2f}, obj={obj:.6f}")

    # ------------------------------------------------------------------
    # 3. Guided hill‑climbing on the full vector
    # ------------------------------------------------------------------
    a = best_a.copy()
    conv = _conv_fft(a, a)                 # current full convolution
    cur_obj = best_obj
    temperature = 0.01                     # very low – we behave almost greedy
    temp_decay = 0.9999

    iteration = 0
    recompute_every = 2000                # recompute conv exactly to avoid drift

    while time.time() < deadline:
        iteration += 1

        # locate the peak of the convolution
        k_max = int(conv.argmax())

        # contributions of each a_i to that peak
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        jdx = k_max - idx
        contrib = a[idx] * a[jdx]

        # choose a source index – bias towards large contributions
        if contrib.sum() > 0 and np.random.rand() < 0.9:
            probs = contrib / contrib.sum()
            i = np.random.choice(idx, p=probs)
        else:
            i = np.random.randint(0, n)

        # choose a destination index – bias towards small contributions
        if np.random.rand() < 0.9:
            # low‑contribution region (use complement of probs)
            low = (contrib.max() - contrib) + 1e-12
            low_probs = low / low.sum()
            j = np.random.choice(idx, p=low_probs)
        else:
            j = np.random.randint(0, n)

        if i == j:
            continue

        # amount of mass to move – a fraction of the source entry
        max_move = min(a[i] * 0.5, a[i] * np.random.rand() * 0.1)
        if max_move <= 0.0:
            continue
        delta = max_move

        # --- incremental convolution update ---------------------------------
        conv_new = _update_convolution(conv, a, i, j, delta)
        new_max = conv_new.max()
        new_obj = float(2.0 * n * new_max)

        # greedy acceptance with tiny simulated‑annealing component
        accept = False
        if new_obj < cur_obj:
            accept = True
        else:
            # very low temperature – rarely accept worse moves
            if np.random.rand() < np.exp((cur_obj - new_obj) / temperature):
                accept = True

        if accept:
            # apply the move to a and conv
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            cur_obj = new_obj

            # keep the best overall solution
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
                print(f"[improve] iter={iteration}, obj={best_obj:.6f}")

        # anneal the temperature (practically stays near zero)
        temperature = max(temperature * temp_decay, 1e-8)

        # periodic exact recomputation to combat floating‑point drift
        if iteration % recompute_every == 0:
            conv = _conv_fft(a, a)
            cur_obj = _objective(a)

    # ------------------------------------------------------------------
    # 4. Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[final] best_obj={best_obj:.6f}, iterations={iteration}")
    # clip to the evaluation routine's safety limits (0‑1000)
    best_a = np.clip(best_a, 0.0, 1000.0)
    return best_a.tolist()
