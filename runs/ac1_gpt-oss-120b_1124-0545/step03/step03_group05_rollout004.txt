def project_simplex(v):
    # v is array, ensure non-negative after projection onto simplex sum=1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑value step‑function sequence.

Algorithm
---------
* Initialise a feasible probability vector `a` (non‑negative, sum = 1).  If a
  global variable `height_sequence_1` is supplied it is interpolated to the
  required length, otherwise a uniform random start is used.
* At each iteration compute the full linear convolution `b = a * a`.
* Approximate the non‑smooth objective `max(b)` by a soft‑max
  (log‑sum‑exp) with temperature `τ`.  The gradient of this smooth
  approximation is `∇F = 2·(a_rev ∗ w)` where `w` is the soft‑max weight
  vector and `a_rev` is `a` reversed.
* Use the Adam optimiser on the gradient of the smooth objective.
  After the Adam step the vector is projected onto the probability simplex
  by clipping to `≥0` and renormalising.
* With a small probability a cheap “mass‑transfer’’ perturbation is applied
  to escape flat plateaux.
* If no improvement is seen for a configurable amount of time, restart from
  a slightly perturbed copy of the current best sequence (or completely
  random if the best is empty) and reset the Adam moments.
* The loop runs until the given time budget (default 1000 s) is exhausted.
  The best sequence found (lowest `evaluate_sequence` value) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT. ``size`` must be ≥ len(u)+len(v)‑1.
    Returns the full convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size1: int) -> np.ndarray:
    """Linear convolution a*a (full length)."""
    return _conv_fft(a, a, size1)


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a random fraction of mass from one index to another.
    The amount transferred is at most `scale * a[i]`.
    """
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # clip and renormalise to stay on the simplex
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= s
    return a_new


def _clip_and_normalise(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex by simple clipping + renorm."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        return np.ones_like(v) / v.shape[0]
    return v / s


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step function within ``budget_s`` seconds.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        ``n`` – length of the sequence (default 6000).  Unused entries are ignored.

    Returns
    -------
    list[float]
        The best sequence found (list of Python floats).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # initialise a feasible distribution on the simplex
    # --------------------------------------------------------------
    try:
        seq0 = globals().get("height_sequence_1", None)
        if seq0 is not None:
            seq0 = np.asarray(seq0, dtype=float)
            if seq0.size != n:
                # linear interpolation to the required length
                x_old = np.linspace(0, 1, seq0.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                a = np.interp(x_new, x_old, seq0)
            else:
                a = seq0.copy()
        else:
            raise KeyError
    except Exception:
        a = np.random.rand(n)

    a = _clip_and_normalise(a)

    # --------------------------------------------------------------
    # pre‑compute FFT lengths (powers of two) – they stay constant
    # --------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)       # for a*a
    size2 = _next_pow2(3 * n - 2)       # for a_rev * weight

    # bookkeeping
    a_rev = a[::-1].copy()

    # --------------------------------------------------------------
    # optimiser state (Adam)
    # --------------------------------------------------------------
    m = np.zeros_like(a)               # first moment
    v = np.zeros_like(a)               # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    # --------------------------------------------------------------
    # hyper‑parameters
    # --------------------------------------------------------------
    alpha = 0.12                # base learning rate for Adam
    tau_factor = 10.0           # initial temperature factor (τ = factor * max_b)
    tau_decay = 0.9997          # exponential decay per iteration
    rand_prob = 0.07            # probability of a random mass‑transfer move
    rand_scale = 0.045          # magnitude of random transfer
    stagnation_seconds = 35.0   # restart if no improvement for this long
    last_improve = start

    # bookkeeping of the best solution seen
    conv_full = _full_convolution(a, size1)
    max_b = conv_full.max()
    cur_score = 2.0 * n * max_b / (a.sum() ** 2)
    best_score = cur_score
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # 1. current convolution and objective
        conv_full = _full_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b / (a.sum() ** 2)

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{t_iter:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # 2. decide the move
        if np.random.rand() < rand_prob:
            # cheap random perturbation
            a = _random_transfer(a, rand_scale)
        else:
            # ---- smooth‑max + Adam gradient step ----
            # temperature for the soft‑max
            tau = max(tau_factor * max_b, 1e-14)
            tau_factor *= tau_decay

            # soft‑max weight vector (scaled to sum = 1)
            weight = np.exp((conv_full - max_b) / tau)
            w_sum = weight.sum()
            if w_sum == 0.0:
                weight.fill(1.0 / weight.size)
            else:
                weight /= w_sum

            # gradient of the smooth objective:  ∇F = 2·(a_rev ∗ weight)
            grad_full = _conv_fft(a_rev, weight, size2)
            grad_center = grad_full[n - 1 : n - 1 + n]      # length‑n slice
            grad = 2.0 * grad_center                        # true gradient (to be minimised)

            # Adam update (gradient descent)
            t_iter += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)

            m_hat = m / (1 - beta1 ** t_iter)
            v_hat = v / (1 - beta2 ** t_iter)

            step = alpha * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = _clip_and_normalise(a)

        # keep the reversed copy up‑to‑date
        a_rev = a[::-1].copy()

        # 3. occasional restart if we are stuck
        if time.time() - last_improve > stagnation_seconds:
            # restart from a slightly perturbed version of the current best
            print(f"[{t_iter:7d}] restart due to stagnation")
            if best_seq is not None and best_seq.size == n:
                noise = 0.01 * np.random.rand(n)
                a = _clip_and_normalise(best_seq + noise)
            else:
                a = np.random.rand(n)
                a = _clip_and_normalise(a)
            a_rev = a[::-1].copy()
            m.fill(0.0)
            v.fill(0.0)
            tau_factor = 10.0
            last_improve = time.time()

    # --------------------------------------------------------------
    # final reporting
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={t_iter}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
