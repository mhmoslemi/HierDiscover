"""
Propose a step‑function (non‑negative sequence) with a low
evaluate_sequence score.

The procedure consists of three stages:

1. **Warm‑start** – use the supplied `height_sequence_1` if it exists,
   otherwise fall back to a simple power‑law (a_i ∝ (i+1+offset)^‑α)
   and an exponential decay (a_i ∝ exp(-λ i)).  A tiny grid search
   over the parameters (α, offset, λ) is performed; the best candidate
   of this cheap sweep becomes the seed for the next stage.

2. **Sub‑gradient descent on the max‑convolution** – the true objective
   (for a normalised sequence) is `2·n·max(conv)`.  Its sub‑gradient at the
   index `k` where the convolution attains the maximum is
   `∂/∂a_i max(conv) = 2·a_{k‑i}` (zero when the index is out of range).
   We repeatedly move the current vector opposite to this sub‑gradient,
   then Euclidean‑project back onto the probability simplex.  A small
   learning‑rate schedule and occasional random mass swaps help to escape
   plateaus.

3. **Finalisation** – the best vector observed throughout the run is
   returned as a plain Python list (the harness will call
   `evaluate_sequence` on it).

The whole routine respects the 1000 s budget, prints progress
information, and uses only NumPy (no external I/O or heavy libraries).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())  # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """Normalised power‑law a_i ∝ (i+1+offset)^{-α} (α>0)."""
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, lam: float) -> np.ndarray:
    """Normalised exponential decay a_i ∝ exp(-λ·i).  λ≥0."""
    if lam <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        idx = np.arange(n, dtype=float)
        a = np.exp(-lam * idx)
        a /= a.sum()
    return a


def _score(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) for a normalised sequence a (Σa=1)."""
    n = a.size
    conv = _conv_fft(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine runs at most ``budget_s`` seconds and returns the best
    sequence found (as a plain Python list).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Unused – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence (non‑negative, sum > 0.01) discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Initial sequence (use the supplied one if possible)
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        init = np.maximum(init, 0.0)
        if init.sum() < 1e-8:
            raise ValueError
        a = init / init.sum()
        n = a.size
        print(f"[load] Using supplied height_sequence_1, n={n}")
    except Exception:
        # fallback: 5 000‑element power‑law with α≈0.5
        n = 5000
        a = _power_sequence(n, alpha=0.5)
        print("[fallback] Using power‑law start, n=5000")

    # make sure we are on the simplex
    a = _project_onto_simplex(a)

    best_seq = a.copy()
    best_score = _score(best_seq)
    print(f"[init] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Light parametric sweep (power‑law + exponential)
    # ------------------------------------------------------------------
    # power‑law grid (α in [0.44,0.56], offset up to 10)
    alphas = np.linspace(0.44, 0.56, 13)       # step ≈0.01
    offsets = np.arange(0, 11, 1)             # 0…10
    for alpha in alphas:

            s = _score(cand)

    # exponential grid (λ in [0, 0.001] – larger λ would concentrate too much mass)
    lambdas = np.linspace(0.0, 0.0015, 16)     # very gentle decay
    for lam in lambdas:
        cand = _exp_sequence(n, lam=lam)

    # ------------------------------------------------------------------
    # 3. Sub‑gradient descent on max‑convolution
    # ------------------------------------------------------------------
    a = best_seq.copy()
    lr = 0.5                     # initial learning rate
    lr_decay = 0.99994
    min_lr = 1e-6
    iteration = 0
    momentum = np.zeros_like(a)

        conv = _conv_fft(a, a)

        grad = np.zeros_like(a)
        lo = max(0, max_idx - (n - 1))

        i_range = np.arange(lo, hi + 1)
        j_range = max_idx - i_range
        grad[i_range] = 2.0 * a[j_range]

        # (g) evaluate exact score
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] iter={iteration} score={cur_score:.6f}")

        # safety check – stop slightly before deadline to avoid overshoot
        if time.time() > deadline - 0.5:
            break

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f}, iterations={iteration}")
    return best_seq.tolist()
