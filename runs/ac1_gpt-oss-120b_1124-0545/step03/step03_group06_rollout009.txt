def _gradient_from_mask(seq, mask):
    """
    Compute gradient = 2 * correlation(mask, seq) on simplex.

    Parameters
    ----------
    seq: np.ndarray (n)
    mask: np.ndarray (2n-1) with values 0 or 1 (or real)

    Returns
    -------
    grad: np.ndarray (n)
    """
    n = seq.shape[0]
    size = _next_pow2(mask.size + seq.size - 1)
    fw = np.fft.rfft(mask, n=size)
    fseq = np.fft.rfft(seq, n=size)
    corr = np.fft.irfft(fw * np.conj(fseq), n=size)
    # extract slice starting at n-1 -> length n
    grad = 2.0 * corr[n-1 : n-1 + n]
    grad -= grad.mean()
    return grad

Fast FFT‑based projected gradient descent for the step‑function
minimisation problem.

The algorithm works in four phases:

1. **Coarse exponent search** – on a short vector (8 k) we scan a
   power‑law a_i ∝ (i+δ)^{-α} and keep the exponent that gives the

   * form a binary mask w and obtain the sub‑gradient  
     g = 2·corr(w, a)   (correlation of the mask with the current vector)
     via a second FFT,
   * make the step tangent to the simplex (zero‑mean) and project back
     onto the probability simplex,
   * slowly decay the step size and shrink `top_k`.  The FFT‑based
     gradient is O(n log n) and allows many thousands of iterations
     within the 1000 s budget.

4. **Random short restarts** – while time permits we launch a few
   quick runs from random power‑law seeds (different α,δ) and keep any
   improvement.

The final sequence is projected onto the simplex once more to guarantee
feasibility and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Normalised non‑negative sequence proportional to (i+1+offset)^{-α}
    for i=0,…,n‑1.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:          # very unlikely, but keep a safe fallback
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution (length 2·n‑1) using a real FFT.
    """
    n = seq.shape[0]
    size = _next_pow2(2 * n - 1)
    ft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(ft * ft, n=size)[: 2 * n - 1]
    conv[conv < 0.0] = 0.0                # eliminate tiny negatives
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}
    (Duchi et al. 2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(seq: np.ndarray) -> float:
    """Objective value for a *normalised* sequence (Σseq == 1)."""
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core gradient routine (FFT‑based sub‑gradient of the max‑convolution)
# ----------------------------------------------------------------------


def _gradient_step_fft(seq: np.ndarray,
                       top_k: int,
                       size_corr: int,
                       mask: np.ndarray,
                       ft_seq: np.ndarray) -> np.ndarray:
    """
    One projected sub‑gradient step.
    Parameters
    ----------
    seq      – current probability vector (sum = 1)
    top_k    – number of largest convolution entries to use
    size_corr– FFT length for the correlation (pre‑computed)
    mask     – reusable buffer of length 2·n‑1 for the binary mask
    ft_seq   – rfft of `seq` already computed with length `size_corr`
    Returns
    -------
    Updated probability vector (still on the simplex).
    """
    n = seq.shape[0]

    # ---- self‑convolution (single FFT, result already in ft_seq)
    conv_full = np.fft.irfft(ft_seq * ft_seq, n=size_corr)[: 2 * n - 1]

    # ---- build binary mask of the `top_k` largest entries
    if top_k >= conv_full.size:
        mask.fill(1.0)
    else:
        mask.fill(0.0)
        idx = np.argpartition(-conv_full, top_k - 1)[:top_k]
        mask[idx] = 1.0

    # ---- correlation w * a (again via FFT)
    ft_mask = np.fft.rfft(mask, n=size_corr)
    corr_full = np.fft.irfft(ft_mask * np.conj(ft_seq), n=size_corr)

    # gradient = 2·corr(w,a) for negative lags → reversed slice
    grad = 2.0 * corr_full[n - 1 :: -1][:n]

    # make it tangent to the affine hull Σx = 1
    grad -= grad.mean()

    # simple fixed step (the caller decides the step size)
    step = _gradient_step_fft.step_size
    new_seq = seq - step * grad
    new_seq = _project_simplex(new_seq)
    return new_seq, conv_full.max()


def _run_gradient(seq: np.ndarray,
                  time_budget: float,
                  *,
                  top_k_start: int = 200,
                  top_k_end: int = 20,
                  step0: float = 1.0,
                  decay: float = 0.9999) -> tuple[np.ndarray, float]:
    """
    Run the FFT‑based projected (sub)‑gradient descent until the time budget
    expires.  Returns the best sequence seen and its score.
    """
    n = seq.shape[0]
    size_corr = _next_pow2(3 * n - 2)
    mask = np.empty(2 * n - 1, dtype=float)

    best_seq = seq.copy()
    best_score = _score(seq)

    # expose step size as a mutable attribute for the helper
    _gradient_step_fft.step_size = step0
    top_k = top_k_start
    it = 0
    start = time.time()

    while time.time() - start < time_budget:
        it += 1

        # FFT of the current vector (used for both conv and correlation)
        ft_seq = np.fft.rfft(seq, n=size_corr)

        # one gradient step
        seq, cur_max = _gradient_step_fft(seq, top_k, size_corr,
                                          mask, ft_seq)
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] it={it}, score={best_score:.6f}")

        # step‑size decay
        _gradient_step_fft.step_size = max(_gradient_step_fft.step_size * decay,
                                          1e-8)

        # slowly shrink the active set
        if it % 2000 == 0 and top_k > top_k_end:
            top_k = max(top_k_end, int(top_k * 0.9))

        # occasional random mass exchange to help escape plateaus
        if it % 500 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(seq[i] * 0.3, _gradient_step_fft.step_size * seq[i])
            seq[i] -= eps
            seq[j] += eps
            seq = _project_simplex(seq)

        # early stop if we already beat the target
        if best_score <= 1.503:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine uses a power‑law start, then an FFT‑accelerated projected
    gradient descent with many thousands of iterations, followed by a few
    random restarts.  It returns before the supplied time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds for the unavoidable overhead before the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Length – a power of two gives optimal FFT performance.
    # ------------------------------------------------------------------
    n = 65536                     # 2^16, fits easily into memory

    # ------------------------------------------------------------------
    # 2. Coarse scan on a small proxy to obtain a good exponent α.
    # ------------------------------------------------------------------
    n_proxy = 8192
    alphas = np.linspace(0.45, 0.65, 41)   # step ≈0.005
    best_alpha = 0.53                     # fallback
    best_proxy_score = np.inf

    for alpha in alphas:
        if time.time() >= deadline:
            break
        seq0 = _power_sequence(n_proxy, alpha)
        score0 = _score(seq0)
        if score0 < best_proxy_score:
            best_proxy_score = score0
            best_alpha = alpha

    # ------------------------------------------------------------------
    # 3. Build the full‑length candidate from the chosen exponent.
    # ------------------------------------------------------------------
    seq = _power_sequence(n, best_alpha)
    # break perfect symmetry with a tiny noise
    seq += np.random.randn(n) * 1e-6
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()

    # ------------------------------------------------------------------
    # 4. Main long‑run gradient descent.
    # ------------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    best_seq, best_score = _run_gradient(seq,
                                         time_budget=remaining,
                                         top_k_start=200,
                                         top_k_end=20,
                                         step0=1.0,
                                         decay=0.9999)
    print(f"[phase‑1] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Short random restarts (if time permits).
    # ------------------------------------------------------------------
    while time.time() < deadline - 10.0:   # keep ~10 s for final polishing
        alpha = np.random.uniform(0.40, 0.70)
        offset = np.random.uniform(0.0, 5.0)
        seq_r = _power_sequence(n, alpha, offset)
        seq_r += np.random.randn(n) * 1e-6
        seq_r = np.maximum(seq_r, 0.0)
        seq_r /= seq_r.sum()

        # a quick descent with tighter parameters
        remaining = max(0.0, deadline - time.time())
        seq_r, score_r = _run_gradient(seq_r,
                                       time_budget=remaining,
                                       top_k_start=80,
                                       top_k_end=20,
                                       step0=0.5,
                                       decay=0.9999)
        if score_r < best_score:
            best_score = score_r
            best_seq = seq_r.copy()
            print(f"[restart] improved to {best_score:.6f}")

        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 6. Final polishing with a very small step size.
    # ------------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    if remaining > 0.0:
        best_seq, best_score = _run_gradient(best_seq,
                                             time_budget=remaining,
                                             top_k_start=20,
                                             top_k_end=20,
                                             step0=0.2,
                                             decay=0.99998)
        print(f"[final] best_score={best_score:.6f}")

    # guarantee feasibility
    best_seq = _project_simplex(best_seq)

    print(f"[return] final_score={best_score:.6f}, length={n}")
    return best_seq.tolist()
