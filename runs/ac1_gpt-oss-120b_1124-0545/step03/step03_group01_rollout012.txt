"""
Search for a non‑negative step‑function minimizing the
evaluate_sequence objective.

The algorithm proceeds as follows:

1. **Seed creation** – a modest collection of analytic families
   (uniform, geometric, power‑law, half‑binomial, inverse‑sqrt,
   arcsine, sine‑window) together with several random soft‑max vectors
   and an optional previously‑found construction.

2. **Adam optimisation of a smoothed max** – the sequence is
   parametrised by logits `v` (softmax(v) lives on the probability simplex).
   A log‑sum‑exp approximation of the maximum of the auto‑convolution is
   minimised with Adam.  The smoothing factor β is increased quickly,
   steering the optimisation toward the true max.

3. **Exact sub‑gradient refinement** – starting from the best Adam result a
   mirror‑descent (exponentiated‑gradient) update using the exact sub‑gradient
   of the max‑convolution is performed.  The learning rate decays as 1/√t.

4. **Light random local search** – a few stochastic “mass‑transfer”
   perturbations are applied to the current best sequence to escape possible
   shallow local minima.

All steps respect the global time budget (default 1000 s).  The best sequence
encountered is stored in ``height_sequence_1`` for reuse in subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Analytic seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx**p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _binomial(n: int) -> np.ndarray:
    """Coefficients of (1‑z)^‑½ truncated to length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a / a.sum()


def _inv_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    return a / a.sum()


def _arcsine(n: int) -> np.ndarray:
    """a_i ∝ 1/√((i+1)*(n-i)).  Symmetric distribution that resembles an arcsine law."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    return a / a.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max loss
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max of the auto‑convolution.
    Returns the best sequence found (as a numpy array) and its
    surrogate objective 2·n·max(conv) (the true objective equals this
    when the sequence is normalised to sum = 1).
    """
    v = np.log(seed + 1e-30)          # logits (unconstrained)

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    # smoothing schedule
    beta = 5.0
    beta_mul = 1.05

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)

        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # increase smoothing
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # back‑prop through the softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 12000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent style refinement using the true sub‑gradient of the
    non‑smoothed max.  The update is an exponentiated‑gradient step that
    stays on the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)               # stabilise numerically
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val

# Simple random local search (few stochastic mass transfers)
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Perform a cheap stochastic hill‑climbing: repeatedly move a tiny
    amount of mass from a randomly chosen index to another.
    Accept the move only if the objective improves.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))
    rng = np.random.default_rng(987654)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # pick a tiny fraction of the source mass
        epsilon = 1e-3 * rng.random() * a[i]
        if epsilon <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= epsilon
        a_new[j] += epsilon
        if a_new[i] < 0.0:                    # safeguard
            continue

        cur_val = 2.0 * n * np.max(_conv_fft(a_new, a_new))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_new.copy()
            a = a_new                              # continue from the improvement
            print(f"[Local] n={n:5d} it={it:5d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (including the half‑binomial family)
# ----------------------------------------------------------------------


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest set of diverse initial vectors of length n."""
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families (a few choices for the ratio r)
    for r in (0.9, 0.95, 0.99):
        seeds.append(_geom(n, r))

    # power‑law families
    for p in (0.3, 0.6, 1.0, 1.5):
        seeds.append(_power(n, p))

    # half‑binomial (coefficients of (1‑z)^‑½)
    seeds.append(_binomial(n))

    # inverse‑sqrt
    seeds.append(_inv_sqrt(n))

    # arcsine‑type distribution
    seeds.append(_arcsine(n))

    # a handful of random soft‑max vectors
    rng = np.random.default_rng(424242)
    for _ in range(6):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # reuse previously discovered construction if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget and returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # possibly reuse a previously computed construction
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – start small, move to larger if time permits
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")
        seeds = _make_seeds(n, previous)

        # ------------------------------------------------------------------
        # Quick scan of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            conv = _conv_fft(s, s)
            val = 2.0 * n * np.max(conv)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam optimisation
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(seeds):
            if time.time() > deadline:
                break
            seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline,
                                                max_iters=8000)
            if seq_adam is not None and val_adam < global_best_val:
                global_best_val = val_adam
                global_best_seq = seq_adam.copy()
                print(f"[overall] n={n} after Adam new best={val_adam:.6f}")

            # ------------------------------------------------------------------
            # Phase 2 – Exact sub‑gradient refinement
            # ------------------------------------------------------------------
            if seq_adam is not None:
                seq_ref, val_ref = _subgradient_refine(seq_adam, n, deadline,
                                                      max_iters=12000)
                if seq_ref is not None and val_ref < global_best_val:
                    global_best_val = val_ref
                    global_best_seq = seq_ref.copy()
                    print(f"[overall] after sub‑grad n={n} new best={val_ref:.6f}")

        # ------------------------------------------------------------------
        # Light random local search starting from the best of this length
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_loc, val_loc = _local_random_search(global_best_seq, n,
                                                    deadline, max_iters=5000)
            if seq_loc is not None and val_loc < global_best_val:
                global_best_val = val_loc
                global_best_seq = seq_loc.copy()
                print(f"[overall] after local search n={n} new best={val_loc:.6f}")

        # keep the current best as seed for the next larger length
        previous = global_best_seq.copy() if global_best_seq is not None else None

    # ----------------------------------------------------------------------
    # Fallback (should never be triggered)
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    # ----------------------------------------------------------------------
    # Final report (optional evaluation via the provided function)
    # ----------------------------------------------------------------------
    try:
        # evaluate_sequence may already be present in the global namespace
        obj = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest objective ≈ {obj:.6f} (length {len(global_best_seq)})")
    except Exception:
        print(f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})")

    # expose the best sequence for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
