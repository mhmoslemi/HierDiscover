"""
Propose a step‑function (non‑negative heights) that minimises
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm has two stages.

1. **Parametric grid search** – a dense search over a three‑parameter
   family a_i ∝ r^i / (i+1+shift)^p  (geometric × power‑law, with an
   optional shift).  Because the evaluation of the fast objective
   `2·n·max(conv)` is performed with an FFT it is cheap, allowing a
   thorough sweep (c controlling r = 1‑c/n, exponent p and shift).

2. **Local refinement** – the best sequence from the grid is taken as a
   start point for an Adam optimiser on the log‑softmax
   parametrisation.  The loss is a smoothed (log‑sum‑exp) approximation
   of the convolution maximum; the smoothing parameter β is increased
   geometrically so the optimiser finally targets the true max.

Both stages are time‑aware: they stop before the supplied deadline
(≈1000 s).  The final sequence is normalised to sum = 1 (the evaluation
function is scale‑invariant) and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (FFT convolution, softmax, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors via FFT.
    Output length is len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (logits → probability simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation
# ----------------------------------------------------------------------


def _refine_sequence(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 4.0,
    beta_mul: float = 1.025,
    lr0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the smoothed maximum of the
    a‑self‑convolution.  Returns the best sequence seen and its fast
    objective value 2·n·max(conv).
    """
    # logits (log‑softmax representation)
    v = np.log(a0 + 1e-30)
    m = np.zeros_like(v)          # first moment
    v2 = np.zeros_like(v)         # second moment

    best_seq = a0.copy()
    cur_conv = _conv_fft(a0, a0)
    best_val = 2.0 * n * np.max(cur_conv)

    beta = start_beta
    it = 0

    while True:
        it += 1
        if time.time() > deadline:
            break

        # Current simplex point
        a = _softmax(v)

        # Convolution and fast objective
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ---- smoothed‑max (log‑sum‑exp) weights ----
        beta *= beta_mul
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = np.sum(w)
        if w_sum == 0.0:          # numerical safety
            continue
        w /= w_sum

        # Gradient w.r.t. a for the smoothed loss
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length = 3·n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]      # central n entries

        # Gradient w.r.t. logits (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)                     # simple annealing
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Parametric family (geometric × power‑law with optional shift)
# ----------------------------------------------------------------------


def _geom_power_shift(n: int, c: float, p: float, shift: float) -> np.ndarray:
    """
    a_i ∝ r^i / (i+1+shift)^p   with r = 1 - c/n   (c > 0).
    Returns a normalised probability vector.
    """
    if c <= 0 or c >= n:
        return None
    r = 1.0 - c / n
    if r <= 0.0:
        return None
    i = np.arange(n, dtype=np.float64)
    # denominator: (i+1+shift)^p, shift can be 0
    denom = (i + 1.0 + shift) ** p
    a = (r ** i) / denom
    s = a.sum()
    if s == 0.0:
        return None
    return a / s
