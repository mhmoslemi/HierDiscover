def propose_candidate(seed=0, budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 5
    best_score = float('inf')
    best_seq = None

    def evaluate(a):
        conv = _conv_fft(a, a)
        return 2 * a.size * conv.max()
    # Try different lengths
    for n in [500, 1000, 2000, 4000, 8000]:
        for alpha in np.linspace(0.35, 0.85, 11):  # step 0.05
            for offset in [0, 0.5, 1, 2, 5]:
                idx = np.arange(1, n+1) + offset
                a = idx**(-alpha)
                a_sum = a.sum()
                if a_sum == 0:
                    continue
                a = a / a_sum
                score = evaluate(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[grid] n={n}, alpha={alpha}, offset={offset}, score={score:.6f}")
                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break
    # Fine-tune with gradient descent starting from best_seq
    if best_seq is not None:
        a = best_seq.copy()
        n = a.size
        lr = 0.1
        momentum = np.zeros_like(a)
        while time.time() < deadline:
            conv = _conv_fft(a, a)
            max_idx = np.argmax(conv)
            # compute subgradient: g_i = 2 * a[max_idx - i] if index valid else 0
            g = np.zeros_like(a)
            for i in range(n):
                j = max_idx - i
                if 0 <= j < n:
                    g[i] = 2 * a[j]
            # momentum
            momentum = 0.9 * momentum + g
            a = a - lr * momentum
            a = _project_onto_simplex(a)
            # reduce lr
            lr = max(lr * 0.9999, 1e-8)
            # evaluate
            score = evaluate(a)
            if score < best_score:

3. **Result** –  The best sequence found during the whole run is returned
   (as a plain python list).  All operations respect the 1000 s time budget
   and use only NumPy; no external files or network access are required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | x_i ≥ 0 , Σ x_i = 1 }.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become negative – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using the FFT.
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _score_from_vec(a: np.ndarray) -> float:
    """
    Compute the exact score used by ``evaluate_sequence`` for a vector that
    already satisfies Σ a_i = 1.  The formula reduces to 2·n·max(conv).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  A small safety margin is kept for the final
        return.
    kwargs : dict
        Unused – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence (non‑negative, Σ a_i > 0.01) discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # 0.  Load a possible external starting point (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        init = np.maximum(init, 0.0)
        if init.sum() > 0.0:
            init /= init.sum()
        else:
            raise ValueError
        best_seq = init.copy()
        best_score = _score_from_vec(best_seq)
        print(f"[init] loaded external sequence, score={best_score:.6f}")
    except Exception:
        # fallback to a simple power‑law (α=0.5) of length 5000
        n0 = 5000
        idx = np.arange(1, n0 + 1, dtype=float)
        a0 = idx ** (-0.5)
        a0 /= a0.sum()
        best_seq = a0.copy()
        best_score = _score_from_vec(best_seq)
        print(f"[init] default power‑law (n={n0}), score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Global grid search over (n, α, offset)
    # ------------------------------------------------------------------
    # candidate lengths – we try a few values, the asymptotic constant is
    # almost length‑independent, but small finite‑size effects can be useful.
    n_candidates = [2000, 4000, 5000, 8000, 12000]

    # exponent grid centred around the known optimum α≈0.5
    alpha_grid = np.linspace(0.40, 0.60, 13)          # step ≈0.016
    offset_grid = np.linspace(0.0, 8.0, 9)           # 0,1,2,...,8

    for n in n_candidates:
        if time.time() > deadline:
            break
        idx_base = np.arange(1, n + 1, dtype=float)   # 1‑based for convenience
        for alpha in alpha_grid:
            # pre‑compute the power for speed
            pow_part = idx_base ** (-alpha)
            for off in offset_grid:
                if time.time() > deadline:
                    break
                # apply offset: (i+off)^{-α}
                a = (idx_base + off) ** (-alpha)
                # normalise to sum = 1
                a_sum = a.sum()
                if a_sum == 0.0:
                    continue
                a /= a_sum
                # evaluate exact score
                cur_score = _score_from_vec(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[grid] n={n}, α={alpha:.3f}, off={off:.1f}, "
                          f"score={best_score:.6f}")
    # ------------------------------------------------------------------
    # 2. Sub‑gradient descent (projected) from the best candidate
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # hyper‑parameters for the sub‑gradient loop
    lr = 0.12            # initial learning rate
    lr_decay = 0.99994
    lr_min = 5e-8
    momentum = np.zeros_like(a)

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # 2a. current convolution and the index where the maximum occurs
        conv = _conv_fft(a, a)
        k_max = int(np.argmax(conv))

        # 2b. sub‑gradient of ``max(conv)`` i.e. 2·shifted copy of ``a``
        g = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        # slice of a that is needed (reversed order)
        rev = a[k_max - np.arange(lo, hi + 1)]
        g[lo:hi + 1] = 2.0 * rev

        # momentum (helps stability)
        momentum = 0.9 * momentum + g

        # gradient step + projection onto simplex
        a = a - lr * momentum
        a = _project_onto_simplex(a)

        # 2c. decay learning rate gradually
        lr = max(lr * lr_decay, lr_min)

        # occasional random mass swap to escape tiny plateaus
        if iteration % 150 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i] * 0.5, a[i] * lr)
            if delta > 0.0:
                a[i] -= delta
                a[j] += delta
                a = _project_onto_simplex(a)

        # 2d. evaluate the exact score and keep the best
        cur_score = _score_from_vec(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[gd] iter={iteration:,}, lr={lr:.2e}, score={best_score:.6f}")

        # early stop when learning rate is tiny and progress stalls
        if lr <= lr_min and iteration > 5000:
            break

    # ------------------------------------------------------------------
    # 3. Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f}, total_iters={iteration:,}")
    # Convert to plain python list as required by the harness
    return best_seq.tolist()
