"""
Propose a step‑function sequence with a small evaluation score.

The algorithm works in three stages:
1. **Generate a diverse set of starts** –  simple symmetric power‑law
   shapes a_i ∝ ((i+1)*(n-i))^{-p/2} (including the uniform case p=0)
   and a few random exponential vectors.  All candidates are scaled to the
   ℓ₁‑norm required by the objective (√(2n)).
2. **Fast projected (exponentiated) gradient descent** on the exact
   objective max(conv(a,a)).  The sub‑gradient of the active entry k* is
   g_i = 2·a_{k*‑i}; we use an exponentiated‑gradient update
   a_i ← a_i·exp(‑η·g_i) followed by a simplex projection.
3. **Simulated annealing** – mass transfers between two indices are
   proposed; the candidate is accepted if it decreases the maximum
   convolution, otherwise with probability exp(‑Δ/T).  Temperature decays
   exponentially until a tiny value.

The whole procedure respects the supplied time budget (default 1000 s) and
keeps the best sequence discovered.  The returned list can be fed directly
to ``evaluate_sequence``.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a non‑negative vector onto the simplex {w ≥ 0, Σ w = s}."""
    if s < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    # standard O(n log n) projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be equal
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (efficient length for FFT convolution)."""
    return 1 << ((2 * n - 1).bit_length())


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Return (max convolution value, index of the maximum) for the self‑convolution.
    Computed by FFT of length ``fft_len``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
def _evaluate_score(max_conv: float, n: int, target_sum: float) -> float:
    """
    Return the value that ``evaluate_sequence`` would produce for a vector
    with the given maximum self‑convolution (the ℓ₁‑norm is forced to
    ``target_sum``).
    """
    return 2.0 * n * max_conv / (target_sum * target_sum)

    ``evaluate_sequence``.  The routine builds a few analytically motivated
    starting points, runs a short exponentiated‑gradient descent, then
    finishes with simulated annealing.
    """
    rng = np.random.default_rng(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s
    # ------------------------------------------------------------------
    # problem size – we keep it moderate (FFT still cheap) but large enough
    # to let the lower bound be close to 1.
    n = 5000
    target_sum = math.sqrt(2.0 * n)              # ℓ₁‑norm required by the objective

    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 1️⃣  Build a portfolio of initial candidates
    candidates = []

    # uniform
    uni = np.full(n, 1.0)
    candidates.append(uni)

    # random exponentials (several seeds)
    for _ in range(3):
        cand = rng.exponential(scale=1.0, size=n)
        candidates.append(cand)

    # power‑law symmetric shapes a_i ∝ ((i+1)*(n-i))^{-p/2}

    i = np.arange(n, dtype=float)
    for p in p_values:
        shape = ((i + 1) * (n - i)) ** (-p / 2.0)
        candidates.append(shape)

    # mixtures of arcsine (p=1) with a uniform background

        arcsine = ((i + 1) * (n - i)) ** (-0.5)
        mixed = lam * arcsine + (1.0 - lam) * 1.0
        candidates.append(mixed)

    # ------------------------------------------------------------------
    # Normalise each candidate to the required ℓ₁‑norm and evaluate it.
    best_seq = None
    best_max = np.inf
    best_score = np.inf

    print("[INIT] n =", n, "target ℓ₁‑norm =", target_sum)

    for idx, cand in enumerate(candidates):
        # project onto the simplex to obtain the correct sum
        cand = _simplex_projection(cand, target_sum)

    gradient_deadline = start_time + budget_s * 0.30
    a = best_seq.copy()
    cur_max, cur_k = _max_convolution(a, fft_len)
    cur_score = _evaluate_score(cur_max, n, target_sum)

    # step‑size schedule
    eta0 = 0.4                      # initial learning rate (relative to total mass)
    it = 0
    last_improve = 0

    while time.time() < gradient_deadline:
        # gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, cur_k - (n - 1))
        i_end = min(cur_k, n - 1)
        idxs = np.arange(i_start, i_end + 1, dtype=int)
        # g_i = 2 * a_{k - i}
        g[idxs] = 2.0 * a[cur_k - idxs]

        # exponentiated‑gradient step
        eta = eta0 / math.sqrt(it + 1.0)
        a_new = a * np.exp(-eta * g)
        a_new = _simplex_projection(a_new, target_sum)

        # evaluate new point
        new_max, new_k = _max_convolution(a_new, fft_len)
        new_score = _evaluate_score(new_max, n, target_sum)

        if new_score < best_score:
            best_score = new_score
            best_max = new_max
            best_seq = a_new.copy()
            print(f"[EGD {it:6d}] new best score={best_score:.6f}  max_conv={best_max:.6f}")

        # keep the iterates
        a = a_new
        cur_max, cur_k = new_max, new_k
        it += 1
        # occasional tiny random perturbation to escape flat plateaus
        if it % 3000 == 0:
            src = rng.integers(n)
            if a[src] > 1e-12:
                dst = rng.integers(n)
                delta = a[src] * 0.02 * rng.random()
                a_pert = a.copy()
                a_pert[src] -= delta
                a_pert[dst] += delta
                a_pert = _simplex_projection(a_pert, target_sum)
                pert_max, _ = _max_convolution(a_pert, fft_len)
                pert_score = _evaluate_score(pert_max, n, target_sum)
                if pert_score < best_score:
                    best_score = pert_score
                    best_max = pert_max
                    best_seq = a_pert.copy()
                    print(f"[EGD {it:6d}] perturbation improved best to {best_score:.6f}")
                a = a_pert
                cur_max = pert_max
        # restart if we see no improvement for a while
        if it - last_improve > 8000:
            a = _simplex_projection(rng.exponential(scale=1.0, size=n), target_sum)
            cur_max, cur_k = _max_convolution(a, fft_len)
            last_improve = it
            print(f"[EGD {it:6d}] restart due to stagnation")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated annealing – use the remaining budget.
    cur_max, cur_k = _max_convolution(best_seq, fft_len)
    cur_score = _evaluate_score(cur_max, n, target_sum)
    a = best_seq.copy()
    best_seq = a.copy()
    best_max = cur_max
    best_score = cur_score

    print(f"[SA-START] initial max_conv={best_max:.6f}  score={best_score:.6f}")

    # temperature schedule
    T0 = best_max * 0.25            # start temperature as a fraction of the objective
    Tmin = best_max * 1e-6
    sa_start = time.time()
    step_counter = 0

    while time.time() < deadline:
        elapsed = time.time() - sa_start
        frac = elapsed / (deadline - sa_start + 1e-12)
        T = T0 * ((Tmin / T0) ** frac)          # exponential cooling

        # propose a mass transfer
        i_src = rng.integers(n)
        if a[i_src] <= 1e-12:
            continue
        i_dst = rng.integers(n)
        if i_src == i_dst:
            continue
        # move up to 15 % of the source mass
        delta = a[i_src] * 0.15 * rng.random()
        a_cand = a.copy()
        a_cand[i_src] -= delta
        a_cand[i_dst] += delta

        cand_max, _ = _max_convolution(a_cand, fft_len)
        cand_score = _evaluate_score(cand_max, n, target_sum)

        if cand_score < cur_score:
            # accept improvement
            a = a_cand
            cur_score = cand_score
            cur_max = cand_max
            if cand_score < best_score:
                best_score = cand_score
                best_max = cand_max
                best_seq = a_cand.copy()
                print(f"[SA {step_counter:7d}] new best score={best_score:.6f}  max_conv={best_max:.6f}")
        else:
            # Metropolis acceptance
            prob = math.exp(-(cand_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                a = a_cand
                cur_score = cand_score
                cur_max = cand_max

        step_counter += 1

    print(f"[DONE] best score={best_score:.6f}  max_conv={best_max:.6f} "
          f"after {it} exponentiated‑gradient iters and {step_counter} SA steps")
    return best_seq.tolist()
