def _gradient_and_max(a):
    conv = _conv_fft(a, a)
    max_idx = int(conv.argmax())
    n = a.shape[0]
    g = np.zeros_like(a)
    if max_idx < n:
        g[:max_idx+1] = 2 * a[max_idx::-1]
    else:
        start = max_idx - (n - 1)
        rev_seg = a[(n-1): (start-1): -1]
        g[start:] = 2 * rev_seg
    return g, max_idx, conv

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    # list of lengths and alphas and offsets to try
    n_list = [1024, 2048, 4096, 5000, 8192]
    alpha_list = [0.3, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7]
    offset_list = [0.0, 0.5, 1.0, 2.0]
    best_score = float('inf')
    best_seq = None

    # initial candidates
    for n in n_list:
        for alpha in alpha_list:
            for offset in offset_list:
                if time.time() >= deadline:
                    break
                a = _power_sequence(n, alpha, offset)
                # evaluate
                conv = _conv_fft(a, a)
                max_conv = conv.max()
                cur_score = 2.0 * n * max_conv
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[grid] n={n}, alpha={alpha}, offset={offset}, score={cur_score:.6f}")

    # if global height_sequence_1 exists, incorporate it
    try:
        hs = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if hs.ndim == 1:
            hs = np.maximum(hs, 0.0)
            hs_sum = hs.sum()
            if hs_sum > 0.0:
                hs = hs / hs_sum
                n = hs.size
                conv = _conv_fft(hs, hs)
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = hs.copy()
                    print(f"[global] n={n}, score={cur_score:.6f}")
    except Exception:
        pass

    # if no best seq found (unlikely) fallback uniform with moderate n
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0/n)

    # Main refinement: subgradient descent
    a = best_seq.copy()
    n = a.shape[0]
    lr = 1.0
    lr_decay = 0.9995
    min_lr = 1e-8
    iter = 0
    while time.time() < deadline:
        iter += 1
        g, k, conv = _gradient_and_max(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        # update best if improved
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[refine] iter={iter}, k={k}, score={cur_score:.6f}")
        # gradient step
        a = a - lr * g
        a = _project_onto_simplex(a)
        # decay learning rate
        lr = max(lr * lr_decay, min_lr)

        # occasional random perturbation
        if iter % 1000 == 0:
            # random mass swap
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i]*0.5, a[i]*lr*10)
            if delta > 0:
                a[i] -= delta
                a[j] += delta
                a = _project_onto_simplex(a)

    # final local search: hill climbing
    seq = best_seq.copy()
    seq_score = best_score
    attempts = 0
    while time.time() < deadline and attempts < 50000:
        i, j = np.random.choice(n, 2, replace=False)
        delta = np.random.uniform(0, 0.01) * seq[i]  # up to 1%
        if delta <= 0:
            continue
        seq[i] -= delta
        seq[j] += delta
        # ensure non-negative - already because delta <= seq[i]
        conv = _conv_fft(seq, seq)
        new_score = 2.0 * n * conv.max()
        if new_score < seq_score:
            seq_score = new_score
            best_seq = seq.copy()
            best_score = new_score
            print(f"[hill] attempt={attempts}, score={new_score:.6f}")
        else:
            # revert
            seq[i] += delta
            seq[j] -= delta
        attempts += 1

    print(f"[final] best_score={best_score:.6f}, n={n}, total_iters={iter}")
    return best_seq.tolist()

"""
Non‑negative step‑function optimisation.

The code searches for a sequence `a` (a probability vector, i.e. a_i ≥ 0,
∑a_i = 1) that minimises the objective used by ``evaluate_sequence``:

    2·n·max(conv(a,a))

where ``conv`` is the full discrete convolution.
The strategy consists of three stages:

1. **Coarse grid search** over a few lengths, power‑law exponents and
   optional offsets.  The best candidate from this cheap sweep is used
   as a starting point.
2. **Sub‑gradient descent** on the exact infinity‑norm of the
   convolution.  The gradient of the maximal entry is computed analytically,
   the iterate is projected back onto the probability simplex, and a
   slowly decaying learning rate is employed.
3. **Hill‑climbing refinement** – random mass transfers between two
   coordinates are accepted only if they improve the exact objective.
   This helps to escape tiny plateaus that the sub‑gradient method may
   miss.

All heavy work (convolution, projection) uses NumPy and FFTs, and the
routine respects the 1000 s time budget by checking a deadline in every
loop.  The best sequence found is returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i ≥ 0 , Σ x_i = 1 }.
    Implementation of Duchi et al., ICML 2008.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real 1‑D arrays using FFT.
    Length of the result is len(x)+len(y)-1.
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())      # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    # tiny negative values may appear because of round‑off – clamp to 0
    np.maximum(conv, 0, out=conv)
    return conv


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Normalised power‑law sequence   a_i ∝ (i+1+offset)^{-α}.
    Returned vector sums to 1 and is non‑negative.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _gradient_and_max(a: np.ndarray):
    """
    Compute the full self‑convolution of ``a``, locate the index of the
    maximal entry and return the exact sub‑gradient of that maximal entry.
    Gradient formula:
        g_j = 2·a_{k-j}    if 0 ≤ k-j < n, else 0
    (where ``k`` is the arg‑max of the convolution).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    k = int(conv.argmax())
    g = np.zeros_like(a)

    if k < n:
        # indices 0..k receive a reversed prefix of length k+1
        g[: k + 1] = 2.0 * a[k :: -1]
    else:
        start = k - (n - 1)                     # first index that receives a contribution
        # reversed segment a[n-1] ... a[start]
        rev_seg = a[(n - 1) : (start - 1) : -1]
        g[start:] = 2.0 * rev_seg
    return g, k, conv


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Runs for at most ``budget_s`` seconds and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # keep a small safety margin

    # ------------------------------------------------------------------
    # 1. Coarse grid search (very cheap)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    n_candidates = [1024, 2048, 4096, 5000, 8192]
    alpha_candidates = [0.30, 0.40, 0.45, 0.50, 0.55, 0.60, 0.70]
    offset_candidates = [0.0, 0.5, 1.0, 2.0]

    for n in n_candidates:
        for alpha in alpha_candidates:
            for offset in offset_candidates:
                if time.time() >= deadline:
                    break
                a = _power_sequence(n, alpha, offset)
                conv = _conv_fft(a, a)
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[grid] n={n:5d}, α={alpha:5.2f}, off={offset:3.1f}, "
                          f"score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate any globally supplied construction (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        hs = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        if hs.ndim == 1:
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 0.0:
                hs /= s
                n = hs.size
                conv = _conv_fft(hs, hs)
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = hs.copy()
                    print(f"[global] n={n:5d}, score={cur_score:.6f}")
    except Exception:
        pass

    # If nothing succeeded (unlikely) fall back to a modest uniform vector
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 3. Sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    lr = 1.0                # initial step size
    lr_decay = 0.9995
    min_lr = 1e-8
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # (a) compute gradient of the current maximal entry
        g, k, conv = _gradient_and_max(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        # (b) keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[refine] iter={iteration:6d}, max_idx={k:5d}, "
                  f"score={cur_score:.6f}")

        # (c) gradient step with projection onto the simplex
        a = a - lr * g
        a = _project_onto_simplex(a)

        # (d) learning‑rate schedule
        lr = max(lr * lr_decay, min_lr)

        # (e) occasional random mass exchange to shake the system
        if iteration % 1000 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i] * 0.5, a[i] * lr * 10.0)
            if delta > 0.0:
                a[i] -= delta
                a[j] += delta
                a = _project_onto_simplex(a)

    # ------------------------------------------------------------------
    # 4. Simple hill‑climbing (random local moves)
    # ------------------------------------------------------------------
    seq = best_seq.copy()
    seq_score = best_score
    attempts = 0
    while time.time() < deadline and attempts < 50000:
        i, j = np.random.choice(n, 2, replace=False)
        # move a tiny fraction of the mass from i to j (up to 1 % of a_i)
        delta = np.random.uniform(0.0, 0.01) * seq[i]
        if delta <= 0.0:
            continue
        seq[i] -= delta
        seq[j] += delta

        conv = _conv_fft(seq, seq)
        new_score = 2.0 * n * conv.max()
        if new_score < seq_score:
            seq_score = new_score
            best_seq = seq.copy()
            best_score = new_score
            print(f"[hill] attempt={attempts:5d}, score={new_score:.6f}")
        else:
            # revert the move
            seq[i] += delta
            seq[j] -= delta

        attempts += 1

    # ------------------------------------------------------------------
    # 5. Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f}, length={n}, "
          f"total_iters={iteration}, hill_steps={attempts}")
    return best_seq.tolist()
