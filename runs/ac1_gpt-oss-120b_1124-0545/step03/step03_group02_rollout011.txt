"""
Propose a candidate step function using a parametric beta-family search
followed by a short Adam‑style soft‑max refinement.

The algorithm:
1. Construct a set of symmetric probability vectors a_i ∝ (i+½)^{α‑1}
   (1‑i‑½)^{α‑1} (i.e. a discretised Beta(α,α) distribution) for a
   range of α around 0.5 (the classic arcsine case).
2. For each α optionally mix the beta shape with a uniform distribution:
   a = (1‑λ)·beta_α + λ·uniform.  Both α and λ are searched on a small grid.
3. Keep the combination with the smallest exact score
   `E = 2·n·max(conv(a,a))`.
4. Starting from the best vector run a few hundred Adam updates on the
   smoothed objective (soft‑max of the convolution) while staying on the
   simplex.  This cheap local search removes the remaining noise.
5. Return the best sequence found before the time budget expires.
"""

import time
import numpy as np

def arcsine_sequence(n: int) -> np.ndarray:
    """Return the classic arcsine shape (Beta(0.5,0.5)) normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n)
    return a / a_sum

def beta_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Return a symmetric Beta(α,α) shape on a grid of n points,
    normalised so that the sum equals 1.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # to avoid NaN when alpha < 1, we rely on np.power which handles positive bases.
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    total = a.sum()
    if total == 0:
        # fallback to uniform
        return np.full(n, 1.0 / n)
    return a / total

def uniform_sequence(n: int) -> np.ndarray:
    """Uniform distribution on n points (sum = 1)."""
    return np.full(n, 1.0 / n)

def exact_score(seq: np.ndarray, n: int) -> float:
    """
    Compute the exact evaluation metric for a probability vector.
    (The evaluation function scales as 2·n·max(conv(seq,seq)).)
    """
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())

def mixed_sequence(beta_seq: np.ndarray, uniform_seq: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination (1‑λ)·beta + λ·uniform, already normalised."""
    a = (1.0 - lam) * beta_seq + lam * uniform_seq
    # re‑normalise to protect from rounding errors
    total = a.sum()
    if total == 0:
        return uniform_seq.copy()
    return a / total

def softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    v_shift = v - np.max(v)
    exp_v = np.exp(v_shift)
    sum_exp = exp_v.sum()
    if sum_exp == 0:
        # avoid division by zero; return uniform
        return np.full_like(v, 1.0 / v.shape[0])
    return exp_v / sum_exp

def gradient_refinement(initial_seq: np.ndarray,
                        n: int,
                        time_deadline: float) -> np.ndarray:
    """
    A short Adam‑style optimisation that minimises the smoothed
    maximum of the convolution while staying on the simplex.
    The routine stops when the current time exceeds `time_deadline`.
    """
    # Parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # schedule parameters (chosen experimentally)
    lr_start, lr_end = 0.08, 0.001
    temp_start, temp_end = 20.0, 300.0

    # Initialise log‑coordinates
    v = np.log(initial_seq + 1e-12)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    # Pre‑compute FFT length (full convolution length = 2n‑1, we need 3n‑2 for intermediate steps)
    fft_len = 1 << ((3 * n - 2).bit_length())

    best_seq = initial_seq.copy()
    best_score = exact_score(best_seq, n)

    it = 0
    while time.time() < time_deadline:
        # progress fraction for schedule
        elapsed = time.time() - (time_deadline - (time_deadline - time.time()))  # dummy placeholder (unused)
        # Instead compute fraction from start of refinement
        # we approximate by dividing iteration count by a large constant
        p = min(1.0, it / 5000.0)

        # Temperature and learning‑rate schedules (linear)
        temp = temp_start * (1.0 - p) + temp_end * p
        lr = lr_start * (1.0 - p) + lr_end * p

        # Current probability vector
        a = softmax(v)               # Σa = 1, a ≥ 0

        # Full convolution a∗a (real FFT)
        fa = np.fft.fft(a, fft_len)
        conv_aa = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        # Soft‑max weighting of convolution entries (smooth max)
        z = temp * conv_aa
        z_max = np.max(z)
        w = np.exp(z - z_max)
        w_sum = w.sum()
        if w_sum == 0:
            w = uniform_sequence(2 * n - 1)
        else:
            w = w / w_sum

        # FFT of weight vector (zero‑padded to fft_len)
        fw = np.fft.fft(w, fft_len)

        # g2 = (w * a) – first n entries of the convolution
        conv_wa = np.real(np.fft.ifft(fw * fa))[: 3 * n - 2]
        g2 = conv_wa[:n]

        # g1 = correlation w ⋆ a (valid part)
        corr_full = np.real(np.fft.ifft(np.conj(fw) * fa))[: 3 * n - 2]
        g1 = corr_full[:n]

        # Gradient of the smoothed objective with respect to a
        g = 2.0 * n * (g1 + g2)

        # Chain rule for the soft‑max parametrisation
        dot_ag = np.dot(a, g)
        grad_v = a * (g - dot_ag)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # Periodic exact evaluation (every few hundred steps)
        if it % 400 == 0:
            cur_score = exact_score(softmax(v), n)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = softmax(v).copy()
                print(f"[refine] iter {it:05d} new best score = {best_score:.6f}")

    return best_seq

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation function.  The routine respects a time budget (default
    1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = int(kwargs.get('n', 2000))

    # ------------------------------------------------------------------
    # 1.  Prepare baseline candidates (arcsine, uniform and Beta families)
    # ------------------------------------------------------------------
    uniform = uniform_sequence(n)
    base_arcsine = arcsine_sequence(n)

    best_seq = base_arcsine.copy()
    best_score = exact_score(best_seq, n)
    print(f"[init] arcsine score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Scan convex mixtures of beta shapes and the uniform vector.
    # ------------------------------------------------------------------
    # Scan over α (Beta shape) and λ (mixing weight).  The grid is chosen
    # to be dense enough to discover improvements while remaining cheap.
    alphas = np.linspace(0.30, 0.70, 81)   # step ≈0.005
    lambdas = np.linspace(0.0, 0.30, 31)   # step ≈0.01

    for lam in lambdas:
        # early exit if time is exhausted
        if time.time() >= deadline:

        for alpha in alphas:
            beta_seq = beta_sequence(n, alpha)
            cand = mixed_sequence(beta_seq, uniform, lam)
            cur_score = exact_score(cand, n)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cand.copy()
                print(f"[grid] α={alpha:.3f} λ={lam:.3f} → score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Optional local refinement using Adam on the smoothed max.
    # ------------------------------------------------------------------

    time_left = deadline - time.time()

        # enforce the minimal sum required by evaluate_sequence
        best_seq += (0.01 - total) / n
        total = best_seq.sum()
    best_seq = best_seq / total   # keep sum = 1 (any scale works)

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, time = {total_time:.1f}s")
    return best_seq.tolist()

```python
"""
Propose a sequence using a parametric Beta(α,α) family with a
shifted grid, followed by a quick Adam‑style refinement on a
soft‑max approximation of the max‑convolution.  The routine respects
the 1000 s budget and returns the best vector found.
"""

import time
import numpy as np

def uniform_sequence(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

def arcsine_sequence(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a_sum = a.sum()
    if a_sum == 0:
        return uniform_sequence(n)
    return a / a_sum

def beta_shifted_sequence(n: int, alpha: float, delta: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) shape on the shifted grid
    x_i = (i + delta) / (n - 1 + 2*delta),  i = 0,…,n‑1.
    The result is normalised to sum = 1.
    """
    x = (np.arange(n, dtype=np.float64) + delta) / (n - 1 + 2.0 * delta)
    # protect against 0^neg by clipping a tiny epsilon
    eps = 1e-18
    x = np.clip(x, eps, 1 - eps)
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    total = a.sum()
    if total == 0:
        return uniform_sequence(n)
    return a / total

def exact_score(seq: np.ndarray, n: int) -> float:
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())

def softmax(v: np.ndarray) -> np.ndarray:
    v_shift = v - np.max(v)
    exp_v = np.exp(v_shift)
    sum_exp = exp_v.sum()
    if sum_exp == 0:
        return np.full_like(v, 1.0 / v.shape[0])
    return exp_v / sum_exp

def gradient_refinement(initial_seq: np.ndarray,
                        n: int,
                        deadline: float) -> np.ndarray:
    """
    Adam optimiser on a smooth surrogate of the max‑convolution.
    Stops when the wall‑clock time reaches ``deadline``.
    """
    start = time.time()
    v = np.log(initial_seq + 1e-12)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_start, lr_end = 0.08, 0.0005
    temp_start, temp_end = 20.0, 300.0

    fft_len = 1 << ((3 * n - 2).bit_length())

    best_seq = softmax(v).copy()
    best_sc = exact_score(best_seq, n)

    it = 0
    while time.time() < deadline:
        # Normalised progress for schedule
        prog = (time.time() - start) / (deadline - start)
        prog = min(1.0, prog)

        temp = temp_start * (1.0 - prog) + temp_end * prog
        lr = lr_start * (1.0 - prog) + lr_end * prog

        a = softmax(v)          # probability vector

        # Full convolution a*a (needed for the smooth‑max weights)
        fa = np.fft.fft(a, fft_len)
        conv_aa = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        # Soft‑max weighting of the convolution entries
        z = temp * conv_aa
        zmax = np.max(z)
        w = np.exp(z - zmax)
        w_sum = w.sum()
        if w_sum == 0:
            w = uniform_sequence(2 * n - 1)
        else:
            w = w / w_sum

        # FFT of weight vector (zero‑padded)
        fw = np.fft.fft(w, fft_len)

        # g2 = (w * a) – first n entries of the linear convolution
        conv_wa = np.real(np.fft.ifft(fw * fa))[: 3 * n - 2]
        g2 = conv_wa[:n]

        # g1 = correlation w ⋆ a (valid part)
        corr_full = np.real(np.fft.ifft(np.conj(fw) * fa))[: 3 * n - 2]
        g1 = corr_full[:n]

        # Gradient w.r.t. a
        g = 2.0 * n * (g1 + g2)

        # Gradient w.r.t. the log‑variables (soft‑max chain rule)
        dot_ag = np.dot(a, g)
        grad_v = a * (g - dot_ag)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # Occasionally evaluate the exact objective
        if it % 400 == 0:
            cur = softmax(v)
            sc = exact_score(cur, n)
            if sc < best_sc:
                best_sc = sc
                best_seq = cur.copy()
                print(f"[refine] iter {it:06d} new best score = {best_sc:.6f}")

    return best_seq

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation function.  The routine respects a time budget (default
    1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = int(kwargs.get('n', 2000))

    # --------------------------------------------------------------
    # 0.  Try to reuse a previously discovered sequence, if available.
    # --------------------------------------------------------------
    global_vars = globals()
    if "height_sequence_1" in global_vars:
        seq0 = np.array(global_vars["height_sequence_1"], dtype=float)
        if seq0.size == n and np.all(seq0 >= 0):
            seq0 = seq0 / seq0.sum()
            best_seq = seq0.copy()
            best_sc = exact_score(best_seq, n)
            print(f"[init] loaded previous sequence, score = {best_sc:.6f}")
        else:
            # fall back to arcsine
            best_seq = arcsine_sequence(n)
            best_sc = exact_score(best_seq, n)
            print(f"[init] fallback arcsine, score = {best_sc:.6f}")
    else:
        best_seq = arcsine_sequence(n)
        best_sc = exact_score(best_seq, n)
        print(f"[init] arcsine, score = {best_sc:.6f}")

    # ----------------------------------------------------------------
    # 1.  Exhaustive grid over (α,δ) – a shifted Beta(α,α) family.
    # ----------------------------------------------------------------
    alphas = np.linspace(0.35, 0.65, 31)   # step ≈ 0.01
    deltas = np.linspace(0.2, 0.9, 15)    # step ≈ 0.05
    for delta in deltas:
        if time.time() >= deadline:
            break
        for alpha in alphas:
            cand = beta_shifted_sequence(n, alpha, delta)
            sc = exact_score(cand, n)
            if sc < best_sc:
                best_sc = sc
                best_seq = cand.copy()
                print(f"[grid] α={alpha:.3f} δ={delta:.3f} → score={sc:.6f}")

    # ----------------------------------------------------------------
    # 2.  A short Adam‑style refinement on the soft‑max surrogate.
    # ----------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        refine_deadline = time.time() + time_left - 2.0   # 2 s safety
        refined = gradient_refinement(best_seq, n, refine_deadline)
        refined_sc = exact_score(refined, n)
        if refined_sc < best_sc:
            best_sc = refined_sc
            best_seq = refined.copy()

"""
Search for a step‑function that minimises

    E(a) = 2·n·max(conv(a,a)) / (sum(a)**2),

by exploring a shifted Beta(α,α) family and a short Adam‑style
refinement on a smooth approximation of the max‑convolution.
The routine respects a 1000 s time budget and returns the best sequence
found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def uniform_sequence(n: int) -> np.ndarray:
    """Uniform distribution on n points (sum = 1)."""
    return np.full(n, 1.0 / n)


def arcsine_sequence(n: int) -> np.ndarray:
    """
    Classic arcsine shape (Beta(0.5,0.5) on a uniform grid).
    Returns a non‑negative vector whose entries sum to 1.
    """
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    total = a.sum()
    if total == 0:
        return uniform_sequence(n)
    return a / total


def beta_shifted_sequence(n: int, alpha: float, delta: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on a shifted grid

        x_i = (i + delta) / (n - 1 + 2·delta),

    i = 0 … n‑1.  The resulting vector is normalised to sum = 1.
    """
    # grid points
    x = (np.arange(n, dtype=np.float64) + delta) / (n - 1 + 2.0 * delta)
    eps = 1e-18
    x = np.clip(x, eps, 1.0 - eps)      # avoid 0⁻ᵃ
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    total = a.sum()
    if total == 0:
        return uniform_sequence(n)
    return a / total


def exact_score(seq: np.ndarray, n: int) -> float:
    """
    Exact evaluation used during the search.
    For a probability vector (sum=1) the objective reduces to 2·n·max(conv).
    """
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    v_shift = v - np.max(v)
    e = np.exp(v_shift)
    s = e.sum()
    if s == 0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def gradient_refinement(initial_seq: np.ndarray,
                        n: int,
                        deadline: float) -> np.ndarray:
    """
    Adam optimiser on a smooth surrogate of the max‑convolution.
    Runs until the wall‑clock time reaches ``deadline``.
    Returns the best probability vector seen.
    """
    start = time.time()
    v = np.log(initial_seq + 1e-12)          # log‑parametrisation (soft‑max)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr_start, lr_end = 0.08, 0.0005
    temp_start, temp_end = 20.0, 300.0

    # FFT length enough for convolution of length 3·n‑2
    fft_len = 1 << ((3 * n - 2).bit_length())

    best_seq = softmax(v).copy()
    best_sc = exact_score(best_seq, n)

    it = 0
    while time.time() < deadline:
        # Normalised progress for schedules
        prog = (time.time() - start) / (deadline - start)
        prog = min(1.0, prog)

        # Linear schedules
        temp = temp_start * (1.0 - prog) + temp_end * prog
        lr = lr_start * (1.0 - prog) + lr_end * prog

        # Current probability vector
        a = softmax(v)

        # Convolution a * a (full length 2n‑1)
        fa = np.fft.fft(a, fft_len)
        conv_aa = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        # Soft‑max weighting -> smooth approximation of the max
        z = temp * conv_aa
        zmax = np.max(z)
        w = np.exp(z - zmax)
        w_sum = w.sum()
        if w_sum == 0:
            w = uniform_sequence(2 * n - 1)
        else:
            w = w / w_sum

        # FFT of the weight vector (zero‑padded)
        fw = np.fft.fft(w, fft_len)

        # g2 = (w * a) – first n entries of linear convolution
        conv_wa = np.real(np.fft.ifft(fw * fa))[: 3 * n - 2]
        g2 = conv_wa[:n]

        # g1 = correlation w ⋆ a (valid part)
        corr_full = np.real(np.fft.ifft(np.conj(fw) * fa))[: 3 * n - 2]
        g1 = corr_full[:n]

        # Gradient w.r.t. a of the smoothed objective
        g = 2.0 * n * (g1 + g2)

        # Chain rule for the soft‑max re‑parameterisation
        dot_ag = np.dot(a, g)
        grad_v = a * (g - dot_ag)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # Periodic exact evaluation
        if it % 400 == 0:
            cur = softmax(v)
            sc = exact_score(cur, n)
            if sc < best_sc:
                best_sc = sc
                best_seq = cur.copy()
                print(f"[refine] iter {it:06d} new best score = {best_sc:.6f}")

    return best_seq


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation function.  The routine respects a time budget (default
    1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    n = int(kwargs.get('n', 2000))

    # --------------------------------------------------------------
    # 0.  Try to reuse a previously discovered sequence, if available.
    # --------------------------------------------------------------
    global_vars = globals()
    if "height_sequence_1" in global_vars:
        seq0 = np.array(global_vars["height_sequence_1"], dtype=float)
        if seq0.size == n and np.all(seq0 >= 0):
            best_seq = seq0 / seq0.sum()
            best_sc = exact_score(best_seq, n)
            print(f"[init] loaded previous sequence, score = {best_sc:.6f}")
        else:
            best_seq = arcsine_sequence(n)
            best_sc = exact_score(best_seq, n)
            print(f"[init] fallback arcsine, score = {best_sc:.6f}")
    else:
        best_seq = arcsine_sequence(n)
        best_sc = exact_score(best_seq, n)
        print(f"[init] arcsine, score = {best_sc:.6f}")

    # --------------------------------------------------------------
    # 1.  Grid search over shifted Beta(α,α) shapes.
    # --------------------------------------------------------------
    alphas = np.linspace(0.35, 0.65, 31)   # includes the classic 0.5
    deltas = np.linspace(0.2, 0.9, 15)    # shift of the interior grid

    for delta in deltas:
        if time.time() >= deadline:
            break
        for alpha in alphas:
            cand = beta_shifted_sequence(n, alpha, delta)
            sc = exact_score(cand, n)
            if sc < best_sc:
                best_sc = sc
                best_seq = cand.copy()
                print(f"[grid] α={alpha:.3f} δ={delta:.3f} → score={sc:.6f}")

    # --------------------------------------------------------------
    # 2.  Short Adam‑style refinement on the smooth surrogate.
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        refine_deadline = time.time() + time_left - 2.0   # 2 s safety margin
        refined = gradient_refinement(best_seq, n, refine_deadline)
        refined_sc = exact_score(refined, n)
        if refined_sc < best_sc:
            best_sc = refined_sc
            best_seq = refined.copy()
            print(f"[refine] final improvement to {best_sc:.6f}")

    # --------------------------------------------------------------
    # 3.  Final post‑processing (non‑negativity, normalisation).
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total < 0.01:                     # enforce the minimal sum required by
        best_seq += (0.01 - total) / n   # evaluate_sequence (sum must be ≥ 0.01)
        total = best_seq.sum()
    best_seq = best_seq / total           # any positive scale works; keep sum = 1

    elapsed = time.time() - start_time
    print(f"[done] best score = {best_sc:.6f}, time = {elapsed:.1f}s")
    return best_seq.tolist()
