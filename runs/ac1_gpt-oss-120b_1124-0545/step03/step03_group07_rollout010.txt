def gradient_softmax(a, tau):
    # a: shape (n,)
    # compute conv = a*a (full)
    # conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len))[:2*n-1]
    conv = full self-convolution.
    # compute softmax weights:
    max_val = conv.max()
    exp_weights = np.exp(tau * (conv - max_val))
    w = exp_weights / exp_weights.sum()
    # gradient = 2 * correlation(w, a)
    w_rev = w[::-1]
    # conv_grad_full = np.convolve(w_rev, a) (length 3n-2)
    # using FFT:
    L = _fft_len_for_n(len(w_rev) + len(a) - 1)
    Fw = np.fft.rfft(w_rev, n=L)
    Fa = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(Fw * Fa, n=L)[: len(w_rev) + len(a) - 1]
    grad = 2 * conv_full[n-1 : n-1 + n]  # length n
    return conv, w, grad

"""
Hybrid optimizer for the step‑function bound.

The algorithm consists of three stages (all running inside the given time
budget):

1. **Conditional‑gradient (Frank–Wolfe) step** – for moderate sizes
   (n ≤ 2000) we solve the linear program

       max sum(b)   s.t.   a ∗ b ≤ M ,   b ≥ 0

   where M = max(a∗a) and “∗” is the linear convolution.  The solution
   b is rescaled to the required ℓ₁–norm and mixed with the current
   vector a; this provably never increases the peak of the self‑convolution
   and yields a strong, global improvement.

2. **Smooth soft‑max gradient descent** – we replace the non‑smooth
   objective max(a∗a) by a temperature‑controlled soft‑max
   `log‑sum‑exp`.  The gradient is a correlation of the soft‑max weights
   with the current sequence.  After each step we project onto the simplex
   { x ≥ 0 , Σx = √(2n) }.  The temperature τ is increased gradually
   (τ → large) so that the procedure first follows a smooth landscape and
   finally refines the true max‑objective.

3. **Simulated annealing** – a classic SA with exponential cooling
   finishes the search, performing random mass transfers between two
   positions and accepting uphill moves with probability exp(−Δ/T).

All three phases keep track of the best sequence seen, and the final
sequence (as a plain Python list) is returned.
"""

import time
import math
import random
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , Σ w = s }."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (used for efficient FFT convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Return (max(conv(a,a)), argmax_index) for the full self‑convolution."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, tau: float, fft_len: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the full self‑convolution of ``a`` and the gradient of the
    temperature‑smoothed objective:

        f_τ(a) = (1/τ)·log Σ_k exp(τ·(a∗a)_k)

    The gradient is   ∇f_τ = 2·(w ⋆ a)   where w are the soft‑max weights.
    """
    n = a.shape[0]

    # ---- full convolution a∗a ----------------------------------------
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

    # ---- soft‑max weights --------------------------------------------
    m = conv_full.max()
    # subtract the max for numerical stability
    exp_w = np.exp(tau * (conv_full - m))
    w = exp_w / exp_w.sum()                     # shape (2n‑1,)

    # ---- gradient = 2·(w ⋆ a)  (cross‑correlation) -------------------
    w_rev = w[::-1]                             # reverse for correlation
    corr_len = len(w_rev) + n - 1
    L = _fft_len_for_n(corr_len)

    fw = np.fft.rfft(w_rev, n=L)
    fa2 = np.fft.rfft(a, n=L)
    corr_full = np.fft.irfft(fw * fa2, n=L)[:corr_len]

    # extract the n entries that correspond to (w ⋆ a)[i] = Σ_j w[i+j] a_j
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]   # shape (n,)

    return conv_full, grad


# ----------------------------------------------------------------------
def _solve_lp_direction(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the linear program
        max Σ b_j   s.t.   a ∗ b ≤ M , b ≥ 0
    and return the optimal b.  If the LP fails we simply return a copy of a.
    """
    n = a.size
    # Build the (2n‑1)×n inequality matrix: (a ∗ b)_k = Σ_j a_{k‑j} b_j
    # Each diagonal (offset d) contains the constant a[d] (for d = 0,…,n‑1)
    data = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(0, n)                    # offsets = 0,…,n‑1
    A_ub = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    b_ub = np.full(2 * n - 1, M)

    # Maximise Σ b_j  → minimise −Σ b_j
    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    # “highs” accepts sparse matrices and is fast for medium‑size LPs.
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"presolve": True})
    if res.success:
        return res.x
    # fallback – keep the current vector unchanged
    return a.copy()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine runs three optimisation phases (conditional‑gradient, smooth
    gradient descent, simulated annealing) inside the allotted time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ----- problem size -------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=float)
    else:
        base = np.random.exponential(scale=1.0, size=2500)

    n = base.size
    target_sum = math.sqrt(2.0 * n)                     # ℓ₁‑norm required by the objective

    # ----- initialise ----------------------------------------------------
    a = base.astype(float)
    a = a / a.sum() * target_sum                         # project onto the simplex

    start = time.time()
    deadline = start + budget_s - 5.0                    # small safety margin

    fft_len = _fft_len_for_n(n)
    cur_val, _ = _max_convolution(a, fft_len)
    best_seq = a.copy()
    best_val = cur_val
    print(f"[INIT] n={n}, initial max(conv)={cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – conditional‑gradient (Frank–Wolfe) using an LP.
    # Only attempted for a size that keeps the LP memory‑friendly.
    if n <= 2000:
        phase1_deadline = start + budget_s * 0.35
        it = 0
        while time.time() < phase1_deadline:
            # recompute the current peak
            cur_val, _ = _max_convolution(a, fft_len)

            # solve the LP to obtain a direction with larger ℓ₁‑mass
            b_dir = _solve_lp_direction(a, cur_val)

            # rescale the direction to the target ℓ₁‑norm
            sum_b = b_dir.sum()
            if sum_b <= 0.0:
                break
            b_scaled = b_dir * (target_sum / sum_b)

            # try a few convex‑combination steps; pick the first that improves
            improved = False
            for t in (0.2, 0.5, 0.8, 1.0):
                a_new = (1.0 - t) * a + t * b_scaled
                val_new, _ = _max_convolution(a_new, fft_len)
                if val_new < cur_val - 1e-12:
                    a = a_new
                    cur_val = val_new
                    improved = True
                    break

            if not improved:
                # a tiny move towards the direction prevents stagnation
                a = a * 0.99 + b_scaled * 0.01
                cur_val, _ = _max_convolution(a, fft_len)

            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[LP {it:5d}] new best max(conv)={best_val:.6f}")
            it += 1

    # ------------------------------------------------------------------
    # Phase 2 – smooth soft‑max gradient descent.
    tau = 0.5           # start with a very smooth approximation
    tau_max = 10.0
    grad_deadline = start + budget_s * 0.75
    it = 0
    while time.time() < grad_deadline:
        # full convolution + gradient of the smoothed objective
        conv_full, grad = _smooth_gradient(a, tau, fft_len)

        # simple diminishing step size
        step = (target_sum * 0.02) / math.sqrt(it + 1.0)
        a_candidate = a - step * grad
        a_candidate = _simplex_projection(a_candidate, target_sum)

        cand_val, _ = _max_convolution(a_candidate, fft_len)

        if cand_val < best_val:
            best_val = cand_val
            best_seq = a_candidate.copy()
            print(f"[GRAD {it:6d}] improved max(conv)={best_val:.6f}")

        # accept if it improves the current peak
        if cand_val < cur_val:
            a = a_candidate
            cur_val = cand_val

        # slowly increase the temperature (more focus on the true max)
        tau = min(tau_max, tau * 1.002)
        it += 1

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing for fine‑grained refinement.
    cur_seq = best_seq.copy()
    cur_val, _ = _max_convolution(cur_seq, fft_len)

    T0 = cur_val * 0.20
    Tmin = cur_val * 1e-5
    step_counter = 0

    while time.time() < deadline:
        # exponential cooling based on the fraction of the total budget used
        frac = (time.time() - start) / budget_s
        T = T0 * ((Tmin / T0) ** frac)

        i = np.random.randint(n)
        if cur_seq[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # transfer up to 10 % of the source mass
        delta = cur_seq[i] * 0.10 * np.random.rand()
        cand_seq = cur_seq.copy()
        cand_seq[i] -= delta
        cand_seq[j] += delta

        cand_val, _ = _max_convolution(cand_seq, fft_len)

        if cand_val < cur_val:
            cur_seq = cand_seq
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand_seq.copy()
                print(f"[SA {step_counter:7d}] new best max(conv)={best_val:.6f}")
        else:
            # Metropolis acceptance for uphill moves
            prob = math.exp(-(cand_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                cur_seq = cand_seq
                cur_val = cand_val

        step_counter += 1

    print(f"[DONE] best max(conv) = {best_val:.6f} after {time.time() - start:.1f}s")
    return best_seq.tolist()
