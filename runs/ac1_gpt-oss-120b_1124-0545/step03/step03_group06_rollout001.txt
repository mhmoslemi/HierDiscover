"""
Propose a step‑function (non‑negative vector on the probability simplex)
that minimizes the evaluation function

    2·n·max(conv(a,a)) / (sum a)^2 .

The algorithm improves on the previous submission by

* using the *exact* sub‑gradient of the max‑convolution (all active
  constraints are taken into account, not only a fixed “top‑k” set);
* projecting after each update onto the simplex with the O(n) Duchi et al.
  algorithm;
* employing a simple step‑size schedule and occasional random mass swaps
  to escape plateaus;
* performing a coarse / fine power‑law sweep to obtain a good start,
  followed by a long exact‑gradient phase, a few random restarts and a
  final polishing phase.

All convolutions needed for the max‑convolution are performed with FFT
(`_conv_fft`).  The routine is completely self‑contained, uses only
NumPy and the standard library and respects a 1000 s overall time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """
    Normalised non‑negative sequence proportional to (i+1+offset)^{-α}.

    Parameters
    ----------
    n : int
        Length of the sequence.
    alpha : float
        Exponent (α>0).
    offset : float
        Small shift that can be used to reduce the first coefficient.

    Returns
    -------
    np.ndarray
        Length‑`n` array with sum equal to 1.
    """
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """
    Self‑convolution using the FFT (real valued). Returns the full vector
    of length ``2*len(seq)-1``.
    """
    n = seq.shape[0]
    size = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    conv[conv < 0] = 0.0                     # numerical noise guard
    return conv


def _conv_fft_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    FFT based convolution of two arbitrary 1‑D arrays.
    Returns the full‑length convolution ``len(a)+len(b)-1``.
    """
    n, m = a.shape[0], b.shape[0]
    size = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    conv[conv < 0] = 0.0                     # numerical noise guard
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1} (algorithm of Duchi et al., 2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_fft(seq: np.ndarray) -> float:
    """
    Exact value of the objective used by ``evaluate_sequence`` for a
    normalised sequence (Σseq == 1). The denominator therefore equals 1.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


def _exact_gradient_step(seq: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step that reduces the *exact* maximal
    entry of the self‑convolution.

    The sub‑gradient is computed from **all** indices where the
    convolution attains its maximum (the “active set”).  This yields a
    true sub‑gradient of the nonsmooth objective.
    """
    n = seq.shape[0]

    # current convolution and the maximal entry
    conv = _conv_fft(seq)
    max_val = conv.max()

    # mask of the active constraints (tolerance avoids missing due to
    # floating‑point noise)
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]

    # build the sub‑gradient (exact)
    grad = np.zeros_like(seq)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1, dtype=int)
        grad[i_range] += seq[k - i_range]

    grad *= 2.0                     # ∂(a*a)[k]/∂a_i = 2·a_{k-i}
    grad -= grad.mean()             # make it tangent to Σx = 1

    new_seq = seq - step * grad
    new_seq = _project_simplex(new_seq)
    return new_seq


def _run_exact_gradient(
    seq: np.ndarray,
    time_budget: float,
    *,
    max_iters: int = 10 ** 9,
    step0: float = 0.5,
    decay: float = 0.9999,
    swap_interval: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent using the *exact* active‑set gradient.
    Returns the best sequence seen during the run and its corresponding
    (exact) score.
    """
    start = time.time()
    best_seq = seq.copy()
    best_score = _score_fft(seq)

    step = step0
    iteration = 0

    while iteration < max_iters and (time.time() - start) < time_budget:
        iteration += 1

        seq = _exact_gradient_step(seq, step)

        # occasional random mass exchange to help escape flat regions
        if swap_interval and iteration % swap_interval == 0:
            i, j = np.random.choice(seq.size, 2, replace=False)
            eps = min(step * seq[i], seq[i] * 0.3)
            seq[i] -= eps
            seq[j] += eps
            seq = _project_simplex(seq)

        cur_score = _score_fft(seq)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        step = max(step * decay, 1e-8)

    return best_seq, best_score
