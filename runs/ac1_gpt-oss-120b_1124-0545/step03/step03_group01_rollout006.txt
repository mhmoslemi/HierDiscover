def _local_search(start_seq, current_val, deadline, max_iters=50000, step_initial=0.005):

    a = start_seq.copy()
    best_seq = a.copy()
    best_val = current_val
    rng = np.random.default_rng(0) # deterministic for reproducibility

    step = step_initial
    accept_cnt = 0
    for it in range(1, max_iters+1):
        if time.time()>deadline: break

        # propose perturbation
        # choose two distinct indices
        i, j = rng.choice(len(a), size=2, replace=False)

"""
Propose a step‑function that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

subject to a_i ≥ 0.  The search consists of three stages:

1. **Diverse seeds** – analytic families (uniform, sine‑window,
   geometric, power‑law, Gaussian, Beta‑distribution) plus a few random
   Dirichlet draws.  All seeds are normalised to sum 1.

2. **Smooth max minimisation** – Adam optimiser works on the
   log‑softmax parametrisation of the sequence.  A temperature‑controlled
   LogSumExp approximates the max, the temperature (β) is increased until
   it is large (≈200) so the optimisation finally targets the true max.
   For each length only the best few seeds are refined for many Adam
   iterations; the remaining seeds are only evaluated.

3. **Exact sub‑gradient refinement** – from the best Adam result we run
   a long mirror‑descent (exponentiated‑gradient) on the simplex using
   the true sub‑gradient of the max‑convolution.

4. **Local random search** – a cheap hill‑climbing / simulated‑annealing
   phase that repeatedly moves a tiny amount of mass between two random
   positions.  It is run until the global time budget (≈1000 s) expires.

All heavy work (convolutions) is performed by a real‑FFT routine
`_conv_fft`.  The routine respects the supplied time budget and returns
the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _beta_dist(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {0,…,n‑1} (continuous then discretised)."""
    t = (np.arange(1, n + 1, dtype=np.float64)) / (n + 1.0)
    a = (t ** (alpha - 1.0)) * ((1.0 - t) ** (beta - 1.0))
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a  # already sums to 1


# ----------------------------------------------------------------------
# Objective helper
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Return the exact value 2·n·max(conv(seq,seq)) / sum(seq)²."""
    n = len(seq)
    if n == 0:
        return float("inf")
    conv = _conv_fft(seq, seq)
    max_b = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Adam optimiser (smooth max)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 5000,
                   lr0: float = 0.2,
                   beta_start: float = 1.0,
                   beta_mul: float = 1.05,
                   beta_cap: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed max (log‑sum‑exp) of the auto‑convolution.
    The parameters are chosen for a fast yet stable convergence.
    Returns the best sequence (on the simplex) and the corresponding
    2·n·max(conv) objective (sum is 1 → the factor “2·n·max(conv)”).
    """
    v = np.log(seed + 1e-30)           # log‑softmax representation
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = beta_start
    best_seq = None
    best_val = float("inf")
    eps_adam = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)    # sum(a) == 1
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:5d} best={cur_val:.6f}")

        # increase smoothing temperature
        beta = min(beta * beta_mul, beta_cap)

        shift = np.max(conv)                # for numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t a (closed form via convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # transform gradient to the log‑softmax parameters
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (exponentiated‑gradient / mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 30000,
                        lr0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the true sub‑gradient of max(conv).
    The update is performed in log‑space (exponentiated‑gradient) which
    automatically respects the simplex constraints.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx] / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # exact sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                 # for stability
        a = np.exp(log_a)
        a /= a.sum()                           # projection onto simplex

    return best_seq, best_val


# ----------------------------------------------------------------------
# Local random search (hill‑climbing / simulated‑annealing)
# ----------------------------------------------------------------------


def _local_search(start_seq: np.ndarray,
                  start_val: float,
                  deadline: float,
                  max_iters: int = 20000,
                  init_step: float = 0.005,
                  rng_seed: int = 314159) -> tuple[np.ndarray, float]:
    """
    Very cheap random walk on the simplex: move a tiny amount of mass
    between two indices.  Occasionally accepts worsening moves according
    to a slowly cooling temperature.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = start_val
    n = len(a)
    rng = np.random.default_rng(rng_seed)

    step = init_step
    acc = 0
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # pick two distinct positions
        i, j = rng.choice(n, size=2, replace=False)

        # mass to transfer (proportional to current step size)
        eps = step * rng.random()
        eps = min(eps, a[i])  # keep a[i] non‑negative

        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        # re‑normalise (numerical drift)
        cand /= cand.sum()

        cand_val = _objective(cand)

        # accept improvement always, otherwise with a small probability
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand.copy()
            a = cand
            acc += 1
        else:
            # simple simulated‑annealing temperature
            T = step / (1.0 + it * 0.001)
            if np.log(rng.random()) < -(cand_val - best_val) / max(T, 1e-12):
                a = cand
                acc += 1

        # adapt step size every 500 iterations
        if it % 500 == 0:
            acc_rate = acc / 500.0
            if acc_rate > 0.2:
                step = min(step * 1.2, 0.02)
            elif acc_rate < 0.02:
                step = max(step * 0.7, 1e-4)
            acc = 0

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generator (extended families)
# ----------------------------------------------------------------------


def _make_seeds_extended(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a modest yet diverse set of starting points (already normalised)."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric & power families (as in the original code)
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
            seeds.append(_sym_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # Gaussian family (different bandwidths)
    for sigma_factor in (0.15, 0.12, 0.10):
        sigma = sigma_factor * n
        seeds.append(_gaussian(n, sigma))

    # Beta‑distribution family
    for a in (0.2, 0.5, 1.0, 2.0):
        for b in (0.2, 0.5, 1.0, 2.0):
            if a > 0 and b > 0:
                seeds.append(_beta_dist(n, a, b))

    # a handful of random Dirichlet draws
    rng = np.random.default_rng(123456)
    for conc in (0.3, 0.7, 1.5):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # reuse a previously discovered solution if it has the correct length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # ensure we have unique seeds (tiny numerical differences are fine)
    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` objective.  The routine respects the time budget,
    uses several optimisation phases, and finally returns the best sequence
    found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the runner
    deadline = start_time + budget_s - 2.0

    # Retrieve a previous construction if available
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – start with the proven 4000, then a larger one
    # ------------------------------------------------------------------
    lengths = [4000, 8000]
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # 1.  Build seeds and evaluate them quickly
        # --------------------------------------------------------------
        seeds = _make_seeds_extended(n, previous)
        for s in seeds:
            val = _objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n} new global best={val:.6f}")

        # --------------------------------------------------------------
        # 2.  Choose the most promising seeds for a longer Adam run
        # --------------------------------------------------------------
        # keep only the best K seeds (K=7 works well)
        K = 7
        seeds_sorted = sorted(seeds, key=_objective)[:K]

        for idx, seed_vec in enumerate(seeds_sorted):
            if time.time() > deadline:
                break
            print(f"[Adam start] n={n} seed #{idx+1}/{len(seeds_sorted)}")
            seq, val = _adam_optimize(
                seed_vec,
                n,
                deadline,
                max_iters=5000,
                lr0=0.2,
                beta_start=1.0,
                beta_mul=1.07,
                beta_cap=200.0,
            )
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall] n={n} new global best={val:.6f}")

        # --------------------------------------------------------------
        # 3.  Exact sub‑gradient refinement on the current best of this n
        # --------------------------------------------------------------
        if global_best_seq is not None and len(global_best_seq) == n:
            print("[SubGrad start] refining on best sequence")
            seq_ref, val_ref = _subgradient_refine(
                global_best_seq,
                n,
                deadline,
                max_iters=30000,
                lr0=0.6,
            )
            if seq_ref is not None and val_ref < global_best_val:
                global_best_val = val_ref
                global_best_seq = seq_ref.copy()
                print(f"[overall] after sub‑grad n={n} best={val_ref:.6f}")

    # ------------------------------------------------------------------
    # 4.  Local random search on the overall best sequence (any length)
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:   # need a tiny window for the final phase
            print("\n[Local search] fine‑tuning with hill‑climbing")
            seq_final, val_final = _local_search(
                global_best_seq,
                global_best_val,
                deadline,
                max_iters=25000,
                init_step=0.005,
                rng_seed=987654,
            )
            if seq_final is not None and val_final < global_best_val:
                global_best_seq = seq_final
                global_best_val = val_final
                print(f"[final] improved to {val_final:.6f}")

    # ------------------------------------------------------------------
    # 5.  Return the best candidate found
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # Fallback – uniform distribution of modest size
        n = 4000
        global_best_seq = _uniform(n)
        print("[fallback] using uniform distribution")

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )
    return global_best_seq.tolist()
