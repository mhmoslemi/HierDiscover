"""
Propose a step‑function (non‑negative heights) with a very low
2·n·max(conv(a,a)) / (∑a)² value.

Algorithm
---------
1. Choose a moderate length (8000).  The objective after fixing the
   ℓ₁‑norm to 1 becomes 2·n·max(conv), i.e. we only need to minimise the
   maximum auto‑convolution M = max(conv(a,a)).
2. Generate a small diverse pool of seeds (uniform, geometric, power‑law,
   mixed, sine‑window and a few Dirichlet samples).  All are normalised
   to sum = 1.
3. For each seed run a long Adam optimisation on a *smoothed* max
   (log‑sum‑exp) of the auto‑convolution.  The smoothing parameter β is
   increased slowly so that the optimisation gradually focuses on the
   true max.  Iterates stay on the probability simplex via a softmax
   parametrisation.
4. From the best Adam result run a mirror‑descent (exponentiated‑gradient)
   refinement that uses the exact sub‑gradient of the true max.  The step
   size decays like 1/√t.
5. After all seeds a cheap random‑local‑search fine‑tuning is performed:
   repeatedly move a small amount of mass from a random index i to a
   random index j and accept the move if the objective improves.
6. The best sequence found is stored in the global variable
   ``height_sequence_1`` (so it can be reused in a later run) and returned
   as a plain list of floats.  The whole routine respects a user supplied
   time budget (default 1000 s) and stops early if the deadline is hit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based linear convolution)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution using a real FFT.
    Length of result = len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a vector on the simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


# ----------------------------------------------------------------------
# Objective (fast version – assumes a is already on the simplex)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """
    Compute the evaluation function for a vector a.
    For speed we use the definition 2·n·max(conv) because sum(a) = 1.
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    M = np.max(conv)
    return float(2.0 * n * M)


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation of the smoothed maximum of the auto‑convolution.
    Returns the best sequence (as a NumPy array) seen and its true objective.
    """
    # Initialise log‑softmax parameters
    v = np.log(seed + 1e-30)

    # Adam moments
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # Adam hyper‑parameters
    lr0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # Smoothing schedule
    beta = 2.0
    beta_mul = 1.03

    best_seq = None
    best_val = np.inf
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                       # a lies on the simplex

        # true objective (fast version)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # keep output reasonably quiet – uncomment for debugging
            # print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ----- smoothed‑max weights (log‑sum‑exp) -----
        beta *= beta_mul
        shift = np.max(conv)                  # numerical shift
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient wrt a of Σ w_k·conv_k -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # ----- convert to gradient wrt v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (exponentiated‑gradient / mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 20000) -> tuple[np.ndarray | None, float]:
    """
    Refine a sequence using the exact sub‑gradient of the true max.
    Mirror‑descent (exponentiated‑gradient) is used, keeping the vector
    on the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # print(f"[SubGrad] it={it:5d} best={cur_val:.6f}")

        # Sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Exponentiated‑gradient (mirror‑descent) update
        lr = 0.5 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)               # numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

    return best_seq, best_val

# Seed generation utilities

def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _random_dirichlet(n: int, alpha: float = 0.5) -> np.ndarray:
    rng = np.random.default_rng()
    x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s != 0 else np.full(n, 1.0 / n)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a modest but diverse set of starting points (already normalised)."""
    seeds: list[np.ndarray] = []

    # uniform baseline
    seeds.append(np.full(n, 1.0 / n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 7):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 7):
        seeds.append(_power(n, p))

    # geometric × power mix
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))

    # smooth sine window
    seeds.append(_sine_window(n))

    # a handful of random Dirichlet vectors
    for _ in range(8):
        seeds.append(_random_dirichlet(n, alpha=0.5))

    # optionally reuse a previously discovered sequence
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds

    # Basic set‑up
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    # slight safety margin – we stop a bit before the hard deadline
    deadline = start_time + budget_s - 1.0

    # (optional) reuse a sequence from a previous run
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Choose a length that offers a good trade‑off between flexibility
    # and per‑iteration cost.
    n = 8000

    # ------------------------------------------------------------------
    # Build initial seeds and evaluate them once (fast)
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous)
    best_seq = None
    best_val = np.inf

    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
    print(f"[Init] Best among raw seeds: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – Adam + exact sub‑gradient on each seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimising seed {idx + 1}/{len(seeds)} ---")

        # 1️⃣ Adam on smoothed max
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline,
                                            max_iters=20000)
        if seq_adam is not None and val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam.copy()
            print(f"[Global] New best after Adam: {best_val:.6f}")

        # 2️⃣ Exact sub‑gradient refinement
        if seq_adam is not None:
            seq_ref, val_ref = _subgradient_refine(seq_adam, n, deadline,
                                                   max_iters=20000)
            if seq_ref is not None and val_ref < best_val:
                best_val = val_ref
                best_seq = seq_ref.copy()
                print(f"[Global] New best after sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Light random local search around the current best sequence
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        step = 0
        while time.time() < deadline and step < 20000:
            step += 1
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                continue
            # Move a small fraction (≈1 %) of mass from i to j
            eps = 0.01 * a[i]
            a_new = a.copy()
            a_new[i] -= eps
            a_new[j] += eps
            # renormalise to keep sum exactly 1
            a_new /= a_new.sum()
            val = _objective(a_new)
            if val < best_val:
                best_val = val
                a = a_new
                best_seq = a_new.copy()
                print(f"[Local] step {step:5d} improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")

    # expose the result for a possible next call
    globals()["height_sequence_1"] = best_seq.copy()

    # Return as plain Python list of floats
    return best_seq.tolist()
