"""
Propose a low‑value step‑function sequence.

Algorithm
---------
1. Initialise a non‑negative vector `a` of length ``n`` (default 6000)
   – if the global variable ``height_sequence_1`` exists we interpolate it,
   otherwise we start from a uniform random vector.  The vector is projected
   onto the probability simplex (sum = 1).

2. Repeatedly minimise a *smooth* surrogate of the peak of the auto‑convolution
   ``c = a * a``.  We use an L‑p approximation:
        w_k ∝ (c_k / max(c))^{p-1}          (p grows from 2 to ≈12)
   The gradient of the surrogate w.r.t. ``a`` is
        ∇ ≈ -2·(a_rev ∗ w)                (full convolution via FFT)
   The descent direction is projected onto the hyperplane Σ∇=0, then an
   Adam‑style update is applied (first‑ and second‑moment estimates).  After
   the update the vector is clipped to non‑negative entries and re‑normalised
   to keep Σa = 1.

3. To escape plateaux we occasionally perform a cheap “mass‑transfer’’ move,
   moving a small fraction of total mass from one index to another.

4. If no improvement has been observed for a configurable amount of time
   (stagnation) we restart from the best sequence seen so far with a small
   random perturbation.

5. The loop runs until the time budget expires (default 1000 s).  The
   best sequence found (lowest value of ``evaluate_sequence``) is returned.

The use of the L‑p approximation (instead of the exponential soft‑max used
in earlier attempts) together with Adam‑style adaptive step sizes gives a
more reliable descent and usually finds a better upper bound within the same
time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution via FFT (real‑valued inputs).
    ``size`` must be ≥ len(u)+len(v)-1.
    Returns the *full* convolution (length len(u)+len(v)-1).
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Convenient wrapper for the auto‑convolution a*a (full length)."""
    return _conv_fft(a, a, size)


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Move a random fraction of the total mass from one coordinate to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer <= 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # re‑normalise (preserves Σa = 1)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / n
    else:
        a_new /= s
    return a_new

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------

    """
    Search for a step‑function sequence of non‑negative heights that minimises

    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

        else:

            d -= d.mean()

            if s == 0.0:
                a[:] = 1.0 / n_len
            else:
                a /= s

            a_rev = a[::-1]

        # ----------------------------------------------------------
        # 3.  restart if we are stuck for too long
        # ----------------------------------------------------------
        if time.time() - last_improve > stagnation_limit:
            # restart from the current best sequence plus a small noise burst
            a = best_seq.copy()
            noise = np.random.randn(n) * 0.005
            a = a + noise
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.random.rand(n)
                a /= a.sum()
            else:
                a /= a_sum
            a_rev = a[::-1]
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[{it:7d}] restart (stagnation) – re‑initialised from best")

        it += 1

    # --------------------------------------------------------------
    # final report
    # --------------------------------------------------------------
    total_time = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total_time:.1f}s")
    # return a *list* of Python floats
    return best_seq.tolist()
