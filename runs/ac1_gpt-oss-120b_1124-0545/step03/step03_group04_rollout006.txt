def run_subgradient(a, n, deadline):
    # a is np.ndarray of length n sum=1
    best_score = 2.0 * n * _conv_fft(a,a).max()
    best_seq = a.copy()
    # Initialize hyperparameters
    lr = 0.5
    lr_decay = 0.9998
    min_lr = 1e-6
    momentum = np.zeros_like(a)
    beta=0.9
    iter = 0
    while time.time() < deadline:
        iter += 1
        conv = _conv_fft(a, a)
        M = conv.max()
        # indices of max (within tolerance to avoid floating rounding)
        # Use tolerance relative to M
        eps = 1e-14
        K = np.where(conv >= M - eps)[0]   # all indices attaining max
        # build w as uniform over K
        w = np.zeros_like(conv)
        if K.size > 0:
            w[K] = 1.0 / K.size
        else:
            # shouldn't happen
            w[conv.argmax()] = 1.0
        # compute v via convolution
        w_rev = w[::-1]
        v_full = _conv_fft(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)
        v = v_full[idx]
        # gradient of max conv approx 2*v (since derivative of conv sum)
        grad = 2.0 * v  # shape (n,)
        # add small L2 regularizer to avoid trivial uniform?
        # optional: grad += 1e-6 * a
        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        # gradient descent step (descending)
        a = a - lr * momentum
        # project onto simplex
        a = _project_onto_simplex(a)
        # decay learning rate
        lr = max(lr * lr_decay, min_lr)
        # occasional random swap
        if iter % 200 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i] * 0.5, a[i] * lr * 10)  # maybe bigger
            if delta > 0:
                a[i] -= delta
                a[j] += delta
                a = _project_onto_simplex(a)

        # compute current score if needed
        if iter % 500 == 0 or time.time() >= deadline:
            cur_conv = _conv_fft(a, a)
            cur_score = 2.0 * n * cur_conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[candidate] iter={iter}, score={best_score:.6f}")
    return best_seq, best_score, iter
```

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0 , Σ x_i = 1 }.
    Implements the algorithm of Duchi et al. (ICML 2008).

    Parameters
    ----------
    v : np.ndarray
        Input vector (any real values).

    Returns
    -------
    np.ndarray
        Projected vector (non‑negative, sum‑1).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using FFT.
    Length = len(x) + len(y) – 1.

    Parameters
    ----------
    x, y : np.ndarray
        Real 1‑D input arrays.

    Returns
    -------
    np.ndarray
        Full convolution.
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _u_shape_distribution(n: int, beta: float = 0.5) -> np.ndarray:
    """
    Symmetric “U‑shaped’’ distribution:
    a_i ∝ ((i+1)*(n-i))^{-β},  i = 0 … n‑1.
    Normalised to sum = 1.
    """
    i = np.arange(n, dtype=float) + 1.0             # 1 … n
    vals = (i * (n - i + 1.0)) ** (-beta)
    s = vals.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return vals / s


def _random_dirichlet(n: int, conc: float = 0.5) -> np.ndarray:
    """Sample a Dirichlet vector with concentration `conc` and normalise."""
    rng = np.random.default_rng()
    samples = rng.gamma(conc, 1.0, size=n)
    total = samples.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return samples / total

    eps: float = 1e-8,
    symmetrize: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises 2·n·max(a*a).

    Returns the best vector found and its (exact) score.
    """
    a = a0.copy()
    # record best
    conv = _conv_fft(a, a)
    best_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        t += 1

        # 1️⃣ current convolution and its maximum
        conv = _conv_fft(a, a)
        M = conv.max()
        # tolerance to collect all truly maximal indices (avoid floating noise)
        tol = max(1e-12, M * 1e-12)
        K = np.where(conv >= M - tol)[0]

        # 2️⃣ build uniform weight on the active set
        w = np.zeros_like(conv)
        if K.size:
            w[K] = 1.0 / K.size
        else:
            w[conv.argmax()] = 1.0

        # 3️⃣ sub‑gradient of max‑convolution:
        #    g = 2 * Σ_{k∈K} shift(a, k)
        #    efficiently via a single convolution with w_rev
        w_rev = w[::-1]
        v_full = _conv_fft(a, w_rev)                 # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)             # map to gradient entries
        grad = 2.0 * v_full[idx]                     # shape (n,)

        # 4️⃣ Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # 5️⃣ gradient step + projection
        a = a - step
        a = _project_onto_simplex(a)

        # 6️⃣ optional symmetry enforcement (helps because optimum is symmetric)
        if symmetrize:
            a = (a + a[::-1]) * 0.5

        # 7️⃣ occasional random mass‑swap to break tiny plateaus
        if t % 500 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i] * 0.5, a[i] * lr * 10.0)
            if delta > 0.0:
                a[i] -= delta
                a[j] += delta
                a = _project_onto_simplex(a)
                if symmetrize:
                    a = (a + a[::-1]) * 0.5

        # 8️⃣ keep exact best score seen so far (no extra np.convolve call)
        if t % 1000 == 0 or time.time() + 1.0 >= deadline:
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[improve] iter={t}, score={best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search the space of non‑negative step‑functions for the smallest
    ``evaluate_sequence`` value.  The routine runs for at most ``budget_s``
    seconds and returns the best sequence discovered.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  A small safety margin is kept.
    kwargs : dict
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence (non‑negative, sum > 0.01) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a few seconds for the harness to receive the result
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1️⃣ Choose a length that balances expressivity and speed.
    #    Power‑of‑two sizes give the fastest FFTs.
    # ------------------------------------------------------------------
    n = 16384                     # ~16k entries (fits comfortably in memory)

    # ------------------------------------------------------------------
    # 2️⃣ Build a set of promising initial points.
    # ------------------------------------------------------------------
    initials = []

    # 2a – the classic U‑shaped power‑law (Beta(0.5,0.5) like)
    initials.append(_u_shape_distribution(n, beta=0.5))

    # 2b – a slightly flatter U‑shape (β=0.3) – gives a different geometry
    initials.append(_u_shape_distribution(n, beta=0.3))

    # 2c – a random Dirichlet (concentration < 1 gives spikier candidates)
    initials.append(_random_dirichlet(n, conc=0.5))

    # 2d – if the harness supplies a previous construction use it
    try:
        import sys

        # ``height_sequence_1`` might exist in the global namespace of the runner
        hs = np.array(sys.modules[__name__].height_sequence_1, dtype=float)  # type: ignore
        # If the provided sequence is shorter or longer, we either truncate or pad
        if hs.size != n:
            if hs.size > n:
                hs = hs[:n]
            else:
                # pad with a tiny amount of uniform mass to keep sum = 1 after normalisation
                pad = np.full(n - hs.size, 1e-12)
                hs = np.concatenate([hs, pad])
        hs = np.maximum(hs, 0.0)
        if hs.sum() == 0.0:
            raise ValueError
        hs /= hs.sum()
        initials.append(hs)
    except Exception:
        pass  # ignore if the global is missing

    # ------------------------------------------------------------------
    # 3️⃣ Optimise each candidate (or a few of them) until time runs out.
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_seq = None
    cand_idx = 0
    while time.time() < deadline and cand_idx < len(initials):
        a0 = initials[cand_idx].copy()
        # small jitter to avoid exact duplicates between runs
        a0 += 1e-8 * np.random.rand(n)
        a0 = _project_onto_simplex(a0)
        a0 = (a0 + a0[::-1]) * 0.5  # enforce symmetry from the start

        # reserve a slice of the remaining budget for this run
        remaining = deadline - time.time()
        # allocate at most 0.4 × remaining to a single candidate (leaves room for
        # later candidates) but never less than 5 s.
        run_deadline = time.time() + max(5.0, 0.4 * remaining)

        seq, score = _adam_optimize(a0, n, run_deadline)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[global] new best from cand {cand_idx} – score={best_score:.6f}")

        cand_idx += 1

    # ------------------------------------------------------------------
    # 4️⃣ Final sanity check and return.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (should never happen)
        best_seq = np.full(n, 1.0 / n)

    # ensure the output respects the evaluate_sequence guards
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total < 0.01:
        # rescale safely
        best_seq = best_seq / total * max(0.01, total)
    print(f"[final] best_score={best_score:.6f}, length={n}")
    return best_seq.tolist()
