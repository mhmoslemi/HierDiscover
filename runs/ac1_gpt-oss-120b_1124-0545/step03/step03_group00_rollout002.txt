def _project_simplex(v, s=1.0):
    # v: 1d array
    # returns projection onto {x >= 0, sum x = s}
    n = len(v)
    if np.sum(v) == s and np.all(v >= 0):
        return v.copy()
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step function (height sequence) minimizing
evaluate_sequence. The algorithm builds a rich collection of analytic
window families (geometric, power‑law, Hann, Hamming, Blackman,
Kaiser, Gaussian, etc.), evaluates them quickly, then refines the best
candidate with two complementary gradient‑based methods:

1. Adam on the log‑softmax parametrisation using a smoothed max loss.
2. A simple projected sub‑gradient descent on the exact max
   convolution term.

Both refinements run until the allotted time expires.  The final
sequence is projected onto the simplex (non‑negative, sum = 1) and
returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.special import i0   # for Kaiser window


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using FFT (real inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i ≥ 0, ∑ x_i = s}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.sum(v) == s and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _kaiser(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = (2.0 * i) / (n - 1) - 1.0  # in [-1, 1]
    a = i0(beta * np.sqrt(1.0 - t * t))
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Gaussian window, sigma = sigma_frac * n."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return a / a.sum()


def _power_window(base: np.ndarray, p: float) -> np.ndarray:
    """Raise a base window to power p and renormalise."""
    a = np.maximum(base, 0.0) ** p
    s = a.sum()
    return a / s if s != 0 else np.full_like(base, 1.0 / base.size)

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_refine(a0: np.ndarray, n: int, deadline: float,
                 start_beta: float = 5.0,
                 beta_mul: float = 1.02,
                 lr0: float = 0.8,
                 max_beta: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.
    Uses a smoothed‑max loss with a geometric β schedule capped at max_beta.
    Returns the best sequence seen and its fast objective value.
    """
    v = np.log(a0 + 1e-30)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    best_seq = a0.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a0, a0))
    beta = start_beta
    it = 0

    while True:
        it += 1
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # increase β but keep it bounded
        beta = min(beta * beta_mul, max_beta)
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of smoothed loss w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n-2
        grad_a = 2.0 * full[n - 1: n - 1 + n]    # central n entries

        # gradient w.r.t. logits (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


def _subgrad_refine(a0: np.ndarray, n: int, deadline: float,
                    max_iters: int = 5000,
                    step0: float = 0.5,
                    tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses the true sub‑gradient (average over all maximising shifts) and
    Euclidean projection onto the simplex after each step.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[subgrad] n={n:5d} it={t:6d} best={cur_val:.6f}")

        # identify all shift indices attaining (or almost attaining) the max
        max_inds = np.where(conv >= cur_max - tol)[0]

        # compute sub‑gradient (average over all active shifts)
        grad = np.zeros_like(a)
        idx = np.arange(n)
        for k in max_inds:
            partner = k - idx
            mask = (partner >= 0) & (partner < n)
            grad[mask] += 2.0 * a[partner[mask]]
        grad /= max(len(max_inds), 1)

        # diminishing step size
        step = step0 / np.sqrt(t)
        a = a - step * grad
        a = _project_simplex(a)      # back to the simplex

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence minimising `evaluate_sequence`.
    The routine respects the supplied time budget (default 1000 s) and
    returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # 4 s safety margin

    # ------------------------------------------------------------------
    # Choose candidate lengths (largest first, fall back if time short)
    # ------------------------------------------------------------------
    lengths = [12000, 8000, 4000]
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"[main] Optimising length = {n}")

        # --------------------------------------------------------------
        # Build a diverse set of seed sequences (all on the simplex)
        # --------------------------------------------------------------
        seed_seqs = []

        # Geometric families
        for c in np.linspace(0.5, 5.0, 12):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_geom(n, r))

        # Power‑law families
        for p in np.linspace(0.0, 2.2, 12):
            seed_seqs.append(_power(n, p))

        # Combined geometric × power families
        for c in np.linspace(0.5, 4.0, 8):
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in np.linspace(0.0, 2.0, 8):
                seed_seqs.append(_geom_power(n, r, p))

        # Symmetric geometric
        for c in np.linspace(0.5, 4.5, 8):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_sym_geom(n, r))

        # Classical windows
        seed_seqs.append(_sine_window(n))
        seed_seqs.append(_hann(n))
        seed_seqs.append(_hamming(n))
        seed_seqs.append(_blackman(n))

        # Kaiser windows (few β values)
        for beta in [0.5, 2.0, 5.0, 8.0, 12.0]:
            seed_seqs.append(_kaiser(n, beta))

        # Gaussian windows (different σ fractions)
        for sigma_frac in [0.05, 0.12, 0.20, 0.35, 0.50]:
            seed_seqs.append(_gaussian(n, sigma_frac))

        # Exponentiate some windows to get sharper/flat shapes
        base_windows = [_sine_window(n), _hann(n), _hamming(n), _blackman(n)]
        for base in base_windows:
            for p in np.linspace(0.3, 2.5, 6):
                seed_seqs.append(_power_window(base, p))

        # Dirichlet draws with various concentrations
        for alpha in [0.2, 0.5, 0.8, 2.0, 5.0]:
            a = np.random.gamma(alpha, 1.0, size=n)
            s = a.sum()
            if s > 0:
                seed_seqs.append(a / s)

        # Random soft‑max seeds
        for _ in range(6):
            v_rand = np.random.randn(n)
            seed_seqs.append(_softmax(v_rand))

        # Re‑use a previously discovered sequence if available
        prev = globals().get("height_sequence_1")
        if prev is not None:
            try:
                a_prev = np.asarray(prev, dtype=np.float64).ravel()
                if a_prev.shape[0] != n:
                    # adjust size by truncation / zero‑padding
                    a_adj = np.zeros(n, dtype=np.float64)
                    m = min(a_prev.shape[0], n)
                    a_adj[:m] = a_prev[:m]
                    a_prev = a_adj
                if a_prev.sum() > 0:
                    a_prev = a_prev / a_prev.sum()
                    seed_seqs.append(a_prev)
            except Exception:
                pass

        # --------------------------------------------------------------
        # Quick scan over seeds (fast objective = 2·n·max(conv))
        # --------------------------------------------------------------
        for idx, a in enumerate(seed_seqs):
            cur_val = 2.0 * n * np.max(_conv_fft(a, a))
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[scan] new global best from seed {idx} = {cur_val:.6f}")

        # --------------------------------------------------------------
        # Refinement phase 1 – Adam optimiser on smoothed max
        # --------------------------------------------------------------
        if global_best_seq is None:
            # fallback uniform start
            global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
            print("[fallback] using uniform start for Adam")

        # Run Adam; parameters are chosen for robustness
        refined_seq, refined_val = _adam_refine(
            global_best_seq,
            n,
            deadline,
            start_beta=5.0,
            beta_mul=1.02,
            lr0=0.8,
            max_beta=200.0,
        )
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[overall] Adam improved global best (len={n}) = {refined_val:.6f}")

        # --------------------------------------------------------------
        # Refinement phase 2 – Projected sub‑gradient descent on exact max
        # --------------------------------------------------------------
        if time.time() < deadline:
            sub_seq, sub_val = _subgrad_refine(
                global_best_seq,
                n,
                deadline,
                max_iters=8000,
                step0=0.7,
                tol=1e-12,
            )
            if sub_val < global_best_val:
                global_best_val = sub_val
                global_best_seq = sub_seq.copy()
                print(f"[overall] Sub‑grad improved global best (len={n}) = {sub_val:.6f}")

        # --------------------------------------------------------------
        # Optional perturbation‑restart loop (if time permits)
        # --------------------------------------------------------------
        while time.time() < deadline:
            noisy_logits = np.log(global_best_seq + 1e-30) + 0.02 * np.random.randn(n)
            perturbed = _softmax(noisy_logits)
            # quick Adam on perturbed start (short‑run)
            p_seq, p_val = _adam_refine(
                perturbed,
                n,
                deadline,
                start_beta=5.0,
                beta_mul=1.02,
                lr0=0.8,
                max_beta=100.0,
            )
            if p_val < global_best_val:
                global_best_val = p_val
                global_best_seq = p_seq.copy()
                print(f"[perturb] improved global best to {p_val:.6f}")

        # --------------------------------------------------------------
        # End of this length – move to next if time permits
        # --------------------------------------------------------------

    # ------------------------------------------------------------------
    # Final sanitisation and return
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # completely fallback: uniform of a moderate size
        n = 4000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("[fallback] final uniform sequence")

    # Ensure exact normalisation (evaluation is scale invariant)
    s = global_best_seq.sum()
    if s > 0:
        global_best_seq = global_best_seq / s

    result = global_best_seq.tolist()
    print(f"[final] best objective (fast) = {global_best_val:.6f}")
    return result
