def propose_candidate(seed=123, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = int(kwargs.get("n", 6000))

    size_conv = _next_pow2(2*n-1)
    size_grad = _next_pow2(3*n-2)

    # Helper: compute objective given a (numpy vector of sum 1)
    def objective(a):
        conv = _conv_fft(a, a, size_conv)
        max_b = conv.max()
        return 2.0 * n * max_b

    # Param search for power law exponent
    best_seq = None
    best_val = float('inf')
    for p in np.linspace(0.1, 2.5, num=25):
        idx = np.arange(n)
        d = np.minimum(idx, n-1-idx)
        a = (d + 1) ** (-p)    # shape decreasing from edges
        a = a.astype(float)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a = a / a_sum
        val = objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    # Also try some random initial points
    for _ in range(5):
        a = np.random.rand(n)
        a = a / a.sum()
        val = objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

    print(f"[init] best after param search = {best_val:.6f}")

    # Now iterative exponentiated gradient
    a = best_seq.copy()
    a_rev = a[::-1].copy()
    it = 0
    last_improve = start

    base_eta = 0.5
    rand_prob = 0.05
    rand_scale = 0.02
    stagnation_restart = 30.0

    while time.time() < deadline:
        # compute convolution and max
        conv = _conv_fft(a, a, size_conv)
        max_b = conv.max()
        cur_val = 2.0 * n * max_b
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{it}] new best {best_val:.6f}")

        # randomness
        if np.random.rand() < rand_prob:
            # random mass transfer
            i, j = np.random.randint(n, size=2)
            while i == j:
                j = np.random.randint(n)
            # transfer a fraction
            delta = rand_scale * a[i] * np.random.rand()
            a[i] -= delta
            a[j] += delta
            a = a / a.sum()
            a_rev = a[::-1].copy()
            it += 1
            continue

        # compute weight for max indices
        eps = 1e-12 * max_b
        mask = conv >= (max_b - eps)
        w = np.zeros_like(conv)
        cnt = mask.sum()
        if cnt == 0:
            # fallback
            w[conv.argmax()] = 1.0
            cnt = 1
        w[mask] = 1.0 / cnt

        # gradient: g = 2 * (a_rev convolve w) (central n)
        g_full = _conv_fft(a_rev, w, size_grad)
        g = g_full[n-1 : n-1 + n]
        gradient = 2.0 * g   # derivative of max (if we assume max = convex combination)
        # step size
        eta = base_eta / np.sqrt(it + 1)
        # exponentiated update: a_i = a_i * exp(-eta * gradient_i)
        # To avoid overflow, clip gradient to range.
        grad_clip = np.clip(gradient, -100.0, 100.0)
        a = a * np.exp(-eta * grad_clip)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0/n)
        else:
            a = a / a_sum
        a_rev = a[::-1].copy()
        it += 1

        # restart if stagnation
        if time.time() - last_improve > stagnation_restart:
            a = np.random.rand(n)
            a = a / a.sum()
            a_rev = a[::-1].copy()
            last_improve = time.time()
            print(f"[{it}] restart due to stagnation")

    print(f"[done] it {it} best {best_val:.6f}")
    return best_seq.tolist()

"""
Propose a low‑value step‑function sequence.

The search consists of three stages:

1. **Parametric initialization** – try a few simple analytic families
   (power‑law decay from the edges, symmetric “polynomial” shapes
   a_i ∝ ((i+1)*(n‑i))**α, etc.).  The best of these is taken as
   the starting point.

2. **Exponentiated‑gradient descent** – the objective is
   `obj(a) = 2·n·max_k (a*a)_k`.  At each iteration we locate the
   (possibly several) indices where the convolution attains its
   maximum, build a weight vector `w` that distributes equal mass over
   those indices and compute the sub‑gradient
   ``g = 2·(a_rev ∗ w)``.  Updating the log‑weights
   (`log a ← log a − η g`) corresponds to a multiplicative update that
   respects non‑negativity; after each step the vector is renormalised
   to unit ℓ₁‑norm (the objective is scale‑invariant).

3. **Exploration & restarts** – with a small probability a cheap
   “mass‑transfer’’ move swaps a random fraction of mass between two
   coordinates; if the algorithm makes no improvement for a while it
   restarts from a fresh random point.

All convolutions are performed by FFT (O(n log n)) which makes it
possible to evaluate hundreds of thousands of candidates within the
1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution via FFT (full length = len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_conv(a: np.ndarray, size: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    return _conv_fft(a, a, size)


# ----------------------------------------------------------------------
# Objective (fast, no clipping – we keep the vector already non‑negative)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, n: int, size_conv: int) -> float:
    """Return 2·n·max(a*a) assuming sum(a)==1."""
    conv = _full_conv(a, size_conv)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence that minimises `evaluate_sequence`.
    The function runs for at most ``budget_s`` seconds and returns the
    best sequence found (list of Python floats, all non‑negative).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5           # safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    size_conv = _next_pow2(2 * n - 1)           # for a*a
    size_grad = _next_pow2(3 * n - 2)           # for a_rev * weight

    # --------------------------------------------------------------
    # helper: evaluate a candidate quickly
    # --------------------------------------------------------------
    def eval_fast(a_vec: np.ndarray) -> float:
        return _objective(a_vec, n, size_conv)

    # --------------------------------------------------------------
    # 1.  Parametric initialisation
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    idx = np.arange(n, dtype=np.float64)

    # ---- family A : distance to the nearer edge, power‑law decay ----
    d_edge = np.minimum(idx, n - 1 - idx)
    for p in np.linspace(0.05, 2.5, 25):
        a_raw = (d_edge + 1.0) ** (-p)
        a = a_raw / a_raw.sum()
        val = eval_fast(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()

    # ---- family B : symmetric polynomial shape ((i+1)*(n-i))^α ----
    base = (idx + 1.0) * (n - idx)
    log_base = np.log(base)
    for alpha in np.linspace(-1.8, 1.8, 37):
        a_raw = np.exp(alpha * log_base)
        a = a_raw / a_raw.sum()
        val = eval_fast(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()

    # ---- family C : sinusoidal shape (positive half‑sine) ----
    a_raw = np.sin(np.pi * (idx + 1.0) / (n + 1.0))
    a = a_raw / a_raw.sum()
    val = eval_fast(a)
    if val < best_val:
        best_val, best_seq = val, a.copy()

    # ---- a few random starts as safety net ----
    for _ in range(8):
        a = np.random.rand(n)
        a /= a.sum()
        val = eval_fast(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()

    print(f"[init] best after param sweep = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Exponentiated‑gradient descent on the simplex
    # --------------------------------------------------------------
    a = best_seq.copy()
    log_a = np.log(a + 1e-30)          # avoid log(0)
    a_rev = a[::-1].copy()
    it = 0
    last_improve = start

    # hyper‑parameters
    base_eta = 0.12                    # controls step size
    rand_prob = 0.07
    rand_scale = 0.025
    stagnation_limit = 28.0            # seconds without improvement → restart

    while time.time() < deadline:
        # ----- current objective -----
        conv_full = _full_conv(a, size_conv)
        max_b = conv_full.max()
        cur_val = 2.0 * n * max_b

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_val:.6f}   time={elapsed:.1f}s")

        # ----- stochastic mass‑transfer move (escape plateaux) -----
        if np.random.rand() < rand_prob:
            i, j = np.random.randint(n, size=2)
            while i == j:
                j = np.random.randint(n)
            if a[i] > 0.0:
                delta = rand_scale * a[i] * np.random.rand()
                a[i] -= delta
                a[j] += delta
                a /= a.sum()
                log_a = np.log(a + 1e-30)
                a_rev = a[::-1].copy()
                it += 1
                continue

        # ----- sub‑gradient of the max‑convolution -----
        eps = max_b * 1e-12
        mask = conv_full >= (max_b - eps)
        cnt = mask.sum()
        if cnt == 0:                     # safety fallback
            mask[conv_full.argmax()] = True
            cnt = 1
        weight = np.zeros_like(conv_full)
        weight[mask] = 1.0 / cnt          # equal weight on all active positions

        # gradient g = 2·(a_rev ∗ weight) (central n entries)
        g_full = _conv_fft(a_rev, weight, size_grad)
        g = g_full[n - 1 : n - 1 + n]     # central part, length n
        grad = 2.0 * g

        # ----- exponentiated update (mirror descent w.r.t. KL) -----
        eta = base_eta / np.sqrt(it + 1.0)
        # clip gradient to avoid overflow in the exponent
        grad_clip = np.clip(grad, -80.0, 80.0)
        log_a -= eta * grad_clip
        # re‑center log‑weights to keep numbers in a stable range
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n)
            log_a = np.log(a)
        else:
            a /= s
            log_a = np.log(a + 1e-30)

        a_rev = a[::-1].copy()
        it += 1

        # ----- restart if stuck for a while -----
        if time.time() - last_improve > stagnation_limit:
            a = np.random.rand(n)
            a /= a.sum()
            log_a = np.log(a)
            a_rev = a[::-1].copy()
            last_improve = time.time()
            print(f"[{it:7d}] restart after stagnation")

    # --------------------------------------------------------------
    # final report
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_val:.6f}, time={total:.1f}s")
    return best_seq.tolist()
