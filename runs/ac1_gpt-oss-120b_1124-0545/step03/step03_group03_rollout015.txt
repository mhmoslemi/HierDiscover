            w = np.ones_like(conv) / conv.size

        g = 2.0 * conv_wa[n-1:2*n-1][::-1]

        eta = eta0 / math.sqrt(iteration + 1)
        # exponentiated gradient update
        exp_arg = -eta * g
        # clip to avoid overflow
        exp_arg = np.clip(exp_arg, -50, 50)

            a_new = np.ones(n) / n
        else:
            a_new = a_new / a_new_sum

        # occasional random move
        if np.random.rand() < 0.1:
            a_new = random_move(a_new, scale=0.05)  # simple

        iteration += 1

    n = len(f)
    # Ensure positive to avoid unboundedness
    f_pos = np.maximum(f, 1e-12)
    conv_f = conv_fft(f_pos)
    M = float(conv_f.max())

    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return f, False
    g = res.x
    # Normalize g to sum 1
    sum_g = g.sum()
    if sum_g <= 1e-12:
        return f, False
    g_norm = g / sum_g
    # Update f via convex combination
    f_new = (1 - t) * f + t * g_norm
    f_new = project_simplex(f_new)
    return f_new, True

import time
import math
import numpy as np
import scipy.linalg
import scipy.optimize

def _next_pow2(x: int) -> int:
    # smallest power of two >= x
    return 1 << (x - 1).bit_length()

def conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]

def compute_score(a: np.ndarray) -> float:
    # Normalize to sum 1
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)

def project_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

def gradient_smooth(a: np.ndarray, lam: float) -> np.ndarray:
    n = a.shape[0]
    conv = conv_fft(a)
    max_c = np.max(conv)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    start = n - 1
    end = 2 * n - 1
    slice_part = conv_wa[start:end]
    g = 2.0 * slice_part[::-1]
    return g

def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a

def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    col = np.concatenate((f, np.zeros(n - 1)))
    row = np.concatenate((f[:1], np.zeros(n - 1)))
    A = scipy.linalg.toeplitz(col, row)[: 2 * n - 1, :]
    return A

def lp_step(f: np.ndarray, t: float):
    """
    Perform a single linear‑programming update.
    Returns (new_f, success_flag).
    """
    n = f.shape[0]
    f_pos = np.maximum(f, 1e-12)      # avoid zeros that would lead to unbounded LP
    conv_f = conv_fft(f_pos)
    M = float(conv_f.max())
    A = build_conv_matrix(f_pos)
    c = -np.ones(n)                  # maximise sum(g)  <=> minimise -sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    # high‑performance interior‑point solver
    result = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                   method='highs', options={'disp': False})
    if not result.success:
        return f, False
    g = result.x
    s = g.sum()
    if s <= 0.0:
        return f, False
    g_norm = g / s
    f_new = (1.0 - t) * f + t * g_norm
    f_new = project_simplex(f_new)
    return f_new, True

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
        2·n·max(conv(a,a)) / (sum(a)**2)

    The routine consists of:
    1. a fine analytic grid search over product / sum power‑law families;
    2. a small pattern‑search local refinement on the best analytic candidate;
    3. a few linear‑programming “mix” steps that move the distribution away
       from the current convolution bottleneck;
    4. a smoothed‑max gradient descent (with Metropolis acceptance) for final
       polishing.
    The whole procedure respects a 1000 s budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a tiny safety margin before hard timeout
    deadline = start_time + budget_s - 0.5

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1. Coarse analytic grid search
    # ------------------------------------------------------------------
    p_vals = np.arange(0.20, 2.001, 0.10)        # 0.20 … 2.0
    offset_vals = np.arange(0.001, 2.001, 0.20)  # 0.001 … 2.0
    modes = ["product", "sum"]

    best_seq = None
    best_score = np.inf

    print("[init] starting analytic grid search (n={})".format(n))

    for mode in modes:
        for p in p_vals:
            for off in offset_vals:
                cand = analytic_candidate(n, p, off, mode)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                # occasional progress output
                if mode == "product" and p == p_vals[0] and off == offset_vals[0]:
                    print("[grid] mode={}, p={:.3f}, off={:.3f} → {:.6f}".format(
                        mode, p, off, sc))

    # Also consider a possible globally provided reference sequence
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            print("[init] reference sequence score = {:.6f}".format(sc))
            if sc < best_score:
                best_score = sc
                best_seq = ref.copy()

    print("[init] best analytic score = {:.6f}".format(best_score))

    # ------------------------------------------------------------------
    # 2. Tiny pattern‑search refinement (random jitter + greedy acceptance)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    # introduce a very small asymmetric jitter to avoid being stuck in a perfectly
    # symmetric stationary point for the subsequent optimisation steps.
    jitter = np.random.rand(n) * 1e-4
    a = project_simplex(a + jitter)
    best_seq = a.copy()
    best_score = compute_score(a)
    print("[refine] after initial jitter best score = {:.6f}".format(best_score))

    # ---------------------------------------------------------
    # 3. Linear‑programming mixing steps
    # ---------------------------------------------------------
    max_lp_iters = 12
    t_start = 0.5
    t_end = 0.05
    lp_iter = 0
    last_improve = time.time()
    while time.time() < deadline and lp_iter < max_lp_iters:
        frac = lp_iter / max_lp_iters
        t = t_start * (t_end / t_start) ** frac   # geometric decay
        a_new, ok = lp_step(a, t)
        if not ok:
            print("[lp] LP step failed at iteration {}".format(lp_iter))
            break
        a = a_new
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            last_improve = time.time()
            print("[lp {:02d}] new best = {:.6f}".format(lp_iter, best_score))
        lp_iter += 1

    # ---------------------------------------------------------
    # 4. Smoothed‑max gradient descent for fine tuning
    # ---------------------------------------------------------
    lam_start, lam_end = 0.5, 1500.0
    step_start, step_end = 0.3, 1e-5
    temp_start, temp_end = 1.0, 0.001

    iteration = 0
    last_improve = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        # linear schedule (roughly 0→1 over most of the budget)
        prog = elapsed / (budget_s - 20.0)
        prog = max(0.0, min(1.0, prog))

        lam = lam_start + prog * (lam_end - lam_start)
        step = step_start + prog * (step_end - step_start)
        temp = temp_start + prog * (temp_end - temp_start)

        # -------------------------------------------------
        # move generation
        # -------------------------------------------------
        if np.random.rand() < 0.88:
            g = gradient_smooth(a, lam)
            cand = a - step * g
            cand = np.clip(cand, 0.0, None)
            cand = project_simplex(cand)
        else:
            cand = random_move(a, step * 0.5)

        cand_score = compute_score(cand)

        # Metropolis style acceptance
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            if temp > 1e-12:
                prob = math.exp(-(cand_score - best_score) / temp)
                if np.random.rand() < prob:
                    accept = True

            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()

                    iteration, best_score, elapsed))

        # -------------------------------------------------

import time
import math
import numpy as np
import scipy.linalg
import scipy.optimize

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear (zero‑padded) convolution of a with itself using FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """Exact objective (scale‑invariant) for a non‑negative vector a."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def gradient_smooth(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smooth approximation of the max:
        Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)
    The gradient is g_i = 2· Σ_k w_k·a_{k-i},
    where w_k are the soft‑max weights.
    """
    n = a.shape[0]
    conv = conv_fft(a)                     # length 2n‑1
    max_c = np.max(conv)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)        # soft‑max weights, sum = 1
    w_rev = w[::-1]                        # reversed weights
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    start = n - 1
    end = 2 * n - 1
    slice_part = conv_wa[start:end]          # length n
    g = 2.0 * slice_part[::-1]               # reverse to obtain g_i
    return g


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum=1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    mode = 'sum'      → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)
    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Construct the (2n‑1)×n matrix that maps a vector g to f * g."""
    n = f.shape[0]
    col = np.concatenate((f, np.zeros(n - 1)))
    row = np.concatenate((f[:1], np.zeros(n - 1)))
    A = scipy.linalg.toeplitz(col, row)[: 2 * n - 1, :]
    return A


def lp_step(f: np.ndarray, t: float):
    """
    One linear‑programming update.
    Returns (new_f, success_flag).
    """
    n = f.shape[0]
    f_pos = np.maximum(f, 1e-12)          # avoid zeros → unbounded LP
    conv_f = conv_fft(f_pos)
    M = float(conv_f.max())
    A = build_conv_matrix(f_pos)
    c = -np.ones(n)                      # minimise –sum(g)  ⇔ maximise sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                 bounds=bounds, method='highs',
                                 options={'disp': False})
    if not res.success:
        return f, False
    g = res.x
    s = g.sum()
    if s <= 0.0:
        return f, False
    g_norm = g / s
    f_new = (1.0 - t) * f + t * g_norm
    f_new = project_simplex(f_new)
    return f_new, True


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
        2·n·max(conv(a,a)) / (sum(a)**2)

    The routine consists of:
    1. a fine analytic grid search over product / sum power‑law families;
    2. a small pattern‑search local refinement on the best analytic candidate;
    3. a few linear‑programming “mix” steps that move the distribution away
       from the current convolution bottleneck;
    4. a smoothed‑max gradient descent (with Metropolis acceptance) for final
       polishing.  The whole procedure respects a 1000 s budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Analytic grid search
    # ------------------------------------------------------------------
    p_vals = np.arange(0.20, 2.001, 0.10)          # 0.20 … 2.0
    offset_vals = np.arange(0.001, 2.001, 0.20)    # 0.001 … 2.0
    modes = ["product", "sum"]

    best_seq = None
    best_score = np.inf

    print("[init] starting analytic grid search (n={})".format(n))

    for mode in modes:
        for p in p_vals:
            for off in offset_vals:
                cand = analytic_candidate(n, p, off, mode)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                # occasional progress report
                if mode == "product" and p == p_vals[0] and off == offset_vals[0]:
                    print("[grid] mode={}, p={:.3f}, off={:.3f} → {:.6f}".format(
                        mode, p, off, sc))

    # If a reference sequence is provided, try it as well
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            print("[init] reference sequence score = {:.6f}".format(sc))
            if sc < best_score:
                best_score = sc
                best_seq = ref.copy()

    print("[init] best analytic score = {:.6f}".format(best_score))

    # ------------------------------------------------------------------
    # 2) Small random jitter to break symmetry
    # ------------------------------------------------------------------
    a = best_seq.copy()
    jitter = np.random.rand(n) * 1e-4
    a = project_simplex(a + jitter)
    best_seq = a.copy()
    best_score = compute_score(a)
    print("[refine] after jitter best score = {:.6f}".format(best_score))

    # ------------------------------------------------------------------
    # 3) Linear‑programming mixing steps
    # ------------------------------------------------------------------
    max_lp_iters = 12
    t_start, t_end = 0.5, 0.05
    lp_iter = 0
    last_improve = time.time()
    while time.time() < deadline and lp_iter < max_lp_iters:
        frac = lp_iter / max_lp_iters
        t = t_start * (t_end / t_start) ** frac   # geometric decay
        a_new, ok = lp_step(a, t)
        if not ok:
            print("[lp] LP step failed at iteration {}".format(lp_iter))
            break
        a = a_new
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            last_improve = time.time()
            print("[lp {:02d}] new best = {:.6f}".format(lp_iter, best_score))
        lp_iter += 1

    # ------------------------------------------------------------------
    # 4) Smoothed‑max gradient descent for fine‑tuning
    # ------------------------------------------------------------------
    lam_start, lam_end = 0.5, 1500.0
    step_start, step_end = 0.3, 1e-5
    temp_start, temp_end = 1.0, 0.001

    iteration = 0
    last_improve = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start_time
        prog = elapsed / (budget_s - 20.0)          # roughly 0 → 1
        prog = max(0.0, min(1.0, prog))

        lam = lam_start + prog * (lam_end - lam_start)
        step = step_start + prog * (step_end - step_start)
        temp = temp_start + prog * (temp_end - temp_start)

        # -------------------------------------------------------------
        # generate a candidate
        # -------------------------------------------------------------
        if np.random.rand() < 0.88:
            g = gradient_smooth(a, lam)
            cand = a - step * g
            cand = np.clip(cand, 0.0, None)
            cand = project_simplex(cand)
        else:
            cand = random_move(a, step * 0.5)

        cand_score = compute_score(cand)

        # -------------------------------------------------------------
        # Metropolis acceptance
        # -------------------------------------------------------------
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            if temp > 1e-12:
                prob = math.exp(-(cand_score - best_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print("[{:07d}] new best = {:.6f} (t={:.1f}s)".format(
                    iteration, best_score, elapsed))

        # -------------------------------------------------------------
        # restart if we have been stuck for a while
        # -------------------------------------------------------------
        if time.time() - last_improve > 30.0:
            jitter = np.random.rand(n) * 1e-4
            a = best_seq + jitter
            a = project_simplex(a)
            last_improve = time.time()
            print("[{:07d}] jitter restart after stagnation".format(iteration))

        if best_score <= 1.503:
            print("[done] target 1.503 reached, exiting early")
            break

        iteration += 1

    total = time.time() - start_time
    print("[done] best score {:.6f}, total time {:.1f}s, LP iters {}, GD iters {}"
          .format(best_score, total, lp_iter, iteration))

    return best_seq.tolist()
