"""
Propose a step‑function (non‑negative heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm improves on the reference implementation by:

*   Adding several new seed families (sparse Dirichlet, symmetric exponential,
    bimodal Gaussian mixtures) to increase diversity.
*   Using a slower β‑schedule for the smooth‑max (β←β·1.03) and a more
    aggressive Adam learning‑rate schedule.
*   Adding a L‑BFGS refinement on the soft‑max parametrisation after Adam.
*   Performing a final mirror‑descent sub‑gradient refinement.
*   Enforcing symmetry as a cheap post‑processing step.
*   Trying a few different lengths (3000–8000) while respecting the overall
    time budget (default 1000 s).

All helper functions are top‑level, there are no closures or lambdas.
The routine respects the deadline and returns the best sequence found as a
plain list of floats.  The best sequence is also stored in the global variable
`height_sequence_1` for possible reuse in later calls.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy import sparse

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT.
    Returns the full result of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns probabilities that sum to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Fast objective (exact max, sum(a)=1)
# ----------------------------------------------------------------------


def _fast_objective(a: np.ndarray) -> float:
    """
    Return 2·n·max(conv(a,a)) for a already normalised to sum(a)=1.
    """
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Smoothed objective and its gradient (w.r.t. soft‑max logits)
# ----------------------------------------------------------------------


def _smooth_obj_grad(v: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Evaluate the smoothed objective

        f(v) = 2·n·logsumexp(beta·conv(a,a)) / beta ,   a = softmax(v)

    and its gradient w.r.t. the logits v.
    """
    a = _softmax(v)
    n = len(a)

    # convolution a * a (full length 2n-1)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)

    # smoothed max via log‑sum‑exp
    # shift by max_c to avoid overflow
    exp_terms = np.exp(beta * (conv - max_c))
    w_sum = np.sum(exp_terms)
    if w_sum == 0.0:          # numerical safety – fallback to uniform weights
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = exp_terms / w_sum

    # gradient of Σ w_k·conv_k w.r.t. a
    rev_a = a[::-1]
    # convolution rev_a * w (full length = 3n-2)
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n - 1:n - 1 + n]          # central n entries

    # chain rule: df/da = (2·n/β) * grad_a
    grad = (2.0 * n / beta) * grad_a

    # convert to gradient w.r.t. logits v using the soft‑max Jacobian
    dot = np.dot(grad, a)
    grad_v = a * (grad - dot)

    # objective value (log‑sum‑exp = max_c + log(w_sum))
    val = 2.0 * n * (max_c + np.log(w_sum) / beta)

    return float(val), grad_v.astype(np.float64)


# ----------------------------------------------------------------------
# Seed families (all returned normalised to sum 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _dirichlet_sparse(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet([alpha] * n)


def _exp_symmetric(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(idx - centre))
    return a / a.sum()


def _gaussian_mixture(n: int, sigma_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Two symmetric Gaussian bumps located at 1/3 and 2/3 of the interval."""
    i = np.arange(n, dtype=np.float64)
    centre1 = n / 3.0
    centre2 = 2.0 * n / 3.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - centre1) / sigma) ** 2)
    a += np.exp(-0.5 * ((i - centre2) / sigma) ** 2)
    return a / a.sum()


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Collect a modest set of diverse starting points."""
    seeds: list[np.ndarray] = []

    # Classic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):

        if r <= 0.0:

        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:

    # Random soft‑max vectors
    rng = np.random.default_rng(12345)

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 3000,
                   beta_start: float = 5.0,
                   beta_mul: float = 1.03) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation of the smooth‑max objective.
    Returns best sequence (as a numpy array) and its exact objective value.
    """
    v = np.log(seed + 1e-30)          # initialise logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = beta_start
    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)

        # exact objective for monitoring (sum(a)=1)
        cur_val = _fast_objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n} it={it:4d} best={cur_val:.6f}")

        # smoothed objective & gradient
        val_s, grad_v = _smooth_obj_grad(v, beta)

        # increase smoothing parameter
        beta *= beta_mul

        # Adam update on logits
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

    return best_seq, best_val

def _lbfgs_refine(v0: np.ndarray,

    Returns the refined sequence and its exact objective.
    """
    n = v0.shape[0]

    # Wrapper that returns value and gradient for scipy.minimize
    def fun(v):
        val, grad = _smooth_obj_grad(v, beta)
        return val, grad

    # time‑aware callback – stops early if we run out of budget
    def callback(xk):
        # No side‑effects needed; just a placeholder for future extensions.
        pass

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n} it={it:5d} best={cur_val:.6f}")

        # exact sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror‑descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                     # numerical safety
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – leave a second for the final print / return
    deadline = start_time + budget_s - 1.0

    # Load previously saved construction (if any)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # Lengths to explore – we start modestly and expand if time permits
    candidate_lengths = [3000, 4000, 6000, 8000]

    global_best_seq: np.ndarray | None = None
    global_best_val = float("inf")

    for n in candidate_lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length {n} ===")

        # ------------------------------------------------------------------
        # Seed generation
        # ------------------------------------------------------------------
        seeds = _make_seeds(n, previous) + _make_extra_seeds(n)

        # ------------------------------------------------------------------
        # Quick scan of raw seeds
        # ------------------------------------------------------------------
        for s in seeds:
            val = _fast_objective(s)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = s.copy()
                print(f"[seed] n={n} new global best={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Adam on smoothed max for each seed
        # ------------------------------------------------------------------
        for idx, seed_vec in enumerate(seeds):
            if time.time() > deadline:
                break
            seq, val = _adam_optimize(seed_vec, n, deadline,
                                      max_iters=2500,
                                      beta_start=5.0,
                                      beta_mul=1.03)
            if seq is not None and val < global_best_val:
                global_best_val = val
                global_best_seq = seq.copy()
                print(f"[overall] n={n} new global best (Adam)={val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – L‑BFGS refinement from a few promising starts
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            # we take the current global best and a couple of generic seeds
            ref_starts = [
                global_best_seq,
                _uniform(n),
                _sine_window(n),
            ]
            for start_seq in ref_starts:
                if time.time() > deadline:
                    break
                v0 = np.log(start_seq + 1e-30)
                # a modest β for the smooth objective (large β ≈ true max)
                beta = 10.0
                seq_ref, val_ref = _lbfgs_refine(v0, beta, deadline,
                                                 maxiter=200)
                if seq_ref is not None and val_ref < global_best_val:
                    global_best_val = val_ref
                    global_best_seq = seq_ref.copy()
                    print(f"[overall] n={n} new global best (L‑BFGS)={val_ref:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – Exact sub‑gradient refinement
        # ------------------------------------------------------------------
        if global_best_seq is not None and global_best_seq.shape[0] == n:
            seq_ref, val_ref = _subgradient_refine(
                global_best_seq, n, deadline, max_iters=12000)
            if seq_ref is not None and val_ref < global_best_val:
                global_best_val = val_ref
                global_best_seq = seq_ref.copy()
                print(f"[overall] n={n} new global best (SubGrad)={val_ref:.6f}")

        # ------------------------------------------------------------------
        # Symmetrisation post‑processing (cheap improvement sometimes)
        # ------------------------------------------------------------------
        if global_best_seq is not None:
            a_sym = (global_best_seq + global_best_seq[::-1]) / 2.0
            a_sym /= a_sym.sum()
            val_sym = _fast_objective(a_sym)
            if val_sym < global_best_val:
                global_best_val = val_sym
                global_best_seq = a_sym.copy()
                print(f"[sym] symmetric version improved to {val_sym:.6f}")

    # ----------------------------------------------------------------------
    # Final sanity check via the provided evaluate_sequence (optional)
    # ----------------------------------------------------------------------
    try:
        # ``evaluate_sequence`` is expected to be present in the global namespace.
        est = evaluate_sequence(global_best_seq.tolist())  # type: ignore
        print(f"Final evaluate_sequence check: {est}")
    except Exception:
        pass

    # Store for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist() if global_best_seq is not None else []

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq) if global_best_seq is not None else 0})")
    return global_best_seq.tolist() if global_best_seq is not None else []
