def propose_candidate(seed=0, budget_s=1000):
    import time, math, random, numpy as np
    set seeds.
    # config
    n = 5000
    target_sum = math.sqrt(2.0 * n)
    # initialize a
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        if base.size != n:
            # pad/truncate
            if base.size > n:
                base = base[:n]
            else:
                base = np.pad(base, (0, n - base.size), 'constant')
    else:
        base = np.random.exponential(scale=1.0, size=n)
    a = base / base.sum() * target_sum

    # Precalc fft_len
    fft_len = 1 << ((2*n - 1).bit_length())
    # time
    start = time.time()
    end = start + budget_s

    # Subgradient phase
    sub_end = start + budget_s * 0.55
    rel_tol = 0.005   # tolerance for active set
    alpha0 = target_sum * 0.05  # bigger step
    it = 0
    best_val, _ = _max_convolution(a, fft_len)
    best_seq = a.copy()
    last_improve_iter = 0

    while time.time() < sub_end:
        conv, max_val = _max_convolution_array(a, fft_len)  # returns full conv and max
        # track best
        if max_val < best_val:
            best_val = max_val
            best_seq = a.copy()
            last_improve_iter = it
            print(f"[SG {it}] new best max conv {best_val:.6f}")

        # active indices
        thresh = max_val - rel_tol * max_val
        active = np.where(conv >= thresh)[0]  # indices
        # compute gradient
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # vectorized: indices = np.arange(i_start, i_end+1)
            idx = np.arange(i_start, i_end + 1, dtype=int)
            # partner indices = k - idx
            g[idx] += 2.0 * a[k - idx]

        # step size
        step = alpha0 / math.sqrt(it + 1)
        a_candidate = a - step * g
        # projection
        a_candidate = _simplex_projection(a_candidate, target_sum)
        # ensure nonnegative (projection does)
        a = a_candidate
        it += 1

        # occasional perturbation
        if it % 5000 == 0:
            # random blend with fresh random vector
            rand_vec = np.random.exponential(scale=1.0, size=n)
            rand_vec = rand_vec / rand_vec.sum() * target_sum
            blend = 0.02  # 2% mixing
            a = (1 - blend) * a + blend * rand_vec
            a = _simplex_projection(a, target_sum)
            print(f"[SG {it}] perturbation applied")

        # stagnation restart
        if it - last_improve_iter > 20000:
            # restart with random vector
            a = np.random.exponential(scale=1.0, size=n)
            a = a / a.sum() * target_sum
            last_improve_iter = it
            print(f"[SG {it}] restart due to stagnation")

    # Simulated annealing phase
    # Use best_seq from subgrad as start
    a = best_seq.copy()
    cur_conv, cur_max = _max_convolution_array(a, fft_len)
    cur_val = cur_max
    best_val = cur_val
    best_seq = a.copy()
    sa_start = time.time()
    # temperature schedule
    T0 = best_val * 0.3
    Tmin = best_val * 1e-6
    # step counter
    sa_iter = 0
    while time.time() < end:
        # update temperature
        elapsed = time.time() - sa_start
        frac = elapsed / (end - sa_start + 1e-12)
        T = T0 * ((Tmin / T0) ** frac)
        # propose move
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # delta: maybe small or random

            a = a_cand
            cur_val = cand_max

            print(f"[SA {sa_iter}] reset to best for exploration")

    # final return

    # evaluate final ratio via evaluate_sequence for debugging
    # optionally call evaluate_sequence before return (not needed)
    return best_seq.tolist()

"""
Propose a step‑function sequence that (almost) minimizes
evaluate_sequence().

Strategy
--------
* Work with a fixed length `n` (default 10 000).  The objective for a
  sequence with ℓ₁‑norm √(2 n) equals the maximal self‑convolution entry,
  because `evaluate_sequence` returns `2·n·max(conv)/(√(2·n))² = max(conv)`.
* **Phase 1 – adaptive projected sub‑gradient descent**.
  The sub‑gradient of `max_k (a*a)[k]` is a convex combination of the
  gradients of the (few) active convolution entries.  We therefore
  collect all indices whose convolution value is within a small relative
  tolerance of the current maximum and form the averaged sub‑gradient.
  An adaptive step size (back‑tracking) together with projection onto the
  simplex (ℓ₁‑norm = √(2·n), a_i ≥ 0) yields fast reduction of the
  objective.
* **Phase 2 – Simulated annealing**.
  Starting from the best vector of Phase 1 we perform many small mass‑
  transfer moves (`a_i → a_j`).  Moves that improve the maximum convolution
  are always accepted; uphill moves are accepted with probability
  `exp(−Δ/T)`.  The temperature decays exponentially.
* **Phase 3 – fine‑grained polishing**.
  A short run of the sub‑gradient method with a very small step size
  cleans up the remaining noise.

All phases respect a total wall‑time of 1000 s.  The routine returns a
plain Python list suitable for `evaluate_sequence`.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project vector v onto the simplex  { w ≥ 0 , Σ w = s }.
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """
    Smallest power‑of‑two ≥ 2·n‑1, used for efficient FFT convolutions.
    """
    return 1 << ((2 * n - 1).bit_length())


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear self‑convolution a*a using an FFT of length fft_len.
    Returns the full array of length 2·len(a)‑1 (real values).
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
def _max_convolution_info(a: np.ndarray, fft_len: int):
    """
    Return (full_conv, max_value, argmax_index).
    """
    conv = _conv_fft(a, fft_len)
    idx = int(np.argmax(conv))
    val = float(conv[idx])
    return conv, val, idx


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimizes the
    evaluation function.  The algorithm consists of three phases:
    (1) adaptive projected sub‑gradient descent on the exact max‑convolution,
    (2) simulated annealing, and
    (3) a short polishing sub‑gradient run.
    """
    np.random.seed(seed)
    random.seed(seed)

    # --------------------------------------------------------------
    # Problem size – larger n gives a lower theoretical bound.
    n = 10000                      # you may adjust; 10 000 fits comfortably
    target_sum = math.sqrt(2.0 * n)   # ℓ₁‑norm required by the objective

    # --------------------------------------------------------------
    # Initialise a.  If a previous construction exists, embed it,
    # otherwise start from a random exponential vector.
    if "height_sequence_1" in globals():
        old = np.asarray(height_sequence_1, dtype=float)
        base = np.zeros(n, dtype=float)
        k = min(old.size, n)
        base[:k] = old[:k]
        # add a tiny random bump to avoid a degenerate start
        base += np.random.exponential(scale=0.1, size=n)
    else:
        base = np.random.exponential(scale=1.0, size=n)

    a = base.astype(float)
    a = a / a.sum() * target_sum          # project onto the simplex

    # --------------------------------------------------------------
    # FFT length (kept constant throughout)
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # Phase 1 – adaptive projected sub‑gradient descent
    start_time = time.time()
    deadline = start_time + budget_s
    sub_end = start_time + budget_s * 0.45   # about 45 % of the time budget

    # hyper‑parameters

    alpha0 = target_sum * 0.07   # initial step size (≈ 7 % of total mass)
    min_alpha = target_sum * 1e-6
    it = 0

    conv, cur_max, _ = _max_convolution_info(a, fft_len)
    best_val = cur_max
    best_seq = a.copy()

    print(f"[INIT] n={n}, target_sum={target_sum:.3f}, init max(conv)={cur_max:.6f}")

    while time.time() < sub_end:
        # --- compute convolution and active set ---
        conv, cur_max, cur_k = _max_convolution_info(a, fft_len)

        # record best
        if cur_max < best_val:

            best_seq = a.copy()
            last_improve = it
            print(f"[SG {it:6d}] new best max(conv) = {best_val:.6f}")

        # select all indices whose value is within rel_tol of the maximum
        thresh = cur_max * (1.0 - rel_tol)

        step = max(alpha0 / math.sqrt(it + 1.0), min_alpha)

        # try the step; if it does not improve, halve it (up to 5 attempts)

            if cand_max <= cur_max:                # accept

                cur_max = cand_max
                break
            else:

        # optional: tiny random perturbation every few thousand iterations
        if it % 4000 == 0 and it > 0:

        # restart if we have not improved for a while

            last_improve = it

        it += 1

    # --------------------------------------------------------------

    a = best_seq.copy()
    _, cur_max, _ = _max_convolution_info(a, fft_len)
    best_val = cur_max
    best_seq = a.copy()

    sa_start = time.time()
    sa_end = deadline                     # run until the overall deadline

    Tmin = best_val * 1e-6                # final temperature

        # --- temperature schedule (exponential cooling) ---
        elapsed = time.time() - sa_start
        frac = elapsed / (sa_end - sa_start + 1e-12)
        T = T0 * ((Tmin / T0) ** frac)

        # --- propose a random transfer of mass ---
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # move at most 5 % of the source mass
        delta = a[i] * 0.05 * np.random.rand()
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta

        # evaluate
        _, cand_max, _ = _max_convolution_info(a_cand, fft_len)

        # --- Metropolis acceptance ---
        if cand_max < cur_max:

            cur_max = cand_max
            if cand_max < best_val:
                best_val = cand_max
                best_seq = a_cand.copy()
