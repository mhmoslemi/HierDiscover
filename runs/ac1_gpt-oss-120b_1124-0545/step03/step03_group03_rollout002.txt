    max_x = np.max(x)

"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Method
------
1. Build a small set of analytic candidates (power‑law families) and keep the
   best one.
2. Represent the distribution a on the probability simplex via a soft‑max
   of an unconstrained vector x (a_i = exp(x_i)/∑exp(x)).
3. Minimise the smooth approximation of the max‑convolution
       Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)
   using a continuation on λ (0.5 → 2000) and a limited‑memory BFGS
   optimiser (SciPy).  The gradient of Φ_λ is computed analytically with
   FFTs and the soft‑max Jacobian maps it to the x‑space.
4. After each λ‑stage evaluate the true objective (2·n·max(conv)) and keep
   the best distribution seen.
5. Perform a short random‑mass‑transfer local search to escape any remaining
   plateaus.

The whole routine respects a 1000 s budget, prints progress and returns the
best sequence found.
"""

import time
import math
import numpy as np

# Low‑level utilities (top‑level, no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a on the simplex.
    Returns 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft_self(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum=1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    mode = 'sum'      → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def compute_phi_and_grad(x: np.ndarray, lam: float, n: int):
    """
    For a given soft‑max parameter vector x, smoothing λ and length n,
    return the smooth max Φ_λ(a) and its gradient w.r.t. x.
    """
    # ----- a = softmax(x) -----
    m = np.max(x)
    exp_x = np.exp(x - m)
    a = exp_x / np.sum(exp_x)

    # ----- convolution a*a -----
    conv = conv_fft_self(a)                # length 2n‑1
    max_c = np.max(conv)

    # ----- soft‑max over convolution entries -----
    shift = conv - max_c                    # improve stability
    exp_term = np.exp(lam * shift)
    sum_exp = np.sum(exp_term)
    phi = max_c + np.log(sum_exp) / lam     # Φ_λ

    w = exp_term / sum_exp                  # weights, sum to 1

    # ----- gradient of Φ_λ w.r.t. a (∂Φ/∂a) -----
    w_rev = w[::-1]                        # reversed weights
    size = _next_pow2(len(w_rev) + n - 1)   # safe FFT length
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)
    # slice the n entries that correspond to the gradient
    slice_part = conv_wa[n - 1: 2 * n - 1]  # length n
    grad_a = 2.0 * slice_part[::-1]        # ∂Φ/∂a

    # ----- map gradient to x‑space using soft‑max Jacobian -----
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)            # ∇_x Φ = J^T ∇_a Φ

    return phi, grad_x

# Main entry point required by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function of length ``n`` (default 2000) that minimises
    the evaluation function.  The routine:

    * builds a small grid of analytic families and keeps the best;
    * runs a smoothed‑max L‑BFGS optimisation with a continuation on λ;
    * performs a short random‑mass‐transfer local search;
    * returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Analytic grid search – provides a solid starting point
    # ------------------------------------------------------------------
    candidates = []

    # power‑law families (product and sum modes)
    for mode in ("product", "sum"):
        for p in (0.45, 0.5, 0.55):
            for off in (0.0, 0.2, 0.5, 1.0):
                off_eff = off if (mode == "sum" or off > 0.0) else 1e-6
                cand = analytic_candidate(n, p, off_eff, mode)
                candidates.append(cand)

    # uniform distribution
    candidates.append(np.ones(n) / n)

    # include the globally‑provided construction if it matches the length
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            s = ref.sum()
            if s > 0:
                candidates.append(ref / s)

    best_a = None
    best_score = np.inf

    for a0 in candidates:
        sc = compute_score(a0)
        if sc < best_score:
            best_score = sc
            best_a = a0.copy()
    print(f"[init] best analytic start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Continuation L‑BFGS on the smoothed max‑convolution
    # ------------------------------------------------------------------
    # Convert the best start into soft‑max parameters
    eps = 1e-12
    x = np.log(np.maximum(best_a, eps))

    # λ schedule (geometric progression from smooth to hard max)
    lam_vals = [0.5, 2.0, 10.0, 50.0, 200.0, 800.0, 2000.0]

    # Try to use SciPy's L‑BFGS‑B; fall back to a manual Adam optimizer if unavailable
    try:
        from scipy.optimize import minimize
        have_scipy = True
    except Exception:                     # pragma: no cover
        have_scipy = False
        print("[info] SciPy not available – will use a simple Adam optimizer instead.")

    for lam in lam_vals:
        if time.time() >= deadline:
            break

        if have_scipy:
            # Heuristic: set max iterations according to remaining time
            remaining = deadline - time.time()
            max_iter = int(min(400, max(20, remaining * 80)))   # very cheap per iteration
            res = minimize(
                compute_phi_and_grad,
                x,
                args=(lam, n),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': max_iter, 'disp': False}
            )
            x = res.x
            a = softmax(x)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                elapsed = time.time() - start_time
                print(f"[lam={lam:.1f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")
        else:
            # ----- simple Adam optimiser (fallback) -----
            alpha = 0.05
            beta1, beta2 = 0.9, 0.999
            eps_opt = 1e-8
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            t = 0
            while time.time() < deadline:
                t += 1
                phi, grad_x = compute_phi_and_grad(x, lam, n)
                m = beta1 * m + (1 - beta1) * grad_x
                v = beta2 * v + (1 - beta2) * (grad_x ** 2)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                step = alpha * m_hat / (np.sqrt(v_hat) + eps_opt)
                x = x - step
                a = softmax(x)
                sc = compute_score(a)
                if sc < best_score:
                    best_score, best_a = sc, a.copy()
                    elapsed = time.time() - start_time
                    print(f"[adam lam={lam:.1f}] new best = {best_score:.6f} (t={elapsed:.1f}s)")
                if t >= 3000:                      # safety stop for this λ
                    break

    # ------------------------------------------------------------------
    # 3) Light random‑mass‑transfer local search (escape possible plateaus)
    # ------------------------------------------------------------------
    a = best_a.copy()
    scale = 0.02  # relative amount moved per trial
    while time.time() < deadline:
        cand = random_move(a, scale)
        sc = compute_score(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            elapsed = time.time() - start_time
            print(f"[local] new best = {best_score:.6f} (t={elapsed:.1f}s)")
            # after an improvement shrink the step size a little
            scale = max(0.001, scale * 0.9)
        else:
            # slowly increase step size again to keep exploration alive
            scale = min(0.05, scale * 1.001)

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, total time = {total_time:.1f}s")
    return best_a.tolist()
