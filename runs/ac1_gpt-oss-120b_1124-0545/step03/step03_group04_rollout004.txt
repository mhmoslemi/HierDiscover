"""
Propose a step‑function (non‑negative sequence) with a low value of
evaluate_sequence().  The routine combines a cheap parametric grid search
(power‑law, exponential and Beta‑shaped sequences) with a simple
simulated‑annealing refinement.  All candidates are normalised to a sum
of 1, because the objective 2·n·max(conv) is homogeneous and the scale
does not matter.  The algorithm respects the allotted time budget and
keeps the best sequence found.

Key ideas
---------
* The optimisation target is 2·n·max(a∗a) for a on the probability simplex.
* Simple analytical families (power‑law, exponential, Beta) already
  contain very good shapes – the grid search can locate a good starting
  point (the known best 1.517… comes from a power‑law with α≈0.5).
* A fast FFT based convolution (full length) supplies the exact objective
  without calling evaluate_sequence() for every trial.
* After the grid phase a short simulated‑annealing / random‑swap hill‑climb
  explores the neighbourhood of the best candidate, escaping shallow
  plateaus and usually giving a modest improvement.
* The code is pure NumPy, uses only top‑level functions and no closures,
  and fits easily into the 1000 s budget.

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Small helper – return the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full 1‑D convolution of real vectors using FFT."""
    size = a.size + b.size - 1
    N = _next_pow_two(size)
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    conv = np.fft.irfft(C, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^{-α}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _exp_sequence(n: int, beta: float) -> np.ndarray:
    """a_i ∝ exp(-β·i), normalised to sum = 1."""
    a = np.exp(-beta * np.arange(n, dtype=float))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _beta_sequence(n: int, p: float, q: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^{p-1}·(n-i)^{q-1}, i=0…n‑1,
    normalised to sum = 1.  p,q>0 produce U‑shaped (p,q<1) or bell‑shaped
    (p,q>1) families.
    """
    i = np.arange(1, n + 1, dtype=float)            # i in [1,n]
    a = (i ** (p - 1.0)) * ((n - i + 1) ** (q - 1.0))
    s = a.sum()
    if s == 0.0 or np.isnan(s):
        return np.full(n, 1.0 / n)
    return a / s


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence for a normalised vector
    (sum == 1).  Returns 2·n·max(conv(a,a)).
    """
    n = a.size
    conv = _conv_fft(a, a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s``
    seconds and returns the best sequence discovered.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  A small safety margin is kept.
    kwargs : dict
        Ignored – present for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence (non‑negative, sum > 0.01) found.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a few seconds free for the harness
    deadline = start + budget_s - 5.0

    # --------------------------------------------------------------
    # 1.  Determine a set of lengths to explore.
    # --------------------------------------------------------------
    # We try a single length that is large enough to give a good bound.
    # If the global variable height_sequence_1 is available we keep its
    # length, otherwise we start from n≈5000 (the size used previously).
    try:
        init = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n0 = init.size
    except Exception:
        n0 = 5000

    # A few additional lengths are cheap enough to test – they may give
    # a tiny edge for particular shapes.
    n_candidates = [n0]
    # add a larger length if time permits (checking later)
    if budget_s > 300 and n0 < 12000:
        n_candidates.append(min(12000, n0 * 2))

    best_score = float("inf")
    best_seq = None

    # --------------------------------------------------------------
    # 2.  Parameter grid search (≈20 % of the budget)
    # --------------------------------------------------------------
    grid_deadline = start + 0.2 * (budget_s - 5.0)   # reserve 20 % for grid

    # Parameter grids – coarse but sufficient for a quick scan.
    alphas = np.linspace(0.2, 1.2, 21)          # power‑law exponent
    offsets = np.linspace(0.0, 5.0, 11)        # offset for power‑law
    betas = np.linspace(0.0, 10.0 / n0, 21)    # exponential decay
    ps = np.linspace(0.2, 2.0, 10)            # beta‑distribution p
    qs = np.linspace(0.2, 2.0, 10)            # beta‑distribution q

    for n in n_candidates:
        # ---- power‑law (with and without offset) -----------------
        for alpha in alphas:
            if time.time() >= grid_deadline:
                break
            a = _power_sequence(n, alpha)
            score = _objective(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                # Debug output
                print(f"[grid] n={n}, power α={alpha:.3f} → score={score:.6f}")

        for alpha in alphas:
            for off in offsets:
                if time.time() >= grid_deadline:
                    break
                a = _power_sequence(n, alpha, offset=off)
                score = _objective(a)
                if score < best_score:
                    best_score, best_seq = score, a.copy()
                    print(f"[grid] n={n}, power α={alpha:.3f}, off={off:.1f} → score={score:.6f}")

        # ---- exponential -----------------------------------------
        for beta in betas:
            if time.time() >= grid_deadline:
                break
            a = _exp_sequence(n, beta)
            score = _objective(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[grid] n={n}, exp β={beta:.6f} → score={score:.6f}")

        # ---- beta‑shaped families --------------------------------
        for p in ps:
            for q in qs:
                if time.time() >= grid_deadline:
                    break
                a = _beta_sequence(n, p, q)
                score = _objective(a)
                if score < best_score:
                    best_score, best_seq = score, a.copy()
                    print(f"[grid] n={n}, beta p={p:.2f}, q={q:.2f} → score={score:.6f}")

    # If the grid phase did not produce any candidate (should never happen)
    if best_seq is None:
        # fallback to a simple uniform distribution
        n = n0
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # --------------------------------------------------------------
    # 3.  Local refinement – simulated annealing / random swaps
    # --------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    # Simple temperature schedule
    temperature = 0.01 * cur_score if cur_score > 0 else 1e-3
    # Decay factor – slow enough to keep the search alive
    temp_decay = 0.999995

    # Main loop – runs until the global deadline.
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        n = cur_seq.size

        # propose a random mass transfer
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        # move a fraction (up to 30 % initially) of the mass from i → j
        max_frac = 0.30
        delta = cur_seq[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # no need to renormalise – total mass unchanged
        new_score = _objective(new_seq)

        # accept if improvement or with a Metropolis probability
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
            cur_seq, cur_score = new_seq, new_score
            # keep a global best
            if new_score < best_score:
                best_score, best_seq = new_score, new_seq.copy()
                print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # cool down
        temperature = max(temperature * temp_decay, 1e-8)

        # occasional larger perturbation to jump out of flat regions
        if iteration % 2000 == 0:
            # draw a fresh random vector from the best‑known family
            n = best_seq.size
            cand = _power_sequence(n, alpha=0.5)  # a familiar shape
            cand_score = _objective(cand)
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                cur_seq, cur_score = cand.copy(), cand_score
                print(f"[reset] iter={iteration}, new seed score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Return the best sequence (as a plain list)
    # --------------------------------------------------------------
    # The evaluation routine will clip negative entries to zero,
    # but we keep everything non‑negative already.
    print(f"[final] best_score={best_score:.6f}, iterations={iteration}")
    return best_seq.tolist()
