"""
Propose a candidate step‐function sequence.

The algorithm works in three stages:

1. **Seed generation** – a large, diverse collection of analytically motivated
   probability vectors (geometric, power‑law, geometric×power, symmetric,
   sine, logistic, block‑uniform, Dirichlet, random softmax) together with the
   previously discovered best sequence (if any).

2. **Fast scan** – evaluate the exact objective
   ``2·n·max(conv(a,a))`` for every seed using an FFT based convolution; keep
   the best seed.

3. **Refinement** – first run an Adam optimiser on the log‑softmax
   parametrisation with a temperature schedule (smooth approximation of the
   maximum).  After the Adam phase, a short projected sub‑gradient descent on
   the *exact* max‑convolution is performed to fine‑tune the solution.
   Small random perturbations and restarts are used while time permits.

All constraints (non‑negative entries, sum‑to‑one) are enforced by a
simple Euclidean projection onto the probability simplex.  The routine respects
the supplied time budget (default 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities (FFT, softmax, projections, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Output length is len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (produces a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative; fallback to uniform.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric series a_i = r^i, normalized."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i = 1/(i+1)^p, normalized."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined geometric × power‑law."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric (two mirrored tails)."""
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _logistic_shape(n: int, alpha: float) -> np.ndarray:
    """Logistic (sigmoid) shape centred at n/2."""
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(alpha * i))
    return a / a.sum()


def _block_uniform(n: int, front_frac: float) -> np.ndarray:
    """Uniform mass on two edge blocks of total width front_frac * n each."""
    block = int(np.round(front_frac * n / 2.0))
    a = np.zeros(n, dtype=np.float64)
    if block > 0:
        a[:block] = 1.0
        a[-block:] = 1.0
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def compute_fast_obj(seq: np.ndarray, n: int) -> float:
    """Return the fast objective 2·n·max(conv(seq,seq))."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Refinement using Adam on a smooth approximation of the max
# ----------------------------------------------------------------------


def _refine_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 5.0,
    beta_mul: float = 1.02,
    lr0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.
    Returns the best sequence encountered and its fast objective value.
    """
    v = np.log(a0 + 1e-30)           # logits
    m = np.zeros_like(v)            # first moment
    v2 = np.zeros_like(v)           # second moment
    best_seq = a0.copy()
    best_val = compute_fast_obj(a0, n)

    beta = start_beta
    it = 0

    while time.time() < deadline:
        it += 1

        # Current probability vector (always lies on the simplex)
        a = _softmax(v)

        # Convolution and objective
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = float(2.0 * n * cur_max)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ---- Smoothed‑max weights (log‑sum‑exp approximation) ----
        beta *= beta_mul
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- Gradient wrt a (smooth loss) ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]          # central n entries

        # ---- Gradient wrt logits v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam update ----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / math.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Refinement using a true sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def _refine_subgradient(
    a0: np.ndarray,
    n: int,
    deadline: float,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    Returns best sequence seen and its objective.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = compute_fast_obj(a, n)
    it = 0

    while time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = float(2.0 * n * cur_max)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[subgrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # Weight vector w = uniform over all maximising indices
        K = np.where(conv == cur_max)[0]
        w = np.zeros_like(conv)
        if K.size > 0:
            w[K] = 1.0 / K.size

        # Gradient of the max (sub‑gradient)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad = 2.0 * full[n - 1 : n - 1 + n]  # central n entries

        # Simple diminishing step‑size
        eta = eta0 / math.sqrt(it)

        # Gradient step + projection onto simplex
        a = project_simplex(a - eta * grad)

    return best_seq, best_val

        perturbed_start = _softmax(noisy_logits)

        # Short Adam run for the perturbed start (use remaining time)
        now = time.time()
        if now >= deadline:
            break
        pert_deadline = min(deadline, now + 15.0)  # at most 15 s per restart
        seq2, val2 = _refine_adam(
            perturbed_start,
            n,
            pert_deadline,
            start_beta=5.0,
            beta_mul=1.02,
            lr0=0.8,
        )
        if val2 < best_val:
            best_val = val2
            best_seq = seq2.copy()
            print(f"[perturb] improved global best to {val2:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    # Normalise to sum‑to‑one (the evaluation function is scale‑invariant)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total
    else:
        best_seq = np.full_like(best_seq, 1.0 / n)

    result = best_seq.tolist()
    print(f"[final] best objective (fast) = {best_val:.6f}")
    return result
