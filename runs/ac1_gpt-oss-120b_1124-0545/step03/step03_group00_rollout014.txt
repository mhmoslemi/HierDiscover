def propose_candidate(budget_s=1000, seed=42):

    start = time.time()
    deadline = start + budget_s - 5.0

    lengths = [20000, 12000, 8000]  # try larger sizes first.

    global_best_seq = None; global_best_val = float('inf')

    for n in lengths:
        if time.time() > deadline: break
        print('Optimising length', n)

        # generate seeds
        seeds = []

        # arcsine (beta 0.5,0.5) if n>2
        try:
            i = np.arange(n, dtype=np.float64)
            a = 1.0/np.sqrt((i+0.5) * (n - i - 0.5))
            a = a / a.sum()
            seeds.append(a)
        except Exception:
            pass

        # beta distribution seeds
        alphas = [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]
        betas = alphas
        for alpha in alphas:
            for beta in betas:
                # compute a_i ∝ (i+1)^(alpha-1) * (n-i)^(beta-1)
                i = np.arange(n, dtype=np.float64) + 1.0
                a = (i ** (alpha-1)) * ((n - i + 1) ** (beta-1))
                # any infinities due to negative exponent at i=0?
                a[np.isinf(a)] = 0.0
                s = a.sum()
                if s > 0:
                    a = a / s
                    seeds.append(a)

        # geometric families
        for c in np.linspace(0.5, 5.0, 10):
            r = 1.0 - c / n
            if r > 0:
                seeds.append(_geom(n, r))

        # power families
        for p in np.linspace(0.0, 3.0, 10):
            seeds.append(_power(n, p))

        # combined geometric-power families
        for c in np.linspace(0.5, 4.0, 6):
            r = 1.0 - c / n
            if r <= 0: continue
            for p in np.linspace(0.0, 2.0, 6):
                seeds.append(_geom_power(n, r, p))

        # symmetric geometric
        for c in np.linspace(0.5, 4.5, 6):
            r = 1.0 - c / n
            if r > 0:
                seeds.append(_sym_geom(n, r))

        # sine window
        seeds.append(_sine_window(n))

        # random Dirichlet samples
        for alpha in [0.1, 0.3, 0.6, 1.5, 3.0]:
            a = np.random.gamma(alpha, 1.0, size=n)
            total = a.sum()
            if total > 0:
                seeds.append(a / total)

        # random softmax seeds
        for _ in range(5):
            v_rand = np.random.randn(n)
            seeds.append(_softmax(v_rand))

        # include previous global best (maybe from previous length)
        if global_best_seq is not None:
            # adjust to length n
            a_prev = np.asarray(global_best_seq, dtype=np.float64)
            if a_prev.shape[0] != n:
                a_adj = np.zeros(n, dtype=np.float64)
                m = min(a_prev.shape[0], n)
                a_adj[:m] = a_prev[:m]
                a_prev = a_adj
            # renormalize
            s = a_prev.sum()
            if s > 0:
                a_prev = a_prev / s
                seeds.append(a_prev)

        # quick scan
        for idx, a in enumerate(seeds):
            cur_val = 2.0 * n * np.max(_conv_fft(a, a))
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[scan] new global best from seed {idx} = {cur_val}")

        # refinement using advanced optimizer
        if global_best_seq is None:
            global_best_seq = np.ones(n) / n
        # Ensure correct length sum=1
        seq = global_best_seq.copy()
        seq = seq / seq.sum()
        refined_seq, refined_val = _refine_sequence_advanced(seq, n, deadline,
                                                             start_beta=0.5,
                                                             beta_mul=1.02,
                                                             lr0=0.8,
                                                             start_lambda_mse=0.5,
                                                             lambda_mse_decay=0.999,
                                                             lambda_ent=0.05)
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[refine] improved best to {refined_val}")

        # perturb restarts while time remains
        while time.time() < deadline:
            noisy_logits = np.log(global_best_seq + 1e-30) + 0.02 * np.random.randn(n)
            pert_seq = _softmax(noisy_logits)
            seq2, val2 = _refine_sequence_advanced(pert_seq, n, deadline,
                                                   start_beta=0.5,
                                                   beta_mul=1.02,
                                                   lr0=0.8,
                                                   start_lambda_mse=0.5,
                                                   lambda_mse_decay=0.999,
                                                   lambda_ent=0.05)
            if val2 < global_best_val:
                global_best_val = val2
                global_best_seq = seq2.copy()
                print(f"[perturb] improved global best to {val2}")

    # final normalisation
    if global_best_seq is None:
        n = 8000
        global_best_seq = np.ones(n)/n

    global_best_seq = np.maximum(global_best_seq, 0.0)
    s = global_best_seq.sum()
    if s > 0:
        global_best_seq = global_best_seq / s

    result = global_best_seq.tolist()
    print(f"[final] objective approx = {global_best_val}")
    return result

"""
Improved optimiser for the step‑function evaluation problem.

The optimiser works as follows:

1. **Rich seed set** –  a collection of analytically‑motivated
   probability vectors (geometric, power‑law, beta‑family, arcsine,
   symmetric, sine‑window) together with a few random Dirichlet draws
   and soft‑max initialisations.

2. **Fast scan** – each seed is evaluated exactly (2·n·max(conv)) and the
   best one is kept.

3. **Advanced refinement** – a projected Adam optimiser on the
   log‑softmax parametrisation.  The loss is a weighted sum of a
   smoothed ‑max (log‑sum‑exp) term and a mean‑squared‑error term that
   encourages the convolution to be flat (close to its average value).
   An optional entropy regulariser further spreads the mass.  The
   smoothing parameter β is increased geometrically, while the MSE
   weight is exponentially decayed, so the optimiser first smooths the
   landscape and finally tightens on the true maximum.

4. **Perturb‑restart loop** – after the main run, a few noisy restarts
   are performed to escape possible local minima.

The whole routine respects a user supplied time budget (default 1000 s)
and returns the best sequence found as a plain list of Python floats.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic linear‑algebra / convolution helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution via FFT.  Output length is len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_dist(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (beta - 1))
    a[np.isinf(a)] = 0.0
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _arcsine_dist(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine law (β=0.5,α=0.5)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return a / a.sum()


# ----------------------------------------------------------------------
# Advanced Adam optimiser on the log‑softmax parametrisation
# ----------------------------------------------------------------------


def _refine_sequence_advanced(
    a0: np.ndarray,
    n: int,
    deadline: float,
    *,
    start_beta: float = 0.5,
    beta_mul: float = 1.02,
    lr0: float = 0.8,
    start_lambda_mse: float = 0.5,
    lambda_mse_decay: float = 0.999,
    lambda_entropy: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the (smoothed‑max + MSE) loss.
    Returns the best sequence found and its fast objective value.
    """
    # initialise logits (log‑softmax)
    v = np.log(a0 + 1e-30)
    m = np.zeros_like(v)          # first moment
    v2 = np.zeros_like(v)         # second moment

    best_seq = a0.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a0, a0))

    beta = start_beta
    lambda_mse = start_lambda_mse
    it = 0
    avg_conv = 1.0 / (2 * n - 1)   # average value of convolution for any sum‑1 vector

    while time.time() < deadline:
        it += 1

        # current probability vector
        a = _softmax(v)

        # convolution and current objective
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ---- smoothed‑max weights ----
        beta *= beta_mul
        shift = cur_max
        w_max = np.exp(beta * (conv - shift))
        w_sum = w_max.sum()
        if w_sum == 0.0:                     # numerical under‑flow – skip this iteration
            continue
        w_max /= w_sum                          # normalised weights (sum to 1)

        # ---- MSE weight (linear term) ----
        w_mse = 2.0 * (conv - avg_conv)          # derivative of ∑(conv-avg)²
        w = w_max + lambda_mse * w_mse           # combined weight vector

        # ---- gradient w.r.t. a (linear combination of conv entries) ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]   # central n entries

        # entropy regulariser (∂/∂a of λ·∑ a·log a)
        if lambda_entropy != 0.0:
            grad_a += lambda_entropy * (np.log(a) + 1.0)

        # ---- gradient w.r.t. logits v (softmax Jacobian) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam update ----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

        # decay the MSE weight
        lambda_mse *= lambda_mse_decay

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  Returns the best sequence (plain Python list)
    found within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin – leave a few seconds for finalisation
    deadline = start_time + budget_s - 5.0

    # try larger lengths first, fall back if time runs out
    lengths = [20000, 12000, 8000]

    global_best_seq: np.ndarray | None = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"[main] Optimising length = {n}")

        # --------------------------------------------------------------
        # 1) Build a diverse set of seed sequences (all on the simplex)
        # --------------------------------------------------------------
        seed_seqs: list[np.ndarray] = []

        # analytically motivated families
        seed_seqs.append(_arcsine_dist(n))
        seed_seqs.append(_sine_window(n))

        # geometric families
        for c in np.linspace(0.5, 5.0, 9):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_geom(n, r))

        # power‑law families
        for p in np.linspace(0.0, 3.0, 9):
            seed_seqs.append(_power(n, p))

        # combined geometric × power families
        for c in np.linspace(0.5, 4.0, 5):
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in np.linspace(0.0, 2.0, 5):
                seed_seqs.append(_geom_power(n, r, p))

        # symmetric geometric
        for c in np.linspace(0.5, 4.5, 5):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_sym_geom(n, r))

        # beta‑family (including the arcsine case α=β=0.5)
        alphas = [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]
        for α in alphas:
            for β in alphas:
                seed_seqs.append(_beta_dist(n, α, β))

        # a few Dirichlet draws
        for α in [0.1, 0.3, 0.6, 1.5, 3.0]:
            a = np.random.gamma(α, 1.0, size=n)
            s = a.sum()
            if s > 0:
                seed_seqs.append(a / s)

        # random soft‑max seeds
        for _ in range(5):
            v_rand = np.random.randn(n)
            seed_seqs.append(_softmax(v_rand))

        # reuse the best sequence from a previous length (truncated / padded)
        if global_best_seq is not None:
            prev = np.asarray(global_best_seq, dtype=np.float64)
            if prev.ndim == 1:
                adj = np.zeros(n, dtype=np.float64)
                m = min(prev.shape[0], n)
                adj[:m] = prev[:m]
                prev = adj
                if prev.sum() > 0:
                    prev = prev / prev.sum()
                    seed_seqs.append(prev)

        # --------------------------------------------------------------
        # 2) Fast scan – pick the seed with the smallest exact objective
        # --------------------------------------------------------------
        for idx, a in enumerate(seed_seqs):
            cur_val = 2.0 * n * np.max(_conv_fft(a, a))
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[scan] new global best from seed {idx} = {cur_val:.6f}")

        # --------------------------------------------------------------
        # 3) Refine the current best seed with the advanced optimiser
        # --------------------------------------------------------------
        if global_best_seq is None:
            # fallback to uniform
            global_best_seq = np.ones(n, dtype=np.float64) / n
            print("[fallback] using uniform start")

        # make sure the seed sums to one (softmax guarantees this)
        seq_start = global_best_seq.copy()
        seq_start = seq_start / seq_start.sum()

        refined_seq, refined_val = _refine_sequence_advanced(
            seq_start,
            n,
            deadline,
            start_beta=0.5,
            beta_mul=1.02,
            lr0=0.8,
            start_lambda_mse=0.5,
            lambda_mse_decay=0.999,
            lambda_entropy=0.05,
        )
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[overall] new global best after refinement (len={n}) = {refined_val:.6f}")

        # --------------------------------------------------------------
        # 4) Perturbed restarts – keep trying while time permits
        # --------------------------------------------------------------
        while time.time() < deadline:
            noisy_logits = np.log(global_best_seq + 1e-30) + 0.02 * np.random.randn(n)
            perturbed = _softmax(noisy_logits)
            seq2, val2 = _refine_sequence_advanced(
                perturbed,
                n,
                deadline,
                start_beta=0.5,
                beta_mul=1.02,
                lr0=0.8,
                start_lambda_mse=0.5,
                lambda_mse_decay=0.999,
                lambda_entropy=0.05,
            )
            if val2 < global_best_val:
                global_best_val = val2
                global_best_seq = seq2.copy()
                print(f"[perturb] improved global best to {val2:.6f}")

        # --------------------------------------------------------------
        # End of this length – continue with the next (if any)
        # --------------------------------------------------------------

    # ------------------------------------------------------------------
    # Final sanity check and return
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 8000
        global_best_seq = np.ones(n, dtype=np.float64) / n
        print("[fallback] final uniform sequence")

    # normalise to sum‑to‑one (scale‑invariant)
    s = np.sum(global_best_seq)
    if s > 0:
        global_best_seq = global_best_seq / s

    result = global_best_seq.tolist()
    print(f"[final] best objective (fast) = {global_best_val:.6f}")
    return result
