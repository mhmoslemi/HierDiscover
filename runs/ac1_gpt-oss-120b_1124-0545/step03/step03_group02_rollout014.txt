def project_simplex(v):
    # v is a 1D array, want projection onto simplex sum=1 and v>=0.
    # Classical algorithm: 
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv/ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

def subgradient_optimize(a_init, time_remain):
    a = a_init.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.time()
    it = 0
    while time.time() - t0 < time_remain:
        conv = np.convolve(a, a)
        max_idx = conv.argmax()
        # gradient of max conv:
        grad = np.zeros_like(a)
        t_start = max(0, max_idx - (n-1))
        t_end = min(n-1, max_idx)
        # indices t_start..t_end inclusive
        # compute corresponding a values reversed
        rev_vals = a[max_idx - np.arange(t_start, t_end+1)]
        grad[t_start:t_end+1] = 2 * rev_vals
        # gradient of objective = 2*n * grad (since objective = 2*n*max conv)
        g = 2 * n * grad
        # step size schedule:
        it += 1

        a = a - step * g
        # projection onto simplex
        a = project_simplex(a)
        # compute score
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[sgd {it}] new best subgradient score {best_score:.6f}")

"""
Algorithm:
- Initialise candidate from previous best or analytic arcsine.
- Randomly sample parameterised families of symmetric step functions (geometric,
  arithmetic, exponential, sine, blend with uniform) and keep the best according to
  the exact objective (computed via FFT).
- Refine the best candidate with a simple sub‑gradient descent on the exact
  max‑convolution objective while projecting onto the probability simplex.
- Return the best sequence found within the time budget.
"""

import time
import numpy as np

def project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    # Implementation based on the algorithm by Wang & Carreira‑Perpiñán (2013).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero/negative – return uniform simplex point
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1] + 1
        theta = cssv[rho - 1] / rho
    w = np.maximum(v - theta, 0)
    return w

def compute_score(a, nfft):
    """Exact evaluation of the objective for a vector a (sum not necessarily 1)."""
    n = a.shape[0]
    # real FFT based convolution
    fa = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(fa * fa, nfft)[: 2 * n - 1]
    max_conv = conv.max()
    s = a.sum()
    return float(2.0 * n * max_conv / (s * s))

def shape_geom(p, n):
    """Geometric (arcsine‑like) family: a_i ∝ ((i+½)(n‑i‑½))^{-p}."""
    i = np.arange(n, dtype=float)
    a = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    a = np.maximum(a, 0.0)
    return a / a.sum()

def shape_arith(p, n):
    """Arithmetic family: a_i ∝ (i+½)^{-p} + (n‑i‑½)^{-p}."""
    i = np.arange(n, dtype=float)
    a = (i + 0.5) ** (-p) + (n - i - 0.5) ** (-p)
    a = np.maximum(a, 0.0)
    return a / a.sum()

def shape_exponential(p, n):
    """Exponential decay from the edges."""
    i = np.arange(n, dtype=float)
    a = np.exp(-p * i) + np.exp(-p * (n - 1 - i))
    a = np.maximum(a, 0.0)
    return a / a.sum()

def shape_sine(p, n):
    """Sine‑based family: a_i ∝ sin(π(i+½)/n)^{p}."""
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * (i + 0.5) / n) ** p
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        # avoid division by zero for pathological p
        a = np.ones_like(a)
    return a / a.sum()

def shape_blend(alpha, n):
    """Blend between uniform and the classic arcsine shape (p=0.5)."""
    uniform = np.ones(n, dtype=float)

Algorithm:
- Initialise from a previously stored solution (if any) or from the classic arcsine shape.
- Randomly sample a few low‑dimensional families of symmetric step functions
  (geometric, arithmetic, exponential, sine‑based and a blend with uniform) and
  keep the best according to the exact objective (computed via FFT).
- Refine the best candidate with a short sub‑gradient descent on the exact
  max‑convolution objective while projecting onto the probability simplex.
- Return the best sequence found before the time budget expires.
"""

def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σv_i = 1, v_i≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – return uniform point
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1] + 1
        theta = cssv[rho - 1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray, nfft: int) -> float:
    """Exact evaluation of the objective for vector a (sum need not be 1)."""
    n = a.shape[0]
    fa = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(fa * fa, nfft)[: 2 * n - 1]
    max_conv = conv.max()
    s = a.sum()
    return float(2.0 * n * max_conv / (s * s))


def shape_geom(p: float, n: int) -> np.ndarray:

    a = np.maximum(a, 0.0)
    return a / a.sum()
