4. If no improvement has been observed for a while a jitter‑restart is
   applied to perturb the current best solution.

The routine respects a time budget (default 1000 s) and returns the best
sequence discovered.
"""

import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to ``x``."""
    return 1 << (x - 1).bit_length()

def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution (zero‑padded) of ``a`` with itself using the FFT.
    Returns an array of length ``2*len(a)-1``.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]

def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one index to another.
    ``scale`` controls the fraction of the source mass that can be moved.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

def hardmax_gradient(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of the objective
        f(a) = max_k (a ∗ a)_k
    evaluated at ``a``.  ``conv`` must be the self‑convolution of ``a``.
    The gradient is
        g_i = 2 * Σ_{k∈I} a_{k‑i}
    where ``I`` are the indices where the convolution attains its maximum.
    """
    n = a.shape[0]
    M = conv.max()
    # indices that are (almost) maximal
    I = np.where(conv >= M - eps)[0]
    g = np.zeros(n, dtype=np.float64)
    for k in I:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.intp)
        # a_{k‑idx} vectorised
        g[idx] += 2.0 * a[k - idx]
    return g

def objective_from_conv(conv: np.ndarray) -> float:
    """Given a convolution array of a vector that already sums to one,
    return the objective 2·n·max(conv)."""
    n = (conv.shape[0] + 1) // 2   # because conv length = 2n‑1
    return float(2.0 * n * conv.max())

def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum=1) analytic candidate.
    ``mode`` = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    ``mode`` = 'sum'      → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)
    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the ``budget_s`` time limit and returns
    the best sequence found.  The default length is 2000, but if a global
    variable ``height_sequence_1`` is present its length is used instead.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose the problem size and an initial point
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if init.ndim != 1:
            init = init.ravel()
        n = init.shape[0]
        a = np.maximum(init, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.ones(n) / n
        else:
            a = a / s
        print(f"[init] using provided height_sequence_1 (n={n})")
    else:
        # fall back to a modest analytic grid (product family)
        n = int(kwargs.get("n", 2000))
        # basic grid – the best from the previous run was around p≈0.6
        a = analytic_candidate(n, p=0.6, offset=0.5, mode="product")
        print("[init] no global sequence – using analytic product candidate")

    # current best
    conv_a = conv_fft(a)
    best_score = objective_from_conv(conv_a)
    best_seq = a.copy()
    print(f"[init] initial objective = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    step_start = 0.30                # initial additive step size
    step_end   = 0.001               # final step size
    temp_start = 0.8                 # initial Metropolis temperature
    temp_end   = 0.001               # final temperature
    random_move_prob = 0.12          # probability to perform a random mass transfer
    restart_cooldown = 25.0          # seconds without improvement before jitter
    eps_grad = 1e-12                 # tolerance for detecting the active set

    iteration = 0
    last_improve = start_time

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 20.0)   # leave a margin for finalisation
        progress = min(1.0, max(0.0, progress))

        # linearly annealed parameters
        step = step_start + progress * (step_end - step_start)
        temperature = temp_start + progress * (temp_end - temp_start)

        # --------------------------------------------------------------
        # Choose a move
        # --------------------------------------------------------------
        if np.random.rand() < random_move_prob:
            cand = random_move(a, scale=step * 0.5)
        else:
            # gradient based move (exact sub‑gradient of the max)
            grad = hardmax_gradient(a, conv_a, eps=eps_grad)
            cand = a - step * grad
            cand = np.clip(cand, 0.0, None)
            cand = project_simplex(cand)

        # --------------------------------------------------------------
        # Evaluate candidate
        # --------------------------------------------------------------
        conv_cand = conv_fft(cand)
        cand_score = objective_from_conv(conv_cand)

        # Metropolis acceptance
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            if temperature > 0.0:
                prob = math.exp(-(cand_score - best_score) / temperature)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = cand
            conv_a = conv_cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:07d}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # --------------------------------------------------------------
        # Jitter restart if we have been stuck
        # --------------------------------------------------------------
        if time.time() - last_improve > restart_cooldown:
            jitter = (np.random.rand(n) - 0.5) * 2e-3   # uniform in [-0.001, 0.001]
            a = best_seq + jitter
            a = project_simplex(a)
            conv_a = conv_fft(a)
            last_improve = time.time()
            print(f"[{iteration:07d}] jitter restart (stagnation)")

        # occasionally report progress (every ~2000 iterations)
        if iteration % 2000 == 0:
            print(f"[{iteration:07d}] current best = {best_score:.6f}, elapsed = {elapsed:.1f}s")

        # early exit if we have reached a target that is clearly below the
        # published bound – the exact target may change, but 1.503 is the
        # reported one.
        if best_score <= 1.503:
            print("[done] target reached")
            break

        iteration += 1

    total_time = time.time() - start_time
    print(f"[done] iterations={iteration}, best_score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
