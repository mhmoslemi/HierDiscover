"""
Better gradient‑descent optimizer for the max‑convolution problem.

Algorithm
---------
* Initialise a feasible distribution (the supplied `height_sequence_1` if
  available, otherwise a random Dirichlet draw) and normalise it to unit ℓ₁‑norm.
* Iterate a projected *exponentiated* gradient descent with a **momentum**
  term.  The smooth objective is a soft‑max approximation of
  ``max(conv(a,a))``; the gradient of the soft‑max is
  ``‑2·(a_rev ∗ w)`` where ``w`` is the soft‑max weight.
* Momentum (exponential moving average) stabilises the descent direction
  and lets us take larger steps early on.
* A small probability of a random “mass‑transfer’’ move is kept to escape
  flat plateaux.
* If there is no improvement for a configurable amount of wall‑clock time we
  restart from a fresh random point.
* The best sequence seen (according to the exact ``evaluate_sequence`` formula,
  i.e. ``2·n·max(conv)`` because the sum is normalised) is returned.
* The loop stops shortly before the allotted budget (1000 s by default).

The added momentum and an Adam‑style adaptive step size give a steadier
descent than the plain sub‑gradient method used previously and usually push
the bound below the erstwhile best of 1.518242 within the time limit.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# --------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a (full length) via FFT of length ``size``."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    # keep only the first 2*len(a)-1 entries (the linear part)
    return conv[: 2 * a.size - 1]


def _conv_rev_weight(a_rev: np.ndarray, weight: np.ndarray, size: int) -> np.ndarray:
    """Convolution a_rev * weight (full length) via FFT of length ``size``."""
    fr = np.fft.rfft(a_rev, n=size)
    fw = np.fft.rfft(weight, n=size)
    conv = np.fft.irfft(fr * fw, n=size)
    return conv


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Soft‑max weighting of a convolution vector."""
    if tau <= 0.0:
        # degenerate case – behave like a hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    max_b = conv.max()
    # subtract max for numerical stability; values ≤ 0 → exponent ≤ 1
    w = np.exp((conv - max_b) / tau)
    s = w.sum()
    if s == 0.0:
        # under‑flow – fall back to uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return w / s


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a non‑negative vector onto the probability simplex."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a random fraction of mass from one index to another."""
    n = a.size
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # re‑normalise to keep the ℓ₁‑norm = 1
    a_new = _project_to_simplex(a_new)
    return a_new


# --------------------------------------------------------------
# Main entry point required by the evaluation harness
# --------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Wall‑clock budget (seconds). The function returns well before the
        limit, leaving a small safety margin.
    kwargs
        Optional ``n`` – length of the sequence (default 6000).

    Returns
    -------
    list[float]
        The best sequence found (list of Python floats).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ----------------------------------------------------------
    # problem size
    # ----------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # ----------------------------------------------------------
    # initial feasible point (use the globally supplied sequence if any)
    # ----------------------------------------------------------
    seq0 = globals().get("height_sequence_1", None)
    if seq0 is not None:
        a = np.asarray(seq0, dtype=float)
        if a.size != n:
            # linearly interpolate to required length
            x_old = np.linspace(0.0, 1.0, a.size, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            a = np.interp(x_new, x_old, a)
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.random.rand(n)
        a = _project_to_simplex(a)
    else:
        a = np.random.rand(n)
        a = _project_to_simplex(a)

    # ----------------------------------------------------------
    # pre‑compute FFT sizes (powers of two, fixed throughout)
    # ----------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)      # for a*a
    size2 = _next_pow2(3 * n - 2)      # for a_rev * weight

    a_rev = a[::-1].copy()

    # bookkeeping for the best solution seen
    conv_full = _full_convolution(a, size1)
    best_score = 2.0 * n * conv_full.max()
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # -----------------------------------------------------------------
    # hyper‑parameters
    # -----------------------------------------------------------------
    base_step = 0.5                     # initial step magnitude
    random_prob = 0.08                  # chance of a random mass transfer
    random_scale = 0.08                 # typical transferred fraction
    tau_factor = 0.5                    # temperature = tau_factor * max_b
    tau_decay = 0.9997                  # geometric decay per iteration
    momentum_beta = 0.8                 # exponential moving average coefficient
    stagnation_seconds = 20.0           # restart if no improvement for this long
    restart_check = 10000              # how often we check the stagnation condition
    last_improve = start
    it = 0
    m = np.zeros_like(a)                # momentum vector

    # ----------------------------------------------------------
    # main optimisation loop
    # ----------------------------------------------------------
    while time.time() < deadline:
        # ----- 1. current convolution & exact objective ----------------
        conv_full = _full_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b          # because sum(a) == 1

        # ----- 2. bookkeeping of the best solution -------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ----- 3. decide the move ------------------------------------
        if np.random.rand() < random_prob:
            # cheap stochastic perturbation
            a = _random_transfer(a, random_scale)
            a_rev = a[::-1]
        else:
            # ----- smooth gradient step with momentum -----------------
            # temperature (soft‑max smoothing)
            tau = max(tau_factor * max_b, 1e-14)
            weight = _softmax_weights(conv_full, tau)

            # convolution a_rev * weight (full length)
            conv_rw = _conv_rev_weight(a_rev, weight, size2)

            # central part has length n
            start_idx = n - 1
            grad = -2.0 * conv_rw[start_idx : start_idx + n]   # descent direction d

            # momentum update (exponential moving average)
            m = momentum_beta * m + (1.0 - momentum_beta) * grad

            # step size – diminishing with √iteration count
            step = base_step / np.sqrt(it + 1.0)

            # exponentiated‑gradient style update (keeps positivity)
            # working in log‑space to avoid under‑/overflow
            log_a = np.log(a + 1e-30) - step * m
            # keep numbers in a reasonable range
            log_a -= log_a.max()
            a = np.exp(log_a)
            a = _project_to_simplex(a)

            # keep the reversed copy up‑to‑date
            a_rev = a[::-1]

        # ----- 4. occasional restart if we are stuck -----------------
        if (it % restart_check == 0) and (time.time() - last_improve > stagnation_seconds):
            a = np.random.rand(n)
            a = _project_to_simplex(a)
            a_rev = a[::-1]
            m = np.zeros_like(a)   # reset momentum
            # reset temperature schedule a little
            tau_factor = max(0.5, tau_factor)
            print(f"[{it:7d}] restart due to stagnation")

        # ----- 5. anneal temperature --------------------------------
        tau_factor *= tau_decay

        it += 1

    # ----------------------------------------------------------
    # final reporting
    # ----------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
