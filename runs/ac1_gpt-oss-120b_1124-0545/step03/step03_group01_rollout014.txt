Define hill climb:

```
def _hill_climb(a, n, deadline, max_iters):
    rng = np.random.default_rng()
    cur = a.copy()
    cur_val = 2*n*np.max(_conv_fft(cur,cur))
    best = cur.copy()
    best_val = cur_val
    eps = 0.01  # step magnitude relative to sum?
    for it in range(max_iters):
        if time.time()>deadline: break
        # propose new by moving small mass between random indices
        i, j = rng.integers(0, n, size=2)
        if i==j: continue
        delta = (rng.random() - 0.5) * eps  # delta in [-eps/2, eps/2]
        # compute max allowed delta to keep nonnegativity
        delta = max(-cur[i] + 1e-12, min(cur[j] - 1e-12, delta))
        if abs(delta) < 1e-12: continue
        new = cur.copy()
        new[i] += delta
        new[j] -= delta
        # renormalization (should already sum constant)
        # evaluate
        val = 2*n*np.max(_conv_fft(new,new))
        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val=val
                best = new.copy()
        # gradually reduce eps
        if it % 1000 == 0:
            eps *= 0.9
    return best, best_val
```

"""
Search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

using a three‑phase strategy:

1. **Randomised seeds** – a diverse family of analytic shapes
   (uniform, power‑law, geometric, symmetric, beta‑like, sine‑window,
   Dirichlet samples) together with the previously found best
   construction (if any).

2. **Adam on a smoothed max** – the sequence is parametrised by a
   log‑softmax vector `v` so that `a = softmax(v)` lives on the
   probability simplex.  The loss is a log‑sum‑exp smooth approximation
   of the max of the auto‑convolution.  The smoothing coefficient is
   increased during optimisation, and Adam updates keep the iterates
   on the simplex automatically.

3. **Exact sub‑gradient refinement** – from the best Adam result a
   mirror‑descent (exponentiated‑gradient) step using the true sub‑gradient
   of the max is performed for many iterations.

4. **Simple stochastic hill‑climbing** – after the deterministic
   refinement a cheap random pair‑wise mass‑transfer neighbourhood search
   is run to squeeze out any remaining improvement.

The whole routine is time‑aware (default budget 1000 s) and returns the
best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Analytic seed families (already normalised to sum 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a / a.sum()


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max of the auto‑convolution.
    Returns the best sequence (as an np.ndarray) and its objective
    2·n·max(conv) – sum(a) is 1 throughout.
    """
    v = np.log(seed + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0            # smoothing coefficient
    beta_mul = 1.03       # grow a bit faster than the reference code

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)                         # stays on the simplex
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights -----
        beta *= beta_mul
        shift = np.max(conv)                     # safe shift
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient w.r.t. a (∇_a Σ w_k·conv_k) -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]     # central n entries

        # ----- convert to gradient w.r.t. v (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update on v -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 40000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent refinement with the exact sub‑gradient of max(conv).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] n={n:5d} it={it:5d} best={cur_val:.6f}")

        # sub‑gradient of max_k conv_k w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a tiny amount of mass between
    two random positions and keep changes that improve the objective.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur, cur))
    best_seq = cur.copy()
    best_val = cur_val

    # step size decays during the walk
    step_scale = 0.01  # initial relative step

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose moving a small amount from i towards j
        donor = i if rng.random() < 0.5 else j
        receiver = j if donor is i else i

        max_transfer = min(cur[donor], step_scale * cur[donor] + 1e-12)
        if max_transfer <= 0:
            continue
        delta = rng.random() * max_transfer  # always positive
        # apply transfer
        new = cur.copy()
        new[donor] -= delta
        new[receiver] += delta
        # check non‑negativity (should hold) and evaluate
        val = 2.0 * n * np.max(_conv_fft(new, new))

        if val < cur_val:
            cur = new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = new.copy()
                print(f"[Hill] n={n:5d} it={it:5d} best={val:.6f}")

        # slowly reduce the step scale
        if it % 1000 == 0:
            step_scale *= 0.9
            step_scale = max(step_scale, 1e-6)

    return best_seq, best_val

def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a modest set of diverse initial vectors (already ℓ₁‑normalised)."""
    seeds: list[np.ndarray] = []

    # elementary baselines
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # symmetric geometric
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # beta‑like shapes (α,β in a few points around 0.5)
    for a in (0.3, 0.5, 0.7):
        for b in (0.3, 0.5, 0.7):
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws
    rng = np.random.default_rng(2023)
    for conc in (0.3, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # a few random soft‑max vectors
    for _ in range(7):
        logits = rng.standard_normal(n)
        seeds.append(_softmax(logits))

    # reuse previously discovered sequence if it matches the length
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds
