def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    n = int(kwargs.get("n", 2000))

    # initialize a
    if "height_sequence_1" in globals():
        a = np.maximum(np.asarray(globals()["height_sequence_1"], dtype=np.float64), 0.0)
        if a.shape[0] != n:
            # If mismatch length, adjust by truncation or pad uniformly.
            if a.shape[0] > n:
                a = a[:n]
            else:
                # pad with small uniform remainder
                extra = np.full(n - a.shape[0], 0.0)
                a = np.concatenate([a, extra])
        # project onto simplex to ensure sum=1
        a = a / a.sum()
    else:
        a = np.ones(n) / n

    # compute convolution
    conv = conv_fft(a)
    max_conv = conv.max()
    cur_score = 2.0 * n * max_conv
    best_a = a.copy()
    best_score = cur_score
    best_iter = 0

    # schedule parameters
    init_step = 0.5  # maximum fraction of selected mass to move
    final_step = 1e-5
    init_temp = 0.5
    final_temp = 1e-4

    # variables for SA
    iteration = 0
    last_improve = start
    stagnation_time = 30.0

    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / budget_s   # 0..1

        step = init_step + progress * (final_step - init_step)
        temp = init_temp + progress * (final_temp - init_temp)

        # random move
        if np.random.rand() < 0.9:
            # mass transfer move
            # pick i where a[i] > 0
            i = np.random.choice(n, p=a)  # sample proportionally to mass? Could increase changes where mass large.
            # For speed, simple random
            # Actually we want i with enough mass; we can just random ; then if a[i]==0 choose again.
            # Simple loop:
            while a[i] == 0.0:
                i = np.random.randint(0, n)
            j = np.random.randint(0, n-1)
            if j >= i:
                j += 1
            max_delta = step * a[i]   # max fraction to move
            if max_delta <= 0:
                continue
            delta = max_delta * np.random.rand()
            # compute new a and conv via incremental
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            # update conv incrementally
            conv_new = conv.copy()
            conv_new[j:j+n] += 2 * delta * a
            conv_new[i:i+n] -= 2 * delta * a
            delta2 = delta * delta
            conv_new[2*j] += delta2
            conv_new[2*i] += delta2
            conv_new[i+j] -= 2 * delta2
            new_max = conv_new.max()
            new_score = 2.0 * n * new_max
        else:
            # subgradient descent step
            # compute subgradient using mask on max indices
            max_val = conv.max()
            mask = (conv == max_val).astype(np.float64)
            mask_sum = mask.sum()

            else:
                w = mask / mask_sum

"""
Propose a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

using a fast simulated‑annealing / sub‑gradient descent hybrid.
The algorithm:

* Initialise from the supplied global construction (height_sequence_1) if
  available, otherwise fall back to a short analytic start.
* Maintain the full convolution a*a (linear, zero‑padded) and the current
  objective value.
* At each iteration either
    – perform a cheap mass‑transfer move (move a small δ from one index to
      another) and update the convolution in O(n) via the exact formula,
    – or take a projected sub‑gradient step for the hard max (the true
      max of the convolution) using FFT to compute the weighted shift.
* Accept moves that improve the score; otherwise accept with a Metropolis
  probability that slowly cools down.
* When no improvement is seen for a while, jitter‑re‑initialise around the
  best solution found so far.
* Stop after the time budget (≈1000 s) or as soon as the target bound
  1.503 is reached.

All helper functions are top‑level, there are no closures or file I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (zero‑padded) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sub‑gradient for the true max (no smoothing)
# ----------------------------------------------------------------------


def subgradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of the objective
        Φ(a) = max_k (a*a)_k
    with respect to a (the sum is assumed to be 1).
    The sub‑gradient is
        g = 2 * Σ_{k∈K} (a shifted by k) / |K|
    where K = {k | conv_k == max(conv)}.
    """
    n = a.shape[0]
    max_val = conv.max()
    mask = (conv == max_val).astype(np.float64)
    w = mask / mask.sum()                     # equal weight on all tight indices

    # weighted shift via FFT (same pattern as gradient_smooth)
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)

    start = n - 1
    slice_part = conv_wa[start:start + n]    # length n
    g = 2.0 * slice_part[::-1]               # reverse to align with a's indices
    return g


# ----------------------------------------------------------------------
# Incremental convolution update for a single mass‑transfer move
# ----------------------------------------------------------------------


def apply_mass_transfer(a: np.ndarray, conv: np.ndarray,
                        i: int, j: int, delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer a positive amount `delta` from index i to index j.
    Returns (new_a, new_conv).  The update of the convolution is performed
    analytically in O(n) without FFTs.
    """
    n = a.shape[0]
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    conv_new = conv.copy()
    # linear contribution 2*delta*a shifted
    conv_new[j:j + n] += 2.0 * delta * a
    conv_new[i:i + n] -= 2.0 * delta * a

    # quadratic δ² term
    d2 = delta * delta
    conv_new[2 * j] += d2
    conv_new[2 * i] += d2
    conv_new[i + j] -= 2.0 * d2

    return a_new, conv_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the evaluation function used by the competition harness.
    The routine respects a ~1000 s time budget and returns the best sequence
    discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Parameters (can be tweaked)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # annealing schedules (linear)
    step_start, step_end = 0.5, 1e-5                # max fraction of source mass to move
    temp_start, temp_end = 0.5, 1e-4                # Metropolis temperature

    # how often we try a true sub‑gradient step (the rest are cheap moves)
    prob_subgrad = 0.08

    # stagnation handling
    stagnation_cooldown = 30.0                      # seconds without improvement → jitter

    # ------------------------------------------------------------------
    # Initial sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        if a0.shape[0] != n:
            # adjust length: truncate or pad with zeros then renormalise
            if a0.shape[0] > n:
                a0 = a0[:n]
            else:
                pad = np.zeros(n - a0.shape[0], dtype=np.float64)
                a0 = np.concatenate([a0, pad])
        a0 = a0 / a0.sum()
        a = a0.copy()
    else:
        # fallback: uniform distribution
        a = np.ones(n, dtype=np.float64) / n

    conv = conv_fft(a)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max

    best_a = a.copy()
    best_score = cur_score
    last_improve = start_time

    iteration = 0
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / budget_s

        # linear annealing of step size and temperature
        step = step_start + progress * (step_end - step_start)
        temp = temp_start + progress * (temp_end - temp_start)

        # --------------------------------------------------------------
        # Choose move type
        # --------------------------------------------------------------
        if np.random.rand() < prob_subgrad:
            # ----- sub‑gradient step for the true max -----
            g = subgradient_max(a, conv)
            a_candidate = a - step * g
            a_candidate = np.clip(a_candidate, 0.0, None)
            a_candidate = project_simplex(a_candidate)

            conv_candidate = conv_fft(a_candidate)
            new_max = conv_candidate.max()
            new_score = 2.0 * n * new_max
        else:
            # ----- cheap random mass‑transfer move -----
            # choose source proportional to current mass – this tends to move
            # mass from large entries.
            i = np.random.choice(n, p=a)
            # ensure the source actually holds some mass
            if a[i] <= 1e-15:
                # fallback to a uniformly random source with non‑zero mass
                nz = np.where(a > 1e-15)[0]
                if nz.size == 0:
                    continue
                i = np.random.choice(nz)

            # choose a different destination uniformly
            j = np.random.randint(0, n - 1)
            if j >= i:
                j += 1

            # amount to move – at most `step` fraction of the source mass
            delta = step * a[i] * np.random.rand()
            if delta <= 0.0:
                continue

            a_candidate, conv_candidate = apply_mass_transfer(a, conv, i, j, delta)
            new_max = conv_candidate.max()
            new_score = 2.0 * n * new_max

        # --------------------------------------------------------------
        # Acceptance test (Metropolis)
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if temp > 1e-12:
                prob = math.exp(-(new_score - cur_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_candidate
            conv = conv_candidate
            cur_score = new_score

            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                last_improve = time.time()
                print(f"[{iteration:07d}] new best = {best_score:.6f} "
                      f"(t={elapsed:.1f}s, step={step:.3e}, temp={temp:.3e})")

        # --------------------------------------------------------------
        # Stagnation handling – jitter around the best so far
        # --------------------------------------------------------------
        if time.time() - last_improve > stagnation_cooldown:
            jitter = (np.random.rand(n) - 0.5) * 1e-4
            a = best_a + jitter
            a = project_simplex(a)
            conv = conv_fft(a)
            cur_score = 2.0 * n * conv.max()
            last_improve = time.time()
            print(f"[{iteration:07d}] jitter restart after {stagnation_cooldown:.0f}s")
            # gradually increase the waiting period
            stagnation_cooldown = min(180.0, stagnation_cooldown * 1.5)

        # early exit if the target bound is already reached
        if best_score <= 1.503:
            print("[done] target reached – exiting early")
            break

        iteration += 1

    total_time = time.time() - start_time
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, "
          f"time={total_time:.1f}s")
    # Return a plain Python list as required by the harness
    return best_a.tolist()
