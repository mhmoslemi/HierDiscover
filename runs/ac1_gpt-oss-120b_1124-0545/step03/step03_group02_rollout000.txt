"""
propose_candidate – two‑stage optimiser for the self‑convolution bound.

Stage 1 – Smooth‑max Adam descent.
    * Parameters are kept on the probability simplex by a soft‑max
      re‑parameterisation a = softmax(v).
    * The true max in the objective 2·n·max(conv(a,a)) is approximated by a
      temperature‑controlled soft‑max (log‑sum‑exp).  The temperature β is
      linearly increased from ~5 to 2000, so the optimisation gradually
      focuses on the actual maximal entry.
    * An Adam update on the unconstrained variables v provides fast,
      adaptive steps.  Restarts are triggered if no improvement is seen
      for a few seconds.

Stage 2 – Local stochastic refinement (simulated‑annealing style).
    * Starting from the best sequence of stage 1, random mass transfers
      between two coordinates are attempted.
    * Each proposal is accepted if it lowers the exact objective or with
      a probability exp(‑Δ/ T) where the temperature T is cooled linearly.
    * This cheap Monte‑Carlo search frequently discovers tiny improvements
      that gradient descent alone may miss.

Both stages respect the 1000 s budget, keep the sequence non‑negative,
and finally return a normalised probability vector (sum = 1).  The
evaluation function is scale‑invariant, so the normalisation does not
affect the final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    v_shift = v - np.max(v)
    exp_v = np.exp(v_shift)
    return exp_v / exp_v.sum()


def initialise_a(n: int, seed: int = 42) -> np.ndarray:
    """
    Initialise a good starting point.
    If a global variable `height_sequence_1` exists and has the correct length,
    reuse it (normalised).  Otherwise fall back to the classic arcsine shape.
    """
    rng = np.random.default_rng(seed)

    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n and a0.sum() > 0.0:
            a0 = a0 / a0.sum()
            return a0

    # classic arcsine shape (inverse‑sqrt)
    i = np.arange(n, dtype=float)
    a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a0 = a0 / a0.sum()
    return a0


def max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return max(conv(a,a)) using FFT (linear convolution)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return conv.max()


def exact_score_from_max(n: int, conv_max: float) -> float:
    """Exact objective value for a probability vector (sum = 1)."""
    return float(2.0 * n * conv_max)


def grad_softmax_obj(v: np.ndarray,
                     a: np.ndarray,
                     conv: np.ndarray,
                     w: np.ndarray,
                     fft_len: int,
                     n: int) -> np.ndarray:
    """
    Gradient of the smoothed objective w.r.t. the log‑parameters v.
    Implements the formula g = 2·n·(g1 + g2) and the soft‑max chain rule.
    """
    # FFT of weight vector
    fw = np.fft.fft(w, fft_len)

    # convolution w * a (linear, length 3n‑2)
    conv_wa_full = np.real(np.fft.ifft(fw * np.fft.fft(a, fft_len)))[: 3 * n - 2]
    g2 = conv_wa_full[:n]                    # first n entries

    # cross‑correlation w ⋆ a (valid part, also length 3n‑2)
    corr_full = np.real(np.fft.ifft(np.conj(fw) * np.fft.fft(a, fft_len)))[: 3 * n - 2]
    g1 = corr_full[:n]                       # first n entries

    # gradient w.r.t. a
    g = 2.0 * n * (g1 + g2)

    # chain rule for soft‑max parametrisation
    dot_ag = np.dot(a, g)
    grad_v = a * (g - dot_ag)
    return grad_v


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of floats) that minimises
    `evaluate_sequence`.  The routine uses a two‑stage optimiser described
    in the module docstring and respects the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    overall_deadline = start_time + budget_s - 0.5

    # Length of the sequence (default 2000)
    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 0) initial point
    # ------------------------------------------------------------------
    a = initialise_a(n, seed=seed)
    v = np.log(a + 1e-12)          # log‑parametrisation for soft‑max

    # ------------------------------------------------------------------
    # 1) Stage 1 – smooth‑max Adam descent
    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    # schedules (linear in the progress variable p∈[0,1])
    lr_start, lr_end = 0.08, 0.001          # learning‑rate
    temp_start, temp_end = 5.0, 2000.0     # soft‑max temperature

    # FFT size – next power of two ≥ 3·n‑2 (required for convolutions below)
    fft_len = 1 << ((3 * n - 2).bit_length())

    # bookkeeping
    best_a = a.copy()
    conv = np.real(np.fft.ifft(np.fft.fft(a, fft_len) ** 2))[: 2 * n - 1]
    best_max = conv.max()
    best_score = exact_score_from_max(n, best_max)
    last_improve = start_time
    it = 0

    # allocate ~85 % of the total budget for stage 1
    stage1_deadline = start_time + 0.85 * budget_s

    while time.time() < stage1_deadline:
        elapsed = time.time() - start_time
        # progress ∈ [0,1] for the whole stage
        p = min(1.0, elapsed / (stage1_deadline - start_time))

        lr = lr_start * (1.0 - p) + lr_end * p
        temp = temp_start * (1.0 - p) + temp_end * p

        # current point on the simplex
        a = softmax(v)

        # linear convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        # smooth‑max weights (log‑sum‑exp)
        z = temp * conv
        z_max = np.max(z)
        w = np.exp(z - z_max)
        w /= w.sum()                     # shape (2n‑1,)

        # gradient w.r.t. v
        grad_v = grad_softmax_obj(v, a, conv, w, fft_len, n)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # exact score (max is already available)
        cur_score = exact_score_from_max(n, conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[{it:08d}] stage‑1 new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # occasional restart when progress stalls
        if it % 4000 == 0:
            if time.time() - last_improve > 20.0:
                v = np.log(best_a + 1e-12) + rng.normal(scale=1e-3, size=n)
                m.fill(0.0)
                v_hat.fill(0.0)
                print(f"[{it:08d}] restart after stagnation (t={elapsed:.1f}s)")

    # ------------------------------------------------------------------
    # 2) Stage 2 – stochastic local refinement (simulated annealing)
    # ------------------------------------------------------------------
    # Use the remaining time budget for the local search.
    remaining_time = overall_deadline - time.time()
    if remaining_time > 5.0:
        a = best_a.copy()
        best_conv_max = best_score / (2.0 * n)

        # SA temperature schedule (linear cooling)
        T_start = best_score * 0.005      # 0.5 % of current score
        T_end = best_score * 1e-6

        sa_it = 0
        phase_start = time.time()
        while time.time() < overall_deadline:
            # progress of the SA phase
            phase_elapsed = time.time() - phase_start
            phase_frac = min(1.0, phase_elapsed / remaining_time)
            T = T_start * (1.0 - phase_frac) + T_end * phase_frac

            # propose a mass transfer between two random coordinates
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            # move a small fraction of a[i] (up to 5 % of its current value)
            max_move = min(a[i], 0.05 * a[i] + 1e-12)
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # evaluate the exact objective for the proposal
            new_max = max_convolution(a_new, fft_len)
            new_score = exact_score_from_max(n, new_max)

            # Metropolis acceptance
            if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (T + 1e-12)):
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    best_conv_max = new_max
                    print(f"[SA {sa_it:07d}] improved to {best_score:.6f} (t={time.time() - start_time:.1f}s)")

            sa_it += 1

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    # Normalise explicitly (tiny numerical drift may appear after SA)
    total = best_a.sum()
    if total == 0.0:
        # fallback to uniform distribution (should never happen)
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    elapsed_total = time.time() - start_time
    print(f"[done] total time={elapsed_total:.1f}s, best score={best_score:.6f}, length={n}")
    return best_a.tolist()
