"""
Propose a step‑function (non‑negative sequence) by
projected Adam optimisation of a smoothed max‑convolution objective.

The algorithm:
  1. Choose several diverse random initialisations (power‑law,
     exponential, Gaussian, uniform, mixtures) on the probability simplex.
  2. For each start run a projected Adam optimiser that minimises the
     soft‑max approximation of max(a∗a).  The true gradient of the
     smoothed objective is 2·(a∗w), where w are the soft‑max weights.
  3. After each Adam step the iterate is projected onto the simplex.
  4. Every few iterations we evaluate the exact objective
     (2·n·max(conv)/(∑a)²) and keep the best sequence seen.
  5. The loop respects the supplied time‑budget (≈1000 s).

The implementation uses only NumPy, no closures and provides the
required `propose_candidate` entry point.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0 , Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be negative → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    The returned length is len(x)+len(y)-1.
    """
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())   # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^(-alpha), normalised to sum‑1."""
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum‑1."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-lam * idx)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _gauss_sequence(n: int, mu: float, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-0.5*((i-mu)/sigma)^2), normalised to sum‑1."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _uniform_sequence(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=float)


def _mix_sequence(a: np.ndarray, b: np.ndarray, p: float) -> np.ndarray:
    """Convex mixture p·a + (1-p)·b, re‑normalised to sum‑1."""
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    mixed = p * a + (1.0 - p) * b
    s = mixed.sum()
    if s == 0.0:
        mixed[:] = 1.0 / mixed.size
    else:
        mixed /= s
    return mixed

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Runs for at most ``budget_s`` seconds
    (a few seconds are kept as safety margin) and returns the best
    sequence found.

    Parameters
    ----------
    seed : int
        Random seed.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Ignored – present for compatibility.

    Returns
    -------
    list[float]
        The best sequence (non‑negative, Σ>0.01) discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a few seconds for the harness to finish gracefully
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine problem size (use the length of a supplied solution if any)
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be defined by the surrounding harness
        init_seq = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = init_seq.size
    except Exception:
        # fallback length
        n = 5000

    # ------------------------------------------------------------------
    # Shared optimisation hyper‑parameters
    # ------------------------------------------------------------------
    # Adam
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 0.08          # base learning rate for Adam

    # Soft‑max smoothing (ε = sigma·max(conv))
    sigma = 0.4
    sigma_decay = 0.9998
    sigma_min = 0.02

    # How often we evaluate the exact objective
    eval_every = 50
    # How often we perform a random mass‑swap to escape plateaus
    swap_every = 200

    best_seq = None

    # Helper to evaluate the exact objective (fast, uses FFT)
    # ------------------------------------------------------------------
    def _exact_score(a_vec: np.ndarray) -> float:
        """Return 2·n·max(a∗a) (sum(a) is 1 after projection)."""
        conv = _conv_fft(a_vec, a_vec)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Initialise best with any known global solution
    # ------------------------------------------------------------------
    try:
        # normalise any supplied solution and compute its score

            s = _exact_score(hs)
            best_score = s
            best_seq = hs.copy()
            print(f"[seed] known solution score={best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Main loop over random restarts (time‑aware)
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        restart += 1
        # ---- 1. random initialisation ---------------------------------
        kind = np.random.choice(['power', 'exp', 'gauss', 'uniform', 'mix'])
        if kind == 'power':

            a = _power_sequence(n, alpha, offset)
        elif kind == 'exp':

            a = _exp_sequence(n, lam)
        elif kind == 'gauss':
            mu = np.random.uniform(0, n - 1)
            sigma0 = np.random.uniform(n / 10.0, n / 2.0)

        else:  # mix

            offset = np.random.uniform(0.0, 5.0)
            pow_seq = _power_sequence(n, alpha, offset)
            uni_seq = _uniform_sequence(n)
            mix_p = np.random.uniform(0.2, 0.8)
            a = _mix_sequence(pow_seq, uni_seq, mix_p)

        a = _project_onto_simplex(a)

        m = np.zeros_like(a)
        v2 = np.zeros_like(a)

        sigma_loc = sigma

        # ---- 3. optimisation loop --------------------------------------
        while time.time() < deadline:
            t += 1

            # ----- a) self‑convolution and max -------------------------
            conv = _conv_fft(a, a)                 # length 2n‑1
            max_conv = conv.max()

            # ----- b) smoothing parameter ε ----------------------------
            sigma_loc = max(sigma_loc * sigma_decay, sigma_min)
            eps = max(sigma_loc * max_conv, 1e-12)

            # ----- c) soft‑max weights --------------------------------

            expo = conv / eps
            expo = np.clip(expo, a_min=None, a_max=50.0)
            w = np.exp(expo)
            w_sum = w.sum()
            if w_sum == 0.0:

            else:
                w /= w_sum

            #   ∇f = 2·(a ∗ w) .  The linear‑convolution implementation
            #   works with the reversed weight vector.

            idx = (2 * n - 2) - np.arange(n)      # pick the correct slice
            v = v_full[idx]
            grad = 2.0 * v                         # **correct** gradient

            # ----- e) Adam update --------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

            a = _project_onto_simplex(a)

            # ----- f) occasional random mass‑swap ----------------------
            if t % swap_every == 0:

                # move a modest fraction (up to 30% of a[i]) from i to j
                delta = min(a[i] * 0.3, a[i] * 0.5)
                if delta > 0.0:
                    a[i] -= delta
                    a[j] += delta
                    a = _project_onto_simplex(a)

            # ----- g) evaluate exact objective from time to time -------
            if t % eval_every == 0:
                cur_score = _exact_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[improve] restart={restart} iter={t} "
                          f"score={best_score:.6f}")

            # ----- h) safety break if we are very close to optimum ---
            # (optional early stop; keep it modest)
            if best_score < 1.503 and time.time() < deadline:
                # we have already reached the target – keep searching a bit
                pass

        # end of a single restart
        # Print a short summary for the restart
        print(f"[restart] {restart} finished, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (convert to Python list for the harness)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution (should never happen)
        best_seq = np.full(n, 1.0 / n, dtype=float)

    print(f"[final] best_score={best_score:.6f}, length={n}")
    return best_seq.tolist()
