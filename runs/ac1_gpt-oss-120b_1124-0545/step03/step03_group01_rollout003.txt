def subgradient_refine(a, n, deadline, max_iters=20000):
    best_a = a.copy()
    best_M = np.max(conv_full(a,a))
    a_curr = a.copy()
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        conv = conv_full(a_curr, a_curr)
        idx_max = int(np.argmax(conv))
        cur_M = conv[idx_max]
        if cur_M < best_M:
            best_M = cur_M
            best_a = a_curr.copy()
            print(f"[Sub] it {it} best M {best_M:.6f}")
        # subgradient g
        g = np.zeros_like(a_curr)
        # Determine overlap indices as earlier
        i_start = max(0, idx_max - (n - 1))
        i_end = min(n - 1, idx_max)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = idx_max - idx_i
        # g_i = 2 * a[rev_idx] (since derivative of a_i a_{j-i} w.r.t a_i appears twice)
        g[idx_i] = 2.0 * a_curr[rev_idx]
        # Mirror descent update: exponentiated gradient
        lr = 0.6 / np.sqrt(it)

        log_a -= np.max(log_a)
        a_curr = np.exp(log_a)
        a_curr /= a_curr.sum()
    return best_a

"""
Propose a step‑function sequence that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm combines three ideas:

1. **Diversified seeds** – a small collection of analytically‑generated
   distributions (uniform, geometric, power‑law, sine, symmetric) and a
   random Dirichlet draw.  The best seed (lowest auto‑correlation peak)
   is selected as a start point.

2. **Linear‑programming “boost”** – given a current sequence `f`,
   we solve the LP

        maximize   1ᵀg
        subject to  f ∗ g ≤ M       (component‑wise)
                    g ≥ 0

   where `M = max(f∗f)`.  The LP finds a direction `g` that
   carries as much mass as possible without increasing the peak
   auto‑correlation.  We mix `f` and the normalised `g` with a scalar
   `t∈[0,1]` that gives the largest reduction of the peak.  The
   procedure is repeated until no further improvement is observed
   or the time budget expires.

3. **Exact sub‑gradient refinement** – starting from the best LP‑enhanced
   sequence, a mirror‑descent (exponentiated‑gradient) update uses the
   true sub‑gradient of the max‑convolution.  This aggressively drives
   the peak down while staying on the probability simplex.

All steps respect the global deadline (default 1000 s) and keep the
vector non‑negative with unit ℓ₁‑norm; `evaluate_sequence` applies the
same scaling internally, so the returned list can be used directly.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, b: np.ndarray | None = None) -> np.ndarray:
    """Linear convolution (full) of a with b using a real FFT."""
    if b is None:
        b = a
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution of a."""
    return float(np.max(_conv_full(a)))


# ----------------------------------------------------------------------
# Seed families (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex via a Dirichlet(1,…,1)."""
    x = rng.gamma(shape=1.0, scale=1.0, size=n)
    s = x.sum()
    return x / s if s != 0 else _uniform(n)


# ----------------------------------------------------------------------
# Conv‑matrix for the LP (f * g ≤ M)
# ----------------------------------------------------------------------


def _conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix A satisfying (f ∗ g)_k = (A g)_k.
    A is lower‑triangular with constant diagonals f[0], f[1], ….
    """
    n = f.size
    m = 2 * n - 1
    # Avoid exact zeros – they would make the LP unbounded.
    f = np.where(f < 1e-12, 1e-12, f)
    data = []
    offsets = []
    for d in range(n):
        # diagonal offset = -d (col – row)
        offsets.append(-d)
        diag_len = m - d
        data.append(np.full(diag_len, f[d], dtype=np.float64))
    return sp.diags(data, offsets, shape=(m, n), format="csr")


# ----------------------------------------------------------------------
# LP‑boost iteration
# ----------------------------------------------------------------------


def _lp_boost(
    a: np.ndarray,
    deadline: float,
    max_iters: int = 12,
) -> np.ndarray:
    """
    Repeatedly solve the LP
          maximize   1ᵀg
          subject to f∗g ≤ M,   g ≥ 0,
    and mix the current sequence `f` with the normalised solution `g`.
    Returns the improved sequence (still on the simplex).
    """
    n = a.size
    f = a.copy()
    best_M = _max_conv(f)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # Build constraint matrix for the current f
        A = _conv_matrix(f)

        # Linear program data
        c = -np.ones(n, dtype=np.float64)          # maximize sum(g)
        b_ub = np.full(2 * n - 1, best_M, dtype=np.float64)

        # Solve with HiGHS (fast for large sparse LPs)
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # If the LP fails, abort the boost phase.
            print(f"[LP] iteration {it}: solver failed – {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print(f"[LP] iteration {it}: trivial solution")
            break

        # Normalise g to stay on the simplex.
        g_norm = g / sum_g

        # ----- line‑search for the best mixing weight t -----
        # coarse grid
        t_grid = np.linspace(0.0, 1.0, 21)
        best_t = 0.0
        best_Mt = best_M
        for t in t_grid:
            new_f = (1 - t) * f + t * g_norm
            Mt = _max_conv(new_f)
            if Mt < best_Mt:
                best_Mt = Mt
                best_t = t

        # refine in the neighbourhood of the best coarse point
        if 0.0 < best_t < 1.0:
            fine_grid = np.linspace(max(0.0, best_t - 0.1), min(1.0, best_t + 0.1), 21)
            for t in fine_grid:
                new_f = (1 - t) * f + t * g_norm
                Mt = _max_conv(new_f)
                if Mt < best_Mt:
                    best_Mt = Mt
                    best_t = t

        if best_t == 0.0:
            # No improvement possible – stop early.
            print(f"[LP] iteration {it}: no improvement (M={best_M:.6f})")
            break

        # Update the current sequence.
        f = (1 - best_t) * f + best_t * g_norm
        best_M = best_Mt
        print(f"[LP] iter {it:2d} – t={best_t: .3f} – new M={best_M:.6f}")

    return f


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror‑descent on the simplex)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start: np.ndarray,
    deadline: float,
    max_iters: int = 20000,
) -> np.ndarray:
    """
    Perform exponentiated‑gradient updates using the true sub‑gradient
    of the max‑convolution.  The iterate stays on the probability simplex.
    """
    n = start.size
    a = start.copy()
    best_a = a.copy()
    best_M = _max_conv(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_full(a)
        k = int(np.argmax(conv))
        cur_M = conv[k]

        if cur_M < best_M:
            best_M = cur_M
            best_a = a.copy()
            print(f"[Sub] it {it:5d} – new best M={best_M:.6f}")

        # Sub‑gradient of max(conv) w.r.t. a (see analysis for indexing)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Exponentiated‑gradient (mirror descent) step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # numerical stability
        a = np.exp(log_a)
        a /= a.sum()                    # project back onto simplex

    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the given
    time budget (default 1000 s) and returns the best sequence found
    as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size – the previous best was at n=4000.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build a modest collection of seeds (all sum to 1).
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 1234)
    seeds: list[np.ndarray] = [
        _uniform(n),
        _geom(n, 0.9),
        _geom(n, 0.95),
        _geom(n, 0.99),
        _power(n, 0.5),
        _power(n, 1.0),
        _power(n, 1.5),
        _sym_geom(n, 0.95),
        _sine_window(n),
        _random_dirichlet(n, rng),
    ]

    # incorporate a previously found solution if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Pick the seed with the lowest initial max‑convolution.
    # ------------------------------------------------------------------
    best_seed = None
    best_M = float("inf")
    for i, s in enumerate(seeds):
        if time.time() > deadline:
            break
        M = _max_conv(s)
        if M < best_M:
            best_M = M
            best_seed = s.copy()
        print(f"[Seed] {i:2d} – M={M:.6f}")

    if best_seed is None:
        best_seed = _uniform(n)

    print(f"[Init] chosen seed has M={best_M:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – LP boost (handles the linear constraints f∗g ≤ M).
    # ------------------------------------------------------------------
    seq_lp = _lp_boost(best_seed, deadline, max_iters=12)

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient refinement (mirrored gradient descent).
    # ------------------------------------------------------------------
    seq_refined = _subgradient_refine(seq_lp, deadline, max_iters=20000)

    # ------------------------------------------------------------------
    # Final sanity check (optional – ensures we are not violating the API).
    # ------------------------------------------------------------------
    try:
        # The evaluator may already be in the global namespace.
        evaluate_sequence  # noqa: F401
    except NameError:
        # If it is missing we simply return the sequence; the caller will
        # evaluate it later.
        pass

    final_seq = seq_refined / seq_refined.sum()   # enforce exact sum = 1
    final_obj = 2.0 * n * _max_conv(final_seq)    # same as evaluate_sequence for sum=1
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")
    return final_seq.tolist()
