"""
Parametric family search for step function minimizing 2*n*max(conv(a,a)).
We restrict to symmetric densities a_i ∝ ((i+0.5)(N-i-0.5))^{-p} mixed with a uniform
component. This family contains the known arcsine shape (p=0.5) which is near‑optimal.
A simple grid search over exponent p and mixing weight ε is performed,
using FFT‑based convolution for fast exact evaluation of the objective.
The best sequence found within the time budget is returned.
"""

import time
import numpy as np

def _conv_max(a: np.ndarray) -> float:
    """Return max of linear convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(conv.max())

def _score(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum=1)."""
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)

def _make_shape(n: int, p: float) -> np.ndarray:
    """Return a symmetric shape a_i ∝ ((i+0.5)(n-i-0.5))^{-p}, normalised."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    raw = denom ** (-p)
    a = raw / raw.sum()
    return a

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation function.
    The algorithm explores a 2‑parameter family (exponent p, uniform mix ε)
    and optionally refines the best candidate with a few Adam updates.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # Choose a length that balances discretisation error and speed.
    # For the full time budget we use a fairly large n.
    # If the remaining time is small we fall back to a modest size.
    if budget_s >= 600:
        n = int(kwargs.get("n", 2000))
        n = max(n, 12000)          # at least 12000 entries
    else:
        n = int(kwargs.get("n", 2000))

    # Pre‑compute the uniform distribution (already normalised)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # Grid for exponent p (around the known arcsine value 0.5)
    p_grid = np.linspace(0.40, 0.60, 41)   # step 0.005
    # Grid for mixing weight ε (fraction of uniform part)
    eps_grid = np.linspace(0.0, 0.20, 21)   # step 0.01

    best_score = float("inf")
    best_a = None
    best_p = None
    best_eps = None

    # ---- Coarse grid search -------------------------------------------------
    for p in p_grid:
        if time.time() >= deadline:
            break
        base = _make_shape(n, p)               # already normalised
        for eps in eps_grid:
            a = (1.0 - eps) * base + eps * uniform
            # a is already on the simplex because both components sum to 1
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                best_p = p
                best_eps = eps
                print(f"[grid] new best= {best_score:.6f} (p={best_p:.4f}, eps={best_eps:.4f})")

    # ---- Local refinement around the best point -----------------------------
    # Reduce step sizes and perform a simple hill‑climb.
    p_step = 0.005
    eps_step = 0.005
    while p_step > 1e-4 and time.time() < deadline:
        improved = False
        for dp in (-p_step, 0.0, p_step):
            p_new = best_p + dp
            if p_new <= 0.0:
                continue
            base_new = _make_shape(n, p_new)
            for de in (-eps_step, 0.0, eps_step):
                eps_new = best_eps + de
                if eps_new < 0.0 or eps_new > 0.3:
                    continue
                a_new = (1.0 - eps_new) * base_new + eps_new * uniform
                sc_new = _score(a_new)
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    best_p = p_new
                    best_eps = eps_new
                    improved = True
                    print(f"[refine] improved to {best_score:.6f} (p={best_p:.5f}, eps={best_eps:.5f})")
        if not improved:
            p_step *= 0.5
            eps_step *= 0.5

    # ---- Optional few Adam‑style updates starting from the best vector -----
    # This step can push the solution a little further without much overhead.
    # We keep the representation in the softmax space to stay on the simplex.
    if best_a is not None and time.time() < deadline:
        # Softmax parametrisation
        v = np.log(best_a + 1e-12)
        m = np.zeros_like(v)
        v_hat = np.zeros_like(v)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        it = 0
        # schedule: we keep a modest learning‑rate because we are already close.
        while time.time() < deadline and it < 2000:
            it += 1
            a = np.exp(v - np.max(v))
            a /= a.sum()
            # exact max – we need its index for sub‑gradient
            fa = np.fft.rfft(a, 1 << ((2 * n - 2).bit_length()))
            conv = np.fft.irfft(fa * fa, 1 << ((2 * n - 2).bit_length()))[: 2 * n - 1]
            k = int(np.argmax(conv))
            # gradient of max‐entry (sub‑gradient)
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            g = np.zeros_like(a)
            g[valid] = 2.0 * a[idx[valid]]
            grad = 2.0 * n * g
            # chain rule through soft‑max
            dot_ag = np.dot(a, grad)
            grad_v = a * (grad - dot_ag)

            # Adam update
            lr = 0.02 / np.sqrt(it)     # simple decreasing rate
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat_corr = v_hat / (1.0 - beta2 ** it)
            v -= lr * m_hat / (np.sqrt(v_hat_corr) + eps)

            # keep the best exact score seen so far
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[adam] it={it:05d} new best={best_score:.6f}")

    # Safety clamp (the evaluation function also clamps, but we keep things tidy)
    best_a = np.maximum(best_a, 0.0)
    total_time = time.time() - start
    print(f"[done] total time={total_time:.1f}s best score={best_score:.6f}")
    return best_a.tolist()

Parametric family search for a step‑function minimizing  

    E(a) = 2·n·max(conv(a,a)) / (sum(a)^2) .

Because the objective is scale‑invariant we restrict ourselves to probability

    a_i ∝ ((i+½)(n−i−½))^{−½} ,

which is already close to optimal.  We explore a two‑parameter family that
contains this shape:

    a(p,ε) = (1‑ε)·a_p + ε·uniform ,

where a_p is the normalized arcsine‑type vector with exponent p and
uniform is the flat distribution.  By scanning p around 0.5 and ε in
[0,0.2] we quickly find a very good candidate.  A small local
refinement (hill‑climbing) is then performed, followed – only for modest
sizes – by a few Adam‑style updates in the soft‑max (log‑parameter)
