"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in two phases:

1. **Extensive analytic grid search** – we test many families of
   simple closed‑form shapes (product, sum of power‐laws and a Beta‑type
   shape) on a fine grid of parameters.  The best result of this stage
   is taken as a starting point.
2. **Exponentiated sub‑gradient descent** – the objective is the
   max‑convolution, whose sub‑gradient at the current maximum index k
   is `g_i = 2·a_{k‑i}` (zero outside the support).  We perform an
   exponentiated update `a_i ← a_i·exp(‑η·g_i)` and renormalise to the
   probability simplex.  A linear schedule for the step size `η`,
   a Metropolis temperature and occasional random mass‑transfer moves
   helps to escape local minima.  The loop runs until the time budget
   (≈ 1000 s) expires, always keeping the best sequence seen.

The final sequence (as a plain Python list) is returned.
"""

# ----------------------------------------------------------------------
# Imports and low‑level helpers (no closures)
# ----------------------------------------------------------------------
import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself via FFT.
    Returns length 2·n‑1 array.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]

def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a.
    The vector is (re)normalised to sum‑1 internally.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s                     # normalise to the simplex
    conv = conv_fft(a)
    max_b = np.max(conv)
    # denominator = (∑a)² = 1, factor 2·n remains
    return float(2.0 * a.shape[0] * max_b)

def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of mass from a random index to another.
    The result is projected back onto the simplex.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Closed‑form candidate.
    mode='product' : a_i ∝ ((i+δ)*(n‑1‑i+δ))^{‑p}
    mode='sum'     : a_i ∝ (i+δ)^{‑p} + (n‑1‑i+δ)^{‑p}
    The vector is normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    # ensure numeric safety for log(0)
    off = max(offset, 1e-12)

    if mode == "product":
        # log a_i = -p * (log(i+off) + log(n‑1‑i+off))
        log_a = -p * (np.log(i + off) + np.log(n - 1 - i + off))
        a = np.exp(log_a - np.max(log_a))
    elif mode == "sum":
        # a_i = (i+off)^{-p} + (n‑1‑i+off)^{-p}
        log_term1 = -p * np.log(i + off)
        log_term2 = -p * np.log(n - 1 - i + off)
        term1 = np.exp(log_term1 - np.max(log_term1))      # stabilise
        term2 = np.exp(log_term2 - np.max(log_term2))
        a = term1 + term2
    else:
        raise ValueError("mode must be 'product' or 'sum'")

    total = a.sum()
    if total == 0.0 or not np.isfinite(total):
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a

def beta_candidate(n: int, alpha: float, beta: float, offset: float) -> np.ndarray:
    """
    Beta‑type shape: a_i ∝ (i+δ)^{α‑1} * (n‑1‑i+δ)^{β‑1}
    """
    i = np.arange(n, dtype=np.float64)
    off = max(offset, 1e-12)
    log_a = (alpha - 1.0) * np.log(i + off) + (beta - 1.0) * np.log(n - 1 - i + off)
    # avoid overflow/underflow
    a = np.exp(log_a - np.max(log_a))
    total = a.sum()
    if total == 0.0 or not np.isfinite(total):
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a

# ----------------------------------------------------------------------
# Sub‑gradient based update (exponentiated)
# ----------------------------------------------------------------------
def subgradient_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)_k at the current maximiser k_max.
    Returns g_i = 2·a_{k_max‑i} for valid indices, else 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    start_idx = max(0, k_max - (n - 1))
    end_idx   = min(n, k_max + 1)
    idx = np.arange(start_idx, end_idx, dtype=int)
    g[idx] = 2.0 * a[k_max - idx]
    return g

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function minimising the given evaluation function.
    Returns the best sequence found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Problem size (default 2000, can be overridden via kwargs)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1) Very fine analytic grid search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    best_mode = None
    best_p = None
    best_off = None

    print("[init] start analytic grid search (n = {})".format(n))

    # product / sum families – p in [0.05, 2.0] step ≈0.01, offset ∈[0,4] step 0.1
    p_vals = np.linspace(0.05, 2.0, 196)          # 196 points
    off_vals = np.linspace(0.0, 4.0, 41)          # 41 points
    for mode in ("product", "sum"):
        for p in p_vals:
            for off in off_vals:
                cand = analytic_candidate(n, p, off, mode)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    best_mode = mode
                    best_p = p
                    best_off = off
        print(f"[grid] mode={mode} completed, best={best_score:.6f}")

    # Beta‑type family (symmetric exponents)
    alpha_vals = np.linspace(0.2, 3.0, 28)   # step ≈0.1
    beta_vals  = np.linspace(0.2, 3.0, 28)
    for alpha in alpha_vals:
        for beta in beta_vals:
            cand = beta_candidate(n, alpha, beta, offset=0.5)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                best_mode = "beta"
                # store parameters for possible debugging
                best_alpha = alpha
                best_beta = beta

    # ------------------------------------------------------------------
    # 2) Incorporate a globally supplied reference if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref /= ref.sum()
            sc = compute_score(ref)
            if sc < best_score:
                best_score = sc
                best_seq = ref.copy()
                best_mode = "global_ref"

    print(f"[init] best analytic start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Exponentiated sub‑gradient descent with annealing
    # ------------------------------------------------------------------
    a = best_seq.copy()                     # current iterate (already on simplex)

    # Schedules (linear in time)
    eta_start, eta_end = 0.5, 5e-4          # step size for exponentiated update
    temp_start, temp_end = 1.0, 0.001      # Metropolis temperature

    iteration = 0
    last_improve = start_time
    restart_cooldown = 30.0                # seconds without progress before jitter

    while time.time() < deadline:
        # --------------------------------------------------------------
        #  a) compute current convolution and its maximiser k
        # --------------------------------------------------------------
        conv = conv_fft(a)
        k_max = int(np.argmax(conv))

        # --------------------------------------------------------------
        #  b) sub‑gradient at the current maximum
        # --------------------------------------------------------------
        g = subgradient_max(a, k_max)

        # --------------------------------------------------------------
        #  c) schedules
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 20.0)          # leave small margin
        eta = eta_start + progress * (eta_end - eta_start)
        temp = temp_start + progress * (temp_end - temp_start)

        # --------------------------------------------------------------
        #  d) decide move type
        # --------------------------------------------------------------
        if np.random.rand() < 0.9:                      # main exponentiated move
            # exponentiated update in log‑domain for numerical stability
            log_a = np.log(a + 1e-32)
            log_a -= eta * g
            log_a -= np.max(log_a)                      # prevent overflow
            cand = np.exp(log_a)
            cand_sum = cand.sum()
            if cand_sum == 0.0:
                cand = a.copy()
            else:
                cand = cand / cand_sum
        else:                                            # occasional random swap
            cand = random_move(a, scale=eta * 0.5)

        cand_score = compute_score(cand)

        # --------------------------------------------------------------
        #  e) Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            if temp > 1e-12:
                prob = math.exp(-(cand_score - best_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = cand
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:07d}] new best = {best_score:.6f}  (t={elapsed:.1f}s)")

        # --------------------------------------------------------------
        #  f) jitter restart if stagnated
        # --------------------------------------------------------------
        if (iteration % 20000 == 0) and (time.time() - last_improve > restart_cooldown):
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = a / a.sum()
            print(f"[{iteration:07d}] jitter restart (stagnation)")

        # --------------------------------------------------------------
        #  g) early exit if target reached
        # --------------------------------------------------------------
        if best_score <= 1.503:
            print("[done] target threshold reached, exiting early")
            break

        iteration += 1

    total_time = time.time() - start_time
    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}, time = {total_time:.1f}s")
    # return as plain Python list
    return best_seq.tolist()
