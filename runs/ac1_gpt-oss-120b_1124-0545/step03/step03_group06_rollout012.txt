def _cross_corr(seq, w):
    n = len(seq)
    L = _next_pow2(n + len(w) - 1)
    fa = np.fft.rfft(seq, n=L)
    fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(fa * np.conj(fw), n=L)
    return corr[:n]

    best_seq = seq.copy()
    best_score = _score_fft(seq)
    step = step0
    gamma = gamma0
    iteration = 0
    start = time.time()

    while iteration < max_iters and (time.time() - start) < time_budget:
        iteration += 1

        # FFT of seq
        FA = np.fft.rfft(seq, n=L)
        # conv = invFFT(FA * FA)
        conv = np.fft.irfft(FA * FA, n=L)[: 2 * n - 1]
        max_conv = conv.max()
        # compute w
        # gamma param ensures softmax; we want w sum = 1 for correlation.
        exp_arg = gamma * (conv - max_conv)   # max becomes zero
        # Use np.exp safely
        w = np.exp(exp_arg)

   The best candidate is kept.
2. Refine the candidate with a projected gradient descent that
   minimises a smooth approximation of the max‑convolution.  The
   approximation uses a soft‑max (log‑sum‑exp) of the self‑convolution
   with a temperature γ that is slowly increased; the gradient is
   obtained by a single FFT‑based cross‑correlation.  After each step
   the vector is projected onto the probability simplex {x ≥ 0,
   Σx=1}.  Occasional random mass swaps and optional symmetrisation
   help escape plateaus.
3. Perform a few short random‑restart runs from random power‑law
   seeds and keep the best result.
4. A final polishing phase with a high γ and a very small step size
   yields the final sequence.

All operations are O(n log n) and fit easily into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution using the FFT (real version)."""
    n = seq.shape[0]
    size = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    # eliminate tiny negatives from round‑off
    conv[conv < 0.0] = 0.0
    return conv


def _score_fft(seq: np.ndarray) -> float:
    """
    Exact objective value for a normalised sequence (Σseq == 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


def _run_opt_gradient(
    seq: np.ndarray,
    time_budget: float,
    *,
    max_iters: int = 10 ** 9,
    step0: float = 0.5,
    decay: float = 0.9999,
    gamma0: float = 0.5,
    gamma_factor: float = 1.002,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a soft‑max approximation of the
    max‑convolution.  The gradient is computed via a single FFT‑based
    cross‑correlation.  The iterate is kept on the simplex.
    """
    n = seq.shape[0]
    # a length that is sufficient for both the convolution and the
    # correlation (≤ 3·n).
    L = _next_pow2(3 * n - 2)

    best_seq = seq.copy()
    best_score = _score_fft(seq)

    step = step0
    gamma = gamma0
    iteration = 0
    start = time.time()

    while iteration < max_iters and (time.time() - start) < time_budget:
        iteration += 1

        # ----- self‑convolution (FFT) -----
        FA = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(FA * FA, n=L)[: 2 * n - 1]
        max_c = conv.max()

        # ----- soft‑max weights -----
        # w_k ∝ exp(gamma * (conv_k - max_c))  → sum w = 1
        exp_arg = gamma * (conv - max_c)     # max_c makes the largest entry zero
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ----- gradient = 2·n·cross‑correlation(seq, w) -----
        FW = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(FA * np.conj(FW), n=L)[:n]
        grad = 2.0 * corr * n

        # ----- gradient step + projection -----
        seq = seq - step * grad
        seq = _project_simplex(seq)

        # optional symmetrisation (helps when the optimum is symmetric)
        if iteration % 10 == 0:
            seq = 0.5 * (seq + seq[::-1])

        # occasional random mass exchange to avoid tiny plateaus
        if iteration % 200 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(step * seq[i], seq[i] * 0.3)
            seq[i] -= eps
            seq[j] += eps
            seq = _project_simplex(seq)

        # ----- bookkeeping -----
        cur_score = _score_fft(seq)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}, gamma={gamma:.3f}")

        # ----- schedules -----
        step = max(step * decay, 1e-12)
        gamma = min(gamma * gamma_factor, 100.0)

    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:

    """

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
