"""
Sub‑gradient descent on the exact maximal auto‑convolution.

The objective (with sum = 1) is   F(a) = 2·n·max(conv(a,a)).
We keep a on the probability simplex using a soft‑max
parameterisation a = softmax(v).  At each iteration we
* compute the full convolution via FFT,
* identify the (few) indices where the maximum is attained,
* obtain the exact sub‑gradient of max(conv) w.r.t. a
  (a sum of shifted copies of a),
* map it to a gradient w.r.t. v,
* apply an Adam update,
* project the best ever seen sequence.

A simple restart (log‑perturbation of the best point) is triggered
when no improvement has been observed for 20 s.  The routine
uses a larger discrete size (default ≈ 12 k) than the reference
solution, which empirically reduces the bound towards the
theoretical limit (~1.5) and is able to beat the target 1.503
within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    v_shift = v - np.max(v)
    e = np.exp(v_shift)
    return e / e.sum()


def initialise_a(n: int, seed: int = 42) -> np.ndarray:
    """
    Initialise a good starting distribution.
    Re‑use a previously discovered sequence if present,
    otherwise fall back to the classic arcsine shape.
    The vector is normalised to sum = 1.
    """
    rng = np.random.default_rng(seed)

    # reuse previous best if it matches the required length
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.size == n and np.all(prev >= 0):
            prev = prev / prev.sum()
            return prev

    # arcsine (inverse‑sqrt) shape
    i = np.arange(n, dtype=float)
    a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a0 = a0 / a0.sum()
    return a0


def exact_score(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function for a probability vector `a`
    (sum = 1).  Uses FFT for speed.
    """
    n = a.shape[0]
    fft_len = 1 << ((3 * n - 2).bit_length())
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence of heights that minimises
    `evaluate_sequence`.  The routine respects the time budget (default
    1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1)  problem size
    # ------------------------------------------------------------------
    # default length – a compromise between quality and speed.
    # The caller may override it via the `n` keyword.
    n = int(kwargs.get("n", 12000))
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 2)  initialise a on the simplex
    # ------------------------------------------------------------------
    a = initialise_a(n, seed=seed)
    v = np.log(a + 1e-12)                     # soft‑max parameters

    # ------------------------------------------------------------------
    # 3)  Adam state and hyper‑parameters
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    lr_start, lr_end = 0.12, 0.001           # linear schedule
    # no explicit temperature – we use the exact max

    # bookkeeping
    best_a = a.copy()
    best_score = exact_score(a)
    last_improve = start_time
    it = 0

    # pre‑allocate FFT length for all transforms (next power of two ≥ 3·n‑2)
    fft_len = 1 << ((3 * n - 2).bit_length())

    # tolerance to decide which convolution entries are “max”
    # (absolute because the objective is scale‑invariant)
    max_tol = 1e-12

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 4)  current point and its convolution
        # ------------------------------------------------------------------
        a = softmax(v)                         # stays on the simplex

        # full self‑convolution using FFT
        fa = np.fft.fft(a, fft_len)
        conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv_full.max()

        # current exact score (no need to divide by sum – sum is 1)
        cur_score = 2.0 * n * max_val

        # ------------------------------------------------------------------
        # 5)  sub‑gradient of the max‑convolution term
        # ------------------------------------------------------------------
        # mask of the maximal entries (could be several because of numerical ties)
        mask = np.zeros(2 * n - 1, dtype=float)
        mask[conv_full >= max_val - max_tol] = 1.0

        # shift‑sum via convolution of reversed mask with a
        rev_mask = mask[::-1]                  # length 2n‑1
        f_rev = np.fft.fft(rev_mask, fft_len)

        # linear convolution (full length = 3n‑2)
        conv_shifts_full = np.real(np.fft.ifft(f_rev * fa))[: 3 * n - 2]

        # Extract the first‑n shifted‑sums in the correct order
        # (see derivation in the analysis section)
        sum_shifts = conv_shifts_full[n - 1: 2 * n - 1][::-1]

        # sub‑gradient of F = 2·n·max(conv) w.r.t. a
        grad_a = 4.0 * n * sum_shifts          # factor 4·n = 2·n * (2 * shift‑sum)

        # ------------------------------------------------------------------
        # 6)  map to gradient w.r.t. the soft‑max parameters v
        # ------------------------------------------------------------------
        dot_ag = np.dot(a, grad_a)
        grad_v = a * (grad_a - dot_ag)

        # ------------------------------------------------------------------
        # 7)  Adam update
        # ------------------------------------------------------------------
        it += 1
        # linear learning‑rate schedule
        p = min(1.0, (time.time() - start_time) / (budget_s - 10.0))
        lr = lr_start * (1.0 - p) + lr_end * p

        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v -= lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # ------------------------------------------------------------------
        # 8)  keep the best ever seen
        # ------------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start_time
            print(f"[{it:07d}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # ------------------------------------------------------------------
        # 9)  occasional stochastic restart if stuck
        # ------------------------------------------------------------------
        if it % 4000 == 0:
            if time.time() - last_improve > 20.0:
                # re‑initialise v near the current best with tiny jitter
                jitter = np.random.randn(n) * 1e-4
                v = np.log(best_a + 1e-12) + jitter
                m.fill(0.0)
                v_hat.fill(0.0)
                print(f"[{it:07d}] restart after stagnation (t={time.time() - start_time:.1f}s)")

    total = time.time() - start_time
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")

    # ------------------------------------------------------------------
    # 10)  final safety – enforce non‑negativity and normalise
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    best_a = best_a / best_a.sum()          # guard against numerical drift

    # expose the sequence under the expected name for possible reuse
    globals()["height_sequence_1"] = best_a.tolist()

    return best_a.tolist()
