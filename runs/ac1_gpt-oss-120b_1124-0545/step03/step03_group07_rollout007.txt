"""
Propose a step‑function sequence by homotopy gradient descent on a smooth
approximation of the max‑convolution, followed by a short hill‑climbing
phase.  The loss is the log‑sum‑exp of the self‑convolution (a smooth
upper bound on the maximum).  Gradients are obtained analytically via
FFT‑based convolutions, and each update is projected onto the
ℓ₁‑simplex (∑a = √(2·n), a ≥ 0).  Adam (first‑order) optimisation is used
with a schedule of increasing smoothing (β) to drive the solution toward
the true minimax objective.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , ∑ w = s }."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)
    n = a.size
    return conv_full[: 2 * n - 1]


def _gradient_logsumexp(
    a: np.ndarray, beta: float, fft_len: int
) -> tuple[float, np.ndarray, float]:
    """
    Smooth surrogate of max(conv) = log‑sum‑exp(β·conv)/β.
    Returns (loss, gradient, current max‐convolution).
    Gradient = 2 * (w * a_rev) / Z, where w_i = exp(β·(conv_i‑maxc)).
    """
    n = a.size
    # 1. self‑convolution
    conv = _self_convolution_fft(a, fft_len)           # length 2n‑1
    maxc = float(conv.max())

    # 2. soft‑max weights (numerically stable)
    w = np.exp(beta * (conv - maxc))
    Z = float(w.sum()) + 1e-30

    # 3. gradient via convolution of w with reversed a
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    f_rev = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * f_rev, n=fft_len)
    # the slice that corresponds to sum_i w_i a_{i‑k}
    start = n - 1
    grad_slice = corr_full[start : start + n]
    grad = 2.0 * grad_slice / Z

    # 4. loss (smooth surrogate of max)
    loss = (maxc + math.log(Z)) / beta
    return loss, grad, maxc


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    evaluate_sequence (i.e. minimise the maximal self‑convolution value).
    A homotopy over the smoothing parameter β combined with Adam updates
    and a final hill‑climbing phase is used.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ---------- problem size ----------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=float)
        n = base.size
        a = base.copy()
    else:
        n = 2500                # generous default
        a = np.random.exponential(scale=1.0, size=n)

    target_sum = math.sqrt(2.0 * n)   # ℓ₁‑norm we enforce (makes objective = max conv)

    # normalise to the simplex
    a = a / a.sum() * target_sum

    # ---------- FFT set‑up ----------
    fft_len = _next_pow2(3 * n - 2)   # enough for conv + weighted conv

    # ---------- optimisation variables ----------
    best_seq = a.copy()
    conv = _self_convolution_fft(a, fft_len)
    best_val = float(conv.max())
    print(f"[INIT] n={n}, target_sum={target_sum:.3f}, init max(conv)={best_val:.6f}")

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0

    # schedule of (beta, learning_rate, max_iters) – beta grows,
    # learning_rate shrinks to keep updates stable.
    schedule = [
        (0.5, 0.6, 3000),
        (1.0, 0.5, 3000),
        (2.0, 0.4, 4000),
        (5.0, 0.3, 5000),
        (10.0, 0.2, 6000),
        (20.0, 0.15, 8000),
        (40.0, 0.1, 10000),
        (80.0, 0.07, 12000),
    ]

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a small safety margin

    # ---------- homotopy gradient descent ----------
    last_improve_iter = 0
    for beta, lr, max_iters in schedule:
        for _ in range(max_iters):
            if time.time() >= deadline:
                break

            loss, grad, cur_max = _gradient_logsumexp(a, beta, fft_len)

            # Adam update
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto simplex (keeps sum = target_sum & non‑negativity)
            a_new = _simplex_projection(a_new, target_sum)

            a = a_new

            # evaluate actual max‑convolution (the true objective)
            conv = _self_convolution_fft(a, fft_len)
            cur_max = float(conv.max())

            if cur_max < best_val - 1e-12:
                best_val = cur_max
                best_seq = a.copy()
                last_improve_iter = t_iter
                print(f"[PHASE β={beta:.1f}] iter {t_iter:6d} new best max(conv)={best_val:.6f}")

            # restart if stuck for a long while
            if t_iter - last_improve_iter > 8000:
                # inject a small random perturbation around the current best
                a = best_seq.copy()
                jitter = np.random.exponential(scale=0.3, size=n)
                a = a + jitter
                a = a / a.sum() * target_sum
                m.fill(0.0)
                v.fill(0.0)
                t_iter = 0
                last_improve_iter = 0
                print(f"[RESTART] β={beta:.1f} after stagnation")

            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # ---------- final hill‑climbing (simple SA) ----------
    cur_max = best_val
    a = best_seq.copy()
    step_counter = 0
    while time.time() < deadline:
        # propose a random transfer of a small fraction of mass
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = a[i] * 0.10 * np.random.rand()
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta
        # projection is unnecessary (ℓ₁‑sum unchanged) but keep non‑negativity
        a_cand = _simplex_projection(a_cand, target_sum)

        conv_cand = _self_convolution_fft(a_cand, fft_len)
        cand_max = float(conv_cand.max())

        if cand_max < cur_max:
            a = a_cand
            cur_max = cand_max
            if cand_max < best_val:
                best_val = cand_max
                best_seq = a_cand.copy()
                print(f"[SA step {step_counter:6d}] new best max(conv)={best_val:.6f}")
        else:
            # accept uphill moves with a tiny temperature that decays
            temperature = best_val * 0.01 * max(0.0, 1.0 - (time.time() - start_time) / budget_s)
            prob = math.exp(-(cand_max - cur_max) / max(temperature, 1e-12))
            if random.random() < prob:
                a = a_cand
                cur_max = cand_max

        step_counter += 1

    print(f"[DONE] best max(conv) = {best_val:.6f} after {t_iter} Adam iters "
          f"and {step_counter} SA steps")
    # Return the best sequence found as a plain list
    return best_seq.tolist()
