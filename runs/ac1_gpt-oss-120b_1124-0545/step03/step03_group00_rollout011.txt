```python
def _refine_sequence(a0, n, deadline, start_beta=1.0, beta_mul=1.03, lr0=0.5):
    v = np.log(a0 + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    best_seq = a0.copy()
    conv = _conv_fft(a0, a0)
    best_val = 2.0 * n * np.max(conv)
    beta = start_beta
    it = 0
    while time.time() < deadline:
        it += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine] n={n} it={it} best={cur_val:.6f}")
        # compute smooth weights
        beta *= beta_mul
        shift = cur_max  # to improve numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = np.sum(w)
        if w_sum == 0:
            continue
        w /= w_sum
        # gradient w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        # full length = 3n-2, core part indices n-1:2n-1
        grad_a = 2.0 * full[n-1 : n-1 + n]
        # gradient w.r.t logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # gradient clipping
        if np.max(np.abs(grad_v)) > 10.0:
            grad_v = grad_v / np.max(np.abs(grad_v)) * 10.0
        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)
        m_hat = m / (1 - 0.9 ** it)
        v2_hat = v2 / (1 - 0.999 ** it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)
    return best_seq, best_val
```

```python
        # include previous solution if exists

        if prev is not None:

            if a_prev.ndim == 1:

            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[scan] new global best from seed {idx} = {cur_val:.6f}")

        refined_seq, refined_val = _refine_sequence(
            global_best_seq, n, deadline,
            start_beta=1.0, beta_mul=1.03, lr0=0.5
        )

        # perturb restarts

            pert_seq = _softmax(noisy_logits)
            seq2, val2 = _refine_sequence(pert_seq, n, deadline,
                                          start_beta=1.0, beta_mul=1.03, lr0=0.5)
            if val2 < global_best_val:
                global_best_val = val2
                global_best_seq = seq2.copy()
                print(f"[perturb] improved global best to {val2:.6f}")

    # final
    if global_best_seq is None:
        n = 4000
        global_best_seq = np.ones(n) / n

    # normalize
    s = np.sum(global_best_seq)
    if s > 0:
        global_best_seq = global_best_seq / s

    result = global_best_seq.tolist()
    print(f"[final] best objective (fast) = {global_best_val:.6f}")
    return result
```

"""
Propose a non‑negative step‑function (height sequence) with a small value of
evaluate_sequence().

The algorithm works in three phases for each chosen length `n`:
1. Generate a diverse pool of analytic seeds (geometric, power‑law,
   beta‑distribution families – including the arcsine law, sine‑power,
   symmetric geometric, Dirichlet draws and random soft‑max vectors).
2. Scan all seeds with a fast FFT‑based convolution to keep the best.
3. Refine the current best using Adam on the log‑softmax parametrisation.
   The loss is a smoothed max (log‑sum‑exp) of the convolution; the
   smoothing parameter β starts small and is increased geometrically.
   After the main refinement we perform cheap perturbed restarts and a
   short smoothing post‑process, always keeping the best sequence seen.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Output length is len(x) + len(y) - 1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (maps ℝⁿ → simplex)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Analytic families of candidates
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _sin_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** p
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _beta_distribution(n: int, a_param: float, b_param: float) -> np.ndarray:
    """
    Symmetric (or asymmetric) Beta distribution evaluated at the mid‑points
    of n equal sub‑intervals of [0,1].
    a_param, b_param > 0.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid 0^neg and 1^neg by clipping
    eps = np.finfo(np.float64).tiny
    x = np.clip(x, eps, 1 - eps)
    a = (x ** (a_param - 1)) * ((1 - x) ** (b_param - 1))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Optimisation core (Adam on log‑softmax)
# ----------------------------------------------------------------------


def _refine_sequence(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 0.5,
    beta_mul: float = 1.04,
    lr0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.
    Returns the best sequence found (as np.ndarray) and its fast objective
    value 2·n·max(conv) (identical to evaluate_sequence when sum==1).
    """
    v = np.log(a0 + 1e-30)               # logits
    m = np.zeros_like(v)                # 1st moment
    v2 = np.zeros_like(v)               # 2nd moment
    best_seq = a0.copy()
    conv = _conv_fft(a0, a0)
    best_val = 2.0 * n * np.max(conv)

    beta = start_beta
    it = 0

    while True:
        if time.time() > deadline:
            break
        it += 1

        a = _softmax(v)                  # probability vector on simplex

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        beta *= beta_mul
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient w.r.t. a (smooth loss) ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]     # central n entries

        # ---- gradient w.r.t. logits (softmax) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- optional clipping for stability ----
        max_abs = np.max(np.abs(grad_v))
        if max_abs > 10.0:
            grad_v *= 10.0 / max_abs

        # ---- Adam update ----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


def _smooth_seq(seq: np.ndarray, kernel: np.ndarray) -> np.ndarray:
    """
    Simple linear smoothing (convolution with `kernel` and renormalisation).
    Guarantees non‑negativity and unit sum.
    """
    sm = np.convolve(seq, kernel, mode="same")
    sm = np.maximum(sm, 0.0)
    total = sm.sum()
    if total > 0.0:
        sm /= total
    return sm


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The routine respects the supplied time budget and
    returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # small safety margin

    # ------------------------------------------------------------------
    # Lengths to try – start with a medium size that allows many iterations.
    # If time permits we also try a larger size.
    # ------------------------------------------------------------------
    lengths = [4000, 8000]   # second length only if enough time remains
    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break

        # reserve a tiny slice of the budget for post‑processing
        length_deadline = deadline - 0.5

        print(f"[main] Optimising length = {n}")

        # --------------------------------------------------------------
        # Build a diverse collection of seed sequences (all on the simplex)
        # --------------------------------------------------------------
        seed_seqs = []

        # Geometric families
        for c in np.linspace(0.5, 5.0, 12):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_geom(n, r))

        # Power‑law families
        for p in np.linspace(0.0, 2.2, 12):
            seed_seqs.append(_power(n, p))

        # Combined geometric × power families
        for c in np.linspace(0.5, 4.0, 8):
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in np.linspace(0.0, 2.0, 8):
                seed_seqs.append(_geom_power(n, r, p))

        # Symmetric geometric
        for c in np.linspace(0.5, 4.5, 8):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_sym_geom(n, r))

        # Smooth sine window (good baseline)
        seed_seqs.append(_sine_window(n))

        # Sine‑power family
        for p in [0.5, 1.0, 2.0, 3.0]:
            seed_seqs.append(_sin_power(n, p))

        # Beta‑distribution families (including arcsine = Beta(0.5,0.5))
        for a_param in [0.5, 0.8, 1.0, 1.5, 2.0]:
            for b_param in [0.5, 0.8, 1.0, 1.5, 2.0]:
                seed_seqs.append(_beta_distribution(n, a_param, b_param))

        # Dirichlet samples (different concentration)
        for alpha in [0.2, 0.5, 0.8, 2.0, 5.0]:
            a = np.random.gamma(alpha, 1.0, size=n)
            s = a.sum()
            if s > 0.0:
                seed_seqs.append(a / s)

        # Random soft‑max seeds
        for _ in range(5):
            v_rand = np.random.randn(n)
            seed_seqs.append(_softmax(v_rand))

        # Re‑use a previously discovered construction if present
        prev = globals().get("height_sequence_1")
        if prev is not None:
            try:
                a_prev = np.asarray(prev, dtype=np.float64)
                if a_prev.ndim == 1:
                    if a_prev.shape[0] != n:
                        # truncate / zero‑pad to the required length
                        a_adj = np.zeros(n, dtype=np.float64)
                        m = min(a_prev.shape[0], n)
                        a_adj[:m] = a_prev[:m]
                        a_prev = a_adj
                    if a_prev.sum() > 0.0:
                        a_prev = a_prev / a_prev.sum()
                        seed_seqs.append(a_prev)
            except Exception:
                pass

        # --------------------------------------------------------------
        # Fast scan: keep the best seed according to the fast objective.
        # --------------------------------------------------------------
        for idx, a in enumerate(seed_seqs):
            cur_val = 2.0 * n * np.max(_conv_fft(a, a))
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[scan] new global best from seed {idx} = {cur_val:.6f}")

        # --------------------------------------------------------------
        # Refinement with Adam starting from the current global best.
        # --------------------------------------------------------------
        if global_best_seq is None:
            # fallback – uniform distribution
            global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

        refined_seq, refined_val = _refine_sequence(
            global_best_seq,
            n,
            length_deadline,
            start_beta=0.5,
            beta_mul=1.04,
            lr0=0.5,
        )
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[overall] new global best after refinement (len={n}) = {refined_val:.6f}")

        # --------------------------------------------------------------
        # Perturbed restarts – cheap, many of them while time permits.
        # --------------------------------------------------------------
        while time.time() < length_deadline:
            noisy_logits = np.log(global_best_seq + 1e-30) + 0.02 * np.random.randn(n)
            pert_seq = _softmax(noisy_logits)
            seq2, val2 = _refine_sequence(
                pert_seq,
                n,
                length_deadline,
                start_beta=0.5,
                beta_mul=1.04,
                lr0=0.5,
            )
            if val2 < global_best_val:
                global_best_val = val2
                global_best_seq = seq2.copy()
                print(f"[perturb] improved global best to {val2:.6f}")

        # --------------------------------------------------------------
        # Optional light smoothing post‑process (may further lower the max).
        # --------------------------------------------------------------
        kernel = np.array([0.2, 0.6, 0.2], dtype=np.float64)
        while time.time() < deadline:
            smoothed = _smooth_seq(global_best_seq, kernel)
            cur_val = 2.0 * n * np.max(_conv_fft(smoothed, smoothed))
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = smoothed
                print(f"[smooth] improved global best to {cur_val:.6f}")
            else:
                break   # no further gain – exit smoothing loop

        # --------------------------------------------------------------
        # End of this length – proceed to the next if time permits.
        # --------------------------------------------------------------

    # ------------------------------------------------------------------
    # Final sanity checks and return
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("[fallback] using uniform sequence")

    # Normalise once more (evaluation is scale‑invariant but we keep sum=1)
    s = np.sum(global_best_seq)
    if s > 0.0:
        global_best_seq = global_best_seq / s

    # Convert to plain Python list
    result = global_best_seq.tolist()
    print(f"[final] best objective (fast) = {global_best_val:.6f}")
    return result
