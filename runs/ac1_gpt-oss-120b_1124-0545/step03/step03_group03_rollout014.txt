"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------

1. **Analytic grid search** – evaluate the classic “product’’ and “sum’’
   families on a fine grid and keep the best vector.

2. **Random mixture search** – sample a mixture of two product families
   (five free parameters) many times.  Each sample is normalised to the
   simplex and the objective is evaluated with a fast FFT convolution.
   The best mixture is remembered.

3. **Exact sub‑gradient descent (Exponentiated Gradient)** – starting from
   the best vector found so far we repeatedly:
   * compute the self‑convolution,
   * locate the index (or indices) where it attains its maximum,
   * build the sub‑gradient `g_i = 2 * sum_{k max} a_{k-i}` (zero where
     the index is out of range),
   * perform an exponentiated‑gradient update
        a_i ← a_i·exp(−η·g_i)   and renormalise to the simplex,
   * accept the move if it improves the score, otherwise accept with a
     temperature‑controlled Metropolis probability.

   The learning rate η and the temperature are annealed linearly.
   Periodically a small random jitter is added when progress stalls.

All helper functions are top‑level (no closures) and the routine respects
the 1000 s budget.  The final best sequence is returned as a plain Python
list.
"""

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself (zero‑padded) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a.

    The vector is expected to be on the simplex (sum = 1).  The function
    normalises it for safety, evaluates its self‑convolution and returns
    2·n·max(conv).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    return float(2.0 * a.shape[0] * np.max(conv))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)

def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """Single product family a_i ∝ ((i+off)*(n‑1‑i+off))^{‑p}."""
    i = np.arange(n, dtype=np.float64)
    off = max(offset, 1e-6)                     # avoid zero division
    a = ((i + off) * (n - 1 - i + off)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """Single sum family a_i ∝ (i+off)^{‑p} + (n‑1‑i+off)^{‑p}."""
    i = np.arange(n, dtype=np.float64)
    off = max(offset, 0.0)
    a = (i + off) ** (-p) + (n - 1 - i + off) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_mixture_product(
    n: int,
    p1: float,
    off1: float,
    p2: float,
    off2: float,
    w: float,
) -> np.ndarray:
    """
    Mixture of two product families:
        a = w·product(p1,off1) + (1‑w)·product(p2,off2)
    """
    i = np.arange(n, dtype=np.float64)
    off1 = max(off1, 1e-6)
    off2 = max(off2, 1e-6)
    a1 = ((i + off1) * (n - 1 - i + off1)) ** (-p1)
    a2 = ((i + off2) * (n - 1 - i + off2)) ** (-p2)
    a1 = np.where(np.isfinite(a1), a1, 0.0)
    a2 = np.where(np.isfinite(a2), a2, 0.0)
    a = w * a1 + (1.0 - w) * a2
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a

def gradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of the objective max(conv(a,a)).
    For every index k attaining the current maximum,
    the contribution to the gradient is 2·a shifted by k.
    """
    n = a.shape[0]
    max_val = np.max(conv)
    ks = np.where(conv == max_val)[0]   # all maximising indices
    g = np.zeros_like(a)
    for k in ks:
        idx = k - np.arange(n)               # k-i
        mask = (idx >= 0) & (idx < n)
        g[mask] += 2.0 * a[idx[mask]]
    return g

    print(f"[mix] finished after {mix_iter} samples, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact sub‑gradient descent with exponentiated updates
    # ------------------------------------------------------------------
    a = best_seq.copy()                     # current iterate (on simplex)
    iteration = 0
    last_improve = time.time()
    restart_cooldown = 30.0                 # seconds without improvement → jitter
    # schedules (linear)
    eta_start, eta_end = 2.0, 1e-4         # learning‑rate for EG
    temp_start, temp_end = 1.0, 1e-3      # Metropolis temperature

    while time.time() < deadline:
        # linear schedules based on elapsed fraction
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 20.0)   # leave a margin at the end
        eta = eta_start + progress * (eta_end - eta_start)
        temp = temp_start + progress * (temp_end - temp_start)

        # compute exact convolution and sub‑gradient
        conv = conv_fft(a)
        g = gradient_max(a, conv)

        # exponentiated‑gradient update
        # avoid NaNs: ensure g is finite
        g = np.where(np.isfinite(g), g, 0.0)

        a_new = a * np.exp(-eta * g)
        # numerical safety – avoid all‑zero due to underflow
        if a_new.sum() == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new = a_new / a_new.sum()

        cand_score = compute_score(a_new)

        # Metropolis acceptance
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            if temp > 1e-12:
                prob = math.exp(-(cand_score - best_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_new
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[{iteration:06d}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # occasional random jitter to escape plateaus
        if (iteration % 20000 == 0) and (time.time() - last_improve > restart_cooldown):
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            print(f"[{iteration:06d}] jitter restart (stagnation)")

        # occasional pure random mass‑transfer move (helps exploration)
        if np.random.rand() < 0.12:
            a = random_move(a, scale=0.5 * eta)

        # early exit if target already reached
        if best_score <= 1.503:
            print("[done] target reached, exiting early")
            break

        iteration += 1

    total_time = time.time() - start_time
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
