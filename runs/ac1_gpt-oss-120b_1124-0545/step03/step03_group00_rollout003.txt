def _refine_subgrad(a0, n, deadline, lr0=0.5):
    v = np.log(a0 + 1e-30)  # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    best_seq = a0.copy()
    conv = _conv_fft(a0, a0); best_val = 2*n*np.max(conv)

    it=0
    while time.time() < deadline:
        it+=1
        a = _softmax(v)      # probabilities

        # compute conv via fft
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[subgrad] n={n} it={it} best={cur_val:.6f}")

        # compute subgradient
        k = np.argmax(conv)
        # compute gradient w.r.t a:
        grad_a = np.zeros_like(a)
        if k < n:
            grad_a[:k+1] = 2 * a[k::-1]
        else:
            offset = k - n + 1
            grad_a[offset:] = 2 * a[n-1:offset-1:-1]
        # gradient w.r.t logits v:
        # as before: grad_v = a * (grad_a - np.dot(grad_a, a))
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)
        m_hat = m / (1.0 - 0.9**it)
        v2_hat = v2 / (1.0 - 0.999**it)
        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val

The algorithm differs from the previous one in two ways:

1. **Richer analytic seed pool** – in addition to the geometric,
   power‑law and sine families it adds a symmetric “U‑shaped’’ family
   a_i ∝ ((i+1)*(n-i))^{-p} (and its mixtures with the uniform
   distribution).  A quick sweep over the shape parameter p and the
   mixing coefficient provides high‑quality initial points.

2. **Two‑phase optimisation** – first a smoothed‑max Adam optimisation
   (as before) and then a *sub‑gradient* Adam optimiser that works on the
   true max‑convolution (no smoothing).  The sub‑gradient is obtained
   analytically from the index where the convolution attains its maximum
   and the update is performed in the log‑softmax (simplex) space.

Both phases respect the global time budget and the best sequence seen
throughout the run is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using FFT (real inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = np.sum(e)
    if s == 0:
        # fallback uniform
        return np.full_like(v, 1.0 / v.size)
    return e / s


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r^i, normalised."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ i^{-p} (i starts at 1), normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    """Combined geometric × power‑law a_i ∝ r^{i-1}/i^{p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric geometric a_i ∝ r^i + r^{n-1-i}."""
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sine_window(n: int) -> np.ndarray:
    """Discrete sine window (peaked in the centre)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _sym_product_power(n: int, p: float) -> np.ndarray:
    """U‑shaped family  a_i ∝ ((i+1)*(n-i))^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i * (n - i + 1)) ** (-p)
    s = a.sum()
    return a / s if s != 0 else np.full(n, 1.0 / n)


def _mix_with_uniform(arr: np.ndarray, lam: float) -> np.ndarray:
    """Mixture λ·Uniform + (1‑λ)·arr, renormalised."""
    uni = np.full_like(arr, 1.0 / arr.size)
    mixed = (1.0 - lam) * arr + lam * uni
    s = mixed.sum()
    return mixed / s if s != 0 else uni


# ----------------------------------------------------------------------
# Optimisation cores
# ----------------------------------------------------------------------


def _refine_sequence(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 5.0,
    beta_mul: float = 1.02,
    lr0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Smoothed‑max Adam optimiser (identical to the reference implementation).
    Returns the best sequence found and its fast objective value.
    """
    v = np.log(a0 + 1e-30)          # logits
    m = np.zeros_like(v)           # first moment
    v2 = np.zeros_like(v)          # second moment
    best_seq = a0.copy()
    conv = _conv_fft(a0, a0)
    best_val = 2.0 * n * np.max(conv)

    beta = start_beta
    it = 0

    while True:
        it += 1
        if time.time() > deadline:
            break

        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[refine] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # staged smoothing
        beta *= beta_mul
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of smoothed loss w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]    # central n entries

        # gradient w.r.t. logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


def _refine_subgrad(
    a0: np.ndarray,
    n: int,
    deadline: float,
    lr0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the *exact* max‑convolution (sub‑gradient).
    Works directly with the index of the current maximum.
    """
    v = np.log(a0 + 1e-30)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    best_seq = a0.copy()
    conv = _conv_fft(a0, a0)
    best_val = 2.0 * n * np.max(conv)

    it = 0
    while time.time() < deadline:
        it += 1
        a = _softmax(v)

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[subgrad] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # index of the current maximum
        k = int(np.argmax(conv))

        # sub‑gradient of conv_k w.r.t. a
        grad_a = np.zeros_like(a)
        if k < n:
            grad_a[: k + 1] = 2.0 * a[k :: -1]      # a[k], a[k‑1], …, a[0]
        else:
            offset = k - n + 1
            # a[n‑1], a[n‑2], …, a[offset]
            grad_a[offset:] = 2.0 * a[n - 1 : offset - 1 : -1]

        # gradient w.r.t. logits (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a height sequence that minimises ``evaluate_sequence``.
    The function respects the supplied time budget (default 1000 s) and
    returns the best sequence found as a plain list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin – the caller may need a few seconds after we return
    deadline = start_time + budget_s - 4.0

    # ------------------------------------------------------------------
    # Choose target lengths (larger first – they tend to give lower bounds)
    # ------------------------------------------------------------------
    possible_lengths = [12000, 8000, 4000]
    global_best_seq = None
    global_best_val = float("inf")

    for n in possible_lengths:
        if time.time() >= deadline:
            break
        print(f"[main] Optimising length = {n}")

        # ------------------------------------------------------------------
        # 1. Build a diverse pool of seed sequences (all already on the simplex)
        # ------------------------------------------------------------------
        seeds = []

        # Geometric families (r = 1 - c/n)
        for c in np.linspace(0.5, 5.0, 12):
            r = 1.0 - c / n
            if r > 0.0:
                seeds.append(_geom(n, r))

        # Power‑law families
        for p in np.linspace(0.0, 2.2, 12):
            seeds.append(_power(n, p))

        # Combined geometric × power families
        for c in np.linspace(0.5, 4.0, 8):
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in np.linspace(0.0, 2.0, 8):
                seeds.append(_geom_power(n, r, p))

        # Symmetric geometric
        for c in np.linspace(0.5, 4.5, 8):
            r = 1.0 - c / n
            if r > 0.0:
                seeds.append(_sym_geom(n, r))

        # Symmetric power‑law (U‑shaped)
        for p in np.linspace(0.1, 2.5, 12):
            seeds.append(_sym_product_power(n, p))

        # Mixtures of the U‑shaped family with the uniform distribution
        for p in np.linspace(0.1, 2.5, 6):
            base = _sym_product_power(n, p)
            for lam in [0.0, 0.25, 0.5, 0.75]:
                seeds.append(_mix_with_uniform(base, lam))

        # Smooth sine window (good baseline)
        seeds.append(_sine_window(n))

        # Dirichlet draws (different concentration parameters)
        for alpha in [0.2, 0.5, 0.8, 2.0, 5.0]:
            a = np.random.gamma(alpha, 1.0, size=n)
            s = a.sum()
            if s > 0:
                seeds.append(a / s)

        # Random soft‑max seeds
        for _ in range(5):
            v_rand = np.random.randn(n)
            seeds.append(_softmax(v_rand))

        # Re‑use the previous best sequence if it exists
        prev = globals().get("height_sequence_1")
        if prev is not None:
            try:
                a_prev = np.asarray(prev, dtype=np.float64).ravel()
                if a_prev.size != n:
                    # adjust size by truncating or zero‑padding
                    a_adj = np.zeros(n, dtype=np.float64)
                    m = min(a_prev.size, n)
                    a_adj[:m] = a_prev[:m]
                    a_prev = a_adj
                if a_prev.sum() > 0:
                    seeds.append(a_prev / a_prev.sum())
            except Exception:
                pass

        # ------------------------------------------------------------------
        # 2. Fast scan: evaluate the exact objective for each seed
        # ------------------------------------------------------------------
        for idx, a in enumerate(seeds):
            conv = _conv_fft(a, a)
            cur_val = 2.0 * n * np.max(conv)
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[scan] new global best from seed {idx} = {cur_val:.6f}")

        # ------------------------------------------------------------------
        # 3. Two‑phase refinement (smooth → sub‑gradient)
        # ------------------------------------------------------------------
        if global_best_seq is None:
            # fallback uniform distribution
            global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
            print("[fallback] using uniform start")

        # split remaining time for this length
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # first half for smoothed Adam, second half for sub‑gradient Adam
        half_deadline = time.time() + remaining * 0.5

        # smoothed phase
        seq_smooth, val_smooth = _refine_sequence(
            global_best_seq,
            n,
            half_deadline,
            start_beta=5.0,
            beta_mul=1.02,
            lr0=0.8,
        )
        if val_smooth < global_best_val:
            global_best_val = val_smooth
            global_best_seq = seq_smooth.copy()
            print(f"[overall] after smooth phase new best (len={n}) = {val_smooth:.6f}")

        # sub‑gradient phase (uses the rest of the time)
        seq_sub, val_sub = _refine_subgrad(
            global_best_seq,
            n,
            deadline,
            lr0=0.5,
        )
        if val_sub < global_best_val:
            global_best_val = val_sub
            global_best_seq = seq_sub.copy()
            print(f"[overall] after sub‑grad phase new best (len={n}) = {val_sub:.6f}")

        # ------------------------------------------------------------------
        # 4. Perturbed restarts (lightweight, use remaining time if any)
        # ------------------------------------------------------------------
        while time.time() < deadline:
            # small random jitter on the logits of the current best
            noisy_logits = np.log(global_best_seq + 1e-30) + 0.02 * np.random.randn(n)
            perturbed = _softmax(noisy_logits)

            # short sub‑gradient refinement (e.g. 8 s) – adapt to remaining budget
            short_dead = time.time() + min(8.0, deadline - time.time())
            seq_tmp, val_tmp = _refine_subgrad(perturbed, n, short_dead, lr0=0.5)

            if val_tmp < global_best_val:
                global_best_val = val_tmp
                global_best_seq = seq_tmp.copy()
                print(f"[perturb] improved global best to {val_tmp:.6f}")

        # End of this length – continue with the next smaller n if time permits
        # ------------------------------------------------------------------

    # ----------------------------------------------------------------------
    # Final sanity check & return
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("[fallback] final uniform sequence")

    # Normalise to sum‑to‑one (evaluation is scale‑invariant)
    s = np.sum(global_best_seq)
    if s > 0:
        global_best_seq = global_best_seq / s

    result = global_best_seq.tolist()
    print(f"[final] best objective (fast) = {global_best_val:.6f}")
    return result
