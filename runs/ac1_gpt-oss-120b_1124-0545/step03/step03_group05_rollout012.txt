def _next_pow2(x):
    return 1 << (x-1).bit_length()

def _conv_max(a, size):
    fu = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(fu * fu, n=size)[:2*len(a)-1]
    max_b = conv_full.max()
    return max_b, conv_full

def _project_simplex(v):
    # Duchi et al.
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _hard_max_gradient(a, max_idxs):
    n = a.shape[0]
    a_rev = a[::-1]
    g = np.zeros_like(a)
    for k in max_idxs:
        shift = k - (n - 1)
        if shift >= 0:
            g[shift:] += a_rev[:n - shift]
        else:
            s = -shift
            g[:n - s] += a_rev[s:]
    g /= len(max_idxs)
    d = -2.0 * g
    return d

def _random_transfer(a, scale):
    n = a.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer <= 0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # re-normalize
    a_new /= a_new.sum()
    return a_new

"""
Propose a low‑value step‑function sequence.

Algorithm
---------
* Initialise a feasible probability vector (sum = 1).  Several seeds are
  tried (global `height_sequence_1`, uniform, power‑law decays) and the
  best of them is kept.
* Repeatedly minimise the quantity  max (a∗a)  on the simplex.
  The exact sub‑gradient of the (non‑smooth) max is used:
      - Find the indices `K` where the current convolution attains its
        maximum.
      - For each `k∈K` the gradient contribution is a shifted copy of the
        reversed vector `a_rev`.  The average of those contributions gives
        a descent direction `d = -2·g`.
* Take a projected gradient step with a diminishing step‑size
  `α = α0 / sqrt(iter+1)`.  The projection is the Euclidean projection
  onto the probability simplex (Duchi‑Shalev‑Shwartz algorithm).
* With a small probability a random “mass‑transfer’’ move is applied to
  escape plateaux.
* If no improvement is observed for a few seconds the state is
  re‑initialised from a fresh random vector (the best solution found so
  far is never discarded).
* The best sequence seen during the allotted time (default 1000 s) is
  returned.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(a: np.ndarray, size_fft: int) -> tuple[float, np.ndarray]:
    """
    Linear convolution a∗a via FFT, returning the maximal entry and the
    full convolution vector.
    """
    fu = rfft(a, n=size_fft)
    full = irfft(fu * fu, n=size_fft)[: 2 * a.size - 1]
    max_val = full.max()
    return max_val, full


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σx_i = 1}.
    Implementation of Duchi et al. (2008), O(n log n).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)

    w = np.maximum(v - theta, 0.0)
    return w


def _hard_max_gradient(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max(a∗a)  at the current point.
    The gradient is the average of the shifted copies of a_rev corresponding
    to the active offsets `max_idxs`.
    Returns the *descent* direction d = -2·g.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    g = np.zeros_like(a)

    for k in max_idxs:
        shift = k - (n - 1)          # signed shift of a_rev
        if shift >= 0:
            g[shift:] += a_rev[: n - shift]
        else:
            s = -shift
            g[: n - s] += a_rev[s:]

    g /= len(max_idxs)               # average over all active constraints
    d = -2.0 * g                     # descent direction
    return d


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Move a random fraction of the mass from one index to another."""
    n = a.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n)
    while i == j:
        j = np.random.randint(n)

    if a[i] <= 0.0:
        return a
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence that minimises ``evaluate_sequence``.
    The function runs for at most ``budget_s`` seconds and returns the best
    sequence found (list of Python floats).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs
        ``n`` – length of the sequence (default 6000).

    Returns
    -------
    list[float]
        The best coefficient sequence (non‑negative, sum = 1).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # build a few initial candidates and keep the best one
    # --------------------------------------------------------------
    candidates = []

    # 1. global hint if available
    seq0 = globals().get("height_sequence_1", None)
    if seq0 is not None:
        seq0 = np.asarray(seq0, dtype=float)
        if seq0.size != n:
            # linear interpolation onto required length
            x_old = np.linspace(0, 1, seq0.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            seq0 = np.interp(x_new, x_old, seq0)
        seq0 = np.clip(seq0, 0.0, None)
        if seq0.sum() == 0.0:
            seq0 = np.random.rand(n)
        seq0 /= seq0.sum()
        candidates.append(seq0)

    # 2. uniform distribution
    candidates.append(np.ones(n) / n)

    # 3. power‑law decays (and their reverses)
    for p in (0.5, 0.8, 1.0, 1.2, 1.5):
        w = np.arange(1, n + 1, dtype=float) ** (-p)
        w /= w.sum()
        candidates.append(w)
        candidates.append(w[::-1])

    # 4. a few random vectors
    for _ in range(3):
        r = np.random.rand(n)
        r /= r.sum()
        candidates.append(r)

    # --------------------------------------------------------------
    # evaluation routine (convolution via FFT)
    # --------------------------------------------------------------
    size_fft = _next_pow2(2 * n - 1)

    def eval_obj(v):
        mb, _ = _conv_max(v, size_fft)
        return 2.0 * n * mb, mb

    best_seq = None
    best_score = np.inf
    for cand in candidates:
        score, _ = eval_obj(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    a = best_seq.copy()
    a_rev = a[::-1]

    # bookkeeping
    max_b, conv_full = _conv_max(a, size_fft)
    best_score = 2.0 * n * max_b
    best_seq = a.copy()
    print(f"[init] n={n}, initial best={best_score:.6f}")

    # --------------------------------------------------------------
    # hyper‑parameters
    # --------------------------------------------------------------
    base_step = 0.5                     # initial step size factor
    rand_prob = 0.07                    # probability of a random transfer
    rand_scale = 0.03                   # magnitude of random transfer
    stagnation_seconds = 20.0           # restart if no progress for this long
    last_improve = time.time()
    it = 0

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # current convolution and objective
        max_b, conv_full = _conv_max(a, size_fft)
        cur_score = 2.0 * n * max_b

        # keep the global best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # -----------------------------------------------------------------
        # Stochastic move vs. gradient step
        # -----------------------------------------------------------------
        if np.random.rand() < rand_prob:
            # cheap random perturbation
            a = _random_transfer(a, rand_scale)
            a_rev = a[::-1]
        else:
            # exact sub‑gradient based on the active set of maximal offsets
            eps = 1e-14 * max_b
            max_idxs = np.where(conv_full >= max_b - eps)[0]
            d = _hard_max_gradient(a, max_idxs)

            # diminishing step size
            step = base_step / np.sqrt(it + 1.0)

            # projected gradient update
            a_candidate = a + step * d
            a_candidate = _project_onto_simplex(a_candidate)
            a = a_candidate
            a_rev = a[::-1]

        # -----------------------------------------------------------------
        # occasional restart if we are stuck
        # -----------------------------------------------------------------
        if time.time() - last_improve > stagnation_seconds:
            a = np.random.rand(n)
            a /= a.sum()
            a_rev = a[::-1]
            print(f"[{it:7d}] restart due to stagnation")
            # reset timer to avoid immediate successive restarts
            last_improve = time.time()

        it += 1

    # --------------------------------------------------------------
    # final reporting
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    # Convert to plain Python floats for the harness
    return best_seq.tolist()
