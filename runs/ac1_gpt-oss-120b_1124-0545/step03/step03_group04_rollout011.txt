```
def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # Choose n (maybe 5000)
    n = 5000

    # generate candidate initial sequences
    init_seqs = []

    # use global height_sequence_1 if available
    try:
        init = np.array(height_sequence_1, dtype=float)
        init = np.maximum(init, 0.0)
        init = init / init.sum() if init.sum()>0 else np.full_like(init, 1.0/init.size)
        if init.size != n:
            # pad or truncate to n
            if init.size < n:
                pad = np.zeros(n - init.size)
                init_seq = np.concatenate([init, pad])
            else:
                init_seq = init[:n]
        else:
            init_seq = init
        init_seqs.append(init_seq)
    except Exception:
        pass

    # power law exponents
    for alpha in np.linspace(0.3, 0.7, 9):
        seq = np.arange(1, n+1) ** (-alpha)
        seq /= seq.sum()
        init_seqs.append(seq)

    # geometric decays
    for c in [0.5, 1.0, 1.5, 2.0]:
        q = 1 - c/n
        seq = ((c/n) * (q ** np.arange(n)))
        seq /= seq.sum()
        init_seqs.append(seq)

    # triangle
    seq = np.linspace(1, 0, n)
    seq = np.maximum(seq, 0)
    seq /= seq.sum()
    init_seqs.append(seq)

    # sine half-wave
    seq = np.sin(np.pi * (np.arange(1, n+1) / (n+1)))
    seq = np.maximum(seq, 0)
    seq /= seq.sum()
    init_seqs.append(seq)

    # random
    for _ in range(5):
        seq = np.random.rand(n)
        seq /= seq.sum()
        init_seqs.append(seq)

    # evaluate all initial seqs, pick best
    best_seq = None
    best_score = float('inf')
    for seq in init_seqs:
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[init] best initial score {best_score}")

    # local search
    a = best_seq.copy()
    cur_score = best_score

    # set annealing parameters
    T0 = 1.0
    T_min = 1e-4
    total_iters = 0
    # define schedule: T = T0 * (T_min/T0) ** (elapsed / total_time)
    while time.time() < deadline:
        # propose move
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 0:
            continue
        # choose fraction α_move
        frac = np.random.rand() * 0.5  # up to 50% of mass at i
        delta = a[i] * frac
        # apply move
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = evaluate_sequence(a_new.tolist())
        total_iters += 1

            a = a_new
            cur_score = new_score

                best_seq = a_new.copy()

            if np.exp(-(new_score - cur_score) / max(T, 1e-12)) > np.random.rand():
                a = a_new
                cur_score = new_score

    print(f"[final] best_score={best_score:.6f}, iterations={total_iters}")
```

```
    max_idx = conv.argmax()
    # compute influences:
    # For each i, get j = max_idx - i
    influence = a.copy()

    i_range = np.arange(low, high+1)
```

assistantfinal**Solution Overview**

The objective

```
score(a) = 2·n·max(a∗a) / (sum(a)²)
```

is invariant under scaling, therefore we can restrict the search to the
probability simplex  `{a_i≥0 , ∑a_i = 1}`.
The function `max(a∗a)` is convex but not smooth.  
A practical way to minimise it is a *greedy‐annealing* scheme that repeatedly

* identifies the convolution index where the current maximum is attained,
* measures how much every coefficient contributes to that maximum,
* moves a small amount of mass from the most influential coefficients
  toward the least influential ones,
* accepts the move if it improves the score or, with a temperature‑controlled probability,
  if it is slightly worse (simulated‑annealing).

All moves keep the ℓ₁‑norm unchanged, so the simplex constraint is satisfied
automatically (only a non‑negativity check is required).  

The algorithm starts from a small pool of well‑known constructions
(power‑law, geometric, triangular, sinusoidal and the existing global
solution if available).  The best of these seeds is taken as the initial
vector and then refined by the greedy‑annealing loop for the whole
time‑budget (≈ 1000 s).  The current best sequence is stored and returned
at the end.

The implementation uses only `numpy` and a tiny FFT based convolution
routine (`_conv_fft`).  No external solvers or file I/O are required.

```python
"""
Greedy‑annealing optimisation of a non‑negative step function.

The algorithm:
1. Build a diverse set of seed sequences (power‑law, exponential,
   triangular, sinusoidal, random and – if present – the global
   `height_sequence_1`).  All seeds are normalised to sum = 1.
2. Pick the seed with the smallest exact objective value.
3. Repeatedly:
   * compute the full self‑convolution a∗a,
   * locate the index k where the convolution attains its maximum,
   * compute for every coefficient i an *influence* = a_i + a_{k‑i}
     (zero if k‑i is out of bounds).  Large influence means that a_i
     contributes heavily to the current maximum.
   * choose a coefficient `i` among the top‑10 % most influential
     indices and a coefficient `j` among the bottom‑10 % least
     influential ones, and move a small fraction of the mass of `i`
     to `j`.
   * accept the move if it improves the objective; otherwise accept
     with probability `exp( –Δ / T )` where the temperature `T`
     gradually cools to zero.
   * keep the best sequence ever seen.
4. Return the best sequence (a plain Python list).

The whole routine respects a 1000 s time budget and does not use any
filesystem or network access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i ≥ 0 , Σ x_i = 1 }.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using FFT – fast for vectors of a few thousand items."""
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


def _power_sequence(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-α}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-alpha)
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a


def _geometric_sequence(n: int, c: float) -> np.ndarray:
    """
    a_i = (c/n)·q^i  with q = 1‑c/n, then normalised to sum = 1.
    The parameter `c` controls the decay speed.
    """
    q = 1.0 - c / n
    i = np.arange(n, dtype=float)
    a = (c / n) * (q ** i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _triangle_sequence(n: int) -> np.ndarray:
    """Linear decline from 1 to 0, normalised."""
    a = np.linspace(1.0, 0.0, n, dtype=float)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    a /= a_sum if a_sum > 0 else 1.0
    return a


def _sine_sequence(n: int) -> np.ndarray:
    """Half‑sine shape on [0,1], normalised."""
    x = np.arange(1, n + 1, dtype=float) / (n + 1)
    a = np.sin(np.pi * x)
    a_sum = a.sum()
    a /= a_sum if a_sum > 0 else 1.0
    return a


def _objective(a: np.ndarray) -> float:
    """Exact objective used by `evaluate_sequence`."""
    n = a.size
    conv = _conv_fft(a, a)
    max_conv = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_conv / (s * s)


def _influence_vector(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    For a given convolution index `k_max`, compute for each coefficient i
    the quantity   a_i + a_{k_max-i}   (the sub‑gradient of the max entry).
    Out‑of‑range partners are ignored.
    """
    n = a.shape[0]
    infl = a.copy()
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    i = np.arange(lo, hi + 1, dtype=int)
    j = k_max - i
    infl[i] += a[j]
    return infl

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The routine respects the given time budget
    (seconds) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # 1.  Choose a length and generate a variety of seed sequences
    # --------------------------------------------------------------
    try:
        # Use the global solution if it exists; adapt its length.
        init = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = init.size
        init = np.maximum(init, 0.0)
        if init.sum() > 0.0:
            init /= init.sum()
        else:
            raise ValueError
    except Exception:
        n = 5000
        init = None

    candidates = []

    if init is not None:
        candidates.append(init)

    # power‑law seeds
    for alpha in np.linspace(0.30, 0.70, 9):
        candidates.append(_power_sequence(n, alpha))

    # geometric / exponential seeds (different decay speeds)
    for c in (0.5, 1.0, 1.5, 2.0, 3.0):
        candidates.append(_geometric_sequence(n, c))

    # triangle and sine shapes
    candidates.append(_triangle_sequence(n))
    candidates.append(_sine_sequence(n))

    # a few random points on the simplex
    for _ in range(5):
        rand = np.random.rand(n)
        rand /= rand.sum()
        candidates.append(rand)

    # --------------------------------------------------------------
    # 2.  Pick the best seed according to the exact objective
    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for seq in candidates:
        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    print(f"[init] length={n}, best seed score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Greedy‑annealing local search
    # --------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score

    # temperature schedule (starts relatively hot, cools to almost zero)
    T_start = 1.0 * best_score        # scale with the initial score
    T_end = 1e-4

    # step‑size schedule (fraction of the source coefficient that is moved)
    step_frac = 0.20
    step_decay = 0.99995
    min_step = 0.001

    # counters for diagnostics
    it = 0
    no_improve = 0
    RESTART_AFTER = 8000            # perturb when stuck for a while

    while time.time() < deadline:
        it += 1

        # ---- compute convolution and the location of the current maximum
        conv = _conv_fft(a, a)
        k_max = int(conv.argmax())
        max_conv = conv[k_max]
        cur_score = 2.0 * n * max_conv / (a.sum() ** 2)

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[improve] iter={it}, score={best_score:.6f}")

        # ---- temperature (linear in log‑time)
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 5.0)
        T = T_start * (T_end / T_start) ** progress

        # ---- influence of each coefficient w.r.t. the active max entry
        infl = _influence_vector(a, k_max)

        # select candidate source (high influence) and target (low influence)
        top_k = max(1, n // 20)               # top 5 %  (≈ n/20)
        low_k = top_k

        # indices with largest / smallest influence
        src_idxs = np.argpartition(-infl, top_k - 1)[:top_k]
        dst_idxs = np.argpartition(infl, low_k - 1)[:low_k]

        i = int(np.random.choice(src_idxs))
        j = int(np.random.choice(dst_idxs))
        if i == j:
            continue

        # amount of mass to move (a fraction of source coefficient)
        delta = a[i] * step_frac
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # safeguard against tiny negative numbers caused by rounding
        if a_new[i] < 0.0:
            a_new[i] = 0.0

        # ---- evaluate new objective
        conv_new = _conv_fft(a_new, a_new)
        max_conv_new = conv_new.max()
        new_score = 2.0 * n * max_conv_new / (a_new.sum() ** 2)

        # ---- acceptance rule (simulated annealing)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability of accepting a worsening move
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if prob > np.random.rand():
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[improve] iter={it}, score={best_score:.6f}")
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---- decay the step fraction (more precise moves later)
        step_frac = max(step_frac * step_decay, min_step)

        # ---- occasional restart to escape a flat plateau
        if no_improve >= RESTART_AFTER:
            # small random jitter on a random subset of coordinates
            jitter_idxs = np.random.choice(n, size=min(50, n), replace=False)
            jitter = (np.random.randn(jitter_idxs.size) *
                      step_frac * a[jitter_idxs])
            a[jitter_idxs] = np.maximum(a[jitter_idxs] + jitter, 0.0)
            # re‑project onto simplex to keep the sum exactly 1
            a = _project_onto_simplex(a)
            no_improve = 0
            print(f"[restart] iter={it}, step_frac={step_frac:.6f}")

    # --------------------------------------------------------------
    # 4.  Return the best sequence discovered
    # --------------------------------------------------------------
    print(f"[final] best_score={best_score:.6f}, total_iters={it}")
    return best_seq.tolist()
```
