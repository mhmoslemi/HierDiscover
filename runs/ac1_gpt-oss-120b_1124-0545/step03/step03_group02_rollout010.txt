def solve_lp_max_sum(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    # Build constraints conv(f, g) <= M
    constraints = []
    # We can compute A matrix
    # using Toeplitz: for k from 0 to 2n-2:
    # conv_f_g_k = sum_{i=max(0, k-n+1)}^{min(n-1, k)} f[i] * g[k-i]
    # but treat k-i as index j, then g_j with coefficient = f[k-j] if 0 <= k-j < n
    # Equivalent to sum_j f[k-j] * g_j, where f index out-of-range => 0.
    for k in range(2 * n - 1):
        # Build coefficient vector for this constraint
        coeff = np.zeros(n)
        # j in range max(0, k-n+1) to min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        # For j in [j_start, j_end], coeff[j] = f[k - j]
        coeff[j_start:j_end+1] = f[k - np.arange(j_start, j_end+1)]
        # Append constraint
        constraints.append(coeff @ g <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        return None
    return np.array(g.value).flatten()
```

"""
Hybrid optimizer for the step‑function bound.

The algorithm proceeds in three phases, each respecting the time budget:

1. **Parametric sweep** – try a family of Beta‑shaped densities
   a_i ∝ x_i^{α‑1}(1−x_i)^{β‑1} (including the classic arcsine case α=β=0.5)
   and keep the best candidate.

2. **Linear‑programming refinement** – given a candidate f,
   solve the LP  max  Σ g_j   s.t.   g ≥ 0  and  f∗g ≤ M,
   where M = max( f∗f ).  The solution g is normalised to a probability
   vector and mixed with f (convex combination).  A few such iterations
   already improve the bound.

3. **Simulated‑annealing / hill‑climbing** – starting from the best vector
   obtained above, repeatedly move a small amount of mass from an index
   that contributes heavily to the current maximal convolution entry to a
   random other index.  The convolution is updated incrementally (O(n))
   and moves are accepted with a Metropolis rule that cools linearly.

The whole search runs for the allotted *budget_s* seconds (default 1000 s)
and returns the best sequence found.  The code requires only ``numpy`` and
optionally ``cvxpy`` (any installed solver is accepted).  All helper
functions are top‑level, no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def initialise_a(n: int, seed: int = 42) -> np.ndarray:
    """
    Initialise a probability vector of length n.
    Reuse a previously found global sequence if it matches the requested length.
    Otherwise fall back to the classic arcsine shape (α=β=0.5).
    """
    rng = np.random.default_rng(seed)

    if "height_sequence_1" in globals():
        a0 = np.array(globals()["height_sequence_1"], dtype=float)
        a0 = np.maximum(a0, 0.0)
        if a0.size == n:
            a0 = a0 / a0.sum()
            return a0

    # arcsine shape (α = β = 0.5)
    i = np.arange(n, dtype=float)
    a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a0 = a0 / a0.sum()
    return a0


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (Σv=1, v≥0)."""
    if v.sum() == 0:
        return np.full_like(v, 1.0 / len(v))
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_max(a: np.ndarray) -> float:
    """Exact max of the linear convolution a * a (full mode)."""
    return np.convolve(a, a).max()


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build matrix A (size (2n‑1)×n) such that (A @ g) = f ∗ g   (full convolution).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    # Fill row by row
    for k in range(rows):
        start = max(0, k - n + 1)
        end = min(k, n - 1)
        # j runs from start .. end (inclusive)
        # coefficient for column j is f[k-j]
        cols = np.arange(start, end + 1)
        A[k, cols] = f[k - cols]
    return A


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear program:
        maximise   Σ g_j
        subject to g ≥ 0  and  f ∗ g ≤ M   (element‑wise)
    Returns the optimiser g as a NumPy array, or None on failure.
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return None

    n = f.shape[0]
    A = build_conv_matrix(f)                     # shape (2n‑1, n)
    g = cp.Variable(n, nonneg=True)             # optimisation variable
    constraints = [A @ g <= M * np.ones(2 * n - 1)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # ECOS works well for dense LPs; fall back to SCS if unavailable.
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:
        prob.solve(solver=cp.SCS, warm_start=True, max_iters=5000)

    if g.value is None:
        return None
    return np.asarray(g.value).flatten()

# Main optimisation entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the provided evaluation function.  The routine respects the given time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # leave a safety margin

    # --------------------------------------------------------------
    # 0) Choose length (default 2000, can be overridden via kwargs)
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    # --------------------------------------------------------------
    # 1) Initial candidate (reuse previous if possible)
    # --------------------------------------------------------------
    a = initialise_a(n, seed=seed)

    # --------------------------------------------------------------
    # 2) Parametric sweep over beta‑shapes (α≈β) and random (α,β)
    # --------------------------------------------------------------
    best_a = a.copy()
    best_score = 2.0 * n * conv_max(best_a)
    print(f"[init] n={n}, score={best_score:.6f}")

    # allocate roughly 15 % of the budget to the sweep
    sweep_deadline = start_time + 0.15 * (budget_s - 1)

    # symmetric sweep (α=β=p)
    for p in np.linspace(0.30, 0.70, 9):
        if time.time() >= sweep_deadline:
            break
        i = np.arange(n, dtype=float)
        cand = 1.0 / ((i + 0.5) * (n - i - 0.5)) ** p
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = 2.0 * n * conv_max(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[sweep] symmetric p={p:.3f} → {sc:.6f}")

    # random (α,β) sweep
    for _ in range(12):
        if time.time() >= sweep_deadline:
            break
        alpha = np.random.uniform(0.15, 2.0)
        beta = np.random.uniform(0.15, 2.0)
        x = (np.arange(n, dtype=float) + 0.5) / n
        cand = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0:
            continue
        cand /= cand.sum()
        sc = 2.0 * n * conv_max(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[sweep] α={alpha:.3f}, β={beta:.3f} → {sc:.6f}")

    # --------------------------------------------------------------
    # 3) Linear‑programming refinement (a few iterations)
    # --------------------------------------------------------------
    lp_deadline = start_time + 0.35 * (budget_s - 1)   # 20 % extra after sweep
    a_lp = best_a.copy()
    while time.time() < lp_deadline:
        M = conv_max(a_lp)
        g_raw = solve_lp_max_sum(a_lp, M)
        if g_raw is None or g_raw.sum() == 0:
            break
        g = g_raw / g_raw.sum()               # normalise to a probability vector
        # convex mixture (tuned for stability)
        t = 0.20
        a_lp = (1.0 - t) * a_lp + t * g
        a_lp = np.maximum(a_lp, 0.0)
        a_lp = a_lp / a_lp.sum()
        sc = 2.0 * n * conv_max(a_lp)
        if sc < best_score:
            best_score, best_a = sc, a_lp.copy()
            print(f"[lp] improvement → {sc:.6f}")
        # stop early if we are not improving
        if time.time() > lp_deadline - 0.1:
            break

    # --------------------------------------------------------------
    # 4) Simulated‑annealing / hill‑climbing refinement
    # --------------------------------------------------------------
    # Prepare structures for fast incremental updates
    a = best_a.copy()
    conv = np.convolve(a, a)                     # full convolution, length 2n‑1
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score
    best_a = a.copy()
    rng = np.random.default_rng(seed + 999)

    # annealing schedule parameters
    T_start = cur_score * 0.08   # initial temperature (roughly 8 % of score)
    it = 0

    while time.time() < deadline:
        # linear cooling based on elapsed fraction of the SA phase
        elapsed = time.time() - start_time
        # fraction over the whole budget (so temperature reaches ≈0 at the end)
        p = min(1.0, elapsed / (budget_s - 0.5))
        T = T_start * (1.0 - p)

        # identify the current maximal convolution index
        k_max = int(np.argmax(conv))

        # contribution of each coordinate to that maximal entry
        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        contrib = np.zeros(n, dtype=float)
        contrib[mask] = a[idx[mask]] * a[k_max - idx[mask]]
        total_contrib = contrib.sum()

        # pick source index proportional to its contribution (or uniform if tiny)
        if total_contrib > 0 and np.random.rand() < 0.9:
            src = rng.choice(n, p=contrib / total_contrib)
        else:
            src = rng.integers(0, n)

        # pick destination uniformly (different from source)
        dst = rng.integers(0, n - 1)
        if dst >= src:
            dst += 1

        # amount to move (fraction of source mass, shrinking as we cool)
        max_delta = a[src]
        if max_delta <= 0.0:
            continue
        step_factor = 0.05 * (1.0 - p) + 0.003   # gradually smaller steps
        delta = max_delta * step_factor * rng.random()
        if delta <= 0.0:
            continue

        # ---------- incremental convolution update ----------
        conv_old = conv.copy()
        a_old_src = a[src]
        a_old_dst = a[dst]

        # linear part of the update (uses the *old* a vector)
        conv[src:src + n] -= 2.0 * delta * a
        conv[dst:dst + n] += 2.0 * delta * a

        # quadratic Δ*Δ contribution
        conv[2 * src] += delta * delta
        conv[2 * dst] += delta * delta
        conv[src + dst] -= 2.0 * delta * delta

        # apply the mass move
        a[src] = a_old_src - delta
        a[dst] = a_old_dst + delta

        # ensure numerical non‑negativity (tiny negatives may appear)
        a[src] = max(a[src], 0.0)

        # evaluate new score
        new_max = conv.max()
        new_score = 2.0 * n * new_max

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        elif T > 0.0:
            accept = np.random.rand() < np.exp(-(new_score - cur_score) / T)

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[SA {it:07d}] new best = {best_score:.6f}")
        else:
            # revert
            a[src] = a_old_src
            a[dst] = a_old_dst
            conv = conv_old

        it += 1

        # occasional re‑normalisation to mitigate drift
        if it % 5000 == 0:
            a = project_simplex(a)
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()

    total_time = time.time() - start_time
    print(f"[done] iterations={it}, best_score={best_score:.6f}, time={total_time:.1f}s")

    # final safety – non‑negative and finite values
    best_a = np.maximum(best_a, 0.0)

    # The evaluation routine caps values at 1000, but we stay well below that.
    return best_a.tolist()
