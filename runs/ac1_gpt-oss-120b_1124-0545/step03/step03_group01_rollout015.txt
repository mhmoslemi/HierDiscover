def _euclidean_proj_simplex(v):
    # algorithm from Duchi et al., "Efficient Projections onto the L1-Ball for Learning in High Dimensions"
    # Input v is a 1-d array, returns projection onto simplex {x >= 0, sum=1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function for the `evaluate_sequence` objective.

The search is performed in three stages:

1. **Diverse seeding** – A fairly large collection of analytic
   families (uniform, power‑laws, geometric, sine/gaussian/Kaiser windows,
   random soft‑max vectors) is generated for a fixed length `n`
   (default 4000).  For each seed the exact objective
   ``2·n·max(conv)`` (sum = 1) is computed and the best seed is kept.

2. **Local refinement** – The current best seed is refined using a
   projected sub‑gradient method.  The sub‑gradient of the *true* max‑
   convolution is used, a decreasing step size ``γ_t = γ₀/√t`` is applied,
   and the iterate is projected onto the probability simplex with an
   Euclidean projection (the O(n log n) algorithm of Duchi et al.).
   This step runs until the global deadline is reached.

3. **Optional continuation** – If time permits a few extra random
   perturbations of the best sequence are tried and re‑refined with the
   same sub‑gradient routine.

All helpers are defined at module level (no closures).  The routine
respects the given time budget (default 1000 s) and returns the best
list of non‑negative heights found.
"""

import time
import numpy as np
import scipy.special as sp
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, projection, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    c = np.fft.irfft(fx * fy, n=N)[:L]
    return c


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – maps ℝⁿ to the probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerically degenerate – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families for seed generation (already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, c: float) -> np.ndarray:
    """Geometric decreasing a_i ∝ r^i with r = 1 - c/n (c>0)."""
    r = 1.0 - c / n
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _geom_power(n: int, c: float, p: float) -> np.ndarray:
    """Mixture r^i / (i+1)^p."""
    r = 1.0 - c / n
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (i - 1)) / (i ** p)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle, σ = sigma_frac·n."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return a / a.sum()


def _kaiser(n: int, beta: float) -> np.ndarray:
    """Kaiser window with shape parameter beta."""
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    x = (i - alpha) / alpha  # in [-1, 1]
    a = sp.i0(beta * np.sqrt(1 - x * x)) / sp.i0(beta)
    return a / a.sum()


def _random_softmax(rng: np.random.Generator, n: int) -> np.ndarray:
    logits = rng.standard_normal(n)
    e = np.exp(logits - np.max(logits))
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Sub‑gradient refinement (euclidean projection)
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution.
    Returns the best sequence encountered and its exact objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Sub] n={n} it={it:5d} best={cur_val:.6f}")

        # exact sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        # decaying step‑size
        lr = 0.5 / np.sqrt(it)  # a tunable constant
        a = _proj_simplex(a - lr * g)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine uses a rich set of analytic seeds,
    evaluates them, and refines the current best with a long
    projected sub‑gradient descent.  It respects the global time budget
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Length to optimise – 4000 gives a good trade‑off between quality
    # and speed (the evaluator can handle larger n but the search gets
    # slower).  One may also try 3000 / 5000 if time permits.
    # ------------------------------------------------------------------
    n = 4000

    # try to reuse a previously good construction
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 1. Build a large and diverse set of seeds (all sum to 1)
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    seeds = []

    # baseline
    seeds.append(_uniform(n))

    # mixed geometric × power
    for c in np.linspace(0.5, 3.0, 5):

    for sigma_frac in [0.08, 0.12, 0.18, 0.25]:   # relative width

    for beta in [0.0, 2.0, 4.0, 6.0, 8.0]:

    # a few random starts
    for _ in range(12):

    # reuse previous if length matches
    if previous is not None and previous.shape == (n,):

    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = float("inf")

    for s in seeds:
        if time.time() > deadline:

        conv = _conv_fft(s, s)
        val = 2.0 * n * np.max(conv)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = s.copy()
            print(f"[seed] n={n} new best={val:.6f}")

        if val_ref < global_best_val:
            global_best_val = val_ref
            global_best_seq = seq_ref.copy()
            print(f"[refine] n={n} improved to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # 4. If time is still left, try a few random perturbations of the best
    #    and re‑refine them – this helps to escape very flat local minima.
    # ------------------------------------------------------------------

        cand = global_best_seq + noise
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
