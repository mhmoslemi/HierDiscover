import numpy as np
import time

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _conv_fft(u, v, size):
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[:len(u) + len(v) - 1]

def _full_convolution(a, size):
    return _conv_fft(a, a, size)

def _simplex_projection(x):
    # keep non‑negative and renormalise to sum = 1
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.shape[0])
    return x / s

def _gradient_active_shifts(a, active_shifts):
    """Sub‑gradient of max(conv(a,a)) for the given active shift indices."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if len(active_shifts) == 0:
        return g
    for l in active_shifts:
        d = l - (n - 1)           # offset of the shift
        if d >= 0:
            # shift to the right
            g[d:] += a[:n - d]
        else:
            # shift to the left
            g[:n + d] += a[-d:]
    # derivative of b_l is 2 * shifted a
    g *= 2.0 / len(active_shifts)
    return g

def _random_transfer(a, scale):
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = _simplex_projection(a_new)
    return a_new

    The algorithm keeps the vector on the probability simplex (non‑negative,
    sum = 1) and iteratively reduces the peak of the auto‑convolution.
    A simple random mass‑transfer move and periodic restarts are used
    to escape plateaux.  The best sequence seen before the time budget
    expires is returned.
    """

    start = time.time()
    deadline = start + budget_s - 0.5

    n = int(kwargs.get("n", 6000))

    # initialise from the supplied global if available, otherwise random
    seq0 = globals().get("height_sequence_1", None)

            x_new = np.linspace(0, 1, n, endpoint=False)
            a = np.interp(x_new, x_old, a)

    a = np.clip(a, 0.0, None)
    a = _simplex_projection(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999

        max_b = conv.max()

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution using a real‑valued FFT.
    ``size`` must be ≥ len(u)+len(v)‑1.
    """
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Convenient wrapper for a*a (full linear convolution)."""
    return _conv_fft(a, a, size)


def _simplex_projection(x: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.shape[0])
    return x / s


def _gradient_active_shifts(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of ``max(conv(a,a))`` for the given active shift indices.
    For a shift ``l`` the gradient is ``2 * a`` shifted by ``l-(n-1)``.
    The result is the average over all active shifts.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    if active_shifts.size == 0:
        return g
    for l in active_shifts:
        d = l - (n - 1)          # offset of the shift
        if d >= 0:
            g[d:] += a[: n - d]  # right shift
        else:
            g[: n + d] += a[-d:]  # left shift
    g *= 2.0 / active_shifts.size   # factor 2 from the derivative
    return g


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Move a random fraction of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return _simplex_projection(a_new)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Adam‑based sub‑gradient optimizer for ``evaluate_sequence``.
    The algorithm works on the probability simplex (non‑negative,
    Σa = 1) and iteratively reduces the peak of the auto‑convolution.
    Random mass‑transfer moves and occasional restarts help to escape
    plateaux.  The best sequence observed before the time budget expires
    is returned.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # initialise a feasible distribution
    # --------------------------------------------------------------
    seq0 = globals().get("height_sequence_1", None)
    if seq0 is not None:
        a = np.asarray(seq0, dtype=float)
        if a.size != n:
            # linear interpolation to required length
            old_x = np.linspace(0, 1, a.size, endpoint=False)
            new_x = np.linspace(0, 1, n, endpoint=False)
            a = np.interp(new_x, old_x, a)
    else:
        a = np.random.rand(n)

    a = _simplex_projection(a)

    # --------------------------------------------------------------
    # pre‑compute FFT length (constant)
    # --------------------------------------------------------------
    size_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Adam state & hyper‑parameters
    # --------------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.15                     # base learning‑rate
    max_step_norm = 0.05           # safety bound on L2‑step size

    prob_random = 0.05             # frequency of random mass‑transfer moves
    random_scale = 0.08            # how much mass is moved
    stagnation_seconds = 20.0      # restart if no improvement for this long

    last_improve = start

    # --------------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------------
    conv = _full_convolution(a, size_fft)
    max_b = conv.max()
    cur_score = 2.0 * n * max_b               # sum(a) == 1 → exact objective
    best_score = cur_score
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    it = 0

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ----- evaluate current peak -----
        conv = _full_convolution(a, size_fft)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        # ----- record improvement -----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ----- stochastic perturbation -----
        if np.random.rand() < prob_random:
            a = _random_transfer(a, random_scale)
            m.fill(0.0)
            v.fill(0.0)          # forget stale moments after a jump
            it += 1
            continue

        # ----- compute sub‑gradient of the maximal convolution entry -----
        tol = max_b * 1e-12                 # tiny relative tolerance
        active = np.where(conv >= max_b - tol)[0]
        g = _gradient_active_shifts(a, active)

        # ----- Adam update on the simplex -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        bias1 = 1.0 - beta1 ** it
        bias2 = 1.0 - beta2 ** it
        m_hat = m / bias1
        v_hat = v / bias2

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        # limit Euclidean norm of the step (helps stability)
        step_norm = np.linalg.norm(step)
        if step_norm > max_step_norm:
            step = step * (max_step_norm / step_norm)

        a = a - step
        a = _simplex_projection(a)

        # ----- stagnation handling -----
        if time.time() - last_improve > stagnation_seconds:
            a = np.random.rand(n)
            a = _simplex_projection(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[{it:7d}] restart due to stagnation")

    # --------------------------------------------------------------
    # final reporting
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
