def build_convolution_matrix(a):
    n = len(a)
    rows = []
    cols = []
    data = []
    for i, ai in enumerate(a):
        # rows: i to i+n-1 inclusive
        rows_i = np.arange(i, i+n)
        cols_i = np.arange(n)
        rows.append(rows_i)
        cols.append(cols_i)
        data.append(np.full(n, ai))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

"""
Hybrid search for a low‑value step‑function.

The objective (up to an irrelevant scale factor) is
    O(a) = 2·n·max(conv(a,a))          (with Σa = 1, a_i ≥ 0)

Key ideas
---------
* 1️⃣ Analytic families – “product’’ and “sum’’ power‑law shapes –
  are scanned on a fairly dense grid.
* 2️⃣ A massive random‑parameter sweep (including convex combinations of the
  two families) is performed until a quarter of the time budget is used.
* 3️⃣ Starting from the best vector found so far, an exact sub‑gradient
  descent on the max‑convolution is run.  The sub‑gradient of
  max_k (a∗a)_k is 2·a shifted by the index where the maximum occurs.
  After each step the vector is projected onto the probability simplex.
* 4️⃣ To avoid stagnation we add occasional random mass‑transfer moves,
  Metropolis acceptance with a cooling temperature, and short jitter
  restarts.

All helper routines are top‑level, there are no closures and the function
`propose_candidate` complies with the expected signature.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using a zero‑padded FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective for a non‑negative vector a.
    The vector is assumed to lie on the simplex (sum = 1).
    Returns 2·n·max(conv(a,a)).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of mass from one random index to another.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum = 1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+off)*(n‑1‑i+off))^{‑p}
    mode = 'sum'      → a_i ∝ (i+off)^{‑p} + (n‑1‑i+off)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        # offset must be > 0 to avoid a division by zero at the ends
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")

    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def mixed_candidate(
    n: int,
    p1: float,
    off1: float,
    p2: float,
    off2: float,
    w: float,
) -> np.ndarray:
    """
    Convex combination of product and sum families.
    w ∈ [0,1] mixes product (weight w) with sum (weight 1‑w).
    The result is normalised to sum = 1.
    """
    prod = analytic_candidate(n, p1, off1, "product")
    sm = analytic_candidate(n, p2, off2, "sum")
    a = w * prod + (1.0 - w) * sm
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.ones(n) / n
    else:
        a = a / a_sum
    return a

        off2 = np.random.uniform(0.0, 3.0)
        w = np.random.rand()
        cand = mixed_candidate(n, p1, max(off1, 1e-6), p2, max(off2, 1e-6), w)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[rand] new best = {best_score:.6f}")

    # incorporate possible external reference
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            tot = ref.sum()
            if tot > 1e-12:
                ref = ref / tot
                sc = compute_score(ref)
                print(f"[init] external reference score = {sc:.6f}")
                if sc < best_score:
                    best_score = sc
                    best_seq = ref.copy()

    print(f"[init] best after initialisation = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Sub‑gradient descent with Metropolis acceptance
    # ------------------------------------------------------------------
    a = best_seq.copy()
    iteration = 0
    last_improve = start_time

    # linear schedules (overall progress 0…1)
    step_start, step_end = 0.5, 0.001
    temp_start, temp_end = 1.0, 0.001

    while time.time() < deadline:
        elapsed = time.time() - start_time
        prog = elapsed / (budget_s - 0.5)   # normalised progress

        step = step_start + prog * (step_end - step_start)
        temp = temp_start + prog * (temp_end - temp_start)

        # -----------------------------------------------------------
        # Choose move: sub‑gradient (80 %) or random mass‑transfer (20 %)
        # -----------------------------------------------------------
        if np.random.rand() < 0.8:
            # exact sub‑gradient of the max‑convolution
            conv = conv_fft(a)
            k_max = int(np.argmax(conv))
            g = np.zeros_like(a)
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            idxs = np.arange(i_low, i_high + 1)
            g[idxs] = 2.0 * a[k_max - idxs]
            cand = a - step * g
            cand = np.clip(cand, 0.0, None)
            cand = project_simplex(cand)
        else:
            cand = random_move(a, step * 0.5)

        cand_score = compute_score(cand)

        # Metropolis acceptance
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            if temp > 1e-12:
                prob = math.exp(-(cand_score - best_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:07d}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # -----------------------------------------------------------
        # Jitter restart if stuck for a while
        # -----------------------------------------------------------
        if iteration % 200000 == 0 and (time.time() - last_improve > 30.0):
            jitter = np.random.rand(n) * 1e-4
            a = best_seq + jitter
            a = project_simplex(a)
            print(f"[{iteration:07d}] jitter restart (stagnation)")

        iteration += 1

    total_time = time.time() - start_time
    print(f"[final] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    # Return a plain Python list
    return best_seq.tolist()
