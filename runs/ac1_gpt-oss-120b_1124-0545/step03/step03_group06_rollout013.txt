"""
Fast “max‑flattening” search for a step‑function.

The idea
---------
* The objective of ``evaluate_sequence`` for a sequence with Σa = 1 is
  `2·n·max(conv(a,a))`.  Hence we want a probability vector whose
  self‑convolution is as flat as possible (its maximum close to the
  average `1/(2n‑1)`).

* We start from the (i+1)^{-½} power‑law which is already known to give
  a good flat convolution.  From there we run a large number of
  projected Adam‑style updates that minimise a smooth approximation of
  the max‑norm:
        L(a) = log‑sum‑exp(γ·conv(a,a)) .
  For `γ → ∞` this tends to the true max, for small `γ` it is a smooth
  surrogate that spreads the gradient over many entries and speeds up
  convergence.

* The gradient of `L` can be written using a correlation of `a` with the
  softmax weights `w_k ∝ exp(γ·conv_k)`.  Both the convolution and this
  correlation are computed with real FFTs in O(n log n).

* After each (Adam) step we simply clip the vector to non‑negative
  values and renormalise – an O(n) “projection” that is sufficient for
  the inner loop.  Before returning we perform one final exact Euclidean
  projection onto the probability simplex (`_project_simplex`) to obey
  the exact constraints of the harness.

* The whole routine respects the 1000 s time budget, chooses a sensible
  length (≈ 3·10⁴) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """Normalised non‑negative sequence proportional to (i+1+offset)^{-α}."""
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Linear self‑convolution using a real FFT (full length 2n‑1)."""
    n = seq.shape[0]
    size = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    conv[conv < 0.0] = 0.0            # remove tiny negatives caused by rounding
    return conv


def _conv_two(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors (full length len(a)+len(b)-1)."""
    size = _next_pow2(a.shape[0] + b.shape[0] - 1)
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: a.shape[0] + b.shape[0] - 1]
    conv[conv < 0.0] = 0.0
    return conv


def _score_fft(seq: np.ndarray) -> float:
    """Exact value of the evaluation function for a normalised sequence."""
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    O(n log n) implementation (Duchi et al. 2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _clip_and_renorm(v: np.ndarray) -> np.ndarray:
    """
    Fast “projection”: clip negatives to zero and renormalise to sum == 1.
    Used inside the inner optimisation loop (O(n)).
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        v[:] = 1.0 / v.size
    else:
        v /= s
    return v


def _softmax(x: np.ndarray, gamma: float) -> np.ndarray:
    """Numerically stable softmax on a 1‑D array, with temperature γ."""
    x = x - x.max()                     # shift for stability
    e = np.exp(gamma * x)
    s = e.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.size)
    return e / s


def _adam_gradient_step(
    seq: np.ndarray,
    m: np.ndarray,
    v: np.ndarray,
    it: int,
    step: float,
    gamma: float,
    beta1: float,
    beta2: float,
    eps: float,
) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    One Adam‑style update that minimises the smooth max objective.
    Returns the updated sequence and the new ADAM buffers.
    """
    n = seq.shape[0]

    # ----- 1. self‑convolution  -----
    conv = _conv_fft(seq)                     # length 2n‑1

    # ----- 2. softmax weights (smooth max) -----
    w = _softmax(conv, gamma)                 # sum(w) == 1

    # ----- 3. gradient = 2 * (a ⋆ w)  (cross‑correlation) -----
    # cross‑correlation a ⋆ w = a convolved with reversed w
    w_rev = w[::-1]
    corr_full = _conv_two(seq, w_rev)        # length 3n‑2
    start = w_rev.size - 1                    # = 2n‑2
    grad = 2.0 * corr_full[start : start + n]  # shape (n,)

    # ----- 4. Adam moment updates -----
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)

    m_hat = m / (1.0 - beta1 ** it)
    v_hat = v / (1.0 - beta2 ** it)

    # ----- 5. parameter update -------
    update = step * m_hat / (np.sqrt(v_hat) + eps)
    new_seq = seq - update

    # fast projection back to the simplex (clip + renorm)
    new_seq = _clip_and_renorm(new_seq)

    return new_seq, m, v

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The method combines a power‑law seed,
    a smooth‑max Adam optimiser, and a final exact simplex projection.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the final projection & return
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # 1. Choose a length that balances quality and speed.
    #    30 000 gives a theoretical lower bound of ≈ 1.00003.
    # --------------------------------------------------------------
    n = 30000

    # --------------------------------------------------------------
    # 2. Initialise from the classic α = 0.5 power‑law (≈ (i+1)^{-1/2}).
    # --------------------------------------------------------------
    seq = _power_sequence(n, alpha=0.5)
    best_seq = seq.copy()
    best_score = _score_fft(seq)

    print(f"[init] n={n}, start_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Adam‑style optimisation with a smooth approximation of max.
    #    Hyper‑parameters are deliberately generous – the loop runs
    #    until the time budget expires.
    # --------------------------------------------------------------
    m = np.zeros_like(seq)          # first moment
    v = np.zeros_like(seq)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step0 = 0.5
    step = step0
    decay = 0.99995
    gamma = 5.0                     # start with a smooth max
    gamma_target = 30.0
    gamma_inc = (gamma_target - gamma) / 200_000  # slow increase

    it = 0
    while time.time() < deadline:
        it += 1

        seq, m, v = _adam_gradient_step(
            seq=seq,
            m=m,
            v=v,
            it=it,
            step=step,
            gamma=gamma,
            beta1=beta1,
            beta2=beta2,
            eps=eps,
        )

        # occasional random mass exchange to escape tiny plateaus
        if it % 500 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps_swap = min(seq[i] * 0.2, seq[j] * 0.2)
            seq[i] -= eps_swap
            seq[j] += eps_swap
            seq = _clip_and_renorm(seq)

        cur_score = _score_fft(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={it:,}, score={best_score:.6f}")

        # schedule updates
        step = max(step * decay, 1e-8)
        gamma = min(gamma_target, gamma + gamma_inc)

    # --------------------------------------------------------------
    # 4. Final polishing – a few thousand cheap projected‑gradient steps
    #    on the exact max (top‑k = 10) to squeeze the last bits.
    # --------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    if remaining > 0.0:
        best_seq, best_score = _run_gradient(
            best_seq,
            time_budget=remaining,
            max_iters=10 ** 9,
            top_k=10,
            step0=0.2,
            decay=0.99998,
        )

    # --------------------------------------------------------------
    # 5. Exact projection onto the probability simplex (ensures validity).
    # --------------------------------------------------------------
    best_seq = _project_simplex(best_seq)

    print(f"[final] best_score={best_score:.6f}, length={n}")
    return best_seq.tolist()
