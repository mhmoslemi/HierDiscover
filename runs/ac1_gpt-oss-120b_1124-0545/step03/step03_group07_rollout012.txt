"""
Hybrid optimization for minimizing max(self‑convolution) of a non‑negative
step‑function.  The algorithm consists of:

1. Projected Adam on a smooth approximation of the ``max`` objective.
   The smooth max is the log‑sum‑exp of the convolution entries with a
   temperature β that is slowly increased.  After each Adam step the
   vector is projected onto the ℓ₁‑simplex {a≥0, Σa = √(2·n)}.

2. Guided hill‑climbing (simulated‑annealing‑style):  
   the shift k where the convolution attains its maximum is identified,
   the contribution a_i·a_{k‑i} of each position is computed, and a
   small amount of mass is transferred from a high‑contribution index
   to a low‑contribution index.  Moves that improve the exact maximum
   are always accepted; otherwise they are accepted with a temperature
   schedule.

Both phases respect the global time budget (≈ 1000 s) and keep the
best sequence encountered.  The returned list can be passed directly to
``evaluate_sequence``.
"""

import time
import math
import random
import numpy as np

def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project onto the simplex {w ≥ 0, Σ w = s}."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_power_of_two(min_len: int) -> int:
    """Smallest power of two ≥ min_len."""
    if min_len <= 1:
        return 1
    return 1 << ((min_len - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution using FFT of given length."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    # truncate to the exact linear convolution size
    return conv[: x.size + y.size - 1]

def _self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (full, length 2·n‑1)."""
    return _conv_fft(a, a, fft_len)

def _contributions(a: np.ndarray, shift: int) -> np.ndarray:
    """Contribution a_i * a_{shift‑i} for each i (0 outside the support)."""
    n = a.shape[0]
    c = np.zeros_like(a)
    i_start = max(0, shift - (n - 1))
    i_end = min(shift, n - 1)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        c[idx] = a[idx] * a[shift - idx]
    return c

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a sequence that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size ------------------------------------------------------
    # Use the previous construction if it exists, otherwise start from scratch.
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=float)
        n = base.size
        a0 = base.copy()
    else:
        # a moderately large size that keeps FFT cheap
        n = 3000
        a0 = np.random.exponential(scale=1.0, size=n)

    target_sum = math.sqrt(2.0 * n)

    # Normalise to the required ℓ₁‑norm
    a = a0.astype(float)
    a = a / a.sum() * target_sum

    # Pre‑compute a single FFT length that can accommodate
    # (2n‑1) + n‑1 = 3n‑2 elements (used for the gradient convolution).
    fft_len = _next_power_of_two(3 * n)

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max Adam descent ----------------------------------
    start_time = time.time()
    deadline = start_time + budget_s
    subgrad_deadline = start_time + budget_s * 0.60  # 60 % for phase 1

    # Adam hyper‑parameters
    lr = 0.5 * target_sum          # step‑size (≈ 2 % of the total mass)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = 0.5                     # start temperature for the soft‑max
    beta_max = 40.0
    beta_inc_factor = 1.001       # multiplicative increase per iteration

    # Evaluate the initial point
    conv = _self_convolution(a, fft_len)
    conv = np.maximum(conv, 0.0)   # numerical safety
    cur_max = float(conv.max())
    best_val = cur_max
    best_seq = a.copy()

    it = 0
    while time.time() < subgrad_deadline:
        # ----- exact convolution & current maximum --------------------
        conv = _self_convolution(a, fft_len)
        conv = np.maximum(conv, 0.0)
        cur_max = float(conv.max())

        # keep the best exact max seen so far
        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()
            print(f"[{it:6d}] Adam best max = {best_val:.6f}")

        # ----- smooth‑max (log‑sum‑exp) --------------------------------
        # shift by the max to avoid overflow
        shift = conv.max()
        exp_arg = beta * (conv - shift)
        # np.exp on float64 for stability
        w = np.exp(exp_arg, dtype=np.float64)
        w_sum = w.sum()
        w /= w_sum                     # softmax weights, sum to 1

        # ----- gradient of smooth‑max (2 * w * a) -----------------------
        grad_full = _conv_fft(w, a, fft_len)   # length 3n‑2
        grad = 2.0 * grad_full[:n]

        # ----- Adam update ---------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # project back onto the simplex (keeps sum = target_sum)
        a = _simplex_projection(a, target_sum)

        # ----- increase soft‑max temperature ---------------------------
        beta = min(beta_max, beta * beta_inc_factor)

        # occasional tiny random perturbation to escape plateaus
        if it % 3000 == 0:
            src = np.random.randint(n)
            if a[src] > 1e-12:
                dst = np.random.randint(n)
                delta = a[src] * 0.03 * np.random.rand()
                a_pert = a.copy()
                a_pert[src] -= delta
                a_pert[dst] += delta
                a_pert = _simplex_projection(a_pert, target_sum)
                conv_pert = _self_convolution(a_pert, fft_len)
                val_pert = float(conv_pert.max())
                if val_pert < cur_max:
                    a = a_pert
                    cur_max = val_pert
                if val_pert < best_val:
                    best_val = val_pert
                    best_seq = a_pert.copy()
                    print(f"[{it:6d}] perturbation improved best to {best_val:.6f}")

        # simple stagnation restart (if no improvement for a long time)
        # we reuse `it` and the last improvement time
        # (implemented by tracking best value improvement; omitted for brevity)

    # ------------------------------------------------------------------
    # Phase 2 – guided hill‑climbing (temperature‑based) ---------------
    cur_val = best_val
    a = best_seq.copy()

    # Temperature schedule for the second phase
    T0 = best_val * 0.25
    Tmin = best_val * 1e-5
    sa_start = time.time()
    last_improve_it = 0
    sa_it = 0

    while time.time() < deadline:
        # recompute exact convolution for the current a
        conv = _self_convolution(a, fft_len)
        conv = np.maximum(conv, 0.0)
        cur_val = float(conv.max())
        k_max = int(conv.argmax())

        # contributions to the maximal entry
        contrib = _contributions(a, k_max)
        src_idx = int(np.argmax(contrib))
        dst_idx = int(np.argmin(contrib))
        if src_idx == dst_idx:
            # fallback: pick a random destination different from src
            dst_idx = (src_idx + np.random.randint(1, n)) % n

        if a[src_idx] <= 1e-12:
            # nothing to move – pick a random source instead
            src_idx = np.random.choice(np.where(a > 1e-12)[0])

        # propose a mass transfer
        delta = a[src_idx] * (0.05 + 0.15 * np.random.rand())  # 5‑%–20 % of source
        a_cand = a.copy()
        a_cand[src_idx] -= delta
        a_cand[dst_idx] += delta

        # evaluate candidate
        conv_cand = _self_convolution(a_cand, fft_len)
        cand_val = float(conv_cand.max())

        # acceptance rule
        if cand_val < cur_val:
            a = a_cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = a_cand.copy()
                last_improve_it = sa_it
                print(f"[SA {sa_it:6d}] new best max = {best_val:.6f}")
        else:
            # temperature that decays exponentially
            elapsed = time.time() - sa_start
            frac = elapsed / (deadline - sa_start + 1e-12)
            T = T0 * ((Tmin / T0) ** frac)
            prob = math.exp(-(cand_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = a_cand
                cur_val = cand_val

        sa_it += 1

        # restart if we have been stuck for many iterations
        if sa_it - last_improve_it > 20000:
            a = np.random.exponential(scale=1.0, size=n)
            a = a / a.sum() * target_sum
            # reset Adam moments for the new random point (optional)
            m.fill(0.0)
            v.fill(0.0)
            last_improve_it = sa_it
            print(f"[SA RESTART] iteration {sa_it}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    print(f"[DONE] best max(conv) = {best_val:.6f}")
    return best_seq.tolist()

"""
Hybrid optimizer for minimizing the evaluation of a non‑negative step function.

The algorithm consists of two phases that respect a 1000 s time budget:

1. **Projected Adam on a smooth approximation of ``max(conv(a,a))``**.
   The smooth max is the log‑sum‑exp of the convolution entries with a
   temperature β that is slowly increased (β₀≈0.5 → β_max≈40).  After every
   Adam step the vector is projected onto the simplex
   {a ≥ 0, ∑a = √(2·n)} so that the ℓ₁‑norm stays constant.

2. **Guided hill‑climbing / simulated‑annealing**.
   The shift k where the exact self‑convolution attains its maximum is
   identified, the contributions a_i·a_{k−i} are computed, and a small
   amount of mass is transferred from the largest‑contribution index to
   the smallest‑contribution index.  Improving moves are always accepted;
   otherwise they are accepted with a temperature that decays exponentially.
   The best sequence over the whole run is returned.

Both phases keep the current best sequence and its exact maximum
convolution value, which coincides with the value returned by
`evaluate_sequence` because the ℓ₁‑norm is normalised.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project ``v`` onto the simplex {w ≥ 0, ∑w = s}."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _next_power_of_two(min_len: int) -> int:
    """Smallest power of two ≥ ``min_len``."""
    if min_len <= 1:
        return 1
    return 1 << ((min_len - 1).bit_length())


# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution ``x*y`` using an FFT of length ``fft_len``."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: x.size + y.size - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution ``a*a`` (length 2·n‑1)."""
    return _conv_fft(a, a, fft_len)


# ----------------------------------------------------------------------
def _contributions(a: np.ndarray, shift: int) -> np.ndarray:
    """Return the array c_i = a_i * a_{shift‑i} (zero outside the support)."""
    n = a.shape[0]
    c = np.zeros_like(a)
    i_start = max(0, shift - (n - 1))
    i_end = min(shift, n - 1)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1)
        c[idx] = a[idx] * a[shift - idx]
    return c


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Choose problem size.
    if "height_sequence_1" in globals():
        # start from the previously discovered construction
        base = np.asarray(height_sequence_1, dtype=float)
        n = base.size
        a = base.copy()
    else:
        # a generous size that still keeps the FFT cheap
        n = 3000
        a = np.random.exponential(scale=1.0, size=n)

    target_sum = math.sqrt(2.0 * n)               # ℓ₁‑norm required by the objective
    a = a.astype(float)
    a = a / a.sum() * target_sum                  # project onto the simplex

    # FFT length that can accommodate (2n‑1)+(n‑1)=3n‑2 elements
    fft_len = _next_power_of_two(3 * n)

    # ------------------------------------------------------------------
    # Phase 1 – projected Adam on a smooth‐max approximation ------------
    start_time = time.time()
    deadline = start_time + budget_s
    # allocate ~80 % of the total budget to the gradient phase
    grad_deadline = start_time + budget_s * 0.80

    # Adam hyper‑parameters (scale‑aware)
    lr = target_sum / n            # ≈ average entry size
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = 0.5                     # start temperature for the soft‑max
    beta_max = 40.0
    beta_factor = 1.001           # multiplicative increase per iteration

    # initial evaluation
    conv = _self_convolution(a, fft_len)
    conv = np.maximum(conv, 0.0)
    best_val = float(conv.max())
    best_seq = a.copy()
    it = 0

    while time.time() < grad_deadline:
        # ----- exact convolution & current max -----------------------
        conv = _self_convolution(a, fft_len)
        conv = np.maximum(conv, 0.0)
        cur_max = float(conv.max())

        # keep the best exact value seen so far
        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()
            print(f"[{it:6d}] Adam best max = {best_val:.6f}")

        # ----- smooth‑max (log‑sum‑exp) -------------------------------
        shift = conv.max()
        exp_arg = beta * (conv - shift)                # stability shift
        w = np.exp(exp_arg)                            # soft‑max weights
        w /= w.sum()

        # ----- gradient of smooth‑max (2·w*a) -------------------------
        grad_full = _conv_fft(w, a, fft_len)           # length 3n‑2
        grad = 2.0 * grad_full[:n]

        # ----- Adam update -------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step                         # descent step
        a = _simplex_projection(a, target_sum)

        # ----- increase soft‑max temperature -------------------------
        beta = min(beta_max, beta * beta_factor)

        # ----- occasional tiny perturbation to escape plateaus -------
        if it % 3000 == 0:
            src = np.random.randint(n)
            if a[src] > 1e-12:
                dst = np.random.randint(n)
                delta = a[src] * 0.03 * np.random.rand()
                a_pert = a.copy()
                a_pert[src] -= delta
                a_pert[dst] += delta
                a_pert = _simplex_projection(a_pert, target_sum)
                conv_pert = _self_convolution(a_pert, fft_len)
                val_pert = float(conv_pert.max())
                if val_pert < cur_max:
                    a = a_pert
                    cur_max = val_pert
                if val_pert < best_val:
                    best_val = val_pert
                    best_seq = a_pert.copy()
                    print(f"[{it:6d}] perturbation improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – guided hill‑climbing / simulated annealing -------------
    cur_val = best_val
    a = best_seq.copy()

    # temperature schedule for the second phase
    T0 = best_val * 0.25
    Tmin = best_val * 1e-5
    sa_start = time.time()
    last_improve_it = 0
    sa_it = 0

    while time.time() < deadline:
        # exact convolution for the current point
        conv = _self_convolution(a, fft_len)
        conv = np.maximum(conv, 0.0)
        cur_val = float(conv.max())
        k_max = int(conv.argmax())

        # contributions to the worst shift
        contrib = _contributions(a, k_max)
        src_idx = int(np.argmax(contrib))
        dst_idx = int(np.argmin(contrib))
        if src_idx == dst_idx:
            dst_idx = (src_idx + np.random.randint(1, n)) % n

        if a[src_idx] <= 1e-12:
            # choose a random non‑zero source instead
            nonzero = np.where(a > 1e-12)[0]
            if nonzero.size == 0:
                break
            src_idx = int(np.random.choice(nonzero))

        # propose a transfer of a modest fraction of the source mass
        delta = a[src_idx] * (0.05 + 0.15 * np.random.rand())   # 5‑% … 20 %
        a_cand = a.copy()
        a_cand[src_idx] -= delta
        a_cand[dst_idx] += delta

        conv_cand = _self_convolution(a_cand, fft_len)
        cand_val = float(conv_cand.max())

        # ----- acceptance -------------------------------------------------
        if cand_val < cur_val:
            a = a_cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = a_cand.copy()
                last_improve_it = sa_it
                print(f"[SA {sa_it:6d}] new best max = {best_val:.6f}")
        else:
            # exponential cooling
            elapsed = time.time() - sa_start
            frac = elapsed / (deadline - sa_start + 1e-12)
            T = T0 * ((Tmin / T0) ** frac)
            prob = math.exp(-(cand_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = a_cand
                cur_val = cand_val

        sa_it += 1

        # if we have been stuck for many iterations, restart from a fresh point
        if sa_it - last_improve_it > 20000:
            a = np.random.exponential(scale=1.0, size=n)
            a = a / a.sum() * target_sum
            last_improve_it = sa_it
            print(f"[SA RESTART] iteration {sa_it}")

    # ------------------------------------------------------------------
    print(f"[DONE] best max(conv) = {best_val:.6f}")
    return best_seq.tolist()
