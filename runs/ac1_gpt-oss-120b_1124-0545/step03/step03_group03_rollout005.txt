"""
Propose a non‑negative step function that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The routine works in three stages:

1. **Exploratory Analytic Grid** – generate many candidate families
   (product / sum power‑law, exponential tails, cosine‑powers,
   Beta‑shaped) and keep the best normalised vector.

2. **Direct Sub‑gradient Descent on the true max‑convolution**
   – the gradient of  max_k (a*a)_k  is simply a shifted slice of a,
   so one FFT per iteration suffices.  The vector is projected onto the
   probability simplex after each step.  A linear schedule for the
   step‑size and a Metropolis temperature give a simulated‑annealing
   flavour; occasional random mass‑transfer moves and jitter restarts
   keep the search from stagnating.

3. **Fine‑grained Random Refinement** – in the last few seconds a
   small Gaussian perturbation is tried many times; any improvement is
   kept.

The whole search respects a 1000 s wall‑time budget and returns the
best sequence found (as a plain list of floats).  All numbers are
non‑negative and sum to one, which makes the evaluation function scale
invariant.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant) for a non‑negative vector a.
    The vector is assumed to lie on the simplex (sum = 1), but a small
    normalisation is performed for safety.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)   # denominator = 1²


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    delta = scale * a[i] * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    return project_simplex(a_new)


def analytic_product(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ ((i+off)*(n‑1‑i+off))^{-p}  (U‑shaped)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_sum(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i+off)^{-p} + (n‑1‑i+off)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def analytic_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * min(i, n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-lam * d)
    total = a.sum()
    a = a / total
    return a


def analytic_cosine(n: int, p: float) -> np.ndarray:
    """a_i ∝ |cos(pi * i / (n‑1))|^{p} (peaks at the ends)."""
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    total = a.sum()
    a = a / total
    return a


def analytic_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{α} * (n‑i)^{β}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    total = a.sum()
    a = a / total
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    the evaluation function.  The routine combines an analytic grid
    search, a fast sub‑gradient descent on the exact max‑convolution,
    occasional random moves and a final stochastic refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5               # tiny safety margin

    # --------------------------------------------------------------
    # 1) Analytic grid search (large and diverse families)
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    best_seq = None
    best_score = np.inf

    print("[init] Analytic grid search, n =", n)

    # product family
    p_vals = np.arange(0.2, 2.01, 0.2)
    off_vals = np.arange(0.01, 2.01, 0.2)   # avoid zero to keep finite
    for p in p_vals:
        for off in off_vals:
            cand = analytic_product(n, p, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
    # sum family
    for p in p_vals:
        for off in off_vals:
            cand = analytic_sum(n, p, off)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
    # exponential tail family
    lam_vals = np.logspace(-3, 1, num=15)   # 0.001 … 10
    for lam in lam_vals:
        cand = analytic_exponential(n, lam)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    # cosine‑power family
    p_vals_cos = np.arange(0.5, 6.1, 0.5)
    for p in p_vals_cos:
        cand = analytic_cosine(n, p)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    # beta‑shape family (α,β) grid
    alpha_vals = np.arange(-0.5, 2.01, 0.5)   # negative exponents give U‑shape
    beta_vals = np.arange(-0.5, 2.01, 0.5)
    for a in alpha_vals:
        for b in beta_vals:
            cand = analytic_beta(n, a, b)
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()

    # if a reference sequence is supplied globally, consider it
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = compute_score(ref)
            print(f"[init] global reference score = {sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = ref.copy()

    print(f"[init] best analytic start score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Sub‑gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    a = best_seq.copy()                     # current iterate (sum = 1)
    last_improve = start_time
    iteration = 0

    # schedule parameters
    step_start, step_end = 0.5, 1e-6
    temp_start, temp_end = 0.5, 0.0
    stagnation_time = 20.0                  # seconds without improvement → jitter

    while time.time() < deadline:
        # current convolution and score
        conv = conv_fft(a)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]
        cur_score = 2.0 * n * max_val

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{iteration:06d}] new best = {best_score:.6f}")

        # sub‑gradient of the active max term
        i = np.arange(n)
        rev_idx = max_idx - i
        mask = (rev_idx >= 0) & (rev_idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev_idx[mask]]

        # linear schedules (progress in [0,1])
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 10.0)
        step = step_start * (1.0 - progress) + step_end * progress
        temp = temp_start * (1.0 - progress) + temp_end * progress

        # gradient move
        cand = a - step * g
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)

        # evaluate candidate (single extra FFT)
        conv_cand = conv_fft(cand)
        cand_score = 2.0 * n * conv_cand.max()

        # Metropolis‑type acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            if temp > 1e-12:
                prob = math.exp(-(cand_score - cur_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = cand

        # occasional random mass‑transfer move
        if np.random.rand() < 0.02:
            a = random_move(a, step * 0.5)

        # random Dirichlet restart (rare)
        if np.random.rand() < 0.005:
            rand_vec = np.random.gamma(1.0, 1.0, size=n)
            rand_vec = rand_vec / rand_vec.sum()
            conv_r = conv_fft(rand_vec)
            score_r = 2.0 * n * conv_r.max()
            if score_r < best_score:
                best_score = score_r
                best_seq = rand_vec.copy()
                a = rand_vec.copy()
                last_improve = time.time()
                print(f"[{iteration:06d}] random sample improved to {best_score:.6f}")

        # jitter restart when we have been stuck for a while
        if time.time() - last_improve > stagnation_time:
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            last_improve = time.time()
            print(f"[{iteration:06d}] jitter restart (stagnation)")

        iteration += 1

    # --------------------------------------------------------------
    # 3) Fine‑grained stochastic refinement (last ~5 s)
    # --------------------------------------------------------------
    refine_deadline = start_time + budget_s - 2.0
    while time.time() < refine_deadline:
        # small Gaussian perturbation around the best so far
        perturb = np.random.normal(scale=5e-4, size=n)
        cand = best_seq + perturb
        cand = np.clip(cand, 0.0, None)
        cand = project_simplex(cand)
        conv_cand = conv_fft(cand)
        cand_score = 2.0 * n * conv_cand.max()
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    total_time = time.time() - start_time
    print(f"[done] best score = {best_score:.6f}, iterations = {iteration}, "
          f"time = {total_time:.1f}s, length = {len(best_seq)}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
