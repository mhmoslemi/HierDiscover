"""
Propose a non‑negative step‑function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Fine analytic grid search** over two parametric families
   (product / sum of power‑law terms) to obtain a strong symmetric start.
   The best grid candidate is kept as the initial distribution.

2. **Exponentiated‑gradient descent with a smoothed‑max surrogate**.
   * The smooth objective is Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k).
   * Its gradient is obtained by FFT‑based convolutions.
   * An EG update   a_i ← a_i·exp(‑η·∇Φ_λ(a)_i)   is performed,
     followed by renormalisation onto the probability simplex.
   * The smoothing parameter λ and the learning rate η are linearly
     annealed from a gentle to a hard regime.
   * After each update the vector is symmetrised (a ← (a+a[::-1])/2) –
     this never increases the max‑convolution and greatly reduces the
     search space.

3. **Stagnation handling** – if no improvement has been seen for a
   configurable time, a tiny random jitter is added and the process
   continues.

Both the exact objective (via `compute_score`) and the surrogate are
computed using FFTs, so each iteration is cheap (≈3 k iterations per
second on a modest laptop).  The routine respects a 1000 s budget and
returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear (zero‑padded) convolution of a with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective (scale‑invariant) for a non‑negative vector a.
    The vector is assumed to lie on the simplex (sum = 1); a tiny
    normalisation is performed for safety.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a = a / s
    conv = conv_fft(a)
    max_b = np.max(conv)
    return float(2.0 * a.shape[0] * max_b)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending order
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_smooth(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        Φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k)
    with respect to a (assuming Σ a_i = 1).  The gradient is
        g_i = 2· Σ_k w_k·a_{k-i},
    where w_k are the soft‑max weights.
    Both the convolution a*a and the weighted convolution
    are obtained via FFT.
    """
    n = a.shape[0]

    # 1. a*a
    conv = conv_fft(a)                     # length 2n‑1

    # 2. soft‑max weights w_k = exp(λ·conv_k) / Σ exp(λ·conv)
    max_c = np.max(conv)
    exp_term = np.exp(lam * (conv - max_c))
    w = exp_term / np.sum(exp_term)        # length 2n‑1, sum = 1

    # 3. weighted convolution w_rev * a  (the factor 2 appears later)
    w_rev = w[::-1]                        # reversed weights
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)

    # Extract the n entries that correspond to the gradient
    start = n - 1
    end = 2 * n - 1                       # exclusive
    slice_part = conv_wa[start:end]       # length n
    g = 2.0 * slice_part[::-1]            # reverse to obtain g_i
    return g


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum=1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    mode = 'sum'      → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")

    a = np.where(np.isfinite(a), a, 0.0)

    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function of length ``n`` (default 2000) that minimises
    the evaluation function.  The routine

    * performs a refined analytic grid search,
    * runs a projected exponentiated‑gradient descent on a smoothed‑max
      surrogate (with annealed smoothing λ and learning‑rate η),
    * enforces symmetry after each update and injects occasional jitter
      when stagnation is detected,
    * returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Parameters and initialisation
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 2000))

    best_seq = None
    best_score = np.inf

    # ---------------------------------------------------------------
    # 1) Refined analytic grid search (symmetric families)
    # ---------------------------------------------------------------
    print("[init] Starting refined analytic grid search (n={})".format(n))

    # Grid resolution – a bit finer than the previous run
    p_vals = np.arange(0.30, 1.01, 0.03)          # 0.30 … 1.00, step 0.03
    offset_vals = np.arange(0.0, 2.01, 0.10)      # 0 … 2.0, step 0.10
    modes = ["product", "sum"]

    for mode in modes:
        for p in p_vals:
            for off in offset_vals:
                # avoid zero offset for product mode (division by zero)
                off_eff = off if (mode == "sum" or off > 0.0) else 1e-3
                cand = analytic_candidate(n, p, off_eff, mode)
                sc = compute_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
    # ------------------------------------------------------------------
    # 2) Optional global reference (if provided by the harness)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=float)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref_sum = ref.sum()
            if ref_sum > 1e-12:
                ref = ref / ref_sum
                sc = compute_score(ref)
                print("[init] Global reference score = {:.6f}".format(sc))
                if sc < best_score:
                    best_score = sc
                    best_seq = ref.copy()

    print("[init] Best analytic start score = {:.6f}".format(best_score))

    # ---------------------------------------------------------------
    # 3) Exponentiated‑gradient descent on smooth max
    # ---------------------------------------------------------------
    a = best_seq.copy()                # keep a on the simplex
    iteration = 0
    last_improve = start_time

    # Schedules (linear in time)
    lam_start, lam_end = 5.0, 2000.0          # smoothing
    eta_start, eta_end = 0.5, 0.01           # learning rate for EG
    jitter_interval = 50000                 # iterations between forced jitters
    stagnation_cooldown = 30.0              # seconds without improvement

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = (elapsed) / (budget_s - 5.0)   # leave some margin

        # linear interpolation of hyper‑parameters
        lam = lam_start + progress * (lam_end - lam_start)
        eta = eta_start + progress * (eta_end - eta_start)

        # -----------------------------------------------------------
        # Gradient of smoothed max and EG update (projected & symmetrised)
        # -----------------------------------------------------------
        g = gradient_smooth(a, lam)                # non‑negative gradient
        # Exponentiated‑gradient step
        factor = np.exp(-eta * g)
        a = a * factor
        # Re‑normalise onto the simplex (still non‑negative)
        s = a.sum()
        if s == 0.0:
            a = np.ones_like(a) / n
        else:
            a = a / s

        # Enforce symmetry (never harms the max‑convolution)
        a = (a + a[::-1]) * 0.5
        a = a / a.sum()                            # re‑normalise after symmetrisation

        # -----------------------------------------------------------
        # Evaluate exact score periodically (every 10 iterations)
        # -----------------------------------------------------------
        if iteration % 10 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                last_improve = time.time()
                print("[{:07d}] new best = {:.6f} (t={:.1f}s)".format(
                    iteration, best_score, elapsed))

        # -----------------------------------------------------------
        # Stagnation handling – add tiny jitter if stuck
        # -----------------------------------------------------------
        if iteration % jitter_interval == 0:
            if time.time() - last_improve > stagnation_cooldown:
                jitter = np.random.rand(n) * 1e-4
                a = a + jitter
                a = project_simplex(a)
                a = (a + a[::-1]) * 0.5
                a = a / a.sum()
                print("[{:07d}] jitter restart (stagnation)".format(iteration))

        # Early exit if we have already reached the target bound
        if best_score <= 1.503:
            print("[done] target reached, exiting early")
            break

        iteration += 1

    total_time = time.time() - start_time
    print("[done] iterations = {}, best score = {:.6f}, time = {:.1f}s".format(
        iteration, best_score, total_time))

    # Return as plain Python list (the harness expects this)
    return best_seq.tolist()
