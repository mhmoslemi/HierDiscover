def refine_subgradient(a, n, time_limit):
    # a is probability vector sum=1.
    best = a.copy(); best_score = compute_score(a)
    start = time.time()
    eta = 0.5
    decay = 0.9995
    steps = 0
    while time.time() < time_limit:
        steps += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        k = conv.argmax()
        # compute subgradient
        g = compute_subgradient(a, k) * (2 * n)  # scaling by 2n (obj derivative)
        # update
        a = a - eta * g
        # projection onto simplex
        a = _project_onto_simplex(a)
        # compute new score
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # decay step size
        eta = max(eta * decay, 1e-6)
        # occasional random perturbation
        if steps % 200 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i]*0.3, 0.001) # small mass
            if delta > 0:
                a[i] -= delta
                a[j] += delta
                a = _project_onto_simplex(a)
    return best, best_score

def power_sequence(n, alpha):
    i = np.arange(1,n+1)
    a = i**(-alpha)
    a /= a.sum()
    return a

def exp_sequence(n, lam):
    i= np.arange(n)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a

def arcsine_sequence(n):
    i = np.arange(n, dtype=np.float64)
    a = 1.0/np.sqrt((i+0.5)*(n-0.5-i))
    a /= a.sum()
    return a

def linear_sequence(n):
    i = np.arange(1,n+1)
    a = i
    a /= a.sum()
    return a

def triangular_sequence(n):
    # increasing then decreasing symmetric
    half = (n+1)//2
    inc = np.arange(1, half+1)
    if n %2 ==0:
        dec = inc[::-1]
    else:
        dec = inc[:-1][::-1]
    a = np.concatenate([inc, dec])
    a = a[:n]
    a = a.astype(float)
    a /= a.sum()
    return a

"""
Propose a non‑negative step‑function (a list of floats) that minimises
``evaluate_sequence``.

The algorithm works in three stages:

1. **Broad exploration** – generate a collection of candidate shapes
   (arcsine, power‑law, exponential, linear, triangular) for several
   sequence lengths (up to 20 000).  Each candidate is normalised to a
   probability vector (sum = 1) and scored with the exact objective
   (computed via FFT, i.e. 2·n·max(conv)/(sum²)).  The best shape is kept.

2. **Sub‑gradient refinement** – starting from the best shape we minimise
   the true non‑smooth objective using a sub‑gradient of the max‑convolution.
   The sub‑gradient for a given maximising index *k* is
   ``∂max/∂a_j = a_{k‑j} + a_{j‑k}`` (out‑of‑range terms are ignored).
   After each step the vector is projected onto the probability simplex
   (Duchi et al. 2008).  A simple learning‑rate schedule and occasional
   random mass‑swaps help escape plateaus.

3. **Light random search** – while time permits we perform tiny random
   mass exchanges on the current best vector and keep any improvement.

All helper functions are defined at the top level; no closures or file I/O
are used.  The routine respects the supplied time budget (default 1000 s)
and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (ICML 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using the FFT (fast for n≥~1e3)."""
    size = x.size + y.size - 1
    N = 1 << ((size - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    Z = X * Y
    conv = np.fft.irfft(Z, n=N)[:size]
    return conv


# ----------------------------------------------------------------------
# Candidate shape generators
# ----------------------------------------------------------------------


def _power_sequence(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-α}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-alpha)
    a /= a.sum()
    return a


def _exp_sequence(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-λ·i), normalised to sum = 1."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-lam * idx)
    a /= a.sum()
    return a


def _linear_sequence(n: int) -> np.ndarray:
    """a_i ∝ i+1, normalised."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx
    a /= a.sum()
    return a


def _triangular_sequence(n: int) -> np.ndarray:
    """Symmetric triangle (increasing then decreasing)."""
    half = (n + 1) // 2
    inc = np.arange(1, half + 1, dtype=float)
    if n % 2 == 0:
        dec = inc[::-1]
    else:
        dec = inc[:-1][::-1]
    a = np.concatenate([inc, dec])
    a = a[:n]
    a /= a.sum()
    return a


def _arcsine_sequence(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine law:
    a_i ∝ 1/√((i+½)(n‑½‑i)), normalised to sum = 1."""
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Scoring helpers
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a probability vector."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    max_b = conv.max()
    # sum(a) == 1 after the projection step
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------


def _max_conv_subgradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) at index ``k``.
    For non‑negative a the sub‑gradient is
        g_j = a_{k‑j}  (if 0≤k‑j<n) + a_{j‑k} (if 0≤j‑k<n).
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    # part 1: a_{k‑j}
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        j = np.arange(lo, hi + 1, dtype=int)
        g[j] = a[k - j]

    # part 2: a_{j‑k}
    if k <= n - 1:
        start = k
        length = n - start
        g[start:] += a[:length]

    return g


# ----------------------------------------------------------------------
# Local refinement (sub‑gradient descent)
# ----------------------------------------------------------------------


def _refine_subgradient(
    a0: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Perform sub‑gradient descent on the exact objective.
    Returns the best vector found and its score.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _objective(a)
    n = a.shape[0]

    # learning‑rate schedule
    lr = 0.5            # initial step size
    lr_decay = 0.9998
    lr_min = 1e-6

    step = 0
    while time.time() < deadline:
        step += 1

        conv = _conv_fft(a, a)
        max_val = conv.max()
        k = int(conv.argmax())

        # sub‑gradient of the objective: 2·n·∂max/∂a
        g = 2.0 * n * _max_conv_subgradient(a, k)

        # gradient step + projection onto the simplex
        a = a - lr * g
        a = _project_onto_simplex(a)

        # keep the exact best
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            # occasional progress report
            print(f"[refine] it={step:,} score={best_score:.6f}")

        # decay learning rate
        lr = max(lr * lr_decay, lr_min)

        # tiny random swap every few hundred steps to escape flat regions
        if step % 250 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta = min(a[i] * 0.1, 0.001)
            if delta > 0:
                a[i] -= delta
                a[j] += delta
                a = _project_onto_simplex(a)

    return best, best_score
