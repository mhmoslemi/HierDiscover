"""
Hybrid optimizer for the step‑function bound.

* Stage 1 – cheap parametric sweep.
  A symmetric “inverse‑product” family
       a_i ∝ ((i+½+Δ)*(n‑i‑½+Δ))^{‑p}
  (p≈½, Δ≥0) covers the classical arcsine shape (p=½,Δ=0) and smooth
  variants that tame the singularities at the ends.
  Each candidate is normalised to Σa=1 and its exact score
       2·n·max(conv(a,a))
  is evaluated with an FFT‑based convolution.  The best candidate is
  kept for the next stage.

* Stage 2 – Adam optimisation on the exact max.
  The objective is scale‑invariant; we keep the iterates on the
  probability simplex by the soft‑max parametrisation a=softmax(v).
  The gradient of the exact max‑convolution is obtained analytically,
  which provides a much more accurate descent direction than a
  log‑sum‑exp surrogate.  An Adam schedule with a linearly decaying
  learning‑rate is used; occasional random restarts escape plateaus.

The whole routine runs for the available time budget (default 1000 s)
and returns the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    v_shift = v - np.max(v)
    e = np.exp(v_shift)
    return e / e.sum()


def conv_max(a: np.ndarray, fft_len: int) -> float:
    """Returns max(conv(a,a)) using an FFT of length `fft_len`."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.shape[0] - 1]
    return float(conv.max())


def conv_array(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (real part) of length 2·n‑1."""
    fa = np.fft.fft(a, fft_len)
    return np.real(np.fft.ifft(fa * fa))[: 2 * a.shape[0] - 1]


def grad_of_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """Gradient of conv(a,a)[k_max] w.r.t. a (exact, no smoothing)."""
    n = a.shape[0]
    g = np.zeros_like(a)

    # term a[k_max - i]
    idx1 = k_max - np.arange(n)
    mask1 = (idx1 >= 0) & (idx1 < n)
    g[mask1] += a[idx1[mask1]]

    # term a[i - k_max]
    idx2 = np.arange(n) - k_max
    mask2 = (idx2 >= 0) & (idx2 < n)
    g[mask2] += a[idx2[mask2]]

    # if the pair i = j occurs (k_max even), we added a[i] twice
    if k_max % 2 == 0:
        i_center = k_max // 2
        if 0 <= i_center < n:
            g[i_center] -= a[i_center]          # keep only a[i] once

    return g


def generate_shape(n: int, p: float, delta: float) -> np.ndarray:
    """
    Symmetric shape a_i ∝ ((i+0.5+Δ)*(n-i-0.5+Δ))^{-p}.
    The vector is normalised to sum = 1.
    """
    i = np.arange(n, dtype=float)
    denom = (i + 0.5 + delta) * (n - i - 0.5 + delta)
    # Guard against overflow/under‑flow for extreme parameters
    with np.errstate(divide='ignore', invalid='ignore'):
        a = np.power(denom, -p)
    # Replace any NaNs (possible if denom==0) by zero
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a


def exact_score(a: np.ndarray) -> float:
    """Exact objective value for a probability vector (Σa = 1)."""
    n = a.shape[0]
    # FFT length – next power of two ≥ 2·n‑1
    fft_len = 1 << ((2 * n - 2).bit_length())
    m = conv_max(a, fft_len)
    return 2.0 * n * m


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects a time budget (default 1000 s)
    and returns the best sequence found.
    """
    # ------------------------------------------------------------------
    # 0) initial setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # Length of the sequence – honour user request when supplied
    n = int(kwargs.get("n", 2000))
    fft_len = 1 << ((2 * n - 2).bit_length())      # for all FFT calls

    # ------------------------------------------------------------------
    # 1) cheap parametric sweep
    # ------------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    # Parameter grids (tuned to give a good coverage around the arcsine shape)
    p_grid = np.linspace(0.35, 0.70, 15)            # around 0.5
    delta_grid = np.linspace(0.0, 2.5, 16)          # offsets to smooth the edges

    # If a previously discovered sequence exists, use it as a fallback
    global_vars = globals()
    if "height_sequence_1" in global_vars:
        cand = np.array(global_vars["height_sequence_1"], dtype=float)
        if cand.size == n:
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            s = exact_score(cand)
            if s < best_score:
                best_score = s
                best_a = cand.copy()
                print(f"[grid] seed sequence score = {s:.6f}")

    # Exhaustive grid search (very cheap)
    for p in p_grid:
        for delta in delta_grid:
            if time.time() >= deadline:
                break
            a = generate_shape(n, p, delta)
            s = exact_score(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
                print(f"[grid] new best score = {s:.6f}  (p={p:.3f}, Δ={delta:.3f})")
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) Adam optimisation on the exact max (soft‑max parametrisation)
    # ------------------------------------------------------------------
    # Initialise logits from the best shape found so far
    if best_a is None:
        # fallback to classic arcsine shape
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        best_a = a0 / a0.sum()
    v = np.log(best_a + 1e-12)          # logits (soft‑max domain)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)

    # Learning‑rate / temperature schedules
    lr_start, lr_end = 0.04, 0.0008
    temp_start, temp_end = 5.0, 3000.0   # soft‑max temperature (for possible smoothing)

    # bookkeeping
    it = 0
    last_improve = start_time
    a_cur = best_a.copy()
    best_score = exact_score(a_cur)
    best_a = a_cur.copy()

    # Pre‑compute the index vector for fast gradient computation
    idx_vec = np.arange(n)

    while time.time() < deadline:
        elapsed = time.time() - start_time
        prog = min(1.0, elapsed / (budget_s - 10.0))

        # --------------------------------------------------------------
        # 2.1) update schedules
        # --------------------------------------------------------------
        lr = lr_start * (1 - prog) + lr_end * prog
        temperature = temp_start * (1 - prog) + temp_end * prog   # not used directly,
                                                                # kept for possible future extensions

        # --------------------------------------------------------------
        # 2.2) forward pass – obtain a on the simplex
        # --------------------------------------------------------------
        a = softmax(v)                     # Σa = 1, a_i ≥ 0

        # --------------------------------------------------------------
        # 2.3) convolution and exact max index
        # --------------------------------------------------------------
        conv_full = conv_array(a, fft_len)      # length 2·n‑1
        max_val = float(conv_full.max())
        k_max = int(conv_full.argmax())

        # --------------------------------------------------------------
        # 2.4) gradient of the exact max (sub‑gradient of the objective)
        # --------------------------------------------------------------
        g_conv = grad_of_max_conv(a, k_max)      # ∂max(conv)/∂a
        grad_obj = 2.0 * n * g_conv              # ∂E/∂a  (E = 2·n·max)

        # Chain‑rule back to logits v (soft‑max)
        dot_av = np.dot(a, grad_obj)
        grad_v = a * (grad_obj - dot_av)

        # --------------------------------------------------------------
        # 2.5) Adam update
        # --------------------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)

        v -= lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # --------------------------------------------------------------
        # 2.6) bookkeeping – keep the best exact score ever seen
        # --------------------------------------------------------------
        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[{it:07d}] new best exact score = {best_score:.6f} (t={elapsed:.1f}s)")

        # --------------------------------------------------------------
        # 2.7) occasional restart if stagnation detected
        # --------------------------------------------------------------
        if it % 3000 == 0:
            if time.time() - last_improve > 25.0:
                # restart from the current best with a small jitter
                jitter = np.random.randn(n) * 1e-3
                v = np.log(best_a + 1e-12) + jitter
                m.fill(0.0)
                v_hat.fill(0.0)
                print(f"[{it:07d}] restart after stagnation (t={elapsed:.1f}s)")

    total_time = time.time() - start_time
    print(f"[done] iterations={it}, best exact score={best_score:.6f}, "
          f"time={total_time:.1f}s")

    # ------------------------------------------------------------------
    # 3) post‑processing – ensure compatibility with evaluate_sequence
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)       # safety net
    # The evaluator caps at 1000, we stay far below that.
    return best_a.tolist()
