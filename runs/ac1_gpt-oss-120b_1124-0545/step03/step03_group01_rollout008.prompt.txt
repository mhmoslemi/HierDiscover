def evaluate_sequence(sequence: list[float]) -> float:
    """

    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    sequence = [min(1000.0, x) for x in sequence]

    max_b = max(b_sequence)

The algorithm works in two phases for each candidate length:

1. **Adam on a smooth max** – we parametrize the step heights by a

   obtained with the log‑sum‑exp trick.  The smoothing parameter β is
   increased quickly (β←β·1.05) so the optimisation soon focuses on the
   true max.  Adam updates keep the iterates on the simplex automatically.

2. **Projected sub‑gradient refinement** – starting from the best Adam
   result we perform a mirror‑descent style update that uses the exact
   sub‑gradient of the true max (the index where the convolution attains
   the maximum).  After each step we project back onto the simplex by
   normalising; the learning rate decays as 1/√t.

Both phases are time‑aware and stop as soon as the global deadline is
reached.  Several seed families (geometric, power‑law, combined,
symmetric, sine‑window and random) are tried for each length (4000 and

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using real FFT.
    Returns the full result of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)

def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n) / n


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n

# Adam optimiser on the smoothed max loss

def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """

    Returns the best sequence (as a numpy array) encountered and its
    objective value (2·n·max(conv) because sum(a)=1).

    beta2 = 0.999
