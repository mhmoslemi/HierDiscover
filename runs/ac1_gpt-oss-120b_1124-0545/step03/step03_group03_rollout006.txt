   to obtain a differentiable objective; λ is increased linearly during
   the run.  Gradient w.r.t. the logits is computed analytically using
   FFTs, and a standard Adam update with linear learning‑rate decay is
   applied.  Periodic jitter and Metropolis‑style acceptance are used
   to escape plateaus.
3) The best sequence found before the time budget expires is returned.

import time

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a with itself (zero‑padded) using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)
    return conv[: 2 * n - 1]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    e = np.exp(x - np.max(x))
    return e / e.sum()


def analytic_candidate(n: int, p: float, offset: float, mode: str) -> np.ndarray:
    """
    Produce a normalised (sum=1) analytic candidate.
    mode = 'product' → a_i ∝ ((i+offset)*(n‑1‑i+offset))^{‑p}
    mode = 'sum'      → a_i ∝ (i+offset)^{‑p} + (n‑1‑i+offset)^{‑p}
    """
    i = np.arange(n, dtype=np.float64)

    if mode == "product":
        a = ((i + offset) * (n - 1 - i + offset)) ** (-p)
    elif mode == "sum":
        a = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    else:
        raise ValueError("mode must be 'product' or 'sum'")

    a = np.where(np.isfinite(a), a, 0.0)

    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def beta_candidate(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric Beta‑distribution shape on the discrete grid.
    a_i ∝ (i/(n-1))^{α-1} (1-i/(n-1))^{β-1}.
    """
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    eps = 1e-12
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, beta - 1.0)
    a = np.where(np.isfinite(a), a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    return a


def gaussian_candidate(n: int, sigma: float) -> np.ndarray:
    """
    Symmetric discrete Gaussian centred at the middle of the array.
    """
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def exponential_candidate(n: int, decay: float) -> np.ndarray:
    """
    Symmetric exponential decay away from the centre.
    """
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-decay * np.abs(i - centre))
    a = a / a.sum()
    return a


def exact_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a normalised (sum=1) vector.
    """
    n = a.shape[0]
    conv = conv_fft(a)
    return float(2.0 * n * np.max(conv))

# Optimisation utilities
# ----------------------------------------------------------------------


def smooth_obj_and_grad(a: np.ndarray, lam: float) -> tuple[float, np.ndarray]:
    """
    Compute the smooth approximation of max(conv) together with its gradient
    w.r.t. a (the gradient of the smoothed maximum, not yet multiplied by 2·n).

    Returns
        smooth_max   –   (1/λ)·log Σ_k exp(λ·conv_k)   (a scalar)
        grad_a       –   ∂smooth_max/∂a   (shape n)
    """
    n = a.shape[0]
    conv = conv_fft(a)                # length 2n‑1
    max_c = np.max(conv)

    # log‑sum‑exp in a numerically stable way
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = np.sum(exp_term)
    smooth_max = max_c + np.log(sum_exp) / lam

    # soft‑max weights
    w = exp_term / sum_exp               # length 2n‑1, sum = 1

    # weighted convolution w_rev * a
    w_rev = w[::-1]
    size = _next_pow2(len(w_rev) + n - 1)
    fft_w = np.fft.rfft(w_rev, n=size)
    fft_a = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(fft_w * fft_a, n=size)

    # slice the n entries that correspond to the gradient
    start = n - 1
    slice_part = conv_wa[start:start + n]     # length n
    grad_a = 2.0 * slice_part[::-1]           # reverse to obtain ∂Φ/∂a_i
    return smooth_max, grad_a


def adam_optimize(x0: np.ndarray,
                  n: int,
                  start_time: float,
                  deadline: float,
                  init_best_a: np.ndarray,
                  init_best_score: float,
                  target_score: float = 1.503,
                  lam_start: float = 0.5,
                  lam_end: float = 2000.0,
                  lr_start: float = 0.05,
                  lr_end: float = 0.001,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  eval_interval: int = 400,
                  stagnation_time: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑it representation.
    Returns the best a (numpy array) and its exact score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = init_best_a.copy()
    best_score = init_best_score
    last_improve = time.time()
    iteration = 0

    while True:
        now = time.time()
        if now >= deadline:
            break

        # progress fraction (0‑1) over the remaining optimisation window
        total_opt_time = deadline - start_time
        elapsed_opt = now - start_time
        progress = elapsed_opt / total_opt_time if total_opt_time > 0 else 1.0
        progress = min(max(progress, 0.0), 1.0)

        # annealed parameters
        lam = lam_start + progress * (lam_end - lam_start)
        lr = lr_start + progress * (lr_end - lr_start)

        # current a on the simplex
        a = softmax(x)

        # smooth objective and its gradient w.r.t a
        smooth_max, grad_a = smooth_obj_and_grad(a, lam)

        # gradient of the *actual* objective (2·n·Φ) w.r.t a
        grad_obj_a = 2.0 * n * grad_a

        # convert gradient to log‑it space (chain rule for softmax)
        dot = np.dot(grad_obj_a, a)
        grad_x = a * (grad_obj_a - dot)

        # optional gradient clipping (helps stability)
        grad_norm = np.linalg.norm(grad_x)
        max_norm = 10.0
        if grad_norm > max_norm:
            grad_x = grad_x * (max_norm / grad_norm)

        # Adam update (bias‑corrected)
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # periodic evaluation of the exact (non‑smoothed) objective
        if iteration % eval_interval == 0:
            a = softmax(x)
            cur_score = exact_score(a)

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                last_improve = time.time()
                print(f"[adam] iter {iteration:07d} – new best {best_score:.6f} (t={now - start_time:.1f}s)")

                if best_score <= target_score:
                    print("[adam] target reached, terminating early")
                    break

        # stagnation handling – jitter restart
        if now - last_improve > stagnation_time:
            # add a tiny Gaussian jitter to the logits
            jitter = np.random.randn(n) * 0.1
            x = np.log(best_a + 1e-12) + jitter
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[adam] jitter restart at iter {iteration}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, n: int = 2000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function of length ``n`` that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

         keeping the best candidate;
      2. refines that candidate with an Adam optimiser on the simplex

    # leave a small margin for the harness
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # Families already provided in the reference implementation
    p_vals = np.arange(0.30, 0.95 + 1e-9, 0.05)          # 0.30 … 0.90
    offset_vals = np.arange(0.0, 1.6, 0.2)              # 0 … 1.4
    modes = ["product", "sum"]

    print("[init] start analytic grid search (n={})".format(n))

    for mode in modes:
        for p in p_vals:
            for off in offset_vals:
                off_eff = off if (mode == "sum" or off > 0.0) else 1e-3
                cand = analytic_candidate(n, p, off_eff, mode)
                sc = exact_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                # occasional progress output
                if (mode == "product") and (p == p_vals[0]) and (off == offset_vals[0]):
                    print(f"[grid] mode={mode}, p={p:.2f}, off={off:.2f} → {sc:.6f}")

    # Additional analytic families ------------------------------------------------
    # Beta‑distribution shaped candidates
    alpha_vals = np.arange(0.2, 1.01, 0.2)   # 0.2,0.4,…,1.0
    beta_vals = np.arange(0.2, 1.01, 0.2)
    for alpha in alpha_vals:
        for beta in beta_vals:
            cand = beta_candidate(n, alpha, beta)
            sc = exact_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()

    # Symmetric Gaussian candidates
    sigma_vals = np.linspace(0.2 * n, 0.6 * n, 7)  # roughly 0.2‑0.6·n

            best_seq = cand.copy()

    # Symmetric exponential candidates
    decay_vals = np.linspace(0.0005, 0.005, 7)  # small decays produce wide shapes
    for decay in decay_vals:

        sc = exact_score(cand)
        if sc < best_score:

            best_seq = cand.copy()

    uniform = np.ones(n, dtype=np.float64) / n
    sc = exact_score(uniform)

    # Global reference if provided
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = exact_score(ref)
            print(f"[init] global reference score = {sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = ref.copy()

    # 2) Adam‑based refinement on simplex (log‑it parametrisation)
    # ------------------------------------------------------------------
    # Initialise logits from the best analytic candidate (add tiny epsilon to avoid log(0))

        stagnation_time=30.0,
    )

    elapsed = time.time() - start_time
    print(f"[done] best score={best_score:.6f}, elapsed={elapsed:.1f}s")
    return best_seq.tolist()

            x = np.log(best_a + 1e-12) + jitter
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[adam] jitter restart at iter {iteration}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     n: int = 2000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function of length ``n`` that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

    The routine:
      1) performs a modest grid search over several analytic families,
         keeping the best candidate;
      2) refines that candidate with an Adam optimiser on the simplex
         (soft‑max parametrisation) using a smooth max surrogate;
      3) injects occasional jitter when progress stalls.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Analytic grid search
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    p_vals = np.arange(0.30, 0.95 + 1e-9, 0.05)   # 0.30 … 0.90
    offset_vals = np.arange(0.0, 1.6, 0.2)       # 0 … 1.4
    modes = ["product", "sum"]

    print("[init] starting analytic grid search (n={})".format(n))

    for mode in modes:
        for p in p_vals:
            for off in offset_vals:
                off_eff = off if (mode == "sum" or off > 0.0) else 1e-3
                cand = analytic_candidate(n, p, off_eff, mode)
                sc = exact_score(cand)
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                if (mode == "product") and (p == p_vals[0]) and (off == offset_vals[0]):
                    print(f"[grid] mode={mode}, p={p:.2f}, off={off:.2f} → {sc:.6f}")

    # Additional families ----------------------------------------------------
    # Beta‑distribution candidates
    alphas = np.arange(0.2, 1.01, 0.2)
    betas = np.arange(0.2, 1.01, 0.2)
    for a in alphas:
        for b in betas:
            cand = beta_candidate(n, a, b)
            sc = exact_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()

    # Symmetric Gaussian candidates
    sigma_vals = np.linspace(0.2 * n, 0.6 * n, 7)
    for sigma in sigma_vals:
        cand = gaussian_candidate(n, sigma)
        sc = exact_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # Symmetric exponential candidates
    decay_vals = np.linspace(0.0005, 0.005, 7)
    for decay in decay_vals:
        cand = exponential_candidate(n, decay)
        sc = exact_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()

    # Uniform baseline
    uniform = np.ones(n, dtype=np.float64) / n
    sc = exact_score(uniform)
    if sc < best_score:
        best_score = sc
        best_seq = uniform.copy()

    # Global reference if it exists
    if "height_sequence_1" in globals():
        ref = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if ref.shape[0] == n:
            ref = np.maximum(ref, 0.0)
            ref = ref / ref.sum()
            sc = exact_score(ref)
            print(f"[init] global reference score = {sc:.6f}")
            if sc < best_score:
                best_score = sc
                best_seq = ref.copy()

    print(f"[init] best analytic start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam‑based refinement on simplex (log‑it parametrisation)
    # ------------------------------------------------------------------
    eps = 1e-12
    x0 = np.log(best_seq + eps)   # initialise logits

    best_seq, best_score = adam_optimize(
        x0=x0,
        n=n,
        start_time=start_time,
        deadline=deadline,
        init_best_a=best_seq,
        init_best_score=best_score,
        target_score=1.503,
        lam_start=0.5,
        lam_end=2000.0,
        lr_start=0.05,
        lr_end=0.001,
        beta1=0.9,
        beta2=0.999,
        eps=1e-8,
        eval_interval=400,
        stagnation_time=30.0,
    )

    elapsed = time.time() - start_time
    print(f"[done] best score={best_score:.6f}, elapsed={elapsed:.1f}s")
    # Return a plain Python list for the harness
    return best_seq.tolist()
