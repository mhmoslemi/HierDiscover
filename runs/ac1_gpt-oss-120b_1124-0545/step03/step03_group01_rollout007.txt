         0.2/√iteration), renormalise to sum = 1 and keep the update only
         if it strictly lowers the true objective.

   The LP uses SciPy’s high‑performance ‘highs’ solver and a dense
   Toeplitz matrix `A` (A_{k,i}=f_{k‑i} for the admissible indices).
   The loop stops when the global deadline (≈ 1000 s) is reached.

The final sequence (a Python list of floats) is returned.  Progress and
intermediate best values are printed to the console.
"""

import time
import numpy as np
from scipy.optimize import linprog  # hiGHS linear programming solver

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.  Returns the full result of length
    len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (sum = 1)."""
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised to sum = 1."""
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i ∝ 1/(i+1)**p, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    """Sinusoidal window a_i = sin(π·(i+1)/(n+1))."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random point on the simplex."""
    x = rng.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max (the same idea as the previous code)
# ----------------------------------------------------------------------


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 2500) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max of the auto‑convolution.
    Returns the best sequence (np.ndarray) found and its objective value
    2·n·max(conv) (the denominator sum(a) is 1 by construction).
    """
    v = np.log(seed + 1e-30)               # log‑softmax parameters

    # Adam moments
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-12

    beta = 5.0
    beta_mul = 1.05

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        t += 1
        a = _softmax(v)

        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:4d} best={cur_val:.6f}")

        # ----- smoothed‑max weights -----
        beta *= beta_mul
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]           # central n entries

        # convert to gradient w.r.t v (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on v
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement utilities
# ----------------------------------------------------------------------


def _build_A(f: np.ndarray, A: np.ndarray) -> None:
    """
    Fill the dense Toeplitz matrix A with entries A[k,i] = f[k-i] when
    0 ≤ k-i < n, otherwise 0.  The matrix has shape (2n‑1, n) and is
    reused across iterations to avoid repeated allocations.
    """
    n = len(f)
    L = 2 * n - 1
    # zero the whole matrix once
    A.fill(0.0)
    # column‑wise fill (fast NumPy slice assignment)
    for i in range(n):
        A[i:i + n, i] = f


def _lp_maximize_sum(f: np.ndarray, M: float, A: np.ndarray) -> np.ndarray | None:
    """
    Solve  maximise Σ g_i   subject to   A @ g ≤ M·1 , g ≥ 0.
    Returns the optimal g (as a dense ndarray) or None if the solver fails.
    """
    n = len(f)
    L = 2 * n - 1

    # A is already built for the current f
    c = -np.ones(n, dtype=np.float64)                # minimise -Σg_i  ⇔ maximise Σg_i
    b_ub = np.full(L, M, dtype=np.float64)

    res = linprog(c,
                  A_ub=A,
                  b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        # The LP might be infeasible only if M is too small; in that case we
        # fall back to the zero vector (which is always feasible).
        return None
    return res.x


def _mix_update(f: np.ndarray, g: np.ndarray, step: float) -> np.ndarray:
    """
    Convex combination (1‑step)·f + step·g followed by renormalisation to sum = 1.
    """
    a = (1.0 - step) * f + step * g
    s = a.sum()
    if s == 0.0:
        # extremely unlikely – fall back to uniform
        a = _uniform(len(f))
    else:
        a /= s
    return a


def _objective(a: np.ndarray) -> float:
    """Exact objective for a normalised sequence (sum = 1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Seed generation (identical to the previous code but trimmed)
# ----------------------------------------------------------------------


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest set of diverse starting points, all normalised to sum = 1."""
    seeds: list[np.ndarray] = [_uniform(n)]

    # geometric families
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))

    # sine window (always a good baseline)
    seeds.append(_sine_window(n))

    # a few random Dirichlet draws
    rng = np.random.default_rng(12345)
    for _ in range(8):
        seeds.append(_random_dirichlet(n, rng))

    # reuse previous construction if present
    if previous is not None and previous.shape == (n,):
        prev = previous.astype(np.float64)
        if prev.sum() > 0:
            seeds.append(prev / prev.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Length to use – 4000 works well and was the length of the
    # previously best construction (height_sequence_1).
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Seed acquisition – try to re‑use the previous best construction if
    # it exists in the global namespace.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    seeds = _make_seeds(n, previous)

    # ------------------------------------------------------------------
    # Quick scan of raw seeds
    # ------------------------------------------------------------------
    global_best = None
    global_best_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < global_best_val:
            global_best_val = val
            global_best = s.copy()
            print(f"[seed] new best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the smoothed max (short but useful)
    # ------------------------------------------------------------------
    seq_adam, val_adam = _adam_optimize(global_best, n, deadline, max_iters=2500)
    if seq_adam is not None and val_adam < global_best_val:
        global_best_val = val_adam
        global_best = seq_adam.copy()
        print(f"[Adam] improved best to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based refinement
    # ------------------------------------------------------------------
    # Pre‑allocate the Toeplitz matrix once (dense) to avoid repeated
    # allocations; we will reuse and overwrite it each iteration.
    A = np.empty((2 * n - 1, n), dtype=np.float64)

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current max of the auto‑convolution
        cur_obj = _objective(global_best)
        conv = _conv_fft(global_best, global_best)
        M = conv.max()

        # solve the LP
        _build_A(global_best, A)
        g = _lp_maximize_sum(global_best, M, A)
        if g is None:
            # shouldn't happen; abort the LP loop
            print("[LP] solver failed – terminating LP refinement")
            break

        # try a decreasing step size until an improvement is observed
        step = 0.2 / np.sqrt(iteration)   # initial guess
        improved = False
        while step > 1e-5:
            candidate = _mix_update(global_best, g, step)
            cand_val = _objective(candidate)
            if cand_val < global_best_val:
                global_best = candidate
                global_best_val = cand_val
                improved = True
                print(f"[LP] it={iteration:4d} step={step:.4f} best={cand_val:.6f}")
                break
            # no improvement – try a smaller step
            step *= 0.5

        if not improved:
            # If no step size helped, we still perform a tiny move to
            # escape possible stagnation.
            global_best = _mix_update(global_best, g, 1e-4)
            # re‑normalise (already done inside _mix_update) – recompute
            # the objective for the next loop.
    # ------------------------------------------------------------------
    # Final sanity check – optionally evaluate with the provided function
    # ------------------------------------------------------------------
    try:
        # The evaluation function may be present in the global namespace.
        from __main__ import evaluate_sequence  # noqa: F401
    except Exception:
        pass

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best)})")
    return global_best.tolist()
