def beta_shape(n, alpha):
    idx = np.arange(n) + 0.5  # use midpoints
    x = idx / n
    a = x**(alpha-1) * (1-x)**(alpha-1)  # may overflow near endpoints for alpha<1
    # Replace infinities / nan with large finite value
    a[np.isinf(a)] = 0
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum == 0: a = np.ones(n) / n
    else: a = a / a_sum
    return a

"""
Optimiser for the non‑negative step‑function objective.
The algorithm proceeds in three stages:

1.  **Extensive parametric sweep** – a family of monotone decreasing
    sequences a_i ∝ (i+1+δ)^{-α} and a_i ∝ exp(-β i) is explored.
    For each candidate the exact objective 2·n·max(conv(a,a)) (with Σa_i=1)
    is evaluated using an FFT‑based convolution.  The best few candidates are
    retained.

2.  **Exact projected sub‑gradient descent** – using the true
    sub‑gradient of the maximal convolution entry (instead of an
    approximation with the top‑k entries) the best candidates are
    refined.  After each step the vector is projected onto the probability
    simplex {x≥0, Σx=1}.  A simple step‑size decay together with occasional
    random mass‑swaps helps escape flat regions.

3.  **Short random restarts** – a handful of randomly perturbed start
    points are refined for a brief budget; the overall best sequence is
    kept and finally polished once more with a small step size.

All heavy work uses real FFT (`np.fft.rfft`) for O(n log n) complexity,
allowing lengths up to ≈ 4·10⁴ inside the 1000 s time limit.
The returned list satisfies the constraints of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Self‑convolution of a real vector using an FFT of size power‑of‑2."""
    n = seq.shape[0]
    size = _next_pow2(2 * n - 1)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    # Clip tiny negative values caused by numerical noise
    conv[conv < 0.0] = 0.0
    return conv


def _score_fft(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a normalized sequence (Σ=1)."""
    n = seq.shape[0]
    conv = _conv_fft(seq)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def _power_sequence(n: int, alpha: float, offset: float = 0.0) -> np.ndarray:
    """Monotone decreasing shape a_i ∝ (i+1+offset)^{-α}, Σ=1."""
    idx = np.arange(1, n + 1, dtype=float) + offset
    a = idx ** (-alpha)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _exp_sequence(n: int, beta: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-β·i), Σ=1."""
    i = np.arange(n, dtype=float)
    a = np.exp(-beta * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient step for max‑convolution
# ----------------------------------------------------------------------


def _exact_subgradient_step(seq: np.ndarray, step: float) -> np.ndarray:
    """
    Projected sub‑gradient step for the objective
    f(seq) = 2·n·max(conv(seq,seq)) using the exact gradient of the
    maximal convolution entry.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq)
    max_val = conv.max()
    # pick one index attaining the maximum (if several pick the first)
    max_idx = int(conv.argmax())

    lo = max(0, max_idx - (n - 1))
    hi = min(max_idx, n - 1)
    i_range = np.arange(lo, hi + 1, dtype=int)
    grad = np.zeros_like(seq)
    # derivative of (seq*seq)_k w.r.t seq[i] is 2·seq[k-i] when i in range
    grad[i_range] = 2.0 * seq[max_idx - i_range]

    # keep us on the affine subspace Σx = 1
    grad -= grad.mean()

    new_seq = seq - step * grad
    new_seq = _project_simplex(new_seq)
    return new_seq, max_val


def _run_exact_gradient(
    seq: np.ndarray,
    time_budget: float,
    *,
    max_iters: int = 10 ** 9,
    step0: float = 0.2,
    decay: float = 0.9999,
    momentum: float = 0.0,
) -> tuple[np.ndarray, float]:
    """
    Run projected sub‑gradient descent using the exact max‑convolution
    sub‑gradient.  Returns the best sequence found and its score.
    """
    start = time.time()
    n = seq.shape[0]
    best_seq = seq.copy()
    best_score = _score_fft(seq)

    step = step0
    v = np.zeros_like(seq)  # momentum accumulator
    iteration = 0

    while iteration < max_iters and (time.time() - start) < time_budget:
        iteration += 1

        # one sub‑gradient step
        seq, max_val = _exact_subgradient_step(seq, step)

        # optional momentum
        if momentum > 0.0:
            # simple momentum averaging (re‑project after momentum)
            v = momentum * v + (1.0 - momentum) * (seq - best_seq)
            seq = best_seq + v
            seq = _project_simplex(seq)

        # occasional random mass exchange to break ties
        if iteration % 250 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(step * seq[i], seq[i] * 0.3)
            seq[i] -= eps
            seq[j] += eps
            seq = _project_simplex(seq)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        step = max(step * decay, 1e-9)

    return best_seq, best_score

    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The procedure combines a wide parametric sweep (power‑law and exponential
    families) with a precise projected sub‑gradient descent on the maximal
    convolution entry.  It returns before the allotted time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the final return
    deadline = start_time + budget_s - 4.0

    # -------------------------------

    # -------------------------------
    n = 20000  # a comfortable size giving a good asymptotic constant

    # -------------------------------------------------
    # 2. Parametric sweep – store the top few candidates
    # -------------------------------------------------
    best_candidates = []  # list of (score, seq)
    max_candidates = 5

    # parameter grids
    alphas = np.linspace(0.35, 0.75, 21)      # step ≈ 0.02
    offsets = np.linspace(0.0, 5.0, 11)       # step = 0.5
    betas = np.linspace(0.0, 2.0, 21)        # exponential decay

    # Helper to keep list of best candidates sorted by score (ascending)
    def _maybe_store(seq, score):
        nonlocal best_candidates
        if len(best_candidates) < max_candidates:
            best_candidates.append((score, seq.copy()))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if score < best_candidates[-1][0]:
                best_candidates[-1] = (score, seq.copy())
                best_candidates.sort(key=lambda x: x[0])

    # sweep power‑law family
    for alpha in alphas:
        if time.time() >= deadline:
            break
        for off in offsets:
            seq = _power_sequence(n, alpha, off)
            score = _score_fft(seq)
            _maybe_store(seq, score)

    # sweep exponential family
    for beta in betas:
        if time.time() >= deadline:
            break
        seq = _exp_sequence(n, beta)
        score = _score_fft(seq)
        _maybe_store(seq, score)

    # report the current best after the sweep
    if best_candidates:
        print("[sweep] best score after param sweep =", best_candidates[0][0])

    # -------------------------------------------------
    # 3. Refine each stored candidate with exact gradient descent
    # -------------------------------------------------
    refined_best_seq = None
    refined_best_score = np.inf

    for init_score, init_seq in best_candidates:
        remaining = max(0.0, deadline - time.time())
        if remaining <= 0.0:
            break
        seq, score = _run_exact_gradient(
            init_seq,
            time_budget=remaining,
            max_iters=20000,
            step0=0.5,
            decay=0.99995,
            momentum=0.0,
        )
        if score < refined_best_score:
            refined_best_score = score
            refined_best_seq = seq.copy()
            print(f"[refine] new best score = {refined_best_score:.6f}")

        # early stopping if we already beat the target
        if refined_best_score <= 1.503:
            break

    # -------------------------------------------------
    # 4. Random restarts – short gradient runs from random seeds
    # -------------------------------------------------
    while time.time() < deadline - 30.0:
        # generate a random start (mix of power‑law and exponential)
        if np.random.rand() < 0.5:
            alpha = np.random.uniform(0.3, 0.8)
            off = np.random.uniform(0.0, 5.0)
            seq = _power_sequence(n, alpha, off)
        else:
            beta = np.random.uniform(0.0, 2.5)
            seq = _exp_sequence(n, beta)

        # inject a tiny Gaussian perturbation to explore the neighbourhood
        seq += np.random.randn(n) * 1e-5
        seq = np.maximum(seq, 0.0)
        seq = _project_simplex(seq)

        remaining = max(0.0, deadline - time.time())
        if remaining <= 0.0:
            break
        seq, score = _run_exact_gradient(
            seq,
            time_budget=remaining,
            max_iters=5000,
            step0=0.3,
            decay=0.9999,
            momentum=0.0,
        )
        if score < refined_best_score:
            refined_best_score = score
            refined_best_seq = seq.copy()
            print(f"[restart] improved to {refined_best_score:.6f}")

        if refined_best_score <= 1.503:
            break

    # -------------------------------------------------
    # 5. Final polishing – long run with a small step size
    # -------------------------------------------------
    if refined_best_seq is not None:
        remaining = max(0.0, deadline - time.time())
        refined_best_seq, refined_best_score = _run_exact_gradient(
            refined_best_seq,
            time_budget=remaining,
            max_iters=10 ** 9,
            step0=0.2,
            decay=0.99998,
            momentum=0.0,
        )

    # make absolutely sure we lie on the simplex
    if refined_best_seq is not None:
        refined_best_seq = _project_simplex(refined_best_seq)

    # -------------------------------------------------
    # 6. Return the result
    # -------------------------------------------------
    if refined_best_seq is None:
        # fallback: uniform distribution (guaranteed feasible)
        refined_best_seq = np.ones(n) / n
        refined_best_score = _score_fft(refined_best_seq)

    print(f"[final] best_score={refined_best_score:.6f}, length={n}")
    return refined_best_seq.tolist()

        # occasional random mass swap to escape plateaus
        if iteration % 250 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = min(step * seq[i], seq[i] * 0.3)
            seq[i] -= eps
            seq[j] += eps
            seq = _project_simplex(seq)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        step = max(step * decay, 1e-9)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main candidate generator required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The procedure combines a wide parametric sweep (power‑law and exponential
    families) with a precise projected sub‑gradient descent on the maximal
    convolution entry.  It returns before the allotted time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a few seconds for the final return
    deadline = start_time + budget_s - 4.0

    # -------------------------------------------------
    # 1. Problem size
    # -------------------------------------------------
    n = 20000                     # comfortable size, gives a good asymptotic constant

    # -------------------------------------------------
    # 2. Parametric sweep – keep the top few candidates
    # -------------------------------------------------
    best_candidates = []          # list of (score, seq)
    max_keep = 5

    def _maybe_store(seq: np.ndarray, score: float):
        """Insert (score,seq) into the sorted best‑candidate list."""
        nonlocal best_candidates
        entry = (score, seq.copy())
        if len(best_candidates) < max_keep:
            best_candidates.append(entry)
            best_candidates.sort(key=lambda x: x[0])
        else:
            if score < best_candidates[-1][0]:
                best_candidates[-1] = entry
                best_candidates.sort(key=lambda x: x[0])

    # parameter grids
    alphas = np.linspace(0.35, 0.75, 21)       # step ≈ 0.02
    offsets = np.linspace(0.0, 5.0, 11)        # step = 0.5
    betas = np.linspace(0.0, 2.0, 21)         # exponential decay

    # sweep power‑law family
    for alpha in alphas:
        if time.time() >= deadline:
            break
        for off in offsets:
            seq = _power_sequence(n, alpha, off)
            score = _score_fft(seq)
            _maybe_store(seq, score)

    # sweep exponential family
    for beta in betas:
        if time.time() >= deadline:
            break
        seq = _exp_sequence(n, beta)
        score = _score_fft(seq)
        _maybe_store(seq, score)

    if best_candidates:
        print("[sweep] best score after param sweep =", best_candidates[0][0])

    # -------------------------------------------------
    # 3. Refine the stored candidates
    # -------------------------------------------------
    refined_best_seq = None
    refined_best_score = np.inf

    for init_score, init_seq in best_candidates:
        remaining = max(0.0, deadline - time.time())
        if remaining <= 0.0:
            break
        seq, score = _run_exact_gradient(
            init_seq,
            time_budget=remaining,
            max_iters=20000,
            step0=0.5,
            decay=0.99995,
        )
        if score < refined_best_score:
            refined_best_score = score
            refined_best_seq = seq.copy()
            print(f"[refine] new best score = {refined_best_score:.6f}")

        if refined_best_score <= 1.503:
            break

    # -------------------------------------------------
    # 4. Random restarts – short gradient runs from random seeds
    # -------------------------------------------------
    while time.time() < deadline - 30.0:
        # random start: either power‑law or exponential
        if np.random.rand() < 0.5:
            alpha = np.random.uniform(0.3, 0.8)
            off = np.random.uniform(0.0, 5.0)
            seq = _power_sequence(n, alpha, off)
        else:
            beta = np.random.uniform(0.0, 2.5)
            seq = _exp_sequence(n, beta)

        # tiny Gaussian jitter to explore the neighbourhood
        seq += np.random.randn(n) * 1e-5
        seq = np.maximum(seq, 0.0)
        seq = _project_simplex(seq)

        remaining = max(0.0, deadline - time.time())
        if remaining <= 0.0:
            break
        seq, score = _run_exact_gradient(
            seq,
            time_budget=remaining,
            max_iters=5000,
            step0=0.3,
            decay=0.9999,
        )
        if score < refined_best_score:
            refined_best_score = score
            refined_best_seq = seq.copy()
            print(f"[restart] improved to {refined_best_score:.6f}")

        if refined_best_score <= 1.503:
            break

    # -------------------------------------------------
    # 5. Final polishing – long run with a very small step size
    # -------------------------------------------------
    if refined_best_seq is not None:
        remaining = max(0.0, deadline - time.time())
        refined_best_seq, refined_best_score = _run_exact_gradient(
            refined_best_seq,
            time_budget=remaining,
            max_iters=10 ** 9,
            step0=0.2,
            decay=0.99998,
        )

    # guarantee exact feasibility
    if refined_best_seq is not None:
        refined_best_seq = _project_simplex(refined_best_seq)

    # fallback – uniform distribution (always feasible)
    if refined_best_seq is None:
        refined_best_seq = np.ones(n) / n
        refined_best_score = _score_fft(refined_best_seq)

    print(f"[final] best_score={refined_best_score:.6f}, length={n}")
    return refined_best_seq.tolist()
