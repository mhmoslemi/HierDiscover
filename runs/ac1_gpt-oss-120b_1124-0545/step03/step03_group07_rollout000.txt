def build_conv_matrix(a):
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        j_start = max(0, k - (n-1))
        j_end = min(k, n-1)
        # indices j from j_start to j_end inclusive
        # correspond to variable indices j
        # coefficient a_[k-j]
        # generate arrays
        js = np.arange(j_start, j_end+1)
        coeffs = a[k - js]
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        data.extend(coeffs.tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A
```

Propose a non‑negative step‑function sequence that minimizes
`evaluate_sequence`.  

The algorithm has two phases:

1️⃣  **Linear‑programming improvement** – starting from an initial
    vector `a` (the previous best construction if available, otherwise a
    random exponential vector) we repeatedly solve the LP

        max   Σ_j b_j
        s.t.  (a * b)_k ≤ M     for all k,
              b_j ≥ 0,

    where `M = max_k (a * a)_k`.  The optimal `b` has the largest possible
    ℓ¹‑norm under the same convolution bound.  After scaling `b` to the
    required total mass √(2·n) we perform a short line‑search on the
    convex combination `a←(1−t)a + t·b` and keep any improvement of `M`.
    The LP is built with a sparse matrix and solved with SciPy’s
    high‑performance “highs” backend; a few (≤ 15) iterations already
    give a noticeable drop in the max‑convolution value.

2️⃣  **Simulated annealing fine‑tuning** – with the remaining time budget
    we run a classic SA that moves a small fraction of mass from one
    coordinate to another, accepting uphill moves with probability
    `exp(‑Δ/T)`.  The temperature decays exponentially from `0.2·current`
    to `1e‑5·current`.  This phase refines the solution found by the LP
    phase.

Both phases respect the global 1000 s time limit and constantly keep the
best sequence seen.  The returned object is a plain Python list of
floats, ready for `evaluate_sequence`.
"""

import time
import math
import random
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex {w ≥ 0, Σ w = s}."""
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1, used for FFT‑based convolutions."""
    return 1 << ((2 * n - 1).bit_length())


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """
    Return (max_convolution_value, argmax_index) for the self‑convolution
    a*a using an FFT of length `fft_len`.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    idx = int(np.argmax(conv))
    val = float(conv[idx])
    return val, idx


# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix `A` such that (A @ b)_k = (a * b)_k,
    i.e. each row implements one convolution inequality.
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []

    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # indices of b participating in (a*b)_k
        js = np.arange(j_start, j_end + 1, dtype=int)
        # corresponding coefficients a_{k-j}
        coeffs = a[k - js]
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        data.extend(coeffs.tolist())

    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A
