Pseudo:

def project_simplex(v):
    # v is array; returns projection onto {x>=0, sum=1}
    # standard algorithm:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Projected sub‑gradient descent for minimizing the max auto‑convolution.

The algorithm works on the probability simplex (sum = 1, a_i ≥ 0).
At each iteration we locate the index where the convolution a*a attains its
maximum, compute the exact sub‑gradient of this max (which is 2·a shifted
by that index), take a diminishing step in the opposite direction and
project back onto the simplex.
Occasional random jitter moves are performed when no improvement is seen
for a while, helping the search escape plateaus.
The routine respects a 1000 s time budget and returns the best sequence
found.
"""

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Duchi et al. algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # Find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive after shift: return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray, fft_len: int) -> tuple[float, np.ndarray]:
    """Return max value of a*a and the full convolution (real)."""
    # using real FFT for speed
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    max_val = conv.max()
    return float(max_val), conv


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """Search for a non‑negative step‑function that minimises `evaluate_sequence`.

    The method is a projected sub‑gradient descent on the simplex,
    followed by occasional stochastic restarts.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    n = int(kwargs.get("n", 2000))
    # pre‑allocate FFT length (next power‑of‑two ≥ 2n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ---------- initial candidates ----------
    candidates = []

    # 1. arcsine shape (inverse‑sqrt)
    i = np.arange(n, dtype=float)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine = arcsine / arcsine.sum()
    candidates.append(arcsine)

    # 2. uniform
    candidates.append(np.full(n, 1.0 / n))

    # 3. random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))

    # 4. previous best if present
    if "height_sequence_1" in globals():
        try:
            prev = np.array(globals()["height_sequence_1"], dtype=float)
            if prev.size == n:
                prev = np.maximum(prev, 0.0)
                prev = prev / prev.sum()
                candidates.append(prev)
        except Exception:
            pass

    # Choose the best among the cheap initial points
    best_a = None
    best_score = np.inf
    for a0 in candidates:
        max_val, _ = _conv_max(a0, fft_len)
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a0.copy()

    # ---------- main optimisation loop ----------
    a = best_a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * _conv_max(a, fft_len)[0]
    last_improve = time.time()
    it = 0
    step0 = 0.4   # initial step size (will decay)

    while time.time() < deadline:
        it += 1

        # compute convolution and locate the maximal entry
        max_val, conv = _conv_max(a, fft_len)
        cur_score = 2.0 * n * max_val

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.time()
            print(f"[{it:07d}] new best score = {best_score:.6f}")

        # sub‑gradient of the maximal convolution entry
        idx = int(np.argmax(conv))
        # compute gradient g_i = 2 * a_{idx-i} for valid indices
        g = np.zeros_like(a)
        j_start = max(0, idx - (n - 1))
        j_end = min(idx, n - 1)
        if j_start <= j_end:
            j = np.arange(j_start, j_end + 1)
            i_idx = idx - j
            g[i_idx] = 2.0 * a[j]

        # diminishing step size
        step = step0 / np.sqrt(it)

        # gradient step on the simplex (projected)
        a_candidate = a - step * g
        a = _project_simplex(a_candidate)

        # occasional stochastic jitter when stuck

                jitter = np.random.randn(n) * 0.002
                a = _project_simplex(best_a + jitter)
                print(f"[{it:07d}] jitter restart after stagnation")

    total = time.time() - start
    print(f"[done] it={it}, best_score={best_score:.6f}, time={total:.1f}s")

    # final safety – enforce non‑negativity and normalisation
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0:
        best_a = np.full_like(best_a, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()
    return best_a.tolist()

"""
Projected sub‑gradient descent (with periodic stochastic jitter) for the
min‑max auto‑convolution problem.

The objective of `evaluate_sequence` is
    E(a) = 2·n·max(conv(a,a)) / (sum(a)**2).

Because the expression is homogeneous we restrict the search to the
probability simplex Σa_i=1, a_i≥0.  For a current vector a the index
j* where conv(a,a) attains its maximum is found.  The sub‑gradient of the
max‑entry with respect to a is

    g_i = 2·a_{j*‑i}  (if 0≤j*‑i<n, else 0).

We step in the opposite direction with a diminishing step size and project
back onto the simplex (Euclidean projection).  The projection uses the
standard O(n log n) algorithm of Duchi et al.  When no improvement is seen
for a while a small random jitter is added to the current best point to
escape plateaus.  All convolutions are evaluated with an FFT (real‑FFT)
so that each iteration is O(n log n) and thousands of iterations fit well
into the 1000 s budget.

The routine respects the time budget, prints occasional progress messages
and finally returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    # Implementation of Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)                # 1‑based indexing
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries would become ≤0
        return np.full_like(v, 1.0 / n)       # uniform distribution
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray, fft_len: int) -> tuple[float, np.ndarray]:
    """
    Return max(conv(a,a)) and the full convolution using a real FFT.
    The convolution length is 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return float(conv.max()), conv

    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects a time budget (default
    1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    n = int(kwargs.get("n", 2000))

    # pre‑allocate FFT length (next power of two ≥ 2·n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 1) build a small pool of cheap starting points
    # ------------------------------------------------------------------
    candidates = []

    # arcsine / inverse‑sqrt shape (known good analytical starter)
    i = np.arange(n, dtype=float)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine = arcsine / arcsine.sum()
    candidates.append(arcsine)

    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet vector
    candidates.append(np.random.dirichlet(np.ones(n)))

    # reuse a previously discovered sequence if it exists

    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s == 0.0:
