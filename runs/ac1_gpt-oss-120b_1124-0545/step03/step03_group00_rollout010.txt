       """Project onto simplex such that sum = 1 and components >= 0."""
       u = np.sort(v)[::-1]
       cssv = np.cumsum(u) - 1
       ind = np.arange(len(v)) + 1
       cond = u - cssv / ind > 0
       rho = ind[cond][-1]
       theta = cssv[cond][-1] / rho
       w = np.maximum(v - theta, 0)

"""
A fast multi‑stage optimiser for the step‑function bound.

Stage 1 – diverse analytic seeds (geometric, power‑law, beta,
symmetric, sine‑window, Hann, Dirichlet, random softmax).

Stage 2 – quick scan of the exact objective (2·n·max(conv)) to pick
the best seed for the current length.

Stage 3 – two complementary gradient refinements:
    * Adam on a *soft‑max* parametrisation with a smoothly‑annealed
      max‑approximation (log‑sum‑exp) – uses the same code as the
      reference but with a slightly more aggressive β‑schedule.
    * A “hard‑max” projected‑gradient descent that always follows the
      true ∞‑norm (active set of maximum autocorrelation entries).
      The update is performed in log‑softmax space, which guarantees
      non‑negativity and exact normalisation.

Stage 4 – a few perturbed restarts from the current best sequence.

All work is bounded by the supplied time budget (default 1000 s) and
the best sequence found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution via FFT.
    Returns a vector of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.ones(n, dtype=np.float64) / n


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n, dtype=np.float64) / n


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n, dtype=np.float64) / n


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return a / a.sum()


def _beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric (or skewed) Beta distribution on the discretised grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid 0^0
    eps = np.finfo(float).eps
    x = np.clip(x, eps, 1 - eps)
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    s = a.sum()
    return a / s if s != 0 else np.ones(n, dtype=np.float64) / n


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _refine_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 8.0,
    beta_mul: float = 1.04,
    lr0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.
    Uses a log‑sum‑exp smoothing of the max.
    Returns the best sequence found and its exact objective value.
    """
    v = np.log(a0 + 1e-30)          # logits
    m = np.zeros_like(v)           # first moment
    v2 = np.zeros_like(v)          # second moment
    best_seq = a0.copy()
    cur_conv = _conv_fft(a0, a0)
    best_val = 2.0 * n * np.max(cur_conv)

    beta = start_beta
    it = 0

    while time.time() < deadline:
        it += 1

        a = _softmax(v)

        # ----- exact objective (for bookkeeping) -----
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(
                f"[adam] n={n:5d} it={it:6d} best={cur_val:.6f}"
            )

        # ----- smooth‑max weights -----
        beta *= beta_mul
        shift = cur_max
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient w.r.t. a (smooth loss) -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]    # central n entries

        # ----- gradient w.r.t. logits v -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update -----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


def _refine_hard_max(
    a0: np.ndarray,
    n: int,
    deadline: float,
    step0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Projected (log‑softmax) gradient descent using the *true* max,
    i.e. the active set of indices where the autocorrelation attains
    its current maximum.  This provides a sharper descent after the
    Adam phase.
    """
    v = np.log(a0 + 1e-30)          # logits, always yields a simplex point
    best_seq = a0.copy()
    conv = _conv_fft(a0, a0)
    best_val = 2.0 * n * np.max(conv)

    it = 0
    while time.time() < deadline:
        it += 1
        a = _softmax(v)

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(
                f"[hard] n={n:5d} it={it:6d} best={cur_val:.6f}"
            )

        # Identify the active set (indices attaining the maximum)
        # a tolerance of 1e‑12 relative to the maximum is sufficient.
        tol = 1e-12
        active = np.where(conv >= cur_max - tol)[0]

        # weight vector for the active set – uniform distribution on it
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)

        # gradient w.r.t. a for the true max (sub‑gradient)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # map to gradient on logits (mirror‑descent on simplex)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # simple diminishing step size
        step = step0 / np.sqrt(it)
        v = v - step * grad_v

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    (default 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Small safety margin – we must return before the external timeout
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Lengths to try – start with a large one, fall back if time runs out
    # ------------------------------------------------------------------
    lengths = [8000, 4000]  # both are fast enough for the budget

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[main] Optimising length = {n}")

        # --------------------------------------------------------------
        # 1.  Build a diverse collection of seed vectors (all on the simplex)
        # --------------------------------------------------------------
        seed_seqs = []

        # Geometric families
        for c in np.linspace(0.5, 5.0, 12):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_geom(n, r))

        # Power‑law families
        for p in np.linspace(0.0, 2.2, 12):
            seed_seqs.append(_power(n, p))

        # Combined geometric × power families
        for c in np.linspace(0.5, 4.0, 8):
            r = 1.0 - c / n
            if r <= 0.0:
                continue
            for p in np.linspace(0.0, 2.0, 8):
                seed_seqs.append(_geom_power(n, r, p))

        # Symmetric geometric
        for c in np.linspace(0.5, 4.5, 8):
            r = 1.0 - c / n
            if r > 0.0:
                seed_seqs.append(_sym_geom(n, r))

        # Classical windows
        seed_seqs.append(_sine_window(n))
        seed_seqs.append(_hann_window(n))

        # Beta‑pdf shapes (both symmetric and skewed)
        for a, b in [(0.8, 0.8), (0.8, 1.2), (1.2, 0.8), (0.5, 0.5), (2.0, 2.0)]:
            seed_seqs.append(_beta_pdf(n, a, b))

        # Dirichlet samples with several concentration parameters
        for alpha in [0.2, 0.5, 0.8, 2.0, 5.0]:
            a = np.random.gamma(alpha, 1.0, size=n)
            s = a.sum()
            if s > 0:
                seed_seqs.append(a / s)

        # Random soft‑max seeds
        for _ in range(6):
            v_rand = np.random.randn(n)
            seed_seqs.append(_softmax(v_rand))

        # If a previous construction exists, import it as a seed
        prev = globals().get("height_sequence_1")
        if prev is not None:
            try:
                a_prev = np.asarray(prev, dtype=np.float64)
                if a_prev.ndim == 1:
                    if a_prev.shape[0] != n:
                        a_adj = np.zeros(n, dtype=np.float64)
                        m = min(a_prev.shape[0], n)
                        a_adj[:m] = a_prev[:m]
                        a_prev = a_adj
                    if a_prev.sum() > 0:
                        a_prev = a_prev / a_prev.sum()
                        seed_seqs.append(a_prev)
            except Exception:
                pass

        # --------------------------------------------------------------
        # 2.  Quick scan – keep the best seed (exact objective)
        # --------------------------------------------------------------
        for idx, a in enumerate(seed_seqs):
            conv = _conv_fft(a, a)
            cur_val = 2.0 * n * np.max(conv)
            if cur_val < global_best_val:
                global_best_val = cur_val
                global_best_seq = a.copy()
                print(f"[scan] new global best from seed {idx} = {cur_val:.6f}")

        # --------------------------------------------------------------
        # 3.  Adam refinement (smooth max)
        # --------------------------------------------------------------
        if global_best_seq is None:
            global_best_seq = np.ones(n, dtype=np.float64) / n
            print("[fallback] uniform start")

        refined_seq, refined_val = _refine_adam(
            global_best_seq,
            n,
            deadline,
            start_beta=8.0,
            beta_mul=1.04,
            lr0=0.8,
        )
        if refined_val < global_best_val:
            global_best_val = refined_val
            global_best_seq = refined_seq.copy()
            print(f"[overall] new global best after Adam (len={n}) = {refined_val:.6f}")

        # --------------------------------------------------------------
        # 4.  Hard‑max (sub‑gradient) refinement – sharpens the tail
        # --------------------------------------------------------------
        # Use whatever time is left after the Adam phase.
        refined_seq2, refined_val2 = _refine_hard_max(
            global_best_seq,
            n,
            deadline,
            step0=0.4,
        )
        if refined_val2 < global_best_val:
            global_best_val = refined_val2
            global_best_seq = refined_seq2.copy()
            print(f"[overall] new global best after hard‑max (len={n}) = {refined_val2:.6f}")

        # --------------------------------------------------------------
        # 5.  Perturbed restarts (few quick Adam runs) while time permits
        # --------------------------------------------------------------
        while time.time() < deadline:
            noisy_logits = np.log(global_best_seq + 1e-30) + 0.01 * np.random.randn(n)
            pert_seq = _softmax(noisy_logits)
            seq2, val2 = _refine_adam(
                pert_seq,
                n,
                deadline,
                start_beta=8.0,
                beta_mul=1.04,
                lr0=0.8,
            )
            if val2 < global_best_val:
                global_best_val = val2
                global_best_seq = seq2.copy()
                print(f"[perturb] improved global best to {val2:.6f}")

        # End of this length – continue with the next length if any.
        print(f"[len={n}] best objective so far = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        global_best_seq = np.ones(n, dtype=np.float64) / n
        print("[fallback] final uniform sequence")

    # Normalise to sum‑to‑one (objective is scale‑invariant)
    s = np.sum(global_best_seq)
    if s > 0:
        global_best_seq = global_best_seq / s

    # Convert to plain list for the scorer
    result = global_best_seq.tolist()
    try:
        # One final evaluation (uses the provided helper)
        final_obj = evaluate_sequence(result)
        print(f"[final] evaluate_sequence = {final_obj:.6f}")
    except Exception:
        print("[final] evaluation failed (should not happen)")

    return result
