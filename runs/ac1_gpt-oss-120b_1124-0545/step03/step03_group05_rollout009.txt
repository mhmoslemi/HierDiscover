        a /= a.sum()

    while time.time() < deadline:

        scores = []
        for a in pop:
            conv = conv_full(a)
            maxb = conv.max()
            score = 2.0 * n * maxb
            scores.append(score)

        # update best

            best_score = scores[i_best]
            best_seq = pop[i_best].copy()
            print(f"[gen {generation}] new best = {best_score}")

        elite_idx = scores.argsort()[:elite]

        # generate new population
        new_pop = []
        # keep elites

"""
Optimisation of a step‑function to minimise 2·n·max(conv(a,a)).
-----------------------------------------------------------

The objective is convex (max of linear forms) and invariant to scaling.
We therefore optimise a probability vector a (a_i ≥ 0, Σa_i = 1).

Algorithm (single‑candidate, exponentiated‑gradient with ADAM‑style
adaptation):
* Initialise a (use the supplied `height_sequence_1` if available,
  otherwise random).
* In each iteration:
  - Compute the full linear convolution a∗a by FFT.
  - Form a smooth approximation of the max using a soft‑max with a
    temperature τ that slowly decays.
  - The gradient of the smooth objective is
        g = 2 · (a_rev ∗ w)          (w = soft‑max weights)
    where a_rev is a reversed copy of a.
  - Update a with a variant of the ADAM optimiser; after the additive
    update we project back onto the probability simplex by clipping to
    ≥0 and renormalising.
  - With probability `rand_prob` we perform a cheap “mass‑transfer’’
    perturbation (move a random fraction of mass from one index to another)
    to help escape plateaus.
  - Track the best sequence seen so far.  If no improvement occurs for
    a while we re‑seed from the current best with a small random jitter.
* The loop runs until the time budget (≈1000 s) is exhausted.
* The best sequence found is returned as a plain Python list.

The use of ADAM together with the exponentiated‑gradient projection
gives faster convergence than the plain projected sub‑gradient method
used previously, while the occasional random transfer adds exploration.
"""

import time
import numpy as np
from typing import List

# --------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# --------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Linear convolution a*a (full length) via FFT.
    The result has length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=size_fft)
    conv_full = np.fft.irfft(fa * fa, n=size_fft)
    # keep only the needed part
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Compute a soft‑max weighting of the convolution entries.
    The weight vector sums to 1 and concentrates on entries near the
    maximum as τ → 0.
    """
    max_val = conv.max()
    # shift for numerical stability; exponent never exceeds 0
    exp_arg = (conv - max_val) / tau
    # exp may under‑flow; safely handle the all‑zero case
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    return w


def _gradient_via_fft(a_rev: np.ndarray, weight: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Compute the gradient ∇J = 2·(a_rev ∗ weight) for the smooth
    objective J(a) = Σ w_k·(a∗a)_k.
    Returns only the central n entries (same length as a_rev).
    """
    fa_rev = np.fft.rfft(a_rev, n=size_fft)
    fw = np.fft.rfft(weight, n=size_fft)
    grad_full = np.fft.irfft(fa_rev * fw, n=size_fft)
    n = a_rev.shape[0]
    # central part of length n (indices n‑1 … 2n‑2)
    grad_center = grad_full[n - 1 : n - 1 + n]
    return 2.0 * grad_center


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """
    Transfer a random fraction of mass from one index to another.
    The amount transferred is ≤ scale·a[i].
    The resulting vector is re‑normalised to sum to 1.
    """
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    s = a_new.sum()
    if s == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= s
    return a_new


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Exact Euclidean projection of v onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection only works on 1‑D arrays.")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a low‑value step function within ``budget_s`` seconds.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        ``n`` – length of the sequence (default 6000).

    Returns
    -------
    list[float]
        The best sequence found (list of Python floats).
    """

    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin for the harness

    # problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 6000))

    # ------------------------------------------------------------------
    seq0 = globals().get("height_sequence_1", None)

        if a.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, a.size, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            a = np.interp(x_new, x_old, a)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.random.rand(n)
            a /= a.sum()
        else:
            a /= s
    else:
        a = np.random.rand(n)
        a /= a.sum()

    # FFT bookkeeping (fixed for the whole run)
    # ------------------------------------------------------------------
    size_fft = _next_pow2(2 * n - 1)  # enough for full convolution

    # ADAM state
    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    alpha = 0.2  # base learning rate for ADAM

    # hyper‑parameters for the smoothing/temperature
    tau_factor = 0.3          # τ = τ_factor * max(conv)
    tau_decay = 0.9995
    min_tau = 1e-12

    # random‑transfer schedule
    rand_prob = 0.12
    rand_scale = 0.05

    # stagnation handling
    stagnation_limit = 120.0  # seconds without improvement → jitter restart
    last_improve = start

    best_score = np.inf
    best_seq = a.copy()
    it = 0

    print(f"[init] n={n}, start score will be computed...")

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --- 1. compute convolution and current score -------------------
        conv = _full_convolution_fft(a, size_fft)          # length 2n‑1
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # --- 2. decide whether to do a random perturbation ------------
        if np.random.rand() < rand_prob:
            # cheap random mass transfer
            a = _random_transfer(a, rand_scale)
        else:
            # --- 2a. smooth max (soft‑max) ----------------------------
            tau = max(tau_factor * max_conv, min_tau)

            weight = _softmax_weights(conv, tau)

            # --- 2b. ADAM gradient step ------------------------------
            a_rev = a[::-1]
            grad = _gradient_via_fft(a_rev, weight, size_fft)   # ∇J

            # ADAM moment updates
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            t = it + 1.0
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            descent = -alpha * m_hat / (np.sqrt(v_hat) + eps_adam)

            a = a + descent

            # project back onto the simplex (clip + renormalise)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a)

            # temperature decay
            tau_factor *= tau_decay

        # --- 3. occasional jitter restart if we are stuck ----------
        if time.time() - last_improve > stagnation_limit:
            # start from the best seen so far with a small random perturbation
            a = best_seq + 0.02 * np.random.rand(n)
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a)
            # reset ADAM moments to avoid stale directions
            m.fill(0.0)
            v.fill(0.0)
            # reset temperature schedule a little larger to explore again
            tau_factor = max(0.3, tau_factor)
            print(f"[{it:7d}] restart (jitter) after {stagnation_limit}s stagnation")

        it += 1

    total_time = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
