"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence().  The strategy is:

1. **Diverse seed generation** – a wide collection of analytically
   motivated probability vectors (geometric, power‑law, combined,
   symmetric, sine, Gaussian, linear‑decay) plus random Dirichlet
   draws and a few soft‑max initialisations.

2. **Fast scan** – evaluate the exact objective ( 2·n·max(conv) )
   for each seed using an FFT convolution and keep the best few.

3. **Hybrid local optimisation** – starting from the current best
   seed, run two complementary refinements under the remaining
   time budget:

   * **Adam on a smoothed max** – the same log‑softmax Adam optimiser
     used previously, but with a more aggressive schedule and a
     configurable start‑beta.
   * **Simulated‑annealing hill‑climbing** – simple pairwise mass
     transfers on the probability vector, accepted with a
     temperature‑controlled Metropolis rule.  This explores the
     highly non‑convex landscape that the max‑convolution creates.

   The best sequence found by either stage is kept.

The algorithm respects the 1000 s time limit, prints progress, and
returns a plain list of floats ready for evaluate_sequence().
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / np.sum(e)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s != 0 else np.ones(n, dtype=np.float64) / n


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n, dtype=np.float64) / n


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s != 0 else np.ones(n, dtype=np.float64) / n


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian_window(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the vector."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s != 0 else np.ones(n, dtype=np.float64) / n


def _fast_objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) – the objective used for fast scans."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _refine_adam(
    a0: np.ndarray,
    n: int,
    deadline: float,
    start_beta: float = 0.5,
    beta_mul: float = 1.01,
    lr0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parametrisation.
    Returns the best sequence (as np.ndarray) and its fast objective.
    """
    # initialise logits
    v = np.log(a0 + 1e-30)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_seq = a0.copy()
    best_val = _fast_objective(a0)

    beta = start_beta
    it = 0

    while time.time() < deadline:
        it += 1

        a = _softmax(v)                     # current simplex point
        cur_val = _fast_objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[adam] n={n:5d} it={it:6d} best={cur_val:.6f}")

        # ---- smoothed‑max weights ----
        beta *= beta_mul
        shift = cur_val / (2.0 * n)          # current max of the convolution
        conv = _conv_fft(a, a)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of smoothed loss w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # gradient w.r.t. logits (softmax)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v2_hat = v2 / (1.0 - 0.999 ** it)

        lr = lr0 / np.sqrt(it)
        v -= lr * m_hat / (np.sqrt(v2_hat) + 1e-12)

    return best_seq, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    deadline: float,
    init_temp: float = 0.05,
    cooling: float = 0.9998,
    step_factor: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Simple pair‑wise mass‑transfer SA on the probability simplex.
    Returns best sequence found and its fast objective.
    """
    a = a0.copy()
    cur_val = _fast_objective(a)
    best_seq = a.copy()
    best_val = cur_val

    temp = init_temp * cur_val               # scale temperature to the current value
    it = 0
    max_step = step_factor

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # transfer a random amount from i to j (cannot go negative)
        max_transfer = min(a[i], max_step)
        if max_transfer <= 0.0:
            continue
        delta = np.random.rand() * max_transfer

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate
        new_val = _fast_objective(a_new)

        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / (temp + 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a_new.copy()
                print(f"[SA] it={it:8d} new best={new_val:.6f}")

        # temperature decay
        temp *= cooling
        # gradually shrink the maximum step size
        if it % 2000 == 0:
            max_step *= 0.9
            max_step = max(max_step, 1e-6)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    The routine respects the supplied time budget (default 1000 s) and returns
    the best sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin so we don't overrun the hard limit
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 8000
    print(f"[main] Optimising length = {n}")

    # ------------------------------------------------------------------
    # 1. Seed generation
    # ------------------------------------------------------------------
    seed_seqs = []

    # uniform baseline
    seed_seqs.append(np.ones(n, dtype=np.float64) / n)

    # geometric families
    for c in np.linspace(0.5, 5.0, 12):
        r = 1.0 - c / n
        if r > 0.0:
            seed_seqs.append(_geom(n, r))

    # power‑law families
    for p in np.linspace(0.0, 2.2, 12):
        seed_seqs.append(_power(n, p))

    # combined geometric × power families
    for c in np.linspace(0.5, 4.0, 8):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 8):
            seed_seqs.append(_geom_power(n, r, p))

    # symmetric geometric (not monotone – kept for completeness)
    for c in np.linspace(0.5, 4.5, 8):
        r = 1.0 - c / n
        if r > 0.0:
            seed_seqs.append(_sym_geom(n, r))

    # smooth sine window
    seed_seqs.append(_sine_window(n))

    # Gaussian windows (different sigmas)
    for sigma in [n / 4.0, n / 6.0, n / 8.0, n / 3.0]:
        seed_seqs.append(_gaussian_window(n, sigma))

    # linear decay (max(0, 1 - α·i/n))
    for alpha in np.linspace(0.0, 0.9, 7):
        a = np.maximum(0.0, 1.0 - alpha * np.arange(n) / n)
        a = a / a.sum()
        seed_seqs.append(a)

    # Dirichlet draws – various concentration parameters
    for alpha in [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]:
        a = np.random.gamma(alpha, 1.0, size=n)
        s = a.sum()
        if s > 0:
            seed_seqs.append(a / s)

    # Random soft‑max seeds
    for _ in range(5):
        v_rand = np.random.randn(n)
        seed_seqs.append(_softmax(v_rand))

    # If a previous construction exists, reuse it (resize/pad if needed)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            a_prev = np.asarray(prev, dtype=np.float64)
            if a_prev.ndim == 1:
                if a_prev.shape[0] != n:
                    if a_prev.shape[0] > n:
                        a_prev = a_prev[:n]
                    else:
                        a_prev = np.concatenate([a_prev, np.zeros(n - a_prev.shape[0])])
                a_prev = np.maximum(0.0, a_prev)
                s = a_prev.sum()
                if s > 0:
                    seed_seqs.append(a_prev / s)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Fast scan – pick the best few seeds
    # ------------------------------------------------------------------
    best_val = float("inf")
    best_seq = None
    for idx, a in enumerate(seed_seqs):
        cur_val = _fast_objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[scan] new best from seed {idx:3d} = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Hybrid refinement
    # ------------------------------------------------------------------
    # Split the remaining budget roughly 55 % for Adam, 45 % for SA.
    remaining = deadline - time.time()
    if remaining <= 0.0:
        # Out of time – return what we have
        return best_seq.tolist()

    adam_deadline = time.time() + remaining * 0.55

    # ---- Adam refinement ----
    seq_adam, val_adam = _refine_adam(
        best_seq,
        n,
        adam_deadline,
        start_beta=0.5,
        beta_mul=1.01,
        lr0=0.8,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = seq_adam.copy()
        print(f"[adam] improved global best to {val_adam:.6f}")

    # ---- Simulated annealing refinement ----
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            deadline,
            init_temp=0.05,
            cooling=0.9998,
            step_factor=0.02,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = seq_sa.copy()
            print(f"[SA] improved global best to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # 4. Final sanity check & return
    # ------------------------------------------------------------------
    # Normalise one last time (objective is scale‑invariant, but this is
    # required by the security checks in evaluate_sequence).
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s
    else:
        best_seq = np.ones(n, dtype=np.float64) / n

    print(f"[final] best fast objective = {best_val:.6f}")
    return best_seq.tolist()
