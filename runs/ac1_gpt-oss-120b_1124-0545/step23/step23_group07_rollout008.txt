def _lbfgs_opt(start_seq, beta_schedule, deadline, maxiter_per_beta=500):
    # start_seq sum=1
    log_a = np.log(start_seq + 1e-30)
    best_seq = start_seq.copy()
    best_val = surrogate(start_seq)
    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        obj_fun = SoftmaxObjective(len(start_seq), beta)
        res = scipy.optimize.minimize(obj_fun, log_a, method='L-BFGS-B',
                                      jac=True,
                                      options={'maxiter': maxiter_per_beta, 'disp': False})
        log_a = res.x
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()
        val = surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # keep log_a for next beta
    return best_seq, best_val

import time, math, numpy as np, scipy.optimize

# Convolution etc.

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _objective_surrogate(seq: np.ndarray) -> float:
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

class SoftmaxObjective:
    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, log_a: np.ndarray):
        """Return (objective, gradient) for L‑BFGS."""
        # stable soft‑max on the convolution
        # build probability vector
        a = np.exp(log_a - np.max(log_a))
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / len(a))
        else:
            a /= a_sum

        conv = _conv_fft(a, a)
        shift = np.max(conv)          # numeric stability
        beta = self.beta

        # soft‑max value  (log‑sum‑exp)/beta
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        softmax = shift + np.log(sum_exp) / beta

        grad_log = a * (grad_a - np.dot(grad_a, a))

        return obj, grad_log

def _lbfgs_opt(start_seq: np.ndarray,
               beta_schedule: list[float],
               deadline: float,
               maxiter_per_beta: int = 500) -> tuple[np.ndarray, float]:
    """Refine a probability vector using a sequence of smoothed‑max L‑BFGS runs."""
    n = len(start_seq)
    # start in log‑space (log‑softmax)
    log_a = np.log(start_seq + 1e-30)

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        obj_fun = SoftmaxObjective(n, beta)

        # SciPy respects time only through maxiter; we also break after each call.
        res = scipy.optimize.minimize(obj_fun,
                                      log_a,
                                      method='L-BFGS-B',
                                      jac=True,
                                      options={'maxiter': maxiter_per_beta,
                                               'disp': False})
        log_a = res.x

        # recover probability vector
        a = np.exp(log_a - np.max(log_a))
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

        # continue with the refined log_a for the next beta
    return best_seq, best_val

def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated gradient) using exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)
    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g

        # exponentiate and renormalise
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Swap‑based simulated annealing on the permutation of entries."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]
        val = _objective_surrogate(ns)

        if val < cur_val or rng.random() < math.exp(-(val - cur_val) / T):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

def _seed_power_shift_grid(n: int) -> list[np.ndarray]:
    """Fine‑grained shifted power‑law seeds."""
    seeds = []
    # exponents in a narrow range that theory predicts to be good
    for p in np.linspace(0.48, 0.62, 15):     # 15 values
        for delta in np.linspace(0.1, 3.0, 20):  # 20 values
            i = np.arange(n, dtype=np.float64)
            a = (i + delta) ** (-p)
            s = a.sum()
            if s > 0:
                a /= s
            else:
                a = np.full(n, 1.0 / n)
            seeds.append(a)
    return seeds

"""
search for a non‑negative step function that minimises evaluate_sequence

Algorithm
---------
1. Build a large pool of analytic seeds.
   * uniform, exponential, symmetric exponential, plain power‑law.
   * a *fine* grid of shifted power‑laws a_i ∝ (i+δ)^(-p) with
     p∈[0.48,0.62] and δ∈[0.1,3] – this family is known to give a flat
     self‑convolution.
   * random Dirichlet samples (various concentrations) and beta‑shaped
     families.
   * the previous best solution (if any) and a few permutations of it.

2. Evaluate all seeds with the cheap surrogate
        S(f)=2·n·max(conv(f,f))   (the ℓ₁‑sum of f is 1)
   and keep the few best ones.

3. Refine each top seed with a cascade of *smoothed‑max* L‑BFGS
   optimisations (β = 10, 50, 200 … 2·10⁵).  The objective is the
   log‑sum‑exp of the convolution, which is a smooth approximation of
   the max.  The gradient is obtained analytically, so each L‑BFGS step
   requires only two FFT convolutions.

4. Polish the result of the L‑BFGS run with an exact exponentiated‑gradient
   (mirror‑descent) that minimises the true max‑convolution.

5. Use a short permutation‑annealing tweak, then evaluate the final vector
   with the supplied ``evaluate_sequence`` and store it in the global
   variable ``height_sequence_1``.
"""

import time
import math
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# FFT convolution (real, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (seq is assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Helper for sorting – no lambda as required by the rules
# ----------------------------------------------------------------------
def _second(item):
    return item[1]


# ----------------------------------------------------------------------
# Smoothed‑max objective used by L‑BFGS
# ----------------------------------------------------------------------
class SoftmaxObjective:
    """
    f(log_a) = 2·n·softmax_beta(conv(a,a)),
    where a = softmax(log_a) and softmax_beta(z) = log_sum_exp(beta*z)/beta.
    The routine returns both value and gradient w.r.t. log_a.
    """

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, log_a: np.ndarray):
        # probability vector from log‑soft‑max
        a = np.exp(log_a - np.max(log_a))
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / self.n)
        else:
            a /= a_sum

        # convolution a*a
        conv = _conv_fft(a, a)
        shift = np.max(conv)                 # for numerical stability
        beta = self.beta

        # soft‑max (log‑sum‑exp) value
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        softmax = shift + np.log(sum_exp) / beta
        obj = 2.0 * self.n * softmax

        # gradient w.r.t. a
        w = exp_beta / sum_exp               # soft‑max weights (length 2n‑1)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)           # length = 3n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # gradient w.r.t. log‑probabilities (log‑softmax Jacobian)
        grad_log = a * (grad_a - np.dot(grad_a, a))

        return obj, grad_log


# ----------------------------------------------------------------------
# Optimiser based on a cascade of smoothed‑max L‑BFGS runs
# ----------------------------------------------------------------------
def _lbfgs_opt(start_seq: np.ndarray,
               beta_schedule: list[float],
               deadline: float,
               maxiter_per_beta: int = 300) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector using a sequence of smoothed‑max L‑BFGS
    optimisations (increasing β).  Returns the best vector found and its
    surrogate value.
    """
    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break

        obj_fun = SoftmaxObjective(n, beta)
        res = scipy.optimize.minimize(obj_fun,
                                      log_a,
                                      method='L-BFGS-B',
                                      jac=True,
                                      options={'maxiter': maxiter_per_beta,
                                               'disp': False})
        log_a = res.x

        a = np.exp(log_a - np.max(log_a))
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Works directly on the (non‑smoothed) objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)
    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active entry (active diagonal)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g

        # re‑exponentiate and renormalise (log‑softmax)
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Swap‑based annealing on the permutation of entries."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        accept = val < cur_val or (rng.random() < math.exp(-(val - cur_val) / T))
        if accept:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Fine grid of shifted power‑law seeds
# ----------------------------------------------------------------------
def _seed_power_shift_grid(n: int) -> list[np.ndarray]:
    """Generate a dense set of (i+δ)^{-p} vectors (ℓ₁‑normalised)."""
    seeds = []
    for p in np.linspace(0.48, 0.62, 15):          # 15 exponents
        for delta in np.linspace(0.1, 3.0, 20):    # 20 shifts
            i = np.arange(n, dtype=np.float64)
            a = (i + delta) ** (-p)
            s = a.sum()
            if s > 0:
                a /= s
            else:
                a = np.full(n, 1.0 / n)
            seeds.append(a)
    return seeds

# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    The best sequence found is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0          # safety margin

    n = 4000                                           # problem size

    # --------------------------------------------------------------
    # 0. Warm‑start with a previously stored solution (if any)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                total = arr.sum()
                if total > 0:
                    prev_vec = arr / total
        except Exception:
            prev_vec = None

    # --------------------------------------------------------------
    # 1. Assemble a rich pool of candidate seeds
    # --------------------------------------------------------------
    seeds: list[np.ndarray] = []

    # basic analytic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.001))
    seeds.append(_exp_decay(n, rate=0.003))
    seeds.append(_sym_exp_decay(n, rate=0.002))
    seeds.append(_power(n, p=0.55))
    seeds.append(_power(n, p=0.70))

    # fine shifted‑power‑law grid (the main source of quality)
    seeds.extend(_seed_power_shift_grid(n))

    # random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.2, 0.8, 2.0):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # beta‑shaped families (random parameters)
    for _ in range(6):
        a = rng.uniform(0.2, 2.5)
        b = rng.uniform(0.2, 2.5)
        seeds.append(_beta_shape(n, a, b))

    # previous solution and a few permutations of it
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(3):
            seeds.append(rng.permutation(prev_vec))

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds, keep the best few
    # --------------------------------------------------------------
    evaluated: list[tuple[np.ndarray, float]] = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=_second)
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed ranking (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Refine the top seeds (L‑BFGS cascade + exact EG polish)
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # smoothing parameters (grow geometrically)
    beta_schedule = [10, 50, 200, 800, 3200, 12800, 51200, 200000]

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget (≈30 % of what is left)
        remaining = deadline - time.time()
        slice_time = remaining * 0.30 / max(1, (top_k - idx))
        sub_deadline = time.time() + slice_time
        print(f"\n[Seed {idx+1}] time slice ≈ {slice_time:.1f}s")

        # --- L‑BFGS cascade on the smoothed max ---
        seq_lbfgs, val_lbfgs = _lbfgs_opt(seed_vec,
                                          beta_schedule,
                                          sub_deadline,
                                          maxiter_per_beta=300)

        # --- exact EG polishing (if time permits) ---
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs,
                                               sub_deadline,
                                               max_iters=1_200_000,
                                               eta0=6.0)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # pick the better local result
        if val_eg < val_lbfgs:
            local_seq, local_val = seq_eg, val_eg
        else:
            local_seq, local_val = seq_lbfgs, val_lbfgs

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final long EG polishing (remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=3_000_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Very short permutation annealing tweak
    # --------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + min(25.0, deadline - time.time())
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Evaluate with the provided function and store the answer
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
