def lp_step(a, M):
    n = len(a)
    # Build A matrix as csr: shape (2n-1, n)
    # For each row k, cols i where i in [max(0, k-(n-1)), min(k, n-1)]
    # A_{k,i} = a[i]
    # We'll use csr building with lists.
    import scipy.sparse as sp
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        for i in range(i_start, i_end+1):
            rows.append(k)
            cols.append(i)
            data.append(a[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # constraints: A @ b <= M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    c = -np.ones(n)  # linprog maximizes? Actually it minimizes c^T b, so set c = -1 to maximize sum(b)
    res = sp.optimize.linprog(c=c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if res.success:
        b = res.x
        # Normalize to sum 1 (if sum=0?) but we will rescale after mixing.
        return b
    else:
        # fallback perhaps just return a?
        return a

def exp_gradient(seq, n, deadline, iters=500000, eta0=5.0):
    a = seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, iters+1):
        if time.time() > deadline:
            break
        # compute gradient via subgradient of max conv
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - 1e-12 * max_val)[0]

        grad = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            grad[mask] += a[rev[mask]]

            grad = 2.0 * grad / active.size

        dot = np.dot(grad, a)
        grad_log = a * (grad - dot)
        eta = eta0 / math.sqrt(it)

        a = _softmax(log_a)

"""
Hybrid LP‑flattening optimiser
--------------------------------

We treat the problem as a linear programme:
    maximise   Σ b_i
    subject to  (a * b)_k ≤ M   for every convolution entry k,
                b_i ≥ 0,
where ``a`` is the current step‑function, ``M`` is its current
maximum convolution value, and ``*`` denotes discrete linear convolution.
The LP gives a vector ``b`` that spreads the mass of ``a`` as evenly
as possible while respecting the same peak‐height ``M``.
After normalising ``b`` we replace or mix it with ``a`` and repeat.
Because the constraint matrix is a Toeplitz matrix with constant
diagonals, it can be built very cheaply with ``scipy.sparse.diags``.
The whole routine respects the 1000 s budget, stores its best result in
the global variable ``height_sequence_1`` and finally returns that list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for the linear programme)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
except Exception:  # pragma: no cover
    _opt = None
    _sp = None

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (∑seq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Linear‑programming step: maximise Σb under (a*b)≤M
# ----------------------------------------------------------------------
def _lp_max_sum(a: np.ndarray, M: float, time_deadline: float) -> np.ndarray | None:
    """
    Solve the LP
        maximise   Σ b_i
        subject to a⋆b ≤ M,  b_i ≥ 0
    where ``a`` is a probability vector (sum = 1) and ``M`` is a scalar.
    Returns the optimal ``b`` (unnormalised) or ``None`` on failure / timeout.
    """
    if _opt is None or _sp is None:
        return None

    n = len(a)
    # Build the Toeplitz‑like constraint matrix A (size (2n‑1) × n)
    # each column i contains the constant a[i] on the diagonal offset i.
    diagonals = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A = _sp.diags(diagonals, offsets, shape=(2 * n - 1, n), format='csc')
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σb  ⇔  minimise -Σb
    c = -np.ones(n, dtype=np.float64)

    # give the solver a small time budget
    time_left = max(0.0, time_deadline - time.time())
    if time_left < 0.5:
        return None

    opts = {'presolve': True, 'time_limit': time_left}
    try:
        res = _opt.linprog(c=c, A_ub=A, b_ub=b_ub,
                           bounds=[(0, None)] * n,
                           method='highs', options=opts)
        if res.success:
            return res.x
        else:
            # silent failure – the caller will simply ignore this step
            print("[LP] solver failed:", res.message)
            return None
    except Exception as exc:  # pragma: no cover
        print("[LP] exception:", exc)
        return None


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s wall‑clock budget, stores the best found
    sequence in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0            # a couple seconds safety margin

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise with the previous best if available, otherwise uniform.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) == n:
        a = np.asarray(prev, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Keep track of the best sequence seen so far.
    # ------------------------------------------------------------------
    best_seq = a.copy()
    best_val = evaluate_sequence(best_seq.tolist())
    print(f"[Init] surrogate = {_objective_surrogate(best_seq):.6f}, true = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop – alternating LP flattening + mixing.
    # ------------------------------------------------------------------
    max_outer_iters = 12                     # comfortably fits into the budget
    outer_it = 0
    while time.time() < deadline and outer_it < max_outer_iters:
        outer_it += 1
        # current maximal convolution value
        conv = _conv_fft(a, a)
        M = float(conv.max())

        # -------- Linear programme -----------------------------------
        sub_deadline = deadline - 0.5                         # leave a tail for polishing
        b_raw = _lp_max_sum(a, M, sub_deadline)
        if b_raw is None:
            print(f"[Iter {outer_it}] LP step failed or timed out – terminating loop.")
            break
        if b_raw.sum() == 0.0:
            print(f"[Iter {outer_it}] LP returned zero vector – abort.")
            break
        b = b_raw / b_raw.sum()                # turn into a probability vector

        # -------- Evaluate a few mixing candidates --------------------
        candidates = [(a, _objective_surrogate(a))]
        mix_alphas = [0.3, 0.5, 0.7, 1.0]      # 1.0 means replace a by b completely
        for alp in mix_alphas:
            cand = (1.0 - alp) * a + alp * b
            cand = np.maximum(cand, 0.0)
            cand_sum = cand.sum()
            if cand_sum == 0.0:
                continue
            cand = cand / cand_sum
            cand_val = _objective_surrogate(cand)
            candidates.append((cand, cand_val))

        # pick the best surrogate among them
        seq_best, sur_best = min(candidates, key=lambda x: x[1])

        # --------‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‑‐
        if sur_best + 1e-12 < _objective_surrogate(best_seq):
            # we have a genuine improvement – update the global best
            best_seq = seq_best.copy()
            best_val = evaluate_sequence(best_seq.tolist())
            a = best_seq.copy()
            print(f"[Iter {outer_it}] NEW global best – surrogate {sur_best:.6f}, true {best_val:.6f}")
        else:
            # No improvement – add a tiny random perturbation to escape possible plateaus
            perturb = a + 0.001 * rng.normal(size=n)
            perturb = np.maximum(perturb, 0.0)
            p_sum = perturb.sum()
            if p_sum > 0.0:
                a = perturb / p_sum
            print(f"[Iter {outer_it}] no improvement, random perturbation applied.")

    # ------------------------------------------------------------------
    # Final polishing: a short exponentiated‑gradient descent on the exact
    # max‑convolution (using the same surrogate as a cheap proxy).
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("\n--- Final exponentiated‑gradient polishing ---")
        # a very light EG (few thousand iterations – cheap)
        log_a = np.log(a + 1e-30)
        best_seq_local = a.copy()
        best_sur_local = _objective_surrogate(a)
        iters = 15_000
        for it in range(1, iters + 1):
            if time.time() > deadline - 1.0:
                break
            conv = _conv_fft(a, a)
            max_val = conv.max()
            active = np.where(conv >= max_val - 1e-12 * max_val)[0]

            # sub‑gradient of the max‑convolution
            g = np.zeros_like(a)
            idx = np.arange(n, dtype=np.int64)
            for k in active:
                rev = k - idx
                mask = (rev >= 0) & (rev < n)
                g[mask] += a[rev[mask]]
            if active.size:
                g = 2.0 * g / active.size

            # mirror‑descent step (exponentiated gradient)
            dot = np.dot(g, a)
            grad_log = a * (g - dot)
            eta = 8.0 / math.sqrt(it)
            log_a -= eta * grad_log
            a = _softmax(log_a)

            cur_sur = _objective_surrogate(a)
            if cur_sur < best_sur_local:
                best_sur_local = cur_sur
                best_seq_local = a.copy()
        # update global best if polishing helped
        if best_sur_local + 1e-12 < _objective_surrogate(best_seq):
            best_seq = best_seq_local.copy()
            best_val = evaluate_sequence(best_seq.tolist())
            print(f"[Polish] improved to surrogate {best_sur_local:.6f}, true {best_val:.6f}")

    # ------------------------------------------------------------------
    # Return the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised", exc)

    print(f"\n=== Finished ===\nBest surrogate ≈ {_objective_surrogate(best_seq):.6f}, "
          f"true evaluate_sequence = {true_val:.6f}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
