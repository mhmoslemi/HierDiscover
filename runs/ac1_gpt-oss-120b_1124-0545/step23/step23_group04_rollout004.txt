"""
A multi‑start, multi‑phase optimiser for the “max‑autocorrelation” problem.

The pipeline for each seed consists of:

0️⃣  Normalisation (sum = 1) and cheap surrogate evaluation.
1️⃣  Adam optimiser on a smooth log‑sum‑exp approximation of the max‑convolution.
    The smoothness parameter β is increased geometrically; a generous
    number of inner steps is performed for the later (large) β values.
2️⃣  Exponentiated‑gradient (mirror‑descent) on the *exact* max‑convolution
    using the sub‑gradient at the active index.  The step size decays as
    η₀/√t and the loop runs until the allocated time slice is exhausted.
3️⃣  An “active‑set LP” step – we maximise Σbⱼ under the linear constraints
    (a∗b)(k) ≤ M for the currently active k‑indices (those where the
    autocorrelation of a attains its maximum M).  The optimiser returns a
    candidate b which is mixed with a (weight t) and kept if it improves
    the surrogate.
4️⃣  A tiny stochastic walk on the simplex (random local search) for
    polishing.

The whole routine is executed on the *top‑k* seeds from a diverse pool
(uniform, arcsine, triangular, half‑sine, several Beta‑shapes,
Dirichlet draws and the previously stored solution).  Time is split
between the seeds, but each stage respects a global deadline, so the
function always returns before the 1000 s budget expires.

The final sequence is stored in the global variable ``height_sequence_1``
and returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
    2·n·max(conv(seq, seq)).  The input is assumed to sum to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _half_sine(n: int) -> np.ndarray:
    """Half‑sine shape: a_i ∝ sin(π·(i+0.5)/n).  Positive and smooth."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = a / a.sum()
    return a


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the discrete set {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _adam_smoothing(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth log‑sum‑exp surrogate of the max‑convolution.
    Returns the best probability vector seen and its exact surrogate value.
    """
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)                # initialise logits
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # Geometric schedule for the smoothness parameter β
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0,
        20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # more inner steps for the larger β values (they dominate the final quality)
        max_iter_beta = 12000 if beta >= 1_000_000 else 8000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                  # current point on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)    # outer factor 2·n

            # convert to gradient in the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.12
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # periodic exact evaluation
            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray, deadline: float,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Exact exponentiated‑gradient (mirror descent) on the true max‑convolution.
    Returns best sequence found and its surrogate.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient at the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % 4000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


def _lp_active_step(a: np.ndarray, M: float,
                    active_idxs: np.ndarray,
                    t: float = 0.15) -> np.ndarray | None:
    """
    Solve a small LP that maximises Σbⱼ under the constraints
        (a ∗ b)(k) ≤ M   for each active k.
    Returns a new candidate a′ = (1‑t)·a + t·b_norm  (already renormalised)
    or None if the LP fails.
    """
    n = a.shape[0]
    rows = active_idxs.size
    if rows == 0:
        return None

    A = np.zeros((rows, n), dtype=np.float64)
    for r, k in enumerate(active_idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        A[r, idx] = a[k - idx]

    b_ub = np.full(rows, M, dtype=np.float64)

    # maximise Σ b_j  <=> minimise -Σ b_j
    c = -np.ones(n, dtype=np.float64)

    try:
        res = linprog(c,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
    except Exception:   # pragma: no cover
        return None

    if not res.success:
        return None

    b = res.x
    s = b.sum()
    if s <= 0.0:
        return None

    b_norm = b / s
    a_new = (1.0 - t) * a + t * b_norm
    a_new = a_new / a_new.sum()
    return a_new


def _local_random_search(start_seq: np.ndarray, deadline: float,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic hill‑climbing on the simplex.
    Each move transfers a small random amount of mass from i to j.
    """
    n = start_seq.size
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # a tiny fraction of the source mass
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found within the allotted budget is stored in the global
    variable ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Load a previously stored solution (if any) and normalise it
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            pass

    # --------------------------------------------------------------
    # Build a diverse seed pool
    # --------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
        _half_sine(n),
    ]

    # symmetric Beta(α,α) for several α
    for α in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        seeds.append(_beta_distribution(n, α))

    # Dirichlet draws with a few concentration parameters
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 0️⃣  Initial surrogate evaluation of the seeds
    # --------------------------------------------------------------
    seed_infos = []
    print("\n--- Initial seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    # Keep a global best across all seeds
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Optimisation loop over the selected seeds
    # --------------------------------------------------------------
    for seed_idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n=== Optimising seed #{seed_idx + 1}/{len(top_seeds)} ===")
        remaining = deadline - time.time()
        if remaining < 30.0:            # not enough time for a full run
            break

        # ---- Adam smoothing (≈20 % of remaining time) ----
        adam_deadline = time.time() + remaining * 0.20
        seq_adam, val_adam = _adam_smoothing(seed_vec, adam_deadline)
        print(f"[Adam] surrogate = {val_adam:.6f}")
        if val_adam < global_best_val:
            global_best_val, global_best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

        # ---- Exponentiated‑gradient polishing (≈55 % of remaining time) ----
        if time.time() < deadline:
            eg_deadline = time.time() + (deadline - time.time()) * 0.55
            seq_eg, val_eg = _exp_gradient_opt(global_best_seq, eg_deadline,
                                               eta0=7.0)   # a slightly larger start step
            print(f"[EG] surrogate = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_val, global_best_seq = val_eg, seq_eg.copy()
                print("   >>> Updated global best (EG)")

        # ---- Active‑set LP step (up to three tries) ----
        lp_attempts = 3
        for attempt in range(lp_attempts):
            if time.time() > deadline:
                break
            conv = _conv_fft(global_best_seq, global_best_seq)
            M = np.max(conv)
            # indices where the autocorrelation is within 0.1 % of the max
            tol = max(1e-9, M * 1e-3)
            active = np.where(conv >= M - tol)[0]
            if active.size == 0:
                break
            # try a few different mixing weights
            for t_mix in (0.15, 0.08, 0.25):
                cand = _lp_active_step(global_best_seq, M, active, t=t_mix)
                if cand is None:
                    continue
                cand_val = _objective_surrogate(cand)
                if cand_val < global_best_val:
                    global_best_val, global_best_seq = cand_val, cand.copy()
                    print(f"[LP] improvement to {cand_val:.6f} with t={t_mix}")
                    # after a successful LP step we break to recompute active set
                    break
            else:
                # no improvement for any t_mix
                continue
            # if we improved, recompute active set on the new point
            # and continue the LP inner loop
        # ---- Final random local refinement (spend the rest of the time) ----
        if time.time() < deadline:
            print("\n--- Final random local refinement ---")
            loc_deadline = deadline
            seq_loc, val_loc = _local_random_search(global_best_seq, loc_deadline,
                                                    sigma=0.012)
            print(f"[Local] surrogate = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_val, global_best_seq = val_loc, seq_loc.copy()
                print("   >>> Updated global best (local)")

        # summary after this seed
        print(f"[Seed #{seed_idx + 1}] best surrogate so far = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Final official evaluation
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next call
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
