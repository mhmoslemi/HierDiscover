   * If a previous solution exists (`height_sequence_1`) it is added
     and also a few of its random permutations.

   Every candidate is ℓ₁‑normalised so that the cheap surrogate
   `2·n·max(conv(a,a))` equals the true ``evaluate_sequence`` value.

2. **Select the best seed**
   Surrogate values are computed for all seeds; the one with the
   smallest surrogate becomes the starting point.

3. **Continuous refinement**
   * **Adam on a smooth log‑sum‑exp approximation** – a schedule of
     increasing “β’’ turns the smooth surrogate into the true max.
   * **Exact exponentiated‑gradient (mirror‑descent)** – works directly
     on the max‑convolution with a diminishing stepsize.
   Both phases are given a fixed share of the remaining budget.

4. **Stochastic fine‑tuning**
   A long simulated‑annealing phase moves mass between coordinates
   (and occasionally swaps two positions).  The temperature is lowered
   linearly from the current surrogate value to a tiny fraction of it.
   The current sequence is always kept feasible (non‑negative, ℓ₁‑sum = 1).

5. **Report**
   The best sequence found is stored in the global variable
   ``height_sequence_1`` and returned as a plain list.  The final
   objective is printed (both surrogate and the true ``evaluate_sequence``).

The whole routine respects the supplied time budget (default 1000 s)
and works with a single thread.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT convolution (real, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (seq is assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^{-p}, i = 0…n‑1, δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine density a_i ∝ 1 / sqrt((i+0.5)*(n-i-0.5))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‐conv."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑probabilities (soft‑max parametrisation)
    log_a = np.log(a + 1e-30)

    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active diagonal
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)          # diminishing stepsize
        log_a -= eta * g                     # mirror step
        a = np.exp(log_a - np.max(log_a))    # log‑softmax
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 20_000) -> tuple[np.ndarray, float]:
    """Adam optimisation of a smooth (log‑sum‑exp) surrogate of max‑conv."""
    if beta_schedule is None:
        beta_schedule = [10, 50, 200, 800, 3_200, 12_800,
                         51_200, 2 * 51_200, 8 * 51_200]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1
            # probabilities (soft‑max)
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            # convolution a*a
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp               # soft‑max weights

            # gradient wrt a (chain rule)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient wrt log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            # keep logs bounded (avoid overflow / underflow)
            log_a = np.clip(log_a, -30.0, 30.0)

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple SA that transfers a tiny amount of mass between two coordinates
    (or swaps two coordinates).  Works directly on the exact surrogate.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-5
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # linear temperature schedule
        frac = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - (deadline - time.time())))
        # simpler: use iteration fraction
        frac = it / max_iters
        T = T_start + (T_end - T_start) * frac

        if rng.random() < 0.1:               # occasional swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:
            # transfer a small amount of mass from i to j
            i = rng.integers(0, n)
            if cur_seq[i] <= 1e-14:
                continue
            max_move = cur_seq[i] * 0.5
            delta = rng.random() * max_move
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            ns = cur_seq.copy()
            ns[i] -= delta
            ns[j] += delta

        val = _objective_surrogate(ns)
        if val < cur_val or rng.random() < math.exp(-(val - cur_val) / max(T, 1e-20)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

        # tiny renormalisation every few thousand steps (numerical drift)
        if it % 5000 == 0:
            s = cur_seq.sum()
            if s > 0:
                cur_seq /= s
            s = best_seq.sum()
            if s > 0:
                best_seq /= s
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Load previous solution if available (helps continuation)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                s = prev_vec.sum()
                if s > 0:
                    prev_vec = prev_vec / s
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build an extensive pool of analytic and random seeds
    # ------------------------------------------------------------------
    seeds = []

    # basic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.0015))
    seeds.append(_exp_decay(n, rate=0.0035))
    seeds.append(_sym_exp_decay(n, rate=0.0025))
    seeds.append(_power(n, p=0.7))
    seeds.append(_power(n, p=1.2))
    seeds.append(_arcsine_shape(n))

    # shifted power‑law – fine grid
    p_grid = np.linspace(0.35, 0.80, 19)            # 0.35 … 0.80 step≈0.025
    delta_grid = np.logspace(-2, 1, 14)            # 0.01 … 10
    for p in p_grid:
        for delta in delta_grid:
            seeds.append(_power_shift(n, p, delta))

    # beta‑shape – moderate grid
    alpha_grid = np.linspace(0.1, 3.0, 13)
    beta_grid = np.linspace(0.1, 3.0, 13)
    for a in alpha_grid:
        for b in beta_grid:
            seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws with various concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution and a few random permutations of it
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(5):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation, keep the best seed
    # ------------------------------------------------------------------
    best_seed = None
    best_seed_val = float('inf')
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        if val < best_seed_val:
            best_seed_val = val
            best_seed = vec
    print(f"\n[Seed selection] best surrogate = {best_seed_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuous refinement (Adam → EG)
    # ------------------------------------------------------------------
    remaining_time = deadline - time.time()
    if remaining_time > 60.0:   # allocate a reasonable slice to Adam
        adam_deadline = time.time() + remaining_time * 0.30
        cand_adam, val_adam = _adam_smooth_opt(best_seed, adam_deadline,
                                               iters_per_beta=25_000)
        print(f"[Adam] surrogate after smoothing = {val_adam:.6f}")
    else:
        cand_adam, val_adam = best_seed, best_seed_val

    remaining_time = deadline - time.time()
    if remaining_time > 30.0:
        eg_deadline = time.time() + remaining_time * 0.45
        cand_eg, val_eg = _exp_gradient_opt(cand_adam, eg_deadline,
                                            max_iters=2_500_000, eta0=8.0)
        print(f"[EG] surrogate after exact optimisation = {val_eg:.6f}")
    else:
        cand_eg, val_eg = cand_adam, val_adam

    # Choose the better of the two refinements as the starting point
    if val_adam < val_eg:
        cur_seq = cand_adam
        cur_val = val_adam
    else:
        cur_seq = cand_eg
        cur_val = val_eg

    best_seq = cur_seq.copy()
    best_val = cur_val
    print(f"[Start of SA] surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_seq, sa_val = _simulated_annealing(best_seq, deadline, max_iters=800_000)
        if sa_val < best_val:
            best_seq, best_val = sa_seq, sa_val
            print(f"[SA] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
