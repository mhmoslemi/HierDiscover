    rng = np.random.default_rng(seed)

    start_time = time.time()

    deadline = start_time + budget_s - 2.0   # safety

    n = int(kwargs.get("n", 4000))

    # Load prev sequence:

    prev_seq = globals().get("height_sequence_1")
    if prev_seq is not None and len(prev_seq) == n:
        prev_vec = np.asarray(prev_seq, dtype=float)
        if prev_vec.sum() > 0.0:
            prev_vec = prev_vec / prev_vec.sum()
        else:
            prev_vec = None
    else:
        prev_vec = None

    # Precompute Beta distributions for a list of alphas.
    beta_alphas = [...]
    beta_vecs = {}
    for a in beta_alphas:
        beta_vecs[a] = _beta_distribution(n, a)

    # Build seed list
    seeds = []
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_arcsine(n))
    # Dirichlet seeds
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        seeds.append(_random_dirichlet(n, conc, rng))

    # Add Beta distributions
    for a in beta_alphas:
        seeds.append(beta_vecs[a])

    # Add mixture seeds: for each pair (a,b) and λ in [0.1,0.3,0.5,0.7,0.9]
    mix_weights = np.linspace(0.1,0.9,9)
    for i, a1 in enumerate(beta_alphas):
        for a2 in beta_alphas[i:]:
            v1 = beta_vecs[a1]
            v2 = beta_vecs[a2]
            for lam in mix_weights:
                mix = lam * v1 + (1 - lam) * v2
                seeds.append(mix)

    # Add random permutations of some seeds to break symmetry:
    for base in (beta_vecs[0.5], beta_vecs[0.2], _uniform(n)):
        perm = rng.permutation(base)
        seeds.append(perm)

    # Add previous vector if any
    if prev_vec is not None:
        seeds.append(prev_vec)

    # Keep best seed by surrogate (max conv)
    best_seq = None
    best_M = np.inf
    best_val = np.inf

    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        # Ensure non-negative, sum 1.
        vec = np.maximum(vec, 0.0)
        total = vec.sum()
        if total == 0:
            continue
        a = vec / total
        M = _max_conv(a)   # compute max conv
        val = 2.0 * n * M
        if val < best_val:

    # We'll attempt full optimization as earlier.

    # Phase A: Adam smoothing with longer schedule
    if best_seq is not None and time.time() < deadline:

        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
Non‑negative step‑function optimiser – enhanced seed generation & multi‑phase
search.

The optimiser builds a large, diverse pool of candidate vectors:
* simple analytical families (uniform, triangular, arcsine)
* Dirichlet draws with many concentration parameters
* Symmetric Beta(α,α) densities for a fine grid of α
* Linear mixtures of two Beta densities (including a fine λ‑grid) and
  shifted‑Beta variants (to break symmetry).

All candidates are normalised to a probability vector (sum = 1).  The
best seed (smallest max‑convolution) is then refined in three timed phases:

1. **Adam on a smooth‑max surrogate** – we increase the smoothness
   parameter β progressively (up to 1e8) and run many inner Adam iterations.
2. **Exact exponentiated‑gradient (mirror‑descent)** on the true max‑convex
   objective, with a larger initial step size (η₀ = 10) and √t decay.
3. **Greedy stochastic local refinement** – small random mass transfers
   (σ ≈ 0.01) are accepted only if they improve the exact objective.

The whole procedure respects the global 1000 s deadline, keeps the best
solution in the global variable ``height_sequence_1`` and returns it as a
plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, softmax, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution by real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _max_conv(seq: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of ``seq``."""
    return float(np.max(_conv_fft(seq, seq)))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on the centre of each bin.
    ``alpha`` may be any positive real number.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of a smooth‑max approximation
    of 2·n·max(conv(a,a)).
    Returns the best probability vector seen and its **exact** surrogate value.
    """
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a schedule that finally reaches a very sharp smooth‑max
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0,
        50_000_000.0, 100_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = 2.0 * n * _max_conv(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # inner budget: more iterations when we still have time
        max_iter_beta = 12000 if (deadline - time.time()) > 30 else 4000

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient d/d a of Σ w·conv
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                        # length = 2n‑1 + n‑1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]       # slice of length n
            grad_a = grad_weighted * (2.0 * n)                # outer factor 2·n

            # mirror‑gradient (log‑softmax space)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 1000 == 0:
                cur_val = 2.0 * n * _max_conv(_softmax(logit))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(logit).copy()

    final_a = _softmax(logit)
    final_val = 2.0 * n * _max_conv(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient on true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Uses the sub‑gradient of the currently active convolution entry.
    Returns the best vector seen and its exact surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = 2.0 * n * _max_conv(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = 2.0 * n * _max_conv(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 100_000,
                         sigma: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic walk on the simplex:
    repeatedly move a small random amount from one coordinate to another.
    Only strictly improving moves are kept.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = 2.0 * n * _max_conv(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = 2.0 * n * _max_conv(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm respects the supplied time budget (≈1000 s) and stores the
    best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load a previous solution (if any) as a seed
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    if prev_seq is not None and len(prev_seq) == n:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
            else:
                prev_vec = None
        except Exception:
            prev_vec = None
    else:
        prev_vec = None

    # ------------------------------------------------------------------
    # Pre‑compute a family of Beta distributions (dense α‑grid)
    # ------------------------------------------------------------------
    beta_alphas = [
        0.02, 0.03, 0.04, 0.05, 0.07, 0.10, 0.15, 0.20,
        0.25, 0.30, 0.35, 0.40, 0.45, 0.50, 0.60,
        0.80, 1.00, 1.20, 1.50, 2.00, 3.00
    ]
    beta_dict = {a: _beta_distribution(n, a) for a in beta_alphas}

    # ------------------------------------------------------------------
    # Build a diverse seed pool (all vectors are normalised to sum = 1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _triangular(n),
        _arcsine(n),
    ]

    # Dirichlet draws
    for conc in (0.05, 0.10, 0.20, 0.50, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Pure Beta families
    for a in beta_alphas:
        seeds.append(beta_dict[a])

    # Linear mixtures of two Betas (coarse λ‑grid)
    lam_grid = np.linspace(0.1, 0.9, 9)
    for i, a1 in enumerate(beta_alphas):
        v1 = beta_dict[a1]
        for a2 in beta_alphas[i:]:
            v2 = beta_dict[a2]
            for lam in lam_grid:
                mix = lam * v1 + (1.0 - lam) * v2
                seeds.append(mix)

    # Shifted‑Beta variants (break symmetry)
    shifts = [n // 4, n // 2]
    for a in (0.50, 0.20, 0.10):
        base = beta_dict[a]
        for s in shifts:
            shifted = np.roll(base, s)
            seeds.append(shifted)
            # also mix shifted & original
            mix = 0.5 * base + 0.5 * shifted
            seeds.append(mix)

    # Include the previous solution if available
    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 0️⃣ Initial evaluation of all seeds – keep the best (exact objective)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    best_M = np.inf

    print("\n--- Seed evaluation (exact max‑conv) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        total = vec.sum()
        if total == 0.0:
            continue
        a = vec / total
        M = _max_conv(a)
        val = 2.0 * n * M
        if val < best_val:
            best_val, best_seq, best_M = val, a.copy(), M
        if (idx + 1) % 50 == 0:
            print(f"  evaluated {idx + 1}/{len(seeds)} seeds – best = {best_val:.6f}")

    print(f"[Init] best exact surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Adam smoothing on the current best seed
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.35 * remaining
        print("\n--- Adam smoothing (smooth‑max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq, best_val = seq_adam, val_adam
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 2️⃣ Exact exponentiated‑gradient polishing (sub‑gradient on true max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        polishing_deadline = deadline - 3.0   # keep a few seconds for the final phase
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, polishing_deadline, eta0=10.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg, val_eg
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 3️⃣ Final greedy random local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        time_left = deadline - time.time()
        # Rough estimate: one attempt ≈ 5e‑4 s
        approx_attempts = int(time_left / 5e-4)
        approx_attempts = max(60_000, min(400_000, approx_attempts))
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, n, deadline,
                                                attempts=approx_attempts,
                                                sigma=0.01)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc, val_loc
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 4️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
