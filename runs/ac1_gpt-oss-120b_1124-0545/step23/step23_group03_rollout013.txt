def _project_simplex(v):
    # from https://doi.org/10.1137/070703136 etc.
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho = np.nonzero(u - css / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    # returns projection onto simplex {x >=0, sum=1}
    # from article: https://doi.org/10.1016/j.cie.2015.11.049
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    # find rho: largest i s.t. u_i > css_i / (i+1)
    rho = np.where(u - css / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _simulated_annealing(start_seq, n, deadline, init_temp_factor=0.02, sigma=0.02):
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    # init temperature
    temp = init_temp_factor * cur_val
    # decays
    cooling = 0.9999  # maybe depends on attempts
    while time.time() < deadline:
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * cur_seq[i] * rng.random()
        if delta <= 0:
            continue

import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _objective(seq):
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _project_simplex(v):
    # fast projection onto probability simplex
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1
    rho_arr = u - css / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = css[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n):
    return np.full(n, 1.0/n, dtype=np.float64)

def _triangular(n):
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half+1)
        right = left[::-1]
    else:
        left = np.arange(1, half+2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a

def _arcsine(n):
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0/np.sqrt((i+0.5)*(n-0.5-i)+eps)
    a /= a.sum()
    return a

def _random_dirichlet(n, conc, rng):
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _beta_distribution(n, alpha):
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha-1.0) * np.power(1.0 - x, alpha-1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        return _uniform(n)
    a /= total
    return a

def _subgradient_descent(start_seq, n, deadline, step0=0.3):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    cur_val = best_val
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        # compute subgradient
        idx = np.arange(n)
        rev_idx = k - idx
        mask = (rev_idx >= 0) & (rev_idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev_idx[mask]]
        eta = step0 / math.sqrt(it)
        a_tmp = a - eta * g
        a = _project_simplex(a_tmp)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _simulated_annealing(start_seq, n, deadline, init_temp_factor=0.02, sigma=0.02):
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    temp = init_temp_factor * cur_val  # initial temperature proportional to obj
    cooling = 0.99995
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * cur_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        new_val = _objective(new_seq)
        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_val - cur_val) / (temp + 1e-12))

            cur_seq = new_seq
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        temp = max(temp * cooling, 1e-12)
    return best_seq, best_val

"""
Optimiser for the step‑function bound.

The algorithm builds a diverse pool of analytic seeds (uniform, triangular,
arcsine, Dirichlet draws and symmetric Beta‑densities).  The best seed is
refined in two stages:

1.  Projected sub‑gradient descent on the *exact* max‑convolution.
    The iteration moves opposite to the sub‑gradient of the current
    maximal convolution entry and projects back onto the probability simplex.
    A small random jitter is applied before the descent to escape the basin
    of the previous solution.

2.  Simulated‑annealing walk: repeatedly transfers a tiny random amount of
    mass from one coordinate to another and accepts worsening moves with a
    temperature that decays geometrically.

Both stages respect a global time budget (≈ 1000 s).  The final sequence is
returned as a list of non‑negative heights and stored in the global variable
``height_sequence_1`` for possible reuse in later runs.
"""

import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution by real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a sequence that already satisfies sum(seq)=1:
        2·n·max(conv(seq,seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x)=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    sv = np.cumsum(u) - 1.0
    rho_arr = u - sv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = sv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on the centre of each bin.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         step0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Euclidean projected sub‑gradient descent on
        f(a) = 2·n·max(conv(a,a))
    with a simplex constraint.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        idx = np.arange(n)
        rev_idx = k - idx
        mask = (rev_idx >= 0) & (rev_idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[rev_idx[mask]]

        eta = step0 / math.sqrt(it)
        a_tmp = a - eta * g
        a = _project_simplex(a_tmp)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine search
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         init_temp_factor: float = 0.02,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic walk with a temperature schedule.
    Proposes a small random mass transfer between two coordinates;
    accepts worsening moves with probability exp(-(Δ)/T).
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    temp = init_temp_factor * cur_val
    cooling = 0.99995

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * cur_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:   # safety
            continue
        new_val = _objective(new_seq)

        if new_val < cur_val:
            accept = True
        else:
            accept = rng.random() < math.exp(-(new_val - cur_val) / (temp + 1e-12))

        if accept:
            cur_seq = new_seq
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

        temp = max(temp * cooling, 1e-12)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the supplied time budget (≈1000 s) and stores the
    best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 1.0

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣  Load a previously stored solution as a possible seed
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣  Build a diverse pool of seeds (all normalised to sum‑1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _triangular(n),
        _arcsine(n),
    ]

    # Dirichlet draws with several concentration parameters
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta distributions – a richer family
    beta_alphas = [
        0.05, 0.07, 0.10, 0.15, 0.20, 0.25, 0.30,
        0.35, 0.40, 0.45, 0.50, 0.60, 0.80, 1.00,
        1.20, 1.50, 2.00
    ]
    for a in beta_alphas:
        seeds.append(_beta_distribution(n, a))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣  Initial evaluation – keep the best seed
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective(vec)
        print(f" seed #{idx + 1:02d}: objective = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best objective = {best_val:.6f}")

    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)

    # ------------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient descent (major chunk of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        sub_deadline = time.time() + 0.6 * remaining
        print("\n--- Projected sub‑gradient descent ---")

        # small random jitter before descent to leave the current basin
        jitter = rng.normal(scale=0.001, size=n)
        start_for_sub = best_seq + jitter
        start_for_sub = np.maximum(start_for_sub, 0.0)
        if start_for_sub.sum() == 0.0:
            start_for_sub = _uniform(n)
        else:
            start_for_sub /= start_for_sub.sum()

        seq_sub, val_sub = _subgradient_descent(start_for_sub, n,
                                                sub_deadline, step0=0.3)
        print(f"[SubGrad] best objective = {val_sub:.6f}")
        if val_sub < best_val:
            best_val, best_seq = val_sub, seq_sub.copy()
            print("   >>> Updated global best (SubGrad)")

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing fine refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                             init_temp_factor=0.02,
                                             sigma=0.02)
        print(f"[SA] best objective = {val_sa:.6f}")
        if val_sa < best_val:
            best_val, best_seq = val_sa, seq_sa.copy()
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 5️⃣  Final evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nFinal surrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
