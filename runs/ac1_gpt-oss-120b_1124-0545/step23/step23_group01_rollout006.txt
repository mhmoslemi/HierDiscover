def _smooth_adam_opt(start_seq, n, deadline, schedule, iters_per_beta=5000, lr0=0.5, beta1=0.9, beta2=0.999, eps=1e-8, eval_interval=200):

   log_a = np.log(start_seq + 1e-30)
   m = np.zeros_like(log_a)
   v = np.zeros_like(log_a)
   best_seq = _softmax(log_a)
   best_val = _objective_surrogate(best_seq)
   t = 0
   for phase_idx, beta in enumerate(schedule):
        # compute allocation based on remaining_time
        # we want total iteration per phase maybe fixed but also adapt:
        # Suppose time left = deadline - time.time()
        # Est iteration time per loop ~ 0.0012, but we don't have accurate.
        # We'll just set max_iters = iters_per_beta
        max_iters = iters_per_beta
        for iter_idx in range(1, max_iters+1):
            if time.time() > deadline:
                break
            a = _softmax(log_a)
            conv = _conv_fft(a, a)  # len L
            # compute w
            shifted = conv - np.max(conv)  # stable
            exp_term = np.exp(beta * shifted)
            w = exp_term / np.sum(exp_term)  # shape L

            # compute weighted gradient via correlation:
            rev_w = w[::-1]
            weighted = _conv_fft(a, rev_w)   # full conv length n+L-1
            L = len(conv)
            g_a = 2.0 * weighted[L - 1 - np.arange(n)]

            # chain rule
            dot = np.dot(g_a, a)
            grad_v = a * (g_a - dot)

            # Adam
            t += 1
            m = beta1 * m + (1 - beta1) * grad_v
            v = beta2 * v + (1 - beta2) * (grad_v ** 2)

            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
            log_a -= step

            # periodic evaluation
            if iter_idx % eval_interval == 0:
                cur_seq = _softmax(log_a)
                cur_val = _objective_surrogate(cur_seq)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_seq.copy()
        if time.time() > deadline:
            break

   return best_seq, best_val

"""
A step‑function optimizer for ``evaluate_sequence``.

The idea
--------
* Work on the simplex by parameterising the vector with a log‑softmax
  (mirror‑descent on the probability simplex).
* Use a *smooth* surrogate of the max‑convolution:
      ϕ_β(a) = 2·n·log‑sum‑exp(β·conv(a,a))/β .
  For β→∞ the surrogate → the true objective.
  The gradient is a weighted sum of the individual convolution‑gradients,
  which can be obtained efficiently with one extra FFT‑convolution.
* Run a multi‑phase Adam optimisation, starting with a small β (very
  smooth) and gradually increasing it (continuation).  This helps to
  escape local minima that trap a pure max‑gradient method.
* After the smoothing phase finish with the original max‑gradient Adam
  and a long exponentiated‑gradient polishing pass (as in the reference
  code).  Finally perform a cheap random local search.
* The routine respects a 1000 s wall‑clock budget, prints progress,
  stores the best vector in ``height_sequence_1`` and returns it as a
  plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective used for comparison: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _triangular(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _sin_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Original (max‑gradient) Adam optimiser (kept for final polishing)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 500_000,
              lr0: float = 0.2,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    t = 0
    for _ in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        a = _softmax(log_a)

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g_a = np.zeros_like(a)
        g_a[idx_i] = 2.0 * a[rev_idx]

        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        if _ % 200 == 0:
            cur_val = _objective_surrogate(_softmax(log_a))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (identical to reference)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (continuation on β)
# ----------------------------------------------------------------------
def _smooth_adam_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     schedule: list[float],
                     iters_per_phase: list[int],
                     lr0: float = 0.5,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8,
                     eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed surrogate
          Φ_β(a) = 2·n·log‑sum‑exp(β·conv)/β .
    A continuation schedule over β is used.
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    t = 0
    for beta, max_iter in zip(schedule, iters_per_phase):
        for it in range(1, max_iter + 1):
            if time.time() > deadline:
                break

            a = _softmax(log_a)

            conv = _conv_fft(a, a)                 # length L = 2n‑1
            # soft‑max weights over convolution entries
            shift = conv - np.max(conv)            # stability
            exp_term = np.exp(beta * shift)
            w = exp_term / np.sum(exp_term)        # shape L

            # weighted gradient = 2 * correlation(a , w)
            rev_w = w[::-1]
            weighted = _conv_fft(a, rev_w)         # length n+L‑1
            L = len(conv)
            g_a = 2.0 * weighted[L - 1 - np.arange(n)]

            # chain rule to log‑softmax coordinates
            dot = np.dot(g_a, a)
            grad_v = a * (g_a - dot)

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
            log_a -= step

            if it % eval_interval == 0:
                cur_seq = _softmax(log_a)
                cur_val = _objective_surrogate(cur_seq)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = cur_seq.copy()

        if time.time() > deadline:
            break

    return best_seq, best_val

    """

    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    n = 4000

    # ------------------------------------------------------------------
    # Build a diversified pool of start vectors
    # ------------------------------------------------------------------
    candidates = []

    # 1) previous best if available
    prev = globals().get("height_sequence_1")

        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                candidates.append(arr / arr.sum())
        except Exception:
            pass

    # 2) exponential decays (front‑heavy)
    for rate in (0.0003, 0.0005, 0.001, 0.002, 0.005):
        candidates.append(_exp_decay(n, rate))

    # 3) reversed exponential (back‑heavy)
    for rate in (0.0005, 0.001):

    # 5) smooth sine‑shape

    # 6) uniform distribution
    candidates.append(_uniform(n))

    # 7) a few Dirichlet draws with different concentration
    for conc in (0.3, 0.7, 2.0):
        cand = rng.dirichlet(alpha=np.full(n, conc))
        candidates.append(cand)

    if not candidates:                                 # sanity fallback
        candidates.append(_exp_decay(n, 0.001))

    # ------------------------------------------------------------------
    # Optimisation parameters (continuation on β)
    # ------------------------------------------------------------------
    schedule_beta = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]
    # baseline iteration budget for each β‑phase (will be scaled)

    best_overall_seq = None

        # scale phase length according to remaining time

            eval_interval=200)

        print(f"  → surrogate after smoothing = {val:.6f}")

        if val < best_overall_val:
            best_overall_val = val

            print(f"  ** new global best (surrogate) = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
