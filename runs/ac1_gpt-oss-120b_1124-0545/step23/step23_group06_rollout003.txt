```
from scipy.optimize import linprog
from scipy.sparse import lil_matrix

def solve_lp_max_sum(f, M):
    n = len(f)
    total_constraints = 2*n-1
    A = lil_matrix((total_constraints, n))
    for k in range(total_constraints):
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # i runs i_min..i_max
        # variable index j = k - i
        for i in range(i_min, i_max+1):
            j = k - i
            A[k, j] = f[i]
    b = np.full(total_constraints, M)
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A.tocsr(), b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback maybe
        return None
    g = res.x
    return g
```

"""
Propose a non‑negative step‑function (probability vector) that minimizes
``evaluate_sequence``.  

The algorithm is a multi‑stage global‑local search:

1. **Broad seeding** – generate a large diverse pool of analytic families
   (uniform, exponential, symmetric exponential, power‑law, beta,
   Dirichlet, random piecewise linear, mixtures) together with the previous
   best solution and random perturbations.

2. **Rapid surrogate filtering** – evaluate the exact surrogate
   ``2·n·max(conv)`` (the sum is forced to 1) for every seed and keep the
   best few (default 5).

3. **Continuation L‑BFGS** – for each top seed run a continuation of
   L‑BFGS on a smooth log‑sum‑exp approximation of the ``max``.  The
   smoothing parameter β is increased geometrically (10 → 8·10⁵) and every
   β is allowed up to 300 L‑BFGS iterations.

4. **Exponentiated‑gradient polish** – after L‑BFGS, a long sub‑gradient
   descent (exponentiated‑gradient) runs until the allocated time expires.

5. **Final polishing** – the current global best is further polished with a
   long EG run on the remaining budget, followed by a cheap continuous
   “mass‑transfer” hill‑climb and a short permutation‑annealing phase.

All steps respect the global deadline (default 1000 s).  The best sequence
found is stored in the global ``height_sequence_1`` variable and returned
as a plain ``list[float]``.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix

# ----------------------------------------------------------------------
# Global deadline – used by helpers to avoid overrunning.
# ----------------------------------------------------------------------
_global_deadline = None


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for our sizes)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return a normalised probability vector.
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    return a / a.sum()


def _mixture_exp(n: int, rng: np.random.Generator) -> np.ndarray:
    rate1 = rng.uniform(1e-4, 5e-3)
    rate2 = rng.uniform(1e-4, 5e-3)
    w = rng.uniform(0.0, 1.0)
    i = np.arange(n, dtype=np.float64)
    a = w * np.exp(-rate1 * i) + (1.0 - w) * np.exp(-rate2 * i)
    return a / a.sum()


# ----------------------------------------------------------------------
# L‑BFGS surrogate objective (log‑sum‑exp smoothing).
# ----------------------------------------------------------------------
class _LbfgsObjective:
    """Callable required by scipy.optimize.minimize.

    Works in log‑softmax space: ``log_a`` are unconstrained parameters,
    ``a = softmax(log_a)`` is the probability vector.
    """

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, log_a: np.ndarray):
        # soft‑max to a probability vector (stable)
        max_log = np.max(log_a)
        exp_shift = np.exp(log_a - max_log)
        a = exp_shift / exp_shift.sum()

        # convolution a∗a
        conv = _conv_fft(a, a)
        shift = np.max(conv)

        # smooth max via log‑sum‑exp
        beta = self.beta
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        w = exp_beta / sum_exp                     # soft weights

        # gradient w.r.t a (2 * convolution of reversed a with w)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3·n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # gradient w.r.t log‑softmax parameters
        grad_log = a * (grad_a - np.dot(grad_a, a))

        # smoothed objective value
        M_beta = shift + np.log(sum_exp) / beta
        obj = 2.0 * self.n * M_beta
        return obj, grad_log


def _lbfgs_opt(start_seq: np.ndarray,
               beta_schedule,
               max_iter_per_beta: int,
               deadline: float):
    """
    Continuation L‑BFGS: run L‑BFGS on a sequence of increasingly sharp
    log‑sum‑exp surrogates.
    """
    log_a = np.log(start_seq + 1e-30)          # stay in log‑softmax space
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break

        obj = _LbfgsObjective(len(start_seq), beta)
        opts = {"maxiter": max_iter_per_beta, "disp": False}
        try:
            res = minimize(fun=obj,
                           x0=log_a,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception:                      # any interruption – stop early
            break

        log_a = res.x
        # back‑project to probability vector
        exp_shift = np.exp(log_a - np.max(log_a))
        a = exp_shift / exp_shift.sum()
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient optimiser (non‑smooth).
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0):
    """
    Sub‑gradient descent on the exact max‑objective using an exponentiated‑
    gradient (log‑space) update.  Stops automatically when ``deadline`` is
    reached.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)
    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # active sub‑gradient (only terms that achieve the max)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        # keep logs bounded
        log_a = np.clip(log_a, -30.0, 30.0)

        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap permutation / mass‑transfer annealing.
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 60_000) -> tuple[np.ndarray, float]:
    """Swap two entries (or transfer a tiny amount) with simulated annealing."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        # tiny transfer instead of full swap – helps continuous landscape
        eps = rng.uniform(0.0, min(ns[i], 0.01))
        ns[i] -= eps
        ns[j] += eps

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation – returns a list of probability vectors.
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator, prev_vec) -> list[np.ndarray]:
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    for rate in np.logspace(-4, -2, 8):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))
    for p in np.linspace(0.2, 2.0, 9):
        seeds.append(_power(n, p))

    # beta shapes
    for _ in range(20):
        a = rng.uniform(0.2, 2.5)
        b = rng.uniform(0.2, 2.5)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws with a spread of concentration parameters
    for conc in (0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 3.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # piecewise‑linear random shapes
    for nb in (3, 6, 12, 20):
        for _ in range(3):
            seeds.append(_piecewise_linear(n, nb, rng))

    # mixtures of two exponentials
    for _ in range(10):
        seeds.append(_mixture_exp(n, rng))

    # perturbations of the above (add small Gaussian noise)
    base = seeds[:30]   # take a subset to keep work reasonable
    for vec in base:
        noisy = vec + rng.normal(scale=0.02, size=n)
        noisy = np.clip(noisy, 0.0, None)
        noisy /= noisy.sum()
        seeds.append(noisy)

    # include previous solution and a few permutations thereof
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(3):
            seeds.append(rng.permutation(prev_vec))

    return seeds


# ----------------------------------------------------------------------
# Main entry point.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The pipeline consists of diverse seeding, surrogate filtering, continuation
    L‑BFGS, exponentiated‑gradient polishing, and a final annealing tweak.
    """
    global _global_deadline
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin
    _global_deadline = start_time + budget_s - 2.0
    deadline = _global_deadline

    # --------------------------------------------------------------
    # Problem size – a few thousand points gives a good trade‑off.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous solution if available.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                if prev_vec.sum() > 0:
                    prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            prev_vec = None

    # --------------------------------------------------------------
    # 1. Build a large pool of seed vectors.
    # --------------------------------------------------------------
    seeds = _generate_seeds(n, rng, prev_vec)

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few.
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda pair: pair[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed ranking (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Local refinement with continuation L‑BFGS (plus EG polish).
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # β schedule – increasingly sharp approximations of the max.
    beta_schedule = [10, 50, 200, 800, 3000, 15000, 60000, 200000, 800000]
    max_iter_per_beta = 300

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate roughly equal share of the remaining time (≈70 % of it)
        remaining = deadline - time.time()
        share = remaining * 0.7 / max(1, (top_k - idx))
        sub_deadline = time.time() + share
        print(f"\n[Seed {idx+1}] allocated time ≈ {share:.1f}s")

        # ----- continuation L‑BFGS -----
        seq_lbfgs, val_lbfgs = _lbfgs_opt(seed_vec,
                                          beta_schedule,
                                          max_iter_per_beta,
                                          sub_deadline)

        # ----- exponentiated‑gradient polish (if time left) -----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs,
                                              sub_deadline,
                                              max_iters=3_000_000,
                                              eta0=6.0)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # keep the better refinement
        if val_lbfgs < val_eg:
            local_seq, local_val = seq_lbfgs, val_lbfgs
        else:
            local_seq, local_val = seq_eg, val_eg

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final long EG polishing on the remaining budget.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=5_000_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Very short final annealing tweak (mass‑transfer + swaps).
    # --------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 12.0  # a dozen seconds is enough
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result.
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
