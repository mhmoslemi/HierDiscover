def _geometric_shape(n, r):
    if r <= 0 or r >= 1:

# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used during optimisation: 2·n·max(conv(seq,seq)).
    Input is assumed to sum to 1 (i.e. a probability vector)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _power_shape(n: int, exp: float) -> np.ndarray:
    """a_i ∝ ((i+½)*(n‑½‑i))**exp   (exp negative ⇒ U‑shaped)."""
    i = np.arange(n, dtype=np.float64)
    base = (i + 0.5) * (n - 0.5 - i)
    # avoid overflow / underflow for extreme exp values
    with np.errstate(over='ignore', divide='ignore'):
        a = np.power(base, exp)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        return _uniform(n)
    return a / a.sum()


def _geometric_shape(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i   (0<r<1).  Normalised to sum‑to‑1."""
    if not (0.0 < r < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth (log‑sum‑exp) approximation of max‑convolution.
    Returns the best probability vector seen and its exact surrogate value.
    """
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0,
        20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        max_iter_beta = 8000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break
            a = _softmax(logit)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv  w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            # convert to gradient in log‑softmax space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.12
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 600 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _exp_gradient_multi(start_seq: np.ndarray,
                       deadline: float,
                       eta0: float = 5.0,
                       eps_frac: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution.
    Uses the averaged sub‑gradient over all (near‑)active indices.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_c = np.max(conv)

        # indices whose value lies within eps of the maximum
        eps = max(eps_frac * max_c, 1e-12)
        active = np.where(conv >= max_c - eps)[0]
        if active.size == 0:          # should never happen, but guard
            active = np.array([np.argmax(conv)], dtype=int)

        # build a weight vector on the full convolution length (2n‑1)
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[active] = 1.0 / active.size               # normalised average

        # Sub‑gradient: g_i = 2 * Σ_{k∈active} a_{k‑i}
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                   # length = n + (2n‑1) – 1
        grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # slice gives g_i

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_weighted
        a = _softmax(log_a)

        if it % 3000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex.
    Each move transfers a small random amount of mass from one coordinate to another.
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found within the allotted budget is stored globally as
    ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Load any previously stored solution (if length matches)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            pass

    # --------------------------------------------------------------
    # 0️⃣  Assemble a large, diverse seed pool
    # --------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # Symmetric Beta seeds (α varies)
    for alpha in np.linspace(0.2, 5.0, 20):
        seeds.append(_beta_distribution(n, alpha))

    # Dirichlet draws with several concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Power‑law shapes (U‑shaped, see _power_shape)
    for exp in np.linspace(-2.0, -0.1, 16):
        seeds.append(_power_shape(n, exp))

    # Simple geometric decreasing shapes
    for r in np.linspace(0.2, 0.8, 7):
        seeds.append(_geometric_shape(n, r))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 1️⃣  Cheap surrogate evaluation of all seeds
    # --------------------------------------------------------------
    seed_infos = []
    print("\n--- Initial seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f" seed #{idx + 1:03d}: surrogate = {val:.6f}")

    if not seed_infos:
        raise RuntimeError("No valid seeds were generated within time budget.")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(8, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate among seeds = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam‑based smooth optimisation (≈20 % of remaining time)
    # --------------------------------------------------------------
    now = time.time()
    if now < deadline:
        remaining = deadline - now
        adam_deadline = now + remaining * 0.20
        print("\n--- Adam smoothing (smooth max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # --------------------------------------------------------------
    # 3️⃣  Multi‑active exponentiated‑gradient polishing (major chunk of time)
    # --------------------------------------------------------------
    now = time.time()
    if now < deadline:
        remaining = deadline - now
        # allocate roughly 70 % of the remaining budget to EG runs
        eg_total = remaining * 0.70
        per_seed = eg_total / top_k
        print("\n--- Multi‑active Exponentiated‑Gradient polishing ---")
        for idx, seed_vec in enumerate(top_seeds):
            eg_deadline = time.time() + per_seed
            seq_eg, val_eg = _exp_gradient_multi(seed_vec, eg_deadline,
                                                 eta0=5.0, eps_frac=1e-6)
            print(f"[EG #{idx+1:02d}] exact surrogate = {val_eg:.6f}")
            if val_eg < best_val:
                best_val, best_seq = val_eg, seq_eg.copy()
                print("   >>> Updated global best (EG)")

    # --------------------------------------------------------------
    # 4️⃣  Final tiny stochastic walk (last seconds)
    # --------------------------------------------------------------
    now = time.time()
    if now < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, deadline,
                                                attempts=300_000,
                                                sigma=0.015)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # --------------------------------------------------------------
    # 5️⃣  Official evaluation with the supplied ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
