"""
Search for a non‑negative step‑function that minimises
``evaluate_sequence``.  The pipeline is:

1.  Build a diverse pool of analytic / random seeds (uniform, exponential,
    power‑law, beta‑shaped, Gaussian, sparse Dirichlet, piecewise‑linear,
    …).  All seeds are normalised to sum = 1.
2.  Keep the best few according to the *exact* surrogate
    ``2·n·max(conv(seq,seq))`` (the factor ``sum(seq)²`` is 1 because of the
    normalisation).
3.  Refine each seed with a continuation L‑BFGS run on a
    log‑sum‑exp smoothing of the max‑convolution.  The β‑schedule is
    longer and the per‑β iteration budget is larger than in the reference
    implementation.
4.  Polish the best candidate with a long exponentiated‑gradient (EG)
    descent on the true (non‑smoothed) objective.
5.  Perform a cheap but massive “mass‑transfer” stochastic local search:
    repeatedly move a small amount of probability mass from a random
    index to another while keeping the sum = 1, accepting only improvements
    (occasionally accepting uphill moves with a simple annealing schedule).
6.  Return the best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global deadline – used throughout the algorithm.
# ----------------------------------------------------------------------
_global_deadline = None


def _time_left() -> float:
    """seconds left until the global deadline (may be negative)."""
    return _global_deadline - time.time() if _global_deadline else -1.0


# ----------------------------------------------------------------------
# Utility FFT‑based convolution (real‑valued, faster than np.convolve).
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact surrogate objective (no denominator because all candidates sum to 1).
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (seq is assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return a normalised probability vector.
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - c))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    pts = np.sort(rng.choice(n, size=num_breaks, replace=False))
    pts = np.concatenate(([0], pts, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(pts))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, pts, vals)
    return a / a.sum()


def _random_sparse(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k random locations with random positive mass, then normalise."""
    pos = rng.choice(n, size=k, replace=False)
    weights = rng.random(k) + 1e-3
    a = np.zeros(n, dtype=np.float64)
    a[pos] = weights
    return a / a.sum()


# ----------------------------------------------------------------------
# L‑BFGS surrogate objective (log‑sum‑exp smoothing of the max‑convolution).
# ----------------------------------------------------------------------
class _LbfgsObjective:
    """Callable required by scipy.optimize.minimize."""

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, log_a: np.ndarray):
        # soft‑max -> probability vector (always sums to 1)
        max_log = np.max(log_a)
        exp_shift = np.exp(log_a - max_log)
        a = exp_shift / exp_shift.sum()

        # convolution a∗a
        conv = _conv_fft(a, a)
        shift = np.max(conv)

        # smooth max via log‑sum‑exp (temperature = 1/β)
        exp_beta = np.exp(self.beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        w = exp_beta / sum_exp

        # gradient of smoothed max w.r.t. a:
        #   ∂/∂a  (Σ w_k Σ_i a_i a_{k‑i}) = 2 * (a_rev * w)   (conv of rev(a) and w)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3·n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # convert gradient to log‑softmax parameters
        grad_log = a * (grad_a - np.dot(grad_a, a))

        # smoothed objective value:
        M_beta = shift + np.log(sum_exp) / self.beta
        obj = 2.0 * self.n * M_beta
        return obj, grad_log


def _lbfgs_opt(start_seq: np.ndarray,
               beta_schedule,
               max_iter_per_beta: int,
               deadline: float):
    """
    Continuation L‑BFGS: run L‑BFGS on a sequence of increasingly sharp
    log‑sum‑exp surrogates.
    """
    log_a = np.log(start_seq + 1e-30)          # stay in log‑softmax space
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break

        obj = _LbfgsObjective(len(start_seq), beta)
        opts = {"maxiter": max_iter_per_beta, "disp": False}
        try:
            res = minimize(fun=obj,
                           x0=log_a,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception:
            break

        log_a = res.x
        exp_shift = np.exp(log_a - np.max(log_a))
        a = exp_shift / exp_shift.sum()
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser (exponentiated gradient).
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0):
    """
    Exponentiated‑gradient descent on the *exact* max‑convolution objective.
    The update is performed in the log‑domain, which automatically keeps the
    vector non‑negative and normalised.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)          # log‑softmax parameters
    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # locate the (maybe several) maximising indices
        conv = _conv_fft(a, a)
        idx_max = int(np.argmax(conv))

        # sub‑gradient of the *unsmoothed* max (active constraint)
        g = np.zeros_like(a)
        i_start = max(0, idx_max - (n - 1))
        i_end = min(n - 1, idx_max)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = idx_max - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)          # diminishing step‑size
        log_a -= eta * g
        # avoid overflow/underflow in the exponent
        log_a = np.clip(log_a, -30.0, 30.0)

        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic “mass‑transfer” local search.
# ----------------------------------------------------------------------
def _mass_transfer_opt(start_seq: np.ndarray,
                       deadline: float,
                       max_no_improve: int = 200_000,
                       temp_start: float = 1e-3,
                       temp_end: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Randomly move a small amount of probability mass from one index to another.
    Accept only improving moves (occasionally uphill moves governed by a tiny
    annealing schedule).  Returns the best sequence seen.
    """
    a = start_seq.copy()
    cur_val = _objective_surrogate(a)
    best_seq = a.copy()
    best_val = cur_val

    n = len(a)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_no_improve:
        it += 1
        # linear cooling for the occasional uphill acceptance
        T = temp_start + (temp_end - temp_start) * (it / max_no_improve)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if a[i] <= 0.0:
            continue

        # move at most 10 % of the mass at i (but never more than a[i])
        delta = rng.random() * min(a[i], 0.1 * a[i])
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # (sum stays 1, non‑negativity guaranteed by construction)
        new_val = _objective_surrogate(new_a)

        if new_val < cur_val or (T > 0 and rng.random() < math.exp(-(new_val - cur_val) / T)):
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_seq = new_a.copy()
                best_val = new_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method builds a large pool of analytic / random seeds, refines the
    most promising ones with continuation L‑BFGS, polishes the global best
    with a long exponentiated‑gradient run, and finally performs a cheap
    stochastic mass‑transfer local search.
    """
    global _global_deadline
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a two‑second safety margin for the final evaluate_sequence call
    _global_deadline = start_time + budget_s - 2.0
    deadline = _global_deadline

    # ------------------------------------------------------------------
    # Problem size – keep a reasonable trade‑off between expressivity
    # and speed.  Use the length of a previously stored solution if it
    # exists, otherwise fall back to 4000.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None and isinstance(previous, list):
        n = len(previous)
        prev_vec = np.maximum(np.asarray(previous, dtype=np.float64), 0.0)
        prev_vec /= prev_vec.sum()
    else:
        n = 4000
        prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build a rich pool of seed vectors.
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _exp_decay(n, 0.0015),
        _exp_decay(n, 0.0035),
        _sym_exp_decay(n, 0.0025),
        _power(n, 0.6),
        _power(n, 1.1),
        _gaussian(n, sigma=0.15 * n),
        _gaussian(n, sigma=0.20 * n),
        _gaussian(n, sigma=0.25 * n),
    ]

    # Beta‑shaped families (a wide grid of parameters).
    for a in (0.2, 0.5, 1.0, 1.5, 2.0):
        for b in (0.2, 0.5, 1.0, 1.5, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # Random Dirichlet draws.
    for conc in (0.02, 0.05, 0.15, 0.5, 1.0, 3.0, 8.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse vectors (few non‑zeros).
    for k in (30, 60, 120):
        for _ in range(2):
            seeds.append(_random_sparse(n, k, rng))

    # Piecewise‑linear random shapes.
    for nb in (5, 10, 20):
        for _ in range(1):
            seeds.append(_piecewise_linear(n, nb, rng))

    # Include the previously found solution (if any) and a few of its permutations.
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(3):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation, keep the most promising seeds.
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() >= deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda p: p[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed ranking (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Local refinement of each top seed with continuation L‑BFGS.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    beta_schedule = [10, 40, 160, 800, 4000, 20000, 100000]   # finer schedule
    max_iter_per_beta = 500                                 # more work per β

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # allocate roughly 15 % of the remaining budget per remaining seed
        share = remaining * 0.15 / max(1, (top_k - idx))
        sub_deadline = time.time() + share
        print(f"\n[Seed {idx+1}] allocated time ≈ {share:.1f}s")

        # ----- continuation L‑BFGS -----
        seq_lbfgs, val_lbfgs = _lbfgs_opt(seed_vec,
                                          beta_schedule,
                                          max_iter_per_beta,
                                          sub_deadline)

        # ----- EG polish (still inside the sub‑deadline) -----
        seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs,
                                           sub_deadline,
                                           max_iters=1_000_000,
                                           eta0=6.0)

        # keep the better of the two refinements
        if val_lbfgs < val_eg:
            local_seq, local_val = seq_lbfgs, val_lbfgs
        else:
            local_seq, local_val = seq_eg, val_eg

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")
        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long EG polishing on the remaining budget.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=5_000_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Stochastic mass‑transfer local search (cheap but many cheap moves).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mass‑transfer local search ---")
        seq_last, val_last = _mass_transfer_opt(global_best_seq,
                                                deadline,
                                                max_no_improve=500_000,
                                                temp_start=1e-3,
                                                temp_end=1e-6)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Local search] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:                    # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
