def adam_opt(a, log_a, deadline, max_iters=2_000_000, step0=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    best_a = a.copy()
    best_val = objective_exact(a)
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        I = np.where(conv == max_val)[0]
        w = np.zeros_like(conv)
        w[I] = 1.0 / len(I)
        grad_a = 2 * _conv_fft(a[::-1], w)  # gradient w.r.t. a
        # convert to gradient w.r.t log_a (softmax param)
        grad_log = a * (grad_a - np.dot(grad_a, a))
        # Adam update
        m = beta1 * m + (1-beta1) * grad_log
        v = beta2 * v + (1-beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        log_a -= step0 * m_hat / (np.sqrt(v_hat) + eps)
        # Clip
        log_a = np.clip(log_a, -30.0, 30.0)
        # Recover a
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()
        cur_val = objective_exact(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
    return best_a, best_val

"""
Propose a candidate step‑function (probability vector) that improves the
upper‑bound 2·n·max(conv(a,a))/sum(a)² returned by ``evaluate_sequence``.

Algorithm
---------

1. **Generate a diverse pool of seeds** – analytic families (uniform,
   exponential, power, beta, sinusoid, cosine‑bump), Dirichlet draws with
   many concentration parameters and the previous best solution (if any).

2. **Select the most promising seeds** using the exact surrogate
   `score = 2·n·max(conv(a,a))` (the scaling by the sum is irrelevant).

3. **Refine each seed** with a continuation L‑BFGS on a *log‑softmax*
   parametrisation.  The objective is the smooth log‑sum‑exp
   approximation of the max‑convolution; a schedule of increasingly sharp
   β values is used and each sub‑problem receives many L‑BFGS iterations.

4. **Polish the current global best** with a long Adam‑optimised sub‑gradient
   descent on the *exact* max‑convolution.  The sub‑gradient is obtained
   from the set of active convolution indices (those attaining the maximum).

5. **Finish with a brief permutation‑annealing** to escape possible local
   minima.

All steps respect the global time budget (default 1000 s) and
periodically check the deadline.  The final candidate is stored in the
global variable ``height_sequence_1`` and returned as a plain list of
floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_exact(a: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sinusoidal(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    return a / a.sum()


def _cosine_bump(n: int, width: float = 0.25) -> np.ndarray:
    """Bump centred in the middle, width ∈ (0,1)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    d = np.abs(i - centre) / n / width
    a = np.cos(np.pi * np.clip(d, 0, 1) / 2.0) ** 2
    a[d > 1] = 0.0
    return a / a.sum()


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS (log‑sum‑exp max)
# ----------------------------------------------------------------------
class _LbfgsSmoothObj:
    """
    Callable for ``scipy.optimize.minimize``.
    Optimises log‑softmax parameters `log_a` to minimise a smooth
    approximation of 2·n·max(conv(a,a)).
    """

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, log_a: np.ndarray):
        # soft‑max to a probability vector
        max_log = np.max(log_a)
        exp_shift = np.exp(log_a - max_log)
        a = exp_shift / exp_shift.sum()

        # convolution a∗a
        conv = _conv_fft(a, a)
        shift = np.max(conv)

        # log‑sum‑exp smoothing
        beta = self.beta
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        w = exp_beta / sum_exp                # soft‑max weights (length 2n‑1)

        # gradient w.r.t. a : 2·conv(rev(a), w)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)           # length 3·n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # chain rule to log‑softmax parameters
        grad_log = a * (grad_a - np.dot(grad_a, a))

        # smoothed max value
        M_beta = shift + np.log(sum_exp) / beta
        obj = 2.0 * self.n * M_beta
        return float(obj), grad_log.astype(np.float64)


def _lbfgs_continuation(start_seq: np.ndarray,
                         beta_schedule,
                         max_iter_per_beta: int,
                         deadline: float):
    """
    Continuation L‑BFGS: for each β in ``beta_schedule`` run a limited‑iteration
    L‑BFGS on the smoothed objective.  Returns the best exact (non‑smooth)
    sequence found together with its exact surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)       # stay in log‑softmax space
    best_seq = start_seq.copy()
    best_val = _objective_exact(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break

        obj = _LbfgsSmoothObj(len(start_seq), beta)
        opts = {"maxiter": max_iter_per_beta, "disp": False}
        try:
            res = minimize(fun=obj,
                           x0=log_a,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception:
            break

        log_a = res.x
        # back‑project to probability simplex
        exp_shift = np.exp(log_a - np.max(log_a))
        a = exp_shift / exp_shift.sum()
        cur_val = _objective_exact(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              deadline: float,
              max_iters: int = 2_000_000,
              step0: float = 0.6,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8):
    """
    Adam on the log‑softmax parameters using the exact sub‑gradient of the
    max‑convolution.  Returns the best sequence found and its exact value.
    """
    a = start_seq.copy()
    n = len(a)
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = a.copy()
    best_val = _objective_exact(a)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        idx_max = np.where(conv == max_val)[0]

        # weight vector for the active set (uniform)
        w = np.zeros_like(conv)
        w[idx_max] = 1.0 / len(idx_max)

        # exact sub‑gradient w.r.t. a
        grad_a = 2.0 * _conv_fft(a[::-1], w)

        # convert to gradient w.r.t. log‑softmax variables
        grad_log = a * (grad_a - np.dot(grad_a, a))

        # Adam update on log‑parameters
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        log_a -= step0 * m_hat / (np.sqrt(v_hat) + eps)

        # keep the log‑parameters numerically stable
        log_a = np.clip(log_a, -30.0, 30.0)

        # project back to simplex
        exp_shift = np.exp(log_a - np.max(log_a))
        a = exp_shift / exp_shift.sum()

        cur_val = _objective_exact(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap permutation‑annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 60_000):
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_exact(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_exact(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence``.  The result is stored in the global variable
    ``height_sequence_1`` and also returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – 4000 points works well for the time budget.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous best (if any) to keep continuity across runs.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                prev_vec /= prev_vec.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build a rich pool of seed vectors.
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _exp_decay(n, 0.0015),
        _exp_decay(n, 0.0035),
        _sym_exp_decay(n, 0.0025),
        _power(n, 0.6),
        _power(n, 1.1),
        _sinusoidal(n),
        _cosine_bump(n, width=0.2),
    ]

    # Dirichlet draws with a wide spread of concentration parameters
    for conc in (0.02, 0.05, 0.15, 0.5, 1.0, 3.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Beta‑shaped families
    for _ in range(8):
        a = rng.uniform(0.2, 2.0)
        b = rng.uniform(0.2, 2.0)
        seeds.append(_beta_shape(n, a, b))

    # Include the previous solution and a few random permutations of it
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(3):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few.
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_exact(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed ranking (exact surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Local refinement with continuation L‑BFGS.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_exact(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # A relatively aggressive β schedule and many iterations per β
    beta_schedule = [5, 20, 80, 300, 1200, 5000, 20000, 100000]
    max_iter_per_beta = 500

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate roughly equal share of the remaining time to each seed
        remaining = deadline - time.time()
        share = remaining * 0.18 / max(1, (top_k - idx))
        sub_deadline = time.time() + share
        print(f"\n[Seed {idx+1}] time share ≈ {share:.1f}s")

        # ---- continuation L‑BFGS ----
        seq_lbfgs, val_lbfgs = _lbfgs_continuation(seed_vec,
                                                   beta_schedule,
                                                   max_iter_per_beta,
                                                   sub_deadline)

        # ---- Adam polish (if still time) ----
        if time.time() < sub_deadline:
            seq_adam, val_adam = _adam_opt(seq_lbfgs,
                                           sub_deadline,
                                           max_iters=800_000,
                                           step0=0.6)
        else:
            seq_adam, val_adam = seq_lbfgs, val_lbfgs

        # choose the better of the two refinements
        if val_lbfgs < val_adam:
            local_seq, local_val = seq_lbfgs, val_lbfgs
        else:
            local_seq, local_val = seq_adam, val_adam

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")
        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long Adam polishing on the remaining budget.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final Adam polishing (remaining budget) ---")
        final_seq, final_val = _adam_opt(global_best_seq,
                                         deadline,
                                         max_iters=2_500_000,
                                         step0=0.5)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation‑annealing tweak (few seconds).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = time.time() + 12.0
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          anneal_deadline,
                                          max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final anneal] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for the next run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
