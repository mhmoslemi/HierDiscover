"""
Propose a step‑function (non‑negative heights) that minimizes
``evaluate_sequence``.  The optimizer works on a relatively large
problem size (n≈12 k) and consists of three stages:

1. **Rich seed generation** – analytic families (power, inverse‑sqrt,
   exponential, beta, Dirichlet) are instantiated and the best few are
   kept according to the exact surrogate 2·n·max(conv(a,a)).
2. **Mirror‑descent on the max‑convolution** – an exponentiated‑gradient
   (EG) method that directly minimises the surrogate by following the
   sub‑gradient at the (almost) active convolution indices.  Momentum
   (Adam‑style) and a decreasing step size stabilise the updates.
3. **Flattening gradient step** – a quadratic loss that forces the whole
   convolution towards its average value, again performed in the
   probability‑simplex using a soft‑max parametrisation.

All work is done on ℓ₁‑normalised vectors, therefore the final result is
scale‑invariant and can be passed unchanged to ``evaluate_sequence``.
The routine respects the 1000 s wall‑clock limit, stores the best found
sequence in the global variable ``height_sequence_1`` and returns it as a
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities (FFT‑based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (the exact quantity that ``evaluate_sequence`` uses,
# up to the normalisation of the vector).
# ----------------------------------------------------------------------
def _surrogate(seq: np.ndarray, n: int) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))   (seq summed to 1)."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser for the max‑convolution objective
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 5_000_000,
                            eta0: float = 15.0,
                            tol: float = 1e-10) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Uses all indices whose value is within ``tol`` relative to the current
    maximum.  Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _surrogate(a, n)

    idx = np.arange(n, dtype=np.int64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set: all entries within tol·max of the maximum
        if max_val == 0:
            active = np.arange(conv.size, dtype=np.int64)
        else:
            active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of max(conv) (averaged over active indices)
        g = np.zeros_like(a)
        if active.size:
            for k in active:
                rev_idx = k - idx
                mask = (rev_idx >= 0) & (rev_idx < n)
                g[mask] += a[rev_idx[mask]]
            g = 2.0 * g / float(active.size)

        # Adam‑like momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient descent that flattens the whole convolution (quadratic loss)
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 1_000_000,
                         eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the loss Σ_k (conv_k – avg)^2  (avg = 1/(2n-1)).
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate(a, n)

    denom = float(2 * n - 1)               # length of the full convolution
    avg_const = 1.0 / denom                # because sum(a)=1 ⇒ (sum a)^2 = 1

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)             # length = 2n‑1
        diff = conv - avg_const

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)     # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent step (soft‑max parameterisation)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Full optimisation pipeline for a single seed
# ----------------------------------------------------------------------
def _full_optimisation(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run EG → flattening → a short simulated‑annealing polish on *start_vec*.
    All stages stop as soon as the shared deadline is reached.
    Returns the best sequence encountered and its surrogate value.
    """
    # 1) exponentiated‑gradient on the max‑convolution
    seq_eg, val_eg = _exp_gradient_opt_multi(start_vec, n, deadline,
                                            max_iters=3_000_000,
                                            eta0=20.0,
                                            tol=1e-12)

    # 2) flattening gradient (quadratic loss)
    if time.time() < deadline:
        seq_flat, val_flat = _flatten_gradient_opt(seq_eg, n, deadline,
                                                   max_iters=600_000,
                                                   eta0=12.0)
    else:
        seq_flat, val_flat = seq_eg, val_eg

    # 3) cheap simulated‑annealing polish (optional)
    if time.time() < deadline:
        # Very light SA – only a few thousand proposals, just in case we
        # happen to discover a slight improvement.
        a = seq_flat.copy()
        best_seq = a.copy()
        best_val = val_flat
        cur_val = best_val
        attempts = 0
        max_attempts = 50_000
        while time.time() < deadline and attempts < max_attempts:
            sigma = 0.001 * (1.0 - attempts / max_attempts)
            ns = a + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                attempts += 1
                continue
            ns /= s
            val = _surrogate(ns, n)
            if val < cur_val:
                a = ns
                cur_val = val
                if val < best_val:
                    best_val = val
                    best_seq = ns.copy()
            attempts += 1
        seq_sa, val_sa = best_seq, best_val
    else:
        seq_sa, val_sa = seq_flat, val_flat

    candidates = [(start_vec, _surrogate(start_vec, n)),
                  (seq_eg, val_eg),
                  (seq_flat, val_flat),
                  (seq_sa, val_sa)]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val

        for _ in range(2):
            seeds.append(_random_dirichlet(n, float(conc), rng))

    # convex mixtures with uniform (helps break symmetry)
    base = _inverse_sqrt(n)
    for alpha in [0.2, 0.5, 0.8]:
        seeds.append(_mix_shapes(_uniform(n), base, float(alpha)))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds, keep the top few.
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _surrogate(vec, n)
        evaluated.append((vec, val))

    if not evaluated:
        raise RuntimeError("No valid seed generated before deadline.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise the global best.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _surrogate(global_best_seq, n)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine each top seed – distribute the remaining budget.
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate a diminishing share; later seeds get a bit more time
        share = remaining * 0.13 / max(1, (top_k - si))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}/{top_k}] optimisation budget ≈ {share:.1f}s")
        seq_opt, val_opt = _full_optimisation(seed_vec, n, sub_deadline, rng)
        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"   *** New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final intensive polishing on the current global best.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing on global best ---")
        final_seq, final_val = _full_optimisation(global_best_seq, n, deadline, rng)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function.
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
