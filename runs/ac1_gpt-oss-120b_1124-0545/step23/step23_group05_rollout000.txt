assistantfinal**Solution Overview**

The evaluation function  

\[
E(a)=\frac{2\,n\max (a*a)}{(\sum a)^2},
\qquad a_i\ge 0,
\]

is *scale‑invariant* – we can normalise any candidate to `sum = 1` and work only with the
max‑convolution term.  
We therefore optimise the surrogate  

\[
S(a)=2\,n\max (a*a)\qquad(\sum a = 1)
\]

and only at the end call the provided `evaluate_sequence` to get the true score.

The code below follows a **two‑stage refinement**:

1. **Adam on a smooth approximation of the max** –  
   `max(conv)` is replaced by a log‑sum‑exp with a temperature β.
   β is increased linearly from 1 to a large value (≈ 10⁷).  
   The optimisation is performed in the *soft‑max* (log‑probability) space,
   guaranteeing positivity and the simplex constraint automatically.
   Adam’s momentum and an adaptive learning‑rate give a fast, stable descent.

2. **Optional simulated‑annealing polish** –  
   a cheap Gaussian perturbation on the simplex, accepting occasional
   uphill moves.  It cleans up any remaining tiny ripples in the convolution.

Both stages respect a shared wall‑clock deadline (default 1000 s).  
The best sequence found is stored in the global variable `height_sequence_1`
and returned as a plain Python list.

```python
"""
propose_candidate – refined Adam + SA optimiser

* works on the scale‑invariant surrogate 2·n·max(conv(a,a))
* optimisation is carried out in log‑softmax space (mirror descent)
  using the Adam optimiser.
* a log‑sum‑exp smoothing of the max is employed, with a linearly
  increasing temperature β that drives the surrogate to the true max.
* after the Adam phase a short simulated‑annealing run is performed
  to polish the result.
* all operations stay on the probability simplex, guaranteeing
  non‑negative coefficients and a sum of 1.
* the routine respects the supplied time budget and stores the
  best sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (the quantity we minimise during optimisation)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(a,a)) for a with sum = 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple analytic families – used only when no previous solution exists
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1) – the classic continuous flat‑conv solution."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 500_000,
                     max_beta: float = 1e7,
                     init_lr: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Adam) on the log‑sum‑exp approximation of max(conv).
    Returns the best sequence seen and its surrogate value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    # work in log‑softmax space
    log_a = np.log(a + 1e-30)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # linear schedule for β
        beta = 1.0 + (max_beta - 1.0) * (t / max_iters)

        # convolution and smooth weights
        conv = _conv_fft(a, a)
        max_c = conv.max()
        # shift for numerical stability (max_c is the maximum)
        exp_beta = np.exp(beta * (conv - max_c))
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp                     # shape (2n‑1,)

        # gradient w.r.t a (see derivation in analysis)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent gradient (∇_log a)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # decaying learning rate
        lr = init_lr / (1.0 + t / 10000.0)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # map back to the simplex
        a = _softmax(log_a)

        # enforce symmetry (empirically optimal for this problem)
        a = 0.5 * (a + a[::-1])

        # renormalise (softmax already sums to 1, symmetry may have perturbed it)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # evaluate surrogate
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress report
        if t % 20000 == 0:
            print(f"[Adam] it={t:,}, β≈{beta:.1e}, surrogate={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight simulated annealing for a final polish
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 250_000,
                         start_sigma: float = 0.001,
                         T_start: float = 0.01,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations of the probability vector.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear temperature schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=len(a))
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

        if attempts % 50000 == 0:
            print(f"[SA] attempts={attempts:,}, best surrogate={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the wall‑clock budget,
    stores the best found sequence in the global variable
    ``height_sequence_1`` and returns it as a Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 1.5

    # ------------------------------------------------------------------
    # problem size – 4000 works well for the time budget
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # initialise from the previous best if present, otherwise fall back
    # to a simple analytic family
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and len(previous) == n:
        base = np.asarray(previous, dtype=np.float64)
        # ensure positivity and normalisation
        base = np.maximum(base, 0.0)
        s = base.sum()
        if s == 0.0:
            base = _inverse_sqrt(n)
        else:
            base = base / s
    else:
        base = _inverse_sqrt(n)

    print(f"\n[Init] start surrogate = {_objective_surrogate(base):.6f}")

    # ------------------------------------------------------------------
    # 1) Adam on smoothed max‑convolution
    # ------------------------------------------------------------------
    sub_deadline = time.time() + (deadline - time.time()) * 0.75
    seq_adam, val_adam = _adam_smooth_opt(
        start_seq=base,
        deadline=sub_deadline,
        rng=rng,
        max_iters=400_000,          # enough iterations for a solid descent
        max_beta=5e6,
        init_lr=0.15,
    )
    print(f"\n[Adam] best surrogate after Adam = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 2) Simulated annealing polishing (uses remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            start_seq=seq_adam,
            deadline=deadline,
            rng=rng,
            max_attempts=300_000,
            start_sigma=0.0008,
            T_start=7e-3,
            T_end=5e-9,
        )
        print(f"\n[SA] best surrogate after SA = {val_sa:.6f}")
    else:
        seq_sa, val_sa = seq_adam, val_adam

    # ------------------------------------------------------------------
    # 3) Choose the best among the two phases
    # ------------------------------------------------------------------
    if val_sa < val_adam:
        best_seq = seq_sa
        best_val = val_sa
    else:
        best_seq = seq_adam
        best_val = val_adam

    # ------------------------------------------------------------------
    # 4) Final official evaluation
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
