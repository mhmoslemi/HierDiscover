def hill_climb(a, conv, deadline):
    n = len(a)
    best_a = a.copy()
    best_val = max(conv)
    cur_a = a
    cur_conv = conv
    cur_val = best_val

    # define parameters
    max_iters = 2000000  # or until time
    start_temp = cur_val * 0.05  # initial temperature relative
    end_temp = cur_val * 1e-6
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute temperature
        T = start_temp * ((end_temp/start_temp) ** (it / max_iters))
        # current max index
        t = int(cur_conv.argmax())
        # compute g_i indices:
        # range of i where i in [max(0, t-(n-1)), min(n-1, t)]
        i_min_idx = max(0, t - (n - 1))
        i_max_idx = min(n - 1, t)
        # slice corresponding a[t-i] values:
        # For i in [i_min_idx, i_max_idx], a[t-i] is a_rev = a[t - i] = a[t - np.arange(i_min_idx, i_max_idx+1)]
        rev_indices = t - np.arange(i_min_idx, i_max_idx+1)
        # but we can compute using a directly:
        # gradient slice = 2 * a[rev_indices]
        g_slice = 2.0 * cur_a[rev_indices]
        # pick i_max as index with max g_slice
        idx_local_max = int(np.argmax(g_slice))
        i = i_min_idx + idx_local_max
        # pick i_min as index with min g_slice (prefer zero)
        idx_local_min = int(np.argmin(g_slice))
        j = i_min_idx + idx_local_min
        if i == j:
            # if same, pick random j
            j = (i + np.random.randint(1, n)) % n

        # step size proportion based on current a[i]
        a_i = cur_a[i]
        # step fraction decays
        fraction = 0.1 * (1 - it / max_iters) + 0.001
        delta = min(a_i * fraction, a_i)
        if delta <= 0:
            continue

        # update conv and a
        # conv slices modifications using old a (cur_a)
        # first subtract i contribution
        cur_conv[i:i+n] -= 2.0 * delta * cur_a
        # add j contribution
        cur_conv[j:j+n] += 2.0 * delta * cur_a

        # unit corrections
        delta2 = delta * delta
        cur_conv[2*i] += delta2
        cur_conv[2*j] += delta2
        cur_conv[i+j] -= 2.0 * delta2

        # update a
        cur_a[i] -= delta
        cur_a[j] += delta

        # ensure non-negativity due to floating errors:
        if cur_a[i] < 0:
            cur_a[i] = 0.0

        # compute new max value
        new_val = cur_conv.max()
        # accept or revert based on temperature
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / (T + 1e-20)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = cur_a.copy()
        else:
            # revert changes
            # Use stored a_i? revert by adding back delta to i and subtract from j.
            # Revert conv modifications: we can just revert by opposite operations.
            cur_conv[i:i+n] += 2.0 * delta * cur_a
            cur_conv[j:j+n] -= 2.0 * delta * cur_a
            cur_conv[2*i] -= delta2
            cur_conv[2*j] -= delta2
            cur_conv[i+j] += 2.0 * delta2
            cur_a[i] += delta
            cur_a[j] -= delta
            # Ensure j mass not negative (should not happen)
        # optionally periodic full recompute of conv and max to avoid drift

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    import time, math, numpy as np

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5 # safety

    # Problem size.
    n = 4000

    # Load previous sequence if available.
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.shape == (n,):
            a0 = np.maximum(arr, 0.0)
            a0 = a0 / a0.sum()
        else:
            a0 = None
    else:
        a0 = None

    # Build seed list.
    seeds = []
    # Uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))
    # Exponential decays
    for rate in (0.001, 0.002, 0.003, 0.004):
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-rate * i)
        seeds.append(a / a.sum())
    # Symmetric exponential
    for rate in (0.0015, 0.0025, 0.0035):
        i = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        a = np.exp(-rate * np.abs(i - centre))
        seeds.append(a / a.sum())
    # Power law
    for p in (0.5, 0.7, 0.9, 1.1):
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / (idx ** p)
        seeds.append(a / a.sum())
    # Beta shape
    for _ in range(5):
        a_param = rng.uniform(0.2, 2.0)
        b_param = rng.uniform(0.2, 2.0)
        i = np.arange(1, n + 1, dtype=np.float64)
        a = (i ** (a_param - 1.0)) * ((n - i + 1) ** (b_param - 1.0))
        seeds.append(a / a.sum())
    # Dirichlet draws
    for conc in (0.1, 0.5, 1.0, 3.0):
        for _ in range(2):
            a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            seeds.append(a)

    if a0 is not None:
        seeds.append(a0)

    # Evaluate surrogate for each seed
    def surrogate(seq):
        conv = np.fft.irfft(np.fft.rfft(seq, n=1<<((2*n-2).bit_length())) *
                           np.fft.rfft(seq, n=1<<((2*n-2).bit_length())))[:2*n-1]
        # we can just use np.convolve directly for ease; it's fine per seed small.
        conv = np.convolve(seq, seq)
        return 2.0 * n * np.max(conv)  # objective surrogate (no scaling needed)

    best_seq = None
    best_val = float('inf')
    for seq in seeds:
        if time.time() > deadline:
            break
        val = surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"Initial best surrogate = {best_val:.6f}")

    # Hill climbing
    a = best_seq.copy()
    conv = np.convolve(a, a)   # full length 2n-1
    cur_val = np.max(conv)
    best_seq = a.copy()
    best_val = cur_val
    print(f"Starting hill climb: cur_val = {cur_val:.6f}")

    max_iters = 2000000   # but will stop early by time
    iteration = 0
    # precompute constant for random numbers
    while time.time() < deadline and iteration < max_iters:
        iteration += 1

        t = int(conv.argmax())   # index of current max convolution

        # Determine range of i where gradient non-zero
        i_start = max(0, t - (n - 1))
        i_end = min(n - 1, t)
        # Compute gradient slice: a[t - i] for i in [i_start, i_end]
        rev_idx = t - np.arange(i_start, i_end + 1, dtype=np.int64)
        grad_slice = a[rev_idx]          # corresponds to a[t-i]
        # pick i with maximum a (largest gradient)
        local_i = int(np.argmax(grad_slice))
        i = i_start + local_i
        if a[i] <= 1e-15:
            # cannot move mass from zero position, choose another random i
            # fallback: random i with a[i]>0
            positive_idxs = np.flatnonzero(a > 1e-15)
            if positive_idxs.size == 0:
                break
            i = rng.choice(positive_idxs)

        # Choose j outside the gradient support (gradient zero) if possible
        # prefer index where a[t-j] = 0
        if i_start > 0:
            j = rng.integers(0, i_start)   # choose from left side
        elif i_end < n - 1:
            j = rng.integers(i_end + 1, n)  # choose from right side
        else:
            # whole range covered, just random different index
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)

        # Determine step size fraction
        # Simple schedule: fraction decreases with iteration count
        frac = 0.2 * (1.0 - iteration / max_iters) + 0.001   # between ~0.2 and 0.001
        delta = min(frac * a[i], a[i])  # ensure not exceed a[i]
        if delta <= 0:
            continue

        # Save slices for revert
        slice_i = conv[i:i+n].copy()
        slice_j = conv[j:j+n].copy()
        val_2i = conv[2*i]
        val_2j = conv[2*j]
        idx_ij = i + j

        conv[i:i+n] -= 2.0 * delta * a

        conv[2*i] += delta2

        if iteration % 5000 == 0:

        true_val = None

"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

The algorithm works as follows:

1. Load the best sequence from the previous run (global variable
   ``height_sequence_1``) if it exists.  Otherwise generate a modest pool of
   analytic / random seeds and keep the best according to the exact surrogate
   ``2·n·max(conv(a,a))``.

2. Starting from that seed run a *mass‑transfer hill‑climbing* routine.
   At each iteration the index that contributes the most to the current
   maximal convolution entry is selected (the “steepest” coordinate) and a
   small amount of probability mass is moved from this index to an index
   that does not affect the current maximum.  The convolution is updated
   with O(n) slice operations, the new maximum is inspected and the move is
   kept only if it reduces the objective.  Periodically the full convolution
   is recomputed (FFT) to avoid numerical drift.

   The step size shrinks with the iteration count, allowing a fine
   approach to a local optimum.

3. After the hill‑climbing phase the best sequence found is evaluated with
   ``evaluate_sequence`` and stored back into ``height_sequence_1`` for future
   runs.

The whole routine respects the 1000 s budget (a small safety margin is kept)
and uses only NumPy – no external files, network or heavy libraries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (fast, O(n log n))."""
    n = a.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv


def _surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used for quick ranking: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    # Full convolution with FFT – accurate and cheap for a single evaluation.
    conv = _full_conv_fft(seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Hill‑climbing core
# ----------------------------------------------------------------------
def _hill_climb(initial_seq: np.ndarray,
                deadline: float,
                max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Perform a deterministic mass‑transfer local search.

    The method repeatedly moves a fraction of the probability mass from the
    index that most contributes to the current maximal convolution entry to a
    distant index (gradient zero).  The convolution is updated by O(n)
    slice operations; the move is kept only if it reduces the maximal entry.
    """
    a = initial_seq.copy()
    n = a.shape[0]

    # current full convolution and its maximal value
    conv = np.convolve(a, a)                     # length 2·n‑1
    cur_max = float(conv.max())
    best_seq = a.copy()
    best_val = cur_max

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ------------------------------------------------------------------
        # locate the index t where the current maximal entry occurs
        # ------------------------------------------------------------------
        t = int(conv.argmax())

        # --------------------------------------------------------------
        # Indices that actually affect conv[t]:
        #   i satisfies 0 ≤ i ≤ n‑1 and 0 ≤ t‑i ≤ n‑1  ⇔  i ∈ [t-(n-1) , t]
        # --------------------------------------------------------------
        i_start = max(0, t - (n - 1))
        i_end   = min(n - 1, t)
        # a[t‑i] for all admissible i (gradient up to a constant factor 2)
        rev = t - np.arange(i_start, i_end + 1, dtype=np.int64)
        grad_slice = a[rev]

        # pick the index with the largest contribution (largest a[t‑i])
        local_i = int(np.argmax(grad_slice))
        i = i_start + local_i

        # if by any chance a[i] is (numerically) zero, pick another positive
        if a[i] <= 1e-18:
            pos = np.flatnonzero(a > 1e-18)
            if pos.size == 0:
                break
            i = int(np.random.choice(pos))

        # --------------------------------------------------------------
        # choose a destination index j where the gradient is zero.
        # Ideally pick it completely outside the active interval.
        # --------------------------------------------------------------
        if i_start > 0:
            j = np.random.randint(0, i_start)               # left side
        elif i_end < n - 1:
            j = np.random.randint(i_end + 1, n)              # right side
        else:
            # interval covers whole domain – pick any different index
            j = np.random.randint(0, n)
            while j == i:
                j = np.random.randint(0, n)

        # --------------------------------------------------------------
        # step size – a decreasing fraction of the source mass
        # --------------------------------------------------------------
        frac = 0.2 * (1.0 - it / max_iters) + 0.001   # between ~0.2 and 0.001
        delta = min(frac * a[i], a[i] * 0.5, 0.02)    # also cap absolute size
        if delta <= 0.0:
            continue

        # --------------------------------------------------------------
        # Save the parts of conv that will be overwritten (to allow revert)
        # --------------------------------------------------------------
        slice_i = conv[i:i + n].copy()
        slice_j = conv[j:j + n].copy()
        val_2i = conv[2 * i]
        val_2j = conv[2 * j]
        idx_ij = i + j
        val_ij = conv[idx_ij]

        # --------------------------------------------------------------
        # Apply the update (using the *old* a for the linear part)
        # --------------------------------------------------------------
        conv[i:i + n] -= 2.0 * delta * a
        conv[j:j + n] += 2.0 * delta * a
        delta2 = delta * delta
        conv[2 * i] += delta2
        conv[2 * j] += delta2
        conv[idx_ij] -= 2.0 * delta2

        a[i] -= delta
        a[j] += delta

        new_max = float(conv.max())

        # --------------------------------------------------------------
        # Keep the move only if it improves the objective
        # --------------------------------------------------------------
        if new_max < cur_max - 1e-14:
            cur_max = new_max
            if new_max < best_val:
                best_val = new_max
                best_seq = a.copy()
        else:
            # revert the whole step
            conv[i:i + n] = slice_i
            conv[j:j + n] = slice_j
            conv[2 * i] = val_2i
            conv[2 * j] = val_2j
            conv[idx_ij] = val_ij
            a[i] += delta
            a[j] -= delta

        # --------------------------------------------------------------
        # Periodically recompute the convolution from scratch to curb drift
        # --------------------------------------------------------------
        if it % 5000 == 0:
            conv = _full_conv_fft(a)
            cur_max = float(conv.max())

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point called by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes the provided
    ``evaluate_sequence`` function.

    The procedure consists of:
      * loading a previous solution (if any),
      * generating a small pool of analytic / random seeds,
      * picking the best seed according to the exact surrogate,
      * refining it with a fast mass‑transfer hill‑climbing optimizer,
      * finally evaluating the result with ``evaluate_sequence`` and storing it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # small safety margin so we finish before the hard timeout
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # problem size – 4000 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Try to continue from the previous run
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    previous_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                previous_vec = np.maximum(arr, 0.0)
                previous_vec /= previous_vec.sum()
        except Exception:
            previous_vec = None

    # ------------------------------------------------------------------
    # Build a modest pool of analytic / random seeds
    # ------------------------------------------------------------------
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for rate in (0.001, 0.002, 0.003, 0.004):
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-rate * i)
        seeds.append(a / a.sum())

    # symmetric exponential
    for rate in (0.0015, 0.0025, 0.0035):
        i = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        a = np.exp(-rate * np.abs(i - centre))
        seeds.append(a / a.sum())

    # power‑law families
    for p in (0.5, 0.7, 0.9, 1.1):
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = 1.0 / (idx ** p)
        seeds.append(a / a.sum())

    # beta‑shaped families
    for _ in range(5):
        a_par = rng.uniform(0.2, 2.0)
        b_par = rng.uniform(0.2, 2.0)
        i = np.arange(1, n + 1, dtype=np.float64)
        a = (i ** (a_par - 1.0)) * ((n - i + 1) ** (b_par - 1.0))
        seeds.append(a / a.sum())

    # Dirichlet draws with various concentrations
    for conc in (0.1, 0.5, 1.0, 3.0):
        for _ in range(2):
            a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            seeds.append(a)

    if previous_vec is not None:
        seeds.append(previous_vec)

    # ------------------------------------------------------------------
    # Quick surrogate evaluation – keep the best seed
    # ------------------------------------------------------------------
    best_seq = None
    best_surrogate = float('inf')
    for seq in seeds:
        if time.time() > deadline:
            break
        val = _surrogate(seq)
        if val < best_surrogate:
            best_surrogate = val
            best_seq = seq.copy()
    print(f"[Init] Best surrogate among seeds = {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # Hill‑climbing refinement
    # ------------------------------------------------------------------
    if best_seq is None:
        raise RuntimeError("No initial sequence could be generated.")

    print("[HillClimb] Starting local optimisation …")
    seq_opt, max_val = _hill_climb(best_seq, deadline)
    print(f"[HillClimb] Finished – best surrogate = {max_val:.6f}")

    # ------------------------------------------------------------------
    # Final (true) evaluation
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(seq_opt.tolist())
        print(f"=== Result ===\nSurrogate ≈ {max_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning – evaluate_sequence raised:", exc)
        true_val = None

    # Store for the next run
    globals()["height_sequence_1"] = seq_opt.tolist()
    return seq_opt.tolist()
