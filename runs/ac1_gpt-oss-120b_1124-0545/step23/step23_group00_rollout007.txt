"""
Improved multi‑stage optimizer for the step‑function attack.

Goal
----
Find a non‑negative vector a of length n (default 4000) that minimises
    2·n·max(conv(a,a)) / (sum(a)²)

Key ideas
---------
* Use several diversified initialisations (previous best, exponential decay,
  random Dirichlet) – this reduces the chance of getting stuck.
* Run a long Adam optimiser on a *log‑softmax* parametrisation with a
  very‑slow β‑annealing schedule (log‑sum‑exp smoothing of the max).
* Follow with an exact exponentiated‑gradient (mirror‑descent) phase that
  works directly on the true max‑convolution.  The step‑size decays as 1/√t
  but we keep iterating until the time budget is almost exhausted.
* Perform an aggressive simulated‑annealing hill‑climb that uses three move
  types (swap, Gaussian perturb, and a targeted “mass‑shift” that moves a
  small amount from the most‑contributing pair of the active offset to a
  low‑impact position).  The temperature schedule is exponential.
* If time permits, do a few short random‑restart Adam refinements.
* Throughout we keep the current best sequence (according to the exact
  surrogate 2·n·max(conv) – i.e. the objective with sum(a)=1) and finally
  evaluate it with the provided black‑box ``evaluate_sequence`` function.

The whole routine stops a few seconds before the 1000 s deadline; the final
sequence is stored in the global variable ``height_sequence_1`` so that a
future run can continue from the best found solution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective when sum(seq)=1, i.e. 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution objective.
    The variable v_local lives in log‑softmax space; a = softmax(v_local)
    is a probability vector with sum=1.
    """
    a = _softmax(v_local)                     # probability vector
    conv = _conv_fft(a, a)                     # length = 2n‑1
    max_c = np.max(conv)
    shift = max_c                               # numeric stability

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                     # weight for each convolution entry
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]      # slice back to length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 500_000,
              beta_start: float = 0.5,
              beta_end: float = 1e8,
              alpha: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate.
    Returns the best probability vector seen and its surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-9

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for β (log‑space linear)
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent on the true max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 12_000_000,
                     eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *true* max‑convolution.  The iterate stays on the
    simplex via a log‑space representation (softmax).  The step size
    decays as η0 / sqrt(t).  Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb with three move types.
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Local search with swaps, Gaussian perturbations and a targeted
    “mass‑shift” move that tries to directly reduce the active convolution
    entry.  Returns the best sequence seen.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # exponential cooling schedule
    T0 = 1e-2 * cur_val
    T1 = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T0 * (T1 / T0) ** (it / max_iters)   # geometric cooling

        move_type = rng.random()
        if move_type < 0.45:                     # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        elif move_type < 0.90:                    # Gaussian perturb
            sigma = 0.002 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                      # targeted mass‑shift
            # Identify most offending pair for the current max offset
            conv = _conv_fft(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            i_start = max(0, max_idx - (n - 1))
            i_end = min(n - 1, max_idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = max_idx - idx_i
            contrib = cur_seq[idx_i] * cur_seq[rev_idx]
            if contrib.size == 0:
                continue
            p = int(idx_i[np.argmax(contrib)])     # donor index
            q = int(rng.integers(0, n))           # random receiver
            if p == q:
                continue
            delta = min(cur_seq[p], 0.5 * cur_seq[p]) * 0.1   # small fraction
            ns = cur_seq.copy()
            ns[p] -= delta
            ns[q] += delta
            # renormalise (the move preserves sum, but rounding may drift)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val

            deadline=deadline,
            rng=rng,
            max_iters=4_500_000,
        )
        if val_hc < best_val:
            best_val, best_seq = val_hc, seq_hc.copy()
            print(f"    Improvement → surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional short random‑restart Adam refinements (if time left)
    # ------------------------------------------------------------------
    while time.time() + 30.0 < deadline:
        print("\n[Stage 4] Random‑restart Adam (final polishing)")
        # inject a modest Gaussian noise into the current best
        noisy = best_seq + 0.005 * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy = noisy / s

        sub_deadline = time.time() + 25.0
        seq_rr, val_rr = _adam_opt(
            start_seq=noisy,
            n=n,
            deadline=sub_deadline,
            max_iters=90_000,
            beta_end=5e7,
            alpha=0.28,
        )
        print(f"    restart surrogate = {val_rr:.6f}")
        if val_rr < best_val:
            best_val, best_seq = val_rr, seq_rr.copy()
            print("    *** improvement found! ***")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the black‑box function
    # ------------------------------------------------------------------
    final_val = None
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {final_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
