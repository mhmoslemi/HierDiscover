"""
Optimiser for the max‑convolution bound.

The algorithm works in three stages on a single promising seed
(the previous best sequence when available, otherwise the classic
inverse‑sqrt shape).  All iterates stay on the probability simplex via
a soft‑max parametrisation.

1. Exponentiated‑gradient (sub‑gradient of the exact max‑convolution).
   A few‑million cheap updates quickly drive the surrogate value down.
2. Smooth‑max optimisation with L‑BFGS‑B.
   We gradually increase the “β’’ parameter of a log‑sum‑exp
   approximation to the max and run many iterations at each β.
3. Simulated‑annealing polishing with a long linear cooling schedule.

The routine respects a 1000 s wall‑clock limit, stores the best found
sequence in the global variable ``height_sequence_1`` and returns it.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Optional scientific imports
# --------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# --------------------------------------------------------------
# Basic numeric utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# --------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# --------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (sum(seq) == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# --------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# --------------------------------------------------------------
# Exponentiated‑gradient on the exact max‑convolution
# --------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 2_500_000,
                      eta0: float = 12.0,
                      tol: float = 1e-10) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    idx = np.arange(n, dtype=np.int64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices where the convolution is within tolerance of the maximum
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient (average over active indices)
        g = np.zeros_like(a)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑like momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# --------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp)
# --------------------------------------------------------------
def _smooth_obj_grad(v: np.ndarray, n: int, beta: float):
    """
    Objective and gradient for the smoothed max (conv) using log‑sum‑exp.
    The parametrisation is v = log‑softmax → a = softmax(v).
    Returns (obj, grad_v).
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)
    shift = conv.max()
    # log‑sum‑exp (stable)
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_beta / sum_exp                      # weights for the smooth max
    rev_a = a[::-1]
    # derivative of max w.r.t. a (convolution with w)
    full = _conv_fft(rev_a, w)                  # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # convert to gradient w.r.t. log‑softmax parameters
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _lbfgs_smooth_optimize(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimise the smoothed max‑convolution via L‑BFGS‑B.
    A schedule of increasing β is used (continuation method).
    """
    if _opt is None:
        # SciPy not available → return the start vector unchanged
        return start_seq

    # initial log‑softmax parameters
    v = np.log(start_seq + 1e-30)

    # continuation schedule (log‑spaced)
    beta_schedule = [
        0.5, 1.0, 2.0, 5.0, 10.0,
        20.0, 50.0, 100.0, 200.0,
        500.0, 1_000.0, 2_000.0,
        5_000.0, 10_000.0, 20_000.0,
        50_000.0, 100_000.0, 200_000.0,
        500_000.0, 1_000_000.0,
    ]

    # small callback to respect the global deadline
    def _callback(x):
        if time.time() > deadline:
            raise KeyboardInterrupt()

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        try:
            res = _opt.minimize(
                fun=_smooth_obj_grad,
                x0=v,
                args=(n, beta),
                method='L-BFGS-B',
                jac=True,
                callback=_callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-13}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta}: {e}")
            break

    return _softmax(v)


# --------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# --------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 1_200_000,
                         start_sigma: float = 0.0015,
                         T_start: float = 0.012,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# --------------------------------------------------------------
# Main optimisation driver (single seed)
# --------------------------------------------------------------
def _optimise_from_seed(seed_vec: np.ndarray,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run the three‑stage optimisation pipeline (EG → L‑BFGS → SA).
    Returns the best sequence and its surrogate value.
    """
    # 1) Exponentiated‑gradient (max‑convolution)
    sub_dead = time.time() + max(0.0, (deadline - time.time()) * 0.35)
    seq_eg, val_eg = _exp_gradient_opt(seed_vec, n, sub_dead,
                                       max_iters=2_200_000,
                                       eta0=14.0,
                                       tol=1e-10)

    # 2) Smooth‑max L‑BFGS
    sub_dead = time.time() + max(0.0, (deadline - time.time()) * 0.45)
    seq_lbfgs = _lbfgs_smooth_optimize(seq_eg, n, sub_dead,
                                       max_iter_per_beta=5000)
    val_lbfgs = _objective_surrogate(seq_lbfgs)

    # 3) Simulated annealing polish
    sub_dead = deadline  # use whatever time is left
    seq_sa, val_sa = _simulated_annealing(seq_lbfgs, n, sub_dead,
                                          rng,
                                          max_attempts=1_100_000,
                                          start_sigma=0.001,
                                          T_start=9e-3,
                                          T_end=5e-9)

    # Pick the best among the three stages (plus the original seed)
    candidates = [
        (seed_vec, _objective_surrogate(seed_vec)),
        (seq_eg, val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_sa, val_sa),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val


# --------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    stores the best found sequence in the global variable
    ``height_sequence_1`` and returns it as a Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # Problem size – a compromise between expressive power and speed
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous best if present (helps continuity)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            tmp = np.asarray(prev, dtype=np.float64)
            if tmp.shape == (n,):
                prev_arr = tmp.copy()
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # Build a tiny seed set (previous best + a classic analytic shape)
    # --------------------------------------------------------------
    seeds = []

    # classic inverse‑sqrt (known strong baseline)
    seeds.append(_inverse_sqrt(n))

    # uniform – helps avoid pathological local minima
    seeds.append(_uniform(n))

    # triangular – a simple alternative
    seeds.append(_triangular(n))

    # a few Gaussian shapes (different spreads)
    for sigma_frac in [0.07, 0.12, 0.20]:
        seeds.append(_gaussian(n, sigma_frac))

    # previous solution if available
    if prev_arr is not None and prev_arr.sum() > 0.0:
        base = prev_arr / prev_arr.sum()
        seeds.append(base)
        # a few random perturbations of the previous best
        for _ in range(4):
            pert = base + 0.006 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0.0:
                seeds.append(pert / s)

    # --------------------------------------------------------------
    # Quick surrogate evaluation of seeds – keep the most promising
    # --------------------------------------------------------------
    evals = []
    for vec in seeds:
        if time.time() > deadline:
            break
        # ensure non‑negative and normalise
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evals.append((vec, val))

    if not evals:
        # fallback – uniform (should never happen)
        best_start = _uniform(n)
    else:
        evals.sort(key=lambda x: x[1])
        best_start = evals[0][0]
        print("\nSeed surrogate values (best first):")
        for i, (_, v) in enumerate(evals[:5]):
            print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # Main optimisation on the chosen seed
    # --------------------------------------------------------------
    print("\n--- Optimisation start ---")
    best_seq, best_val = _optimise_from_seed(best_start, n, deadline, rng)
    print(f"\n[Result] best surrogate after pipeline = {best_val:.6f}")

    # --------------------------------------------------------------
    # Final evaluation using the supplied black‑box
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
