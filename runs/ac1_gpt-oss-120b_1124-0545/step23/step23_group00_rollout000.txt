"""
Step‑function optimisation for minimise 2·n·max(conv(a,a)) / (sum(a)²).

Approach
--------
1.  Initialise with a decreasing exponential (or reuse the previous best).
2.  Multiple short Adam runs on a smooth log‑sum‑exp surrogate of the
    max‑convolution (log‑softmax parametrisation guarantees non‑negativity
    and the simplex constraint).  The best Adam result is kept.
3.  Additive projected sub‑gradient descent on the *exact* max‑convolution
    (sub‑gradient of the active entry).  Each step is projected onto the
    probability simplex.
4.  Simulated‑annealing hill‑climb that uses two move types:
        – swap two coefficients,
        – transfer a small amount of mass from one coefficient to another.
    The temperature cools linearly; worsening moves are accepted with the
    Metropolis rule.
5.  Finally evaluate the best sequence with the provided black‑box
    ``evaluate_sequence`` and store it for possible continuation runs.

All stages respect the global time‑budget (≈1000 s).  The code only uses
NumPy (and the already‑available ``evaluate_sequence``); no file I/O or
network access is performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv(seq,seq))  (sum(seq)=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)  (normalised to sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0.0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – used by Adam
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution objective.
    The optimisation variables are logits v such that a = softmax(v) (sum = 1).
    """
    a = _softmax(v_local)                     # probability vector
    conv = _conv_fft(a, a)                    # length = 2n‑1
    max_c = np.max(conv)
    shift = max_c

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient wrt a
    w = exp_beta / sum_exp                    # weights over the convolution bins
    rev_a = a[::-1]
    # convolution of rev_a (length n) with w (length 2n‑1) gives length 3n‑2
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n - 1 : n - 1 + n]     # slice back to length n

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 200_000,
              beta_start: float = 0.5,
              beta_end: float = 1_000_000.0,
              alpha: float = 0.5) -> tuple[np.ndarray, float]:
    """Adam on the smooth surrogate.  Returns best probability vector seen."""
    v = np.log(start_seq + 1e-30)     # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for smoothing parameter β
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    # shift by the mean to improve numerical stability
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen, but fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise to counter tiny rounding errors
    w_sum = w.sum()
    if w_sum > 0.0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


# ----------------------------------------------------------------------
# Additive projected sub‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------
def _proj_subgradient_opt(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 2_000_000,
                          eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution.
    Each iteration uses the sub‑gradient of the active entry,
    then projects back onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        a_candidate = a - eta * g
        a = _project_onto_simplex(a_candidate)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (swap + mass‑transfer moves)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly swaps two coefficients or transfers a small
    amount of mass from one coefficient to another.  A decreasing temperature
    accepts worsening moves with probability exp(-(Δ)/T).
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule (linear cooling)
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                     # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                         # transfer move
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_seq[i] == 0.0:
                continue
            frac = rng.random() * 0.10               # up to 10 % of source mass
            delta = frac * cur_seq[i]
            ns = cur_seq.copy()
            ns[i] -= delta
            ns[j] += delta

        # the move preserves the sum, but due to rounding we re‑normalise
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            continue
        ns /= s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val
