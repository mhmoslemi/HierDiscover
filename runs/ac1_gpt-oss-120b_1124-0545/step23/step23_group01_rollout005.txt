    log_a = np.log(start_seq + 1e-30)
    m=zeros; v=zeros; λ=λ0
    best_seq = _softmax(log_a); best_val = _objective_surrogate(best_seq)
    for t in range(1, max_iters+1):
        if time.time()>deadline: break
        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        # compute smoothed gradient weights
        exp_term = np.exp(λ * (conv - max_c))
        sum_exp = np.sum(exp_term)
        if sum_exp == 0:
            weights = np.full_like(conv, 1.0/len(conv))
        else:
            weights = exp_term / sum_exp
        # gradient w.r.t a:
        a_rev = a[::-1]
        corr = _conv_fft(weights, a_rev)
        grad_a = 2.0 * corr[n-1 : n-1+n]
        # chain rule to log space
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)
        # Adam
        m = β1*m + (1-β1)*grad_log
        v = β2*v + (1-β2)*(grad_log**2)
        m_hat = m/(1-β1**t)
        v_hat = v/(1-β2**t)
        step = η0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step
        # optionally evaluate surrogate (actual) every few iter
        if t % 1000 == 0:
            cur_val = _objective_surrogate(_softmax(log_a))
            if cur_val < best_val:

                best_seq = _softmax(log_a).copy()
        # increase λ
        λ = min(λ*λ_factor, λ0 * (λ_factor**max_iters)) # but maybe just λ *= λ_factor each iteration.

        # Could also cap λ to some max, e.g., 20000.

    return best_seq, best_val

"""
Optimisation of a non‑negative step‑function (probability vector) for the
objective

    2·n·max(conv(a,a)) / (sum a)^2            (scale‑invariant)

The code uses two phases:

1. **Smoothed Adam** – the non‑smooth max is replaced by a
   log‑sum‑exp (temperature λ).  λ is increased linearly from a small
   value (λ≈0.1) to a large one (λ≈2·10⁴);  the gradient of the smooth
   surrogate is a weighted combination of the gradients of all
   convolution entries.  The optimisation runs in the log‑softmax
   (mirror‑descent) parametrisation and employs Adam moments.

2. **Final polishing** – after the smoothing phase the exact max‑convolution
   gradient (active‑index sub‑gradient) is used (again with Adam) to
   fine‑tune the solution.

Multiple diverse seeds (previous best, exponential decays, triangular,
uniform, random Dirichlet draws) are examined; the best sequence found
within the time budget (≈1000 s) is returned and stored in the global
variable ``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real inputs → real output)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective used for ranking candidates:
       2·n·max(conv(seq,seq))  (sum(seq) is 1 by construction)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _triangular(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Adam optimiser for the *exact* max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 500_000,
              lr0: float = 0.2,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam mirror descent on the true max‑convolution objective.
    Returns the best probability vector seen and its objective value.
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active entry (same as in the reference code)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g_a = np.zeros_like(a)
        g_a[idx_i] = 2.0 * a[rev_idx]

        # chain rule for log‑softmax variables
        dot = np.dot(g_a, a)
        grad_log = a * (g_a - dot)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        if t % 5000 == 0:
            cur_val = _objective_surrogate(_softmax(log_a))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax, log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_adam_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     lambda0: float = 0.1,
                     lambda_max: float = 20000.0,
                     eta0: float = 0.5,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate
        F_λ(a) = 2·n· (log Σ exp(λ·conv_i)) / λ .
    λ is increased linearly from `lambda0` to `lambda_max` over the run.
    Returns the best probability vector seen and its true (non‑smooth)
    objective value.
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current temperature
        lam = lambda0 + (lambda_max - lambda0) * t / max_iters

        a = _softmax(log_a)

        # convolution a * a
        conv = _conv_fft(a, a)

        # log‑sum‑exp (stable)
        max_c = np.max(conv)
        exp_shift = np.exp(lam * (conv - max_c))
        sum_exp = np.sum(exp_shift)
        if sum_exp == 0.0:
            # all weights would be zero – fall back to uniform weighting
            weights = np.full_like(conv, 1.0 / conv.size)
        else:
            weights = exp_shift / sum_exp          # Σ weights = 1

        # gradient w.r.t a : 2 * cross‑correlation(weights, a)
        # cross‑correlation = (weights ⋆ a)[i] = Σ_j weights_{i+j} a_j
        a_rev = a[::-1]
        corr_full = _conv_fft(weights, a_rev)        # length = 3n‑2
        grad_a = 2.0 * corr_full[n - 1:n - 1 + n]    # slice of length n

        # chain rule for log‑softmax variables
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # occasional true‑objective evaluation
        if t % 5000 == 0:
            cur_val = _objective_surrogate(_softmax(log_a))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(log_a).copy()

    # final evaluation before returning
    cur_val = _objective_surrogate(_softmax(log_a))
    if cur_val < best_val:
        best_val = cur_val
        best_seq = _softmax(log_a).copy()

    return best_seq, best_val
