    Returns np.inf if the input is invalid.

        # any other non-integer/non-float types (like strings or complex numbers).

            return np.inf

        if np.isnan(x) or np.isinf(x):

    sequence = [float(x) for x in sequence]

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq))  (seq sums to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)

    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()

    idx = np.arange(1, n + 1, dtype=np.float64)

    i = np.arange(1, n + 1, dtype=np.float64)

class _LbfgsObjective:
    """Callable needed by scipy.optimize.minimize."""

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, log_a: np.ndarray):
        # soft‑max to a probability vector
        max_log = np.max(log_a)
        exp_shift = np.exp(log_a - max_log)
        a = exp_shift / exp_shift.sum()

        beta = self.beta

        sum_exp = np.sum(exp_beta)

        # smoothed objective value: M_β = shift + log(sum_exp)/β

        return obj, grad_log


def _lbfgs_opt(start_seq: np.ndarray,

def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,

    cur_seq = start_seq.copy()

    best_seq = cur_seq.copy()
    best_val = cur_val

    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)

            cur_val = val
            if val < best_val:

                best_val = val
