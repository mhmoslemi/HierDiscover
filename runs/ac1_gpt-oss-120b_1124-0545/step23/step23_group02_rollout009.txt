"""
Improved step‑function optimiser (≈ 1 000 s)

Goal: minimise
    2·n·max(conv(a,a)) / (sum(a)²)
for a non‑negative vector a.  The objective is invariant under scaling, so we
optimise over the probability simplex (∑a = 1).  The algorithm works as follows:

1. Build a small, diverse seed set (uniform, power‑law, exponential,
   a “Fourier‑sqrt’’ seed obtained by solving  a*a ≈ uniform  in the
   frequency domain, random Dirichlet draws and the previous best if any).

2. For each seed run a *soft‑max‑gradient* phase:
   – minimise a smooth approximation of the max‑convolution
     (soft‑max with a temperature that is annealed to 0);
   – the gradient is obtained analytically by a correlation of the current
     distribution with the soft‑max weights;
   – updates are performed in the log‑softmax (exponentiated‑gradient) space,
     keeping the iterate on the simplex automatically.

3. Keep the best sequence seen across all seeds; stop early as soon as the
   surrogate drops below the target 1.503.

4. Return the best sequence and store it in the global variable
   ``height_sequence_1`` for warm‑starts in later runs.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _corr_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear correlation (a ⋆ w)[lag] = Σ_i a_i w_{i+lag} .
    Returns the full‑length correlation array.
    """
    L = _next_pow2(len(a) + len(w) - 1)
    Fa = np.fft.fft(a, n=L)
    Fw = np.fft.fft(w, n=L)
    corr = np.fft.ifft(Fa * np.conj(Fw)).real
    return corr[: len(a) + len(w) - 1]


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (used for quick ranking)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (∑seq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed constructions
# ----------------------------------------------------------------------
def _fourier_sqrt_seed(n: int) -> np.ndarray:
    """
    Solve a*a ≈ uniform on {0,…,2n‑2} in the Fourier domain:
        A(ω)^2 ≈ FT[uniform].
    The resulting vector is clipped to non‑negative values and
    ℓ₁‑normalised.
    """
    L = _next_pow2(2 * n - 1)
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)
    Ft = np.fft.fft(target, n=L)
    sqrt_Ft = np.sqrt(Ft)                     # principal complex square‑root
    a_complex = np.fft.ifft(sqrt_Ft).real[:n]  # take the real part
    a = np.maximum(a_complex, 0.0)            # enforce non‑negativity
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


def _make_power_seed(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    a /= a.sum()
    return a


def _make_exp_seed(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Core optimiser: soft‑max gradient on the exact max‑convolution
# ----------------------------------------------------------------------
def _softmax_max_opt(start_vec: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     eta0: float = 8.0,
                     T0: float = 0.2,
                     T_min: float = 1e-7,
                     beta_mom: float = 0.9,
                     check_every: int = 200) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient on the smoothed max of the convolution.
    Returns the best vector encountered and its surrogate value.
    """
    log_a = np.log(start_vec + 1e-30)           # log‑softmax parameters
    momentum = np.zeros_like(start_vec)

    best_seq = None
    best_val = np.inf

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)                    # a stays on the simplex
        conv = _conv_fft(a, a)                 # linear convolution
        max_conv = conv.max()

        # temperature schedule (geometric decay)
        T = max(T_min, T0 * (0.9995 ** it))

        # soft‑max weights over the convolution entries
        # (stable formulation)
        shift = conv - max_conv
        w = np.exp(shift / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient of the smooth max w.r.t. a:
        #   ∂/∂a  Σ_k w_k (a*a)_k = 2 * (a ⋆ w)[0:n]
        grad = 2.0 * _corr_fft(a, w)[:n]

        # Adam‑like momentum (only the first‑order term)
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad

        # step‑size schedule (inverse‑sqrt)
        eta = eta0 / math.sqrt(it)

        # log‑space update (exponentiated‑gradient)
        log_a = log_a - eta * momentum

        # occasional evaluation on the true surrogate
        if it % check_every == 0:
            val = _objective_surrogate(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
            # early exit if we have already crossed the target
            if best_val < 1.503:
                break

    # fallback to the last iterate if we never improved
    if best_seq is None:
        best_seq = _softmax(log_a)
        best_val = _objective_surrogate(best_seq)

    return best_seq, best_val
