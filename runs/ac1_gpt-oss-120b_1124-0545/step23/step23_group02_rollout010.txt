def smooth_max_opt(start_seq, n, deadline, max_iters=800000, eta0=0.1, lambda0=1.0, lambda_final=500.0):
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = softmax(v)
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        # compute softmax weights
        lam = lambda0 * (lambda_final / lambda0) ** (it / max_iters)
        # stable compute exp(lam*conv - max)
        maxc = conv.max()
        exp_vals = np.exp(lam * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # compute gradient w.r.t a
        # grad = 2 * conv_rev[(len(w)-1)-i]
        rev_w = w[::-1]
        grad_full = np.convolve(rev_w, a)  # full int conv of length 3n-2
        # Extract gradient segment
        L = len(w)  # 2n-1
        # grad_i = 2 * grad_full[L-1 - i] for i in range(n)
        grad_a = 2.0 * grad_full[L-1 : L-1 - n : -1]   # need proper slicing
        # Actually slice: indices from (L-1 - 0) down to (L-1 - (n-1))
        # So gradient vector = grad_full[(L-1) : (L - n) : -1]
        grad_a = 2.0 * grad_full[(L-1) : (L - n - 1) : -1] # inclusive of L-n? check.
        # Let's compute using np.arange: idx = L-1 - np.arange(n)
        idx = (L-1) - np.arange(n)
        grad_a = 2.0 * grad_full[idx]
        # chain rule for v
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1-beta1) * grad_v
        v_hat = beta2 * v_hat + (1-beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1**it)  # bias-corrected
        v_hat_corr = v_hat / (1 - beta2**it)

        # evaluate surrogate

        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:

                return best_seq, best_val
            a = _softmax(log_a)
            conv = _conv_fft(a, a)
            c = lam * conv
            c_max = c.max()
            exp_c = np.exp(c - c_max)
            sum_exp = exp_c.sum()
            w = exp_c / sum_exp
            # gradient
            rev_w = w[::-1]
            full = _conv_fft(rev_w, a)
            idx = (len(w)-1) - np.arange(n)
            grad_a = 2.0 * full[idx]
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            # momentum update
            eta = 0.8 / math.sqrt(it_total+1)
            momentum = beta * momentum + (1-beta) * grad_v
            log_a -= eta * momentum
            # evaluate
            cur_seq = _softmax(log_a)
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
            it_total += 1
    return best_seq, best_val

"""
Improved step‑function optimiser (≈ 1 000 s budget)

The objective 2·n·max(conv(a,a))/(∑a)² is scale‑invariant, so we work on the
probability simplex (∑a=1).  The core idea is to minimise a *smoothed*
approximation of the maximum convolution value:

    Φ_λ(a) = (1/λ)·log(∑_k exp(λ·(a∗a)_k))

For small λ the function is smooth and encourages the whole convolution to
be flat; as λ grows it converges to the true max.  We minimise Φ_λ with a
log‑softmax parametrisation (v↦softmax(v)) using SciPy’s L‑BFGS‑B
(if available) or a simple Adam‑style mirror‑descent fallback.  The values
of λ are increased in stages (λ∈[0.5,2,10,50,200,800]) so that the optimiser
first flattens the convolution and then focuses on the peak.  After the
smoothed‑max phase we polish the result with the exact max‑subgradient
method from the previous baseline (exponentiated‑gradient with an active‑set).

The routine respects the 1000 s wall‑clock limit, stores the best sequence
found in the global variable ``height_sequence_1`` and returns it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (∑=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, ∑a=1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv(seq,seq)) for ∑seq=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _uniform(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Smoothed‑max optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _smoothmax_lbfgs_opt(start_vec: np.ndarray,
                         n: int,
                         deadline: float,
                         lam_schedule,
                         max_iter_per_lambda: int = 250):
    """
    Optimises the smooth‑max surrogate Φ_λ via L‑BFGS‑B (if SciPy is present)
    or a simple Adam‑style mirror‑descent fallback.
    Returns the best sequence (∑=1) and its surrogate value.
    """
    # log‑softmax parameters
    v = np.log(start_vec + 1e-30)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper that evaluates Φ_λ and its gradient w.r.t. v
    # ------------------------------------------------------------------
    def _eval(v_local, lam):
        a = _softmax(v_local)
        conv = _conv_fft(a, a)                     # a∗a
        # ---- smooth max ----
        c = lam * conv
        c_max = c.max()
        exp_c = np.exp(c - c_max)                  # stability
        sum_exp = exp_c.sum()
        phi = (np.log(sum_exp) + c_max) / lam      # Φ_λ(a)

        w = exp_c / sum_exp                        # ∂Φ/∂conv = softmax(lam·conv)

        # ---- gradient wrt a (2·(w∗a) with proper index handling) ----
        rev_w = w[::-1]
        full = _conv_fft(rev_w, a)                # full convolution length = 3n‑2
        L = len(w)                                # = 2n‑1
        idx = (L - 1) - np.arange(n)              # indices giving ∑_j w_{i+j} a_j
        grad_a = 2.0 * full[idx]

        # ---- chain rule for v (log‑softmax) ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return phi, grad_v

    # ------------------------------------------------------------------
    # L‑BFGS branch (if SciPy is available)
    # ------------------------------------------------------------------
    if _opt is not None:
        for lam in lam_schedule:
            if time.time() > deadline:
                break

            # Wrap the objective for SciPy (returns (f,grad))
            def fun(v_loc):
                # abort if out of budget
                if time.time() > deadline:
                    raise KeyboardInterrupt()
                f, g = _eval(v_loc, lam)
                return f, g

            # small callback to enforce the deadline
            def callback(_):
                if time.time() > deadline:
                    raise KeyboardInterrupt()

            try:
                res = _opt.minimize(fun=fun,
                                    x0=v,
                                    method='L-BFGS-B',
                                    jac=True,
                                    callback=callback,
                                    options={'maxiter': max_iter_per_lambda,
                                             'ftol': 1e-12,
                                             'disp': False})
                v = res.x
            except KeyboardInterrupt:
                # time ran out – keep current v
                pass
            # record best after this λ‑stage
            cur_seq = _softmax(v)
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
        return best_seq, best_val

    # ------------------------------------------------------------------
    # Fallback: simple Adam‑style mirror descent on the smooth‑max
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    eps = 1e-8
    it_total = 0

    for lam in lam_schedule:
        if time.time() > deadline:
            break
        for _ in range(max_iter_per_lambda):
            if time.time() > deadline:
                break
            it_total += 1
            a = _softmax(v)
            conv = _conv_fft(a, a)

            c = lam * conv
            c_max = c.max()
            exp_c = np.exp(c - c_max)
            sum_exp = exp_c.sum()
            w = exp_c / sum_exp

            rev_w = w[::-1]
            full = _conv_fft(rev_w, a)
            L = len(w)
            idx = (L - 1) - np.arange(n)
            grad_a = 2.0 * full[idx]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update on v (log‑parameters)
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** it_total)
            v_hat_corr = v_hat / (1.0 - beta2 ** it_total)
            step = 0.8 / (np.sqrt(v_hat_corr) + eps)
            v -= step * m_hat

            # record best sequence seen so far
            cur_seq = _softmax(v)
            cur_val = _objective_surrogate(cur_seq)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact max‑subgradient optimiser (exponentiated gradient, mirrors the baseline)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 800_000,
                            eta0: float = 8.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution (multi‑active set).
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of the max (averaged over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑like momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val
