"""
Propose a step‑function with a very low upper‑bound for ``evaluate_sequence``.

Algorithm
---------
1️⃣  Build a *large* and *diverse* pool of simplex seeds.
    • Uniform, triangular and arcsine‑like families.
    • Arcsine‑like with an extra exponent γ (a_i ∝ ((i+½)(n‑½‑i))⁻ᵞ).
    • Symmetric Beta(α,α) for many α.
    • A few mixtures of arcsine and uniform.
    • Random Dirichlet draws (different concentrations).

    All seeds are normalised to sum = 1 and scored with the cheap surrogate
    ``2·n·max(conv(a,a))``.  The ten best seeds are kept.

2️⃣  **Adam smoothing** – optimise the *smooth* max‑convolution.
    The smooth max is  `log‑sum‑exp(β·conv)` with a schedule of β that ends at
    1e8.  For each β we run many Adam steps (≈12 000) to let the optimiser
    settle before the next β is introduced.

3️⃣  **Adam on the exact max** – we keep Adam (instead of plain
    exponentiated‑gradient) but now the gradient is the true sub‑gradient of
    the max‑convolution (the active index only).  This combines the stability
    of Adam with the exact objective.

4️⃣  **Simulated annealing walk** – a tiny stochastic walk on the simplex.
    Proposals transfer a small random amount of mass from one coordinate to
    another.  Improvements are always accepted; worsening moves are accepted
    with probability exp(−Δ/T) where the temperature T decays geometrically.
    This stage helps escape shallow local minima that Adam can miss.

5️⃣  Finally the candidate is passed to the supplied ``evaluate_sequence``
    and stored globally as ``height_sequence_1`` for the next call.

The whole routine respects the 1000 s budget (a 2‑second safety margin) and
prints progress information.  All helper functions are top‑level, use only
NumPy and the standard library, and avoid any filesystem or network I/O.
"""

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # should never happen, but fall back to uniform
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2·n·max(conv(seq, seq))   (seq is assumed to sum to 1)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a *probability* vector that sums to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _arcsine_like(n: int, gamma: float = 0.5) -> np.ndarray:
    """
    Generalised arcsine family:
        a_i ∝ ((i+½)(n‑½‑i))⁻ᵞ
    gamma=0.5 reproduces the classic arcsine distribution.
    """
    i = np.arange(n, dtype=np.float64)
    eps = 1e-15
    denom = (i + 0.5) * (n - 0.5 - i) + eps
    a = np.power(denom, -gamma)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix(a: np.ndarray, b: np.ndarray, lam: float) -> np.ndarray:
    """Mixture lam·a + (1‑lam)·b, renormalised to sum = 1."""
    mixed = lam * a + (1.0 - lam) * b
    return mixed / mixed.sum()

def _adam_smoothing(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the *smooth* max‑convolution.
    Returns the best probability vector found and its exact (non‑smoothed)
    surrogate value.
    """
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)               # initialise logits
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a richer β‑schedule – we finish at 1e8 to obtain a very sharp max
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0, 50_000_000.0, 1_00_000_000.0
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # inner loop – a bit longer than the original code
        max_iter_beta = 12_000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)               # current point on the simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv wrt a  (see original code)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)        # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            # convert to gradient in log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.12
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _adam_subgrad_opt(start_seq: np.ndarray,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Adam on the *exact* max‑convolution.
    Sub‑gradient is taken at the active index (the index attaining the max).
    """
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    a = _softmax(logit)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)          # basic decay
        logit = logit - eta * m_hat / (np.sqrt(v_hat) + eps)

        a = _softmax(logit)

        if it % 2500 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.02,
                         decay: float = 0.9997,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny simulated‑annealing walk on the simplex.
    - Transfer a small random amount of mass from i → j.
    - Accept improvements always.
    - Accept worsening moves with probability exp(-Δ/T).
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)

    best_seq = cur.copy()
    best_val = cur_val

    T = init_temp * cur_val          # temperature scales with current objective

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # amount to move: a fraction of the source coordinate
        delta = sigma * cur[i] * rng.random()
        if delta <= 0.0:
            continue

        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        if new[i] < 0.0:
            continue

        new_val = _objective_surrogate(new)

        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(new_val - cur_val) / T) if T > 0 else 0.0
            accept = rng.random() < prob

        if accept:
            cur, cur_val = new, new_val
            # temperature cooling
            T *= decay

            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

    return best_seq, best_val

    # ------------------------------------------------------------------
    # 0️⃣ Assemble a rich seed pool
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _triangular(n)]

    # classic arcsine (γ = 0.5) and a few neighbours
    for gamma in (0.4, 0.45, 0.5, 0.55, 0.6):
        seeds.append(_arcsine_like(n, gamma))

    # Beta‑shaped seeds (α varies)
    for α in (0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7, 0.9, 1.2, 1.5, 2.0):
        seeds.append(_beta_distribution(n, α))

    # Dirichlet draws with several concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Mixtures of arcsine and uniform – a cheap way of exploring intermediate shapes
    arcs = _arcsine_like(n, 0.5)
    unif = _uniform(n)
    for lam in (0.2, 0.4, 0.6, 0.8):
        seeds.append(_mix(arcs, unif, lam))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣ Initial cheap surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    seed_infos = []
    print("\n--- Evaluating seed pool (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")

    # Keep the ten best seeds for the next stage
    seed_infos.sort(key=lambda x: x[0])
    top_k = min(10, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Adam smoothing on the *smooth* max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.30
        print("\n--- Adam smoothing (smooth max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 3️⃣ Adam on the exact max‑convolution (sub‑gradient Adam)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 3.0          # keep a few seconds for polishing
        print("\n--- Adam on exact max‑convolution (sub‑gradient) ---")
        seq_eg, val_eg = _adam_subgrad_opt(best_seq, eg_deadline, eta0=5.0)
        print(f"[Adam‑Exact] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (Adam‑Exact)")

    # ------------------------------------------------------------------
    # 4️⃣ Simulated‑annealing walk (final refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, deadline,
                                              init_temp=0.02,
                                              decay=0.9996,
                                              sigma=0.02)
        print(f"[SA] exact surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_val, best_seq = val_sa, seq_sa.copy()
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 5️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next call and return
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
