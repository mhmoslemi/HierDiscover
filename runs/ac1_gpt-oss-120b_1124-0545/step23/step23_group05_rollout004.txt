"""
Improved step‑function optimizer.

Main ideas
~~~~~~~~~~
* The objective  2·n·max(conv(a,a))  depends only on the *shape* of a,
  i.e. on a normalized vector (sum = 1).  For the continuous problem the
  optimal shape is a(t) ∝ 1/√t, which is approximated by a_i ∝ 1/√i.
  Small changes of the discretisation (shifted index, different power)
  can improve the discrete maximum noticeably.
* We therefore build an extensive **parameteric seed set**:
      – pure power laws a_i ∝ (i+α)^‑p   with p≈0.5 and a shift α∈[‑0.4,0.4]
      – symmetric mixtures of a decreasing and its reversal (mountain
        shape) which tend to flatten the autocorrelation.
      – classic window families, random Dirichlet draws and the existing
        seeds from the reference implementation.
* All seeds are normalised and their *surrogate* value
  f(a)=2·n·max(conv(a,a)) is computed instantly (FFT, O(n log n)).
  The ten best seeds are kept.
* Each of those seeds is refined with a **four‑stage pipeline** identical
  to the reference code, but the exponentiated‑gradient (EG) stage is
  given a larger budget (more iterations, Adam momentum).  The final
  polishing stage (simulated annealing) is also run longer.
* The whole search respects a 1000 s wall‑clock limit, prints progress,
  stores the best sequence in the global variable ``height_sequence_1``
  and finally returns it as a plain Python list.

The optimizer is deterministic only through the supplied ``seed``
parameter; the random generator is seeded once at the start.
"""

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate (sum(seq)=1): 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Analytic families (all return ℓ₁‑normalised vectors)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ 1/(i+1)^p."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_offset(n: int, p: float, offset: float) -> np.ndarray:
    """
    a_i ∝ (i+1+offset)^‑p   (offset may be negative, must keep argument >0).
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    # enforce positivity – values that become non‑positive are set to a tiny epsilon
    idx = np.where(idx > 0, idx, 1e-12)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _symmetric_power(n: int, p: float) -> np.ndarray:
    """
    Convex mixture of a decreasing power law and its reverse:
        a = (dec + rev(dec))/2   and renormalised.
    """
    dec = _power(n, p)
    rev = dec[::-1]
    a = (dec + rev) * 0.5
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, renormalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window – requires scipy.signal."""
    if _sigwin is None:
        return _uniform(n)
    a = _sigwin.dpss(n, nw, Kmax=1, sym=True)
    a = np.abs(a[0])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 1_000_000,
                            eta0: float = 12.0,
                            tol: float = 1e-10) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient on the exact max‑convolution.
    Uses all indices whose value is within ``tol`` relative of the current max.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices whose conv is close to the maximum
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient (average over active indices)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)

        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]

        if active.size:
            g = 2.0 * g / active.size

        # Adam‑like momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 800_000,
                         eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the quadratic loss Σ (conv_k - avg)².
    It tries to flatten the whole convolution.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # length of full convolution

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when sum=1
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent step (natural gradient on simplex)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """
    Smooth‑max optimisation based on a log‑sum‑exp surrogate.
    Optimised with L‑BFGS‑B in the log‑softmax parametrisation.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 300_000,
                         start_sigma: float = 0.001,
                         T_start: float = 8e-3,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbation SA with a linear temperature schedule.
    Returns the best sequence observed before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


def _full_optimisation(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Full four‑stage pipeline (EG → L‑BFGS → flatten → SA) applied to *start_vec*.
    All phases respect the shared *deadline*.  Returns the best sequence
    obtained and its surrogate value.
    """
    # 1) exponentiated‑gradient (max‑convolution)
    seq_eg, val_eg = _exp_gradient_opt_multi(
        start_vec.copy(),
        n,
        deadline,
        max_iters=2_500_000,
        eta0=12.0,
        tol=1e-10,
    )

    # 2) smooth‑max L‑BFGS
    if time.time() < deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(
            seq_eg,
            n,
            deadline,
            beta_schedule=None,
            max_iter_per_beta=3000,
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, val_eg

    # 3) flattening gradient
    if time.time() < deadline:
        seq_flat, val_flat = _flatten_gradient_opt(
            seq_lbfgs,
            n,
            deadline,
            max_iters=800_000,
            eta0=8.0,
        )
    else:
        seq_flat, val_flat = seq_lbfgs, val_lbfgs

    # 4) simulated‑annealing polish
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            seq_flat,
            n,
            deadline,
            rng,
            max_attempts=300_000,
            start_sigma=0.001,
            T_start=8e-3,
            T_end=5e-9,
        )
    else:
        seq_sa, val_sa = seq_flat, val_flat

    candidates = [
        (start_vec, _objective_surrogate(start_vec)),
        (seq_eg, val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_flat, val_flat),
        (seq_sa, val_sa),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects a 1000 s wall‑clock budget, stores the best found
    sequence in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate length gives a good trade‑off between quality
    # and computational cost.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if present – helps continuity across runs.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1️⃣  Build a rich seed set
    # ------------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # classic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_cosine_window(n))

    # power‑law families (including the classic inverse‑sqrt)
    for p in [0.45, 0.5, 0.55, 0.6, 0.7, 1.0, 1.5]:
        seeds.append(_power(n, p))

    # inverse‑sqrt (continuous flat‑conv solution)
    seeds.append(_inverse_sqrt(n))

    # --------------------------------------------------------------
    # New parametric families (offset power laws & symmetric mixtures)
    # --------------------------------------------------------------
    for p in np.linspace(0.45, 0.65, 9):            # 0.45 … 0.65
        for off in np.linspace(-0.40, 0.40, 9):     # -0.4 … 0.4
            seeds.append(_power_offset(n, float(p), float(off)))

    for p in [0.5, 0.55, 0.6]:
        seeds.append(_symmetric_power(n, p))

    # Exponential decays
    for rate in np.logspace(-5, -0.2, 22):
        seeds.append(_exp_decay(n, float(rate)))

    # Gaussian windows
    for sigma_frac in np.linspace(0.05, 0.35, 7):
        seeds.append(_gaussian(n, float(sigma_frac)))

    # Beta shapes (wide variety)
    alphas = np.linspace(0.01, 5.0, 11)
    betas = np.linspace(0.01, 5.0, 11)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, float(a), float(b)))

    # Logistic shapes
    for scale in [0.3, 0.6, 1.2, 2.5]:
        for mid in [n * 0.2, n * 0.5, n * 0.8]:
            seeds.append(_logistic_shape(n, float(scale), float(mid)))

    # DPSS windows (if scipy.signal windows are available)
    if _sigwin is not None:
        for nw in [2.5, 3.5, 4.5, 5.5]:
            seeds.append(_dpss_window(n, float(nw)))

    # Random Dirichlet draws – several concentration values
    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, float(conc), rng))

    # Mixtures with uniform – break symmetry
    for alpha in [0.1, 0.3, 0.5, 0.7]:
        base = _exp_decay(n, 1e-3)
        seeds.append(_mix_shapes(_uniform(n), base, float(alpha)))

    # Include reversed copies of every seed (helps address edge effects)
    rev_seeds = [s[::-1].copy() for s in list(seeds)]
    seeds.extend(rev_seeds)

    # Perturbations of the previously best solution (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        base = prev_arr / prev_arr.sum()
        seeds.append(base)
        for _ in range(6):
            pert = base + 0.006 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2️⃣  Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Refine each top seed – allocate a modest slice of the remaining budget
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # each seed gets roughly an equal share of the rest (≈15 % per seed)
        share = remaining * 0.15 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}/{top_k}] optimisation budget ≈ {share:.1f}s")
        seq_opt, val_opt = _full_optimisation(seed_vec, n, sub_deadline, rng)

        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"   *** New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final intensive polishing on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final intensive polishing on global best ---")
        final_seq, final_val = _full_optimisation(
            global_best_seq,
            n,
            deadline,
            rng,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
