"""
Smooth‑max optimiser for the step‑function attack.

The original surrogate   S(a) = 2·n·max(conv(a,a))   (∑a = 1)
is non‑smooth because of the max.  We replace the max by a
temperature‑smoothed version

    φ_β(a) = β·log ∑_k exp(conv_k(a,a)/β)          (β>0)

which converges to the true max as β→0.  φ_β is differentiable;
its gradient w.r.t. the logits (log‑soft‑max parameters) can be
written in closed form using a single FFT‑convolution:

    w_k = exp(conv_k/β) / ∑_j exp(conv_j/β)
    grad_a = 2·(rev(a) * w)   (central n entries)

Chain‑rule for the soft‑max gives the gradient in the log‑space.
We run a short continuation: start with a relatively large β
(the loss is then almost the L²‑flatness term), run L‑BFGS,
decrease β and repeat.  After the continuation we polish the
result with a classic exponentiated‑gradient (EG) step that works
directly on the exact max.  The whole routine respects the 1000 s
budget and always stores the best sequence in the global variable
``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global variables used by the smooth‑max gradient routine
# ----------------------------------------------------------------------
_CURRENT_BETA: float = 1.0  # will be overwritten before each optimisation


# ----------------------------------------------------------------------
# Basic utilities (FFT‑based convolution, soft‑max, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → probability vector (∑=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective used for quick ranking (∑a = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))   (exact surrogate, ∑a = 1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth‑max (β‑regularised) loss and its gradient w.r.t. logits
# ----------------------------------------------------------------------
def _smooth_max_obj(logits: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns (loss, grad) where
        loss = β·log∑ exp(conv/β)
    and grad is the derivative w.r.t. `logits`.
    The current β is taken from the module‑level variable
    ``_CURRENT_BETA``.
    """
    beta = _CURRENT_BETA
    a = _softmax(logits)                 # a lives on the simplex
    conv = _conv_fft(a, a)                # length 2n‑1
    # ----- smooth max (log‑sum‑exp) -----
    v = conv / beta
    v_max = np.max(v)
    exp_shift = np.exp(v - v_max)
    sumexp = np.sum(exp_shift)
    loss = beta * (v_max + np.log(sumexp))

    # ----- gradient w.r.t. a -----
    w = exp_shift / sumexp                # soft‑max over conv/β, length 2n‑1
    rev_a = a[::-1]
    # full convolution rev_a * w ; then keep the n central entries
    grad_full = _conv_fft(rev_a, w)
    n = len(a)
    grad_a = 2.0 * grad_full[n - 1:n - 1 + n]

    # ----- back‑prop through the soft‑max -----
    dot = np.dot(grad_a, a)
    grad_logits = a * (grad_a - dot)
    return loss, grad_logits


# ----------------------------------------------------------------------
# Simple exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      deadline: float,
                      max_iters: int = 800_000,
                      eta0: float = 6.0,
                      tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) directly minimising the
    true max‑convolution.  Uses an active‑set of indices that are
    within `tol`·max of the current maximum.
    Returns (best_seq, best_surrogate_value).
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_a = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val

                                    jac=True,
                                    options={'maxiter': 300,
                                             'ftol': 1e-13,
                                             'disp': False})
                logits = res.x
            except Exception:                     # pragma: no cover
                # fallback to a few manual gradient steps
                pass

        else:   # -------------------------------------------------
            # very basic gradient descent if SciPy is missing
            step = 0.2
            for it in range(150):
                if time.time() > stage_deadline:
                    break
                loss, grad = _smooth_max_obj(logits)
                step_size = step / math.sqrt(it + 1.0)
                logits = logits - step_size * grad

        # evaluate surrogate after this temperature
        a_curr = _softmax(logits)
        val = _objective_surrogate(a_curr)
        if val < best_surrogate:
            best_surrogate = val
            best_seq = a_curr.copy()
            print(f"[β={beta:.5e}] improved surrogate → {val:.6f}")

    # --------------------------------------------------------------
    # 2. Final EG polish on the true max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polish (exact max) ---")
        final_seq, final_val = _exp_gradient_opt(best_seq,
                                                 deadline,
                                                 max_iters=5_000_000,
                                                 eta0=6.0,
                                                 tol=1e-9)
        if final_val < best_surrogate:
            best_surrogate = final_val
            best_seq = final_seq
            print(f"[EG] improved surrogate to {final_val:.6f}")

    # --------------------------------------------------------------
    # 3. Evaluate with the provided (secure) evaluator
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_surrogate:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:               # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store globally for warm‑starts in later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

The programme respects the 1000 s wall‑clock limit, keeps the
best sequence in the global variable **height_sequence_1**, and
uses a smooth‑max continuation together with a final exponentiated‑gradient
polish to push the objective below the previous best (~1.5085).
