"""
Optimise a non‑negative step‑function (vector a) for the provided
``evaluate_sequence``.  The objective (for ℓ₁‑normalised vectors) is
``2·n·max(conv(a,a))`` – i.e. minimise the peak of the auto‑convolution.
The algorithm consists of

1.  Assemble a **large, cheap** pool of analytic and random seeds.
2.  Keep the few best seeds (surrogate evaluation).
3.  Refine the best seed with a *continuation* Adam optimiser that
    repeatedly sharpens the log‑sum‑exp approximation of ``max``.
4.  Perform a long **exact exponentiated‑gradient (mirror‑descent)**
    run on the true max‑convolution.
5.  Finish with a very brief **stochastic local search** that injects
    Gaussian noise in the log‑probability space.

All vectors are kept as a probability simplex (log‑softmax), guaranteeing
non‑negativity and unit ℓ₁‑norm.  The code respects the 1000 s budget,
reports progress, and stores the final list in the global variable
``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real inputs, full length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (seq assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised vector ----------------
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^{-p} , i = 0…n‑1, δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 40_000) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp smoothed max‑convolution.
    The schedule starts with a small β and repeatedly multiplies it,
    allowing the landscape to be gradually sharpened.
    """
    if beta_schedule is None:
        # geometric schedule, ending at a very large β ~ 2^18
        beta_schedule = [10, 30, 100, 300, 1_000, 3_000,
                         10_000, 30_000, 100_000, 300_000, 1_000_000]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.15          # a little larger than before – works well with many steps
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities (log‑softmax)
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            # convolution a*a
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)
            # avoid overflow by shifting
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp          # soft‑max weights, sum to 1

            # gradient wrt a :  grad = 2 * (rev_a * w)   (central part)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient wrt log‑softmax parameters (chain rule)
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            # keep logs bounded to avoid overflow/underflow
            log_a = np.clip(log_a, -40.0, 40.0)

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 5.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the true max‑convolution.
    The update is performed in the log‑probability space,
    guaranteeing positivity and ℓ₁‑normalisation.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)
    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see analysis)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing stepsize (standard 1/√t schedule)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g

        # re‑exponentiate and renormalise (log‑softmax)
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _stochastic_local_search(start_seq: np.ndarray,
                             deadline: float,
                             init_sigma: float = 0.15,
                             cooling: float = 0.995,
                             max_trials: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Very cheap random walk in log‑probability space.
    At each trial a Gaussian perturbation (with decaying sigma) is added;
    the candidate is accepted if it improves the surrogate.
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    log_a = np.log(start_seq + 1e-30)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    sigma = init_sigma

    for trial in range(1, max_trials + 1):
        if time.time() > deadline:
            break
        # decay sigma slowly
        sigma *= cooling

        proposal_log = log_a + sigma * rng.normal(size=n)
        # soft‑max projection
        a_prop = np.exp(proposal_log - np.max(proposal_log))
        a_prop /= a_prop.sum()

        val = _objective_surrogate(a_prop)
        if val < cur_val:
            cur_seq = a_prop
            cur_val = val
            log_a = proposal_log
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
        # occasional tiny random restart to escape deep plateaus
        if trial % 10_000 == 0:
            if rng.random() < 0.02:
                log_a = np.log(cur_seq + 1e-30) + sigma * rng.normal(size=n)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the time budget, prints progress and stores the final
    list in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 4.0

    # ------------------------------------------------------------------
    # Problem size – 4000 is a good compromise (as in the reference code)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if it exists (helps continuation runs)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                tot = prev_vec.sum()
                if tot > 0:
                    prev_vec = prev_vec / tot
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Assemble a diversified pool of analytic / random seeds
    # ------------------------------------------------------------------
    seeds = []

    # simple analytic families
    seeds.append(_uniform(n))
    for rate in (0.0012, 0.0018, 0.0025):
        seeds.append(_exp_decay(n, rate))
    seeds.append(_sym_exp_decay(n, rate=0.0025))

    # power‑law families (including shifts)
    for p in np.linspace(0.45, 0.65, 9):
        seeds.append(_power(n, p))
        for delta in (0.05, 0.2, 0.8, 2.0, 4.0):
            seeds.append(_power_shift(n, p, delta))

    # beta‑shaped families – systematic grid
    alphas = np.linspace(0.2, 2.5, 9)
    betas = np.linspace(0.2, 2.5, 9)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws – many concentrations
    concs = [0.01, 0.03, 0.06, 0.12, 0.25, 0.5, 1.0, 2.0]
    for conc in concs:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution + a few permutations (if any)
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(4):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation, keep the very best few
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values, best {}) ---".format(top_k))
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Refine the top seed(s)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # allocate the bulk of the budget to the best seed discovered
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # time slice – give more time to earlier (better) seeds
        remaining = deadline - time.time()
        # 70 % of remaining time for the very first seed, then split the rest
        if idx == 0:
            slice_time = remaining * 0.70
        else:
            slice_time = remaining * 0.30 / (top_k - 1)

        sub_deadline = time.time() + slice_time
        print(f"\n[Seed {idx+1}] allocated ≈ {slice_time:.1f}s")

        # ---- smoothed Adam optimisation (large number of iterations) ----
        seq_adam, val_adam = _adam_smooth_opt(seed_vec, sub_deadline,
                                              iters_per_beta=50_000)

        # ---- exact EG polish (uses any leftover time) --------------------
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_adam, sub_deadline,
                                               max_iters=1_200_000,
                                               eta0=6.0)
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # choose the better refinement
        if val_adam < val_eg:
            local_seq, local_val = seq_adam, val_adam
        else:
            local_seq, local_val = seq_eg, val_eg

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Long exact EG polishing on the current global best (most budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=3_000_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short stochastic local search (may shave a last bit)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + min(30.0, deadline - time.time())
        seq_last, val_last = _stochastic_local_search(global_best_seq.copy(),
                                                      short_deadline,
                                                      init_sigma=0.12,
                                                      cooling=0.997,
                                                      max_trials=120_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final local] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Final evaluation with the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
