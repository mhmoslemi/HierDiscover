"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm (≈1000 s budget)
--------------------------------
1. **Base pool** – a few elementary seeds (uniform, exponential, previous result).
2. **Coarse scan** – exhaustive search of the one‑parameter family  
   a_i ∝ (i+δ)^(-p)  (the “shifted power‑law”) on a coarse p/δ grid.
3. **Fine scan** – around the best coarse parameters a finer grid is examined.
4. **Local random perturbations** – log‑multiplicative noise is added to the
   best vector and accepted if it improves the surrogate.
5. **Exact mirror‑descent (exponentiated‑gradient)** – the best candidate is
   polished with many EG iterations (sub‑gradient of the active max entry).
6. **Adam on a smooth max** – a short Adam run on a log‑sum‑exp surrogate,
   starting from the EG result.
7. **Permutation annealing** – a very short simulated‑annealing phase that
   swaps two entries at a time.
8. The final vector is stored in ``height_sequence_1`` and returned.

All intermediate evaluations use the cheap surrogate  
``2·n·max(conv(a,a))`` (exact when ∑a = 1).  The total time budget is
respected by checking the deadline before every heavyweight step.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based linear convolution (real, full length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (seq is assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (always return an ℓ₁‑normalised vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^{-p}, i = 0…n‑1,  δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random ℓ₁‑normalised vector."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution sub‑gradient)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (log‑softmax space)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # initialise log‑probabilities (log‑softmax)
    log_a = np.log(a + 1e-30)

    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # compute full self‑convolution and locate the active index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see derivation in the description)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        # re‑exponentiate and renormalise (log‑softmax)
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑sum‑exp smoothing of the max)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 20_000) -> tuple[np.ndarray, float]:
    """Adam on a smooth (log‑sum‑exp) surrogate of the max‑convolution."""
    if beta_schedule is None:
        beta_schedule = [10, 50, 200, 800, 3_200, 12_800, 51_200]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities (log‑softmax)
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            # full convolution a*a
            conv = _conv_fft(a, a)

            # soft‑max (log‑sum‑exp) weights
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp

            # gradient wrt a (via convolution with reversed a)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient wrt log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            log_a = np.clip(log_a, -30.0, 30.0)   # keep numbers numerically stable

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Swap‑based permutation optimiser, useful for a final razor‑edge tweak."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        accept = val < cur_val or (rng.random() < math.exp(-(val - cur_val) / T))
        if accept:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  The routine respects the time budget and stores
    the best found sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a nice trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Load possible previous solution (helps continuation)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                total = prev_vec.sum()
                if total > 0:
                    prev_vec = prev_vec / total
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Small generic pool of seeds (analytic + random)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _exp_decay(n, 0.0015),
        _exp_decay(n, 0.0035),
        _sym_exp_decay(n, 0.0025),
    ]

    if prev_vec is not None:
        seeds.append(prev_vec)

    # a few Dirichlet draws (very low concentration -> spiky, high -> flat)
    for conc in (0.02, 0.1, 0.5, 2.0):
        seeds.append(_random_dirichlet(n, conc, rng))

    # ------------------------------------------------------------------
    # 2. Coarse grid scan over the shifted power‑law family
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    best_p = None
    best_delta = None

    coarse_ps = np.linspace(0.45, 0.80, 18)          # step ≈0.02
    coarse_ds = np.linspace(0.10, 5.00, 25)          # step ≈0.20

    print("\n--- Coarse grid scan (shifted power law) ---")
    for p in coarse_ps:
        for delta in coarse_ds:
            if time.time() > deadline:
                break
            vec = _power_shift(n, p, delta)
            val = _objective_surrogate(vec)
            if val < best_val:
                best_val = val
                best_seq = vec
                best_p = p
                best_delta = delta

    print(f"Coarse best surrogate = {best_val:.6f} (p={best_p:.4f}, δ={best_delta:.4f})")

    # ------------------------------------------------------------------
    # 3. Fine grid around the best coarse parameters
    # ------------------------------------------------------------------
    if best_p is not None:
        fine_ps = np.arange(best_p - 0.02, best_p + 0.021, 0.001)   # about 41 points
        fine_ds = np.arange(best_delta - 0.20, best_delta + 0.201, 0.01)  # about 41 points
        print("\n--- Fine grid scan ---")
        for p in fine_ps:
            if p <= 0:
                continue
            for delta in fine_ds:
                if delta <= 0:
                    continue
                if time.time() > deadline:
                    break
                vec = _power_shift(n, float(p), float(delta))
                val = _objective_surrogate(vec)
                if val < best_val:
                    best_val = val
                    best_seq = vec
                    best_p = float(p)
                    best_delta = float(delta)

        print(f"Fine‑grid best surrogate = {best_val:.6f} (p={best_p:.4f}, δ={best_delta:.4f})")

    # ------------------------------------------------------------------
    # 4. Random log‑multiplicative perturbations (local search)
    # ------------------------------------------------------------------
    print("\n--- Local random perturbations ---")
    perturb_iters = 3000
    for it in range(perturb_iters):
        if time.time() > deadline:
            break
        # multiplicative log‑normal noise; keep the vector normalised
        noise = rng.normal(scale=0.03, size=n)
        cand = best_seq * np.exp(noise)
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_seq = cand
            if it % 200 == 0:
                print(f"  iter {it:4d}: new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Exact EG polishing – use whatever time remains
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        slice_time = remaining * 0.60                     # 60 % of remaining budget
        sub_deadline = time.time() + slice_time
        print(f"\n--- Exponentiated‑gradient polishing (≈{slice_time:.1f}s) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, sub_deadline,
                                           max_iters=2_500_000,
                                           eta0=6.0)
        if val_eg < best_val:
            best_val = val_eg
            best_seq = seq_eg
            print(f"  EG improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Short Adam smoothing (helps to escape any remaining tiny plateau)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        slice_time = remaining * 0.30
        sub_deadline = time.time() + slice_time
        print(f"\n--- Adam smoothing (≈{slice_time:.1f}s) ---")
        seq_adam, val_adam = _adam_smooth_opt(best_seq, sub_deadline,
                                              iters_per_beta=10_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam
            print(f"  Adam improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Very short permutation‑annealing tweak (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + min(30.0, deadline - time.time())
        print("\n--- Final permutation annealing (≤30 s) ---")
        seq_perm, val_perm = _perm_anneal(best_seq.copy(),
                                          short_deadline,
                                          max_iters=100_000)
        if val_perm < best_val:
            best_val = val_perm
            best_seq = seq_perm
            print(f"  Permutation tweak improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Final evaluation with the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
