"""
Optimiser for the step‑function objective
    2·n·max(conv(a,a)) / (∑a)² .

The algorithm works on the probability simplex (∑a = 1) and uses a
smooth approximation of the max‑convolution together with an Adam‑style
mirror‑descent on the log‑softmax parameters.  The schedule for the
smoothness parameter λ starts very low (a wide soft‑max) and is
gradually increased so that the optimiser first sees a smooth landscape
and finally focuses on the true maximum.  After the main optimisation
a short simulated‑annealing polish is applied.  The best sequence found
is stored in the global variable ``height_sequence_1`` and returned.

The implementation respects the 1000 s wall‑clock budget, uses only
NumPy (no external files) and works for any length (default n=4000).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, ∑ = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------
def _smooth_max_opt(start_seq: np.ndarray,
                    deadline: float,
                    max_iters: int = 500_000,
                    lam_start: float = 5.0,
                    lam_end: float = 800.0,
                    lr_start: float = 0.12,
                    lr_end: float = 0.005) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror‑descent on the log‑softmax parametrisation.
    The objective is a smoothed version of max(conv) obtained by a
    temperature‑controlled soft‑max (parameter λ).  λ grows linearly with
    elapsed time, turning the surrogate gradually into the true max.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    v = np.log(a + 1e-30)          # log‑softmax parameters

    m = np.zeros_like(v)          # first moment
    v2 = np.zeros_like(v)         # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ---------- convolution ----------
        conv = _conv_fft(a, a)
        max_conv = conv.max()

        # ---------- smooth‑max weights ----------
        elapsed = time.time() - start_time
        ratio = elapsed / total_time
        lam = lam_start + (lam_end - lam_start) * ratio
        w = np.exp(lam * (conv - max_conv))      # max term becomes 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / len(w))
        else:
            w = w / w_sum

        # ---------- gradient of the smoothed max ----------
        rev_a = a[::-1]
        conv_rev = _conv_fft(rev_a, w)          # full convolution
        # slice the part that corresponds to the gradient entries
        g = 2.0 * conv_rev[len(a) - 1:len(a) - 1 + len(a)]

        # ---------- chain rule (log‑softmax) ----------
        grad = a * (g - np.dot(g, a))

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        lr = lr_start + (lr_end - lr_start) * ratio
        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

        # ---------- project back to the simplex ----------
        a = _softmax(v)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (cheap, works on the surrogate)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 80_000,
                         start_sigma: float = 0.0015,
                         T_start: float = 0.015,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    """
    Simple Gaussian perturbation SA with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the wall‑clock budget, stores the best found sequence
    in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin of a couple of seconds for final evaluation
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – 4000 works well and is the size used by the baseline
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Initialise a seed library (previous best + analytic families)
    # ------------------------------------------------------------------
    seeds = []

    # previous solution, if any (helps warm‑start)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,) and arr.sum() > 0.0:
                arr = np.maximum(arr, 0.0)
                arr = arr / arr.sum()
                seeds.append(arr)
        except Exception:
            pass

    # analytic families – two‑parameter beta shapes are especially useful
    seeds.append(_uniform(n))
    for a_ in (0.02, 0.05, 0.1, 0.2):
        for b_ in (0.02, 0.05, 0.1, 0.2):
            seeds.append(_beta_shape(n, a_, b_))

    # a few exponential / power‑law candidates
    for rate in np.logspace(-5, -0.5, 8):
        seeds.append(_exp_decay(n, float(rate)))
    for p in np.linspace(0.0, 4.0, 7):
        seeds.append(_power(n, float(p)))

    # random Dirichlet draws with various concentrations
    for conc in (0.001, 0.01, 0.1, 0.5):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # make sure every seed lives on the simplex
    clean_seeds = []
    for s in seeds:
        s = np.maximum(s, 0.0)
        ss = s.sum()
        if ss > 0.0:
            clean_seeds.append(s / ss)
    seeds = clean_seeds

    # ------------------------------------------------------------------
    # 1️⃣ Quick surrogate evaluation – keep the best few seeds
    # ------------------------------------------------------------------
    evaluated = [(s, _objective_surrogate(s)) for s in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [s for s, _ in evaluated[:top_k]]

    print("\n--- Top seed surrogate values ---")
    for i, (_, v) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {v:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Global best initialisation
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Process each seed (smooth‑max Adam → short SA)
    # ------------------------------------------------------------------
    remaining = len(top_seeds)
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a roughly equal share of the remaining time
        secs_left = deadline - time.time()
        share = secs_left * 0.18 / max(1, remaining)
        sub_deadline = time.time() + share
        remaining -= 1

        # ----- a) smooth‑max Adam optimisation -------------------------
        seq_opt, val_opt = _smooth_max_opt(seed_vec, sub_deadline)

        # ----- b) short simulated‑annealing polish --------------------
        if time.time() < sub_deadline - 0.3:
            seq_sa, val_sa = _simulated_annealing(seq_opt, n,
                                                  sub_deadline - 0.3,
                                                  rng,
                                                  max_attempts=60_000,
                                                  start_sigma=0.0012,
                                                  T_start=0.012,
                                                  T_end=3e-9)
            if val_sa < val_opt:
                seq_opt, val_opt = seq_sa, val_sa

        # keep the best of this seed
        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"[Seed {idx+1}/{top_k}] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final long polishing (another Adam run on the current best)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("\n--- Final long polishing (Adam) ---")
        final_seq, final_val = _smooth_max_opt(global_best_seq,
                                               deadline - 5.0,
                                               max_iters=800_000,
                                               lam_start=5.0,
                                               lam_end=1200.0,
                                               lr_start=0.08,
                                               lr_end=0.004)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Official evaluation (the true objective)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f},"
              f" true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible warm‑starts in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
