def propose_candidate(seed=123, budget_s=1000, **kwargs):

    rng = np.random.default_rng(seed)
    start_time = time.time()
    # minimal margin
    deadline = start_time + budget_s - 5

    n = 4000

    # get previous solution if any
    prev_seq = None
    if globals().get("height_sequence_1") is not None:
        try:
            arr = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if arr.shape[0] == n:
                arr = np.maximum(arr, 0)
                if arr.sum() > 0:
                    prev_seq = arr/arr.sum()
        except Exception:
            prev_seq = None

    seeds = []

    # analytic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.001))
    seeds.append(_exp_decay(n, rate=0.003))
    seeds.append(_sym_exp_decay(n, rate=0.002))
    # power law
    for p in np.linspace(0.45,0.70,7):  # inclusive
        for delta in (0.0, 0.1, 0.5, 1.0, 2.0):
            seeds.append(_power_shift(n, p, delta+1e-12))
        # symmetric power law
        for delta in (0.0, 0.1, 0.5, 1.0):
            seeds.append(_power_shift_sym(n, p, delta+1e-12))

    # arcsine shape approx (beta=0.5)
    for delta in (0.0, 0.5, 1.0, 2.0):
        seeds.append(_arcsine_shape(n, delta+1e-12))

    # beta shape random
    for _ in range(6):
        a = rng.uniform(0.2,2.5)
        b = rng.uniform(0.2,2.5)
        seeds.append(_beta_shape(n,a,b))

    # random Dirichlet
    for conc in (0.02, 0.05, 0.1, 0.3, 0.6, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous and permutations
    if prev_seq is not None:
        seeds.append(prev_seq)
        for _ in range(5):
            seeds.append(rng.permutation(prev_seq))

    # optionally convex combos of some top seeds after evaluation later.

    # evaluate surrogate
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec,_ in evaluated[:top_k]]

    print("Top seeds surrogate values:")
    for i,(vec,val) in enumerate(evaluated[:top_k]):
        print(f"  {i+1}: {val:.6f}")

    # now refinement
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"Initial best surrogate = {global_best_val:.6f}")

    # time allocation
    while time.time() < deadline and top_seeds:
        # compute remaining seeds
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        per_seed_time = remaining * 0.8 / len(top_seeds)
        # pop first seed
        seed_seq = top_seeds.pop(0)
        sub_deadline = time.time() + per_seed_time
        print(f"\nRefining seed, allocated {per_seed_time:.1f}s")
        # Adam smoothing
        seq_adam, val_adam = _adam_smooth_opt(seed_seq, sub_deadline,
                                              beta_schedule=[10,50,200,800,3200,12800,51200],
                                              iters_per_beta=12000)
        # EG polish
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_adam, sub_deadline,
                                               max_iters=500_000,
                                               eta0=6.0)
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # pick better
        if val_adam < val_eg:

"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Build a very diverse pool of analytic seeds:
   * Uniform, exponential, symmetric exponential decay.
   * Shifted power laws  a_i ∝ (i+δ)^{-p}  (p≈0.45…0.70, δ∈[0,5]).
   * *Symmetric* power laws  a_i ∝ (i+δ)^{-p}+(n‑1‑i+δ)^{-p}.
   * Discrete arcsine shape  a_i ∝ 1/√((i+δ)(n‑1‑i+δ)).
   * Beta‑shaped families, random Dirichlet draws and permutations of the
     previously found best solution (if any).

2. Evaluate every seed with the cheap surrogate
   `` 2·n·max(conv(seq,seq)) `` (the surrogate equals the true
   ``evaluate_sequence`` value after ℓ₁‑normalisation).

3. Keep the best few seeds (≤ 5) and refine each of them:
   * A short Adam optimiser on a log‑sum‑exp smoothed version of the
     max‑convolution (multiple β phases, ≈ 80 k iterations total).
   * A longer exact exponentiated‑gradient (mirror‑descent) polish that
     works on the true non‑smooth objective.

   The time budget is split equally among the selected seeds; most of the
   remaining time is used for one final long EG polishing of the globally
   best candidate, followed by a brief permutation‑annealing tweak.

4. The best sequence obtained is stored in the global variable
   ``height_sequence_1`` and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT convolution (real, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (seq is assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^{-p} , i = 0…n‑1, δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift_sym(n: int, p: float, delta: float) -> np.ndarray:
    """Symmetric version: a_i ∝ (i+δ)^{-p} + (n‑1‑i+δ)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p) + (n - 1 - i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_shape(n: int, delta: float) -> np.ndarray:
    """Discrete arcsine density: a_i ∝ 1/√((i+δ)(n‑1‑i+δ))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + delta) * (n - 1 - i + delta))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on a log‑sum‑exp smooth surrogate of max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 12_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth (log‑sum‑exp) surrogate of the max‑convolution.
    """
    if beta_schedule is None:
        beta_schedule = [10, 50, 200, 800, 3_200, 12_800, 51_200]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities via log‑softmax
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            conv = _conv_fft(a, a)

            # smooth max using log‑sum‑exp
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp  # soft‑max weights over the full conv

            # gradient wrt a (see derivation in the previous code)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # slice back to length n

            # gradient wrt log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            log_a = np.clip(log_a, -30.0, 30.0)   # keep logs bounded

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution (sub‑gradient on the active entry).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # initialise log‑probabilities (log‑softmax coordinates)
    log_a = np.log(a + 1e-30)

    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step size
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g

        # log‑softmax projection
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """Pure permutation optimiser – swaps two entries."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        accept = val < cur_val or (rng.random() < math.exp(-(val - cur_val) / T))
        if accept:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val
