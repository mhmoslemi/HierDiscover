def _exp_gradient_opt_precise(start_seq, n, deadline, max_iters=1_000_000, eta0=10.0):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta = 0.9
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # tolerance decreasing with iteration: rel_tol = 1e-12 * (1 + it/1e6)? maybe small constant.
        # We'll take only indices within absolute tolerance 1e-12*max_val.
        tol = 1e-12
        active = np.where(conv >= max_val - tol * max_val)[0]
        # compute subgradient
        g = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            rev_idx = k - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size
        # update momentum
        momentum = beta * momentum + (1 - beta) * g
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * momentum
        a = _softmax(log_a)
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    return best_seq, best_val

"""
Improved step‑function optimiser.

Main ideas
-----------
* richer family of analytic seeds – a power law with a tunable offset
  (a_i ∝ 1/(i+offset)^p) – which can fine‑tune the early‑index
  mass that dominates the central convolution peak.
* use the same multi‑stage pipeline as the reference (EG → smooth L‑BFGS
  → flattening → SA) but with more aggressive budgets for the most
  promising seeds and a tighter tolerance in the exponentiated‑gradient
  (sub‑gradient) phase.
* a final “polishing” phase that repeatedly perturbs the best‑found
  sequence in log‑softmax space and runs a short EG optimisation,
  accepting any improvement.
* all operations stay on the probability simplex (soft‑max
  re‑parameterisation) so the objective is scale‑invariant.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Optional scientific imports
# --------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

try:
    from scipy.special import i0 as _i0
except Exception:  # pragma: no cover
    _i0 = None

try:
    from scipy.signal import windows as _sigwin
except Exception:  # pragma: no cover
    _sigwin = None

# --------------------------------------------------------------
# Numerical utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq))  (seq is ℓ₁‑normalised)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# --------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    a = a / a.sum()
    return a


def _power_offset(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ 1/(i+offset)^p   with i = 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64) + offset
    a = 1.0 / (i ** p)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture (1‑α)·a + α·b, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    if s == 0.0:
        return a.copy()
    return mixed / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# --------------------------------------------------------------
# Exponentiated‑gradient optimiser (max‑convolution surrogate)
# --------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 2_000_000,
                      eta0: float = 12.0,
                      tol: float = 1e-12,
                      beta_mom: float = 0.9) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    idx = np.arange(n, dtype=np.int64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices where conv is within (tol * max_val) of the maximum
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient: average over the active indices
        g = np.zeros_like(a)
        for k in active:
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += a[rev[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑style momentum on the sub‑gradient
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# --------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# --------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """L‑BFGS optimisation of the smooth‑max (log‑sum‑exp) surrogate."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 200.0,
            500.0, 1_000.0, 2_000.0, 5_000.0, 10_000.0,
            20_000.0, 50_000.0, 100_000.0, 200_000.0,
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of the smooth‑max term
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'ftol': 1e-12,
                         'disp': False}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            # any solver failure – abort the schedule
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# --------------------------------------------------------------
# Flatten‑gradient optimiser (quadratic loss on the whole convolution)
# --------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 800_000,
                         eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on Σ_k (conv_k – avg)²  – tries to flatten the whole convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # length of full convolution
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom     # equals 1/denom for sum=1
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)    # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent (projected onto the simplex via log‑softmax)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# --------------------------------------------------------------
# Simulated annealing (final polish)
# --------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 300_000,
                         start_sigma: float = 0.001,
                         T_start: float = 8e-3,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    """Simple SA with a linear cooling schedule."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

# Full optimisation pipeline for a single seed
# --------------------------------------------------------------
def _full_optimisation(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run EG → L‑BFGS → flattening → SA on a copy of *start_vec*.
    All stages respect the shared *deadline*.
    Returns the best sequence encountered and its surrogate value.
    """
    # 1) exponentiated gradient on the exact max‑convolution
    seq_eg, val_eg = _exp_gradient_opt(
        start_vec.copy(),
        n,
        deadline,
        max_iters=2_500_000,
        eta0=14.0,
        tol=1e-12,
    )

    # 2) smooth L‑BFGS
    if time.time() < deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(
            seq_eg,
            n,
            deadline,
            beta_schedule=None,
            max_iter_per_beta=4000,
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, val_eg

    # 3) flattening gradient (quadratic loss)
    if time.time() < deadline:
        seq_flat, val_flat = _flatten_gradient_opt(
            seq_lbfgs,
            n,
            deadline,
            max_iters=800_000,
            eta0=9.0,
        )
    else:
        seq_flat, val_flat = seq_lbfgs, val_lbfgs

    # 4) simulated annealing polish
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            seq_flat,
            n,
            deadline,
            rng,
            max_attempts=300_000,
            start_sigma=0.001,
            T_start=8e-3,
            T_end=5e-9,
        )
    else:
        seq_sa, val_sa = seq_flat, val_flat

    candidates = [
        (start_vec, _objective_surrogate(start_vec)),
        (seq_eg, val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_flat, val_flat),
        (seq_sa, val_sa),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])

# --------------------------------------------------------------
# Helper: generate a rich collection of seeds
# --------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds = []

    # basic analytic families
    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))

    # power‑law with tunable offset
    offsets = [0.0, 0.001, 0.005, 0.01, 0.03, 0.07]          # fraction of n
    powers = [0.5, 0.55, 0.6, 0.65, 0.7]
    for p in powers:
        for of in offsets:
            off = of * n
            seeds.append(_power_offset(n, p, off))

    # exponential decays (single)
    for rate in np.logspace(-6, -0.2, 20):
        seeds.append(_exp_decay(n, rate))

    # mixtures with uniform
    for alpha in [0.1, 0.3, 0.5, 0.7, 0.9]:
        seeds.append(_mix_shapes(_uniform(n), _inverse_sqrt(n), alpha))

    # random Dirichlet draws (wide range of concentration)
    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # reversed copies (helps exploit symmetry)
    rev = [s[::-1].copy() for s in list(seeds)]
    seeds.extend(rev)

    return seeds
