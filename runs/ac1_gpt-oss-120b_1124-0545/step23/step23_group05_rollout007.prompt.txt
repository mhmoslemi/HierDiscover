def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# Mirror‑descent on the exact max‑convolution (multi‑active)

def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,

                            eta0: float = 10.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:

    Exponentiated‑gradient on the exact max‑convolution.
    Uses all indices within a relative tolerance of the current maximum.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    # momentum (Adam‑like) – helps traverse flat regions
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient (average over active indices)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑like momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Gradient–descent that flattens the convolution (quadratic loss)
# --------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 800_000,
                         eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the loss Σ_k (conv_k - avg)^2 .
    The aim is to make the whole convolution as flat as possible.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # length of full convolution
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when sum=1
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent step
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Smooth‑max L‑BFGS optimiser (log‑sum‑exp)
# --------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """
    Optimise a smooth max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space to keep the iterate on the simplex.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)

# Simulated‑annealing refinement (accepts occasional worsenings)
# --------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.0015,
                         T_start: float = 0.012,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val
