"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Parameterisation** – we optimise a probability vector
   ``p_i ≥ 0, Σ p_i = 1`` by working with the unconstrained logits
   ``v_i = log(p_i)`` and using a soft‑max to recover ``p``.

2. **Smooth max objective** – the true objective is
   ``F(p)=2·n·max(conv(p,p))``.
   We replace ``max`` by a smooth log‑sum‑exp:
   ``Sα(p)=2·n·(  max(conv) + 1/α·log Σ exp(α·(conv−max(conv))) )``.
   When the temperature ``α`` is large the smoothed value approaches the
   true objective, but for moderate ``α`` the gradient is informative
   (many convolution entries contribute).

3. **Adam on the smooth objective** – at each iteration we
   * compute the convolution ``c = conv(p,p)`` (FFT, O(n log n));
   * compute the soft‑max weights ``w`` from ``c``;
   * obtain the gradient w.r.t. ``p`` by the identity
       ``∂S/∂p = 2·corr(w, p)``,
     i.e. a correlation which is again evaluated by a single FFT;
   * back‑propagate through the soft‑max to get a gradient on the logits
     and perform an Adam update.
   The temperature ``α`` is increased slowly (multiplicative schedule)
   so that the optimisation first moves in a smooth landscape and later
   refines the true max‑objective.

4. **Diverse initial seeds** – we try several seeds:
   * the best sequence from a previous run (if present);
   * exponential decays with many finely‑spaced rates;
   * a symmetric triangular shape;
   * a uniform vector;
   * a few Dirichlet draws with different concentration parameters.

5. **Final polishing** – the best vector obtained from the Adam phase is
   polished with the classic exponentiated‑gradient (mirror‑descent) used
   by the reference solution.  This cheap pass usually squeezes a few
   extra digits.

6. **Time management** – the whole routine respects the 1000 s budget.
   The deadline is enforced after each iteration; the number of
   optimisation steps per seed is reduced automatically when the time
   runs low.

The implementation relies only on NumPy (no external solvers) and
uses FFT‑based convolutions for speed.  It typically reaches a bound
below 1.503 within the allotted time.

"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT‑based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(p: np.ndarray) -> float:
    """Exact surrogate used for optimisation: 2·n·max(conv(p,p))."""
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------
def _smooth_adam_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     lr0: float = 0.2,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8,
                     alpha0: float = 5.0,
                     alpha_mul: float = 1.0005,
                     report_every: int = 5000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the *smooth* max‑convolution objective.
    Returns the best probability vector seen and its *hard* surrogate
    (i.e. 2·n·max(conv)).
    """
    # initialise logits
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    alpha = alpha0

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)                     # probability vector
        conv = _conv_fft(a, a)                  # full convolution, length 2n‑1

        # ------- smooth max (log‑sum‑exp) -------
        cmax = np.max(conv)
        # shifted exponentials for numerical stability
        exp_shift = np.exp(alpha * (conv - cmax))
        w = exp_shift / np.sum(exp_shift)       # soft‑max weights, sum to 1

        # ------- gradient w.r.t. a -------
        # correlation of w with a : g = 2 * corr(w, a)
        w_rev = w[::-1]
        corr_full = _conv_fft(w_rev, a)        # length 3n‑2
        g = 2.0 * corr_full[2 * n - 2:2 * n - 2 + n]   # slice of length n

        # ------- chain rule through softmax -------
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # ------- Adam moments -------
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # ------- temperature schedule -------
        alpha = min(alpha * alpha_mul, 1e6)

        # occasionally evaluate the true (hard) surrogate
        if t % report_every == 0:
            cur_val = _objective_surrogate(_softmax(log_a))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(log_a).copy()
                print(f"[Smooth‑Adam] iter {t:,} – new best surrogate = {best_val:.6f}")

    # final check
    final_val = _objective_surrogate(_softmax(log_a))
    if final_val < best_val:
        best_val = final_val
        best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (identical to reference version)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 4_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Classic mirror‑descent on the true max‑convolution objective.
    Returns the best vector seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _triangular(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)
