def _exp_gradient_opt(start_seq, n, deadline, eta0=5.0):
    a = start_seq.copy()
    # ensure positivity
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    log_a = np.log(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)  # length 2n-1
        k = np.argmax(conv)
        # compute gradient at positions i satisfying i + j = k.
        # g_i = 2 a_{k - i} if 0 <= k-i < n else 0.
        # Using vectorized approach
        # positions i range max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]
        # update log a:
        eta = eta0 / np.sqrt(it)
        log_a = log_a - eta * g
        # project back to simplex via softmax
        a = _softmax(log_a)
        # optionally re-normalize due to numerical error:
        a = a / a.sum()
        # evaluate surrogate less frequently e.g., every 20 iter
        if it % 20 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    return best_seq, best_val

"""
Non‑negative step‑function optimiser (high‑precision variant).

The idea:
* The objective 2·n·max(conv(a,a)) (with Σa = 1) is minimised.
* For a large number of points the optimal shape is close to the
  continuous arcsine law (Beta(½,½)).  Increasing the discretisation size
  reduces the discretisation error and brings the value close to the
  conjectured limit 1.5.
* We therefore start from a family of symmetric Beta‑distributions,
  evaluate a cheap FFT‑based surrogate (exact for the chosen normalisation)
  and keep the best shape.
* A short mirror‑descent (exponentiated‑gradient) on the true max‑convolution
  refines the candidate further.
* The whole search respects the external 1000 s deadline and stores the
  best sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, softmax, surrogate)
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for the surrogate)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search (Σa = 1):
        2 * n * max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# --------------------------------------------------------------
# Distribution generators (all already sum to 1)
# --------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on the centre of each bin.
    For α = 0.5 we obtain the arcsine law.
    """
    i = np.arange(n, dtype=np.float64) + 0.5   # centre of the i‑th bin
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# --------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of true max)
# --------------------------------------------------------------

def _exp_gradient_opt(start_seq: np.ndarray,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    The loop runs until ``deadline`` expires.
    Returns the best sequence seen and its surrogate value.
    """
    a = np.copy(start_seq)
    a = np.maximum(a, 1e-30)          # avoid log(0)
    a = a / a.sum()
    log_a = np.log(a)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    n = len(a)

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)               # length 2n‑1
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx_i
        g[idx_i] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)                 # stay on the simplex
        a = a / a.sum()                     # re‑normalise against numerical drift

        if it % 20 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# --------------------------------------------------------------
# Light‑weight random mixing (tiny set of candidates)
# --------------------------------------------------------------

def _mix(seq1: np.ndarray, seq2: np.ndarray, w: float) -> np.ndarray:
    """Convex combination (1‑w)*seq1 + w*seq2 – both already normalised."""
    return (1.0 - w) * seq1 + w * seq2


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine works with a larger discretisation (≈12 k points) because
    the discretisation error dominates the current gap.  Starting from a
    family of symmetric Beta‑distributions we keep the best surrogate,
    polish it with exact exponentiated‑gradient, and finally evaluate the
    true objective.  The best sequence is stored globally as
    ``height_sequence_1`` and also returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a safety margin for the external watchdog
    deadline = start_time + budget_s - 2.0
    print(f"[Info] Global deadline ≈ {deadline - start_time:.1f}s from start")

    # ----------------------------------------------------------
    # 1️⃣ Choose a problem size large enough to beat the target.
    # ----------------------------------------------------------
    # The known value at n=4000 is ≈1.5085.
    # Empirically the error scales roughly like C / n (C≈34).  To reach
    # 1.503 we need n ≳ 1.2·10⁴.  We cap the size at 15000 to keep the
    # final direct convolution affordable (≈2·10⁸ operations → < 1 s).
    n_requested = int(kwargs.get("n", 4000))
    n = max(8000, n_requested)          # never go below 8 k for safety
    approx_needed = int(34000 / (1.5085 - 1.503) + 1)   # ≈12000
    n = max(n, min(15000, approx_needed))
    print(f"[Setup] Using sequence length n = {n}")

    # ----------------------------------------------------------
    # 2️⃣ Build a small diverse pool of seeds (all sum to 1).
    # ----------------------------------------------------------
    seeds = []

    # core Beta shapes around the arcsine law
    for a in (0.45, 0.48, 0.50, 0.52, 0.55):
        seeds.append(_beta_distribution(n, a))

    # simple distributions for robustness
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # mixtures of arcsine with uniform – a few mixing weights
    arcsine = _beta_distribution(n, 0.5)
    uniform = _uniform(n)
    for w in (0.05, 0.10, 0.15, 0.20, 0.30):
        seeds.append(_mix(arcsine, uniform, w))

    # ----------------------------------------------------------
    # 3️⃣ If a previous solution exists (same length), add it as seed.
    # ----------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n:
                if arr.sum() > 0:
                    prev_norm = arr / arr.sum()
                    seeds.append(prev_norm)
        except Exception:
            pass   # ignore malformed previous solution

    # ----------------------------------------------------------
    # 4️⃣ Select the best surrogate among the seeds.
    # ----------------------------------------------------------
    best_seq = None
    best_surrogate = np.inf
    print("\n--- Seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        # numeric safety – ensure a proper probability vector
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        print(f" seed #{idx+1:02d}: surrogate = {val:.6f}")
        if val < best_surrogate:
            best_surrogate = val
            best_seq = vec.copy()
    print(f"[Init] Best surrogate after seeding = {best_surrogate:.6f}")

    # ----------------------------------------------------------
    # 5️⃣ Exact exponentiated‑gradient polishing.
    # ----------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        # allocate roughly half of the remaining time to polishing
        rem = deadline - time.time()
        pol_deadline = time.time() + 0.45 * rem
        seq_polished, val_polished = _exp_gradient_opt(best_seq,
                                                       deadline=pol_deadline,
                                                       eta0=5.0)
        print(f"[EG] polished surrogate = {val_polished:.6f}")
        if val_polished < best_surrogate:
            best_seq = seq_polished
            best_surrogate = val_polished
            print("   >>> Updated global best after EG")

    # ----------------------------------------------------------
    # 6️⃣ Final official evaluation with the provided ``evaluate_sequence``.
    # ----------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_surrogate:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:               # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

This implementation follows the constraints:

* All helper functions are top‑level, no closures or lambdas.
* Only `numpy` and the standard library are used (no filesystem/network I/O).
* The algorithm respects the 1000 s deadline, keeps the best sequence
  in the global variable `height_sequence_1`, and finally returns it as
  a plain list ready for `evaluate_sequence`.
