"""
Optimisation of a non‑negative step function (probability vector) to
minimise ``evaluate_sequence``.

Approach
--------
*   Work with a fixed length ``n`` (default 8000) and enforce a simplex
    constraint via a soft‑max parametrisation: a = softmax(log_a).
*   Optimise the smooth surrogate
        M_β(a) = (1/β) log Σ_k exp(β·conv_k(a,a))
    where conv = a∗a.  As β → ∞ the surrogate converges to
    max_k(conv_k) and the original objective 2·n·max(conv) is recovered.
*   Use a continuation on β (geometric increase) and a long Adam run on the
    log‑softmax parameters.  The gradient of the surrogate is obtained from
    the convolution of the reversed vector with the soft‑max weights.
*   Several analytic and random seeds are generated, the most promising are
    refined, and the overall best is finally polished with a longer Adam phase.
*   The whole routine respects the global time‑budget (default 1000 s) and
    stores the best sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper functions – pure and top‑level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast for many calls)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(a,a)) (a assumed normalised, sum=1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic / random seed families (all return normalised probability vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    return a / a.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation
# ----------------------------------------------------------------------


def _adam_opt(start_seq: np.ndarray,
              deadline: float,
              max_iters: int = 2_000_000,
              init_lr: float = 0.5,
              beta_start: float = 5.0,
              beta_mult: float = 3.0,
              epochs: int = 12) -> Tuple[np.ndarray, float]:
    """
    Perform a continuation Adam optimisation on the smooth max surrogate.
    Returns (best_sequence, best_surrogate_value).
    """
    n = len(start_seq)
    # initialise log‑softmax parameters
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = start_seq.copy()
    best_val = _objective_surrogate(best_a)

    it = 0
    # each epoch uses a larger β (sharper approximation of max)
    for epoch in range(epochs):
        beta = beta_start * (beta_mult ** epoch)
        iters_this_epoch = max_iters // epochs
        for _ in range(iters_this_epoch):
            it += 1
            if time.time() > deadline:
                return best_a, best_val

            # current primal vector via soft‑max
            max_log = np.max(log_a)
            exp_shift = np.exp(log_a - max_log)
            a = exp_shift / exp_shift.sum()

            # convolution a∗a and its max
            conv = _conv_fft(a, a)
            shift = np.max(conv)                 # = max(conv)
            # smooth weights (log‑sum‑exp)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp               # length 2n‑1, sums to 1

            # gradient w.r.t. a: 2 * (rev_a ∗ w) restricted to n entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t. log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update on log_a
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            log_a -= init_lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep logs bounded to avoid overflow / underflow
            log_a = np.clip(log_a, -30.0, 30.0)

            # evaluate surrogate (no extra convolution needed – we already have conv)
            cur_val = 2.0 * n * shift
            if cur_val < best_val:
                best_val = cur_val
                # recompute a from the current log_a for storage
                max_log = np.max(log_a)
                exp_shift = np.exp(log_a - max_log)
                best_a = exp_shift / exp_shift.sum()

    return best_a, best_val

        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
