"""
Propose a step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)²)

The approach is a *continuation* on a smooth approximation of the max.
For a probability vector a (∑a=1) we minimise

    F_τ(a) = (1/τ)·log Σ_k exp( τ·(a*a)_k )                (τ>0)

When τ→∞ the loss tends to the exact max‑convolution.
F_τ is smooth for any finite τ, and its gradient can be written in closed
form using only FFT‑based convolutions, which makes it cheap to evaluate.
We start from a rich library of analytic seeds, run a few L‑BFGS
optimisations with increasing τ (τ=5,…,1280) and finally polish the
best candidate with the exponentiated‑gradient (EG) method that works on
the true max‑convolution.  All updates stay on the simplex via a
log‑softmax re‑parameterisation.  The whole search respects the 1000 s
wall‑clock budget and stores the best sequence in the global variable
``height_sequence_1``.
"""

import time
import math

# ----------------------------------------------------------------------
# Optional SciPy import (required for L‑BFGS)

# ----------------------------------------------------------------------
# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(L log L))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate (exact) objective for a probability vector (∑a = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) for ∑seq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ i⁻ᵖ (i starts from 1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    # initialise log‑softmax parameters
    v = np.log(start_vec + 1e-30)

    # convenience for deadline handling
    def _time_check():
        if time.time() > deadline:
            raise KeyboardInterrupt()

    for τ in taus:
        _time_check()

        # ----- objective and gradient for a given τ -----
        def fun(v_local: np.ndarray):
            a = _softmax(v_local)                     # prob. vector, ∑a = 1
            conv = _conv_fft(a, a)                    # linear conv, length 2n‑1

            # ----- smooth approximation of max(conv) -----
            z = τ * conv
            z_max = np.max(z)
            exp_z = np.exp(z - z_max)                 # shift for numerical stability
            sum_exp = np.sum(exp_z)
            loss = (np.log(sum_exp) + z_max) / τ      # = (1/τ)·log Σ exp(τ·conv)

            # ----- gradient w.r.t. a -----
            w = exp_z / sum_exp                        # soft‑max weights, Σ w = 1
            # g = Σ_k w_k·∂conv_k/∂a  =  conv(w, a)[:n] + a
            conv_w_a = _conv_fft(w, a)[:n]
            g = conv_w_a + a

            # ----- chain rule to log‑softmax parameters -----
            dot = np.dot(g, a)
            grad_v = a * (g - dot)                     # Jacobian of softmax

            return float(loss), grad_v.astype(np.float64)

        # ----- callback to respect the deadline -----
        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        # ------- run L‑BFGS (or fallback) -------
        if _opt is not None:
            try:
                res = _opt.minimize(fun=fun,
                                    x0=v,
                                    method='L-BFGS-B',
                                    jac=True,
                                    callback=callback,
                                    options={'maxiter': maxiter_per_tau,
                                             'disp': False,
                                             'ftol': 1e-13})
                v = res.x
            except KeyboardInterrupt:
                # time limit – keep current iterate
                break
        else:
            # very simple gradient descent fallback (rare)
            eta = 0.5
            for it in range(maxiter_per_tau):
                _time_check()
                loss, grad = fun(v)
                v -= eta * grad
                eta = eta * 0.99  # decay

    return _softmax(v)


# ----------------------------------------------------------------------
# Exponentiated‑gradient (EG) refinement on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 600_000,
                      eta0: float = 6.0,
                      tol: float = 1e-9) -> np.ndarray:
    """
    Mirror‑descent on the true max‑convolution (multi‑active set).
    Returns the best sequence encountered before the deadline.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of the max (average over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑like momentum
        momentum = beta * momentum + (1.0 - beta) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s wall‑clock
    budget, stores the best found sequence in the global variable
    ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a moderate size that balances expressiveness and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Restore previous best (if any) to warm‑start
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                s = arr.sum()
                if s > 0:
                    prev_vec = arr / s
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build a diverse seed library
    # ------------------------------------------------------------------
    seeds = []

    # analytic families
    seeds.append(_uniform(n))

    # U‑shaped Beta families – explore very small parameters
    for a in (0.02, 0.05, 0.1, 0.2):
        for b in (0.02, 0.05, 0.1, 0.2):
            seeds.append(_beta_shape(n, a, b))

    # Exponential decays (log‑spaced)
    for rate in np.logspace(-6, -0.3, 14):
        seeds.append(_exp_decay(n, rate))

    # Power‑law tails
    for p in np.linspace(0.0, 5.0, 11):
        seeds.append(_power_law(n, p))

    # Logistic shapes – mass pushed toward the ends
    for scale in (0.5, 1.0, 2.0):
        for mid in (n * 0.15, n * 0.85):
            seeds.append(_logistic_shape(n, scale, mid))

    # Dirichlet draws with a range of concentration parameters
    for conc in (0.001, 0.005, 0.01, 0.05, 0.1):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # previous best (if any) and perturbations thereof
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(4):
            pert = prev_vec + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds, keep the K best
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seeding (n={}) ---".format(n))
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  seed #{i+1}: surrogate = {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Process each seed: smooth‑L‑BFGS → EG polishing
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        print(f"\n[Seed {si+1}/{top_k}] starting optimisation")
        remaining = deadline - time.time()
        # allocate a modest share of the remaining budget for this seed
        seed_deadline = time.time() + remaining * 0.14

        # ---- a) smooth continuation using L‑BFGS ----
        a_smooth = _smooth_opt(seed_vec, n, seed_deadline)
        val_smooth = _objective_surrogate(a_smooth)
        print(f"  after smooth L‑BFGS: surrogate = {val_smooth:.6f}")

        # ---- b) EG polishing on the exact max (if time permits) ----
        if time.time() < seed_deadline:
            a_final = _exp_gradient_opt(a_smooth, n, seed_deadline,
                                        max_iters=800_000,
                                        eta0=8.0, tol=1e-9)
            val_final = _objective_surrogate(a_final)
            print(f"  after EG polish:   surrogate = {val_final:.6f}")
        else:
            a_final, val_final = a_smooth, val_smooth

        # keep the best among the three (seed, smooth, final)
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (a_smooth, val_smooth),
            (a_final, val_final),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"  local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"  *** new global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long EG polishing on the overall best (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq = _exp_gradient_opt(global_best_seq, n, deadline,
                                      max_iters=5_000_000,
                                      eta0=6.0, tol=1e-9)
        final_val = _objective_surrogate(final_seq)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate against the provided ``evaluate_sequence`` and store
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
