The program searches for a non‑negative step function (a probability vector that sums to 1) that minimises  

\[
\Phi(a)=\frac{2n\;\max\limits_{k}(a*a)_k}{(\sum_i a_i)^2},
\]

which is exactly the value returned by `evaluate_sequence`.  
The optimisation is performed in three phases:

1. **Diverse seeds** – a large pool of candidate shapes (uniform, arcsine, triangular, many β‑shapes and Dirichlet draws) is generated and evaluated with the cheap surrogate
   \(\hat\Phi(a)=2n\max(a*a)\) (the denominator is 1 because we normalise the vectors).

2. **Momentum exponentiated‑gradient descent** – a mirror‑descent algorithm on the *exact* maximum of the convolution.  
   The active index `k = argmax(a*a)` supplies a sub‑gradient  
   \(g_i = 2\,a_{k-i}\) (when the pair exists).  
   The update is performed in log‑space with Adam‑style momentum, which keeps the iterate on the simplex and converges fast on this non‑smooth problem.

3. **Fine‑grained local polishing** – a greedy transfer of a tiny amount of mass from the coordinate that contributes most to the current maximum towards the coordinate that contributes least, followed by a short stochastic walk.  
   Both steps keep the sequence on the simplex and accept only improving moves.

Time is split automatically: the first part (seed generation) is cheap, the second part receives the bulk of the budget, and the third part uses the remaining seconds.  
The best sequence found is stored in the global variable `height_sequence_1` and returned as a plain Python list.

Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Generate a large and diverse set of seed distributions (uniform,
   arcsine, triangular, many symmetric Beta(α,α) shapes and Dirichlet
   draws).  All seeds are normalised to sum = 1 and evaluated with the
   cheap surrogate 2·n·max(conv(a,a)).  The five best seeds are kept.

2. For each of the selected seeds run a momentum exponentiated‑gradient
   descent on the *exact* max‑convolution:
        • compute the active index k = argmax(conv(a,a))
        • sub‑gradient g_i = 2·a_{k‑i} on the support of k
        • work in log‑space, apply Adam‑style momentum and a
          diminishing step‑size η_t = η₀/√t
        • after every few thousand iterations evaluate the exact surrogate
          and keep the best iterate.
   The running time is split equally among the top seeds.

3. Greedy refinement – repeatedly move a small fraction of mass from the
   coordinate with the largest sub‑gradient to the one with the smallest
   sub‑gradient (only if the move improves the exact surrogate).  This
   quickly removes the last “bottleneck’’ of the max‑convolution.

4. A final tiny stochastic walk (local random search) is performed for the
   last few seconds.

5. The best sequence is finally evaluated with the supplied
   ``evaluate_sequence`` and stored globally as ``height_sequence_1``.
"""

    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate: 2·n·max(conv(seq, seq)).
    ``seq`` is assumed to sum to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the discrete set {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

# Optimisation primitives
# ----------------------------------------------------------------------


def _exp_gradient_momentum(start_seq: np.ndarray,
                           deadline: float,
                           eta0: float = 5.0,
                           beta_mom: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam‑style momentum on the exact max‑convolution.
    ``start_seq`` must sum to 1.
    Returns the best sequence seen and its exact surrogate value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    # initialise log‑probabilities (mirror variables)
    logit = np.log(a + 1e-30)
    # momentum accumulator on the *log‑gradient*
    mom = np.zeros_like(logit)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active max‑convolution
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        # mirror‑gradient (log‑space) as required by exponentiated‑gradient
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # momentum update (Adam‑style without bias‑correction for speed)
        mom = beta_mom * mom + (1.0 - beta_mom) * grad_log

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        # update in log‑space and map back to the simplex
        logit = logit - eta * mom
        a = _softmax(logit)

        # periodic exact evaluation
        if it % 5000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        # early exit if we already beat the target (≈1.503)
        if best_val < 1.503:
            break

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val
    return best_seq, best_val


def _greedy_refine(start_seq: np.ndarray,
                   deadline: float,
                   attempts: int = 120_000,
                   delta_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy transfer of a small amount of mass from the coordinate that
    contributes most to the current max‑convolution towards the one that
    contributes least.  Only improving moves are kept.
    """
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    rng = np.random.default_rng()
    while attempts > 0 and time.time() < deadline:
        attempts -= 1

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        # indices with the largest / smallest gradient
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))

        if a[i_max] <= 1e-15:
            continue

        # transfer a tiny fraction of the mass of i_max to i_min
        delta = delta_frac * a[i_max] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta

        # numerical safety
        if new_a[i_max] < 0.0:
            continue

        val = _objective_surrogate(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a  # continue from the improved point
    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex.  Moves that improve the exact
    surrogate are kept.
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function of length ``n`` that minimises
    ``evaluate_sequence``.  The best sequence found within the allotted
    time is stored globally as ``height_sequence_1`` and returned as a plain
    list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Load any previously stored solution (if length matches)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = np.maximum(0.0, arr)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            pass

    # --------------------------------------------------------------
    # 0️⃣  Assemble a diverse seed pool
    # --------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # Symmetric Beta(α,α) with many α values (U‑shapes and bell‑shapes)
    alphas = np.concatenate([
        np.linspace(0.05, 0.55, 11),   # very U‑shaped
        np.linspace(0.6, 1.4, 9),     # around the arcsine
        np.linspace(1.5, 2.5, 7)      # bell‑shaped
    ])
    for a in alphas:
        seeds.append(_beta_distribution(n, float(a)))

    # Dirichlet draws with several concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 1️⃣  Cheap surrogate evaluation of all seeds
    # --------------------------------------------------------------
    seed_infos = []
    print("\n--- Evaluating initial seed pool (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(0.0, vec)
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f"seed #{idx + 1:02d}: surrogate = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_k = min(5, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Momentum exponentiated‑gradient on the exact max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline:
        # give each top seed an equal slice of the remaining time
        remaining = deadline - time.time()
        per_seed_deadline = time.time() + remaining / top_k
        print("\n--- Momentum EG optimisation (exact max) ---")
        for sidx, seed_vec in enumerate(top_seeds, 1):
            if time.time() > deadline:
                break
            print(f"  EG run #{sidx} (time slice ≈ {per_seed_deadline - time.time():.1f}s)")
            seq, val = _exp_gradient_momentum(seed_vec, per_seed_deadline)
            print(f"   → surrogate after EG = {val:.6f}")
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print("   >>> Updated global best (EG)")

            # ----------------------------------------------------------
            # 2️⃣‑b  Greedy refinement of the EG result
            # ----------------------------------------------------------
            if time.time() < deadline:
                seq2, val2 = _greedy_refine(seq, deadline)
                if val2 < best_val:
                    best_val, best_seq = val2, seq2.copy()
                    print("   >>> Updated global best (greedy)")

    # --------------------------------------------------------------
    # 3️⃣  Final random local refinement (last few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, deadline,
                                                attempts=250_000,
                                                sigma=0.015)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # --------------------------------------------------------------
    # 4️⃣  Official evaluation with the supplied ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
