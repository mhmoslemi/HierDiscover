def _adam_softmax_opt(start_seq, n, deadline, max_iters=300_000, lr0=0.2, target_alpha=200.0):
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v_ = np.zeros_like(log_a) # rename not to clash
    best_seq = _softmax(log_a)
    best_hard = _objective_surrogate(best_seq) # hard max (2n*max conv)
    # We'll use schedule: at iteration t compute alpha = min(target_alpha, max(1.0, t * target_alpha / max_iters))
    # could use quadratic schedule.
    for t in range(1, max_iters+1):
        if time.time() > deadline: break
        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        # compute smoothed gradient weight
        # choose alpha based on iteration
        # Let's use shape: alpha = 1 + (target_alpha - 1)*(t / max_iters)**2
        alpha = 1.0 + (target_alpha - 1.0) * (t / max_iters)**2
        # compute weights
        shifted = conv - max_conv  # to improve numeric
        w = np.exp(alpha * shifted)
        w_sum = np.sum(w)
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / len(w))
        else:
            w /= w_sum

        # compute gradient wrt a:
        g_a = 2.0 * _cross_corr_w_a(w, a)

        # chain rule for log-softmax parameters
        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_ = beta2 * v_ + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v_ / (1.0 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_a -= step

        # evaluate hard objective
        cur_hard = _objective_surrogate(_softmax(log_a))
        if cur_hard < best_hard:
            best_hard = cur_hard
            best_seq = _softmax(log_a).copy()

    return best_seq, best_hard

def _adam_softmax_opt(start_seq, n, deadline, max_iters=500_000, lr0=0.2, beta1=0.9, beta2=0.999, eps=1e-8, target_alpha=200.0):
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v_hat = np.zeros_like(log_a)
    best_seq = None
    best_val = float('inf')
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        # record best based on hard objective
        cur_val = float(2.0 * n * max_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute smoothing weight
        progress = t / max_iters
        alpha = 1.0 + (target_alpha - 1.0) * (progress ** 2)
        # compute softmax weights for all conv entries
        shifted = conv - max_conv
        w = np.exp(alpha * shifted)
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # gradient wrt a
        g_a = 2.0 * _cross_corr_w_a(w, a)
        # gradient for log-softmax
        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)
        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v**2)
        m_hat = m / (1.0 - beta1**t)
        v_hat_corr = v_hat / (1.0 - beta2**t)
        step = lr0 * m_hat / (np.sqrt(v_hat_corr) + eps)
        log_a -= step
    # after loop, if best_seq is None (should not), compute from log_a
    if best_seq is None:
        best_seq = _softmax(log_a)
        best_val = float(2.0 * n * np.max(_conv_fft(best_seq, best_seq)))
    return best_seq, best_val

"""
Candidate proposer for the “step‑function” challenge.

Algorithm
---------
1. Build a diverse pool of probability vectors (length 4000):
   - the previous best (if any);
   - several exponential decays;
   - a symmetric triangular spike;
   - a “U‑shape” (high at the edges, low in the centre);
   - a few Beta‑distribution shapes (including the heavy‑tailed 0.5‑0.5);
   - the uniform distribution;
   - a handful of random Dirichlet draws.

2. For each seed run a *smoothed* Adam optimiser.
   The objective is
        F(a) = 2·n·max(a*a)                (n = 4000, sum(a)=1)
   and we replace the hard max by a soft‑max with a temperature α that
   grows from 1 to a large value (≈200).  This yields a differentiable
   surrogate whose gradient can be written in closed form:
        ∇_a F_s = 2·(w ★ a) ,
   where w_i = softmax_α( (a*a)_i ) and ★ denotes the (linear)
   cross‑correlation.  The optimisation variables are the log‑softmax
   parameters `v` (so that a = softmax(v) never violates non‑negativity
   or the simplex constraint).

3. The best sequence from step 2 is refined with the *hard‑max*
   Adam optimiser (the same routine used in the reference solution)
   to push the true max‑convolution down further.

4. Finally an exponentiated‑gradient (mirror descent) pass is applied
   for a few million iterations – this is the “polishing” phase from
   the reference code.

All steps respect the 1000 s time budget; the routine prints a short
log and stores the best vector under the name ``height_sequence_1`` so
that later runs can continue from it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a single FFT (autocorrelation)."""
    L = len(a) + len(a) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv


def _corr_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (non‑negative lags) of x and y.
    Returns the full correlation of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    corr = np.fft.irfft(fx * np.conj(fy), n=N)[:L]
    return corr


def _cross_corr_w_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """g_i = Σ_j w_{i+j}·a_j  (cross‑correlation restricted to i∈[0,n))."""
    # linear correlation, take the first n entries (non‑negative lags)
    return np.real(_corr_fft(w, a)[: len(a)])


# ----------------------------------------------------------------------
# Simple probability primitives
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (guaranteed to sum to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate used for monitoring: 2·n·max(conv(a,a))."""
    n = len(a)
    conv = _conv_self_fft(a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _triangular(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _v_shape(n: int) -> np.ndarray:
    """U‑shaped distribution: zero in the centre, linear increase to the edges."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.abs(i - centre) / centre
    a = np.maximum(a, 1e-12)          # avoid exact zeros for the log‑step
    s = a.sum()
    return a / s


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Symmetric Beta density on a regular grid, normalised to sum 1."""
    eps = 1e-12
    x = np.linspace(eps, 1.0 - eps, n, dtype=np.float64)
    vals = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    s = vals.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return vals / s


# ----------------------------------------------------------------------
# Adam optimiser with a *soft‑max* surrogate
# ----------------------------------------------------------------------
def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      lr0: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      target_alpha: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed objective.
    Returns the best probability vector (sum = 1) seen and its *hard*
    surrogate value 2·n·max(conv).
    """
    log_a = np.log(start_seq + 1e-30)               # initialise in log‑space
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        # ----- convolution and soft‑max weights -----
        conv = _conv_self_fft(a)                     # a*a
        max_conv = np.max(conv)

        # linear schedule for the temperature α (quadratic growth)
        progress = t / max_iters
        alpha = 1.0 + (target_alpha - 1.0) * (progress ** 2)

        # soft‑max over the convolution entries (shifted for stability)
        shifted = conv - max_conv                     # max element becomes 0
        w = np.exp(alpha * shifted)
        w_sum = np.sum(w)
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a -----
        g_a = 2.0 * _cross_corr_w_a(w, a)            # shape (n,)

        # ----- chain rule for log‑softmax parameters -----
        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)                     # ∇_v

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # parameter update
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # ----- keep the best hard‑max value seen -----
        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val

# Adam optimiser on the *hard* max (the version used in the reference)
# ----------------------------------------------------------------------
def _adam_hard_opt(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 300_000,
                   lr0: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the true max‑convolution objective (sub‑gradient only on the
    active index).  Returns best sequence and its surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)
        conv = _conv_self_fft(a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g_a = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g_a[idx_i] = 2.0 * a[rev_idx]

        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val

# Exponentiated‑gradient polishing (identical to the reference version)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Classic mirror‑descent (EG) on the *hard* max‑convolution objective.
    Returns best sequence and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_self_fft(a)
        k = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

    best_overall_val = float("inf")
    total_cands = len(candidates)

    for idx, cand in enumerate(candidates, start=1):
        if time.time() > deadline - 10:
            print("Time almost exhausted – leaving seed loop early.")
            break

        # give each seed a modest amount of work (≈30 k–150 k iters)
        remaining = deadline - time.time()
        est_iters = min(150_000,
                        max(30_000,
                            int(remaining * 1500 / (total_cands - idx + 1))))
        print(f"[Seed {idx}/{total_cands}] iterations ≈{est_iters}")

        seq, val = _adam_softmax_opt(cand, n, deadline,
                                     max_iters=est_iters,
                                     lr0=0.2,
                                     target_alpha=80.0)
        print(f"    → surrogate (hard max) after soft‑max Adam = {val:.6f}")

        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"    ** new global best surrogate = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # Second stage: longer refinement on the current best (hard‑max Adam)
    # ------------------------------------------------------------------
    if best_overall_seq is not None and time.time() < deadline - 10:
        remaining_iters = int((deadline - time.time()) * 1200)  # rough guess
        remaining_iters = max(200_000, remaining_iters)
        print("\n--- Refinement with hard‑max Adam (long run) ---")
        seq, val = _adam_hard_opt(best_overall_seq, n, deadline,
                                  max_iters=remaining_iters,
                                  lr0=0.15)
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"    → improved global best after hard‑max Adam = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with exponentiated gradient (as in the reference)
    # ------------------------------------------------------------------
    if best_overall_seq is not None and time.time() < deadline:
        print("\n--- Polishing with exponentiated‑gradient (hard max) ---")
        polished_seq, polished_val = _exp_gradient_opt(
            best_overall_seq, n, deadline,
            max_iters=4_000_000,
            eta0=12.0)
        if polished_val < best_overall_val:
            best_overall_seq = polished_seq
            best_overall_val = polished_val
            print(f"[Polish] improved surrogate = {best_overall_val:.6f}")
        else:
            print(f"[Polish] no improvement (best stays {best_overall_val:.6f})")

    # ------------------------------------------------------------------
    # Evaluate with the provided secure function and store result
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_overall_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:      # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
