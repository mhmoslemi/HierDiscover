def _build_support(d, n, rng, max_trials=30, timeout=None):
    best_support = []
    theoretical_max = int(math.sqrt(2 * n * d))
    start = time.time()
    for trial in range(max_trials):
        if timeout is not None and time.time() > timeout:
            break
        perm = rng.permutation(n)

            # update contributions

"""
Step‑function optimizer based on sparse support with bounded sum‑multiplicity.

Idea
----
* The evaluation metric is 2·n·max(conv(a,a)) / (sum(a)²) and is scale‑invariant.
* If the support S = {i : a_i>0} has size K and every ordered sum i+j
  appears at most `d` times (i.e. the multiplicity of each convolution entry
  ≤ d), then
        max(conv) ≤ d / K²                (weights are uniform 1/K)
        objective ≤ 2·n·d / K² .
  For n = 4000 and the target 1.503 we need roughly
        K ≥ sqrt(2·n·d / 1.503).
  With d = 4 this gives K ≈ 150; the theoretical capacity allows up to
  K ≤ sqrt(2·n·d) ≈ 179.  Hence a set with ≤4 collisions and size ≳150
  already beats the previous best score.

* We build such a set by a greedy random‑order construction.
  For each candidate index we check whether adding it would keep every
  sum‑multiplicity ≤ d.  The test is inexpensive (O(|S|)) and we repeat the
  greedy process many times, keeping the largest set found.

* The final step is a short mirror‑descent (exponentiated‑gradient) polishing
  on the exact max‑convolution objective, starting from the uniform‑weight
  vector on the selected support.  This usually gives a modest further
  improvement without destroying the low‑collision structure.

The whole routine respects the 1000 s budget, prints progress, stores the
result in the global ``height_sequence_1`` for possible continuation, and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogates 2·n·max(conv(seq,seq))  (sum(seq) = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Greedy construction of a low‑collision support set
# ----------------------------------------------------------------------
def _build_best_support(d: int,
                        n: int,
                        rng: np.random.Generator,
                        deadline: float,
                        max_trials: int = 200) -> list[int]:
    """
    Greedy random‑order construction of a set S⊂{0,…,n‑1}
    such that each ordered sum i+j (0≤i,j<n) occurs at most `d` times.
    Returns the largest set found before `deadline`.
    """
    best_support: list[int] = []
    theoretical_max = int(math.sqrt(2 * n * d))

    trial = 0
    while time.time() < deadline and trial < max_trials:
        trial += 1
        perm = rng.permutation(n)                     # random order of candidates
        support: list[int] = []
        mult = np.zeros(2 * n - 1, dtype=np.uint8)    # multiplicities of ordered sums

        for idx in perm:
            if time.time() > deadline:
                break

            # self‑pair (i,i) contributes a single ordered term
            if mult[2 * idx] + 1 > d:
                continue

            # check whether adding `idx` would violate the bound for any sum
            ok = True
            for s in support:
                if mult[s + idx] + 2 > d:   # two ordered terms (s,idx) and (idx,s)
                    ok = False
                    break
            if not ok:
                continue

            # –––‑ add idx –––‑
            for s in support:
                mult[s + idx] += 2
            support.append(int(idx))
            mult[2 * idx] += 1

        if len(support) > len(best_support):
            best_support = support
            print(f"[Support] trial {trial}: new best size = {len(best_support)}")
            # early exit if we are essentially at the theoretical limit
            if len(best_support) >= theoretical_max - 2:
                break

    return best_support


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (mirror descent on the max‑conv)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact objective.
    Works on the simplex via a log‑softmax update (mirror descent).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = max_idx - i_range
        g[i_range] = 2.0 * a[j_range]          # factor 2 accounts for ordered pairs

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (length 4000) that minimises
    ``evaluate_sequence``.  The algorithm builds a sparse support with at most
    four collisions per convolution entry, assigns uniform weight on that
    support, and finally runs a short exponentiated‑gradient polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0        # small safety margin

    n = 4000
    d = 4                                        # max ordered multiplicity per sum

    # ------------------------------------------------------------------
    # 1. Build a low‑collision support set
    # ------------------------------------------------------------------
    print("[Stage 1] Greedy low‑collision support construction")
    support = _build_best_support(d, n, rng, deadline,
                                  max_trials=200)
    K = len(support)
    if K == 0:
        raise RuntimeError("Failed to construct a non‑empty support set.")

    print(f"  Support size obtained: {K}")

    # ------------------------------------------------------------------
    # 2. Initialise the sequence: uniform weight on the support
    # ------------------------------------------------------------------
    seq = np.zeros(n, dtype=np.float64)
    w = 1.0 / K
    seq[support] = w
    # normalise just in case (should already be 1)
    seq_sum = seq.sum()
    if seq_sum == 0.0:
        raise RuntimeError("Zero sum after initialisation.")
    seq /= seq_sum

    # ------------------------------------------------------------------
    # 3. Optional polishing with exact EG (a few seconds only)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Stage 2] Exact exponentiated‑gradient polishing")
        seq, val = _exp_gradient_opt(seq, n, deadline,
                                     max_iters=150_000,
                                     eta0=5.0)
        print(f"  Polishing surrogate value: {val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final evaluation with the provided black‑box function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {_objective_surrogate(seq):.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible continuation runs
    globals()["height_sequence_1"] = seq.tolist()
    return seq.tolist()
