"""
Advanced step‑function optimizer – deep smoothed L‑BFGS

The core idea:
* Work on the ℓ₁‑simplex using a soft‑max (logits → probabilities) to keep the
  variables non‑negative and automatically normalised.
* Approximate the hard max in the objective
      2·n·max(conv(a,a))
  by a smooth log‑sum‑exp (soft‑max) with a schedule of increasing β.
  For a fixed β the objective is
      2·n·[ shift + log Σ exp(β·(conv−shift))/β ]
  with shift = max(conv) to keep the exponential stable.
* Use SciPy’s L‑BFGS‑B on this smooth surrogate.  For each β we run a
  substantial number of L‑BFGS iterations (up to a few thousand) and keep the
  best iterate according to the exact (hard‑max) surrogate.
* Run the optimiser from several diverse seeds (previous best, the classic
  inverse‑sqrt shape, a random Dirichlet draw) and keep the overall best
  sequence.
* Finally polish the result with a short exponentiated‑gradient (EG) phase
  that directly reduces the true max‑convolution.

All work respects the supplied wall‑clock budget (default 1000 s) and stores
the best sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# SciPy is optional – the optimiser needs it, otherwise we fall back to a
# simple EG loop.
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) for sum(seq)==1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – provide good starting points
# ----------------------------------------------------------------------
def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Extremely sparse random draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Deep smoothed L‑BFGS optimiser
# ----------------------------------------------------------------------
def _lbfgs_smooth_opt(
    start_a: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule,
    max_iter_per_beta: int = 3000,
):
    """
    Minimise the smoothed max‑convolution objective using L‑BFGS‑B.
    Returns the best sequence (ℓ₁‑normalised) seen and its exact surrogate.
    """
    if _opt is None:
        # Fall back to a tiny EG loop – should never happen on the platform.
        return _exp_gradient_opt(start_a, n, deadline)

    # Log‑softmax parameters
    v = np.log(start_a + 1e-30)
    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            # stable soft‑max over the convolution values
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun,
                v,
                method="L-BFGS-B",
                jac=True,
                callback=callback,
                options={"maxiter": max_iter_per_beta, "ftol": 1e-14, "disp": False},
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception at β={beta_val}: {e}")
            break

        a_cur = _softmax(v)
        cur_val = _objective_surrogate(a_cur)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a_cur.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple exponentiated‑gradient (EG) optimiser – used for a final polish
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_a: np.ndarray, n: int, deadline: float,
                     max_iters: int = 500_000, eta0: float = 8.0, tol: float = 1e-10):
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    Returns the best sequence encountered and its exact surrogate value.
    """
    a = start_a.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient (average over active indices)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for k in active:
            rev = k - idx
            mask = (rev >= 0) & (rev < n)
            g[mask] += a[rev[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑style momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a -= eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

            max_iter_per_beta=2500,
        )
        print(f"  → L‑BFGS surrogate = {val_lbfgs:.8f}")

        # 2) short EG polishing (helps escape tiny plateaus)
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(
                start_a=seq_lbfgs,
                n=n,
                deadline=sub_deadline,
                max_iters=400_000,
                eta0=6.0,
                tol=1e-12,
            )
            print(f"  → EG polish   surrogate = {val_eg:.8f}")
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # keep the best of the two phases
        candidate_seq, candidate_val = (seq_eg, val_eg) if val_eg < val_lbfgs else (seq_lbfgs, val_lbfgs)

        if candidate_val < global_best_val:
            global_best_val = candidate_val
            global_best_seq = candidate_seq.copy()
            print(f"   *** New global best surrogate = {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Final intensive polishing on the current global best (use remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline and global_best_seq is not None:
        print("\n--- Final polishing on global best ---")
        final_seq, final_val = _exp_gradient_opt(
            start_a=global_best_seq,
            n=n,
            deadline=deadline,
            max_iters=800_000,
            eta0=4.0,
            tol=1e-12,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.8f}")

    # --------------------------------------------------------------
    # Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.8f}, true evaluate_sequence = {true_val:.8f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
