"""
Improved search for a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Problem size** – Use n = 4000 (the same size as before).  The
   objective is homogeneous, therefore we work with vectors that sum to 1
   (probability vectors).  The true score is 2·n·max(conv(a,a)).

2. **Initial pool** – A small set of diverse seeds:
   * the best sequence from a previous run (global ``height_sequence_1``);
   * a few exponential decays;
   * a symmetric triangular shape;
   * a handful of Dirichlet draws (α = 0.5 gives spiky patterns).

   All seeds are normalised to sum = 1.

3. **Stage 1 – Smoothed Adam** – For each seed we run Adam on the
   *smoothed* max‑convolution objective.
   The smoothing uses a soft‑max over the convolution entries with a
   temperature parameter α that is increased from a small value (≈0.1)
   to a larger value (≈30) during the run.  The gradient of the
   smoothed objective is the *cross‑correlation* between the current
   probability vector ``a`` and the soft‑max weights ``p``.  Using the
   FFT we obtain both the convolution and the cross‑correlation in
   O(n log n) time per iteration.

   The optimisation is performed in the log‑soft‑max domain
   (variables ``v`` with ``a = softmax(v)``) so that non‑negativity
   and the simplex constraint are satisfied automatically.

4. **Stage 2 – Precise Adam** – The best vector from Stage 1 is refined
   with a short Adam run that uses the *exact* max‑convolution gradient
   (the previous approach).  This sharpens the solution after the
   smoothing has already moved us out of poor local minima.

5. **Stage 3 – Exponentiated‑gradient polishing** – Finally a long
   decreasing‑step EG pass (the same routine used in the reference
   solution) is applied to squeeze a few extra digits.

6. **Time management** – The total budget (default 1000 s) is split
   among the candidates.  The loop monitors the wall‑clock and stops
   before the deadline, otherwise the current best vector is returned.

The whole routine is self‑contained, uses only NumPy and respects the
time limit.  It returns a list of non‑negative heights that should
improve on the previous bound (≈ 1.5085) and aim for the target
≈ 1.503.

# FFT utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _cross_corr_fft(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (a ⋆ p) for non‑negative lags.
    Returns the vector g of length len(a) where
        g[i] = 2 * Σ_k p_k * a_{k-i}   (indices outside [0,n) are zero).
    """
    n = a.shape[0]
    m = p.shape[0]          # = 2*n-1
    L = n + m - 1           # = 3*n-2
    N = _next_pow2(L)

    A = np.fft.rfft(a, n=N)
    P = np.fft.rfft(p, n=N)

    # cross‑correlation via A * conj(P)
    corr = np.fft.irfft(A * np.conj(P), n=N)[:L]

    # Positive lags start at index m‑1
    start = m - 1
    g = 2.0 * corr[start:start + n]
    return g


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used for optimisation: 2·n·max(conv(seq,seq)).
    The function returns max(conv) – the factor 2·n is constant for a
    fixed n and can be applied at the end.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _triangular(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

                     lr0: float = 0.2,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8,
                     alpha_start: float = 0.1,
                     alpha_end: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max (temperature α) smoothing of the
    max‑convolution.  Returns the best probability vector seen and its
    maximal convolution value (not multiplied by 2·n).
    """
    # log‑softmax variables
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    a = _softmax(log_a)
    best_seq = a.copy()
    best_max = _objective_surrogate(a)   # max(conv)

    # constant FFT size covering the longest linear operation
    N = _next_pow2(3 * n - 2)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        # ----- convolution a * a -----
        A = np.fft.rfft(a, n=N)
        conv_full = np.fft.irfft(A * A, n=N)[: 2 * n - 1]

        # temperature schedule
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iters)

        # ----- soft‑max weights over the convolution -----
        max_c = conv_full.max()
        shifted = conv_full - max_c               # for numerical stability
        exp_term = np.exp(alpha * shifted)
        p = exp_term / exp_term.sum()             # length 2n‑1, sums to 1

        # ----- gradient: cross‑correlation a ⋆ p -----
        g = _cross_corr_fft(a, p)                 # length n

        # chain rule for log‑softmax variables
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # keep the best seen by the *true* max (not the smoothed one)
        cur_max = conv_full.max()
        if cur_max < best_max:
            best_max = cur_max
            best_seq = _softmax(log_a).copy()

    return best_seq, best_max


# ----------------------------------------------------------------------
# Adam on the *exact* max‑convolution (as in the reference code)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 80_000,
              lr0: float = 0.2,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that uses the gradient of the *active* convolution entry
    (i.e. the true max).  Returns the best sequence and its maximal
    convolution value.
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    a = _softmax(log_a)
    best_seq = a.copy()
    best_max = _objective_surrogate(a)

    N = _next_pow2(2 * n - 1)  # size needed for a*a only

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        # convolution a*a
        A = np.fft.rfft(a, n=N)
        conv_full = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
        k = int(np.argmax(conv_full))

        # gradient of the active entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        # chain rule
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        cur_max = conv_full.max()
        if cur_max < best_max:
            best_max = cur_max
            best_seq = _softmax(log_a).copy()

    return best_seq, best_max


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (identical to the reference version)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 4_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns best sequence and its maximal convolution value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_max = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    N = _next_pow2(2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        A = np.fft.rfft(a, n=N)
        conv_full = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
        k = int(np.argmax(conv_full))

        # sub‑gradient of the active entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_max = conv_full.max()
        if cur_max < best_max:
            best_max = cur_max
            best_seq = a.copy()

    return best_seq, best_max

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (length 4000) that minimises
    ``evaluate_sequence``.  The routine builds a small pool of seeds,
    refines each with a two‑stage Adam optimiser (smoothed → exact
    gradient), finishes with a long exponentiated‑gradient polishing,
    and returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin (~5 s) for the final evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem dimension
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build initial candidate pool
    # ------------------------------------------------------------------
    candidates = []

    # previous best, if present
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                candidates.append(arr / arr.sum())
        except Exception:
            pass

    # exponential decays
    for rate in (0.0005, 0.001, 0.002, 0.005, 0.01):
        candidates.append(_exp_decay(n, rate))

    # triangular shape
    candidates.append(_triangular(n))

    # uniform baseline
    candidates.append(_uniform(n))

    # a few random Dirichlet draws
    for _ in range(3):
        cand = rng.dirichlet(alpha=np.full(n, 0.5))
        candidates.append(cand)

    # ------------------------------------------------------------------
    # Optimise each candidate
    # ------------------------------------------------------------------
    best_overall_seq = None
    best_overall_max = float("inf")   # store maximal convolution (smaller → better)

    # Rough per‑candidate budget (will be trimmed by remaining time)
    rough_iters_stage1 = 200_000
    rough_iters_stage2 = 80_000

    for idx, cand in enumerate(candidates, start=1):
        # Normalise to a probability vector
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s

        # Check remaining time before starting a new candidate
        time_left = deadline - time.time()
        if time_left < 15.0:
            break

        # Allocate iteration budget proportionally to the time left
        iters1 = min(rough_iters_stage1,
                     max(20_000,
                         int((time_left - 30.0) / (len(candidates) - idx + 1) * 0.6)))
        iters2 = min(rough_iters_stage2,
                     max(10_000,
                         int((time_left - 30.0) / (len(candidates) - idx + 1) * 0.4)))

        print(f"[Candidate {idx}/{len(candidates)}] "
              f"stage‑1 iters = {iters1}, stage‑2 iters = {iters2}")

        # ---- Stage 1 : smoothed Adam ----
        seq1, max1 = _adam_opt_smooth(cand, n, deadline,
                                      max_iters=iters1,
                                      lr0=0.2,
                                      alpha_start=0.1,
                                      alpha_end=30.0)

        print(f"   → max(conv) after stage‑1 = {max1:.6e}")

        # ---- Stage 2 : exact‑max Adam ----
        seq2, max2 = _adam_opt(seq1, n, deadline,
                               max_iters=iters2,
                               lr0=0.15)

        print(f"   → max(conv) after stage‑2 = {max2:.6e}")

        # Update global best
        if max2 < best_overall_max:
            best_overall_max = max2
            best_overall_seq = seq2.copy()
            print(f"   ** new global best (max(conv)) = {best_overall_max:.6e}")

    # ------------------------------------------------------------------
    # Final polishing (exponentiated‑gradient)
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # fallback – a simple exponential decay
        best_overall_seq = _exp_decay(n, rate=0.001)

    # Polishing pass – use remaining time
    if time.time() < deadline:
        print("\n--- Polishing with exponentiated‑gradient ---")
        polished_seq, polished_max = _exp_gradient_opt(
            best_overall_seq, n, deadline,
            max_iters=4_000_000, eta0=12.0)

        if polished_max < best_overall_max:
            best_overall_seq = polished_seq
            best_overall_max = polished_max
            print(f"[Polish] improved max(conv) = {best_overall_max:.6e}")
        else:
            print("[Polish] no improvement over current best")

    # ------------------------------------------------------------------
    # Evaluate the true objective via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
        surrogate_val = 2.0 * n * best_overall_max
        print(f"\n=== Final result ===\n"
              f"Surrogate (2·n·max) ≈ {surrogate_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store the best sequence for possible continuation runs
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
