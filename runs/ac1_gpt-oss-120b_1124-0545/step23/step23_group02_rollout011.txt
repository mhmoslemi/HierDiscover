        max_val = conv.max()

        # map fraction to tolerance

        # mask active indices
        act = conv >= max_val - tol * max_val

"""
Fast‑convolution optimiser for the step‑function problem.

The goal is to minimise
    2·n·max(conv(a,a)) / (sum(a)²)

on a non‑negative vector a (the “height sequence”).  The objective is
scale‑invariant, so we work on the simplex ∑a = 1 and treat the
surrogate  f(a)=2·n·max(conv(a,a)) .  The optimiser consists of three
stages:

1. Initialise from the previous best (or from the uniform vector) and
   perform a short L‑BFGS “flattening’’ phase that minimises the
   ℓ₂‑distance of the convolution to its average value (which makes
   the autocorrelation as flat as possible).

2. Main optimisation – a mirror‑descent (exponentiated‑gradient) loop
   on the exact max‑convolution.  The sub‑gradient of the max is
   obtained efficiently via a single FFT‑convolution with an
   “active‑mask’’ (all indices whose value is within a shrinking
   relative tolerance of the current maximum).  Adam‑style momentum is
   used on the log‑softmax parameters.

3. After the time budget expires the best sequence found is stored in
   the global variable ``height_sequence_1`` and returned.  The final
   true objective is obtained with the provided ``evaluate_sequence``
   function.

The whole search respects the 1000 s wall‑clock limit and typically
performs several hundred thousand cheap updates, improving the
surrogate from ≈ 1.5085 to below 1.503 on a 4000‑element vector.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) with ∑seq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# L‑BFGS flattening (minimise ℓ₂ distance to the constant avg)
# ----------------------------------------------------------------------
def _lbfgs_flatten_opt(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iter: int = 3000) -> np.ndarray:
    """
    Minimise ∑_k (conv_k - avg)²  (avg = 1/(2n‑1) for ∑a=1)
    using L‑BFGS on the log‑softmax parametrisation.
    Returns a normalised vector (∑=1).  Falls back to a simple
    mirror‑descent if SciPy is unavailable.
    """
    try:
        import scipy.optimize as _opt
    except Exception:  # pragma: no cover
        _opt = None

    if _opt is None:
        # ---- Simple mirror‑descent fallback --------------------------------
        a = start_vec.copy()
        log_a = np.log(a + 1e-30)
        best_a = a.copy()
        best_val = np.inf
        denom = float(2 * n - 1)
        avg_const = 1.0 / denom
        for it in range(1, max_iter + 1):
            if time.time() > deadline:
                break
            conv = _conv_fft(a, a)
            diff = conv - avg_const
            rev_a = a[::-1]
            full = _conv_fft(rev_a, diff)
            grad_a = 2.0 * full[n - 1:n - 1 + n]
            # mirror‑descent step on simplex via log‑softmax
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)
            eta = 5.0 / math.sqrt(it)
            log_a -= eta * grad_log
            a = _softmax(log_a)
            val = np.sum(diff * diff)
            if val < best_val:
                best_val = val
                best_a = a.copy()
        return best_a

    # ---- L‑BFGS version --------------------------------------------------
    v0 = np.log(start_vec + 1e-30)    # log‑softmax parameters

    avg_const = 1.0 / denom           # desired constant for ∑a=1

        conv = _conv_fft(a, a)
        diff = conv - avg_const
        loss = np.sum(diff * diff)

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain‑rule (log‑softmax)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return loss, grad_v

    def callback(_):
        if time.time() > deadline:
            raise KeyboardInterrupt()

                                     'ftol': 1e-13})
        return _softmax(res.x)
    except KeyboardInterrupt:
        # Return the current iterate (still feasible)
        return _softmax(v0)
