def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      lr0: float = 0.5,
                      beta0: float = 0.001,
                      beta_max: float = 20.0,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8):
    # ensure positivity
    start_seq = np.maximum(start_seq, 1e-30)
    log_a = np.log(start_seq)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = _softmax(log_a)
        # compute conv a*a
        conv_aa = _conv_fft(a, a)
        max_b = np.max(conv_aa)
        # compute softmax weights
        beta = beta0 + (beta_max - beta0) * (t / max_iters)
        # stable exponent
        exp_term = np.exp(beta * (conv_aa - max_b))
        w = exp_term / np.sum(exp_term)
        # gradient wrt a
        a_rev = a[::-1]
        conv_wa_rev = _conv_fft(w, a_rev)  # length 3n-2
        start = n - 1
        grad_a = 2.0 * conv_wa_rev[start:start + n]
        # chain rule
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # Adam moments
        m = beta1 * m + (1 - beta1) * grad_v
        v = beta2 * v + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step
        # evaluate current surrogate (max)
        cur_val = np.max(_conv_fft(_softmax(log_a), _softmax(log_a))) * 2 * n  # optional compute directly
        # Actually compute surrogate via helper
        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()
    return best_seq, best_val

"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence(seq).

Algorithm – smooth‑max Adam + hard‑max polishing
------------------------------------------------
1. **Parameterisation** – a = softmax(v) guarantees a_i ≥ 0 and Σa_i = 1,
   which makes the objective scale‑invariant.

2. **Smooth approximation of the max** – for the convolution b = a∗a we
   use a temperature‑controlled soft‑max
        p_k = exp(β·b_k) / Σ_j exp(β·b_j) .
   The loss L_β = Σ_k p_k·b_k smoothly approximates max(b).  Its
   gradient w.r.t. a is
        ∇_a L_β = 2·(p * a_rev)[n‑1 : n‑1+n]            # a_rev = a[::-1]
   (full convolution followed by a slice).

3. **Adam optimisation in log‑space** – we keep the Adam moments for
   the gradient on v (the log‑softmax variables).  β is increased
   linearly from a tiny value (≈10⁻³) to a fairly large one (≈20),
   gradually turning the smooth loss into the true max.

4. **Candidate pool** – the previous best sequence (if any) together
   with a few analytic families (exponential decays, a symmetric
   triangle, random Dirichlet draws).  Each candidate is optimised
   with the procedure above; the best among them is kept.

5. **Hard‑max polishing** – a final mirror‑descent pass (identical to
   the reference implementation) uses the exact max‑gradient to
   squeeze the remaining few digits.

6. **Time management** – the routine respects the 1000 s budget,
   aborting early if the deadline is approached.  The best sequence
   found is stored in the global variable ``height_sequence_1`` for
   possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for length ≤ 2ⁿ)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _triangular(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (log‑softmax variables)
# ----------------------------------------------------------------------
def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      lr0: float = 0.2,
                      beta0: float = 0.001,
                      beta_max: float = 20.0,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth approximation of the max‑convolution.
    Returns the best probability vector (sum = 1) and the corresponding
    surrogate value 2·n·max(conv(a,a)).
    """
    # initialise log‑softmax variables
    start_seq = np.maximum(start_seq, 1e-30)
    log_a = np.log(start_seq)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    a = _softmax(log_a)
    conv_aa = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv_aa)
    best_seq = a.copy()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        # --- convolution a*a -------------------------------------------------
        conv_aa = _conv_fft(a, a)            # length 2n‑1
        max_b = np.max(conv_aa)              # max entry of a*a

        # --- soft‑max weights over the convolution entries --------------------
        beta = beta0 + (beta_max - beta0) * (t / max_iters)
        # stable exponentiation
        exp_term = np.exp(beta * (conv_aa - max_b))
        w = exp_term / exp_term.sum()        # w_k ≈ softmax_β(b_k)

        # --- gradient w.r.t. a ------------------------------------------------
        a_rev = a[::-1]
        conv_wr = _conv_fft(w, a_rev)       # length 3n‑2
        start = n - 1
        grad_a = 2.0 * conv_wr[start:start + n]

        # --- chain rule for log‑softmax --------------------------------------
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # --- Adam moments ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # --- parameter update -------------------------------------------------
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # --- keep the best vector seen ---------------------------------------
        cur_val = 2.0 * n * max_b
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Hard‑max polishing (same as the reference code)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the true max‑gradient.
    Returns the best sequence and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry (same derivation as before)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val
