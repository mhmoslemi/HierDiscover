"""
Propose a candidate step‑function (non‑negative heights) that minimises
``evaluate_sequence``.

The algorithm uses a single long Adam optimisation on a smooth
log‑sum‑exp approximation of the max‑convolution.  A schedule of
increasing smoothing parameters β is employed; the final β is
very large so the softmax becomes essentially a hard max, i.e. we
optimise the true objective.  The optimisation works in the
log‑softmax (probability‑vector) parametrisation, guaranteeing non‑negativity
and unit sum at all times.  A single well‑chosen seed (the previous best
solution padded to a larger length) is refined for the whole time budget.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Global deadline – used by inner loops to avoid overrunning.
# ----------------------------------------------------------------------
_global_deadline: float = 0.0


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using real FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(seq)==1):
        2 * n * max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (always return a normalised probability vector).
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the β‑smoothed log‑sum‑exp surrogate.
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   beta_schedule: List[float],
                   steps_per_beta: int,
                   final_beta: float,
                   final_steps: int,
                   lr: float,
                   beta1: float,
                   beta2: float,
                   eps: float,
                   deadline: float) -> Tuple[np.ndarray, float]:
    """
    Optimise ``start_seq`` (already normalised to sum==1) using Adam on a
    smoothed max‑convolution objective.  Returns the best sequence found
    and its exact surrogate value.
    """
    n = len(start_seq)
    # work in log‑softmax space
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    t = 0

    # track the best we have seen (exact surrogate)
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    def _step(beta: float) -> bool:
        """Perform a single Adam step for a given β.  Returns False if out of time."""
        nonlocal log_a, m, v, t, best_seq, best_val

        if time.time() > deadline:
            return False

        # ---- soft‑max to a probability vector (sum == 1) ----
        shift_log = np.max(log_a)
        exp_shift = np.exp(log_a - shift_log)
        a = exp_shift / exp_shift.sum()

        # ---- convolution a * a (full) ----
        conv = _conv_fft(a, a)

        # ---- β‑softmax weights for the smoothed max ----
        max_conv = np.max(conv)
        # subtract max for stability
        exp_beta = np.exp(beta * (conv - max_conv))
        sum_exp = exp_beta.sum()
        w = exp_beta / sum_exp

        # ---- gradient w.r.t. a (2 * rev_a * w) ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3·n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # slice length n

        # ---- chain‑rule to obtain gradient w.r.t. log‑softmax parameters ----
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep log‑parameters bounded (helps numeric stability)
        log_a = np.clip(log_a, -30.0, 30.0)

        # ---- occasional exact evaluation ----
        if t % 1000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"    [Iter {t}] improved surrogate = {best_val:.6f}")

        return True

    # ------------------------------------------------------------------
    # Coarse schedule (moderate β) – many cheap steps.
    # ------------------------------------------------------------------
    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if not _step(beta):
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # Final fine phase with a very sharp β (almost hard max).
    # ------------------------------------------------------------------
    for _ in range(final_steps):
        if not _step(final_beta):
            break
        if time.time() > deadline:
            break

    # make sure the best_seq is a proper probability vector
    best_seq = np.clip(best_seq, 0.0, None)
    best_seq /= best_seq.sum() + 1e-30
    best_val = _objective_surrogate(best_seq)
    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

    Strategy
    --------
    1.  Use the previously found solution (if any) padded to a larger length
        (8000) as the main starting point; also generate a few analytic
        seeds (uniform, exponentials, Dirichlet draws).
    2.  Pick the seed with the smallest surrogate value.
    3.  Run a long Adam optimisation on a smoothed surrogate
        (log‑sum‑exp of the self‑convolution) with a schedule of β values.
        The final β is huge, making the surrogate essentially the true max.
    4.  Return the best sequence found.
    """
    global _global_deadline
    rng = np.random.default_rng(seed)

    start_time = time.time()
    _global_deadline = start_time + budget_s - 2.0   # safety margin
    deadline = _global_deadline

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more freedom to flatten the convolution.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise a pool of seeds.
    # ------------------------------------------------------------------
    seeds: List[np.ndarray] = [
        _uniform(n),
        _exp_decay(n, 0.001),
        _exp_decay(n, 0.004),
        _sym_exp_decay(n, 0.0025),
        _power(n, 0.8),
        _power(n, 1.2),
    ]

    # Dirichlet draws with different concentration parameters.
    for conc in (0.02, 0.1, 0.5, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Include the previous solution (if present) padded to length n.
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.ndim == 1:
                # pad or truncate to required length
                if prev_arr.size < n:
                    padded = np.zeros(n, dtype=np.float64)
                    padded[:prev_arr.size] = prev_arr
                    prev_vec = padded
                else:
                    prev_vec = prev_arr[:n]
                prev_vec = np.clip(prev_vec, 0.0, None)
                if prev_vec.sum() > 0:
                    prev_vec = prev_vec / prev_vec.sum()
                    seeds.append(prev_vec)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Quick surrogate evaluation – keep the best seed.
    # ------------------------------------------------------------------
    evaluated: List[Tuple[np.ndarray, float]] = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda p: p[1])
    best_seed, best_surrogate = evaluated[0]
    print(f"\n--- Chosen seed surrogate = {best_surrogate:.6f}")

    # ------------------------------------------------------------------
    # Adam optimisation schedule.
    # ------------------------------------------------------------------
    beta_schedule = [5, 20, 80, 300, 1000, 5000, 20000, 100000]
    steps_per_beta = 15000            # moderate number of steps per β
    final_beta = 500000               # almost a hard max
    final_steps = 200000              # long fine‑tuning phase
    lr = 0.07                         # Adam learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    print("\n--- Starting Adam optimisation ---")
    opt_seq, opt_val = _adam_optimize(best_seed,
                                      beta_schedule,
                                      steps_per_beta,
                                      final_beta,
                                      final_steps,
                                      lr,
                                      beta1,
                                      beta2,
                                      eps,
                                      deadline)

    # ------------------------------------------------------------------
    # Final evaluation with the provided function.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(opt_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate = {opt_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for the next run
    globals()["height_sequence_1"] = opt_seq.tolist()
    return opt_seq.tolist()
