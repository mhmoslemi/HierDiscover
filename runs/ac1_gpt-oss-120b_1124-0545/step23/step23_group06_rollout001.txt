"""
Propose a candidate step‑function (non‑negative vector) that minimises
``evaluate_sequence``.
The algorithm combines a rich pool of analytic seeds, a continuation
L‑BFGS optimiser on a smooth log‑sum‑exp surrogate of the max‑convolution,
a long exponentiated‑gradient polish and finally a stochastic “mass‑move”
annealing on the remaining budget.

Key ideas compared with the previous baseline:
*   many more analytic families (Gaussian, truncated Pareto, mixtures);
*   a more aggressive β‑schedule and twice as many L‑BFGS iterations;
*   a custom gradient‑descent refinement that can run for hundreds of
    thousands of iterations;
*   a cheap stochastic “mass‑move’’ annealing that transfers a small amount
    of probability mass from one index to another (instead of swapping);
*   dynamic time allocation – the best seed receives most of the remaining
    time.

All helper functions are top‑level; no closures or lambdas are used.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global deadline – shared with the L‑BFGS objective to avoid overruns.
# ----------------------------------------------------------------------
_global_deadline = None


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast for many calls)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate: 2·n·max(conv(seq,seq))  (seq is already normalised to sum=1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – each returns a **probability vector** (sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _gaussian(n: int, sigma: float, shift: float = 0.0) -> np.ndarray:
    """Discrete Gaussian centred at shift (may be non‑integer)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - shift) / sigma) ** 2)
    return a / a.sum()


def _pareto(n: int, shape: float, scale: float = 1.0) -> np.ndarray:
    """Truncated Pareto on {0,…,n‑1}; values ∝ (scale + i)^(-shape)."""
    i = np.arange(n, dtype=np.float64)
    a = (scale + i) ** (-shape)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    return a / a.sum()


def _mixture_two(n: int,
                 rng: np.random.Generator,
                 fam1,
                 fam2,
                 **kwargs) -> np.ndarray:
    """Random convex combination of two analytic families."""
    w = rng.random()
    a1 = fam1(n, **kwargs[0]) if isinstance(kwargs[0], dict) else fam1(n, **kwargs[0])
    a2 = fam2(n, **kwargs[1]) if isinstance(kwargs[1], dict) else fam2(n, **kwargs[1])
    a = w * a1 + (1.0 - w) * a2
    return a / a.sum()


# ----------------------------------------------------------------------
# L‑BFGS surrogate objective (log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
class _LbfgsObjective:
    """Callable compatible with scipy.optimize.minimize."""

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, log_a: np.ndarray):
        # soft‑max mapping to a probability vector (stable)
        m = np.max(log_a)
        exp_shift = np.exp(log_a - m)
        a = exp_shift / exp_shift.sum()

        conv = _conv_fft(a, a)
        shift = np.max(conv)                     # used for numerical stability
        beta = self.beta

        # smooth max via log‑sum‑exp (weights sum to 1)
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        w = exp_beta / sum_exp                     # weight vector of length 2n‑1

        # gradient w.r.t. a : 2 * conv(reversed a, w) (central part)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # gradient w.r.t. log‑softmax parameters
        grad_log = a * (grad_a - np.dot(grad_a, a))

        # smoothed objective value: M_β = shift + log(sum_exp)/β
        M_beta = shift + np.log(sum_exp) / beta
        obj = 2.0 * self.n * M_beta
        return float(obj), grad_log.astype(np.float64)


def _lbfgs_opt(start_seq: np.ndarray,
               beta_schedule,
               max_iter_per_beta: int,
               deadline: float):
    """
    Continuation L‑BFGS: run L‑BFGS on a sequence of increasingly sharp
    log‑sum‑exp surrogates.
    Returns best vector (normalised) and its surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)          # stay in log‑softmax space
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break

        obj = _LbfgsObjective(len(start_seq), beta)
        opts = {"maxiter": max_iter_per_beta, "disp": False}
        try:
            res = minimize(fun=obj,
                           x0=log_a,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception:                      # any interruption – stop early
            break

        log_a = res.x
        # back‑project to probability vector
        m = np.max(log_a)
        exp_shift = np.exp(log_a - m)
        a = exp_shift / exp_shift.sum()
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (sub‑gradient) optimiser – long polish
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0):
    """
    Mirrors the original EG polish but runs longer and with a simple
    schedule. Operates in the log‑space to keep the variables in the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)
    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active max entry (see original code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        log_a = np.clip(log_a, -30.0, 30.0)

        # project back to simplex via soft‑max
        m = np.max(log_a)
        exp_shift = np.exp(log_a - m)
        a = exp_shift / exp_shift.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic “mass‑move’’ annealing – cheap global tweaker
# ----------------------------------------------------------------------
def _mass_move_anneal(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 200_000):
    """
    At each iteration pick two indices i≠j and transfer a small ε of mass
    from i to j (if possible). Acceptance follows a standard simulated‑annealing
    schedule on the surrogate objective.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # Temperature schedule – start at the current value and decay geometrically
    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng()
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T_start * (T_end / T_start) ** (it / max_iters)

        # choose donor and receiver
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] < 1e-12:
            continue

        # ε drawn from a truncated exponential that respects the donor mass
        eps = rng.exponential(scale=0.01)
        eps = min(eps, cur_seq[i] * 0.9)  # keep donor non‑zero

        ns = cur_seq.copy()
        ns[i] -= eps
        ns[j] += eps

        val = _objective_surrogate(ns)
        if val < cur_val or rng.random() < math.exp(-(val - cur_val) / max(T, 1e-12)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

# Helper for sorting by second element
# ----------------------------------------------------------------------
def _second(pair):
    return pair[1]

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``. The pipeline:

    1. Assemble a diverse pool of analytic / random seeds.
    2. Keep the most promising few (surrogate value).
    3. Refine each with a continuation L‑BFGS (log‑sum‑exp surrogate).
    4. Polish the best candidate with a long exponentiated‑gradient run.
    5. Apply a cheap stochastic mass‑move annealing on the remaining time.
    6. Return the best vector (as a plain Python list) and store it globally.
    """
    global _global_deadline
    rng = np.random.default_rng(seed)
    start_time = time.time()
    _global_deadline = start_time + budget_s - 2.0            # safety margin
    deadline = _global_deadline

    # --------------------------------------------------------------
    # Problem size – a moderate length that balances resolution & speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load a previous solution (if any) to keep continuity across runs.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            prev_vec = None

    # --------------------------------------------------------------
    # 1. Build a generous pool of seed vectors.
    # --------------------------------------------------------------
    seeds = [
        _uniform(n),
        _exp_decay(n, 0.0015),
        _exp_decay(n, 0.0035),
        _sym_exp_decay(n, 0.0025),
        _power(n, 0.6),
        _power(n, 1.1),
        _beta_shape(n, 0.7, 0.7),
        _beta_shape(n, 1.5, 1.5),
        _gaussian(n, sigma=200.0, shift=n / 2),
        _gaussian(n, sigma=600.0, shift=n / 2),
        _pareto(n, shape=1.2, scale=1.0),
    ]

    # Dirichlet draws – breadth of concentration parameters.
    for conc in (0.02, 0.05, 0.15, 0.5, 1.0, 3.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random piecewise‑linear shapes.
    for nb in (5, 10, 20):
        for _ in range(2):
            seeds.append(_piecewise_linear(n, nb, rng))

    # Mixtures of two families (Gaussian + exponential, etc.)
    for _ in range(4):
        w = rng.random()
        g = _gaussian(n, sigma=rng.uniform(80.0, 600.0), shift=n / 2)
        e = _exp_decay(n, rate=rng.uniform(0.001, 0.004))
        mix = w * g + (1.0 - w) * e
        seeds.append(mix / mix.sum())

    # Include the previous solution and a few of its permutations.
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(3):
            seeds.append(rng.permutation(prev_vec))

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few.
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=_second)
    top_k = min(7, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed ranking (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Local refinement with continuation L‑BFGS.
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # A more aggressive schedule and more inner iterations.
    beta_schedule = [10, 50, 200, 1000, 5000, 20000, 100000]
    max_iter_per_beta = 500

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate roughly equal share of the remaining time to remaining seeds,
        # but give a small bonus to the best seed (idx=0) later.
        remaining = deadline - time.time()
        share = remaining * 0.15 / max(1, (top_k - idx))
        sub_deadline = time.time() + share
        print(f"\n[Seed {idx+1}] time budget ≈ {share:.1f}s")

        seq_lbfgs, val_lbfgs = _lbfgs_opt(seed_vec,
                                          beta_schedule,
                                          max_iter_per_beta,
                                          sub_deadline)

        # short EG polish to clean up any leftover sub‑gradient noise
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs,
                                              sub_deadline,
                                              max_iters=300_000,
                                              eta0=6.0)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # keep the better local result
        if val_lbfgs < val_eg:
            local_seq, local_val = seq_lbfgs, val_lbfgs
        else:
            local_seq, local_val = seq_eg, val_eg

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")
        if local_val < global_best_val:
            global_best_seq = local_seq

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if time.time() < deadline:

        final_seq, final_val = _exp_gradient_opt(global_best_seq,

            global_best_val = final_val

    # --------------------------------------------------------------
    # 5. Very cheap stochastic mass‑move annealing.
    # --------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 25.0          # a few seconds
        print("\n--- Mass‑move annealing (final tweak) ---")
        seq_last, val_last = _mass_move_anneal(global_best_seq.copy(),
                                               short_deadline,

        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final anneal] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result.
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"

    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
