"""
search for a low‑value step function.

The script follows three stages:

1️⃣  **Coarse parametrised search** – generates a few hundred
    simple decreasing shapes (uniform, exponential, power‑law and a
    product of both).  Each candidate is normalised to sum = 1 and the
    exact surrogate 2·n·max(conv)/sum² is evaluated (the sum is 1, so
    this is simply 2·n·max(conv)).  The best few candidates are kept.

2️⃣  **Deterministic refinement** – the current best candidate is refined
    with a limited‑iteration L‑BFGS optimisation on a smooth
    approximation of the L∞‑objective.  The smooth surrogate is the
    log‑sum‑exp smoothing used in the reference implementation, but we
    keep the smoothing parameter β very large (1e5) so that the
    optimisation essentially follows a sub‑gradient of the true max
    convolution.  This step usually drops the surrogate by a few‑
    thousandths.

3️⃣  **Stochastic hill‑climbing** – a long simulated‑annealing loop
    that repeatedly perturbs the vector.  Three move types are used:
      – Transfer a small fraction of mass from one index to another.
      – Add a tiny Gaussian perturbation to all entries (followed by
        renormalisation).
      – Swap two entries.
    Worsening moves are accepted with probability exp(‑Δ/T) where the
    temperature T is linearly cooled from 1 % of the current objective
    down to 1e‑6 × objective.  The best vector seen during this phase
    replaces the current best.

After the time budget (≈ 1000 s) the final vector is evaluated with the
provided ``evaluate_sequence`` black‑box and stored in the global
variable ``height_sequence_1`` for possible continuation runs.

All helper functions are top‑level, no closures or lambda‑expressions
are used, and the code works with only ``numpy`` (and ``scipy`` if
available) plus the already‑imported ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# --------------------------- FFT utilities ---------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# --------------------------- objective helpers ---------------------------

def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate: 2·n·max(conv(seq,seq)) / (sum(seq)²).
    ``seq`` is assumed non‑negative.  The function is cheap because the
    convolution is performed with FFT.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_c = np.max(conv)
    s = np.sum(seq)
    if s < 0.01:          # protects against division‑by‑zero (mirrors evaluate_sequence)
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Smoothed max‑convolution objective with its gradient
    (log‑sum‑exp smoothing).  The variables are the log‑softmax
    parameters ``v_local``; the returned gradient is w.r.t. these
    parameters.
    """
    a = _softmax(v_local)                     # probability vector (sum=1)
    conv = _conv_fft(a, a)                    # length = 2n‑1
    max_c = np.max(conv)
    shift = max_c                               # for numerical stability

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                     # weights over the convolution bins
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]      # slice back to length n

    # chain rule: ∂obj/∂v = a * (grad_a - (grad_a·a))
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# --------------------------- L‑BFGS refinement ---------------------------

def _lbfgs_opt(start_seq: np.ndarray,
               n: int,
               max_iter: int = 200) -> np.ndarray:
    """
    One‑shot L‑BFGS optimisation of the smoothed surrogate.
    The optimisation is performed on the log‑softmax variables.
    ``beta`` is fixed to a large value (1e5) so the surrogate is a very
    close approximation of the true ``max``.
    """
    try:
        from scipy.optimize import minimize
    except Exception as exc:                     # scipy unavailable – fall back
        print("SciPy not available, skipping L‑BFGS stage:", exc)
        return start_seq

    beta = 1e5

    v0 = np.log(start_seq + 1e-30)

    def fun(v):
        """Return (objective, gradient) for L‑BFGS."""
        obj, grad = _smooth_obj_and_grad(v, n, beta)
        return obj, grad

    res = minimize(fun,
                   v0,
                   method="L-BFGS-B",
                   jac=True,
                   options={"maxiter": max_iter, "disp": False})

    a_opt = _softmax(res.x)
    return a_opt


# --------------------------- hill‑climbing ---------------------------

def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 600_000) -> np.ndarray:
    """
    Simulated‑annealing hill‑climbing with three move types:
        * transfer a small fraction of mass from i → j,
        * add a tiny Gaussian perturbation to all entries,
        * swap two entries.
    The temperature drops linearly; the best vector ever seen is returned.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule (linear cooling)
    T_start = 0.01 * cur_val       # start at 1 % of current objective
    T_end = 1e-6 * cur_val         # finish almost zero

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear cooling
        T = T_start + (T_end - T_start) * (it / max_iters)

        # ---- move selection -------------------------------------------------
        r = rng.random()
        if r < 0.5:                                              # mass transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # transfer at most 1 % of the source mass
            frac = rng.uniform(0.001, 0.01)
            amt = cur_seq[i] * frac
            ns = cur_seq.copy()
            ns[i] -= amt
            ns[j] += amt

        elif r < 0.8:                                            # Gaussian perturbation
            # noise standard deviation decays with the iteration count
            sigma = 0.001 * (1.0 - it / max_iters) + 1e-5
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.clip(ns, 0.0, None)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s                                            # renormalise

        else:                                                    # swap
            i, j = rng.integers(0, n, size=2)
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        # --------------------------------------------------------------------
        val = _objective_surrogate(ns)

        if val < cur_val:                     # greedy accept
            cur_seq = ns
            cur_val = val
            if val < best_val:                # record global best
                best_seq = ns.copy()
                best_val = val
        else:                                 # Metropolis criterion
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq

    # exponential decay
    for lam in np.logspace(-5, -1, 30):
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a /= a.sum()
        candidates.append(a)

    # power‑law decay
    for p in np.linspace(0.0, 3.0, 30):
        a = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
        a /= a.sum()
        candidates.append(a)

    # product of exponential and power‑law (a modest grid)
    for lam in np.logspace(-5, -2, 10):
        for p in np.linspace(0.5, 2.0, 5):
            a = np.exp(-lam * np.arange(n, dtype=np.float64))
            a *= 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0.0:
                a /= s
                candidates.append(a)

    # ---------------------------------------------------------------
    # evaluate all candidates and keep the best one
    # ---------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for seq in candidates:
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    print(f"[Stage 0] Best surrogate after grid search = {best_val:.6f}")

    # ---------------------------------------------------------------
    # 2. Deterministic refinement with L‑BFGS (if time permits)
    # ---------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Stage 1] L‑BFGS refinement")
        seq_refined = _lbfgs_opt(best_seq, n, max_iter=200)
        val_refined = _objective_surrogate(seq_refined)
        print(f"   L‑BFGS surrogate = {val_refined:.6f}")
        if val_refined < best_val:
            best_val = val_refined
            best_seq = seq_refined.copy()

    # ---------------------------------------------------------------
    # 3. Intensive simulated‑annealing hill‑climb
    # ---------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Stage 2] Hill‑climb (transfer / noise / swap)")
        seq_hc = _hill_climb(best_seq, n, deadline, rng, max_iters=600_000)
        val_hc = _objective_surrogate(seq_hc)
        print(f"   Hill‑climb surrogate = {val_hc:.6f}")
        if val_hc < best_val:
            best_val = val_hc
            best_seq = seq_hc.copy()

    # ---------------------------------------------------------------
    # 4. Final evaluation with the black‑box function
    # ---------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:       # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # make the result globally available for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
