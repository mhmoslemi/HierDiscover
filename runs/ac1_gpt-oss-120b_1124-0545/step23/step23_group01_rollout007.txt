"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence(seq).

Algorithm
---------
1. Choose a problem size (n = 4000 by default).  The objective is
   homogeneous, so we work on the simplex (∑a = 1).

2. Build a tiny pool of seeds (previous best, exponential decays,
   triangular, uniform, a few Dirichlet draws).

3. For each seed run a *smooth* Adam optimiser.
   • The true objective is 2·n·max(conv(a,a)).  The non‑smooth max
     is replaced by a soft‑max with a temperature ε:
         𝑀ε(a) = (1/ε)·log ∑ₖ exp(ε·(a*a)ₖ) .
   • Its gradient can be written in closed form and is computed
     by two FFT‑based convolutions per iteration.
   • A schedule of increasing ε (0.5 → 2 → 5 → 10 → 20) first
     smooths the landscape and then refines the true max.

4. After the smoothing phases a short *hard‑max* Adam run (the
   original gradient on the active index) is performed to polish
   the solution.

5. The best vector found over all seeds is finally refined by an
   exponentiated‑gradient (mirror‑descent) pass (the same routine
   used in the reference solution).

6. The true objective is evaluated with the provided
   ``evaluate_sequence`` function, the result is printed and the
   vector is stored in ``height_sequence_1`` for possible continuation
   runs.

All FFT‑based convolutions use a power‑of‑two length for speed.
The total runtime never exceeds the given budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT.
    Returns array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convenient wrapper returning the full convolution."""
    return _conv_fft_full(x, y)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used during optimisation: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _triangular(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (soft‑max relaxation of the max)
# ----------------------------------------------------------------------
def _adam_opt_smooth(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     eps_schedule,
                     lr0: float = 0.2,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps_adam: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the *smoothed* max‑convolution
    objective.  ``eps_schedule`` is a list of (epsilon, iters) pairs.
    Returns the best sequence (probability vector) and its surrogate.
    """
    # Log‑softmax parameters
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    a = _softmax(log_a)
    best_seq = a.copy()
    best_val = _surrogate(a)

    total_iters = 0
    for eps_val, phase_iters in eps_schedule:
        for _ in range(phase_iters):
            if time.time() > deadline:
                return best_seq, best_val
            total_iters += 1

            a = _softmax(log_a)

            # 1) self‑convolution
            conv_self = _conv_fft(a, a)                # length 2n‑1

            # 2) soft‑max weights over the convolution entries
            scaled = eps_val * conv_self
            max_scaled = np.max(scaled)
            exp_scaled = np.exp(scaled - max_scaled)
            p = exp_scaled / np.sum(exp_scaled)        # shape (2n‑1,)

            # 3) gradient of the smoothed max w.r.t. a
            a_rev = a[::-1]                            # reversed a
            conv_rev = _conv_fft_full(a_rev, p)        # length 3n‑2
            grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]    # slice n entries

            # 4) chain rule for log‑softmax parameters
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # 5) Adam update
            m = beta1 * m + (1.0 - beta1) * grad_v
            v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** total_iters)
            v_hat = v / (1.0 - beta2 ** total_iters)

            step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
            log_a -= step

            # 6) surrogate evaluation (exact max, not the smoothed one)
            cur_val = _surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val

def _adam_opt_hard(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 100_000,
                   lr0: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps_adam: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam on the true max‑convolution (sub‑gradient at the active index)."""
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _surrogate(best_seq)

    total_iters = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        total_iters += 1

        a = _softmax(log_a)

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active entry
        g_a = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g_a[idx_i] = 2.0 * a[rev_idx]

        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** total_iters)
        v_hat = v / (1.0 - beta2 ** total_iters)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        log_a -= step

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (as in the reference solution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 4_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 123,

                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  The routine runs for at most ``budget_s``

    variable ``height_sequence_1`` and returns it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # reserve a tiny safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem dimension (keep the well‑studied size unless we decide otherwise)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build a modest pool of seeds
    # ------------------------------------------------------------------
    candidates = []

    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                candidates.append(arr / arr.sum())
        except Exception:
            pass

    for rate in (0.0005, 0.001, 0.002, 0.005):
        candidates.append(_exp_decay(n, rate))

    candidates.append(_triangular(n))
    candidates.append(_uniform(n))

    for _ in range(2):
        candidates.append(rng.dirichlet(alpha=np.full(n, 0.5)))

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    # schedule for the smoothed Adam phases (ε, iterations)
    eps_schedule = [
        (0.5, 50_000),
        (2.0, 50_000),
        (5.0, 100_000),
        (10.0, 100_000),
        (20.0, 100_000),
    ]

    best_overall_seq = None
    best_overall_val = float("inf")

    # ------------------------------------------------------------------
    # Process each seed
    # ------------------------------------------------------------------
    for idx, cand in enumerate(candidates, start=1):
        # Normalise to a probability vector
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:

        cand = cand / cand.sum()

        time_left = deadline - time.time()
