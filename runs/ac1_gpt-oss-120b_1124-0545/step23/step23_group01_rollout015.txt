The optimisation problem is  

\[
\min_{a\ge 0,\;\sum a_i=1}
\;2\,n\;\max_k (a*a)[k] ,
\qquad (a*a)[k]=\sum_i a_i a_{k-i}.
\]

`evaluate_sequence` computes exactly the quantity above, therefore the
inner product can be ignored – we only have to find a probability vector
with a small *maximum* autocorrelation.

The previous implementation used a sub‑gradient on the *active*
convolution index only.  
We improve on it by

1. **Smoothing the objective** – minimise the *ℓ₂*‑norm of the whole
   convolution (`∑ (a*a)[k]²`).  This objective is smooth and its gradient
   can be written as a single (cross‑)convolution, i.e. it can be evaluated
   with two FFTs per iteration.
   Minimising this surrogate flattens the whole autocorrelation curve,
   pushing the maximal entry down.

2. **Continuation** – after the ℓ₂‑phase we switch to the original
   max‑convolution sub‑gradient (the “max‑phase”) to polish the result.

3. **Adam on the log‑softmax parametrisation** – the variables are
   `v = log(a)` and `a = softmax(v)`.  Adam gives fast, adaptive updates
   while the softmax guarantees non‑negativity and the simplex constraint.

4. **Multiple seeds & local perturbations** – we start from a diverse pool
   (previous best, exponential decays, triangular, uniform, Dirichlet
   draws).  After the main optimisation a few random perturbations are
   tried to escape possible local minima.

All heavy work is performed with FFTs of size
`N = 2^⌈log₂(3·n‑2)⌉` (≈ 16384 for `n=4000`), therefore each iteration is
≈ O(N log N) and easily fits into the 1000 s budget.

The final sequence is stored in the global variable `height_sequence_1`
so that future runs can continue from the best found solution.

```python
"""
Hybrid optimiser for the step‑function problem.

Stage 1 – smooth ℓ₂‑objective:
    minimise   Σ_k (a*a)[k]²
    gradient  g = 4·(a ⋆ (a*a))  (cross‑correlation)
    optimisation via Adam on the log‑softmax parameters.

Stage 2 – max‑phase polishing:
    minimise   max_k (a*a)[k]  (sub‑gradient on the active index).

The code respects the 1000 s time limit, uses only NumPy, and stores the
best vector in the global name ``height_sequence_1`` for later runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _correlation_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Non‑negative lag correlation a ⋆ b  (length = len(a)).
    a : length n   (probability vector)
    b : length L   (full convolution, L = 2*n-1)
    Returns c[j] = Σ_i a_i·b_{i+j},   j = 0 … n‑1.
    """
    n = len(a)
    L = len(b)
    N = _next_pow2(n + L - 1)
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    corr = np.fft.irfft(A * np.conj(B), n=N)[:n]
    return corr


# ----------------------------------------------------------------------
# Simple helpers --------------------------------------------------------
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax, guarantees a non‑negative vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used for comparison:
        2·n·max(conv(seq,seq))
    (seq is already normalised to sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses ----------------------------------
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _triangular(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Adam optimiser for ℓ₂‑objective (smooth phase) -----------------------
# ----------------------------------------------------------------------
def _adam_l2_opt(start_seq: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 200_000,
                 lr0: float = 0.5,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth objective  Σ_k (a*a)[k]².
    Returns the best probability vector seen and its *max‑convolution*
    surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    # FFT size large enough for both a*a (length 2n‑1) and the correlation
    N = _next_pow2(3 * n - 2)          # ≈ 16384 for n=4000
    L = 2 * n - 1                     # length of full convolution

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        # forward FFT of a (once per iteration)
        FA = np.fft.rfft(a, n=N)

        # a*a in frequency domain
        conv_fft = FA * FA
        conv = np.fft.irfft(conv_fft, n=N)[:L]          # time‑domain convolution

        # correlation a ⋆ conv  (size n)
        corr = np.fft.irfft(FA * np.conj(conv_fft), n=N)[:n]

        # gradient of Σ conv²  :  g = 4·corr
        grad_a = 4.0 * corr

        # map to gradient w.r.t. log‑softmax variables
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # parameter update
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # evaluate the *true* surrogate (max‑convolution) to keep the best
        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser for max‑convolution (original sub‑gradient) ----------
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 150_000,
              lr0: float = 0.2,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the *exact* max‑convolution objective.
    Gradient uses only the active index (the one attaining the maximum).
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g_a = np.zeros_like(a)
        g_a[idx] = 2.0 * a[rev]

        # chain rule through the softmax
        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (as in the reference solution) ---
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 4_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient (mirror descent) on the true max‑convolution.
    Many tiny steps are cheap because only the active index is needed.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine uses:
      * several diverse seeds,
      * a smooth ℓ₂‑phase (Adam on Σ(conv²)),
      * a max‑phase (Adam on the exact max‑convolution),
      * a final exponentiated‑gradient polish,
      * a few random perturbations to escape local minima.
    The best vector is stored globally as ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    n = 4000

    # ------------------------------------------------------------------
    # Build a pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                candidates.append(arr / arr.sum())
        except Exception:
            pass

    for rate in (0.0005, 0.001, 0.002, 0.005):
        candidates.append(_exp_decay(n, rate))

    candidates.append(_triangular(n))
    candidates.append(_uniform(n))

    for _ in range(2):
        candidates.append(rng.dirichlet(alpha=np.full(n, 0.5)))

    if not candidates:
        candidates.append(_exp_decay(n, 0.001))

    # ------------------------------------------------------------------
    # Optimise each seed
    # ------------------------------------------------------------------
    best_overall_seq = None
    best_overall_val = float("inf")

    for idx, cand in enumerate(candidates, start=1):
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()

        time_left = deadline - time.time()
        if time_left < 12.0:
            break

        # allocate a rough budget per candidate (dynamic)
        # 40 % of remaining time → smooth ℓ₂ phase, 40 % → max‑phase,
        # remaining → possible perturbations.
        smooth_budget = min(150_000, int(time_left * 0.4 / 0.001))
        smooth_budget = max(smooth_budget, 10_000)

        print(f"[Candidate {idx}/{len(candidates)}] ℓ₂‑phase, max iters = {smooth_budget}")

        seq_l2, val_l2 = _adam_l2_opt(cand, n, deadline, max_iters=smooth_budget)

        print(f"   → surrogate after ℓ₂‑phase = {val_l2:.6f}")

        # --- max‑phase -------------------------------------------------
        time_left = deadline - time.time()
        if time_left < 8.0:
            seq = seq_l2
            val = val_l2
        else:
            max_budget = min(120_000, int(time_left * 0.4 / 0.001))
            max_budget = max(max_budget, 5_000)
            print(f"   max‑phase, max iters = {max_budget}")
            seq, val = _adam_opt(seq_l2, n, deadline, max_iters=max_budget)
            print(f"   → surrogate after max‑phase = {val:.6f}")

        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   ** new global best = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with exponentiated‑gradient
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        best_overall_seq = _exp_decay(n, 0.001)
        best_overall_seq = best_overall_seq / best_overall_seq.sum()

    if time.time() < deadline:
        print("\n--- Polishing with exponentiated‑gradient ---")
        polished_seq, polished_val = _exp_gradient_opt(
            best_overall_seq, n, deadline,
            max_iters=3_500_000, eta0=12.0)
        if polished_val < best_overall_val:
            best_overall_seq = polished_seq
            best_overall_val = polished_val
            print(f"[Polish] improved surrogate = {best_overall_val:.6f}")
        else:
            print(f"[Polish] no improvement (best unchanged)")

    # ------------------------------------------------------------------
    # Small random perturbations to escape possible local minima
    # ------------------------------------------------------------------
    attempts = 3
    for att in range(attempts):
        time_left = deadline - time.time()
        if time_left < 5.0:
            break
        eps = 0.02 * (0.5 ** att)          # decreasing perturbation size
        noise = rng.normal(scale=eps, size=n)
        pert = np.maximum(best_overall_seq + noise * best_overall_seq, 0.0)
        if pert.sum() == 0.0:
            continue
        pert = pert / pert.sum()
        max_it = min(40_000, int(time_left * 0.2 / 0.001))
        seq_tmp, val_tmp = _adam_opt(pert, n, deadline, max_iters=max_it)
        if val_tmp < best_overall_val:
            best_overall_seq = seq_tmp
            best_overall_val = val_tmp
            print(f"[Perturb {att}] improved surrogate = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # Evaluate the true objective via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_overall_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible continuation runs
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
```
