def _hill_climb(start_seq, n, deadline, rng, max_iters=600_000):
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start)*(it/max_iters)
        move_type = rng.random()
        if move_type < 0.4:
            # swap mass values
            i, j = rng.integers(0, n, size=2)
            if i != j:
                ns = cur_seq.copy()
                ns[i], ns[j] = ns[j], ns[i]
            else:
                continue
        elif move_type < 0.8:
            # Gaussian perturb
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s
        else:
            # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            amount = cur_seq[i] * (0.1 * rng.random())  # up to 10% of source
            if amount <= 0.0:
                continue
            ns = cur_seq.copy()
            ns[i] -= amount
            ns[j] += amount
            ns = np.maximum(ns, 0.0)
            ns = ns / ns.sum()
        val = _objective_surrogate(ns)
        if val < cur_val - 1e-12:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val

"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

under the constraints a_i ≥ 0.
The objective is scale‑invariant, therefore we work on the probability
simplex (∑a_i = 1).  The algorithm proceeds in several stages:

1.  Build a diverse pool of *structured* initial vectors (exponential,
    power‑law, plateau+exponential, uniform and the previously best
    sequence if it exists).

2.  For each start run a short Adam optimiser on a smooth log‑sum‑exp
    surrogate of the max‑convolution.  The surrogate is
        f_β(a) = 2·n·(M + log Σ exp(β·(c−M))/β)
    where c = a∗a and M = max(c).  β is increased geometrically from
    0.5 to 1e6, gradually turning the surrogate into the true max.

3.  Polish the Adam result with an *exact* sub‑gradient mirror‑descent
    (exponentiated‑gradient) that works directly on the true max‑convolution.

4.  Perform a simulated‑annealing / hill‑climbing phase with three
    move types (swap, small Gaussian perturbation, and a directed mass‑transfer)
    and a linear temperature schedule.

5.  If time permits, run a few cheap random‑restart Adam refinements
    around the current best solution.

All heavy work uses an FFT‑based full convolution (O(n log n)).
The routine respects the 1000 s budget; the final best vector is stored
in the global ``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(seq,seq)) for a sequence that sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), then normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _power_law(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-α}, then normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-alpha)
    return a / a.sum()


def _plateau_exponential(n: int, plateau_frac: float, rate: float) -> np.ndarray:
    """Plateau of constant height for the first `plateau_frac`·n entries,
    then an exponential tail.  Returned vector is normalised to sum = 1."""
    k = max(1, int(plateau_frac * n))
    a = np.empty(n, dtype=np.float64)
    a[:k] = 1.0 / k
    tail_len = n - k
    if tail_len > 0:
        tail = np.exp(-rate * np.arange(tail_len, dtype=np.float64))
        tail = tail / tail.sum() * (1.0 - a[:k].sum())
        a[k:] = tail
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – needed for Adam optimisation
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution objective.
    The variables are the unconstrained log‑softmax parameters `v_local`.
    """
    a = _softmax(v_local)                     # probability vector (sum = 1)
    conv = _conv_fft(a, a)                     # length = 2n‑1
    max_c = np.max(conv)
    shift = max_c                               # stability shift

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # Gradient w.r.t. a (see derivation in the original code)
    w = exp_beta / sum_exp                     # soft‑max weights over convolution bins
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]      # slice back to length n

    # Chain rule to obtain gradient w.r.t. v (log‑softmax variables)
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 150_000,
              beta_start: float = 0.5,
              beta_end: float = 1_000_000.0,
              alpha: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns the best probability vector seen and its (exact) surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # initialise log‑softmax variables
    m = np.zeros_like(v)                  # first moment
    v2 = np.zeros_like(v)                 # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam update (log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (exponentiated‑gradient) on the true objective
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution objective.
    Operates in log‑softmax space to keep the iterate on the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active convolution bin
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb with three move types
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly applies:
        * swap of two entries,
        * tiny Gaussian perturbation of the whole vector,
        * directed mass‑transfer from one index to another.
    Worsening moves are accepted with a linearly decreasing temperature.
    Returns the best vector found.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T_start + (T_end - T_start) * (it / max_iters)

        move = rng.random()
        if move < 0.40:                         # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]

        elif move < 0.80:                       # small Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        else:                                   # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_seq[i] == 0.0:
                continue
            # transfer up to 10 % of the source mass
            amount = cur_seq[i] * (0.1 * rng.random())
            ns = cur_seq.copy()
            ns[i] -= amount
            ns[j] += amount
            ns = np.maximum(ns, 0.0)
            ns = ns / ns.sum()

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # Metropolis acceptance
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # problem dimension used on the leaderboard
    n = 4000

    # --------------------------------------------------------------
    # 1. Assemble a pool of diverse initial vectors
    # --------------------------------------------------------------
    init_pool = []

    # exponential decays with a range of rates
    for rate in np.linspace(5e-4, 8e-3, 6):
        init_pool.append(_exp_decay(n, rate))

    # power‑law shapes
    for alpha in np.linspace(0.0, 2.0, 5):
        init_pool.append(_power_law(n, alpha))

    # plateau + exponential tails
    for frac in (0.05, 0.10, 0.20):
        init_pool.append(_plateau_exponential(n, plateau_frac=frac, rate=0.0015))

    # uniform (all equal)
    init_pool.append(np.full(n, 1.0 / n))

    # reuse the previously best sequence if it exists
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                init_pool.append(np.maximum(arr, 0.0) / np.maximum(arr.sum(), 1e-30))
        except Exception:
            pass

    # --------------------------------------------------------------
    # 2. Run the optimisation pipeline on each starter (as long as time permits)
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    for idx, init_seq in enumerate(init_pool):
        if time.time() > deadline:
            break

        print(f"[Stage 1 – start #{idx + 1}/{len(init_pool)}]")
        # guarantee a proper probability vector
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.sum() == 0.0:
            init_seq = np.full(n, 1.0 / n)
        else:
            init_seq = init_seq / init_seq.sum()

        # ---- Adam on smoothed surrogate ---------------------------------
        seq_a, val_a = _adam_opt(
            start_seq=init_seq,
            n=n,
            deadline=deadline,
            max_iters=100_000,          # a fairly long run but still cheap
            beta_start=0.5,
            beta_end=1_000_000.0,
            alpha=0.5,
        )
        print(f"   Adam  surrogate = {val_a:.6f}")

        # ---- Exact sub‑gradient polishing --------------------------------
        if time.time() < deadline:
            seq_g, val_g = _exp_gradient_opt(
                start_seq=seq_a,
                n=n,
                deadline=deadline,
                max_iters=1_000_000,
                eta0=5.0,
            )
            print(f"   EG    surrogate = {val_g:.6f}")
            if val_g < val_a:
                seq_a, val_a = seq_g, val_g

        # ---- Hill‑climb ----------------------------------------------------
        if time.time() < deadline:
            seq_h, val_h = _hill_climb(
                start_seq=seq_a,
                n=n,
                deadline=deadline,
                rng=rng,
                max_iters=250_000,
            )
            print(f"   HC    surrogate = {val_h:.6f}")
            if val_h < val_a:
                seq_a, val_a = seq_h, val_h

        # keep the best over all starts
        if val_a < best_val:
            best_val = val_a
            best_seq = seq_a.copy()
            print(f"   *** New global best surrogate = {best_val:.6f} ***")

    # --------------------------------------------------------------
    # 3. Short random‑restart Adam refinements (budget permitting)
    # --------------------------------------------------------------
    while time.time() + 30.0 < deadline:
        if best_seq is None:
            break
        noisy = best_seq + 0.002 * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy = noisy / s

        sub_deadline = time.time() + 20.0
        seq_r, val_r = _adam_opt(
            start_seq=noisy,
            n=n,
            deadline=sub_deadline,
            max_iters=60_000,
            beta_start=0.5,
            beta_end=200_000.0,
            alpha=0.5,
        )
        print(f"   Restart Adam surrogate = {val_r:.6f}")
        if val_r < best_val:
            best_val = val_r
            best_seq = seq_r.copy()
            print(f"   *** Improvement after restart! new best = {best_val:.6f} ***")

    # --------------------------------------------------------------
    # 4. Final evaluation with the supplied black‑box function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print("\n=== Finished ===")
        print(f"Best surrogate ≈ {best_val:.6f}")
        print(f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
