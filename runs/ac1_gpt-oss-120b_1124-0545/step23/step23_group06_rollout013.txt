The algorithm is a hybrid of three ideas:

1. **LP‑based improvement** – for a moderate size vector (n≈2000) we
   repeatedly solve the linear programme  
        maximise   Σ g_j   subject to   f * g ≤ M ,   g ≥ 0,
   where ``M`` is the current maximum of the autocorrelation ``f*f``.
   The solution (after normalising to Σ g = 1) is taken as a new
   candidate; the bound ``M`` is strictly reduced after a few iterations.
   This corresponds to the “active‑set” iteration described in the
   statement and works very well for bringing the surrogate value close
   to the theoretical optimum.

2. **Upscaling + high‑resolution polishing** – the best low‑resolution
   vector is interpolated to the target size (n=4000) and further refined
   with a continuation L‑BFGS on a smooth log‑sum‑exp surrogate
   (gradually sharpening the β‑parameter) followed by a long
   exponentiated‑gradient (EG) polish and a short permutation‑annealing
   tweak.

3. **Multiple analytic seeds** – a small pool of analytic shapes
   (uniform, exponential, power‑law, beta‑shaped, Dirichlet draws and the
   previous solution) is used to initialise the LP phase; the best
   result is retained.

All functions are top‑level, there are no closures, and the total run‑time
is bounded by the supplied ``budget_s`` (default 1000 s).  The final result
is stored in the global variable ``height_sequence_1`` and also returned
as a plain ``list[float]``.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, power‑of‑2 padding)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq))  (seq is already normalised)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all normalise to sum‑1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# L‑BFGS surrogate optimiser (log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
class _LbfgsObjective:
    """Callable needed by scipy.optimize.minimize."""

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, log_a: np.ndarray):
        # soft‑max to a probability vector
        max_log = np.max(log_a)
        exp_shift = np.exp(log_a - max_log)
        a = exp_shift / exp_shift.sum()

        # convolution a∗a
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        beta = self.beta

        # smooth max via log‑sum‑exp
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        w = exp_beta / sum_exp

        # gradient w.r.t. a (2 * convolution of reversed a with w)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3·n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # gradient w.r.t. log‑softmax parameters
        grad_log = a * (grad_a - np.dot(grad_a, a))

        # smoothed objective value: M_β = shift + log(sum_exp)/β
        M_beta = shift + np.log(sum_exp) / beta
        obj = 2.0 * self.n * M_beta
        return obj, grad_log


def _lbfgs_opt(start_seq: np.ndarray,
               beta_schedule,
               max_iter_per_beta: int,
               deadline: float):
    """
    Continuation L‑BFGS: run L‑BFGS on a sequence of increasingly sharp
    log‑sum‑exp surrogates.
    """
    log_a = np.log(start_seq + 1e-30)          # stay in log‑softmax space
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break

        obj = _LbfgsObjective(len(start_seq), beta)
        opts = {"maxiter": max_iter_per_beta, "disp": False}
        try:
            res = scipy.optimize.minimize(fun=obj,
                                          x0=log_a,
                                          method="L-BFGS-B",
                                          jac=True,
                                          options=opts)
        except Exception:                      # any interruption – stop early
            break

        log_a = res.x
        exp_shift = np.exp(log_a - np.max(log_a))
        a = exp_shift / exp_shift.sum()
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser (exponentiated gradient)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)
    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        log_a = np.clip(log_a, -30.0, 30.0)

        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap permutation annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 60_000):
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# LP‑based improvement (maximise Σg under f*g≤M)
# ----------------------------------------------------------------------
def _lp_indices(n: int):
    """Return row/col index arrays for the convolution constraint matrix."""
    rows = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    cols = np.repeat(np.arange(n), n)
    return rows, cols


def _solve_lp_max_sum(f: np.ndarray, M: float,
                      row_idx: np.ndarray, col_idx: np.ndarray):
    """Solve  max Σ g  s.t.  f * g ≤ M ,  g ≥ 0  (LP in standard form)."""
    n = len(f)
    data = np.tile(f, n)                     # length n²
    A_ub = scipy.sparse.csr_matrix((data, (row_idx, col_idx)),
                                    shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    c = -np.ones(n, dtype=np.float64)        # maximise Σg  ⇔ minimise −Σg
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def _lp_optimize(start_seq: np.ndarray,
                 max_iters: int,
                 deadline: float,
                 row_idx: np.ndarray,
                 col_idx: np.ndarray):
    """
    Repeated LP improvement: start from ``start_seq`` (sum=1), compute M,
    solve the LP, renormalise the solution, and repeat while the surrogate
    improves.
    """
    f = start_seq.copy()
    best_f = f.copy()
    best_val = _objective_surrogate(f)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv_f = _conv_fft(f, f)
        M = conv_f.max()
        g_raw = _solve_lp_max_sum(f, M, row_idx, col_idx)
        if g_raw is None:
            break
        if g_raw.sum() == 0:
            break

        g = g_raw / g_raw.sum()
        conv_g = _conv_fft(g, g)
        M_g = conv_g.max()
        val = 2.0 * len(g) * M_g

        if val < best_val - 1e-12:
            best_val = val
            best_f = g.copy()

        # stop if the bound does not decrease
        if M_g >= M - 1e-12:
            break

        f = g

    return best_f, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The pipeline:

    1. generate a modest set of analytic / random low‑resolution seeds;
    2. improve each with a few LP iterations (maximising mass under the
       current autocorrelation bound);
    3. pick the best, upscale to the target size (n=4000);
    4. polish with continuation L‑BFGS, a long EG run and a short anneal;
    5. return the best sequence.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem sizes
    # ------------------------------------------------------------------
    n_small = 2000
    n_large = 4000

    # ------------------------------------------------------------------
    # Pre‑compute LP index arrays (only depends on n_small)
    # ------------------------------------------------------------------
    row_idx, col_idx = _lp_indices(n_small)

    # ------------------------------------------------------------------
    # Build a pool of low‑resolution analytic / random seeds
    # ------------------------------------------------------------------
    seeds_small = [
        _uniform(n_small),
        _exp_decay(n_small, 0.001),
        _exp_decay(n_small, 0.003),
        _exp_decay(n_small, 0.006),
        _power(n_small, 0.5),
        _power(n_small, 0.8),
        _power(n_small, 1.0),
    ]

    # a few Dirichlet draws with diverse concentrations
    for conc in (0.02, 0.1, 0.5, 1.0, 3.0):
        for _ in range(2):
            seeds_small.append(_random_dirichlet(n_small, conc, rng))

    # Beta‑shaped families
    for _ in range(6):
        a = rng.uniform(0.2, 2.0)
        b = rng.uniform(0.2, 2.0)
        seeds_small.append(_beta_shape(n_small, a, b))

    # Previous solution (if any) down‑sampled
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 1:
                x_old = np.linspace(0.0, 1.0, arr.size)
                x_new = np.linspace(0.0, 1.0, n_small)
                down = np.interp(x_new, x_old, arr)
                down = np.maximum(down, 0.0)
                if down.sum() > 0:
                    down = down / down.sum()
                else:
                    down = _uniform(n_small)
                seeds_small.append(down)
        except Exception:
            pass

    print(f"[Info] Generated {len(seeds_small)} low‑resolution seeds (n={n_small})")

    # ------------------------------------------------------------------
    # LP improvement on each seed
    # ------------------------------------------------------------------
    best_seq_small = None
    best_val_small = float('inf')

    for idx, seed_seq in enumerate(seeds_small):
        if time.time() >= deadline:
            break

        print(f"\n[Seed {idx+1}/{len(seeds_small)}] LP refinement (max 8 iters)")
        seq_lp, val_lp = _lp_optimize(seed_seq,
                                      max_iters=8,
                                      deadline=deadline,
                                      row_idx=row_idx,
                                      col_idx=col_idx)

        print(f"  → surrogate = {val_lp:.6f}")
        if val_lp < best_val_small:
            best_val_small = val_lp
            best_seq_small = seq_lp.copy()
            print("        *** new best (small)")

    if best_seq_small is None:
        # fallback – very unlikely
        best_seq_small = _uniform(n_small)

    # ------------------------------------------------------------------
    # Upscale to final size
    # ------------------------------------------------------------------
    print("\n[Upscaling] Interpolating to final dimension")
    x_small = np.linspace(0.0, 1.0, n_small)
    x_large = np.linspace(0.0, 1.0, n_large)
    seq_large = np.interp(x_large, x_small, best_seq_small)
    seq_large = np.maximum(seq_large, 0.0)
    seq_large = seq_large / seq_large.sum()

    # ------------------------------------------------------------------
    # High‑resolution polishing: L‑BFGS → EG → permutation anneal
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Refinement] L‑BFGS continuation")
        beta_schedule = [10, 80, 400, 2000, 10000, 50000, 200000]
        max_iter_per_beta = 500
        seq_refined, val_refined = _lbfgs_opt(seq_large,
                                              beta_schedule,
                                              max_iter_per_beta,
                                              deadline)
        print(f"  L‑BFGS surrogate = {val_refined:.6f}")
    else:
        seq_refined = seq_large
        val_refined = _objective_surrogate(seq_refined)

    if time.time() < deadline:
        print("\n[Refinement] Exponentiated‑gradient polish")
        seq_eg, val_eg = _exp_gradient_opt(seq_refined,
                                           deadline,
                                           max_iters=2_500_000,
                                           eta0=6.0)
        if val_eg < val_refined:
            seq_refined, val_refined = seq_eg, val_eg
        print(f"  EG surrogate = {val_refined:.6f}")

    if time.time() < deadline:
        print("\n[Refinement] Short permutation annealing")
        seq_final, val_final = _perm_anneal(seq_refined,
                                           deadline,
                                           max_iters=60_000)
        if val_final < val_refined:
            seq_refined, val_refined = seq_final, val_final
        print(f"  final surrogate = {val_refined:.6f}")

    # ------------------------------------------------------------------
    # Evaluate with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(seq_refined.tolist())
        print("\n=== Finished ===")
        print(f"Surrogate ≈ {val_refined:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                    # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for future runs
    globals()["height_sequence_1"] = seq_refined.tolist()
    return seq_refined.tolist()
