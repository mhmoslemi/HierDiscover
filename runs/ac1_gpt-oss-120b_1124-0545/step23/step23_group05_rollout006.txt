"""
Advanced optimizer for the step‑function “flat‑convolution’’ problem.

The objective of ``evaluate_sequence`` is

    2·n·max(conv(a,a)) / (sum(a)²)      (sum(a) > 0)

and is invariant to scaling, so we normalise every candidate to sum = 1.
The optimiser works in three phases (all respecting a global time budget):

1. **Rich random‑parameter seed search** – families such as beta‑shapes,
   power laws, mixed exponentials, Gaussians, etc. are sampled
   extensively; the best surrogate (= 2·n·max(conv)) is kept.

2. **Adam‑optimisation of a smooth max‑convolution** – we minimise
   a log‑sum‑exp smooth approximation of the max using a soft‑max
   (log‑softmax) parametrisation.  The smoothing parameter β is grown
   gradually (β‑schedule).  Adam‑style momentum and a diminishing step‑size
   provide a stable descent on the high‑dimensional simplex.

3. **Local stochastic polishing** – tiny mass transfers between random
   coordinates are accepted only when they improve the true surrogate.
   The step‑size cools linearly, acting as a final fine‑grained hill‑climb.

The whole pipeline finishes within the supplied time budget (default
≈ 1000 s) and stores the best sequence found in the global variable
``height_sequence_1``.  The returned list can be fed directly to the
provided ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric primitives
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq))  (seq is already ℓ₁‑normalised)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mixed_exponential(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    e1 = np.exp(-r1 * np.arange(n, dtype=np.float64))
    e2 = np.exp(-r2 * np.arange(n, dtype=np.float64))
    a = (1.0 - alpha) * e1 + alpha * e2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Randomised seed search
# ----------------------------------------------------------------------
def _random_seed_search(n: int, deadline: float, rng: np.random.Generator):
    """Explore a large random family space until the deadline."""
    best_seq = None
    best_val = np.inf
    families = ["beta", "power", "mixedexp", "gauss", "invsqrt"]
    while time.time() < deadline:
        fam = rng.choice(families)
        if fam == "beta":
            alpha = 10 ** rng.uniform(-3.0, 1.0)   # 0.001 … 10
            beta = 10 ** rng.uniform(-3.0, 1.0)
            seq = _beta_shape(n, alpha, beta)
        elif fam == "power":
            p = rng.uniform(0.2, 2.5)
            seq = _power(n, p)
        elif fam == "mixedexp":
            r1 = 10 ** rng.uniform(-5.0, -0.2)
            r2 = 10 ** rng.uniform(-5.0, -0.2)
            alpha = rng.random()
            seq = _mixed_exponential(n, r1, r2, alpha)
        elif fam == "gauss":
            sigma_frac = rng.uniform(0.02, 0.4)
            seq = _gaussian(n, sigma_frac)
        else:               # invsqrt with a tweak
            p = rng.uniform(0.3, 1.2)
            i = np.arange(1, n + 1, dtype=np.float64)
            a = 1.0 / (i ** p)
            seq = a / a.sum()
        val = _surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq
    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimisation of a smooth max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray, deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Minimise a log‑sum‑exp smooth approximation of max(conv)."""
    n = len(start_seq)
    # initialise log‑softmax parameters
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)           # first‑moment accumulator
    b1 = 0.9                       # momentum constant

    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq)

    # β‑schedule – from very smooth to almost exact max
    betas = [
        0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0,
        200.0, 500.0, 1000.0, 2000.0, 5000.0,
        1e4, 2e4, 5e4, 1e5
    ]

    total_iters = 0
    for beta in betas:
        # give each β a time slice proportional to the remaining budget
        if time.time() >= deadline:
            break
        time_per_beta = (deadline - time.time()) / max(1, len(betas) - betas.index(beta))
        start_beta = time.time()
        # a modest upper bound on inner iterations (so we never get stuck)
        max_inner = int(min(3000, time_per_beta / 0.0015)) + 1
        for _ in range(max_inner):
            if time.time() >= deadline:
                break
            a = _softmax(v)                     # ℓ₁‑normalised candidate
            conv = _conv_fft(a, a)
            maxc = conv.max()
            # smooth‑max via log‑sum‑exp
            shift = maxc                         # ensures numerical stability
            expbeta = np.exp(beta * (conv - shift))
            w = expbeta / expbeta.sum()
            # gradient w.r.t. a (see derivation in the comment block above)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]
            # gradient w.r.t. log‑softmax parameters
            grad_v = a * (grad_a - np.dot(grad_a, a))

            total_iters += 1
            lr = 0.2 / math.sqrt(total_iters)   # simple diminishing step‑size
            m = b1 * m + (1.0 - b1) * grad_v
            v = v - lr * m

            cur_val = 2.0 * n * maxc            # true surrogate (no smoothing)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        # end inner while
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple stochastic local search (hill climbing)
# ----------------------------------------------------------------------
def _local_search(start_seq: np.ndarray, deadline: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Tiny mass transfers; accept only improvements (with a tiny chance to escape)."""
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _surrogate(a)
    start_time = time.time()
    # linear cooling of the transfer magnitude
    init_step = 0.02  # 2 % of total mass
    while time.time() < deadline:
        elapsed = time.time() - start_time
        remaining = max(1e-9, deadline - time.time())
        frac = remaining / (deadline - start_time + 1e-12)
        step = init_step * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = step * rng.random()
        if a[i] < eps:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # renormalise to guard against rounding drift
        a_new /= a_new.sum()
        val = _surrogate(a_new)
        if val + 1e-12 < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new
        else:
            # occasional uphill move (tiny probability) – helps escape flat plateaus
            if rng.random() < 1e-4:
                a = a_new
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects a wall‑clock limit (default 1000 s) and stores the
    best found sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # a small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 works well for the benchmark
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previously‑found construction if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_seq = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_seq = arr / arr.sum()
        except Exception:
            prev_seq = None

    # ------------------------------------------------------------------
    # 1. Assemble a modest set of deterministic seeds
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _triangular(n),
        _inverse_sqrt(n),
        _power(n, 0.5),
        _power(n, 0.7),
        _power(n, 1.0),
        _power(n, 1.5),
        _exp_decay(n, 1e-3),
        _gaussian(n, 0.15),
        _beta_shape(n, 0.2, 0.2),
        _beta_shape(n, 0.5, 0.5),
    ]
    if prev_seq is not None:
        seeds.append(prev_seq)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of deterministic seeds
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    for s in seeds:
        if time.time() > deadline:
            break
        v = s / s.sum()
        val = _surrogate(v)
        if val < best_val:
            best_val = val
            best_seq = v.copy()
    print(f"[Init] best deterministic surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Randomised parametric search (≈15 % of remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        search_deadline = time.time() + 0.15 * remaining
        seq_rand, val_rand = _random_seed_search(n, search_deadline, rng)
        if val_rand < best_val:
            best_val = val_rand
            best_seq = seq_rand.copy()
            print(f"[Random] new best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam optimisation of smooth max‑convolution (≈70 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        opt_deadline = time.time() + 0.70 * remaining
        seq_opt, val_opt = _adam_optimize(best_seq, opt_deadline, rng)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt.copy()
            print(f"[Adam] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final local stochastic polishing (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_fin, val_fin = _local_search(best_seq, deadline, rng)
        if val_fin < best_val:
            best_val = val_fin
            best_seq = seq_fin.copy()
            print(f"[Local] final surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:          # pragma: no cover
        print("Warning – evaluate_sequence raised:", exc)

    print("\n=== Finished ===")
    print(f"Best surrogate ≈ {best_val:.6f}")
    print(f"True evaluate_sequence = {true_val:.6f}" if true_val is not None else "True value unavailable")

    # store globally for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
