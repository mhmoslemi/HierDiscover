"""
Non‑negative step‑function optimiser (≈ 1000 s).

The search proceeds in three stages:

1. **Diversified seeding** – uniform, triangular, Dirichlet and symmetric
   Beta‑distributions (including the previous best solution).

2. **Continuous optimisation** – a two‑phase optimiser on the probability
   simplex:
   * Adam on a smooth‑max surrogate (soft‑max of the convolution values) with
     an aggressively increasing temperature schedule.
   * Exact exponentiated‑gradient (mirror‑descent) on the true max‑convolution.

3. **LP‑based polishing** – solve a linear programme
   `max 1ᵀg   s.t.   Conv(f,g) ≤ M , g ≥ 0`  where `M = max Conv(f,f)`.
   The LP yields a vector `g` with a *larger* L₁‑norm while respecting the
   same convolution bound.  After normalising `g` to unit mass we mix it with
   the current iterate using a back‑tracking line‑search that guarantees the
   max‑convolution does not increase.  This step typically reduces the
   objective further than pure gradient methods.

Finally a cheap greedy local walk (mass transfers between two coordinates)
is performed and the best sequence is returned.  The global variable
``height_sequence_1`` stores the result for possible warm‑starts on the next
invocation.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, soft‑max, surrogate)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution by real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq))
    ``seq`` is assumed to be a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _max_conv(seq: np.ndarray) -> float:
    """Maximum entry of the self‑convolution."""
    return float(np.max(_conv_fft(seq, seq)))


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    tot = a.sum()
    if tot == 0.0:
        return _uniform(n)
    a /= tot
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on a log‑softmax parametrisation of a smooth‑max."""
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # Temperature schedule – start gentle, end extremely sharp.
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0, 20000.0,
        50000.0, 100000.0, 200000.0,
        500000.0, 1_000_000.0, 2_000_000.0,
        5_000_000.0, 10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # inner budget grows with the remaining time
        max_iter_beta = 6000 if (deadline - time.time()) > 40 else 1500

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                 # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights (log‑sum‑exp)
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)         # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            # mirror‑gradient on the log‑softmax space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact check
            if it % 700 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final evaluation
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated gradient) on  max(conv(a,a))."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 100_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """Greedy stochastic walk on the simplex – only strictly improving moves."""
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming polishing (max‑convolution bound preserving)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Construct the matrix A (shape = (2n‑1, n)) such that
    (A @ g)[k] = (conv(f, g))[k]   for all k.
    """
    n = len(f)
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    idx = 0
    for i in range(n):
        rows[idx: idx + n] = np.arange(i, i + n, dtype=np.int32)
        cols[idx: idx + n] = i
        data[idx: idx + n] = f
        idx += n

    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_polish(seq: np.ndarray,
               current_M: float,
               deadline: float,
               mix_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Solve   max 1ᵀg   s.t.   Conv(seq, g) ≤ current_M , g ≥ 0.
    The LP solution g may have L₁‑norm > 1.
    After normalising g to unit mass we mix it back into `seq`
    using a back‑tracking line‑search that guarantees the max‑convolution
    does not increase.
    Returns the updated sequence and its new max‑convolution.
    """
    n = len(seq)
    if time.time() > deadline:
        return seq, current_M

    # Build the (sparse) convolution matrix once.
    A = _build_conv_matrix(seq)

    # Linear programme: minimise   -1ᵀg   (i.e. maximise 1ᵀg)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, current_M, dtype=np.float64)

    # Use HiGHS (fast for large sparse LPs)
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})

    if not res.success:
        # LP failed – fall back to the original sequence.
        return seq, current_M

    g = res.x
    sum_g = g.sum()
    if sum_g <= 1.0 + 1e-9:
        # No improvement possible.
        return seq, current_M

    # Normalise to unit mass (the same normalisation used for all candidates)
    g_norm = g / sum_g

    # Mix g_norm into seq – back‑track to keep the max‑convolution ≤ current_M.
    t = max(min(mix_factor, 1.0), 0.0)          # start with a moderate step size
    while t > 1e-5:
        cand = (1.0 - t) * seq + t * g_norm
        cand_M = _max_conv(cand)
        if cand_M <= current_M * (1.0 + 1e-12):
            seq = cand
            current_M = cand_M
            break
        t *= 0.5

    return seq, current_M

def propose_candidate(seed: int = 123,
