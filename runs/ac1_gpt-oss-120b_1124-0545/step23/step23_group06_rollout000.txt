"""
Propose a candidate step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.  The strategy builds a large pool of random and analytic
seeds for a moderate length (8000), keeps the best few by the fast FFT‑based
surrogate, then performs a continuation L‑BFGS optimisation on a
log‑sum‑exp smooth approximation of the max‑convolution.  After that a
long exponentiated‑gradient (EG) polish is applied, followed by a tiny
permutation‑annealing tweak.  The previous best solution (global variable
``height_sequence_1``) is up‑sampled and used as a seed, which helps the
search escape the 4000‑point local optimum found earlier.

All heavy work uses FFT‑convolution for speed and respects the global
deadline so the routine always returns a valid sequence before the 1000 s
budget expires.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors using FFT (real→real)."""
    lx = len(x)
    ly = len(y)
    L = lx + ly - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) where sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return a normalised probability vector
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    return a / a.sum()


def _upsample(prev: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised vector to a new length."""
    old_n = len(prev)
    if old_n == new_n:
        return prev.copy()
    old_pos = np.linspace(0, old_n - 1, old_n)
    new_pos = np.linspace(0, old_n - 1, new_n)
    up = np.interp(new_pos, old_pos, prev)
    up = np.maximum(up, 0.0)
    up /= up.sum()
    return up


# ----------------------------------------------------------------------
# Continuation L‑BFGS on a log‑sum‑exp surrogate of the max
# ----------------------------------------------------------------------


class _LbfgsObjective:
    """Callable for scipy.optimize.minimize (value & gradient)."""

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, log_a: np.ndarray):
        # soft‑max to a probability vector
        max_log = np.max(log_a)
        exp_shift = np.exp(log_a - max_log)
        a = exp_shift / exp_shift.sum()

        # convolution a∗a
        conv = _conv_fft(a, a)
        shift = np.max(conv)                     # true max value

        # smooth max via log‑sum‑exp
        beta = self.beta
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        w = exp_beta / sum_exp                    # weights that sum to 1

        # gradient w.r.t a (2 * convolution of reversed a with w)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length = 3*n-2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # gradient w.r.t. log‑softmax parameters (mirrored KL‑gradient)
        grad_log = a * (grad_a - np.dot(grad_a, a))

        # smoothed objective value
        M_beta = shift + np.log(sum_exp) / beta
        obj = 2.0 * self.n * M_beta
        return float(obj), grad_log


def _lbfgs_opt(start_seq: np.ndarray,
               beta_schedule,
               max_iter_per_beta: int,
               deadline: float):
    """
    Continuation L‑BFGS on a sequence of increasing beta values.
    Returns the best sequence (as a numpy array) and its surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)
    best_seq = start_seq.copy()
    best_val = _surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        obj = _LbfgsObjective(len(start_seq), beta)
        opts = {"maxiter": max_iter_per_beta, "disp": False}
        try:
            res = minimize(fun=obj,
                           x0=log_a,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception:  # any error → abort further optimization
            break

        log_a = res.x
        # back‑project to simplex
        exp_shift = np.exp(log_a - np.max(log_a))
        a = exp_shift / exp_shift.sum()
        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (sub‑gradient) optimiser – works directly on the
# non‑smoothed max objective.
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0):
    """
    Sub‑gradient descent with KL‑mirror (exponentiated gradient).
    Optimises the true objective 2 n·max(conv(a,a)).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a)
    log_a = np.log(a + 1e-30)
    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see earlier code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        log_a = np.clip(log_a, -30.0, 30.0)  # keep numbers stable

        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap permutation‑annealing tweak (swap only)
# ----------------------------------------------------------------------


def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 60_000):
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]
        val = _surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """

    The pipeline:

    1. Build a fairly large pool of analytic / random seeds (length ≈ 8000).
    2. Keep the top few according to the fast FFT surrogate.
    3. Refine each with a continuation L‑BFGS on a log‑sum‑exp surrogate.
    4. Polish the best candidate with a long exponentiated‑gradient run.
    5. Finish with a short permutation‑annealing tweak.
    6. Return the best sequence (stored globally as ``height_sequence_1``).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin
    global_deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – 8000 points give more degrees of freedom.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Load previous solution (if any) and up‑sample it.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.ndim == 1:
                prev_vec = _upsample(arr, n)
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of seed vectors.
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _exp_decay(n, 0.0012),
        _exp_decay(n, 0.0030),
        _sym_exp_decay(n, 0.0022),
        _power(n, 0.7),
        _power(n, 1.2),
        _beta_shape(n, 0.5, 0.5),
        _beta_shape(n, 1.5, 0.8),

    # Dirichlet draws with a broad range of concentration parameters.
    for conc in (0.01, 0.03, 0.07, 0.2, 0.6, 1.5, 4.0):

    top_k = min(6, len(evaluated))

    # ------------------------------------------------------------------

        sub_deadline = time.time() + share

        # ----- continuation L‑BFGS -----
        seq_lbfgs, val_lbfgs = _lbfgs_opt(seed_vec,
