        max_iters = 200_000  # heuristic
        for t in range(max_iters):
            if time.time() > deadline: break
            a = _softmax(log_a)
            conv = _conv_fft(a, a)
            # compute objective O = (sum conv^p)^(1/p)
            c_pow = np.power(conv, p)
            S = np.sum(c_pow)

            w = np.power(conv, p - 1)  # conv^{p-1}
            # compute gradient term:
            # O^{1-p}
            scale = 2.0 * (O ** (1.0 - p))  # note O^{1-p}

            # convolve a with w_rev (full)

            idx = (len(w) - 1) - np.arange(n)
            grad_a = scale * corr[idx]
            # compute gradient w.r.t log a:
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            # Adam updates

            cur_surrogate = _objective_surrogate(a)

"""
Step‑function optimizer (≈1000 s budget)
=======================================

The aim is to minimise

    2·n·max(conv(a,a)) / (sum(a)**2)

for a non‑negative vector `a`.  The value is invariant under scaling,
so we normalise every candidate to sum to 1 and minimise the surrogate

    S(a) = 2·n·max(conv(a,a))

which is exactly the value returned for a normalised vector.

Algorithm
---------

1. **Seed pool** – a few diverse initial vectors (previous best,
   exponential decays, a symmetric triangular shape, the uniform
   distribution and a handful of Dirichlet draws).

2. **Fast coarse optimisation** – each seed is refined with a
   short Adam run on the *hard* max‑convolution sub‑gradient
   (only one FFT per iteration).  The three best seeds are kept.

3. **Continuation with a smoothed objective** –  
   For the selected seeds we run Adam on the *p‑norm* of the
   convolution:

       Lp(a) = ( Σ_i conv_i^p )^{1/p}    ,   p = 2,4,8,16,32

   `Lp` smoothly approximates the max, providing a more stable
   gradient.  For each `p` we perform many iterations (two FFTs per
   step) and gradually decrease the learning rate (`lr ∝ 1/√p`).
   The best vector over all stages is retained.

4. **Final polishing** – an exponentiated‑gradient (mirror‑descent)
   pass on the true max‑convolution objective (as in the reference
   solution) for a few million low‑step‑size iterations.

5. **Local random perturbations** – a cheap hill‑climbing phase that
   moves a tiny amount of mass between random coordinates and keeps
   any improving move.

6. **Evaluation** – the best vector is fed to ``evaluate_sequence``,
   printed, stored globally as ``height_sequence_1`` and returned.

All steps respect the global time limit (default 1000 s).  The code
uses only NumPy and runs on a single CPU (the FFT implementation is
already multithreaded if the underlying BLAS/LAPACK supports it).

"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a vector summing to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _triangular(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Adam on the *hard* max‑convolution sub‑gradient
# ----------------------------------------------------------------------
def _adam_max_opt(start_seq: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 100_000,
                  lr0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent using the sub‑gradient of the active
    max‑convolution entry.
    Returns the best probability vector seen and its surrogate.
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))          # active index

        # gradient of the active entry w.r.t. a:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g_a = np.zeros_like(a)
        g_a[idx_i] = 2.0 * a[rev_idx]

        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # keep the best surrogate seen
        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam on the *smoothed* p‑norm of the convolution
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     p_schedule: list[int],
                     iters_per_stage: int = 80_000,
                     lr0: float = 0.3,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam with a continuation on the p‑norm of the convolution.
    Gradient of Lp(a) = ( Σ conv_i^p )^{1/p} is used.
    Returns the best probability vector seen and its surrogate.
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)
    total_iter = 0

    for p in p_schedule:
        # learning‑rate scaled with p (larger p → smaller steps)
        lr = lr0 / math.sqrt(p)

        for _ in range(iters_per_stage):
            if time.time() > deadline:
                break
            total_iter += 1

            a = _softmax(log_a)

            # ----- compute smoothed gradient -----
            conv = _conv_fft(a, a)                     # length 2n‑1
            # Lp‑norm
            pow_conv = np.power(conv, p)
            S = np.sum(pow_conv)
            O = S ** (1.0 / p)                         # Lp‑norm value

            # w = conv^{p‑1}
            w = np.power(conv, p - 1)

            # correlation = convolution(a, reversed w)
            w_rev = w[::-1]
            corr_full = _conv_fft(a, w_rev)            # length 3n‑2
            idx = (len(conv) - 1) - np.arange(n)       # map to gradient entries
            grad_a = 2.0 * (O ** (1.0 - p)) * corr_full[idx]

            # chain rule for log‑softmax variables
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_v
            v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** total_iter)
            v_hat = v / (1.0 - beta2 ** total_iter)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            log_a -= step

            # ----- keep best surrogate (max‑convolution) -----
            cur_conv = _conv_fft(_softmax(log_a), _softmax(log_a))
            cur_val = float(2.0 * n * np.max(cur_conv))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(log_a).copy()

        print(f"   [Smooth stage p={p}] best surrogate = {best_val:.6f}")

        if time.time() > deadline:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (hard max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 4_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient (mirror descent) on the true max‑convolution.
    Returns the best vector seen and its surrogate.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Small local random‑search hill climbing
# ----------------------------------------------------------------------
def _local_random_search(seq: np.ndarray,
                         n: int,
                         deadline: float,
                         steps: int = 2_000,
                         epsilon: float = 0.02) -> np.ndarray:
    """
    Randomly moves a small amount of mass between two coordinates.
    Accepts any improvement in the hard surrogate.
    """
    a = seq.copy()
    best_val = _objective_surrogate(a)
    rng = np.random.default_rng()

    for _ in range(steps):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] < 1e-12:
            continue
        delta = epsilon * a[i] * rng.random()
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur_val = _objective_surrogate(a_new)
        if cur_val < best_val:
            best_val = cur_val
            a = a_new
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (vector) that minimises the
    provided ``evaluate_sequence`` function.  The routine respects the
    time budget, stores the best vector globally as ``height_sequence_1``
    and returns it as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a few seconds spare for the final evaluation / printing
    deadline = start_time + budget_s - 5.0

    n = 4000

    # ------------------------------------------------------------------
    # Build the initial pool of seeds
    # ------------------------------------------------------------------
    candidates = []

    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                candidates.append(arr / arr.sum())
        except Exception:
            pass

    # exponentials with several rates
    for rate in (0.0003, 0.0006, 0.0012, 0.0025, 0.004):
        candidates.append(_exp_decay(n, rate))

    # triangular and uniform
    candidates.append(_triangular(n))
    candidates.append(_uniform(n))

    # Dirichlet draws – a handful with varied concentration
    for alpha in (0.3, 0.7, 1.5):
        cand = rng.dirichlet(alpha=np.full(n, alpha))
        candidates.append(cand)

    # ensure at least one candidate
    if not candidates:
        candidates.append(_exp_decay(n, rate=0.001))

    print(f"[Setup] generated {len(candidates)} initial candidates")

    # ------------------------------------------------------------------
    # Phase 1 – cheap Adam (max‑subgradient) on all candidates
    # ------------------------------------------------------------------
    cheap_iters = 15_000
    cheap_results = []

    for idx, cand in enumerate(candidates, start=1):
        if time.time() > deadline - 30.0:
            break
        print(f"[Phase‑1] candidate {idx}/{len(candidates)} – short Adam")
        seq, val = _adam_max_opt(cand, n, deadline,
                                 max_iters=cheap_iters,
                                 lr0=0.2)
        cheap_results.append((seq, val))

    # keep the three best seeds for the expensive stage
    cheap_results.sort(key=lambda x: x[1])
    top_seeds = [obj[0] for obj in cheap_results[:3]]
    print(f"[Phase‑1] keeping {len(top_seeds)} best seeds for refinement")

    # ------------------------------------------------------------------
    # Phase 2 – continuation on smoothed p‑norms
    # ------------------------------------------------------------------
    best_overall_seq = None
    best_overall_val = float("inf")
    p_schedule = [2, 4, 8, 16, 32]
    # allocate remaining time roughly evenly over the top seeds
    remaining_time = deadline - time.time()
    if remaining_time < 10.0:
        raise RuntimeError("Not enough time left after phase‑1")

    # each seed gets a share of the remaining budget (seconds)
    time_per_seed = remaining_time / max(1, len(top_seeds))
    for idx, seed_seq in enumerate(top_seeds, start=1):
        seed_deadline = time.time() + time_per_seed
        print(f"[Phase‑2] refining seed {idx}/{len(top_seeds)}")
        # number of iterations per p‑stage is dynamically limited by the time budget
        iters_per_stage = 80_000
        seq, val = _adam_smooth_opt(seed_seq, n, seed_deadline,
                                    p_schedule=p_schedule,
                                    iters_per_stage=iters_per_stage,
                                    lr0=0.3)
        print(f"   → refined surrogate = {val:.6f}")
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()

    # ------------------------------------------------------------------
    # Phase 3 – final polishing (hard max‑convolution)
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        best_overall_seq = _uniform(n)

    if time.time() < deadline:
        print("\n--- Polishing with exponentiated‑gradient (hard objective) ---")
        polished_seq, polished_val = _exp_gradient_opt(
            best_overall_seq, n, deadline,
            max_iters=4_000_000, eta0=12.0)
        if polished_val < best_overall_val:
            best_overall_seq = polished_seq
            best_overall_val = polished_val
            print(f"[Polish] improved surrogate = {best_overall_val:.6f}")
        else:
            print("[Polish] no improvement")

    # ------------------------------------------------------------------
    # Phase 4 – cheap random local search (optional, cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Local random hill‑climbing ---")
        best_seq = _local_random_search(best_overall_seq, n, deadline,
                                        steps=3_000, epsilon=0.015)
        best_val = _objective_surrogate(best_seq)
        if best_val < best_overall_val:
            best_overall_seq = best_seq
            best_overall_val = best_val
            print(f"[Local] improved surrogate = {best_overall_val:.6f}")
        else:
            print("[Local] no improvement")

    # ------------------------------------------------------------------
    # Final evaluation with the provided black‑box function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_overall_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
