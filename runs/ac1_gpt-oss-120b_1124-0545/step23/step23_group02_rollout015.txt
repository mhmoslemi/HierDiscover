def propose_candidate(seed=123,budget_s=1000, **kwargs):

    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    n = 4000  # maybe adjust to 5000 if time.

    # Retrieve previous best if any.
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.shape == (n,):
            prev_vec = arr / arr.sum()

    # Build seeds:
    seeds = []
    seeds.append(_uniform(n))
    # Beta shapes grid more fine
    for a in np.logspace(-3, 0, 8):
        for b in np.logspace(-3, 0, 8):
            seeds.append(_beta_shape(n, a, b))
    # Exp decays
    for rate in np.logspace(-6, -0.3, 10):
        seeds.append(_exp_decay(n, rate))
    # Power laws
    for p in np.linspace(0., 6., 13):
        seeds.append(_power(n, p))
    # Logistic
    for scale in (0.2,0.5,1.0,2.0):
        for midpoint in (int(n*0.25), int(n*0.5), int(n*0.75)):
            seeds.append(_logistic_shape(n, scale, midpoint))
    # Random Dirichlet
    for conc in (0.0005, 0.001,0.005,0.01,0.05,0.1):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)))
    if prev_vec is not None:
        seeds.append(prev_vec)
        # add perturbed variants
        for _ in range(3):
            pert = prev_vec + 0.005 * rng.normal(size=n)
            pert = np.maximum(pert,0)
            s=pert.sum()
            if s>0:
                seeds.append(pert/s)

    # Evaluate surrogates and keep top K (maybe 12)
    evaluated = []
    for vec in seeds:
        if time.time()>deadline:
            break
        vec = np.maximum(vec,0)
        s=vec.sum()
        if s==0: continue
        vec=vec/s
        val=_objective_surrogate(vec)
        evaluated.append((vec,val))
    if not evaluated:
        raise RuntimeError("No seeds evaluated")
    evaluated.sort(key=lambda x:x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec,_ in evaluated[:top_k]]
    print(f"Top {top_k} seeds (surrogate):", [f"{val:.6f}" for _,val in evaluated[:top_k]])

    # Optimize each seed with smoothed gradient descent.
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"Initial global surrogate: {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time()>deadline:
            break
        # set individual deadline: allocate fraction of remaining time
        remaining = deadline - time.time()
        # each seed gets maybe 1/(remaining seeds) fraction * 0.6 time

        share = remaining * 0.5 / max(1, (top_k - idx))
        sub_deadline = time.time() + share

        # Run smooth gradient descent
        seq, val_surrogate = _smooth_max_grad_descent(seed_vec, n, sub_deadline, rng)

            seq2,val2=seq,val_surrogate
        # maybe SA polishing

        candidates = [(seed_vec,_objective_surrogate(seed_vec)),

                     (seq2,val2),
                     (seq3,val3)]
        local_seq, local_val = min(candidates, key=lambda x:x[1])
        print(f"[Seed {idx+1}/{top_k}] local best surrogate {local_val:.6f}")
        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f" New global best {global_best_val:.6f}")

    # final heavy polishing using smoothed gradient descent with high alpha then exact EG
    if time.time()<deadline:
        # start from global best
        seq, val = _smooth_max_grad_descent(global_best_seq, n, deadline, rng,

def _smooth_max_grad_descent(start_vec, n, deadline, rng, start_alpha=0.1, final_alpha=2000.0, max_iters=400000, base_eta=5.0):
    a = start_vec.copy()
    log_a = np.log(a + 1e-30)
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_val = _objective_surrogate(a)

    # schedule alpha (geometric)
    # maybe increase linearly in log-space: alpha = start_alpha * (final_alpha/start_alpha)**(float(it)/max_iters)
    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        # Determine alpha
        alpha = start_alpha * (final_alpha/start_alpha) ** (float(it)/max_iters)

def _smooth_max_opt(start_vec, n, deadline, rng, max_iters=300000, base_eta=1.0, start_alpha=0.1, final_alpha=500.0):
    a = start_vec.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        alpha = start_alpha * (final_alpha/start_alpha) ** (float(it)/max_iters)
        # compute conv once
        a = _softmax(log_a) # ensure simplex
        conv = _conv_fft(a, a)
        # compute weight w
        # to avoid overflow, shift by max(alpha*conv)
        max_ac = alpha * np.max(conv)
        exp_term = np.exp(alpha * conv - max_ac)
        w = exp_term / exp_term.sum()
        # gradient
        grad_a = 2.0 * _conv_fft(a[::-1], w)[n-1:n-1+n]
        grad = 2.0 * n * grad_a
        dot = np.dot(grad, a)
        grad_log = a * (grad - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = base_eta / (np.sqrt(v_hat) + eps)
        log_a -= step * m_hat

        # evaluate surrogate
        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val

"""
Improved step‑function optimiser (≈ 1000 s budget)

The evaluation objective is scale‑invariant, therefore the search is performed
on the probability simplex (∑a = 1).  The optimiser consists of the following
phases:

1.  Build a rich library of seed vectors (uniform, beta‑shapes,
    exponential tails, power‑laws, logistic, DPSS windows and random
    Dirichlet draws).  Keep the K best seeds according to the cheap
    surrogate  2·n·max(conv(a,a)).

2.  For each seed run a *smooth‑max* L‑BFGS optimiser.  The nonsmooth max
    is approximated by the log‑sum‑exp with a temperature α which is
    increased geometrically (α ≈ 5 → 10 → 30 → 100 → 400).  The optimisation
    is performed on log‑softmax parameters, stays on the simplex
    automatically and uses exact gradients (convolution via FFT).

3.  Refine the obtained vectors with the original sub‑gradient mirror‑descent
    optimiser (exponentiated‑gradient), which directly minimises the true
    max‑convolution objective.

4.  A short simulated‑annealing polish finishes each candidate.

5.  The overall best sequence is finally polished with a long
    exponentiated‑gradient run.

All stages respect the 1000 s wall‑clock budget, printing progress so that
the best sequence is stored in the global variable ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

try:
    from scipy.signal.windows import dpss as _dpss
except Exception:  # pragma: no cover
    _dpss = None

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (∑seq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i⁻ᵖ (i starts from 1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_shape(n: int, nw: float) -> np.ndarray:
    """Principal DPSS window (positive version) normalised to sum = 1."""
    if _dpss is None:
        return _uniform(n)
    try:
        w = _dpss(n, nw, Kmax=1, sym=True)[0]
    except Exception:
        return _uniform(n)
    w = np.maximum(w, 0.0)
    s = w.sum()
    return w / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _lbfgs_smooth_max(start_vec: np.ndarray,
                      n: int,
                      deadline: float,
                      alphas: list[float] = (5.0, 20.0, 80.0, 300.0, 1000.0),
                      max_iter_per_alpha: int = 250) -> np.ndarray:
    """
    Minimises 2·log‑sum‑exp(α·conv) / α  (α increasing) using L‑BFGS on the
    log‑softmax parameters.  Returns the best (ℓ₁‑normalised) vector found.
    """
    # initialise log‑softmax variables
    v = np.log(start_vec + 1e-30)

    # helper: callback to abort when time runs out
    def _cbk(_):
        if time.time() > deadline:
            raise KeyboardInterrupt()

    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    for alpha in alphas:
        if time.time() > deadline:
            break

        # loss / gradient for a given alpha
        def _fun(vpar):
            a = _softmax(vpar)
            conv = _conv_fft(a, a)                     # length 2n‑1
            # log‑sum‑exp in a numerically stable way
            z = alpha * conv
            m = np.max(z)
            expz = np.exp(z - m)
            sumexp = expz.sum()
            loss = (2.0 / alpha) * (m + np.log(sumexp))

            w = expz / sumexp                           # weights sum to 1
            # gradient w.r.t. a (exact): 4 * conv_fft(a_rev, w)
            grad_a = 4.0 * _conv_fft(a[::-1], w)[n - 1:n - 1 + n]

            # chain‑rule through soft‑max (log‑parameter)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return loss, grad_v

        if _opt is not None:
            try:
                res = _opt.minimize(fun=_fun,
                                    x0=v,
                                    method='L-BFGS-B',
                                    jac=True,
                                    callback=_cbk,
                                    options={'maxiter': max_iter_per_alpha,
                                             'disp': False,
                                             'ftol': 1e-12})
                v = res.x
            except KeyboardInterrupt:
                # time out – keep current iterate
                pass
            except Exception:
                # fallback to next alpha if something goes wrong
                pass

    seq = _softmax(v)
    val = _objective_surrogate(seq)
    if val < best_val:
        best_seq = seq
        best_val = val
    return best_seq


# ----------------------------------------------------------------------
# Exponentiated‑gradient on the exact max‑convolution (multi‑active)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 800_000,
                            eta0: float = 8.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses all indices whose value is within `tol`·max of the current maximum.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient (average over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑like momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 120_000,
                         start_sigma: float = 0.0015,
                         T_start: float = 0.015,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s wall‑clock budget, stores the best found
    sequence in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # problem size – a few thousand entries give a good trade‑off
    n = 4000

    # ------------------------------------------------------------------
    # Restore previous best if it exists (helps warm‑starts)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            pa = np.asarray(previous, dtype=np.float64)
            if pa.shape == (n,):
                s = pa.sum()
                if s > 0:
                    prev_arr = pa / s
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Build a compact, diverse seed set
    # ------------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # analytic families
    seeds.append(_uniform(n))

    # beta shapes – dense grid (U‑shaped and bell‑shaped)
    for a in np.logspace(-2.5, 0.0, 6):
        for b in np.logspace(-2.5, 0.0, 6):
            seeds.append(_beta_shape(n, a, b))

    # exponential decays (heavy‑tail)
    for rate in np.logspace(-6, -0.3, 10):
        seeds.append(_exp_decay(n, rate))

    # power‑law tails
    for p in np.linspace(0.0, 5.0, 9):
        seeds.append(_power(n, p))

    # logistic (sigmoid) shapes placed at both ends
    for scale in (0.5, 1.0, 2.0):
        for mid in (int(n * 0.2), int(n * 0.5), int(n * 0.8)):

    # DPSS windows (different time‑bandwidth products)

    # random Dirichlet draws (various concentrations)
    for conc in (0.0005, 0.001, 0.01, 0.1, 0.5):

    # previous best (if any) and tiny perturbations thereof

        vec = np.maximum(vec, 0.0)

        if s == 0.0:

        vec = vec / s
        val = _objective_surrogate(vec)

    if not evaluated:
        raise RuntimeError("No valid seeds were generated.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a time slice for this seed (≈ 15% of the remaining budget)

        share = remaining * 0.15

        print(f"\n[Seed {si+1}/{top_k}] optimisation window ends at {time.strftime('%H:%M:%S', time.localtime(sub_deadline))}")

        # ----- a) smooth‑max L‑BFGS -------------------------------------------------
        seq_lb = _lbfgs_smooth_max(seed_vec, n, sub_deadline)

        # ----- b) exponentiated‑gradient on the exact max ---------------------------
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt_multi(seq_lb,
                                                    n,
                                                    sub_deadline,
                                                    max_iters=500_000,
                                                    eta0=10.0,
                                                    tol=1e-9)
        else:
            seq_eg, val_eg = seq_lb, val_lb

        else:

        ]

        if local_best_val < global_best_val:
