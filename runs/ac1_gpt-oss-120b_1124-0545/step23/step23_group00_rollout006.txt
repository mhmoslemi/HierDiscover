import cvxpy as cp
def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    conv_f_g = cp.conv(cp.Parameter(f, constant=True), g)
    constraints = [conv_f_g <= M]
    # Also define maybe lower bound 0
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, max_iters=2000)
    # return solution
    g_val = g.value
    if g_val is None:
        return None
    return np.maximum(g_val,0)

    deadline = start + budget_s - 5

    # Load previous seq
    seq = previous or exponential
    seq = seq / seq.sum()
    n = len(seq)

    best_seq = seq

    while time.time() < deadline:

        conv_self = np.convolve(seq, seq)
        M = np.max(conv_self)

            break

import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    f_param = cp.Parameter(n, value=f, nonneg=True)
    conv_fg = cp.conv(f_param, g)
    constraints = [conv_fg <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-9, reltol=1e-9)
    except Exception as e:
        print("LP solve exception:", e)

"""
propose_candidate
-----------------

A two‑phase optimizer for the step‑function problem.

Phase 1 – start from the best known sequence (or a simple exponential decay) and
ensure it is a probability vector (sum = 1).

Phase 2 – iterative linear‑programming refinement.
    * For the current vector f we compute M = max(conv(f,f)).
    * We solve the LP

          maximise   sum(g)
          subject to  f * g ≤ M   (entry‑wise convolution)
                      g ≥ 0

      The LP is solved with SciPy’s *highs* interior‑point method; the constraint
      matrix is a sparse Toeplitz matrix built once per iteration.

    * If the LP yields a feasible g with sum(g)>1 we normalise it to a
      probability vector ĝ = g / sum(g) and form a convex combination
      f′ = (1‑t)·f + t·ĝ (t∈{0.5,0.3,0.2,0.1,0.05}).
      The combination is accepted when the true objective
      2·n·max(conv(f′,f′)) decreases.
    * The process repeats until no improvement is possible or the time budget
      expires.

The algorithm exploits the fact that the bound “max conv(f,g) ≤ M” is linear in
g, while the objective depends only on the self‑convolution of f.  By
alternately expanding the feasible mass in g and mixing back into f we obtain
monotone reductions of the max‑convolution, pushing the score below the
previous best (≈ 1.5085) and towards the target 1.503.

The final sequence is stored in the global variable ``height_sequence_1`` for
possible continuation runs.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _max_conv(seq: np.ndarray) -> float:
    """Return max(conv(seq, seq))."""
    return float(np.max(np.convolve(seq, seq)))


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) (sum(seq)=1)."""
    n = len(seq)
    return 2.0 * n * _max_conv(seq)


def _build_conv_matrix(f: np.ndarray, rows_cache=None, cols_cache=None):
    """
    Build the sparse matrix A (shape (2n‑1, n)) that implements the linear map
    g ↦ conv(f, g).  For each column j the constant coefficient f[j] appears on
    the diagonal with offset j.
    """
    n = len(f)
    if rows_cache is None or cols_cache is None:
        # static row/col pattern, independent of the actual values of f
        i = np.arange(n, dtype=np.int32)
        j = np.arange(n, dtype=np.int32)
        rows = (i[:, None] + j[None, :]).reshape(-1)
        cols = np.tile(j, n)
    else:
        rows, cols = rows_cache, cols_cache
    data = np.tile(f, n)                       # f[i] repeated n times
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A, rows, cols


def _solve_lp_max_sum_g(f: np.ndarray, M: float, rows_cache, cols_cache):
    """
    Solve   maximise sum(g)   s.t.   conv(f,g) ≤ M, g ≥ 0.
    Returns the optimal vector g (may be None if the solver fails).
    """
    n = len(f)
    # Build the constraint matrix A (conv(f,g) = A @ g)
    A, rows, cols = _build_conv_matrix(f, rows_cache, cols_cache)

    c = -np.ones(n)                     # linprog minimises; -sum(g) → maximise sum(g)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    bounds = [(0.0, None)] * n

    # Use HiGHS (fast, sparse‑matrix aware)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if not res.success or res.x is None:
        print("   LP failed:", res.message)
        return None
    return np.maximum(res.x, 0.0)


def _mix_and_evaluate(f: np.ndarray, g_norm: np.ndarray, t: float) -> tuple[np.ndarray, float]:
    """
    Return (f_new, obj_new) where f_new = (1‑t)·f + t·g_norm and obj_new is the
    exact objective value after renormalising (renormalisation is a no‑op
    because both summands sum to 1, but we keep it for numerical safety).
    """
    f_new = (1.0 - t) * f + t * g_norm
    # enforce non‑negativity / tiny rounding errors
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s == 0.0:
        return f, np.inf
    f_new /= s
    obj_new = _objective(f_new)
    return f_new, obj_new

# ----------------------------------------------------------------------

        cur_seq = best_seq.copy()
        cur_obj = best_obj
        T_start = 1e-4 * cur_obj
        T_end   = 1e-7 * cur_obj
        max_iters = 300_000
        for it in range(1, max_iters + 1):
            if time.time() > deadline:
                break
            T = T_start + (T_end - T_start) * (it / max_iters)

            if rng.random() < 0.6:                    # swap move
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    ns = cur_seq.copy()
                    ns[i], ns[j] = ns[j], ns[i]
                else:
                    continue
            else:                                     # tiny Gaussian blur
                sigma = 0.0005 / np.sqrt(it + 1)
                ns = cur_seq + sigma * rng.normal(size=n)
                ns = np.maximum(ns, 0.0)
                s = ns.sum()
                if s == 0.0:
                    continue
                ns /= s

            ns_obj = _objective(ns)
            if ns_obj < cur_obj:
                cur_seq, cur_obj = ns, ns_obj
                if cur_obj < best_obj:
                    best_seq, best_obj = cur_seq, cur_obj
            else:
                # accept occasional uphill moves
                if T > 0.0 and rng.random() < np.exp(-(ns_obj - cur_obj) / T):
                    cur_seq, cur_obj = ns, ns_obj

    # --------------------------------------------------------------
    # 5.  Final evaluation with the black‑box function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {best_obj:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
