    v = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    m = np.zeros_like(a)
    v2 = np.zeros_like(a)
    beta1 = 0.9; beta2 = 0.999; eps=1e-8
    t=0
    lam_start = 1.0
    lam_end = max(20.0, n*0.5)   # maybe n*0.5 for n=4000 gives 2000.
    # compute for each iteration
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(v)
        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        # compute weight w via softmax weighted by lambda
        lam = lam_start + (lam_end - lam_start) * (t / max_iters)
        # subtract max for numerical stability
        diff = lam * (conv - max_val)
        # compute exp of diff
        # To avoid overflow with large lam, using np.exp(diff) may underflow for large negative diff but that's ok.
        w = np.exp(diff)
        sum_w = w.sum()
        if sum_w == 0.0:
            # fallback to uniform weights
            w = np.full_like(conv, 1.0 / len(conv))
        else:
            w /= sum_w
        # compute gradient w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1 : n-1+n]
        # gradient w.r.t v (log-softmax)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1-beta1) * grad_v
        v2 = beta2 * v2 + (1-beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1**t)

        base_lr = 0.15

        # update best

    return best_seq, best_val

"""
Smooth‑max optimiser for the convolution‑peak problem.

The objective (with ∑a = 1) is   F(a) = 2·n·max_k (a * a)_k .
We minimise a smooth approximation

    Sλ(a) = (1/λ)·log( Σ_k exp( λ·(a*a)_k ) ),

which converges to max when λ → ∞ .
Using the log‑softmax parametrisation a = softmax(v) guarantees the simplex
constraint and enables back‑propagation of the gradient.
A schedule raises λ from a modest value to ≈n, turning the optimisation
from an L₂‑flattening phase into a genuine max‑reduction phase.
The optimiser runs as many Adam‑steps as time permits and keeps the best
vector seen (measured by the exact surrogate 2·n·max(conv)).
The routine builds a diverse seed library (uniform, beta‑U‑shapes,
power‑law, exponentials, DPSS windows if SciPy is present, random
Dirichlet draws and the previous best), keeps the best few seeds and
optimises each of them with the smooth‑max method.  A final long run
polishes the overall best candidate.

The global variable ``height_sequence_1`` stores the best list of heights
found so that a later invocation can warm‑start from it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for DPSS windows (if available)
# ----------------------------------------------------------------------
try:
    from scipy.signal import windows as _windows
except Exception:  # pragma: no cover
    _windows = None

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real → real)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → probability vector (∑=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(a,a)) with ∑a = 1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine‑like shape (β(½,½) density) – useful seed."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n - i + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_seeds(n: int):
    """Generate a few DPSS windows (positive, normalised)."""
    if _windows is None:
        return []
    seeds = []
    for NW in (2.0, 3.0, 4.0):
        try:
            vec = _windows.dpss(M=n, NW=NW, Kmax=1, sym=False)[0]
            vec = np.abs(vec)                 # first DPSS is non‑negative
            sm = vec.sum()
            if sm > 0:
                seeds.append(vec / sm)
        except Exception:
            continue
    return seeds


# ----------------------------------------------------------------------
# Smooth‑max optimiser (softmax of the convolution)
# ----------------------------------------------------------------------
def _smoothmax_optimize(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 1_000_000,
                        lam_start: float = 1.0,
                        lam_factor: float = 3.0) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best sequence seen and its exact surrogate value.
    """
    a = start_seq.copy()
    # initialise log‑softmax parameters
    v = np.log(a + 1e-30)

    # Adam state
    m = np.zeros_like(a)
    v2 = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # schedule for λ : λ_t = lam_start * (lam_factor * n) ** (t / max_iters)
    lam_max = lam_start * lam_factor * n

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current probability vector
        a = _softmax(v)

        # full auto‑convolution
        conv = _conv_fft(a, a)               # length 2n‑1
        max_c = np.max(conv)

        # smooth‑max weights (softmax on λ·conv, stabilised by max)
        lam = lam_start + (lam_max - lam_start) * (it / max_iters)
        # use max_c for numerical stability
        diff = lam * (conv - max_c)
        w = np.exp(diff)
        sw = w.sum()
        if sw == 0.0:                     # extremely unlikely
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= sw                       # now Σ w = 1

        # gradient wrt a  (2·rev_a * w, sliced to length n)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n - 1: n - 1 + n]

        # chain rule for log‑softmax parametrisation
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        # base learning rate – a modest value works well for all n
        base_lr = 0.15
        step = base_lr * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))

        v -= step * m_hat / (np.sqrt(v2_hat) + eps)

        # evaluate the exact surrogate (cheap, uses the max already computed)
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s wall‑clock budget,
    stores the best sequence in the global variable ``height_sequence_1``
    and returns it as a Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Build a diverse seed library
    # ------------------------------------------------------------------
    seeds = [_uniform(n)]

    # Analytic families
    for a in (0.05, 0.1, 0.2, 0.5):
        for b in (0.05, 0.1, 0.2, 0.5):
            seeds.append(_beta_shape(n, a, b))

    for p in (0.0, 0.3, 0.6, 1.0, 2.0, 3.0):
        seeds.append(_power(n, p))

    for rate in np.logspace(-5, -0.5, 8):
        seeds.append(_exp_decay(n, rate))

    for scale in (0.5, 1.0, 2.0):
        for mid in (n * 0.25, n * 0.75):
            seeds.append(_logistic_shape(n, scale, mid))

    seeds.append(_arcsine_shape(n))

    # DPSS windows (if SciPy is present)
    seeds.extend(_dpss_seeds(n))

    # Random Dirichlet draws – a few concentrations
    for conc in (0.001, 0.01, 0.1, 0.5):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # Warm‑start from the previous best (if it matches the current n)
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                sm = arr.sum()
                if sm > 0:
                    seeds.append(arr / sm)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Quick surrogate evaluation of all seeds, keep the top few
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed evaluation (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k], 1):
        print(f"  #{i}: {v:.6f}")

    # ------------------------------------------------------------------
    # Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Optimise each top seed (smooth‑max + Adam)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a fair share of the remaining time
        remaining = deadline - time.time()
        share = remaining * 0.12 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        seq_opt, val_opt = _smoothmax_optimize(seed_vec,
                                                n,
                                                sub_deadline,
                                                max_iters=300_000)
        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"[Seed {si+1}/{top_k}] new global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final long polishing on the overall best (as much time as left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long polishing ---")
        final_seq, final_val = _smoothmax_optimize(global_best_seq,
                                                   n,
                                                   deadline,
                                                   max_iters=2_000_000)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation via the provided evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
