def _adam_subgrad_opt(start_seq, deadline, lr=0.2, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)  # initial logits
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    it = 0
    best_seq = _softmax(logit)  # initial
    best_val = _objective_surrogate(best_seq)

    while time.time() < deadline:
        it += 1
        a = _softmax(logit)
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # subgradient
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        rev_idx = k - idx
        g[idx] = 2.0 * a[rev_idx]

        # logit gradient
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 2000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq = final_a
        best_val = final_val

    return best_seq, best_val
```

def _directed_local_search(start_seq, deadline, attempts=200000, step_factor=0.02):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    n = len(a)
    while attempts>0 and time.time() < deadline:
        attempts -= 1
        # compute conv and active index
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idxs = np.arange(i_start, i_end+1)
        rev_idx = k - idxs
        g_vals = 2.0 * a[rev_idx]
        i_max = idxs[np.argmax(g_vals)]
        i_min = idxs[np.argmin(g_vals)]
        if i_max == i_min or a[i_max] < 1e-12:
            continue
        delta = step_factor * a[i_max] * np.random.rand()
        if delta > a[i_max]:
            delta = a[i_max]
        new_seq = a.copy()
        new_seq[i_max] -= delta
        new_seq[i_min] += delta
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
            a = new_seq  # accept
    return best_seq, best_val
```

"""
Advanced step‑function optimisation.

The search proceeds in four phases:

1️⃣  A diverse seed pool (uniform, arcsine, triangular, many Beta‑shapes,
    Dirichlet draws and the previously stored solution) is generated and
    evaluated with the cheap surrogate 2·n·max(conv(a,a)).

2️⃣  Starting from the best seed we run Adam on the *exact* sub‑gradient
    of the max‑convolution (the true objective).  Adam’s adaptive steps
    give a faster, more stable descent than the plain exponentiated‑gradient
    used previously.

3️⃣  A directed local search repeatedly moves a small amount of mass from a
    coordinate that most hurts the active max‑convolution to a coordinate
    that helps most.  This is a cheap greedy refinement that can squeeze out
    the final few thousandths.

4️⃣  The final sequence is evaluated with the supplied ``evaluate_sequence``
    and stored globally as ``height_sequence_1`` for warm‑starting the next
    call.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Softmax that stays on the simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate: 2·n·max(conv(seq, seq)).
    The input is assumed to sum to 1 (i.e. a probability vector).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth (log‑sum‑exp) approximation of the max‑convolution.
    Returns best probability vector seen and its exact surrogate value.
    """
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0,
        20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        max_iter_beta = 8000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break
            a = _softmax(logit)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.12
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 600 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _adam_subgrad_opt(start_seq: np.ndarray,
                     deadline: float,
                     lr: float = 0.20,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser using the *exact* sub‑gradient of the max‑convolution.
    Works directly on logits (softmax parametrisation of the simplex).
    """
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    it = 0
    a = _softmax(logit)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    while time.time() < deadline:
        it += 1
        a = _softmax(logit)
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active max‑convolution entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx
        g[idx] = 2.0 * a[rev_idx]

        # gradient w.r.t. logits (mirror map of the simplex)
        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 2000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _directed_local_search(start_seq: np.ndarray,
                           deadline: float,
                           attempts: int = 250_000,
                           step_factor: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy local refinement: repeatedly move a small amount of mass from the
    index that most *hurts* the current max‑convolution to the index that most
    *helps* it (according to the active gradient).
    """
    n = len(start_seq)
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx
        g_vals = 2.0 * a[rev_idx]          # gradient magnitudes for the active term

        i_max = idx[np.argmax(g_vals)]
        i_min = idx[np.argmin(g_vals)]
        if i_max == i_min or a[i_max] <= 0.0:
            continue

        delta = step_factor * a[i_max] * np.random.rand()
        if delta > a[i_max]:
            delta = a[i_max]

        new_a = a.copy()
        new_a[i_max] -= delta
        new_a[i_min] += delta

        val = _objective_surrogate(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a        # accept the improvement immediately

    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 300_000,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Unbiased tiny stochastic walk on the simplex (kept for fallback).
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found within the allotted budget is stored globally as
    ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Load previous solution if it matches the required length
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 0️⃣ Assemble a diverse seed pool
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # Beta‑shaped seeds (α varies, including more extreme values)
    for α in (0.05, 0.08, 0.1, 0.12, 0.15, 0.2, 0.3, 0.4,
              0.5, 0.7, 0.9, 1.2, 1.5, 2.0, 2.5, 3.0):
        seeds.append(_beta_distribution(n, α))

    # Dirichlet draws with various concentrations (including very spiky)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣ Initial evaluation of seeds (cheap surrogate)
    # ------------------------------------------------------------------
    seed_infos = []
    print("\n--- Initial seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_k = 5
    top_seeds = [vec for _, vec in seed_infos[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Adam smoothing (smooth max) – about 25 % of remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        smooth_deadline = time.time() + remaining * 0.25
        print("\n--- Adam smoothing (smooth max) ---")
        seq_smooth, val_smooth = _adam_smoothing(best_seq, smooth_deadline)
        print(f"[Smooth] best exact surrogate = {val_smooth:.6f}")
        if val_smooth < best_val:
            best_val, best_seq = val_smooth, seq_smooth.copy()
            print("   >>> Updated global best (smooth)")

    # ------------------------------------------------------------------
    # 3️⃣ Adam sub‑gradient optimisation (major part of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = time.time() + (deadline - time.time()) * 0.55
        print("\n--- Adam sub‑gradient optimisation (exact max) ---")
        seq_adam, val_adam = _adam_subgrad_opt(best_seq, subgrad_deadline,
                                               lr=0.20)
        print(f"[Adam‑SG] exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam‑SG)")

    # ------------------------------------------------------------------
    # 4️⃣ Directed greedy local refinement (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Directed greedy local refinement ---")
        seq_dir, val_dir = _directed_local_search(best_seq, deadline,
                                                  attempts=300_000,
                                                  step_factor=0.02)
        print(f"[Directed] exact surrogate = {val_dir:.6f}")
        if val_dir < best_val:
            best_val, best_seq = val_dir, seq_dir.copy()
            print("   >>> Updated global best (directed)")

    # ------------------------------------------------------------------
    # 5️⃣ Final random walk (fallback polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, deadline,
                                                attempts=200_000,
                                                sigma=0.015)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (random)")

    # ------------------------------------------------------------------
    # 6️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
