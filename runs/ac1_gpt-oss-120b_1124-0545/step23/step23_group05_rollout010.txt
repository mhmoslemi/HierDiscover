Main ideas

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (sum(seq) == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1) – the classic solution of the continuous flat‑conv problem."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate*i) , i=0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    """Sigmoid shape, optionally shifted."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mixed_exponential(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    """Convex mixture of two exponentials."""
    e1 = np.exp(-r1 * np.arange(n, dtype=np.float64))
    e2 = np.exp(-r2 * np.arange(n, dtype=np.float64))
    a = (1 - alpha) * e1 + alpha * e2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet distribution (the whole vector is already normalised)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window – requires scipy.signal."""
    if _sigwin is None:
        return _uniform(n)
    a = _sigwin.dpss(n, nw, Kmax=1, sym=True)
    a = np.abs(a[0])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_exp_shape(n: int, p: float, lam: float) -> np.ndarray:
    """Two‑parameter shape a_i ∝ i⁻ᵖ·e⁻ˡᵃᵐᵇ𝑑𝑎·(i‑1).  Handles lam≈0 gracefully."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (-p)) * np.exp(-lam * (i - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Mirror‑descent on the exact max‑convolution (multi‑active)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 2_000_000,
                            eta0: float = 15.0,
                            tol: float = 1e-12,
                            beta_mom: float = 0.95) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient on the exact max‑convolution.
    Uses all indices whose value is within *tol*·max.
    Returns the best sequence found and its surrogate value.
    """
    a = start_seq.copy()
    # work in log‑softmax space
    log_a = np.log(a + 1e-30)

    momentum = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set: indices that are “almost’’ maximal
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient (average over active indices)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑like momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient‑descent that flattens the convolution (quadratic loss)
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 2_000_000,
                         eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the loss Σ_k (conv_k - avg)² .
    The aim is to make the whole convolution as flat as possible.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)                # length of full convolution
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)                 # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent step
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    """
    Optimise a smooth max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space to keep the iterate on the simplex.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 300_000,
                         start_sigma: float = 0.0015,
                         T_start: float = 0.012,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Full optimisation pipeline for a single seed
# ----------------------------------------------------------------------
def _full_optimisation(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator,
                       quick: bool = False) -> tuple[np.ndarray, float]:
    """
    Run a multi‑stage optimisation on *start_vec*.
    If *quick* is True we only run the exponentiated‑gradient phase
    (big budget) – useful for the per‑seed stage.
    Otherwise we run EG → L‑BFGS → flatten → SA (full polishing).
    Returns the best sequence encountered and its surrogate value.
    """
    # 1) exponentiated‑gradient (max‑convolution)
    seq_eg, val_eg = _exp_gradient_opt_multi(
        start_vec.copy(),
        n,
        deadline,
        max_iters=5_000_000,
        eta0=15.0,
        tol=1e-12,
        beta_mom=0.95,
    )

    if quick or time.time() >= deadline:
        return seq_eg, val_eg

    # 2) smooth L‑BFGS
    seq_lbfgs = _lbfgs_smooth_optimize(
        seq_eg,
        n,
        deadline,
        beta_schedule=None,
        max_iter_per_beta=3000,
    )
    val_lbfgs = _objective_surrogate(seq_lbfgs)

    # 3) flattening gradient (longer than the reference implementation)
    seq_flat, val_flat = _flatten_gradient_opt(
        seq_lbfgs,
        n,
        deadline,
        max_iters=2_000_000,
        eta0=8.0,
    )

    # 4) simulated annealing polish
    seq_sa, val_sa = _simulated_annealing(
        seq_flat,
        n,
        deadline,
        rng,
        max_attempts=300_000,
        start_sigma=0.001,
        T_start=8e-3,
        T_end=5e-9,
    )

    candidates = [
        (start_vec, _objective_surrogate(start_vec)),
        (seq_eg, val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_flat, val_flat),
        (seq_sa, val_sa),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, stores the best found sequence in
    the global variable ``height_sequence_1`` and returns it as a Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – we keep it moderate while still allowing a fine shape
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # ------------------------------------------------------------------
    # 1. Build a rich seed set (many analytic families + random draws)
    # ------------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # ---- basic analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_cosine_window(n))
    seeds.append(_kaiser_window(n, beta=8.0))
    seeds.append(_dpss_window(n, nw=3.5))

    # ---- power families ---------------------------------------------------------
    for p in [0.4, 0.5, 0.6, 0.8, 1.0, 1.5, 2.0]:
        seeds.append(_power(n, p))

    # ---- classic inverse‑sqrt ----------------------------------------------------
    seeds.append(_inverse_sqrt(n))

    # ---- exponential families ----------------------------------------------------
    rates = np.logspace(-5, -0.2, 20)
    for r in rates:
        seeds.append(_exp_decay(n, r))

    # ---- mixed exponentials ------------------------------------------------------
    for r1 in rates[:10]:
        for r2 in rates[10:]:
            seeds.append(_mixed_exponential(n, r1, r2, alpha=0.3))

    # ---- Gaussian windows ---------------------------------------------------------
    for sigma_frac in np.linspace(0.05, 0.35, 7):
        seeds.append(_gaussian(n, sigma_frac))

    # ---- beta‑shape grid ---------------------------------------------------------
    betas = np.linspace(0.01, 10.0, 25)
    alphas = np.linspace(0.01, 10.0, 25)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # ---- logistic shapes ---------------------------------------------------------
    for scale in [0.3, 0.6, 1.2, 2.5]:
        for mid in [n * 0.2, n * 0.5, n * 0.8]:
            seeds.append(_logistic_shape(n, scale, mid))

    # ---- dense two‑parameter power‑exponential sweep ----------------------------
    p_vals = np.linspace(0.3, 1.2, 16)               # exponent p
    lam_vals = np.logspace(-5, -2, 16)               # decay λ
    for p in p_vals:
        for lam in lam_vals:
            seeds.append(_power_exp_shape(n, p, lam))

    # ---- random Dirichlet draws ---------------------------------------------------
    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(10):
            seeds.append(_random_dirichlet(n, conc, rng))

    # ---- mixtures with uniform ----------------------------------------------------
    for alpha in [0.1, 0.3, 0.5, 0.7]:
        base = _inverse_sqrt(n)
        seeds.append(_mix_shapes(_uniform(n), base, alpha))

    # ---- reversed versions --------------------------------------------------------
    rev_seeds = [s[::-1].copy() for s in list(seeds)]
    seeds.extend(rev_seeds)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds (keep only the best few)
    # ------------------------------------------------------------------
    top_k = 20
    evaluated = []       # list of (vector, surrogate) pairs, kept sorted

    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec /= s                       # ℓ₁‑normalise
        val = _objective_surrogate(vec)

        # keep a small sorted buffer of size top_k
        if len(evaluated) < top_k:
            evaluated.append((vec, val))
            evaluated.sort(key=lambda x: x[1])
        else:
            # if better than worst stored entry, replace it
            if val < evaluated[-1][1]:
                evaluated[-1] = (vec, val)
                evaluated.sort(key=lambda x: x[1])

    print("\nTop seeds after fast screening (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:min(8, len(evaluated))]):
        print(f"  #{i+1}: {v:.6f}")

    # Keep the best few seeds for the heavy optimisation phase
    top_seeds = [vec for vec, _ in evaluated[:min(8, len(evaluated))]]

    # ------------------------------------------------------------------
    # 3. Global best initialise
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine each top seed – allocate time dynamically
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # we reserve about 50 % of the remaining budget for the per‑seed phase,
        # split equally among the yet‑to‑process seeds.
        share = remaining * 0.5 / max(1, (len(top_seeds) - si))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}/{len(top_seeds)}] optimisation budget ≈ {share:.1f}s")
        seq_opt, val_opt = _full_optimisation(seed_vec, n, sub_deadline, rng, quick=True)

        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"   *** New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final intensive polishing on the current global best (remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final intensive polishing on global best ---")
        final_seq, final_val = _full_optimisation(
            global_best_seq,
            n,
            deadline,
            rng,
            quick=False,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
