"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Broad seeding** – generate a large, diverse pool of analytically‑driven
   candidates (uniform, exponentials, power‑laws, shifted power‑laws,
   beta‑shapes, random Dirichlet draws, triangular / two‑block shapes).  
   All vectors are ℓ₁‑normalised (sum = 1) so that the cheap surrogate
   ``2·n·max(conv(a,a))`` coincides with the true objective.

2. **Fine sweep of the shifted‑power‑law family** – this family
   ``a_i ∝ (i+δ)^{-p}`` is known to give very flat self‑convolutions.
   We sweep ``p`` on a fine grid (step 0.005) and ``δ`` on a log‑grid,
   evaluating each seed with the surrogate (≈µs per evaluation).

3. **Select the best few seeds** (≈6).  For every pair of the top three,
   also add convex combinations (weights 0.3, 0.5, 0.7) to enrich the
   candidate set.

4. **Refinement** – each selected seed is improved by two stages:
   * *Adam‐smooth* optimisation on a log‑sum‑exp approximation of the
     max‑convolution (fast early progress).
   * *Exact exponentiated‑gradient* (mirror descent) using the true
     sub‑gradient of the max‑convolution.  The learning‑rate schedule
     ηₜ = η₀/√t with η₀≈8 gives stable convergence.

   Time is split among seeds (more budget for higher‑ranked seeds).
   The best refined sequence is kept globally.

5. **Final polishing** – a long exponentiated‑gradient run on the
   current global best consumes the remaining budget, followed by a
   short permutation‑annealing pass (swap moves) that can shave the last
   bits.

All heavy loops honour a global deadline (default 1000 s).  The final
sequence (list of floats) is stored in ``height_sequence_1`` for possible
reuse in subsequent runs.

The implementation is pure NumPy, uses only the provided ``evaluate_sequence``
and respects all competition constraints.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# FFT‑based convolution (real, full linear convolution)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors):
#   2·n·max(conv(a,a))
# ------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Assumes `a.sum() == 1`."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Analytic families – always return ℓ₁‑normalised vectors
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^{-p}, i = 0…n‑1, δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric tent shape, linear rise then fall."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _two_block(n: int, k_frac: float, ratio: float) -> np.ndarray:
    """
    First block (length = floor(k_frac * n)) has height h1,
    second block height h2 = h1 / ratio, chosen so total ℓ₁‑mass = 1.
    """
    k = max(1, int(np.floor(k_frac * n)))
    # let h2 = h1 / ratio
    # k*h1 + (n-k)*h2 = 1  =>  k*h1 + (n-k)*h1/ratio = 1
    # h1 * (k + (n - k)/ratio) = 1
    denom = k + (n - k) / ratio
    h1 = 1.0 / denom
    h2 = h1 / ratio
    a = np.empty(n, dtype=np.float64)
    a[:k] = h1
    a[k:] = h2
    return a


# ------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution)
# ------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in log‑space (log‑softmax) to enforce positivity and ℓ₁‑norm.
    """
    a = start_seq.copy()
    n = len(a)
    # initialise log‑probabilities
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # re‑normalise the probability vector (numerically stable)
        max_log = np.max(log_a)
        a = np.exp(log_a - max_log)
        a /= a.sum()

        # full self‑convolution and the active index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see discussion above)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # stepsize schedule ηₜ = η₀ / √t
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g

        # keep logs bounded (avoid overflow/underflow)
        log_a = np.clip(log_a, -30.0, 30.0)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ------------------------------------------------------------
# Smoothed Adam optimiser (log‑sum‑exp smoothing of the max)
# ------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 20_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth (log‑sum‑exp) surrogate of the max‑convolution.
    The parameter ``beta`` controls the smoothness; we increase it geometrically.
    """
    if beta_schedule is None:
        beta_schedule = [10, 25, 80, 320, 1280, 5120, 20480, 81920]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.15
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities (log‑softmax)
            max_log = np.max(log_a)
            a = np.exp(log_a - max_log)
            a /= a.sum()

            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp  # soft‑max weights, length = 2n‑1

            # gradient wrt a
            rev_a = a[::-1]
            # full convolution rev_a * w has length 3n‑2 (but we only need the centre n)
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient wrt log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            log_a = np.clip(log_a, -30.0, 30.0)

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser – swaps two entries.
    Helps to escape tiny local traps at the end of the search.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        accept = val < cur_val or (rng.random() < math.exp(-(val - cur_val) / T))
        if accept:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects a total wall‑clock budget (default 1000 s) and
    stores the best found vector in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # small safety margin

    # Problem size – 4000 is a good compromise (the earlier best used this)
    n = 4000

    # --------------------------------------------------------
    # 0. Load previous solution (if any) for warm‑start
    # --------------------------------------------------------
    prev_vec = None
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                total = prev_vec.sum()
                if total > 0:
                    prev_vec = prev_vec / total
        except Exception:
            prev_vec = None

    # --------------------------------------------------------
    # 1. Build a rich pool of seeds (analytic + random)
    # --------------------------------------------------------
    seeds = []

    # basic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.0010))
    seeds.append(_exp_decay(n, rate=0.0025))
    seeds.append(_sym_exp_decay(n, rate=0.0020))
    seeds.append(_triangular(n))
    seeds.append(_power(n, p=0.55))
    seeds.append(_power(n, p=0.70))

    # shifted power‑law grid – fine sweep (p step 0.005, δ log‑grid)
    p_grid = np.arange(0.45, 0.66 + 1e-9, 0.005)      # 0.45 .. 0.66
    delta_grid = np.logspace(-1, 1.3, num=15)       # 0.1 … ~20
    for p in p_grid:
        for delta in delta_grid:
            seeds.append(_power_shift(n, p=float(p), delta=float(delta)))

    # beta‑shapes (randomised)
    for _ in range(8):
        a = rng.uniform(0.2, 2.5)
        b = rng.uniform(0.2, 2.5)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws (various concentrations)
    for conc in (0.02, 0.06, 0.15, 0.4, 1.0, 2.5):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # two‑block shapes (different block ratios)
    for k_frac in (0.1, 0.3, 0.5):
        for ratio in (2.0, 5.0, 10.0):
            seeds.append(_two_block(n, k_frac, ratio))

    # previous solution & permutations
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(4):
            seeds.append(rng.permutation(prev_vec))

    # --------------------------------------------------------
    # 2. Quick surrogate evaluation, keep the best few
    # --------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------
    # 3. Enrich top set with convex combinations (to explore
    #    the convex hull of the best candidates)
    # --------------------------------------------------------
    enriched = top_seeds.copy()
    for i in range(min(3, len(top_seeds))):
        for j in range(i + 1, min(3, len(top_seeds))):
            a = top_seeds[i]
            b = top_seeds[j]
            for w in (0.3, 0.5, 0.7):
                combo = w * a + (1 - w) * b
                combo /= combo.sum() if combo.sum() > 0 else 1.0
                enriched.append(combo)

    # Keep only the most promising ones (keep size manageable)
    enriched_evals = [(v, _objective_surrogate(v)) for v in enriched]
    enriched_evals.sort(key=lambda x: x[1])
    refined_seeds = [v for v, _ in enriched_evals[:top_k]]

    # --------------------------------------------------------
    # 4. Refinement of the selected seeds
    # --------------------------------------------------------
    global_best_seq = refined_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # allocate a fraction of the remaining budget to each seed
    remaining_total = deadline - time.time()
    # weights: give more budget to higher‑ranked seeds
    weight_vec = np.array([0.4, 0.25, 0.15, 0.10, 0.07, 0.03])
    weight_vec = weight_vec[:len(refined_seeds)]
    weight_vec = weight_vec / weight_vec.sum()

    for idx, seed_vec in enumerate(refined_seeds):
        if time.time() >= deadline:
            break
        slice_time = (deadline - time.time()) * weight_vec[idx]
        sub_deadline = time.time() + slice_time
        print(f"\n[Seed {idx+1}] allocated time ≈ {slice_time:.1f}s")

        # --- short Adam smoothing (fast early improvement) ---
        adam_deadline = time.time() + min(20.0, slice_time * 0.4)
        seq_adam, val_adam = _adam_smooth_opt(seed_vec, adam_deadline,
                                              iters_per_beta=12_000)

        # --- exact EG polish (uses the rest of the slice) ---
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_adam, sub_deadline,
                                               max_iters=800_000,
                                               eta0=8.0)
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # pick the better of the two refinements
        if val_adam < val_eg:
            local_seq, local_val = seq_adam, val_adam
        else:
            local_seq, local_val = seq_eg, val_eg

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------
    # 5. Final long EG polishing (use whatever time remains)
    # --------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=2_200_000,
                                                 eta0=8.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------
    # 6. Very short permutation tweak (may shave a bit more)
    # --------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + min(20.0, deadline - time.time())
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # --------------------------------------------------------
    # 7. Final evaluation with the provided function
    # --------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
