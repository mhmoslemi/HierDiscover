def _arcsine_integrated(n):
    i = np.arange(n+1, dtype=np.float64)
    # compute cdf values at endpoints: cdf_i = (2/π) * arcsin(sqrt(i/n))
    # Actually endpoints for i from 0..n, inclusive.
    lo = i[:-1] / n
    hi = i[1:] / n
    cdf_lo = (2.0/np.pi) * np.arcsin(np.sqrt(lo))
    cdf_hi = (2.0/np.pi) * np.arcsin(np.sqrt(hi))
    diff = cdf_hi - cdf_lo
    # clip to avoid zero due to rounding
    diff = np.maximum(diff, 0.0)
    return diff
```

def _subgradient_and_max(seq):
    conv = _conv_fft(seq, seq)
    max_c = conv.max()
    # tolerance: 1e-15 relative?
    eps = 1e-12
    active = np.where(conv >= max_c - eps)[0]
    # gradient
    n = len(seq)
    g = np.zeros_like(seq)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        g[idx] += 2.0 * seq[k - idx]
    return g, max_c

"""
Non‑negative step‑function optimiser (improved).

The optimiser builds a high‑quality seed – the **exact arcsine law**
discretised by integrating the continuous CDF on each bin (this is the
optimal continuous distribution for the problem).  From this seed we run a
mirror‑descent on the true max‑convolution (sub‑gradient of 
max_i (a*a)_i).  The update is performed in log‑space (exponentiated‑gradient)
so the simplex constraint is satisfied automatically.  A simple
random‑swap refinement is applied at the end.  The whole procedure respects
the 1000 s budget and stores the best sequence in the global variable
``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (real inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Entropy‑mirror (softmax) – guarantees a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # should never happen, but protect against division by zero
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq)).
    ``seq`` must already sum to 1 (probability vector).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _arcsine(n: int) -> np.ndarray:
    """Mid‑point discretisation of the arcsine law (rough version)."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _arcsine_integrated(n: int) -> np.ndarray:
    """
    Exact arcsine law, discretised **by integrating the CDF** on each bin.
    The continuous pdf is 1/(π sqrt(x(1‑x))) on (0,1), with CDF
        F(x) = (2/π) * arcsin(sqrt(x)).
    The probability of bin i (width 1/n) is F((i+1)/n) – F(i/n).
    """
    # bin edges in [0,1]
    edges = np.linspace(0.0, 1.0, n + 1, dtype=np.float64)
    lo = edges[:-1]
    hi = edges[1:]
    # CDF values
    cdf_lo = (2.0 / np.pi) * np.arcsin(np.sqrt(lo))
    cdf_hi = (2.0 / np.pi) * np.arcsin(np.sqrt(hi))
    a = cdf_hi - cdf_lo
    # Numerical safety
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        # fallback – uniform
        return _uniform(n)
    a /= a_sum
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) discretised on bin centres.
    For α=0.5 this coincides with the arcsine law (mid‑point version).
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) minimising
    2·n·max(conv(a,a)).
    Uses the exact sub‑gradient of the max‑convolution.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # stay in the simplex via softmax
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # pre‑allocate the weight vector for the gradient (length = 2·n‑1)
    w = np.zeros(2 * n - 1, dtype=np.float64)

    it = 0
    while time.time() < deadline:
        # ---- current convolution and active set ----
        conv = _conv_fft(a, a)                # length 2·n‑1
        max_c = conv.max()
        # indices where the max is attained (within a tiny tolerance)
        tol = max_c * 1e-12 + 1e-15
        active = np.where(conv >= max_c - tol)[0]

        # ---- sub‑gradient of max(conv) ----
        # w holds a weight 1 for every active index
        w.fill(0.0)
        w[active] = 1.0
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)           # length = n + (2·n‑1) – 1 = 3·n‑2
        g = 2.0 * full[n - 1:n - 1 + n]       # gradient of Σ_active conv

        # ---- mirror‑descent update (log‑space) ----
        it += 1
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        # ---- occasional exact evaluation ----
        if it % 200 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Subgrad] iter {it:6d}, best surrogate = {best_val:.6f}")

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_val, best_seq = final_val, a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight greedy local refinement
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer walk on the simplex.
    Only strictly improving moves are kept.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises the provided ``evaluate_sequence`` function.

    The routine:
        1. builds a diverse pool of seeds (including the exact integrated
           arcsine law, which is near‑optimal);
        2. keeps the best seed according to the exact surrogate;
        3. refines it with a long sub‑gradient mirror‑descent;
        4. applies a final greedy random local search;
        5. evaluates the true objective and stores the result in the global
           variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a tiny safety margin
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – we deliberately use a larger n than the default 4000
    # because the asymptotic optimum of 2·n·max(conv) is ≈ 1.5 and a larger
    # n brings us closer to that limit.
    # ------------------------------------------------------------------
    n_requested = int(kwargs.get("n", 4000))
    n = max(8000, n_requested)          # at least 8000 bins

    # ------------------------------------------------------------------
    # 0️⃣ Load a previously stored solution (if any) as a possible seed
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
            else:
                # if length differs we simply ignore it – a new seed will be built
                prev_vec = None
        except Exception:
            prev_vec = None
    else:
        prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a diverse seed pool – all vectors are normalised to sum 1
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _triangular(n),
        _arcsine(n),               # midpoint version (fallback)
        _arcsine_integrated(n),    # exact CDF‑based arcsine – best known seed
    ]

    # Dirichlet draws (different concentration levels)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # A handful of symmetric Beta densities around the optimal α=0.5
    for a in (0.45, 0.48, 0.50, 0.52, 0.55):
        seeds.append(_beta_distribution(n, a))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Initial evaluation – keep the best seed
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.clip(vec, 0.0, None)   # enforce non‑negativity
        vec = vec / vec.sum()           # normalise to a probability vector
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Sub‑gradient mirror descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Sub‑gradient mirror descent (exact max) ---")
        # allocate roughly half of the remaining budget
        remaining = deadline - time.time()
        sub_deadline = time.time() + 0.45 * remaining
        seq_sub, val_sub = _subgradient_descent(best_seq, n, sub_deadline,
                                                eta0=5.0)
        print(f"[Subgrad] best surrogate after descent = {val_sub:.6f}")
        if val_sub < best_val:
            best_val, best_seq = val_sub, seq_sub.copy()
            print("   >>> Updated global best (sub‑gradient)")

    # ------------------------------------------------------------------
    # 4️⃣ Greedy random local refinement (tiny transfers)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Final greedy local refinement ---")
        # number of attempts adapts to remaining time
        time_left = deadline - time.time()
        # empirically about 0.5 ms per attempt (including FFT)
        approx_attempts = int(time_left / 0.0005)
        approx_attempts = max(100_000, min(500_000, approx_attempts))
        seq_loc, val_loc = _local_random_search(best_seq, n, deadline,
                                                attempts=approx_attempts,
                                                sigma=0.02)
        print(f"[Local] best surrogate after local search = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 5️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None

        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation (if any)
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
