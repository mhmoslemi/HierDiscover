"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. **Initial pool** – reuse any previous solution and a handful of
   analytic families (uniform, exponentials, power‑laws, beta‑shapes, etc.).
2. **Exploration** – random‑sample a large number of *shifted beta*
   and *shifted power‑law* families (≈2 000 candidates).  Each candidate
   is ℓ₁‑normalised (sum = 1) and evaluated by the cheap surrogate  

        S(a) = 2·n·max(conv(a,a))          (exact for Σa = 1)

   The best ``TOP_K`` vectors are kept.
3. **Refinement** – run a long exact exponentiated‑gradient (mirror‑descent)
   optimisation on the best candidate.  The optimiser works in the
   log‑softmax space, guaranteeing non‑negativity and ℓ₁‑normalisation.
4. **Final polishing** – a short permutation‑based simulated‑annealing
   pass (swap two entries) may shave a little more.
5. **Evaluation** – call the provided ``evaluate_sequence`` on the final
   vector, store it in ``height_sequence_1`` and return it.

The search is fully time‑bounded (default 1000 s).  All heavy work is
performed with NumPy + FFT, which is fast enough to evaluate thousands
of candidates within the budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT convolution (real, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (seq is assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^{-p} , i = 0…n‑1, δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shifted(n: int, alpha: float, beta: float, delta: float) -> np.ndarray:
    """
    a_i ∝ (i+δ)^{α-1} (n‑1‑i+δ)^{β-1},
    a discretised Beta(α,β) density with a positive shift δ.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (alpha - 1.0) * (n - 1 - i + delta) ** (beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in log‑softmax space, thus maintains non‑negativity
    and Σa = 1 automatically.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # initialise log‑probabilities
    log_a = np.log(a + 1e-30)

    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and index of the active diagonal (the current max)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of the active entry (2·a_{max_idx - i})
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Diminishing stepsize
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g

        # re‑exponentiate and renormalise (log‑softmax)
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Pure permutation optimiser – swaps two entries."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        accept = val < cur_val or (rng.random() < math.exp(-(val - cur_val) / T))
        if accept:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget (default 1000 s) and stores the
    best vector in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 4.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – keep 4000 (good trade‑off)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load any previous solution (if present) – useful for warm restarts
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                total = prev_vec.sum()
                if total > 0:
                    prev_vec = prev_vec / total
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build an initial diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _exp_decay(n, rate=0.0015),
        _exp_decay(n, rate=0.0035),
        _sym_exp_decay(n, rate=0.0025),
        _power_shift(n, p=0.55, delta=1.0),
        _beta_shifted(n, alpha=0.5, beta=0.5, delta=1.0),
    ]

    # add a few Dirichlet draws
    for conc in (0.02, 0.08, 0.2, 0.6, 1.0, 2.5):
        seeds.append(_random_dirichlet(n, conc, rng))

    # add previous solution(s) if available
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(3):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of the handcrafted seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    # ------------------------------------------------------------------
    # 3. Random exploration of analytic families (shifted beta & power‑law)
    # ------------------------------------------------------------------
    TOP_K = 12                     # keep the best K vectors from random search
    best_candidates = []           # list of (val, vec) sorted increasingly

    def consider(vec: np.ndarray):
        val = _objective_surrogate(vec)
        if len(best_candidates) < TOP_K:
            best_candidates.append((val, vec))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if val < best_candidates[-1][0]:
                best_candidates[-1] = (val, vec)
                best_candidates.sort(key=lambda x: x[0])

    # allocate roughly half of the total budget for random exploration
    explore_deadline = time.time() + (deadline - time.time()) * 0.48

    while time.time() < explore_deadline:
        # pick a family at random
        r = rng.random()
        if r < 0.45:
            # shifted power‑law
            p = rng.uniform(0.44, 0.66)                     # fine grid around 0.55
            delta = 10 ** rng.uniform(-2, 0.5)               # 0.01 … ~3.16
            vec = _power_shift(n, p, delta)
        elif r < 0.90:
            # shifted beta (allows asymmetry)
            alpha = rng.uniform(0.35, 1.15)
            beta = rng.uniform(0.35, 1.15)
            delta = 10 ** rng.uniform(-2, 0.5)
            vec = _beta_shifted(n, alpha, beta, delta)
        else:
            # random Dirichlet (broad exploration)
            conc = rng.uniform(0.02, 3.0)
            vec = _random_dirichlet(n, conc, rng)

        consider(vec)

    # Merge handcrafted seeds into the candidate list
    for vec, val in evaluated:
        consider(vec)

    # ------------------------------------------------------------------
    # 4. Pick the best candidate from the pool
    # ------------------------------------------------------------------
    best_candidates.sort(key=lambda x: x[0])
    best_seq, best_val = best_candidates[0][1].copy(), best_candidates[0][0]
    print("\n--- Exploration summary ---")
    for i, (v, _) in enumerate(best_candidates[:TOP_K]):
        print(f"  Rank {i+1}: surrogate = {v:.6f}")

    print(f"\n[Init] best surrogate after exploration = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Refinement via exact exponentiated‑gradient (mirror‑descent)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # allocate the majority of the remaining budget to EG
        eg_deadline = deadline - 2.0   # keep a couple of seconds for final polishing
        print("\n--- EG refinement (remaining time) ---")
        refined_seq, refined_val = _exp_gradient_opt(best_seq, eg_deadline,
                                                     max_iters=3_000_000,
                                                     eta0=7.0)
        if refined_val < best_val:
            best_seq, best_val = refined_seq, refined_val
            print(f"[EG] improved surrogate to {best_val:.6f}")
        else:
            print(f"[EG] no improvement (best still {best_val:.6f})")

    # ------------------------------------------------------------------
    # 6. Very short permutation‑based annealing (swap moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + min(30.0, deadline - time.time())
        print("\n--- Final permutation anneal ---")
        perm_seq, perm_val = _perm_anneal(best_seq.copy(), perm_deadline,
                                         max_iters=120_000)
        if perm_val < best_val:
            best_seq, best_val = perm_seq, perm_val
            print(f"[Perm] improved surrogate to {best_val:.6f}")
        else:
            print(f"[Perm] no improvement (best still {best_val:.6f})")

    # ------------------------------------------------------------------
    # 7. Final evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
