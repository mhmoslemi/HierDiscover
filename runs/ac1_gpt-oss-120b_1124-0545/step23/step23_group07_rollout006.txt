"""
Propose a step–function (non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Build a very diverse pool of candidate vectors:
   * classic analytic families (uniform, exponential, beta, …)
   * a *fine* grid of the shifted power‑law  a_i ∝ (i+δ)^‑p
     with p∈[0.52,0.63] (step 0.005) and δ logarithmically spaced
     in [1e‑2, 10].
   * a handful of Dirichlet draws.

2. Evaluate all candidates with the cheap surrogate
   ``2·n·max(conv(a,a))`` (the ℓ₁‑sum is normalised to 1) and keep the
   best few (top‑8).

3. For each top‑seed run two optimisation stages:
   * a short Adam optimiser on a *smooth* (log‑sum‑exp) approximation
     of the max‑convolution – this quickly lowers the objective;
   * a series of L‑BFGS‑B runs on the same smooth surrogate with an
     increasing “β” schedule (β = 1, 5, 20, 80, 320, 1280, 5120).
     The parametrisation used is the log‑softmax of the coefficients,
     i.e.  a_i = exp(u_i)/∑_j exp(u_j).  This gives a true gradient
     (no sub‑gradient) and converges faster than the original
     exponentiated‑gradient.

4. Keep the best vector seen over all seeds.

5. Final polishing:
   * a long exponentiated‑gradient (exact sub‑gradient of the true
     max‑convolution) run on the current best vector;
   * a tiny permutation‑annealing sweep (swap two entries).

6. Return the best vector (the global name ``height_sequence_1`` is
   updated) and print progress / timings.

All heavy work is done with FFT‑based convolutions (real FFT), and the
total wall‑clock time is limited to the supplied budget (default 1000 s).
"""

import time
import math
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# FFT based full linear convolution (real vectors)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact once the ℓ₁‑sum is 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (seq is assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^{-p}, i = 0…n‑1, δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 15_000) -> tuple[np.ndarray, float]:
    """Adam on a smoothed max‑convolution (log‑sum‑exp)."""
    if beta_schedule is None:
        beta_schedule = [1, 5, 20, 80, 320, 1280, 5120]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities (log‑softmax)
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            # full convolution a*a
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp        # soft‑max weights (length = 2n‑1)

            # gradient wrt a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)   # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient wrt log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            # keep logs bounded (avoid overflow)
            log_a = np.clip(log_a, -30.0, 30.0)

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Global placeholder used by the L‑BFGS objective
# ----------------------------------------------------------------------
CURRENT_BETA = None   # will be set before each L‑BFGS call


def _lbfgs_obj(log_u: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Return (objective, gradient) for the smooth max‑convolution
    with the current value of CURRENT_BETA.
    Parameters are log‑softmax weights: a = softmax(log_u).
    """
    beta = CURRENT_BETA
    n = len(log_u)

    # convert to probabilities (stable softmax)
    max_log = np.max(log_u)
    exp_shift = np.exp(log_u - max_log)
    a = exp_shift / exp_shift.sum()

    # convolution a*a
    conv = _conv_fft(a, a)

    # smooth max via log‑sum‑exp (stable w.r.t. beta)
    max_conv = np.max(conv)
    # weights w_i = exp(beta*(conv_i - max_conv)) / Z
    w_raw = np.exp(beta * (conv - max_conv))
    Z = np.sum(w_raw)
    w = w_raw / Z                     # shape (2n‑1)

    # smooth max value
    smooth_max = max_conv + np.log(Z) / beta
    obj = 2.0 * n * smooth_max        # ℓ₁‑sum of a is 1

    # gradient wrt a :   ∂smooth_max/∂a = 2 * conv_rev(w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)       # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule: ∂obj/∂log_u = a * (grad_a - (grad_a·a))
    dot = np.dot(grad_a, a)
    grad_log = a * (grad_a - dot)

    return obj, grad_log


def _lbfgs_optimize(start_seq: np.ndarray,
                    beta_schedule,
                    deadline: float,
                    max_iter_per_stage: int = 60) -> tuple[np.ndarray, float]:
    """
    L‑BFGS optimiser on the smooth surrogate.
    start_seq must be ℓ₁‑normalised (sum = 1, all ≥0).
    """
    global CURRENT_BETA
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    # initialise log‑parameters (softmax)
    log_u = np.log(best_seq + 1e-30)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        CURRENT_BETA = beta

        # SciPy minimise expects a callable returning (f, grad)
        res = minimize(_lbfgs_obj,
                       log_u,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter_per_stage,
                                'disp': False})
        log_u = res.x
        # recover a from log_u
        max_log = np.max(log_u)
        exp_shift = np.exp(log_u - max_log)
        a = exp_shift / exp_shift.sum()
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

    CURRENT_BETA = None
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient optimiser (sub‑gradient of the true max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated gradient) on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # initialise log‑probabilities
    log_a = np.log(a + 1e-30)

    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and index of the active diagonal (the current max)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing stepsize
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        # re‑exponentiate and renormalise (log‑softmax)
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """Pure permutation optimiser – swaps two entries."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        accept = (val < cur_val) or (rng.random() < math.exp(-(val - cur_val) / T))
        if accept:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Refinement of the top seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget (roughly equal parts)
        remaining = deadline - time.time()
        slice_time = remaining * 0.20 / max(1, (top_k - idx))
        sub_deadline = time.time() + slice_time
        print(f"\n[Seed {idx + 1}] allocated time ≈ {slice_time:.1f}s")

        # ---- stage A: short Adam on smooth surrogate ----
        seq_adam, val_adam = _adam_smooth_opt(seed_vec, sub_deadline,
                                              iters_per_beta=10_000)

        # ---- stage B: L‑BFGS with increasing β ----
        if time.time() < sub_deadline:
            beta_sched = [1, 5, 20, 80, 320, 1280, 5120]
            seq_lbfgs, val_lbfgs = _lbfgs_optimize(seq_adam,
                                                   beta_sched,
                                                   sub_deadline,
                                                   max_iter_per_stage=50)
        else:
            seq_lbfgs, val_lbfgs = seq_adam, val_adam

        # pick the better of the two refinements
        if val_adam < val_lbfgs:
            local_seq, local_val = seq_adam, val_adam
        else:
            local_seq, local_val = seq_lbfgs, val_lbfgs

        print(f"[Seed {idx + 1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long exact EG polishing (use all remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=2_500_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation tweak (may shave a bit more)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + min(30.0, deadline - time.time())
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=100_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
