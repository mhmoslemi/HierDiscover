"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Build a large, diverse pool of analytic seeds (uniform, exponential,
   power‑law, shifted power‑law, beta, Dirichlet, plus the previous
   best solution).  A fine grid over the exponent `p` and shift `δ`
   of the shifted power‑law is explored because theory predicts this
   family to be near‑optimal.

2. Evaluate all seeds with the *surrogate* objective
        S(a) = 2·n·max(conv(a,a))   (∑a = 1)
   and keep the best few.

3. For each top seed run a two‑stage optimiser:
   * **Adam on a smooth max** – the max‑convolution is approximated by a
     soft‑max with an increasing temperature schedule.
   * **Exact exponentiated‑gradient (mirror‑descent)** – a sub‑gradient
     of the true max is taken.  Unlike the previous code we use the
     *average* sub‑gradient over **all active constraints** (all indices
     within a tiny tolerance of the maximum).  This yields a more
     informative descent direction and helps flatten the convolution.

4. A final long EG polishing run is performed on the current global
   best, using any remaining time budget.

5. A very short random local search (mass transfer between two
   coordinates) is applied as a final tweak.

All vectors are kept ℓ₁‑normalised; therefore the surrogate value equals
the true ``evaluate_sequence`` value.  The best sequence is stored in the
global variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based full linear convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))   (seq is assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^{-p}, i = 0…n‑1, δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on a smooth (log‑sum‑exp) approximation of max‑conv.
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 15_000) -> tuple[np.ndarray, float]:
    """Adam on a smooth (log‑sum‑exp) surrogate of the max‑convolution."""
    if beta_schedule is None:
        beta_schedule = [10, 50, 200, 800, 3_200, 12_800,
                         51_200, 204_800, 819_200]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            conv = _conv_fft(a, a)

            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp          # soft‑max weights

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            grad_log = a * (grad_a - np.dot(grad_a, a))

            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            log_a = np.clip(log_a, -30.0, 30.0)

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (mirror‑descent) optimiser.
# Uses *average* sub‑gradient over all active (nearly maximal) indices.
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                           deadline: float,
                           max_iters: int = 1_500_000,
                           eta0: float = 5.0,
                           act_tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *exact* max‑convolution.
    The sub‑gradient is averaged over all indices within ``act_tol`` of the
    maximal value (active set).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)   # log‑softmax parameters
    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set (within relative tolerance)
        tol = max(act_tol, max_val * 1e-12)
        active = np.where(conv >= max_val - tol)[0]

        # uniform weights over the active set
        w = np.zeros_like(conv)
        w[active] = 1.0 / max(len(active), 1)

        # gradient of the weighted sum of conv entries
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # gradient in log‑space (mirror‑descent with KL‑divergence)
        grad_log = a * (grad_a - np.dot(grad_a, a))

        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad_log
        # keep logs bounded (prevent overflow)
        log_a = np.clip(log_a, -30.0, 30.0)

        # recompute a from log‑softmax
        a = np.exp(log_a - np.max(log_a))
        a_sum = a.sum()
        if a_sum == 0.0:
            # numerical fallback – revert to best known
            a = best_seq.copy()
            log_a = np.log(a + 1e-30)
            continue
        a /= a_sum

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Very short random local search (mass transfer between two coordinates)
# ----------------------------------------------------------------------
def _random_local_search(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 120_000,
                         eps_factor: float = 0.02) -> tuple[np.ndarray, float]:
    """Hill‑climbing: move a small fraction of mass from i to j."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    n = len(a)
    rng = np.random.default_rng()

    for it in range(max_iters):
        if time.time() > deadline:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = eps_factor * a[i] * rng.random()
        if eps <= 0.0:
            continue
        ns = a.copy()
        ns[i] -= eps
        ns[j] += eps
        if ns[i] < 0.0:
            continue
        val = _objective_surrogate(ns)
        if val < best_val:
            best_val = val
            best_seq = ns.copy()
            a = ns      # accept as current state

    return best_seq, best_val

        # ---- Exact EG polish (average sub‑gradient) ----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt_multi(seq_adam, sub_deadline,
                                                    max_iters=800_000,
                                                    eta0=6.0)
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # keep the better of the two refinements
        if val_adam < val_eg:
            local_seq, local_val = seq_adam, val_adam
        else:
            local_seq, local_val = seq_eg, val_eg

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long EG polishing (use all remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt_multi(global_best_seq,
                                                       deadline,
                                                       max_iters=2_200_000,
                                                       eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short random local tweak
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + min(30.0, deadline - time.time())
        seq_last, val_last = _random_local_search(global_best_seq.copy(),
                                                  short_deadline,
                                                  max_iters=120_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final local] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
