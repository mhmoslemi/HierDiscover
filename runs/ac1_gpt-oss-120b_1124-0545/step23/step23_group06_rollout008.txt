    used_sums = set()
    positions = []
    for pos in range(n):

        if (2*pos) in used_sums:

        positions.append(pos)

"""
Sidon‑set based search for a step‑function that minimises
``evaluate_sequence``.

The convolution of a non‑negative vector a with itself satisfies
∑_s (a∗a)[s] = (∑ a_i)^2 = 1, hence the average entry of a∗a is
≈ 1/(2n).  The theoretical lower bound for the objective
2·n·max(a∗a) is therefore 1.  If the support of a is a B₂ (Sidon)
set – i.e. all sums i+j are distinct – then each convolution entry
contains at most one product a_i a_j (twice for i≠j).  With equal
weights w = 1/K on a support of size K the maximal entry is
2·w² = 2/K² and the objective becomes 4·n/K².
Thus the problem reduces to **maximising the size K of a Sidon set**
inside {0,…,n‑1}.  The combinatorial bound is
K(K+1)/2 ≤ 2n‑1 → K ≤ ⌊(√(16n‑7)‑1)/2⌋  (125 for n=4000).

The algorithm repeatedly:
  1. draws a random ordering of the indices,
  2. greedily builds a Sidon set respecting the ordering,
  3. builds the probability vector (equal weight on the set),
  4. measures the true objective (via FFT convolution).

The search runs until the time budget expires or an objective below
1.5 (the current target) is found.  The best vector is stored in the
global ``height_sequence_1`` variable and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast linear convolution using real FFT.
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Greedy construction of a B₂ (Sidon) set for a given ordering.
# ----------------------------------------------------------------------
def _build_sidonic_sequence(
    n: int,
    max_K: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray | None, int, float]:
    """
    Try one random ordering and greedy construction of a B₂ set.
    Returns (sequence, K, objective) where objective = 2·n·max(conv).
    If no element can be added, returns (None, 0, inf).
    """
    order = np.arange(n, dtype=np.int32)
    rng.shuffle(order)

    # bool array for used sums: index s corresponds to sum == s (0 … 2n‑2)
    used = np.zeros(2 * n - 1, dtype=bool)
    positions: list[int] = []

    for pos_np in order:
        pos = int(pos_np)

        # self‑sum already used → cannot take this index
        if used[2 * pos]:
            continue

        # check pairwise sums with the current set
        conflict = False
        for p in positions:
            if used[p + pos]:
                conflict = True
                break
        if conflict:
            continue

        # accept the position
        positions.append(pos)
        used[2 * pos] = True
        for p in positions[:-1]:
            used[p + pos] = True

        if len(positions) >= max_K:
            break

    K = len(positions)
    if K == 0:
        return None, 0, float("inf")

    # build the probability vector (equal weight on each selected index)
    seq = np.zeros(n, dtype=np.float64)
    w = 1.0 / K
    seq[positions] = w

    # exact objective using convolution
    conv = _conv_fft(seq, seq)
    max_conv = float(np.max(conv))
    obj = 2.0 * n * max_conv
    return seq, K, obj


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The search builds large Sidon sets (maximising the support size K) and
    returns the best probability vector discovered within the time budget.
    """
    start_time = time.time()
    # small safety margin so we do not overrun the allotted 1000 s
    deadline = start_time + budget_s - 2.0

    n = 4000                                  # problem size used in earlier runs
    # theoretical maximum size of a B₂ set inside {0,…,n‑1}
    max_K = int((math.isqrt(16 * n - 7) - 1) // 2)   # =125 for n=4000
    # size sufficient to beat the target 1.503 (4·n/K² < 1.5 ⇒ K≥104)
    target_K = 104

    rng = np.random.default_rng(seed)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    best_K = 0

    # One deterministic trial (ascending order) – cheap and often good.
    # ------------------------------------------------------------------
    seq_det, K_det, val_det = _build_sidonic_sequence(
        n, max_K, np.random.default_rng(seed + 1)
    )
    if seq_det is not None and val_det < best_val:
        best_seq, best_K, best_val = seq_det, K_det, val_det
        print(f"[Deterministic] K={K_det}, surrogate={val_det:.6f}")

    # ------------------------------------------------------------------
    # Randomised greedy search.
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        seq, K, val = _build_sidonic_sequence(n, max_K, rng)
        if seq is None:

        # keep the best (smaller objective) – also record larger K
        if K > best_K or val < best_val:

    # ------------------------------------------------------------------

        best_val = 2.0 * n * np.max(_conv_fft(best_seq, best_seq))
        best_K = n
        print("Fallback to uniform sequence.")

    # Final verification with the official evaluator.
    # ------------------------------------------------------------------
    try:

            f"True evaluate_sequence   = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible reuse in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
