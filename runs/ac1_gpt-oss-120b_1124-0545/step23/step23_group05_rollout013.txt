"""
Advanced step‑function optimizer (fresh approach)

The new optimizer differs from the reference implementation in several ways:

* A very cheap random‑search phase over a wide family of parametric shapes
  (power‑laws, shifted power‑laws, double‑power, beta, Gaussian,
  exponential‑mixture, logistic, Dirichlet draws, …).  The 20 best
  candidates (according to the exact surrogate 2·n·max(conv)) are kept.
* Each of these top seeds is refined by a fast *max‑convolution*
  sub‑gradient descent (exponentiated‑gradient on the simplex).  The
  gradient is the exact sub‑gradient of the max‑convolution, not a smooth
  surrogate.
* A smooth‑max L‑BFGS‑B stage (log‑sum‑exp) follows the EG step,
  improving convergence when the objective is close to non‑smooth.
* A “flattening’’ gradient stage minimises the squared deviation of the
  whole convolution from its average – this drives the convolution to a
  flat profile.
* Finally a short simulated‑annealing polish is performed.
* Time is dynamically split between the phases; the last remaining
  seconds are spent on an intensive flattening of the current global
  best.

All iterates stay on the probability simplex, so the final objective
(2·n·max(conv)) is invariant to scaling.  The routine respects the
1000 s wall‑clock budget, stores the best sequence in the global variable
``height_sequence_1`` and returns it as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(a) + len(b) - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vector must be ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))  (assumes sum(seq)=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i^{‑p}  (i = 1 … n)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a / a.sum()


def _power_shifted(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i + shift)^{‑p}."""
    idx = np.arange(1, n + 1, dtype=np.float64) + shift
    a = idx ** (-p)
    return a / a.sum()


def _double_power(n: int, p: float, q: float, shift1: float, shift2: float) -> np.ndarray:
    """a_i ∝ (i+shift1)^{‑p}·(n‑i+1+shift2)^{‑q}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = (i + shift1) ** (-p)
    right = (n - i + 1 + shift2) ** (-q)
    a = left * right
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    return a / a.sum()


def _mixed_exponential(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    e1 = np.exp(-r1 * i)
    e2 = np.exp(-r2 * i)
    a = (1.0 - alpha) * e1 + alpha * e2
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_with_uniform(base: np.ndarray, alpha: float) -> np.ndarray:
    return _uniform(len(base)) * (1.0 - alpha) + base * alpha


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _exp_gradient_max(conv_start: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 200_000,
                      eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient on the *exact* max‑convolution.
    Uses the true sub‑gradient from the currently maximal convolution entry.
    """
    a = conv_start.copy()
    # work in log‑space to stay on the simplex
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current vector on the simplex
        a = _softmax(log_a)

        # compute convolution and active index
        conv = _conv_fft(a, a)
        k_max = int(conv.argmax())

        # sub‑gradient of max(conv) w.r.t. a
        grad = np.copy(a)                     # term a_j
        rev = k_max - np.arange(n)            # index j' = k_max - j
        mask = (rev >= 0) & (rev < n)
        grad[mask] += a[rev[mask]]            # term a_{k-j}

        # step‑size (diminishing)
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update in log‑space
        log_a = log_a - eta * grad

        # evaluate surrogate
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Quadratic “flattening’’ loss (Σ (conv – avg)²) – mirror‑descent on log‑space
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000,
                         eta0: float = 8.0) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)      # length of full convolution

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom      # = 1/denom when sum=1
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)      # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent step on the simplex
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log

        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) optimisation via L‑BFGS‑B
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500) -> np.ndarray:
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)   # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simple simulated‑annealing polishing (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 250_000,
                         start_sigma: float = 0.0015,
                         T_start: float = 0.008,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear temperature schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Full optimisation pipeline for a single seed (quick + deep phases)
# ----------------------------------------------------------------------
def _full_optimize_pipeline(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    # 1) fast EG on the exact max‑convolution
    seq_eg, val_eg = _exp_gradient_max(
        start_vec.copy(), n, deadline,
        max_iters=150_000,
        eta0=6.0,
    )

    # 2) smooth‑max L‑BFGS (optional, if SciPy is present)
    if time.time() < deadline and _opt is not None:
        seq_lbfgs = _lbfgs_smooth_optimize(seq_eg, n, deadline)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, val_eg

    # 3) flattening gradient (quadratic loss)
    if time.time() < deadline:
        seq_flat, val_flat = _flatten_gradient_opt(
            seq_lbfgs, n, deadline,
            max_iters=250_000,
            eta0=9.0,
        )
    else:
        seq_flat, val_flat = seq_lbfgs, val_lbfgs

    # 4) simulated‑annealing polish
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            seq_flat, n, deadline, rng,
            max_attempts=200_000,
            start_sigma=0.001,
            T_start=0.006,
            T_end=5e-9,
        )
    else:
        seq_sa, val_sa = seq_flat, val_flat

    # Pick the best among all stages
    candidates = [
        (start_vec, _objective_surrogate(start_vec)),
        (seq_eg, val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_flat, val_flat),
        (seq_sa, val_sa),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence``.  The routine respects the time budget,
    stores the best found sequence in the global variable ``height_sequence_1``
    and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – moderate (large enough for a good bound)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if any) for continuity
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = prev_arr / prev_arr.sum()
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # ------------------------------------------------------------------
    # 1. Build a diverse seed set (analytic families + random parametric)
    # ------------------------------------------------------------------
    seeds = []

    # Analytic families (already normalised)
    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))
    seeds.append(_power(n, 0.7))
    seeds.append(_power(n, 1.2))

    # Additional families with random parameters
    for _ in range(150):
        typ = rng.integers(0, 7)
        if typ == 0:          # shifted power‑law
            p = rng.uniform(0.3, 2.5)
            shift = rng.uniform(0.0, 200.0)
            seeds.append(_power_shifted(n, p, shift))
        elif typ == 1:        # double power‑law
            p = rng.uniform(0.3, 2.5)
            q = rng.uniform(0.3, 2.5)
            sh1 = rng.uniform(0.0, 200.0)
            sh2 = rng.uniform(0.0, 200.0)
            seeds.append(_double_power(n, p, q, sh1, sh2))
        elif typ == 2:        # beta shape
            a = rng.uniform(0.01, 15.0)
            b = rng.uniform(0.01, 15.0)
            seeds.append(_beta_shape(n, a, b))
        elif typ == 3:        # exponential decay
            rate = 10 ** rng.uniform(-6, -0.2)   # log‑uniform
            seeds.append(_exp_decay(n, rate))
        elif typ == 4:        # Gaussian window
            sigma = rng.uniform(0.02, 0.4)
            seeds.append(_gaussian(n, sigma))
        elif typ == 5:        # logistic shape
            scale = rng.uniform(0.2, 5.0)
            mid = rng.uniform(0, n)
            seeds.append(_logistic_shape(n, scale, mid))
        elif typ == 6:        # mixture of two exponentials
            r1 = 10 ** rng.uniform(-6, -0.2)
            r2 = 10 ** rng.uniform(-6, -0.2)
            alpha = rng.random()
            seeds.append(_mixed_exponential(n, r1, r2, alpha))

    # Random Dirichlet draws (wide range of concentration)
    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Mixture with uniform (helps break symmetry)
    for base in [_inverse_sqrt(n), _power(n, 0.9), _exp_decay(n, 1e-3)]:
        for alpha in [0.1, 0.3, 0.5]:
            seeds.append(_mix_with_uniform(base, alpha))

    # Reversed copies
    seeds += [s[::-1].copy() for s in list(seeds)]

    # Include the previous best if available
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(5):
            pert = prev_arr + 0.005 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds, keep the top K
    # ------------------------------------------------------------------
    K = 20
    best_candidates = []  # list of (value, vector)

    def consider(vec):
        a = np.maximum(vec, 0.0)
        s = a.sum()
        if s == 0.0:
            return
        a = a / s
        val = _objective_surrogate(a)
        if len(best_candidates) < K:
            best_candidates.append((val, a))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if val < best_candidates[-1][0]:
                best_candidates[-1] = (val, a)
                best_candidates.sort(key=lambda x: x[0])

    for vec in seeds:
        if time.time() > deadline * 0.5:   # spend ~half time on seeding
            break
        consider(vec)

    print("\nTop seeds after random‑search phase (surrogate values):")
    for i, (val, _) in enumerate(best_candidates):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 3. Refine each top seed (dynamic budget allocation)
    # ------------------------------------------------------------------
    global_best_seq = best_candidates[0][1].copy()
    global_best_val = best_candidates[0][0]
    num_seeds = len(best_candidates)

    for idx, (seed_val, seed_vec) in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        # give each seed a fair slice of the remaining time
        remaining = deadline - time.time()
        share = remaining * 0.7 / max(1, (num_seeds - idx))
        sub_deadline = time.time() + share
        print(f"\n[Refine seed #{idx+1}/{num_seeds}] budget ≈ {share:.1f}s")
        seq_opt, val_opt = _full_optimize_pipeline(seed_vec, n, sub_deadline, rng)
        if val_opt < global_best_val:
            global_best_val = val_opt
            global_best_seq = seq_opt.copy()
            print(f"   *** New global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final intensive polishing on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final intensive polishing on global best ---")
        final_seq, final_val = _full_optimize_pipeline(
            global_best_seq, n, deadline, rng
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
