def _adam_smooth_opt_log(log_a, betas, iters_per_beta, start_deadline):
    best_log_a = log_a.copy()
    best_val = np.inf
    n = len(log_a)
    # Adam states
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr_global = 0.2 # base learning
    t = 0
    for beta in betas:
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                break
            t += 1
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # compute surrogate
            cur_val = 2 * n * shift
            if cur_val < best_val:
                best_val = cur_val
                best_log_a = log_a.copy()
            # compute w
            exp_beta = np.exp(beta * (conv - shift))
            w = exp_beta / np.sum(exp_beta)
            # gradient wrt a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]
            # gradient wrt log_a
            grad_log = a * (grad_a - np.dot(grad_a, a))
            # Adam update
            m = beta1*m + (1-beta1)*grad_log
            v = beta2*v + (1-beta2)*(grad_log**2)
            m_hat = m/(1-beta1**t)
            v_hat = v/(1-beta2**t)
            lr = lr_global/ (np.sqrt(v_hat) + eps)
            log_a -= lr * m_hat
            # clip logs to avoid overflow
            log_a = np.clip(log_a, -30, 30)
        if time.time() > deadline:
            break
    # final best log_a

"""
Improved search for a low‐value step‑function.

Algorithm
---------
1. Generate a diverse pool of ℓ₁‑normalised seeds (uniform,
   exponential / symmetric exponentials, pure/shifted power laws,
   beta‑shapes, Dirichlet draws, and a simple piecewise power law).
   Keep the 5 best according to the cheap surrogate
        s(p)=2·n·max(conv(p,p))   (∑p=1).

2. Optimise the best seed with a *continuation* scheme:
   * Adam on a smooth log‑sum‑exp approximation of the max‑convolution.
     The smoothing parameter β is increased exponentially
     (β∈[5,…,1.3·10⁶]) while each β gets many (≈30 k) Adam steps.
   * The final stage switches to a hard‑max exponentiated‑gradient
     (mirror‑descent) that directly minimises the true max‑convolution.
   * If time permits a brief permutation‑annealing sweep is performed.

   All updates are carried out in the log‑softmax space so that
   positivity and the ℓ₁‑constraint are automatically satisfied.

3. The best sequence found is stored globally as ``height_sequence_1``,
   evaluated with the provided ``evaluate_sequence`` routine and returned
   as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT convolution (real, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (seq is assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Helper: soft‑max of a log‑vector (guarantees positivity, sum=1)
# ----------------------------------------------------------------------
def _softmax(log_vec: np.ndarray) -> np.ndarray:
    x = np.exp(log_vec - np.max(log_vec))
    return x / np.sum(x)


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^{-p} , i = 0…n‑1, δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_power(
    n: int,
    p1: float,
    p2: float,
    delta1: float,
    delta2: float,
    split_ratio: float = 0.5,
) -> np.ndarray:
    """Two‑piece power law, normalised."""
    split = int(split_ratio * n)
    i = np.arange(n, dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[:split] = (i[:split] + delta1) ** (-p1)
    a[split:] = (i[split:] + delta2) ** (-p2)
    s = a.sum()
    return a / s if s > 0.0 else _uniform(n)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _adam_smooth_opt_log(
    log_a: np.ndarray,
    beta_schedule: list[float],
    iters_per_beta: int,
    deadline: float,
    lr0: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate  (1/β)·log‑sum‑exp(β·conv).
    Operates directly on log‑probabilities (soft‑max re‑normalisation).
    Returns the best log‑vector seen and its surrogate value.
    """
    n = len(log_a)
    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_log = log_a.copy()
    best_val = np.inf

    for beta in beta_schedule:
        # run a fixed number of iterations for this β (stop early if out of time)
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                return best_log, best_val

            t += 1

            a = _softmax(log_a)
            conv = _conv_fft(a, a)
            shift = np.max(conv)                     # max(conv) = current max‑conv
            cur_val = 2.0 * n * shift                # exact surrogate for monitoring
            if cur_val < best_val:
                best_val = cur_val
                best_log = log_a.copy()

            # smooth soft‑max weights over the whole convolution
            exp_beta = np.exp(beta * (conv - shift))
            w = exp_beta / np.sum(exp_beta)

            # gradient of the smooth max w.r.t. a (2·a * w convolution)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 3n‑2
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # gradient w.r.t. log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            # keep log‑values bounded to avoid overflow/underflow
            log_a = np.clip(log_a, -40.0, 40.0)

    return best_log, best_val


# ----------------------------------------------------------------------
# Exact max‑convolution optimiser (exponentiated‑gradient / mirror descent)
# ----------------------------------------------------------------------
def _exp_gradient_opt_log(
    log_a: np.ndarray,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 6.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution (hard max).
    Works in the log‑softmax space; returns best log‑vector and its
    surrogate value seen.
    """
    n = len(log_a)
    a = _softmax(log_a)
    best_log = log_a.copy()
    best_val = np.inf

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        cur_val = 2.0 * n * max_val
        if cur_val < best_val:
            best_val = cur_val
            best_log = log_a.copy()

        # sub‑gradient for the active diagonal (hard max)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        grad_log = a * (g - np.dot(g, a))

        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad_log
        a = _softmax(log_a)

    return best_log, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves)
# ----------------------------------------------------------------------
def _perm_anneal(seq: np.ndarray, deadline: float, max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Very short swap‑only simulated annealing, used as a final polishing step.
    """
    n = len(seq)
    cur_seq = seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        accept = val < cur_val or (rng.random() < math.exp(-(val - cur_val) / T))
        if accept:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

        for _ in range(2):

                for d2 in (0.5, 2.0):

    # include the previous solution if present

    # ------------------------------------------------------------------

    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Optimisation of the best seed (most of the remaining budget)
    # ------------------------------------------------------------------
    # start from the best seed
    best_start = top_seeds[0].copy()
    log_best = np.log(best_start + 1e-30)

    # continuation schedule – smoothly increase β
    beta_schedule = [
        5.0,
        20.0,
        80.0,
        320.0,
        1280.0,
        5120.0,
        20480.0,
        81920.0,
        327680.0,
        1_310_720.0,
    ]

    # allocate ~85 % of the remaining budget to smooth Adam,
    # the rest will be used for the exact EG + a tiny permutation tweak.
    remaining = deadline - time.time()
    smooth_deadline = time.time() + 0.84 * remaining
    print(f"\n[Phase 1] Smooth Adam optimisation, target deadline in {smooth_deadline - time.time():.1f}s")

    # a modest number of iterations per β; the loop will break early if out of time
    iters_per_beta = 30_000
    log_best, smooth_val = _adam_smooth_opt_log(
        log_best,
        beta_schedule=beta_schedule,
        iters_per_beta=iters_per_beta,
        deadline=smooth_deadline,
        lr0=0.12,
    )
    # translate back to probabilities for monitoring
    prob_best = _softmax(log_best)
    print(f"[Phase 1] Best surrogate after smoothing = {_objective_surrogate(prob_best):.6f}")

    # ------------------------------------------------------------------
    # 4. Exact max‑convolution polishing (exponentiated‑gradient)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        eg_deadline = time.time() + 0.93 * remaining
        print(f"\n[Phase 2] Exact max‑convolution EG polishing, deadline in {eg_deadline - time.time():.1f}s")
        log_best, eg_val = _exp_gradient_opt_log(
            log_best,
            deadline=eg_deadline,
            max_iters=3_000_000,
            eta0=6.0,
        )
        prob_best = _softmax(log_best)
        print(f"[Phase 2] Best surrogate after EG = {_objective_surrogate(prob_best):.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation tweak (might shave a little more)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0:
        perm_deadline = time.time() + min(20.0, remaining)
        print("\n[Phase 3] Final permutation annealing")
        seq_last, val_last = _perm_anneal(prob_best.copy(), perm_deadline, max_iters=80_000)
        if val_last < _objective_surrogate(prob_best):
            prob_best = seq_last
            print(f"[Phase 3] Improved surrogate to {val_last:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(prob_best.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {_objective_surrogate(prob_best):.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = prob_best.tolist()
    return prob_best.tolist()
