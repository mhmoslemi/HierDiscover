def _adam_opt(start_seq, deadline, max_iters=200000, lr=0.1):
    a = start_seq.copy()
    n = len(a)
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    # we define beta schedule: increase linearly from 10 to 200000 across max_iters
    beta_start = 10.0
    beta_end = 200000.0
    while it < max_iters and time.time() < deadline:
        it += 1
        # adjust beta proportionally
        beta = beta_start + (beta_end - beta_start) * (it / max_iters)
        # compute conv
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        # weight exponentials
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        w = exp_beta / sum_exp
        # gradient w.r.t a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]
        # gradient w.r.t log_a
        grad_log = a * (grad_a - np.dot(grad_a, a))
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step
        # clip log_a
        log_a = np.clip(log_a, -30.0, 30.0)
        # project to simplex via softmax
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()
        # evaluate surrogate occasionally
        if it % 1000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                # optionally print progress
                # print(f"  Adam iter {it}, val {cur_val:.6f}")
    return best_seq, best_val

"""
Improved local‑search for the “max‑convolution” step‑function problem.

The algorithm works with probability vectors (non‑negative, sum = 1) and
optimises the smooth log‑sum‑exp surrogate of the objective
    f(a) = 2·n·max_k (a*a)[k]
using a continuation scheme.  For each candidate we run a long
L‑BFGS optimisation on increasingly sharp surrogates, followed by a
high‑precision Adam‑style mirror‑descent (which works directly in
log‑softmax space).  The best result is polished with a tiny
permutation‑annealing phase and finally evaluated with the provided
``evaluate_sequence`` function.

The code respects the 1000 s budget, prints progress, and stores the
best sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Global deadline (set in propose_candidate)
# ----------------------------------------------------------------------
_global_deadline = None


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (efficient for our sizes)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) (seq assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for seeding – always return a normalised probability vector.
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    return a / a.sum()


def _sin_shape(n: int) -> np.ndarray:
    """A half‑wave sin shape (always non‑negative)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a = np.maximum(a, 0.0)
    return a / a.sum()


# ----------------------------------------------------------------------
# Smoothed loss (log‑sum‑exp) and its gradient in log‑softmax space.
# ----------------------------------------------------------------------
def _smooth_loss_and_grad(log_a: np.ndarray, beta: float):
    """
    Returns (loss, grad_log) where loss = 2·n·M_β and
    grad_log = ∂loss/∂log_a  (log‑softmax parameters).
    """
    max_log = np.max(log_a)
    exp_shift = np.exp(log_a - max_log)          # soft‑max numerator
    a = exp_shift / exp_shift.sum()              # probability vector

    conv = _conv_fft(a, a)
    shift = np.max(conv)

    # soft‑max over the convolution entries
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = np.exp(0.0) + np.sum(exp_beta) - np.max(exp_beta)  # ensure numerical safety
    sum_exp = np.sum(exp_beta)
    w = exp_beta / sum_exp                       # weights, sum to 1

    # gradient w.r.t. a (2 * rev_a * w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    n = len(a)
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # gradient w.r.t. log‑softmax parameters
    grad_log = a * (grad_a - np.dot(grad_a, a))

    loss = 2.0 * n * (shift + np.log(sum_exp) / beta)
    return loss, grad_log


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smoothed surrogate (continuation).
# ----------------------------------------------------------------------
class _LbfgsObjective:
    """Callable for scipy.optimize.minimize: returns (value, gradient)."""

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, log_a: np.ndarray):
        loss, grad = _smooth_loss_and_grad(log_a, self.beta)
        return loss, grad


def _lbfgs_opt(start_seq: np.ndarray,
               beta_schedule,
               max_iter_per_beta: int,
               deadline: float):
    """
    Continuation L‑BFGS: sequentially solves the smoothed problem for
    increasing β.  Returns the best probability vector found.
    """
    log_a = np.log(start_seq + 1e-30)
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break

        obj = _LbfgsObjective(len(start_seq), beta)
        opts = {"maxiter": max_iter_per_beta, "disp": False}
        try:
            res = minimize(fun=obj,
                           x0=log_a,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception:
            break

        log_a = res.x
        # back‑project to probability vector
        exp_shift = np.exp(log_a - np.max(log_a))
        a = exp_shift / exp_shift.sum()
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam‑style mirror descent in log‑softmax space.
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              deadline: float,
              max_iters: int = 500_000,
              lr: float = 0.2,
              beta_start: float = 10.0,
              beta_end: float = 200_000.0):
    """
    Adam optimiser on the log‑softmax parameters using a linearly
    increasing β (continuation).  Returns the best probability vector found.
    """
    a = start_seq.copy()
    n = len(a)
    log_a = np.log(a + 1e-30)

    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # linear schedule for β
        beta = beta_start + (beta_end - beta_start) * (it / max_iters)

        # compute gradient of the smoothed loss
        loss, grad_log = _smooth_loss_and_grad(log_a, beta)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # keep log‑softmax bounded (prevents overflow)
        log_a = np.clip(log_a, -30.0, 30.0)

        # back‑project to simplex
        exp_shift = np.exp(log_a - np.max(log_a))
        a = exp_shift / exp_shift.sum()

        # occasional evaluation of the true surrogate (every 2000 iters)
        if it % 2000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap permutation annealing (swap moves only).
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 100_000):
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        # linear temperature schedule
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – called by the benchmark harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The pipeline:
        1. build a diverse pool of analytic / random seeds;
        2. keep a few promising candidates (surrogate);
        3. refine each with a continuation L‑BFGS + Adam;
        4. polish the global best with a long Adam run;
        5. finish with a short permutation‑annealing tweak;
        6. store and return the best sequence.
    """
    global _global_deadline
    rng = np.random.default_rng(seed)
    start_time = time.time()
    _global_deadline = start_time + budget_s - 2.0   # safety margin
    deadline = _global_deadline

    # ------------------------------------------------------------------
    # Choose a problem size.  4000 is a good trade‑off for the current budget.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution if it exists (helps continuity across runs).
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.size > 0:
                if arr.shape[0] != n:
                    # resample by linear interpolation to the required length
                    xs_old = np.linspace(0, 1, arr.shape[0])
                    xs_new = np.linspace(0, 1, n)
                    arr = np.interp(xs_new, xs_old, arr)
                prev_vec = np.maximum(arr, 0.0)
                if prev_vec.sum() > 1e-12:
                    prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build a modest pool of seed vectors.
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _exp_decay(n, 0.0015),
        _exp_decay(n, 0.0035),
        _sym_exp_decay(n, 0.0025),
        _power(n, 0.6),
        _power(n, 1.1),
        _beta_shape(n, 0.7, 0.7),
        _beta_shape(n, 2.0, 2.0),
        _sin_shape(n),
    ]

    # Dirichlet draws with a wide range of concentration parameters.
    for conc in (0.02, 0.05, 0.15, 0.5, 1.0, 3.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random piecewise‑linear shapes.
    for nb in (5, 10):
        for _ in range(2):
            seeds.append(_piecewise_linear(n, nb, rng))

    # Incorporate the previous solution and a few permutations thereof.
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(3):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few.
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed ranking (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Local refinement: L‑BFGS continuation + Adam polish.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # parameters for the continuation schedules
    beta_schedule = [10, 80, 400, 2000, 10000, 50000, 200000]
    max_iter_per_beta = 800  # more inner iterations than the previous run

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a generous slice of the remaining budget
        remaining = deadline - time.time()
        share = remaining * 0.25 / max(1, (top_k - idx))
        sub_deadline = time.time() + share
        print(f"\n[Seed {idx+1}] allocated ≈ {share:.1f}s for refinement")

        # ----- L‑BFGS continuation -----
        seq_lbfgs, val_lbfgs = _lbfgs_opt(seed_vec,
                                          beta_schedule,
                                          max_iter_per_beta,
                                          sub_deadline)

        # ----- Adam polish (if time permits) -----
        if time.time() < sub_deadline:
            seq_adam, val_adam = _adam_opt(seq_lbfgs,
                                          sub_deadline,
                                          max_iters=200_000,
                                          lr=0.15)
        else:
            seq_adam, val_adam = seq_lbfgs, val_lbfgs

        # pick the better local result
        if val_adam < val_lbfgs:
            local_seq, local_val = seq_adam, val_adam
        else:
            local_seq, local_val = seq_lbfgs, val_lbfgs

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")
        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long Adam polishing on the remaining time budget.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long Adam polishing ---")
        final_seq, final_val = _adam_opt(global_best_seq,
                                         deadline,
                                         max_iters=500_000,
                                         lr=0.12)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation‑annealing tweak.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = time.time() + min(15.0, deadline - time.time())
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          anneal_deadline,
                                          max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
