def _smooth_max_opt(start_vec, n, deadline, max_iters=200_000, eta0=6.0, alpha_start=4.0, alpha_end=200.0):
    a = start_vec.copy()
    log_a = np.log(a + 1e-30)
    # momentum optional
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        # compute conv
        conv = _conv_fft(a, a)
        # compute smoothing parameter alpha linearly increasing
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        # compute softmax weights
        # To avoid overflow, subtract max
        maxc = np.max(alpha * conv)
        expw = np.exp(alpha * conv - maxc)
        p = expw / expw.sum()
        # gradient of smoothed max
        rev_a = a[::-1]
        grad_full = _conv_fft(rev_a, p)
        grad_a = 2.0 * grad_full[n-1:n-1+n]
        # step size
        eta = eta0 / np.sqrt(it)
        # exponentiated gradient update
        log_a -= eta * grad_a
        a = _softmax(log_a)
    smooth_val = np.max(conv)  # compute actual max
    return a, smooth_val

"""
Propose a step‑function that minimises `evaluate_sequence`.

Algorithm
---------
* Work on the probability simplex (∑a = 1) – the objective is
  scale‑invariant, so we optimise a probability vector.
* Build a modest but diverse library of seeds (uniform, beta‑U‑shapes,
  exponential, power‑law, logistic and a few Dirichlet draws).  Keep the
  best few according to the cheap surrogate  2·n·max(conv(a,a)).
* For each seed run a **smooth‑max** descent:
      – use the log‑sum‑exp smoothing of max(conv) with a temperature
        parameter α that grows from a small value to a large one.
      – gradient of the smoothed objective is obtained analytically by a
        convolution with the soft‑max weights.
      – the optimisation is performed in the log‑softmax
        (exponentiated‑gradient) parametrisation using an Adam‑style
        update, which stays on the simplex automatically.
* After the smooth‑max phase run a short **active‑set exponentiated‑gradient**
  (as in the previous baseline) to polish the exact max value.
* Optionally finish with a tiny simulated‑annealing run.
* Keep the best sequence seen over all seeds, evaluate it with the
  provided `evaluate_sequence`, store it in a global variable
  `height_sequence_2` and return it.

All heavy work (convolutions) is done with a real FFT, so each
iteration costs O(n log n).  Time‑budget handling guarantees the
function returns before the 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (real → real)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, ∑a = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) (∑seq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on the discrete set {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i⁻ᵖ (i starts from 1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smooth‑max optimiser (log‑sum‑exp smoothing of the max)
# ----------------------------------------------------------------------
def _smooth_max_opt(start_vec: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 400_000,
                    eta0: float = 5.0,
                    alpha_start: float = 4.0,
                    alpha_end: float = 250.0) -> tuple[np.ndarray, float]:
    """
    Minimises the smoothed max of the convolution using an Adam‑style
    exponentiated‑gradient on the log‑softmax parametrisation.
    Returns the best vector found and its *raw* max(conv) value.
    """
    a = start_vec.copy()
    # initialise log‑softmax parameters
    log_a = np.log(a + 1e-30)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_a = a.copy()
    best_val = np.max(_conv_fft(a, a))

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current convolution
        conv = _conv_fft(a, a)

        # temperature schedule – linearly grow α
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / max_iters)

        # soft‑max weights for the smoothed max
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        expw = np.exp(scaled - max_scaled)          # stability
        p = expw / expw.sum()

        # gradient of the smoothed objective (see derivation in docstring)
        rev_a = a[::-1]
        grad_full = _conv_fft(rev_a, p)            # length 3n‑2
        grad_a = 2.0 * grad_full[n - 1:n - 1 + n]  # slice of length n

        # Adam update on the log‑softmax parameters
        m = beta1 * m + (1.0 - beta1) * grad_a
        v = beta2 * v + (1.0 - beta2) * (grad_a * grad_a)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step‑size schedule (decays like 1/√t)
        step = eta0 / (math.sqrt(it) + 1e-12)

        log_a -= step * m_hat / (np.sqrt(v_hat) + eps)

        # back to the simplex
        a = _softmax(log_a)

        # keep the best seen vector (measured with the *true* max)
        cur_val = np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient on the exact max‑convolution (active‑set)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 800_000,
                            eta0: float = 8.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses all indices whose value is within `tol`·max of the current
    maximum (a multi‑active set).  Returns the best sequence encountered
    and its surrogate value (= 2·n·max(conv)).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of the max (average over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑like momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 80_000,
                         start_sigma: float = 0.0010,

    """Gaussian perturbation SA with a linear temperature schedule."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises the provided ``evaluate_sequence`` objective.
    The routine respects a wall‑clock budget of ``budget_s`` seconds,
    stores the best sequence in the global variable ``height_sequence_2``,
    prints progress information and finally returns the best sequence.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more freedom but also higher cost.
    # 5000 works well for the time budget.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Restore previous best if it exists (helps warm‑starts across runs)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_2")
    prev_arr = None
    if previous is not None:
        try:
            pa = np.asarray(previous, dtype=np.float64)
            if pa.shape == (n,):
                prev_arr = pa / pa.sum() if pa.sum() > 0 else None
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Build a compact, diverse seed library
    # ------------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # basic analytic families
    seeds.append(_uniform(n))

    # beta‑U‑shapes – explore very small parameters (U‑shaped)
    for a in (0.05, 0.1, 0.2, 0.5):
        for b in (0.05, 0.1, 0.2, 0.5):
            seeds.append(_beta_shape(n, a, b))

    # exponential decays (log‑spaced rates)
    for rate in np.logspace(-5, -0.5, 12):
        seeds.append(_exp_decay(n, rate))

    # power‑law tails
    for p in np.linspace(0.0, 4.0, 9):
        seeds.append(_power(n, p))

    # logistic (sigmoid) shapes – shifted to place mass at the ends
    for scale in (0.5, 1.0, 2.0):
        for mid in (n * 0.2, n * 0.8):
            seeds.append(_logistic_shape(n, scale, mid))

    # random Dirichlet draws (various concentrations)
    for conc in (0.001, 0.01, 0.1, 0.5):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # maybe keep the previous best and tiny perturbations
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(3):
            pert = prev_arr + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds, keep the K best (K=6)
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise the global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Process each seed (smooth‑max → active‑set EG → optional SA)
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a fraction of the remaining budget for this seed
        remaining = deadline - time.time()
        # roughly 15 % of the remaining time for each seed (shrinking)
        share = remaining * 0.15 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}/{top_k}] start optimisation, sub‑deadline in {share:.1f}s")

        # ----- a) smooth‑max descent (focuses directly on max(conv)) -----
        seq_smooth, max_conv = _smooth_max_opt(seed_vec,
                                                n,
                                                sub_deadline,
                                                max_iters=300_000,
                                                eta0=6.0,
                                                alpha_start=4.0,
                                                alpha_end=250.0)
        val_smooth = _objective_surrogate(seq_smooth)
        print(f"    after smooth‑max: surrogate = {val_smooth:.6f}")

        # ----- b) active‑set EG refinement (exact max) --------------------
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt_multi(seq_smooth,
                                                    n,
                                                    sub_deadline,
                                                    max_iters=600_000,
                                                    eta0=9.0,
                                                    tol=1e-9)
        else:
            seq_eg, val_eg = seq_smooth, val_smooth

        # ----- c) optional short simulated‑annealing polish -------------
        hill_deadline = sub_deadline - 0.25
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_eg,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=70_000,
                                                  start_sigma=0.0010,
                                                  T_start=0.010,
                                                  T_end=1e-8)
        else:
            seq_sa, val_sa = seq_eg, val_eg

        # pick the best among the candidates for this seed
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_smooth, val_smooth),
            (seq_eg, val_eg),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long polishing (smooth‑max + active‑set) on the best so far
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long polishing ---")
        # give the remaining time to a final smooth‑max run
        final_seq, _ = _smooth_max_opt(global_best_seq,
                                       n,
                                       deadline,
                                       max_iters=800_000,
                                       eta0=5.0,
                                       alpha_start=4.0,
                                       alpha_end=300.0)
        # finish with an active‑set EG pass
        final_seq, final_val = _exp_gradient_opt_multi(final_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=2_000_000,
                                                       eta0=5.0,
                                                       tol=1e-9)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_2"] = global_best_seq.tolist()
    return global_best_seq.tolist()
