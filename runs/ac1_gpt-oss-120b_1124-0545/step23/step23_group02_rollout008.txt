import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx*fy, n=N)[:L]
    return conv

def _softmax(v):
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

def _objective_surrogate(seq):
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _beta_shape(n, alpha, beta):
    i = np.arange(1, n+1, dtype=np.float64)
    a = (i**(alpha-1.0))*((n - i + 1)**(beta-1.0))
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s

def _exp_decay(n, rate):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s

def _power_law(n, p):
    idx = np.arange(1, n+1, dtype=np.float64)
    a = 1.0/(idx**p)
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s

def _logistic_shape(n, scale, midpoint):
    i = np.arange(n, dtype=np.float64)
    a = 1.0/(1.0 + np.exp(-(i - midpoint)/scale))
    s = a.sum()
    if s <= 0:
        return _uniform(n)
    return a / s

def _sin_shape(n):
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s

def _triangle_shape(n):
    i = np.arange(n, dtype=np.float64)
    a = np.minimum(i+1, n-i)
    s = a.sum()
    return a / s

def _parabolic_shape(n):
    i = np.arange(1, n+1, dtype=np.float64)
    a = i * (n - i + 1)
    s = a.sum()
    return a / s

def _hann_shape(n):
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n-1)))
    s = a.sum()
    return a / s

def _hamming_shape(n):

    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n-1))
    s = a.sum()
    return a / s

    a = (0.42 - 0.5 * np.cos(2*np.pi*i/(n-1)) + 0.08 * np.cos(4*np.pi*i/(n-1)))
    s = a.sum()
    return a / s

def _random_dirichlet(n, rng):
    conc = 10**rng.uniform(-3, 1)  # between 0.001 and 10
    vec = rng.dirichlet(np.full(n, conc))
    return vec.astype(np.float64)

def _adam_opt(start_seq, n, deadline,
              max_iters=200_000,
              lr=0.3,
              beta1=0.9,
              beta2=0.999,
              eps=1e-8,
              tol=1e-12,
              patience=2000):
    """Mirror‑descent with Adam on the exact max‑convolution.
    Returns (best_seq, best_surrogate)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    no_improve = 0

    idx = np.arange(n, dtype=np.int64)

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # select active indices (within relative tolerance)
        active = np.where(conv >= max_val - max(tol * max_val, 1e-15))[0]

        if active.size:
            # gradient of max (average over active)
            grad = np.zeros_like(a)
            for k in active:
                rev_idx = k - idx
                mask = (rev_idx >= 0) & (rev_idx < n)
                grad[mask] += a[rev_idx[mask]]
            grad = (2.0 / active.size) * grad
        else:
            grad = np.zeros_like(a)

        # Adam update on log‑softmax parameters
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                # stop early if we are stuck
                break

    return best_seq, best_val

"""
Heuristic optimiser for the “max‑convolution” step‑function problem.

The idea is to generate a diverse set of analytic and random seeds,
pick the most promising few (by the cheap surrogate 2·n·max(conv)),
and then minimise the exact objective with a mirror‑descent variant that
optimises the true max‑convolution (Ada‑gradient on the log‑softmax
parameters).  The optimiser works on the probability simplex, so the
scale‑invariance of the original score is respected.  A final long polish
is performed from the current best vector.  The best sequence is stored
in the global variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Core numeric utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# --------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# --------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (∑seq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# --------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ i⁻ᵖ (i starts from 1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    """Logistic (sigmoid) shape."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sin_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s


def _triangle_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.minimum(i + 1, n - i)
    s = a.sum()
    return a / s


def _parabolic_shape(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    s = a.sum()
    return a / s


def _hann_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s


def _hamming_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s


def _blackman_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s


def _random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with a random concentration parameter."""
    conc = 10 ** rng.uniform(-3, 1)   # between 0.001 and 10
    vec = rng.dirichlet(np.full(n, conc))
    return vec.astype(np.float64)


# --------------------------------------------------------------
# Adam‑based mirror‑descent on the exact max‑convolution
# --------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 200_000,
              lr: float = 0.3,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8,
              tol: float = 1e-12,
              patience: int = 2000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam on the exact max‑convolution.
    Returns the best sequence (sum = 1) encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # log‑softmax parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    no_improve = 0

    idx = np.arange(n, dtype=np.int64)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices where the convolution is (almost) maximal
        active = np.where(conv >= max_val - max(tol * max_val, 1e-15))[0]

        if active.size:
            grad = np.zeros_like(a)
            for k in active:
                rev_idx = k - idx
                mask = (rev_idx >= 0) & (rev_idx < n)
                grad[mask] += a[rev_idx[mask]]
            grad = (2.0 / active.size) * grad
        else:
            grad = np.zeros_like(a)

        # Adam update on the log‑softmax parameters
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                break

    return best_seq, best_val

    rng = np.random.default_rng(seed)
