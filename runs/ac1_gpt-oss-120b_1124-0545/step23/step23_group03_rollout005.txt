def _guided_local_search(start_seq, n, deadline, attempts=200000, sigma=0.02):
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    for it in range(attempts):
        if time.time()>deadline:
            break
        conv = _conv_fft(best_seq, best_seq)
        max_idx = int(np.argmax(conv))
        # contributions
        i_start = max(0, max_idx-(n-1))
        i_end = min(max_idx, n-1)
        idx_i = np.arange(i_start, i_end+1)
        rev_idx = max_idx - idx_i
        contrib = best_seq[idx_i] * best_seq[rev_idx]
        total = contrib.sum()
        if total <= 0:
            probs = None
        else:
            probs = contrib / total
        # choose source i weighted by contrib
        i = rng.choice(idx_i, p=probs) if probs is not None else rng.integers(i_start, i_end+1)
        # choose destination j random not equal i
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val

"""
Non‑negative step‑function optimiser (v2).

The optimiser builds a large pool of diverse seed vectors (uniform, triangular,
arcsine, many Beta‑shapes, mixtures of two Betas, Dirichlet draws) and keeps the
best one (evaluated with the exact surrogate 2·n·max(conv(p,p)) for a
probability vector p).  From this seed it runs a fast **projected gradient
descent** on the true max‑convolution (sub‑gradient for the active index) and
projects every iterate onto the probability simplex.  After the gradient phase
a guided random local walk (mass transfer from coordinates that dominate the
current peak to random destinations) is performed until the time budget is
exhausted.  The final sequence is stored in the global variable
``height_sequence_1`` and returned as a plain list.

All steps respect the external 1000 s deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(p: np.ndarray) -> float:
    """Exact surrogate used in the search:
       2 * n * max(conv(p, p)),   where p already sums to 1.
    """
    n = len(p)
    conv = _conv_fft(p, p)
    return float(2.0 * n * np.max(conv))


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density discretised on cell centres."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of two independent symmetric Beta(α,α) components."""
    a1 = rng.uniform(0.05, 2.0)
    a2 = rng.uniform(0.05, 2.0)
    w = rng.random()
    p1 = _beta_distribution(n, a1)
    p2 = _beta_distribution(n, a2)
    mix = w * p1 + (1.0 - w) * p2
    mix /= mix.sum()
    return mix

def _projected_gradient_descent(start: np.ndarray,
                                n: int,
                                deadline: float,
                                eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Sub‑gradient is taken at the active index ``k = argmax(conv)``.
    The iterate is projected onto the simplex after each step.
    """
    a = start.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size (η_t = η0 / sqrt(t))
        eta = eta0 / math.sqrt(it)
        a_tmp = a - eta * g
        a = _simplex_projection(a_tmp)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


def _guided_local_search(start: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic walk that preferentially removes mass from coordinates
    that contribute heavily to the current peak of the convolution.
    """
    rng = np.random.default_rng()
    best_seq = start.copy()
    best_val = _objective_surrogate(best_seq)

    for _ in range(attempts):
        if time.time() > deadline:
            break

        # current peak information
        conv = _conv_fft(best_seq, best_seq)
        k = int(np.argmax(conv))

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        contrib = best_seq[idx_i] * best_seq[rev_idx]
        total = contrib.sum()
        # probability proportional to contribution (avoid division by zero)
        if total > 0.0:
            probs = contrib / total
            src = rng.choice(idx_i, p=probs)
        else:
            src = rng.integers(i_start, i_end + 1)

        # random destination (different from source)
        dst = rng.integers(0, n)
        while dst == src:
            dst = rng.integers(0, n)

        # transfer a modest fraction of the source mass
        delta = sigma * best_seq[src] * rng.random()
        if delta <= 0.0 or delta > best_seq[src]:
            continue

        new_seq = best_seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta
        # safeguard non‑negativity (should hold)
        if new_seq[src] < 0.0:
            continue

        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (probability vector) that minimises
    ``evaluate_sequence``.  The function respects the provided time budget (≈
    1000 s) and stores the best sequence in the global variable
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin so that the external timeout never cuts us off
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0️⃣ Use the previously found solution as an additional seed (if any)
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a large and diverse seed pool (all vectors are normalised)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
    ]

    # a fine grid of symmetric Beta laws (including very sharp U‑shapes)
    beta_alphas = [
        0.05, 0.07, 0.10, 0.12, 0.15, 0.18, 0.20, 0.25,
        0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60, 0.70,
        0.80, 0.90, 1.00, 1.20, 1.40, 1.60, 1.80, 2.00
    ]
    for a in beta_alphas:
        seeds.append(_beta_distribution(n, a))

    # mixtures of two Beta components – a cheap way to explore richer shapes
    for _ in range(30):
        seeds.append(_beta_mixture(n, rng))

    # Dirichlet draws with a broad range of concentration parameters
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Initial evaluation of all seeds – keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_val = math.inf

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec.astype(np.float64)
        vec /= vec.sum()
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()

    if best_seq is None:
        # Fallback – uniform distribution (should never happen)
        best_seq = _uniform(n)
        best_val = _objective_surrogate(best_seq)

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Projected gradient descent on the exact max–convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        pgd_deadline = time.time() + 0.55 * remaining  # keep some time for refinement
        print("\n--- Projected gradient descent (exact max) ---")
        seq_pgd, val_pgd = _projected_gradient_descent(best_seq, n, pgd_deadline,
                                                       eta0=2.0)
        print(f"[PGD] best surrogate = {val_pgd:.6f}")
        if val_pgd < best_val:
            best_seq, best_val = seq_pgd, val_pgd
            print("   >>> Updated global best (PGD)")

    # ------------------------------------------------------------------
    # 4️⃣ Guided random local refinement (mass transfer from dominant pairs)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        # Heuristic: one attempt takes ~5e‑4 s → estimate number of attempts
        approx_attempts = int(time_left / 0.0005)
        approx_attempts = max(100_000, min(500_000, approx_attempts))
        print("\n--- Guided random local refinement ---")
        seq_loc, val_loc = _guided_local_search(best_seq, n, deadline,
                                                attempts=approx_attempts,
                                                sigma=0.02)
        print(f"[Local] best surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc, val_loc
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 5️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
