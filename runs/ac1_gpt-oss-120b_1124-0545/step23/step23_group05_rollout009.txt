"""
Propose a step‑function (non‑negative heights) that minimises the
provided ``evaluate_sequence`` function.

Algorithm
---------

* Build a *rich* set of candidate shapes – uniform, inverse‑sqrt,
  simple power laws, shifted power‑law families ``a_i ∝ (i+Δ)^{-p}``,
  mixed exponential‑power families and a few Dirichlet draws.
  All candidates are ℓ₁‑normalised (sum = 1).

* Evaluate the *exact* surrogate  2·n·max(conv(a,a))  for each candidate
  (the surrogate is identical to the true objective after the
  normalisation performed by ``evaluate_sequence``).

* Pick the best candidate and run a long local optimisation pipeline
  that respects the remaining wall‑clock budget:

    1. **Exponentiated‑gradient (EG)** on the exact max‑convolution.
       Uses an active‑set sub‑gradient and Adam‑style momentum.
    2. **Smooth‑max L‑BFGS** (log‑sum‑exp surrogate) for fine tuning.
    3. **Flattening gradient** that reduces the variance of the whole
       convolution (quadratic loss).
    4. **Simulated‑annealing** polish.

  The stages are time‑aware; they stop as soon as the global deadline
  is reached.

* The best sequence found is stored in the global variable
  ``height_sequence_1`` and returned as a plain Python list.

The implementation deliberately keeps all helper functions at the top
level (no closures, no lambdas) and works with or without SciPy
installed.  It uses at most two CPU cores (the code is single‑threaded
but respects the 1000 s wall‑clock limit).

"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# --------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# --------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (sum(seq) == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# --------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    """Classic a_i ∝ 1/√(i+1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _pow_shift(n: int, p: float, shift: float) -> np.ndarray:
    """a_i ∝ (i+shift)^{-p} – a flexible power‑law family."""
    i = np.arange(1, n + 1, dtype=np.float64) + shift
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mixed_exponential(n: int, r1: float, r2: float, alpha: float) -> np.ndarray:
    """Convex mixture of two exponentials."""
    e1 = np.exp(-r1 * np.arange(n, dtype=np.float64))
    e2 = np.exp(-r2 * np.arange(n, dtype=np.float64))
    a = (1.0 - alpha) * e1 + alpha * e2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate*i) , i=0..n-1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# --------------------------------------------------------------
# Mirror‑descent on the exact max‑convolution (multi‑active)
# --------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 5_000_000,
                            eta0: float = 15.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient on the exact max‑convolution.
    Uses all indices within a relative tolerance of the current maximum.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices where conv is within tol of the maximum
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient (average over active indices)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# --------------------------------------------------------------
# Gradient–descent that flattens the convolution (quadratic loss)
# --------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 1_200_000,
                         eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the loss Σ_k (conv_k - avg)^2 .
    The aim is to make the whole convolution as flat as possible.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # length of full convolution
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # equals 1/denom when sum=1
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# --------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp)
# --------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """
    Optimise a smooth max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space to keep the iterate on the simplex.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception:
            break

    return _softmax(v)


# --------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# --------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 600_000,
                         start_sigma: float = 0.001,
                         T_start: float = 0.01,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# --------------------------------------------------------------
# Full optimisation pipeline (single seed, full budget)
# --------------------------------------------------------------
def _full_optimisation_one_seed(start_vec: np.ndarray,
                                n: int,
                                deadline: float,
                                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run the four stages (EG → smooth L‑BFGS → flatten → SA) on *start_vec*.
    All stages respect the shared *deadline*.
    Returns the best sequence encountered and its surrogate value.
    """
    # 1) Exponentiated‑gradient on the exact max‑convolution
    seq_eg, val_eg = _exp_gradient_opt_multi(
        start_vec.copy(),
        n,
        deadline,
        max_iters=5_000_000,
        eta0=15.0,
        tol=1e-9,
    )

    # 2) Smooth L‑BFGS
    if time.time() < deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(
            seq_eg,
            n,
            deadline,
            beta_schedule=None,
            max_iter_per_beta=4000,
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, val_eg

    # 3) Flattening gradient
    if time.time() < deadline:
        seq_flat, val_flat = _flatten_gradient_opt(
            seq_lbfgs,
            n,
            deadline,
            max_iters=1_200_000,
            eta0=10.0,
        )
    else:
        seq_flat, val_flat = seq_lbfgs, val_lbfgs

    # 4) Simulated annealing polish
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            seq_flat,
            n,
            deadline,
            rng,
            max_attempts=600_000,
            start_sigma=0.0008,
            T_start=0.012,
            T_end=5e-9,
        )
    else:
        seq_sa, val_sa = seq_flat, val_flat

    # Choose the best among the intermediate results
    candidates = [
        (start_vec, _objective_surrogate(start_vec)),
        (seq_eg, val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_flat, val_flat),
        (seq_sa, val_sa),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val


# --------------------------------------------------------------
# Helper for sorting without lambda
# --------------------------------------------------------------
def _by_val(item):
    return item[1]


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    stores the best found sequence in the global variable
    ``height_sequence_1`` and returns it as a Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 3.0          # safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate but sufficiently large n
    # --------------------------------------------------------------
    n = 5000

    # --------------------------------------------------------------
    # Load previous solution if present (helps continuity)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Build a diverse seed set
    # --------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # basic analytic families
    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))

    # simple power families
    for p in [0.5, 0.6, 0.7, 0.8, 1.0]:
        seeds.append(_power(n, p))

    # shifted power‑law grid
    for p in np.linspace(0.4, 0.8, 9):
        for shift in np.linspace(0.0, 20.0, 7):
            seeds.append(_pow_shift(n, float(p), float(shift)))

    # mixed exponential‑power families
    for rate1 in np.logspace(-5, -0.5, 6):
        for rate2 in np.logspace(-5, -0.5, 6):
            if rate2 <= rate1:
                continue
            seeds.append(_mixed_exponential(n, float(rate1), float(rate2), alpha=0.3))

    # pure exponential decays
    for rate in np.logspace(-5, -0.3, 8):
        seeds.append(_exp_decay(n, float(rate)))

    # random Dirichlet draws – wide range of concentration
    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # include the previous best (if dimensions match)
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(5):
            pert = prev_arr + 0.006 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # --------------------------------------------------------------
    # Normalise all seeds and discard degenerate ones
    # --------------------------------------------------------------
    clean_seeds = []
    for s in seeds:
        s = np.maximum(s, 0.0)
        tot = s.sum()
        if tot == 0.0:
            continue
        clean_seeds.append(s / tot)

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in clean_seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        # Fallback – should never happen
        best_initial = _uniform(n)
        best_initial_val = _objective_surrogate(best_initial)
        print("[Fallback] using uniform seed")
    else:
        evaluated.sort(key=_by_val)
        best_initial, best_initial_val = evaluated[0]
        print("\nTop seed surrogate value: {:.6f}".format(best_initial_val))

    # --------------------------------------------------------------
    # 3. Intensive local optimisation on the best seed
    # --------------------------------------------------------------
    print("\n--- Full optimisation on the best seed ---")
    seq_opt, val_opt = _full_optimisation_one_seed(best_initial, n, deadline, rng)
    print("Local optimum surrogate = {:.6f}".format(val_opt))

    # --------------------------------------------------------------
    # 4. If time remains, try a short re‑optimisation from the result
    # --------------------------------------------------------------
    while time.time() < deadline - 30.0:
        print("\n--- Additional polishing stage (remaining {:.1f}s) ---".format(
            deadline - time.time()))
        seq_opt, val_opt = _full_optimisation_one_seed(seq_opt, n, deadline, rng)
        print("Polished surrogate = {:.6f}".format(val_opt))

    # --------------------------------------------------------------
    # 5. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(seq_opt.tolist())
        print("\n=== Finished ===")
        print("Surrogate ≈ {:.6f}, true evaluate_sequence = {:.6f}".format(val_opt, true_val))
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally and return
    globals()["height_sequence_1"] = seq_opt.tolist()
    return seq_opt.tolist()
