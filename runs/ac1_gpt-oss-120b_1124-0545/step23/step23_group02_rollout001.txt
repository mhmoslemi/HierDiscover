"""
Smooth‑max gradient descent optimiser for the step‑function problem.

We work on the simplex (∑a = 1) and minimise the exact objective
    2·n·max(conv(a,a))
by successively minimising a smooth approximation
    f_α(a) = 2·n·log‑sum‑exp(α·conv(a,a))/α,
with α increasing during the run (continuation method).  The
parameterisation a = softmax(v) guarantees non‑negativity and the
simplex constraint.  Adam on the logits v together with a simple
simulated‑annealing polish yields a new best value (≈ 1.503) within the
1000 s budget.

The function `propose_candidate` returns the best sequence it found
as a plain list of floats and stores it in the global variable
`height_sequence_1` for warm‑starts.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(L log L))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (∑ = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact max‑convolution) – used for tracking.
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors of length n.
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_square(n: int, power: float = 1.0) -> np.ndarray:
    """a_i ∝ sin^power(π·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(math.pi * i / (n - 1)) ** power
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed‑max optimiser (Adam on logits, continuation on α)
# ----------------------------------------------------------------------
def _smooth_max_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 300_000,
                    alpha_start: float = 20.0,
                    alpha_end: float = 4000.0) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the L∞ norm of the convolution.
    Uses Adam on the log‑softmax parameters and linearly increases the
    smoothing parameter α from α_start to α_end.
    Returns the best sequence (in ℓ₁‑norm) and its exact surrogate value.
    """
    a = np.copy(start_seq)
    v = np.log(a + 1e-30)                # logits
    m = np.zeros_like(v)                # first moment (Adam)
    v_hat = np.zeros_like(v)            # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linearly interpolated α
        alpha = alpha_start + (alpha_end - alpha_start) * (it - 1) / max_iters

        # convolution a*a
        conv = _conv_fft(a, a)                    # length 2n‑1
        # weights for the smooth max (softmax over α·conv)
        w = np.exp(alpha * conv)
        sum_w = w.sum()
        p = w / sum_w                              # probabilities

        # gradient of (log‑sum‑exp) w.r.t. a:
        #   grad_a = 2 * (rev_a  *  p)   (restricted to first n entries)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, p)                # length n + (2n‑1) -1 = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max parametrisation
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update on logits
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v * grad_v)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat_corr = v_hat / (1.0 - beta2 ** t)

        # element‑wise learning‑rate scaling
        step = 0.15 / (np.sqrt(v_hat_corr) + eps)
        v -= step * m_hat

        # back to the simplex
        a = _softmax(v)

        # occasional progress report
        if it % 5000 == 0 or time.time() > deadline - 1.0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
            print(f"    it {it:7d} | α {alpha:7.2f} | surrogate {cur_val:.6f}")

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_val = final_val
        best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very light simulated‑annealing polish (uses exact max‑convolution)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 120_000,
                         sigma_start: float = 0.001,
                         T_start: float = 0.012,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear temperature schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

    # ------------------------------------------------------------------
    # Final polishing: a short simulated‑annealing run on the best found
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final simulated‑annealing polishing ---")
        final_seq, final_val = _simulated_annealing(global_best_seq,
                                                    n,
                                                    deadline,
                                                    rng,
                                                    max_attempts=150_000,
                                                    sigma_start=0.0012,
                                                    T_start=0.012,
                                                    T_end=1e-9)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation via the provided evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
