"""
Propose a 4000‑element non‑negative step function that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Initial seed** – use the best sequence obtained in a previous run
   (global ``height_sequence_1``) if it exists, otherwise fall back to a
   simple exponential decay.  The vector is normalised to sum to 1
   (the objective is scale‑invariant).

2. **Local stochastic search** – a simulated‑annealing style walk that
   repeatedly transfers a small amount of mass from one index *i* to a
   different index *j*.  The convolution ``b = a*a`` is maintained
   incrementally:
   * ``b[i:i+n] += d_i * a`` and ``b[j:j+n] += d_j * a`` (where
     ``d_i = -δ`` and ``d_j = +δ``) add the linear part of the
     convolution update.
   * ``b[2*i] += d_i**2``, ``b[2*j] += d_j**2`` and
     ``b[i+j] += 2*d_i*d_j`` add the quadratic term.
   * After each trial the new maximum of ``b`` is examined.
   * Moves that improve the objective are always kept; worsening moves
     are accepted with probability ``exp(-(Δ)/T)`` where the temperature
     ``T`` decays linearly to 0.

   The step size δ shrinks from a few percent of the total mass to
   ``1e‑5`` during the run, allowing an aggressive exploration at the
   beginning and fine‑grained polishing later.

3. **Adam refinement** – the best vector found by the local search is
   fed to the Adam optimiser from the reference solution (optimising the
   exact max‑convolution surrogate).  Only a modest number of iterations
   are performed, respecting the remaining time.

4. **Exponentiated‑gradient polishing** – a final long run of the mirror‑
   descent routine from the reference code is applied to squeeze out a
   few extra digits.

5. The resulting vector is evaluated with the provided ``evaluate_sequence``
   function, stored in ``height_sequence_1`` for possible continuation,
   and returned.

The whole routine respects the 1000 s budget (leaving a few seconds
margin) and uses only NumPy (no external solvers).  It works with a
single thread but may use up to two CPUs if NumPy’s BLAS is multi‑threaded.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full convolution (used only for the initial state)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used for optimisation: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses (fallbacks)
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Adam optimiser (taken from the reference solution)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 300_000,
              lr0: float = 0.2,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g_a = np.zeros_like(a)
        g_a[idx_i] = 2.0 * a[rev_idx]

        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (reference implementation)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Local stochastic search (simulated annealing with incremental convolution)
# ----------------------------------------------------------------------
def _local_search(a_init: np.ndarray,
                  n: int,
                  deadline: float,
                  rng_seed: int = 0) -> tuple[np.ndarray, float]:
    """Return a vector (sum=1) that locally minimises max(conv(a,a))."""
    rng = np.random.default_rng(rng_seed)

    a = a_init.copy()
    L = 2 * n - 1
    b = _conv_fft(a, a)               # full convolution, length 2n‑1

    cur_val = 2.0 * n * np.max(b)     # current surrogate value
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    max_step = 0.05          # start moving up to 5 % of mass
    min_step = 1e-5          # final precision
    T0 = 1e-4                # start temperature for uphill moves

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0.0:
            break
        frac = elapsed / total                     # 0 → 1

        # linearly decay step size and temperature
        step = max_step * (1.0 - frac) + min_step * frac
        T = T0 * (1.0 - frac)

        # pick a source index with enough mass
        i = rng.integers(0, n)
        if a[i] <= step:
            # fallback to any index that has mass > step
            cand = np.where(a > step)[0]
            if cand.size == 0:                     # nothing to move → stop
                break
            i = int(rng.choice(cand))

        delta = min(step, a[i])                    # never go negative

        # pick a distinct destination index
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # remember old values for a quick revert
        a_i_old = a[i]
        a_j_old = a[j]

        d_i = -delta
        d_j = delta

        # ---- apply the move (linear + quadratic part) ----
        # linear part: d_i * a  and d_j * a  (a is still the old vector)
        b[i:i + n] += d_i * a
        b[j:j + n] += d_j * a

        # quadratic part:
        b[2 * i] += d_i * d_i
        b[2 * j] += d_j * d_j
        b[i + j] += 2.0 * d_i * d_j

        # update the probability vector
        a[i] = a_i_old + d_i
        a[j] = a_j_old + d_j

        # evaluate new objective
        new_max = b.max()
        new_val = 2.0 * n * new_max

        # acceptance test
        accept = False
        if new_val < cur_val:
            accept = True
        elif T > 0.0:
            prob = math.exp(-(new_val - cur_val) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # ---- revert ----
            a[i] = a_i_old
            a[j] = a_j_old

            # because ``a`` is now back to its original values we can
            # simply subtract the same contributions we added
            b[i:i + n] -= d_i * a
            b[j:j + n] -= d_j * a
            b[2 * i] -= d_i * d_i
            b[2 * j] -= d_j * d_j
            b[i + j] -= 2.0 * d_i * d_j
        # end of one iteration
    return best_a, best_val
