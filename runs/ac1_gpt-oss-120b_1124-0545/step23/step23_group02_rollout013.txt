def _smooth_max_opt(start_seq, n, deadline, max_iters=2_000_000, eta0=5.0, beta_start=0.5, beta_growth=1.001):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m_beta = beta_start
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        # smooth max weights
        maxc = conv.max()
        # compute weighted softmax
        exp_vals = np.exp(m_beta * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        rev_a = a[::-1]
        full = _conv_fft(w, rev_a)
        grad_a = 2.0 * full[n-1:n-1+n]

        # mirror descent
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # momentum (adam-ish)
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad_log
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum

        # Evaluate surrogate occasionally
        if it % 1000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
        # Update beta
        m_beta = min(m_beta * beta_growth, 2000.0)  # cap
    return best_seq, best_val

"""
Improved step‑function optimiser (≈ 1000 s budget)

Goal: minimise   2·n·max(conv(a,a)) / (∑a)²   for a non‑negative vector a.
The objective is scale‑invariant, so we work on the probability simplex
(∑a = 1).  The search combines three ideas that were missing or only
lightly used in the earlier version:

1.  A **large, diversified seed set** – besides the classic uniform,
    beta, exponential, power‑law and logistic families we also add
    Slepian (DPSS) windows (if SciPy is present) and a massive random
    sweep of the beta parameters (log‑uniform).  This uncovers shapes
    that the hand‑picked grid missed.

2.  A **smoothing‑max optimiser** – instead of a sub‑gradient on the
    hard max we minimise a differentiable approximation
        M_β(a)=β⁻¹·log∑ₖexp(β·convₖ(a,a))
    whose gradient is a weighted sum of all convolution entries.
    Using a mirror‑descent (soft‑max) parametrisation with momentum
    yields faster descent towards flatter auto‑convolutions.

3.  A **time‑aware schedule** – the total wall‑clock budget is split
    between seed generation, a quick L²‑flattening (L‑BFGS on the
    constant‑convolution loss) and a long polishing phase with the
    smoothing‑max optimiser.  The remaining time is spent on a brief
    simulated‑annealing polish.

All updates stay on the simplex via a log‑softmax re‑parameterisation.
The best sequence seen is stored globally as ``height_sequence_1`` and
returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (DPSS windows & L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

try:
    from scipy.signal.windows import dpss as _dpss
    _has_dpss = True
except Exception:  # pragma: no cover
    _has_dpss = False

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, ∑ = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) (∑seq = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ i⁻ᵖ (i starts from 1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_shape(n: int, nw: float) -> np.ndarray:
    """Slepian / DPSS window (sym=False for full length)."""
    w = _dpss(n, nw, Kmax=1, sym=False).real
    w = np.maximum(w, 0.0)
    s = w.sum()
    return w / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the flat‑convolution loss
# ----------------------------------------------------------------------
def _lbfgs_flatten_opt(start_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iter: int = 2000) -> np.ndarray:
    """Minimise ∑(conv‑avg)² via L‑BFGS on log‑softmax parameters."""
    if _opt is None:
        # fall back to a tiny mirror‑descent (same as in the reference code)
        a = start_vec.copy()
        log_a = np.log(a + 1e-30)
        best_a = a.copy()
        best_val = np.inf
        denom = float(2 * n - 1)
        for it in range(1, max_iter + 1):
            if time.time() > deadline:
                break
            conv = _conv_fft(a, a)
            avg = 1.0 / denom
            diff = conv - avg
            rev_a = a[::-1]
            full = _conv_fft(rev_a, diff)
            grad_a = 2.0 * full[n - 1:n - 1 + n]
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)
            eta = 5.0 / math.sqrt(it)
            log_a -= eta * grad_log
            a = _softmax(log_a)
            val = np.sum(diff * diff)
            if val < best_val:
                best_val = val
                best_a = a.copy()
        return best_a

    # L‑BFGS case
    v0 = np.log(start_vec + 1e-30)

    denom = float(2 * n - 1)
    avg_const = 1.0 / denom

    def fun(v):
        a = _softmax(v)
        conv = _conv_fft(a, a)
        diff = conv - avg_const
        loss = np.sum(diff * diff)

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        return loss, grad_v

    def callback(_):
        if time.time() > deadline:
            raise KeyboardInterrupt()

    try:
        res = _opt.minimize(fun=fun,
                            x0=v0,
                            method='L-BFGS-B',
                            jac=True,
                            callback=callback,
                            options={'maxiter': max_iter,
                                     'disp': False,
                                     'ftol': 1e-13})
        return _softmax(res.x)
    except KeyboardInterrupt:
        return _softmax(v0)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 800_000,
                            eta0: float = 8.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (active set)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of the max (average over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a -= eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothing‑max optimiser (differentiable approximation of max)
# ----------------------------------------------------------------------
def _smooth_max_opt(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 3_000_000,
                    eta0: float = 6.0,
                    beta_start: float = 0.5,
                    beta_growth: float = 1.001,
                    beta_cap: float = 2000.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on f_β(a)=β⁻¹·log∑ₖexp(β·convₖ(a,a)).
    Uses a soft‑max over the convolution entries, momentum and a schedule that
    slowly raises β (the approximation becomes tighter).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9
    beta = beta_start

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)
        conv = _conv_fft(a, a)
        max_c = conv.max()
        # numerically stable soft‑max weights
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()

        # gradient of the smoothed objective
        rev_a = a[::-1]
        full = _conv_fft(w, rev_a)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # momentum (Adam‑like)
        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad_log
        eta = eta0 / math.sqrt(it)

        log_a -= eta * momentum

        # occasional progress check
        if it % 2000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

        # raise β gradually
        beta = min(beta * beta_growth, beta_cap)

    # final evaluation
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_val, best_seq = final_val, a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 90_000,
                         start_sigma: float = 0.0012,
                         T_start: float = 0.012,
                         T_end: float = 3e-9) -> tuple[np.ndarray, float]:
    """Gaussian perturbations with a linear temperature schedule."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        if ns.sum() == 0.0:
            attempts += 1
            continue
        ns /= ns.sum()

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s wall‑clock budget, stores the best found
    sequence in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size – 4000 is a good trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Warm‑start from a previous run if it exists
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            pa = np.asarray(previous, dtype=np.float64)
            if pa.shape == (n,):
                s = pa.sum()
                if s > 0:
                    prev_arr = pa / s
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Build a huge and varied seed pool
    # --------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_beta_shape(n, 0.2, 0.2))
    seeds.append(_beta_shape(n, 0.1, 0.1))
    seeds.append(_exp_decay(n, 1e-3))
    seeds.append(_power(n, 0.5))
    seeds.append(_logistic_shape(n, 1.0, n * 0.2))
    seeds.append(_logistic_shape(n, 1.0, n * 0.8))

    # DPSS windows (if available)
    if _has_dpss:
        for nw in (2.0, 3.0, 4.0, 5.0):
            try:
                seeds.append(_dpss_shape(n, nw))
            except Exception:
                pass

    # previous best (if any)
    if prev_arr is not None:
        seeds.append(prev_arr)

    # --------------------------------------------------------------
    # Randomised sweep of parameterised families (≈ 0.12 · budget)
    # --------------------------------------------------------------
    sweep_deadline = start_time + 0.12 * budget_s
    while time.time() < sweep_deadline:
        # Beta shape – log‑uniform alphas / betas
        alpha = 10 ** rng.uniform(-4, 0.5)   # 1e‑4 … ≈ 3
        beta = 10 ** rng.uniform(-4, 0.5)
        seeds.append(_beta_shape(n, alpha, beta))

        # Exponential decay – log‑uniform rate
        rate = 10 ** rng.uniform(-6, 0)      # 1e‑6 … 1
        seeds.append(_exp_decay(n, rate))

        # Power‑law – uniform exponent
        p = rng.uniform(-0.5, 4.5)
        seeds.append(_power(n, p))

        # Logistic – random scale & midpoint
        scale = 10 ** rng.uniform(-0.5, 1.5)      # ≈ 0.3 … 30
        midpoint = rng.uniform(0.2 * n, 0.8 * n)
        seeds.append(_logistic_shape(n, scale, midpoint))

        # Dirichlet draws with random concentration
        conc = 10 ** rng.uniform(-4, 0)          # 1e‑4 … 1
        seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    # --------------------------------------------------------------
    # 2. Cheap surrogate evaluation of all seeds, keep the top few
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Process each seed (flatten → EG → SA)
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a modest slice of the remaining budget
        remaining = deadline - time.time()
        share = remaining * 0.15 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        # a) L‑BFGS flattening (L² deviation from constant)
        seq_flat = _lbfgs_flatten_opt(seed_vec, n, sub_deadline, max_iter=2000)

        # b) Exponentiated‑gradient on the exact max‑convolution
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt_multi(seq_flat,
                                                    n,
                                                    sub_deadline,
                                                    max_iters=500_000,
                                                    eta0=9.0,
                                                    tol=1e-9)
        else:
            seq_eg, val_eg = seq_flat, _objective_surrogate(seq_flat)

        # c) Short simulated‑annealing polish
        hill_deadline = sub_deadline - 0.2
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_eg,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=70_000,
                                                  start_sigma=0.0011,
                                                  T_start=0.011,
                                                  T_end=5e-9)
        else:
            seq_sa, val_sa = seq_eg, val_eg

        # pick the best among the three candidates
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_flat, _objective_surrogate(seq_flat)),
            (seq_eg, val_eg),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long polishing – smoothing‑max optimiser
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final smoothing‑max polishing ---")
        final_seq, final_val = _smooth_max_opt(global_best_seq,
                                               n,
                                               deadline,
                                               max_iters=2_500_000,
                                               eta0=6.0,
                                               beta_start=0.5,
                                               beta_growth=1.001,
                                               beta_cap=1500.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final smooth‐max] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided evaluate_sequence()
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
