def compute_conv_counts(weights, positions, n):
    # returns array conv length 2*n-1 (float)
    k = len(weights)
    conv = np.zeros(2*n-1, dtype=np.float64)
    for i in range(k):
        wi = weights[i]
        pi = positions[i]
        for j in range(k):
            conv[pi + positions[j]] += wi * weights[j]
    return conv

def _simulated_annealing(start_vec, n, deadline, start_temp=0.5, end_temp=0.01):
    vec = start_vec.copy()
    logit = np.log(vec + 1e-30)
    best_vec = vec.copy()
    best_val = _objective_surrogate(best_vec)
    cur_val = best_val
    cur_logit = logit.copy()
    while time.time() < deadline:
        t_frac = (deadline - time.time()) / (deadline - start_time)
        temp = end_temp + (start_temp - end_temp) * t_frac
        # propose
        step = 0.5 * temp
        proposal_logit = cur_logit + step * rng.normal(size=n)
        proposal = _softmax(proposal_logit)
        val = _objective_surrogate(proposal)
        if val < cur_val or rng.random() < math.exp(-(val - cur_val)/temp):
            cur_logit = proposal_logit
            cur_val = val
            if val < best_val:
                best_val = val
                best_vec = proposal.copy()
    return best_vec, best_val

    exponentiated‑gradient (mirror‑descent) on the *exact* max‑convolution
    (sub‑gradient at the active index) for a few hundred iterations and
    finish with a tiny greedy random‑local search.  The best sequence found
    over all restarts is stored.

3️⃣  Finally a short simulated‑annealing walk in the log‑softmax space
    (perturbing the logits) tries to escape any remaining local optimum.

All phases respect the global 1000 s deadline.  The global variable
``height_sequence_1`` is updated so that a later call can start from the
current best solution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, softmax, surrogate objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution by real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq))
    ``seq`` must already be a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _sparse_random_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Create a sparse probability vector.
    The support size is chosen roughly between 0.8·√n and 2·√n.
    The non‑zero entries are given a Dirichlet draw (conc=0.5) to avoid
    exact uniformity – a little randomness helps the later optimiser.
    """
    sqrt_n = int(math.sqrt(n))
    k = rng.integers(int(0.8 * sqrt_n), int(2.0 * sqrt_n) + 1)
    support = rng.choice(n, size=k, replace=False)
    weights = rng.dirichlet(np.full(k, 0.5))
    a = np.zeros(n, dtype=np.float64)
    a[support] = weights
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient on the true max‑convolution (mirror descent)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Greedy random local refinement (small mass transfers)
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 100_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic walk on the simplex:
    repeatedly move a small random amount from one coordinate to another.
    Only strictly improving moves are kept.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing in log‑softmax space
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         start_temp: float = 0.5,
                         end_temp: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Simple SA that perturbs the logits of the current distribution.
    Accepts worse moves with probability exp(-(Δ)/T).
    """
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    cur_logit = np.log(start_seq + 1e-30)
    cur_val = best_val

    t0 = time.time()
    while time.time() < deadline:
        # linearly decreasing temperature
        frac = (deadline - time.time()) / (deadline - t0)
        temp = end_temp + (start_temp - end_temp) * max(frac, 0.0)

        # propose a new state
        step_scale = 0.5 * temp
        prop_logit = cur_logit + step_scale * rng.normal(size=n)
        prop_seq = _softmax(prop_logit)

        prop_val = _objective_surrogate(prop_seq)

        if prop_val < cur_val or rng.random() < math.exp(-(prop_val - cur_val) / max(temp, 1e-12)):
            cur_logit = prop_logit
            cur_val = prop_val
            if prop_val < best_val:
                best_val = prop_val
                best_seq = prop_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm respects the supplied time budget (≈1000 s) and stores the best
    sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣  Load a previously stored solution (if any)
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    best_seq = None
    best_val = np.inf
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                arr = arr / arr.sum()
                val = _objective_surrogate(arr)
                best_seq, best_val = arr.copy(), val
                print(f"[Prev] surrogate = {val:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Build a diverse pool of seeds (all sum to 1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
    ]

    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    beta_alphas = [
        0.05, 0.07, 0.10, 0.15, 0.20, 0.25, 0.30,
        0.35, 0.40, 0.45, 0.50, 0.60, 0.80, 1.00,
        1.20, 1.50, 2.00,
    ]
    for a in beta_alphas:
        seeds.append(_beta_distribution(n, a))

    # a few sparse random starts – they often discover low‑peak patterns
    for _ in range(8):
        seeds.append(_sparse_random_seed(n, rng))

    if best_seq is not None:
        seeds.append(best_seq)   # keep the previous best as a seed

    # ------------------------------------------------------------------
    # 2️⃣  Quick evaluation of the seed pool
    # ------------------------------------------------------------------
    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.asarray(vec, dtype=np.float64)
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s < 1e-12:
            continue
        vec /= s
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, vec.copy()
            print("   >>> New global best from seeds")

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Multi‑restart sub‑gradient optimisation
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        restart += 1
        # pick a seed – 50 % chance to reuse the current best,
        # otherwise a fresh random (dense or sparse) seed.
        if rng.random() < 0.5:
            seed_vec = best_seq.copy()
        else:
            # random choice among the pre‑generated seeds (excluding the best)
            cand = rng.choice(seeds)
            seed_vec = np.asarray(cand, dtype=np.float64)
            seed_vec = np.maximum(seed_vec, 0.0)
            s = seed_vec.sum()
            if s < 1e-12:
                continue
            seed_vec /= s

        # allocate a slice of the remaining budget to this restart
        time_left = deadline - time.time()
        if time_left < 1.0:
            break
        # give the sub‑gradient phase ~70 % of the slice, the rest for local search
        grad_deadline = time.time() + 0.70 * time_left
        seq_g, val_g = _exp_gradient_opt(seed_vec, n, grad_deadline, eta0=5.0)

        # a few thousand greedy local moves (cheap)
        local_deadline = deadline
        seq_l, val_l = _local_random_search(seq_g, n, local_deadline,
                                            attempts=80_000, sigma=0.02)

        if val_l < best_val:
            best_val, best_seq = val_l, seq_l.copy()
            print(f"[Restart {restart:03d}] new best surrogate = {best_val:.6f}")

        # stop if we have already exhausted most of the budget
        if deadline - time.time() < 1.0:
            break

    # ------------------------------------------------------------------
    # 4️⃣  Final simulated‑annealing refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Simulated annealing refinement ---")
        sa_deadline = deadline
        seq_sa, val_sa = _simulated_annealing(best_seq, n, sa_deadline,
                                              rng, start_temp=0.5, end_temp=0.01)
        if val_sa < best_val:
            best_val, best_seq = val_sa, seq_sa.copy()
            print(f"[SA] improved surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
