def simulated_annealing(current_seq, deadline):
    n = len(current_seq)
    best_seq = current_seq.copy()
    best_val = objective(best_seq)
    cur_seq = current_seq.copy()
    cur_val = best_val
    # configure schedule: T0 = (initial_val - maybe)??? Use T0 = 1e-3? Actually objective values around 1.5. So T0 maybe 0.02. Use cooling factor 0.995 per iteration.
    T = (best_val) * 0.1  # maybe.
    rng = np.random.default_rng()
    iterations = 0
    while time.time() < deadline:
        # pick i,j
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.random() * 0.01 * cur_seq[i]  # small proportion
        if delta > cur_seq[i]: 
            delta = cur_seq[i]
        # random distribution of delta between i and j (transfer)
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # due to rounding, ensure non-negative
        if new_seq[i] < 0:
            continue
        # compute new objective
        new_val = objective(new_seq)
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            cur_seq = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()
        # cooling
        iterations += 1
        if iterations % 2000 == 0:
            # reduce temperature
            T *= 0.95
    return best_seq, best_val

"""
Improved step‑function optimisation.

Strategy
--------
1️⃣  **Broad parametric sweep** – generate many candidate shapes on the
    simplex (uniform, triangular, arcsine and a dense grid of symmetric
    Beta(α,α) distributions, especially for very small α ∈ [0.02, 0.30]).
    Each candidate is evaluated with the cheap surrogate
        2·n·max(conv(a,a))           (sum(a)=1)

2️⃣  **Exact mirror‑descent polishing** – the best few candidates are
    refined with exponentiated‑gradient (mirror descent) on the *true*
    max‑convolution (sub‑gradient at the active index).  The routine runs
    until the allocated time budget is exhausted.

3️⃣  **Simulated‑annealing final walk** – starting from the best refined
    sequence we perform a temperature‑controlled random walk on the
    simplex.  Small mass transfers between random coordinates are accepted
    either if they improve the surrogate or with probability
        exp(-(Δ / T)).  The temperature is slowly cooled.

4️⃣  The final vector (still normalised to sum = 1) is fed to the provided
    ``evaluate_sequence`` function.  The best sequence is stored in the
    global variable ``height_sequence_1`` for reuse in later calls.

The whole search respects the 1000 s budget; each stage checks the
deadline and stops early if time runs out.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, O(N log N))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate: 2·n·max(conv(seq, seq)).
    The input is assumed to be normalised (sum == 1) and non‑negative.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors, i.e. sum == 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) on the discrete set {0,…,n‑1}.
    Works for α>0 (including α<0.5, which gives heavier tails).
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, alpha - 1.0)
    a = a / a.sum()
    return a


def _cosine_end_peaked(n: int) -> np.ndarray:
    """Shape that is large at the ends and small in the centre."""
    i = np.arange(n, dtype=np.float64)
    # (1+cos) is maximal at the ends (i=0,n-1) and minimal in the middle.
    a = 1.0 + np.cos(2.0 * np.pi * i / (n - 1))
    a = a - a.min() + 1e-12   # shift to be strictly positive
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 5.0,
                     max_iters: int = 4_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the *exact* max‑convolution.
    The optimisation runs until the deadline or the iteration limit.
    Returns the best sequence seen and its surrogate value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # Sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % 5000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing random walk on the simplex
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         init_temp: float = 0.05,
                         cooling_rate: float = 0.95,
                         step_factor: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny SA on the simplex.
    Each move transfers a small random amount of mass from i → j.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    T = cur_val * init_temp  # start temperature proportional to current value
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose a transfer from i to j
        max_delta = cur_seq[i] * step_factor
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:               # safety
            continue

        new_val = _objective_surrogate(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp((cur_val - new_val) / max(T, 1e-12)):
            cur_seq = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()

        it += 1
        # cooling schedule
        if it % 2000 == 0:
            T *= cooling_rate
            if T < 1e-12:
                T = 1e-12

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – the function that will be called
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found within the time budget is stored in the global
    variable ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Load any previously stored solution (if length matches)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 0️⃣  Build a diverse seed pool
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
        _cosine_end_peaked(n)
    ]

    # 1️⃣  Fine grid of symmetric Beta(α,α) – especially very small α
    alphas = np.concatenate([
        np.linspace(0.02, 0.30, 15),   # dense in the region that was omitted before
        np.linspace(0.35, 0.90, 12),   # intermediate values
        np.linspace(1.0, 2.0, 11)      # heavier centre (for completeness)
    ])
    for a in alphas:
        seeds.append(_beta_distribution(n, float(a)))

    # 2️⃣  Random Dirichlet draws (different concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(rng.dirichlet(np.full(n, conc)).astype(np.float64))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣  Initial cheap evaluation of all seeds (surrogate)
    # ------------------------------------------------------------------
    seed_infos = []
    print("\n--- Seed pool evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(0.0, vec)          # just in case
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f" seed #{idx + 1:03d}: surrogate = {val:.6f}")

    if not seed_infos:
        raise RuntimeError("No seeds were evaluated before deadline.")

    seed_infos.sort(key=lambda x: x[0])
    top_k = 5
    top_seeds = [vec for _, vec in seed_infos[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"\n[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Exact exponentiated‑gradient polishing on the best seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:   # ensure we have something to work with
        eg_deadline = time.time() + remaining * 0.70   # allocate ~70 % to EG
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        for sid, seed_vec in enumerate(top_seeds):
            if time.time() > eg_deadline:
                break
            print(f"  EG run #{sid + 1}/{len(top_seeds)}")
            seq_eg, val_eg = _exp_gradient_opt(seed_vec, eg_deadline,
                                                eta0=7.0,
                                                max_iters=5_000_000)
            print(f"   → surrogate = {val_eg:.6f}")
            if val_eg < best_val:
                best_val, best_seq = val_eg, seq_eg.copy()
                print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 3️⃣  Final simulated‑annealing refinement (last seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, deadline,
                                              init_temp=0.07,
                                              cooling_rate=0.94,
                                              step_factor=0.015)
        print(f"   → surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_val, best_seq = val_sa, seq_sa.copy()
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 4️⃣  Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print("\n=== Finished ===")
        print(f"Surrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
