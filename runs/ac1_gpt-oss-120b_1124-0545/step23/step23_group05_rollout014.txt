"""
Step‑function optimiser – aggressive single‑seed refinement.

The optimiser builds a modest pool of analytic “seed” vectors (uniform,
triangular, power‑law, exponential‑decay, Gaussian, Beta‑shapes,
logistic, Dirichlet draws and the previously best known sequence).
All seeds are ℓ₁‑normalised (sum = 1).  The top few candidates are
refined with a *four‑stage* pipeline:

1. **Exponentiated‑gradient (EG)** – minimise the exact max‑convolution.
   Two passes are performed: a wide‑tol pass ( tol ≈ 1 %) to obtain a
   coarse flattening, followed by a tight‑tol pass ( tol ≈ 10⁻¹⁰) for
   fine‑grained improvement.  Momentum (Adam‑style) speeds‑up the
   descent.

2. **Smooth‑max L‑BFGS** – optimise a log‑sum‑exp surrogate of the max
   using a schedule of inverse‑temperatures β that grows up to 10⁶.
   The optimisation works in the log‑softmax (probability‑simplex)
   parametrisation, guaranteeing feasibility.

3. **Quadratic flattening** – minimise the variance of the convolution
   ( Σ (cₖ–avg)² ) by mirror‑descent, driving the whole auto‑convolution
   towards a flat profile.

4. **Simulated‑annealing (SA)** – a final stochastic polish that
   repeatedly perturbs the log‑weights with a Gaussian kernel and
   accepts worsening moves according to a linear cooling schedule.

The whole pipeline respects the global time‑budget (≈ 1000 s).  The
best sequence found is stored in the global variable ``height_sequence_1``
and returned as a plain ``list[float]`` suitable for ``evaluate_sequence``.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# --------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (sum(seq)==1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# --------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _dpss_window(n: int, nw: float) -> np.ndarray:
    if _sigwin is None:
        return _uniform(n)
    a = _sigwin.dpss(n, nw, Kmax=1, sym=True)
    a = np.abs(a[0])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# --------------------------------------------------------------
# Exponentiated‑gradient on the exact max‑convolution (multi‑active)
# --------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 1_000_000,
                            eta0: float = 10.0,
                            tol: float = 1e-10) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) targeting max(conv(a,a)).
    ``tol`` determines the width of the active set:
    conv ≥ max · (1 – tol).  Returns the best sequence seen.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # loosen tolerance a little to keep a reasonable active set
        active = np.where(conv >= max_val - tol * max_val)[0]

        # gradient of the (averaged) active max‑convolution terms
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑style momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# --------------------------------------------------------------
# Gradient‑descent that flattens the convolution (quadratic loss)
# --------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 800_000,
                         eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on Σₖ (convₖ – avg)².
    Drives the whole convolution towards a flat profile.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # length of full convolution
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when sum=1
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Mirror‑descent (Kullback–Leibler geometry)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 10_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# --------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# --------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.0015,
                         T_start: float = 0.012,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# --------------------------------------------------------------
# Deep optimisation pipeline for a single seed
# --------------------------------------------------------------
def _deep_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Aggressive refinement consisting of:
    1) EG (wide tolerance) → 2) EG (tight tolerance) → 3) L‑BFGS →
    4) flattening gradient → 5) simulated annealing.
    Each stage respects the shared deadline.
    Returns the best vector encountered and its surrogate value.
    """

    # ---- 1. EG – wide tolerance (focus on a broader active set) ----
    seq, val = _exp_gradient_opt_multi(
        start_vec.copy(),
        n,
        deadline,
        max_iters=800_000,
        eta0=15.0,
        tol=0.01,                     # 1 % of the max → many active indices
    )

    if time.time() >= deadline:
        return seq, val

    # ---- 2. EG – tight tolerance (polish the peak) ----
    seq, val = _exp_gradient_opt_multi(
        seq,
        n,
        deadline,
        max_iters=800_000,
        eta0=12.0,
        tol=1e-10,
    )

    if time.time() >= deadline:
        return seq, val

    # ---- 3. Smooth‑max L‑BFGS ----
    seq = _lbfgs_smooth_optimize(
        seq,
        n,
        deadline,
        beta_schedule=None,
        max_iter_per_beta=3000,
    )
    val = _objective_surrogate(seq)

    if time.time() >= deadline:
        return seq, val

    # ---- 4. Flattening gradient (quadratic loss) ----
    seq, val = _flatten_gradient_opt(
        seq,
        n,
        deadline,
        max_iters=800_000,
        eta0=10.0,
    )
    if time.time() >= deadline:
        return seq, val

    # ---- 5. Simulated Annealing polish ----
    seq, val = _simulated_annealing(
        seq,
        n,
        deadline,
        rng,
        max_attempts=400_000,
        start_sigma=0.001,
        T_start=8e-3,
        T_end=5e-9,
    )
    return seq, val


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s wall‑clock limit, stores the best found
    sequence in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # --------------------------------------------------------------
    # Problem size – a moderate but expressive length
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Load previous solution if present (helps continuity)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            tmp = np.asarray(previous, dtype=np.float64)
            if tmp.shape == (n,):
                prev_arr = tmp / tmp.sum() if tmp.sum() > 0 else None
        except Exception:
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Build a compact but diverse seed pool
    # --------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # Analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_inverse_sqrt(n))

    # power‑law tails
    for p in [0.5, 0.8, 1.0, 1.5]:
        seeds.append(_power(n, p))

    # exponential decays
    for rate in [1e-4, 5e-4, 1e-3, 5e-3, 1e-2]:
        seeds.append(_exp_decay(n, rate))

    # Gaussian windows
    for sigma_frac in [0.1, 0.2, 0.3]:
        seeds.append(_gaussian(n, sigma_frac))

    # Beta shapes (a few representative points)
    for a, b in [(0.5, 0.5), (2.0, 2.0), (0.2, 5.0), (5.0, 0.2)]:
        seeds.append(_beta_shape(n, a, b))

    # Logistic shapes
    for scale in [0.5, 1.0]:
        for mid_frac in [0.3, 0.5, 0.7]:
            seeds.append(_logistic_shape(n, scale, mid_frac * n))

    # Random Dirichlet draws
    for conc in [0.01, 0.1, 0.5]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # DPSS windows (if available)
    if _sigwin is not None:
        for nw in [2.5, 3.5, 4.5]:
            seeds.append(_dpss_window(n, nw))

    # Include the previous best (if any) and a few perturbed copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(4):
            pert = prev_arr + 0.006 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        raise RuntimeError("No valid seeds were generated.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(4, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Refine the top seeds – allocate the bulk of the remaining budget
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a decreasing share of the remaining time (more for the best)
        remaining = deadline - time.time()
        if si == 0:
            share = remaining * 0.60         # 60 % of what is left for the best seed
        else:
            # split the rest evenly among the remaining seeds
            share = remaining * 0.40 / (top_k - 1)

        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}/{top_k}] optimisation budget ≈ {share:.1f}s")
        seq_opt, val_opt = _deep_optimize(seed_vec, n, sub_deadline, rng)

        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"   *** New global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final intensive polishing on whatever is currently best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final intensive polishing on global best ---")
        final_seq, final_val = _deep_optimize(global_best_seq, n, deadline, rng)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final] improved surrogate to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
