def _simulated_annealing(a0, conv0, deadline, rng):
    n = len(a0)
    a = a0.copy()
    conv = conv0.copy()
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    best_iter = 0

    total_time = deadline - time.time()
    # set schedule parameters
    T0 = 1e-5  # initial temperature magnitude
    T_end = 1e-9
    step_start = 0.5
    step_end = 0.01

    # Stats
    it = 0

    while time.time() < deadline:
        it += 1
        # compute fraction of time elapsed
        elapsed = time.time() - start_time # need start_time param.
        # Actually pass start_time for schedule.
        fraction = (elapsed) / total_time
        T = T0 * (1 - fraction) + T_end * fraction
        step_frac = step_start * (1 - fraction) + step_end * fraction

        # random source selection weighted by a
        i = rng.choice(n, p=a)
        # random dest
        j = rng.integers(0, n)
        if j == i:
            continue
        delta = step_frac * a[i] * rng.random()
        if delta <= 0:
            continue

        a_old = a.copy()
        conv_old = conv.copy()
        # apply move
        apply_move_inplace(a, conv, i, j, delta)

                best_a = a.copy()
                # optionally reset best_conv
        else:
            # reject
            a[:] = a_old
            conv[:] = conv_old

    return best_a, best_max

"""
Advanced step‑function optimiser.

The optimiser proceeds in three phases:

1️⃣  Build a diverse pool of seed distributions (uniform, triangular,
    beta‑shaped and Dirichlet draws) and keep the best few according
    to the cheap surrogate 2·n·max(conv).

2️⃣  Refine the current best using the existing Adam‑smoothing and
    exact exponentiated‑gradient (EG) procedures – these give fast
    progress on a smooth landscape.

3️⃣  Finish with a fast simulated‑annealing (SA) walk that updates the
    convolution incrementally (O(n) per move).  The SA uses a
    temperature schedule and a step‑size schedule that shrink
    continuously, allowing a very large number of cheap moves and
    fine‑grained polishing of the solution.

The final sequence is evaluated with the supplied ``evaluate_sequence``
and stored globally as ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)).  Input assumed to sum to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (same pool as the reference implementation)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

    m = np.zeros_like(logit)

    it = 0

        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0,
        20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        max_iter_beta = 8000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break
            a = _softmax(logit)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)

            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.12
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 600 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val
    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 5.0,
                     max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % 5000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (new component)
# ----------------------------------------------------------------------


def _apply_move_inplace(a: np.ndarray,
                        conv: np.ndarray,
                        i: int,
                        j: int,
                        delta: float) -> None:
    """
    Transfer ``delta`` from index i to index j.
    Both ``a`` (probability vector) and ``conv`` (full auto‑convolution)
    are updated **in‑place** according to the exact quadratic formula.
    """
    n = a.shape[0]
    # linear part (uses the *old* a – caller must keep a copy if revert is needed)
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    # quadratic correction
    idx2i = 2 * i
    idx2j = 2 * j
    idxij = i + j
    if idx2i < conv.size:
        conv[idx2i] += delta * delta
    if idx2j < conv.size:
        conv[idx2j] += delta * delta
    if idxij < conv.size:
        conv[idxij] -= 2.0 * delta * delta

    # finally move the mass
    a[i] -= delta
    a[j] += delta


def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Fast SA that works directly on the *exact* max‑convolution.
    The convolution is updated incrementally (O(n) per move) which
    makes it possible to perform many cheap steps.
    Returns the best probability vector seen and its surrogate value.
    """
    a = start_seq.copy()
    n = a.shape[0]
    conv = _conv_fft(a, a)                      # exact convolution on the simplex
    cur_max = np.max(conv)

    best_a = a.copy()
    best_max = cur_max

    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    # schedule parameters (tuned experimentally)
    T_start = 2e-5          # initial temperature
    T_end = 1e-9            # final temperature
    step_start = 0.4        # early moves may shift up to 40 % of source mass
    step_end = 1e-4         # final moves are extremely fine‑grained

    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = T_start * (1.0 - frac) + T_end * frac
        step = step_start * (1.0 - frac) + step_end * frac

        # source chosen proportionally to current mass (big buckets move more often)
        i = rng.choice(n, p=a)
        # random destination, must differ from source
        j = rng.integers(0, n)
        if j == i:
            continue

        # proposed transfer – never exceed the available mass at i
        delta = step * a[i] * rng.random()
        if delta <= 0.0:
            continue

        # keep a copy of the *old* a for a cheap revert if needed
        a_old = a.copy()
        conv_old = conv.copy()

        _apply_move_inplace(a, conv, i, j, delta)
        new_max = np.max(conv)

        # Metropolis acceptance
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(T, 1e-15))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
        else:
            # reject – restore the previous state (no need to recompute)
            a[:] = a_old
            conv[:] = conv_old

        # occasional re‑normalisation / drift correction
        if it % 5000 == 0:
            s = a.sum()
            if abs(s - 1.0) > 1e-12:
                a /= s
            # re‑compute convolution to kill accumulated floating error
            conv = _conv_fft(a, a)
            cur_max = np.max(conv)

    return best_a, float(2.0 * n * best_max)

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found within the allotted budget is stored in the global
    variable ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Load any previously stored solution (if length matches)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            pass

    # --------------------------------------------------------------
    # 0️⃣ Assemble a diverse seed pool
    # --------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    for α in (0.3, 0.4, 0.5, 0.7, 0.9, 1.2, 1.5, 2.0):
        seeds.append(_beta_distribution(n, α))

    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 1️⃣ Initial evaluation of seeds (cheap surrogate)
    # --------------------------------------------------------------
    seed_infos = []
    print("\n--- Initial seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_k = 5
    top_seeds = [vec for _, vec in seed_infos[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Adam smoothing (≈15 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.15
        print("\n--- Adam smoothing (smooth max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # --------------------------------------------------------------
    # 3️⃣ Exact exponentiated‑gradient polishing (≈40 % of time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = time.time() + (deadline - time.time()) * 0.40
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, eg_deadline, eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # --------------------------------------------------------------
    # 4️⃣ Simulated‑annealing polishing (bulk of the remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, deadline, rng)
        print(f"[SA] exact surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_val, best_seq = val_sa, seq_sa.copy()
            print("   >>> Updated global best (SA)")

    # --------------------------------------------------------------
    # 5️⃣ Very tiny stochastic walk (final polishing, a few seconds)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final tiny random walk ---")
        seq_loc, val_loc = _local_random_search(best_seq, deadline,
                                                attempts=200_000,
                                                sigma=0.008)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # --------------------------------------------------------------
    # 6️⃣ Official evaluation with the supplied ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
