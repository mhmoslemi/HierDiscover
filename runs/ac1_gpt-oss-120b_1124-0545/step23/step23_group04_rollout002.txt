Pseudo:

```
seq = start_seq.copy()
best_seq = seq.copy()
best_val = objective(seq)
rng = np.random.default_rng()
T0 = 0.1
T_end = 1e-5
total_iters = 0
while time < deadline and total_iters < max_iter:
    # propose move:
    i, j = rng.integers(0, n, size=2)
    if i == j: continue
    delta = seq[i] * rng.random() * 0.2 # up to 20% of mass at i
    if delta == 0: continue
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    if new_seq[i] < 0: continue
    val = objective(new_seq)
    dval = val - best_val
    T = T0 * (T_end/T0) ** (total_iters / max_iter)
    if dval < 0 or rng.random() < np.exp(-dval / T):
        seq = new_seq
        if val < best_val:
            best_val = val
            best_seq = new_seq.copy()
    total_iters += 1

"""
Improved step‑function optimisation.

The search consists of three phases:

1️⃣  Fast, diverse seeding (uniform / arcsine / triangular / Beta /
    Dirichlet) – the best few seeds (by the exact surrogate 2·n·max(conv))
    are kept.

2️⃣  Adam optimisation on a smooth log‑sum‑exp approximation of the
    max‑convolution.  The schedule for the smoothness parameter β is
    progressively enlarged; we run this for the first ~35 % of the time
    budget (more iterations per β than the previous baseline).

3️⃣  Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution
    for the bulk of the remaining time.  A larger initial step‑size
    (η₀=8) and a simple restart‑on‑stagnation strategy are used.

4️⃣  Tiny simulated‑annealing walk on the simplex as a final polishing
    step (≈10 % of the time).

All intermediate optimisation uses the cheap surrogate
    2·n·max(conv(a,a))   (exact when the sequence is normalised to sum = 1).
At the end the best sequence found is evaluated with the provided
``evaluate_sequence`` and stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))  (seq must sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seeding helpers
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth (log‑sum‑exp) surrogate.
    Returns the best probability vector seen and its exact surrogate value.
    """
    n = len(start_seq)
    # initialise logits from the starting distribution
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a generous schedule (more β values and more iterations per β)
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0,
        20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # allocate a time‑budget per β – if we run out we leave early
        max_iter_beta = 12_000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current point on the simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)       # outer factor 2·n

            # convert to gradient in the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.12
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # periodic exact evaluation to preserve the best
            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final evaluation
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 8.0,
                     max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the *exact* max‑convolution.
    Uses the sub‑gradient at the active index.  If no improvement is
    observed for a long stretch we inject a small random perturbation.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    it = 0
    last_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % 4000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                last_improve = it

        # restart on stagnation (no improvement for ~200 k iterations)
        if it - last_improve > 200_000:
            # a tiny random jitter that stays on the simplex
            jitter = rng.normal(scale=0.02, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            if a.sum() == 0:
                a = _uniform(n)
            else:
                a = a / a.sum()
            log_a = np.log(a + 1e-30)
            last_improve = it

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projects a vector onto the probability simplex (non‑negative, sum = 1)."""
    # implementation of the algorithm from
    # https://arxiv.org/abs/1101.6081 (section 2)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         max_attempts: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Tiny simulated‑annealing walk on the simplex.
    The temperature decays exponentially; we accept worsening moves
    according to the Metropolis criterion.
    """
    n = len(start_seq)
    seq = start_seq.copy()
    best_seq = seq.copy()
    best_val = _objective_surrogate(best_seq)
    rng = np.random.default_rng()

    # annealing schedule parameters
    T0 = 0.12
    T_end = 6e-5
    total_iters = 0

    while time.time() < deadline and total_iters < max_attempts:
        total_iters += 1
        # linearly map iteration to [0,1] for temperature decay
        tau = total_iters / max_attempts
        T = T0 * (T_end / T0) ** tau

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a random fraction (up to 30 %) of mass from i → j
        delta = seq[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue

        val = _objective_surrogate(new_seq)
        dval = val - best_val
        if dval < 0 or rng.random() < math.exp(-dval / max(T, 1e-12)):
            seq = new_seq
            if val < best_val:
                best_val = val
                best_seq = new_seq.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found within the allotted budget is stored in the global
    variable ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Load any previously stored solution (if length matches)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            pass

    # --------------------------------------------------------------
    # 0️⃣  Assemble a diverse seed pool
    # --------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # Beta‑shaped seeds (α varies)
    for α in (0.3, 0.4, 0.5, 0.7, 0.9, 1.2, 1.5, 2.0):
        seeds.append(_beta_distribution(n, α))

    # Dirichlet draws with several concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 1️⃣  Initial cheap evaluation of seeds (surrogate)
    # --------------------------------------------------------------
    seed_infos = []
    print("\n--- Initial seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            continue
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")

    if not seed_infos:
        raise RuntimeError("No valid seeds were generated.")

    seed_infos.sort(key=lambda x: x[0])
    top_k = 3
    top_seeds = [vec for _, vec in seed_infos[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam smoothing (≈35 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.35
        print("\n--- Phase A: Adam smoothing (smooth max) ---")
        seq_adam, val_adam = _adam_optimize(best_seq, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq, best_val = seq_adam.copy(), val_adam
            print("   >>> Updated global best (Adam)")

    # --------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient polishing (≈55 % of remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = deadline - 2.0  # keep a couple of seconds for final tweaks
        print("\n--- Phase B: Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, eg_deadline,
                                           eta0=8.0,
                                           max_iters=5_000_000)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg.copy(), val_eg
            print("   >>> Updated global best (EG)")

    # --------------------------------------------------------------
    # 4️⃣  Final tiny simulated‑annealing refinement (≈10 % of time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase C: Simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, deadline,
                                              max_attempts=300_000)
        print(f"[SA] exact surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_seq, best_val = seq_sa.copy(), val_sa
            print("   >>> Updated global best (SA)")

    # --------------------------------------------------------------
    # 5️⃣  Official evaluation with the supplied ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
