def _targeted_hill_climb(start_seq, n, deadline, rng, max_iters=400_000):
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule progressively reduces acceptance of worsening moves
    T_start = 5e-4 * cur_val # maybe bigger
    T_end = 1e-6 * cur_val
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        # compute gradient from max conv entry
        conv = _conv_fft(cur_seq, cur_seq)
        max_idx = int(np.argmax(conv))

        # subgradient indices
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        # gradient values:
        grad = 2.0 * cur_seq[rev_idx]    # shape same as idx_i

        # choose source and target extremes
        # source = index with largest grad (i.e., larger a at rev position)
        src_local = np.argmax(grad)
        tgt_local = np.argmin(grad)
        src = idx_i[src_local]
        tgt = idx_i[tgt_local]
        if src == tgt or cur_seq[src] <= 0:
            # fallback to random swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:
            # transfer delta from src to tgt
            delta = min(cur_seq[src] * 0.1, 0.001)  # up to 10% of src, not too large
            ns = cur_seq.copy()
            ns[src] -= delta
            ns[tgt] += delta

        # renorm to sum=1 (should remain because delta transferred)
        # just to correct rounding
        s = ns.sum()
        if s == 0:
            continue
        ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            # accept with probability exp(-(val - cur_val)/T)
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val)/T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val
```

"""
Propose a step‑function (non‑negative sequence) that minimises
      2·n·max(conv(a,a)) / (sum(a)²)

The algorithm blends three ideas:

1.  A short Adam optimisation on a smoothed (log‑sum‑exp) surrogate
    of the max‑convolution.  This quickly finds a good region.

2.  A *targeted* hill‑climb that repeatedly moves a small amount of mass
    from the positions that contribute most to the current maximal
    convolution entry toward the positions that contribute least.
    The move is accepted with a simple simulated‑annealing schedule.

3.  A generic simulated‑annealing hill‑climb (swap / Gaussian perturb /
    block‑reallocation moves) that explores the neighbourhood of the best
    solution found so far.

If time permits a few short random‑restart Adam runs are performed as a
polish.

All steps respect the 1000 s budget; the best sequence is stored in the
global variable ``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) – used for Adam optimisation
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (surrogate objective, gradient w.r.t. the log‑softmax vars)
    for the smooth approximation of  max(conv(a,a)).
    """
    a = _softmax(v_local)                     # a_i ≥ 0, Σa_i = 1
    conv = _conv_fft(a, a)                    # length = 2n‑1

    max_c = np.max(conv)
    shift = max_c                              # numerical stabiliser

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (see derivation in the original solution)
    w = exp_beta / sum_exp                     # weights over convolution bins
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]     # slice back to length n

    # chain rule through softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 200_000,
              beta_start: float = 0.5,
              beta_end: float = 1_000_000.0,
              alpha: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate.
    Returns the best probability vector seen and its (exact) surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax parameters
    m = np.zeros_like(v)                  # first moment
    v2 = np.zeros_like(v)                 # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for the smoothing parameter β
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam update in the log‑space
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Targeted hill‑climb that moves mass from “heavy” to “light” contributors
# ----------------------------------------------------------------------
def _targeted_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly reduces the contribution of the indices
    that dominate the current maximal convolution entry.
    A simple simulated‑annealing schedule allows occasional uphill moves.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule (linear cooling)
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T_start + (T_end - T_start) * (it / max_iters)

        # compute current max‑convolution entry and its active sub‑gradient
        conv = _conv_fft(cur_seq, cur_seq)
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        grad = 2.0 * cur_seq[rev_idx]               # sub‑gradient on the simplex

        # pick source (largest grad) and target (smallest grad)
        src_local = int(np.argmax(grad))
        tgt_local = int(np.argmin(grad))
        src = idx_i[src_local]
        tgt = idx_i[tgt_local]

        if src == tgt or cur_seq[src] <= 0.0:
            # fall back to a random swap when the gradient move is not possible
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:
            # transfer a modest amount of mass from src to tgt
            delta = min(cur_seq[src] * 0.07, 0.001)   # at most 7 % of src (capped)
            ns = cur_seq.copy()
            ns[src] -= delta
            ns[tgt] += delta

        # renormalise (only to cure rounding errors)
        ns = ns / ns.sum()

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            # accept worsening move with probability exp(-(Δ)/T)
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Generic simulated‑annealing hill‑climb (swap / perturb / block moves)
# ----------------------------------------------------------------------
def _enhanced_hill_climb(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Local search with three move types:
        * swap two entries,
        * add a tiny Gaussian perturbation,
        * re‑allocate a small random block of mass.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T_start + (T_end - T_start) * (it / max_iters)

        r = rng.random()
        if r < 0.4:                     # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        elif r < 0.8:                   # Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s
        else:                           # block re‑allocation (Dirichlet‑like)
            k = rng.integers(2, 7)      # choose a small block size 2‑6
            idxs = rng.choice(n, size=k, replace=False)
            total = cur_seq[idxs].sum()
            if total == 0.0:
                continue
            # generate a random distribution on the block (Dirichlet(1,…,1))
            new_props = rng.random(k)
            new_props = new_props / new_props.sum()
            ns = cur_seq.copy()
            ns[idxs] = total * new_props

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val
