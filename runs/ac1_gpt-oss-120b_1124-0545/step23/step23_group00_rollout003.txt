Implementation:

def _proj_simplex(v):
    """Project onto simplex {x >=0, sum=1}."""
    # Duchi et al. algorithm.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)

def subgradient_descent(start_seq, n, deadline):
   a = start_seq.copy()
   best_seq = a.copy()
   best_val = _objective_surrogate(a)  # but we compute via conv
   it = 0
   eta0 = 1.0
   while time.time() < deadline:
       it += 1
       eta = eta0 / math.sqrt(it)
       conv = _conv_fft(a, a)
       max_idx = int(np.argmax(conv))
       cur_val = 2.0 * n * np.max(conv)
       if cur_val < best_val:
           best_val = cur_val
           best_seq = a.copy()
       # compute gradient
       idx = max_idx - np.arange(n)
       mask = (idx >= 0) & (idx < n)
       g = np.zeros(n)
       g[mask] = 2.0 * a[idx[mask]]
       # step
       a_new = a - eta * g
       # project onto simplex
       a = _proj_simplex(a_new)
   return best_seq, best_val

def _proj_simplex(v):
    # project v onto simplex {x >= 0, sum x = 1}
    # Duchi et al., algorithm:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _hill_climb_extended(start_seq, n, deadline, rng, max_iters=800_000):
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val
    T_start = 1e-2 * cur_val
    T_end = 1e-6 * cur_val
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)
        # decide move
        if rng.random() < 0.5:
            # swap
            i, j = rng.integers(0, n, size=2)
            ns = cur.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:
            # transfer
            # pick source with high a
            src = rng.choice(n, p=cur/cur.sum())
            # pick target (different) uniformly
            tgt = rng.integers(0, n)
            while tgt == src:
                tgt = rng.integers(0, n)
            delta = min(cur[src] * 0.3, 0.001)  # small fraction
            ns = cur.copy()
            ns[src] -= delta
            ns[tgt] += delta
        val = _objective_surrogate(ns)
        if val < cur_val:
            cur = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0 and rng.random() < math.exp(-(val - cur_val)/T):
                cur = ns
                cur_val = val
    return best_seq, best_val

"""
Step‑function optimisation for the ``evaluate_sequence`` benchmark.

Approach
--------
1. Initialise from the previously found best sequence (if any) or from an
   exponential decay, normalised to sum = 1.
2. **Projected sub‑gradient descent** on the exact objective
   ``max(conv(a,a))``.
   – At each iteration the active convolution index *k* (the one attaining the
     maximum) is identified.
   – A sub‑gradient of the max‑convolution w.r.t. ``a`` is
     ``g_i = 2·a_{k−i}`` (zero outside the support).
   – A step ``a ← a – η·g`` is taken and the result is projected back onto the
     probability simplex (non‑negative, sum = 1) via the O(n log n) Duchi et al.
     projection.
   – Step size follows ``η_t = η₀ / sqrt(t)``.
   The loop runs until the allocated time is exhausted, constantly keeping the
   best vector found.
3. **Intensive simulated‑annealing hill‑climb** that explores two types of
   neighbourhood moves:
   * swap two entries,
   * transfer a small fraction of mass from one index to another.
   Worsening moves are accepted with a linearly cooling temperature.
4. **Short Adam refinements** (the same log‑softmax Adam used previously) are
   launched as random‑restart “polishing” steps while time permits.
5. The final sequence is evaluated with the provided ``evaluate_sequence`` and
   stored in ``height_sequence_1`` for possible continuation runs.

The algorithm respects the 1000 s budget (a 5 s safety margin) and aims to
lower the surrogate value below the previously obtained 1.5085, targeting the
goal of ≤ 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based full linear convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ∈ ℝⁿ | x_i ≥ 0, Σx_i = 1}.
    Implements the algorithm from Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)) (sum(seq) = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         eta0: float = 0.5,
                         max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent with simplex projection.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # step size schedule
        eta = eta0 / math.sqrt(it)

        # compute convolution and the active index
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # active index (the one attaining the max)
        k = int(np.argmax(conv))

        # sub‑gradient of max⁡conv at a (g_i = 2·a_{k−i} if in range)
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # gradient step + simplex projection
        a = _proj_simplex(a - eta * g)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (swap / mass‑transfer moves)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Local search with two move types:
    – swap two coordinates,
    – transfer a tiny fraction of mass from a source index to a target index.
    Accepts worsening moves with a linearly cooling temperature.
    Returns the best sequence found.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule (log‑linear)
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:          # swap move (most common)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                             # mass‑transfer move
            # pick a source index with probability ∝ a (so likely a large entry)
            src = rng.choice(n, p=cur_seq / cur_seq.sum())
            # pick a distinct target uniformly
            tgt = rng.integers(0, n)
            while tgt == src:
                tgt = rng.integers(0, n)
            # transfer a random fraction (up to 10 %) of the source mass
            delta = cur_seq[src] * 0.1 * rng.random()
            ns = cur_seq.copy()
            ns[src] -= delta
            ns[tgt] += delta

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq, cur_val = ns, val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # Metropolis acceptance
            if T > 0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq, cur_val = ns, val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed surrogate (log‑sum‑exp of the max)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Objective and gradient of the log‑sum‑exp smoothed max‑convolution.
    Operates on log‑softmax variables `v_local`.
    """
    a = _softmax(v_local)                     # probability vector (sum = 1)
    conv = _conv_fft(a, a)                     # length = 2n‑1
    max_c = np.max(conv)
    shift = max_c                               # stabilisation

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a (weights for the smoothed max)
    w = exp_beta / sum_exp                     # shape (2n‑1,)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]        # slice back to n components

    # chain rule: ∂obj/∂v = a * (grad_a - (grad_a·a))
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 200_000,
              beta_start: float = 0.5,
              beta_end: float = 1_000_000.0,
              alpha: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed surrogate.
    Returns the best sequence (probability vector) and its surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax parameters
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for the smoothing parameter β
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine combines projected sub‑gradient descent, an intensive
    simulated‑annealing hill‑climb, and optional Adam refinements.
    Returns the best sequence discovered before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin – keep a few seconds for the final evaluation call
    deadline = start_time + budget_s - 5.0

    n = 4000  # problem size used by the benchmark

    # ------------------------------------------------------------------
    # 1. Initialise – reuse previous best if available
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                init_seq = np.maximum(arr, 0.0)
                init_seq = init_seq / init_seq.sum()
            else:
                init_seq = None
        except Exception:
            init_seq = None
    else:
        init_seq = None

    if init_seq is None:
        # exponential decay as a simple baseline (rate tuned for n=4000)
        i = np.arange(n, dtype=np.float64)
        exp_rate = 0.001
        a = np.exp(-exp_rate * i)
        init_seq = a / a.sum()

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient descent (core optimisation)
    # ------------------------------------------------------------------
    print("[Stage 1] Projected sub‑gradient descent")
    seq_sgd, val_sgd = _subgradient_descent(
        start_seq=init_seq,
        n=n,
        deadline=deadline,
        eta0=0.5,
        max_iters=5_000_000,
    )
    print(f"   best surrogate after SGD = {val_sgd:.6f}")

    # ------------------------------------------------------------------
    # 3. Intensive hill‑climb (swap / mass‑transfer moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n[Stage 2] Hill‑climb (swap / transfer)")
        seq_hc, val_hc = _hill_climb(
            start_seq=seq_sgd,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=800_000,
        )
        print(f"   best surrogate after HC = {val_hc:.6f}")
        if val_hc < val_sgd:
            seq_sgd, val_sgd = seq_hc, val_hc

    # ------------------------------------------------------------------
    # 4. Short random‑restart Adam refinements while time permits
    # ------------------------------------------------------------------
    while time.time() + 30.0 < deadline:
        print("\n[Stage 3] Random‑restart Adam polishing")
        noisy = seq_sgd + 0.003 * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy = noisy / s

        sub_deadline = time.time() + 25.0
        seq_rr, val_rr = _adam_opt(
            start_seq=noisy,
            n=n,
            deadline=sub_deadline,
            max_iters=80_000,
            beta_start=0.5,
            beta_end=200_000.0,
            alpha=0.5,
        )
        print(f"   restart surrogate = {val_rr:.6f}")
        if val_rr < val_sgd:
            seq_sgd, val_sgd = seq_rr, val_rr
            print("   *** improvement found ***")

    # ------------------------------------------------------------------
    # 5. Final evaluation with the black‑box function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(seq_sgd.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {val_sgd:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = seq_sgd.tolist()
    return seq_sgd.tolist()
