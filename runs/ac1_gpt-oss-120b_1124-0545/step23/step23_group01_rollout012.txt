def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used for optimisation: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law: p_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return a / a.sum()


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max‑convolution objective
# ----------------------------------------------------------------------
def _grad_softmax_max(log_a: np.ndarray, n: int, beta: float) -> np.ndarray:
    """
    Gradient of (1/β)·log Σ_k exp(β·conv(a,a)[k]) w.r.t. log‑softmax
    parameters `log_a`.  Returned gradient is ∂F/∂log_a.
    """
    a = _softmax(log_a)                 # probability vector
    conv = _conv_fft(a, a)               # full self‑convolution (len 2n‑1)

    # softmax weights over the convolution entries
    scaled = beta * conv
    max_scaled = np.max(scaled)
    w = np.exp(scaled - max_scaled)
    w /= w.sum()                         # w sums to 1, shape (2n‑1,)

    # g_a = 2 Σ_k w_k · shift(a)   (size n)
    a_rev = a[::-1]
    w_a = _conv_fft(w, a_rev)            # length 3n‑2
    g_a = 2.0 * w_a[n - 1 : n - 1 + n]    # slice → length n

    # Chain rule through the softmax
    dot = np.dot(g_a, a)
    grad_log = a * (g_a - dot)
    return grad_log


def _adam_softmax_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 200_000,
                      lr0: float = 0.5,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      beta_start: float = 0.5,
                      beta_end: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation on the smoothed max‑convolution objective.
    Returns the best simplex vector found and its surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)      # initialise in log‑space
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linearly increase the temperature β
        beta = beta_start + (beta_end - beta_start) * (t / max_iters)

        # gradient of the smoothed objective
        grad = _grad_softmax_max(log_a, n, beta)

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # evaluate current surrogate (exact, not smoothed)
        cur_seq = _softmax(log_a)
        cur_val = _objective_surrogate(cur_seq)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (identical to the reference version)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low
    evaluate_sequence value.  The routine builds a diverse set of
    seeds (arcsine, uniform, exponentials, random Dirichlet, …),
    refines each seed with a smoothed ADAM optimiser, and finally
    applies a long exponentiated‑gradient polishing pass.
    The best sequence found is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin before the hard deadline
    deadline = start_time + budget_s - 5.0

    n = 4000

    # ------------------------------------------------------------------
    # Build a pool of seeds
    # ------------------------------------------------------------------
    candidates = []

    # 1) arcsine – the most promising analytic shape
    candidates.append(_arcsine_distribution(n))

    # 2) previous best, if any
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                candidates.append(np.maximum(arr, 0.0) / np.sum(np.maximum(arr, 0.0)))
        except Exception:
            pass

    # 3) uniform
    candidates.append(_uniform(n))

    # 4) a few exponential decays
    for rate in (0.0005, 0.001, 0.002, 0.005):
        candidates.append(_exp_decay(n, rate))

    # 5) convex combinations of arcsine + uniform (coarse sweep)
    arcs = _arcsine_distribution(n)
    uni = _uniform(n)
    for alpha in np.linspace(0.0, 1.0, 11):
        cand = alpha * arcs + (1.0 - alpha) * uni
        candidates.append(cand / cand.sum())

    # 6) random Dirichlet draws (lightweight)
    for _ in range(3):
        cand = rng.dirichlet(alpha=np.full(n, 0.5))
        candidates.append(cand)

    # ------------------------------------------------------------------
    # Optimise each candidate with the smoothed Adam optimiser
    # ------------------------------------------------------------------
    best_overall_seq = None
    best_overall_val = float("inf")

    remaining_candidates = len(candidates)

    for idx, cand in enumerate(candidates, start=1):
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()

        time_left = deadline - time.time()
        if time_left < 10.0:
            print("[Info] Time exhausted, stopping candidate loop.")
            break

        # allocate iterations proportional to remaining time
        max_iters = int(min(250_000,
                            max(20_000, (time_left - 20.0) /
                                      max(1, remaining_candidates - idx + 1))))
        print(f"[Candidate {idx}/{len(candidates)}] max_iters={max_iters}")

        seq, val = _adam_softmax_opt(cand, n, deadline,
                                     max_iters=max_iters,
                                     lr0=0.5,
                                     beta_start=0.5,
                                     beta_end=8.0)

        print(f"   surrogate after Adam = {val:.6f}")

        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   *** new global best (surrogate) = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with exponentiated‑gradient (high‑iteration pass)
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # fallback – uniform if everything failed
        best_overall_seq = _uniform(n)

    if time.time() < deadline:
        print("\n--- Polishing with exponentiated‑gradient ---")
        polished_seq, polished_val = _exp_gradient_opt(
            best_overall_seq, n, deadline,
            max_iters=4_000_000,
            eta0=12.0)
        if polished_val < best_overall_val:
            best_overall_seq = polished_seq
            best_overall_val = polished_val
            print(f"[Polish] improved surrogate = {best_overall_val:.6f}")
        else:
            print(f"[Polish] no improvement (best surrogate unchanged)")

    # ------------------------------------------------------------------
    # Evaluate the true objective via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_overall_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible continuation runs
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
