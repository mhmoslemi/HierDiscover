Standard algorithm:

def _project_simplex(v):
    u = np.sort(v)[::-1]
    sv = np.cumsum(u) - 1
    rho = np.where(u - sv/(np.arange(1,len(v)+1)) > 0)[0][-1]
    theta = sv[rho]/(rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence(seq).

Algorithm
---------
* The objective is 2·n·max(conv(a,a)) for a probability vector a.
* We minimise the *soft‑max* approximation of the maximum:
      fα(a) = 2·n· Σ_k w_k·c_k ,   w_k = exp(α·c_k) / Σ_j exp(α·c_j) ,
  where c = conv(a,a).  As α → ∞ the function approaches the true
  objective, but for moderate α the gradient is smooth and informative.
* For each seed we run a projected gradient descent on the simplex:
      a_{t+1} = Π_{simplex}( a_t – η_t·∇fα(a_t) )
  with a step‑size η_t = η₀/√t and a linear schedule α_t∈[α_start,α_end].
  The gradient ∇fα is obtained analytically via two FFT‑based convolutions.
* Several diverse seeds are tried (previous best, exponential decays,
  a triangular shape, uniform, a few Dirichlet samples).  The best
  result of the descent runs is kept.
* A final “polish” phase runs the exponentiated‑gradient (mirror‑descent)
  optimiser used in the reference solution – it is very effective at
  squeezing the last few digits.
* All work respects the 1000 s wall‑time budget; the routine stops a few
  seconds before the deadline and stores the best vector in the global
  variable ``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)) – used for monitoring."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # tiny renormalisation to avoid accumulated numerical drift
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic families for initial guesses (simple, cheap)
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _triangular(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Gradient of the soft‑max‑approximated objective
# ----------------------------------------------------------------------
def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a fα where w are the soft‑max weights (sum to 1).
    Uses three FFT‑based convolutions (a with w and a with w reversed).
    """
    n = a.shape[0]

    # term2 = (a * w)[i + n - 1]   – ordinary convolution
    conv2_full = _conv_fft(a, w)                     # length 3n‑2
    term2 = conv2_full[n - 1 : 2 * n - 1]            # slice of length n

    # term1 = (a * w_rev)[2n‑2‑i] – correlation style
    w_rev = w[::-1]
    conv1_full = _conv_fft(a, w_rev)                 # length 3n‑2
    term1 = conv1_full[2 * n - 2 - np.arange(n)]

    grad = 2.0 * n * (term1 + term2)
    return grad


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_max_descent(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 5000,
                        alpha_start: float = 0.5,
                        alpha_end: float = 15.0,
                        lr0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max approximation of the max
    convolution objective.
    """
    a = start_seq.copy()
    a = a / a.sum()                 # ensure exact simplex
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # linear schedule for α (smoothness)
    if max_iters > 1:
        alphas = np.linspace(alpha_start, alpha_end, max_iters)
    else:
        alphas = np.full(1, alpha_end)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current smoothing parameter
        alpha = alphas[t - 1]

        # convolution a*a
        conv = _conv_fft(a, a)                     # length 2n‑1

        # soft‑max weights (stable)
        tmp = alpha * conv
        tmp_max = np.max(tmp)
        w = np.exp(tmp - tmp_max)
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        # gradient w.r.t. a
        grad = _grad_from_weights(a, w)

        # diminishing step‑size
        step = lr0 / math.sqrt(t)

        # projected gradient step
        a = _project_onto_simplex(a - step * grad)

        # monitor true surrogate (the real objective)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Exponentiated‑gradient polishing (identical to reference)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 4_000_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = np.exp(log_a)
        a = _project_onto_simplex(a)          # keep on simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

            pass

    # exponential decays with several rates
    for rate in (0.0005, 0.001, 0.002, 0.005):
        seeds.append(_exp_decay(n, rate))

    # triangular shape
    seeds.append(_triangular(n))

    # uniform
    seeds.append(_uniform(n))

    # a couple of random Dirichlet draws (spiky)
    for _ in range(2):
        seeds.append(rng.dirichlet(alpha=np.full(n, 0.5)))

    # safety – ensure at least one seed
    if not seeds:
        seeds.append(_exp_decay(n, 0.001))

    # ----------------------- optimise each seed --------------------
    best_overall_seq = None
    best_overall_val = float("inf")

    remaining = len(seeds)
    for idx, seed_vec in enumerate(seeds, start=1):
        if time.time() > deadline - 2.0:
            print("[Time] No time left for further seeds.")
            break

        # Normalise to the simplex
        seed_vec = np.maximum(seed_vec, 0.0)
        if seed_vec.sum() == 0.0:
            continue
        seed_vec = seed_vec / seed_vec.sum()

        # Allocate iterations dynamically – keep a few seconds for polishing
        time_left = deadline - time.time()
        # aim for at most 6 seconds per seed (adjustable)
        max_iters = int(min(6000, max(2_000, time_left / remaining * 0.8)))
        max_iters = max(max_iters, 1_000)   # never too tiny

        print(f"[Seed {idx}/{len(seeds)}] start – max iters = {max_iters}")

        seq, val = _smooth_max_descent(seed_vec,
                                       n,
                                       deadline,
                                       max_iters=max_iters,
                                       alpha_start=0.5,
                                       alpha_end=15.0,
                                       lr0=0.2)

        print(f"  -> surrogate after smooth descent = {val:.6f}")

        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"  ** new global best (surrogate) = {best_overall_val:.6f}")

        remaining -= 1

    # ----------------------- final polishing -----------------------
    if best_overall_seq is None:
        # fallback – a simple exponential decay
        best_overall_seq = _exp_decay(n, 0.001)
        best_overall_seq = best_overall_seq / best_overall_seq.sum()
        best_overall_val = _objective_surrogate(best_overall_seq)

    if time.time() < deadline:
        print("\n--- Polishing with exponentiated‑gradient ---")
        polished_seq, polished_val = _exp_gradient_opt(best_overall_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=4_000_000,
                                                       eta0=12.0)
        if polished_val < best_overall_val:
            best_overall_seq = polished_seq
            best_overall_val = polished_val
            print(f"[Polish] improved surrogate = {best_overall_val:.6f}")
        else:
            print(f"[Polish] no improvement (best surrogate unchanged)")

    # ----------------------- final evaluation ---------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_overall_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:               # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
