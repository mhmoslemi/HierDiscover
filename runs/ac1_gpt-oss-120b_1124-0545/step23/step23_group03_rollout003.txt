"""
Non‑negative step‑function optimiser (improved).

The optimiser works in several stages:

1️⃣  Build a rich pool of candidate distributions:
    • Uniform, triangular, arcsine.
    • A dense grid of symmetric Beta(α,α) densities (α∈[0.05,2]).
    • Random mixtures of two Beta members (weight w∈[0,1]).

2️⃣  Keep the best candidate according to the exact surrogate
    S(p)=2·n·max(conv(p,p)) (the same value returned by
    ``evaluate_sequence`` when the vector is normalised).

3️⃣  Refine the best candidate with three optimisation passes,
    all respecting the global deadline (≈ 1000 s):
    • Adam on a smooth‑max surrogate (log‑sum‑exp) with a long β‑schedule.
    • Exponentiated‑gradient (mirror‑descent) on the true max‑convolution.
    • Simulated‑annealing random local moves that occasionally accept
      uphill steps to escape shallow minima.

4️⃣  The final distribution is returned (and stored in the global
    variable ``height_sequence_1``) and also evaluated with the provided
    ``evaluate_sequence`` function.

All operations are performed on normalised vectors (sum = 1) – the
objective is scale‑invariant, so this does not affect the final score.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, softmax, surrogate)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution by real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used during the search:
       2·n·max(conv(seq,seq)), where ``seq`` sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return probability vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α, α) density discretised on bin centres."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Adam optimisation on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """Adam on the log‑softmax space of the log‑sum‑exp surrogate.
       Returns the best probability vector and its exact surrogate."""
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a fairly long schedule that ends with a very sharp β
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
        1000.0, 2000.0, 5000.0, 20000.0, 50000.0,
        100000.0, 200000.0, 500000.0, 1_000_000.0,
        2_000_000.0, 5_000_000.0, 10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # number of inner iterations depends on remaining time
        max_iter_beta = 6000 if (deadline - time.time()) > 30 else 1500

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)               # probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a (see derivation in analysis)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)        # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)            # outer factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient on true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

# Simulated‑annealing random local refinement
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         sigma: float = 0.02,
                         init_T: float = 5e-4,
                         final_T: float = 1e-7) -> tuple[np.ndarray, float]:
    """Greedy random walk that accepts uphill moves with a temperature schedule."""
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)

    best_seq = cur_seq.copy()
    best_val = cur_val

    start_stage = time.time()
    total_stage = max(deadline - start_stage, 1e-6)

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_stage
        T = init_T * (1.0 - elapsed / total_stage) + final_T * (elapsed / total_stage)

        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] <= 0.0:
            continue
        delta = sigma * cur_seq[i] * rng.random()
        if delta <= 0.0:
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        if new_seq[i] < 0.0:
            continue

        new_val = _objective_surrogate(new_seq)

        # accept if better, otherwise with probability exp(-(Δ)/T)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            if T > 0.0 and rng.random() < math.exp(-(new_val - cur_val) / T):
                accept = True

        if accept:
            cur_seq = new_seq
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()

    return best_seq, best_val

    # problem size (default 4000, can be overridden via kwargs)
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣ Load a previously stored solution (if any) as a potential seed
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and np.sum(arr) > 0.0:
                prev_vec = arr / np.sum(arr)
            else:
                prev_vec = None
        except Exception:
            prev_vec = None
    else:
        prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a rich seed pool (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
    ]

    # a dense grid of symmetric Beta(α,α) densities
    beta_grid = np.linspace(0.05, 2.0, 200)
    beta_vectors = [_beta_distribution(n, a) for a in beta_grid]

    # add the whole grid to the seed pool (these are already normalised)
    seeds.extend(beta_vectors)

    # optionally add the previous best solution
    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Initial evaluation of deterministic seeds
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Deterministic seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
        print(f" seed #{idx + 1:03d}: surrogate = {val:.6f}")

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Random mixtures of two Beta members (exploratory phase)
    # ------------------------------------------------------------------
    mix_deadline = deadline - 0.4 * (deadline - time.time())
    if time.time() < mix_deadline:
        print("\n--- Random mixture exploration ---")
        attempts = 0
        while time.time() < mix_deadline:
            attempts += 1
            # pick two beta vectors at random
            i1 = rng.integers(len(beta_vectors))
            i2 = rng.integers(len(beta_vectors))
            w = rng.random()
            vec = w * beta_vectors[i1] + (1.0 - w) * beta_vectors[i2]  # already sums to 1
            val = _objective_surrogate(vec)
            if val < best_val:
                best_val = val
                best_seq = vec.copy()
                print(f"  new best after {attempts} mixes: {best_val:.6f}")
                # keep exploring – the loop continues

    # ------------------------------------------------------------------
    # 4️⃣ Adam smoothing (smooth‑max surrogate) on the current best seed
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Adam smoothing (smooth‑max) ---")
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.45 * remaining
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 5️⃣ Exact exponentiated‑gradient polishing (sub‑gradient on true max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        polishing_deadline = deadline - 5.0   # keep a few seconds for final steps
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, polishing_deadline,
                                           eta0=5.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 6️⃣ Simulated‑annealing local refinement (allows uphill moves)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Simulated‑annealing local refinement ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, n, deadline,
                                              sigma=0.02,
                                              init_T=5e-4,
                                              final_T=1e-7)
        print(f"[SA] exact surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_val, best_seq = val_sa, seq_sa.copy()
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 7️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
