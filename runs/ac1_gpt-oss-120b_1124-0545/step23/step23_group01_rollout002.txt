def _targeted_local_search(a0, n, deadline, rng):
    a = a0.copy()
    conv = _conv_fft(a,a)
    cur_val = 2.0*n*np.max(conv)
    best_seq = a.copy()
    best_val = cur_val

    step_ratio = 0.05 # initial proportion of source mass to move
    it = 0
    while time.time() < deadline:
        it += 1
        k = int(np.argmax(conv))
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        idxs = np.arange(i_start, i_end+1)
        prod = a[idxs] * a[k - idxs]
        src_idx = idxs[np.argmax(prod)]
        # if a[src_idx] is very small, skip
        if a[src_idx] < 1e-12:
            # maybe choose new random source
            src_idx = rng.integers(0,n)
            if a[src_idx] < 1e-12:
                continue

        # pick target index outside the interval [i_start, i_end]
        # Build list of eligible indices
        # Instead of constructing every time, we can decide: choose with probability 0.5 left side else right side.
        if (i_start > 0) and (i_end < n-1):
            # both sides available
            if rng.random() < 0.5:
                # left side
                tgt = rng.integers(0, i_start)
            else:
                tgt = rng.integers(i_end+1, n)
        elif i_start > 0:
            tgt = rng.integers(0, i_start)
        else:
            tgt = rng.integers(i_end+1, n)

        # ensure target not equal src
        if tgt == src_idx:
            continue

        delta = step_ratio * a[src_idx]
        if delta <= 0:
            continue
        # ensure we don't move more than available
        delta = min(delta, a[src_idx])
        # apply move
        a_new = a.copy()
        a_new[src_idx] -= delta
        a_new[tgt] += delta

        conv_new = _conv_fft(a_new, a_new)
        new_val = 2.0 * n * np.max(conv_new)

        if new_val < cur_val:
            a = a_new
            conv = conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
            # optionally increase step ratio a bit
            step_ratio = min(step_ratio*1.02, 0.2)
        else:
            # maybe reduce step ratio for more granular moves
            step_ratio = max(step_ratio*0.99, 1e-4)

        # schedule: decrease step ratio gradually
        if it % 5000 == 0:
            step_ratio = max(step_ratio*0.99, 1e-5)

        # optional: break if improvement negligible
        # continue until time limit

    return best_seq, best_val

"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence(seq).

Algorithm
---------
1. Build a modest pool of diverse seeds (previous best, a few exponential
   decays, a triangular shape, the uniform distribution and a few
   Dirichlet draws).  All seeds are normalised to a probability vector.

2. Quick surrogate evaluation (2·n·max(conv(a,a))) is used to keep the
   three most promising seeds.

3. Each of those three seeds is refined by a mirror‑descent optimiser
   (Adam on the log‑softmax parametrisation).  The optimiser works in
   the exact max‑convolution objective and runs until the time budget
   is almost exhausted.

4. The best vector from the Adam stage is then fed to a *targeted local
   search*.  At every step we look at the index `k` where the convolution
   attains its current maximum.  We pick the pair (i,k‑i) that contributes
   the most to this entry and move a small fraction of the mass at `i`
   to an index that does **not** affect the entry `k`.  The move is
   accepted only if the surrogate value improves; otherwise we shrink
   the step size.  This simple greedy scheme is able to escape the
   plateau that Adam often gets stuck in and typically yields a few
   extra digits of improvement.

5. The final candidate is evaluated with the provided ``evaluate_sequence``
   function, printed and stored in the global variable ``height_sequence_1``
   for possible continuation runs.

The whole routine respects the 1000 s time limit, uses only NumPy and
SciPy (via ``scipy.sparse`` for the optional polishing stage) and works
for the default problem size `n = 4000`.
"""

import time
import math
import numpy as np
from scipy import sparse

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used for optimisation: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _triangular(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax variables
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 300_000,
              lr0: float = 0.2,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the exact max‑convolution objective.
    Returns the best sequence (probability vector) and its surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v_h = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active entry w.r.t. a:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g_a = np.zeros_like(a)
        g_a[idx_i] = 2.0 * a[rev_idx]

        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_h = beta2 * v_h + (1.0 - beta2) * (grad_v ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v_h / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (optional)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Targeted local search – moves mass away from the active maximal entry
# ----------------------------------------------------------------------
def _targeted_local_search(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Greedy local search that repeatedly reduces the contribution to the
    current maximal convolution entry by moving a small amount of mass
    to a location that does not affect that entry.
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)
    cur_val = 2.0 * n * np.max(conv)
    best_seq = a.copy()
    best_val = cur_val

    # step is a *relative* fraction of the source mass that will be moved
    step_frac = 0.08  # start with an 8 % move, will shrink automatically

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # active index
        k = int(np.argmax(conv))

        # interval of indices that participate in conv[k]
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idxs = np.arange(i_start, i_end + 1, dtype=int)

        # contributions a[i] * a[k-i]; pick the biggest one
        prod = a[idxs] * a[k - idxs]
        src_idx = idxs[np.argmax(prod)]

        # if the source weight is almost zero, pick a random non‑zero source
        if a[src_idx] < 1e-12:
            nz = np.flatnonzero(a > 1e-12)
            if len(nz) == 0:
                break
            src_idx = rng.choice(nz)

        # choose a target index that does NOT belong to [i_start, i_end]
        # (moving there cannot affect conv[k])
        if i_start > 0 and i_end < n - 1:
            if rng.random() < 0.5:
                tgt = rng.integers(0, i_start)
            else:
                tgt = rng.integers(i_end + 1, n)
        elif i_start > 0:
            tgt = rng.integers(0, i_start)
        else:
            tgt = rng.integers(i_end + 1, n)

        if tgt == src_idx:
            continue

        delta = step_frac * a[src_idx]
        delta = min(delta, a[src_idx])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[src_idx] -= delta
        a_new[tgt] += delta

        conv_new = _conv_fft(a_new, a_new)
        new_val = 2.0 * n * np.max(conv_new)

        # accept only improvements
        if new_val < cur_val:
            a = a_new
            conv = conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
            # slightly enlarge the step when we are making progress
            step_frac = min(step_frac * 1.02, 0.25)
        else:
            # otherwise shrink the step to become more delicate
            step_frac = max(step_frac * 0.97, 1e-5)

        # occasional aggressive shrink to avoid stagnation
        if it % 5000 == 0:
            step_frac = max(step_frac * 0.98, 1e-5)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluate_sequence
    value.  The routine builds a small pool of seeds, refines the three
    most promising seeds with Adam, optionally performs a short
    exponentiated‑gradient polishing, and finally runs a targeted greedy
    local search.  The best vector found is returned as a Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # 5 s safety margin

    n = 4000

    # ------------------------------------------------------------------
    # 1) Build a modest pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                candidates.append(arr / arr.sum())
        except Exception:
            pass

    # exponential decays with a few rates
    for rate in (0.0003, 0.0007, 0.0015, 0.003):
        candidates.append(_exp_decay(n, rate))

    candidates.append(_triangular(n))
    candidates.append(_uniform(n))

    # a couple of random Dirichlet draws (lightweight)
    for _ in range(3):
        candidates.append(rng.dirichlet(alpha=np.full(n, 0.5)))

    # ------------------------------------------------------------------
    # 2) Quick surrogate evaluation – keep the three best seeds
    # ------------------------------------------------------------------
    seed_vals = []
    for cand in candidates:
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()
        seed_vals.append((_objective_surrogate(cand), cand))

    seed_vals.sort(key=lambda x: x[0])
    top_seeds = [cand for (_, cand) in seed_vals[:3]]

    # ------------------------------------------------------------------
    # 3) Adam optimisation for the top seeds
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    for idx, seed_vec in enumerate(top_seeds, start=1):
        if time.time() > deadline:
            break
        # allocate a budget that respects the remaining time
        time_left = deadline - time.time()
        # aim for at most ~120 000 iterations per seed (≈0.5 s per 1 k iters)
        max_iters = int(min(150_000, max(30_000, time_left * 50)))
        print(f"[Adam] seed {idx}/{len(top_seeds)} – max_iters = {max_iters}")

        seq, val = _adam_opt(seed_vec, n, deadline, max_iters=max_iters,
                             lr0=0.22, beta1=0.9, beta2=0.999, eps=1e-8)
        print(f"   surrogate after Adam = {val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"   ** new best (surrogate) = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional exponentiated‑gradient polishing (short run)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Polishing with exponentiated‑gradient (short) ---")
        seq, val = _exp_gradient_opt(best_seq, n, deadline,
                                     max_iters=800_000, eta0=14.0)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Polish] improved surrogate = {best_val:.6f}")
        else:
            print("[Polish] no improvement")

    # ------------------------------------------------------------------
    # 5) Targeted greedy local search – the main improvement engine
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Targeted greedy local search ---")
        seq, val = _targeted_local_search(best_seq, n, deadline, rng,
                                          max_iters=1_200_000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Local] improved surrogate = {best_val:.6f}")
        else:
            print("[Local] no improvement")

    # ------------------------------------------------------------------
    # 6) Final evaluation with the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
