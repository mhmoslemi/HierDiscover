"""
Improved step‑function search for the evaluation function

The code follows a multi‑stage strategy:

1. **Broad low‑dimensional sweep** – generate a few hundred candidate
   shapes (pure exponential, power‑law and mixtures of two exponentials)
   and keep the one with the smallest surrogate value
   ``2·n·max(conv(a,a))`` (the true objective for ∑a=1).

2. **Adam on a smooth max‑convolution surrogate** – the non‑smooth
   max is replaced by a log‑sum‑exp with a temperature β that grows from
   0.5 to 1e9.  The optimisation variable is the log‑softmax vector,
   guaranteeing non‑negativity and the simplex constraint.

3. **Exact exponentiated‑gradient polishing** – sub‑gradient descent on the
   true max‑convolution with a diminishing stepsize.

4. **Long simulated‑annealing hill‑climb** – swaps and tiny Gaussian
   perturbations with a slowly cooling temperature.

5. **A second polishing pass + cheap random‑restart Adam refinements**.

All stages respect the overall 1000 s budget and check the wall‑clock
regularly.  The final sequence is stored in the global variable
``height_sequence_1`` for possible continuation runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate for 2·n·max(conv(seq,seq)) with ∑seq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _pow_law_decay(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) – needed for Adam optimisation
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution objective
    using log‑sum‑exp smoothing.
    """
    a = _softmax(v_local)                     # probability vector (sum = 1)
    conv = _conv_fft(a, a)                     # length = 2n‑1
    max_c = np.max(conv)
    shift = max_c                               # for numerical stability

    # w holds the soft‑max weights over the convolution entries
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    # guard against under‑flow (should be ≈1 because the max entry gives 1)
    if sum_exp == 0.0:
        w = np.zeros_like(exp_beta)
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_beta / sum_exp

    # gradient wrt a : 2 * (a_rev * w)   (convolution of reversed a with w)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]      # slice back to length n

    # chain rule through the soft‑max
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 200_000,
              beta_start: float = 0.5,
              beta_end: float = 1_000_000.0,
              alpha: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed surrogate.
    Returns the best probability vector found and its surrogate value.
    """
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v -= step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient (sub‑gradient) optimisation
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Operates on the simplex via a log‑space (softmax) update.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (double count for i≠j)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (swap / perturb moves)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Local stochastic search with swaps and Gaussian perturbations.
    Accepts worsening moves according to a linear cooling schedule.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                   # tiny Gaussian move
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq, cur_val = ns, val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq, cur_val = ns, val

    return best_seq, best_val

    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                init_seq = np.maximum(arr, 0.0)
                init_seq = init_seq / init_seq.sum()
            else:
                init_seq = None
        except Exception:
            init_seq = None
    else:
        init_seq = None

    # --------------------------------------------------------------
    # 1. Broad low‑dimensional sweep (exponential / power‑law / mixture)
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    # a) pure exponentials
    for rate in np.logspace(-4, -2, num=20):
        seq = _exp_decay(n, rate)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    # b) power‑law decays
    for p in np.linspace(0.2, 5.0, num=25):
        seq = _pow_law_decay(n, p)
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val, best_seq = val, seq

    # c) mixtures of two exponentials
    rates = np.logspace(-4, -2, num=8)
    for r1 in rates:
        for r2 in rates:
            e1 = _exp_decay(n, r1)
            e2 = _exp_decay(n, r2)
            for w in np.linspace(0.0, 1.0, 5):
                seq = w * e1 + (1.0 - w) * e2
                seq = seq / seq.sum()
                val = _objective_surrogate(seq)
                if val < best_val:
                    best_val, best_seq = val, seq

    # d) keep the earlier initialisation if it was better
    if init_seq is not None:
        v = _objective_surrogate(init_seq)
        if v < best_val:
            best_val, best_seq = v, init_seq

    print(f"[0] best surrogate after sweep = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Adam optimisation (smooth surrogate)
    # --------------------------------------------------------------
    print("\n[1] Adam optimisation (smooth surrogate)")
    seq_adam, val_adam = _adam_opt(
        start_seq=best_seq,
        n=n,
        deadline=deadline,
        max_iters=500_000,
        beta_start=0.5,
        beta_end=1e9,
        alpha=0.5,
    )
    print(f"    surrogate after Adam = {val_adam:.6f}")

    # --------------------------------------------------------------
    # 3. Exact exponentiated‑gradient polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n[2] Exact EG polishing")
        seq_eg, val_eg = _exp_gradient_opt(
            start_seq=seq_adam,
            n=n,
            deadline=deadline,
            max_iters=3_000_000,
            eta0=20.0,
        )
        print(f"    surrogate after EG = {val_eg:.6f}")
        if val_eg < val_adam:
            seq_adam, val_adam = seq_eg, val_eg

    # --------------------------------------------------------------
    # 4. Hill‑climb with simulated‑annealing moves
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n[3] Hill‑climb (swap / perturb)")
        seq_hc, val_hc = _hill_climb(
            start_seq=seq_adam,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=1_200_000,
        )
        print(f"    surrogate after HC = {val_hc:.6f}")
        if val_hc < val_adam:
            seq_adam, val_adam = seq_hc, val_hc

    # --------------------------------------------------------------
    # 5. Second EG polishing (fine‑tuning)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n[4] Second EG polishing")
        seq_eg2, val_eg2 = _exp_gradient_opt(
            start_seq=seq_adam,
            n=n,
            deadline=deadline,
            max_iters=2_000_000,
            eta0=30.0,
        )
        print(f"    surrogate after EG2 = {val_eg2:.6f}")
        if val_eg2 < val_adam:
            seq_adam, val_adam = seq_eg2, val_eg2

    # --------------------------------------------------------------
    # 6. Cheap random‑restart Adam refinements while time allows
    # --------------------------------------------------------------
    while time.time() + 30.0 < deadline:
        print("\n[5] Random‑restart Adam")
        noisy = seq_adam + 0.002 * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy = noisy / s

        sub_deadline = time.time() + 25.0
        seq_rr, val_rr = _adam_opt(
            start_seq=noisy,
            n=n,
            deadline=sub_deadline,
            max_iters=100_000,
            beta_start=0.5,
            beta_end=2e5,
            alpha=0.5,
        )
        print(f"    restart surrogate = {val_rr:.6f}")
        if val_rr < val_adam:
            seq_adam, val_adam = seq_rr, val_rr
            print("    *** improvement found ***")

    # --------------------------------------------------------------
    # 7. Final evaluation with the provided black‑box function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(seq_adam.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {val_adam:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = seq_adam.tolist()
    return seq_adam.tolist()
