def _subgrad_active(a, conv, eps=1e-12):
    max_val = conv.max()
    active = np.where(conv >= max_val - eps)[0]  # indices
    n = len(a)
    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # slice length = i_end - i_start +1
        # corresponding reversed a indices:
        rev_slice = a[i_end - np.arange(i_end - i_start + 1)]
        g[i_start:i_end+1] += 2.0 * rev_slice
    return g

def _exp_gradient_opt_full(start_seq, deadline, max_iters=2000000, eta0=10.0, eps=1e-12):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)
    n = len(a)
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        # compute conv
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # find active indices w.r.t tolerance
        active_idx = np.where(conv >= max_val - eps)[0]
        # compute subgradient g
        g = np.zeros_like(a)
        for k in active_idx:
            i0 = max(0, k - (n - 1))
            i1 = min(n - 1, k)

            # length = i1 - i0 + 1

            best_seq = a.copy()

"""
Improved step‑function optimiser
--------------------------------
* Builds a large pool of analytic and random seeds (uniform,
  exponential, power‑law, shifted power‑law, beta, Dirichlet).
* Keeps the best few by the cheap surrogate 2·n·max(conv(a,a))
  (a is ℓ₁‑normalised, so the surrogate equals the true
  ``evaluate_sequence`` value).
* Refines each top seed with a *mirror‑descent* (exponentiated‑gradient)
  optimiser that uses the true sub‑gradient of the max‑convolution.
  The sub‑gradient is taken as the **average** of all active
  (max‑attaining) convolution entries, which gives a more stable
  direction than using a single index.
* After the refinements, creates many convex combinations (mixtures)
  of the refined vectors – convexity guarantees the objective can only
  improve.  The best mixture is again polished by a final EG run.
* The whole workflow respects the 1000 s budget; progress and
  intermediate surrogate values are printed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT convolution (real, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (seq is assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^{-p} , i = 0…n‑1, δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) optimiser on the true max‑conv.
# Uses the sub‑gradient averaged over *all* active (max‑attaining) indices.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                      deadline: float,
                      max_iters: int = 2_500_000,
                      eta0: float = 10.0,
                      tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑softmax parameters
    log_a = np.log(a + 1e-30)

    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current convolution and active set
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol)[0]

        # compute sub‑gradient (average over the active set)
        g = np.zeros_like(a)
        for k in active:
            i0 = max(0, k - (n - 1))
            i1 = min(n - 1, k)
            # reversed slice of a that contributes to this k
            rev = a[i0:i1 + 1][::-1]
            g[i0:i1 + 1] += 2.0 * rev
        g /= max(1, active.size)   # average

        # stepsize
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g

        # keep log‑values bounded to avoid overflow/underflow
        log_a = np.clip(log_a, -30.0, 30.0)

        # mapping back to the simplex
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple smoothed Adam optimiser (log‑sum‑exp surrogate of max)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 15_000) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp smoothed surrogate of the max‑convolution."""
    if beta_schedule is None:
        beta_schedule = [10, 50, 200, 800, 3_200, 12_800, 51_200]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities (log‑softmax)
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            conv = _conv_fft(a, a)

            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp          # soft‑max weights

            # gradient wrt a (smooth max)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)      # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient wrt log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            log_a = np.clip(log_a, -30.0, 30.0)

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Convex mixing of several candidate vectors
# ----------------------------------------------------------------------
def _mix_seeds(seeds, num_mix: int, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Random convex combinations of the given seeds (all sum to 1)."""
    if not seeds:
        raise ValueError("No seeds to mix")
    k = len(seeds)
    best_seq = None
    best_val = float("inf")
    for _ in range(num_mix):
        weights = rng.dirichlet(np.ones(k))
        mix = np.zeros_like(seeds[0])
        for w, s in zip(weights, seeds):
            mix += w * s
        val = _objective_surrogate(mix)
        if val < best_val:
            best_val = val
            best_seq = mix.copy()
    return best_seq, best_val

# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 80_000) -> tuple[np.ndarray, float]:
    """Pure permutation optimiser – swaps two entries."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        accept = val < cur_val or (rng.random() < math.exp(-(val - cur_val) / T))
        if accept:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val
