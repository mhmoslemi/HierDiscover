def random_powerlaw(n):

    i = np.arange(n)+1
    seq = (i**(-p))
    seq = seq / seq.sum()
    return seq

def random_stretched_exp(n):
    rate = 10**np.random.uniform(-5, -2)
    beta = np.random.uniform(0.5, 3.0)
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-rate * (i**beta))
    seq = seq / seq.sum()
    return seq

def random_piecewise_exp(n):
    cutoff = np.random.randint(100, n-100)
    rate1 = 10**np.random.uniform(-5, -2)
    rate2 = 10**np.random.uniform(-5, -2)
    i = np.arange(n)
    seq = np.empty(n)
    seq[:cutoff] = np.exp(-rate1 * i[:cutoff])
    seq[cutoff:] = np.exp(-rate2 * (i[cutoff:] - cutoff))
    seq = seq / seq.sum()
    return seq

"""
Hybrid random‑parameter search + Adam optimisation + sub‑gradient polishing.

The algorithm works in three phases within the 1000 s budget:

1. Initialise from the previously stored best sequence (if any) or from a
   simple exponential decay.  Its surrogate objective (2·n·max(conv)) is printed.

2. Random‑restart optimisation:
   * Generate a candidate shape from several families
     (exponential, power‑law, stretched‑exponential, piecewise‑exponential,
      or a mixture of two exponentials).
   * Normalise it to sum = 1.
   * Run the Adam optimiser on the smooth log‑sum‑exp surrogate of the max‑convolution.
   * Keep the best sequence seen so far.

3. Refinement:
   * A long run of projected sub‑gradient descent (the “exponentiated‑gradient”
     step) on the exact max‑convolution objective.
   * A final simulated‑annealing hill‑climb (swap / tiny perturbation moves).

The final sequence is evaluated with the provided ``evaluate_sequence`` function,
stored in the global ``height_sequence_1`` and returned as a plain list of
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution via real FFT (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector (sum=1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact (scale‑invariant) objective 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) needed for Adam optimisation
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) of the smoothed max‑convolution objective
    using a log‑sum‑exp approximation.
    """
    a = _softmax(v_local)                     # probability vector (sum=1)
    conv = _conv_fft(a, a)                     # length = 2n‑1
    max_c = np.max(conv)
    shift = max_c                               # for numerical stability

    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                     # weights over convolution bins
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]      # slice back to length n

    # chain rule through the softmax
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 200_000,
              beta_start: float = 0.5,
              beta_end: float = 1_000_000.0,
              alpha: float = 0.5) -> tuple[np.ndarray, float]:
    """Adam on the smooth surrogate – returns best probability vector seen."""
    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    m = np.zeros_like(v)                  # first moment
    v2 = np.zeros_like(v)                 # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for the smoothing parameter β
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v = v - step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (exponentiated‑gradient) on max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑space representation for the simplex projection
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (swap / small Gaussian perturb)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """Local stochastic search that occasionally accepts worsening moves."""
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule (linear cooling)
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap two positions (most common)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                   # tiny Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # accept with probability exp(-(val‑cur_val)/T)
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random shape generators (all return a normalised non‑negative vector)
# ----------------------------------------------------------------------
def _gen_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = 10 ** rng.uniform(-5, -2)        # 1e‑5 … 1e‑2
    return _exp_decay(n, rate)


def _gen_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.1, 5.0)
    i = np.arange(n, dtype=np.float64) + 1.0
    a = i ** (-p)
    return a / a.sum()


def _gen_stretched_exp(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = 10 ** rng.uniform(-5, -2)
    beta = rng.uniform(0.5, 3.0)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (i ** beta))
    return a / a.sum()


def _gen_piecewise_exp(n: int, rng: np.random.Generator) -> np.ndarray:
    cutoff = rng.integers(200, n - 200)
    rate1 = 10 ** rng.uniform(-5, -2)
    rate2 = 10 ** rng.uniform(-5, -2)
    i = np.arange(n, dtype=np.float64)
    a = np.empty(n, dtype=np.float64)
    a[:cutoff] = np.exp(-rate1 * i[:cutoff])
    a[cutoff:] = np.exp(-rate2 * (i[cutoff:] - cutoff))
    return a / a.sum()


def _gen_mixed_exp(n: int, rng: np.random.Generator) -> np.ndarray:
    w = rng.random()
    rate1 = 10 ** rng.uniform(-5, -2)
    rate2 = 10 ** rng.uniform(-5, -2)
    i = np.arange(n, dtype=np.float64)
    a = w * np.exp(-rate1 * i) + (1.0 - w) * np.exp(-rate2 * i)
    return a / a.sum()

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (length 4000) that minimises
    ``evaluate_sequence``.  Returns the best sequence found before the time
    budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # small safety margin

    n = 4000

    # ------------------------------------------------------------------
    # 0. Initialise from previous run if possible
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, (list, tuple)) and len(prev) == n:
        try:
            init_seq = np.asarray(prev, dtype=np.float64)
            init_seq = np.maximum(init_seq, 0.0)
            s = init_seq.sum()
            if s > 0.0:
                init_seq = init_seq / s
            else:
                init_seq = _exp_decay(n, 0.001)
        except Exception:
            init_seq = _exp_decay(n, 0.001)
    else:
        init_seq = _exp_decay(n, 0.001)       # simple exponential start

    best_seq = init_seq.copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] surrogate = {best_val:.6f}")

    # helper to update global best
    def _maybe_update(seq: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective_surrogate(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"   *** new best surrogate = {best_val:.6f} ***")

    # ------------------------------------------------------------------
    # 1. Random‑restart Adam optimisation
    # ------------------------------------------------------------------
    families = [_gen_exponential, _gen_powerlaw, _gen_stretched_exp,
                _gen_piecewise_exp, _gen_mixed_exp]

    restart_counter = 0
    max_restarts = 10                     # tune for the time budget

    while time.time() < deadline - 30 and restart_counter < max_restarts:
        # Occasionally perturb the current best instead of a fresh random draw
        if rng.random() < 0.2:
            seq0 = best_seq + 0.001 * rng.normal(size=n)
            seq0 = np.maximum(seq0, 0.0)
            s = seq0.sum()
            if s > 0.0:
                seq0 = seq0 / s
            else:
                # fallback to a random shape
                seq0 = rng.choice(families)(n, rng)
        else:
            gen = rng.choice(families)
            seq0 = gen(n, rng)

        # Adam optimisation on the smooth surrogate
        seq_adam, val_adam = _adam_opt(
            start_seq=seq0,
            n=n,
            deadline=deadline,
            max_iters=150_000,
            beta_start=0.5,
            beta_end=1_000_000.0,
            alpha=0.5,
        )
        _maybe_update(seq_adam)

        restart_counter += 1

    # ------------------------------------------------------------------
    # 2. Exact sub‑gradient polishing (exponentiated‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        print("\n[Stage EG] Exact sub‑gradient polishing")
        seq_eg, val_eg = _exp_gradient_opt(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=1_500_000,
            eta0=10.0,
        )
        _maybe_update(seq_eg)

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        print("\n[Stage HC] Simulated‑annealing hill‑climb")
        seq_hc, val_hc = _hill_climb(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            rng=rng,
            max_iters=800_000,
        )
        _maybe_update(seq_hc)

    # ------------------------------------------------------------------
    # 4. Final evaluation with the black‑box function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store for possible continuation runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```

This implementation performs a diversified random‑parameter search, aggressive Adam optimisation on the smooth surrogate, followed by an exact sub‑gradient phase and a final annealed hill‑climb, all within the 1000 s time budget. It keeps the best sequence seen, prints progress, and finally returns the sequence ready for the black‑box evaluation.
