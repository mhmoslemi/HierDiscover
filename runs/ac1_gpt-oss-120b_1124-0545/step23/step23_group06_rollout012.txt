"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  The algorithm first tries a *linear‑programming
mixing* step (the “LP + mix” method described in the problem statement)
to push mass into positions where the current autocorrelation is low.
After a few cheap LP updates we polish the result with a long
exponentiated‑gradient descent and finish with a short permutation‑anneal.
If a previous solution (`height_sequence_1`) exists it is used as the
starting point; otherwise we start from a uniform distribution.
The whole search respects the supplied time budget (default 1000 s).
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Helper for fast convolution (real FFT, length 2n‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(a: np.ndarray) -> float:
    """
    Exact surrogate: 2·n·max(conv(a,a))   (a is assumed to sum to 1).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Build the (2n‑1) × n matrix that encodes   conv(f,g) ≤ M .
# Row k, column j entry = f_{k-j}   (zero if index out of range).
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return a sparse matrix A with A @ g = conv(f,g) (full convolution)."""
    n = f.shape[0]
    # indices for a dense n×n block; we will trim to the required shape.
    i = np.arange(n, dtype=np.int32)                     # row inside f
    j = np.arange(n, dtype=np.int32)                     # column index (g)
    # Every column j contributes f[i] at row i+j
    row_idx = (i[:, None] + j[None, :]).astype(np.int32).ravel()
    col_idx = np.repeat(j, n)
    data = np.tile(f, n)                                 # f repeated for each column
    # shape = (2n-1, n) – rows that fall outside are never accessed because
    # i+j ∈ [0, 2n‑2] automatically.
    A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimizer (sub‑gradient on the exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    n = len(a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and its maximal entry
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the (unique) active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size decays like O(1/√t)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        # keep numbers in a safe range
        log_a = np.clip(log_a, -30.0, 30.0)

        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap permutation annealing (swap only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 60_000) -> tuple[np.ndarray, float]:
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    T_start = cur_val
    T_end = cur_val * 1e-6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

    # Number of LP steps we will attempt – keep it modest to respect time.
    max_lp_iters = 12
    # Mixing coefficients (decrease to stay safe)
    t_schedule = [0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01, 0.005]

    for it in range(max_lp_iters):
        if time.time() > deadline - 150.0:   # reserve ~150 s for polishing/anneal
            break
        M = float(np.max(_conv_fft(f, f)))   # current max autocorrelation value

        # Build the convolution matrix A such that A @ g = conv(f,g)
        # This can be memory‑intensive; we free it as soon as possible.
        build_start = time.time()
        A = _build_conv_matrix(f)
        build_time = time.time() - build_start
        # print diagnostic
        print(f"[LP {it+1}] built matrix in {build_time:.2f}s (M={M:.6f})")

        # Right‑hand side: all entries ≤ M (tiny epsilon for numeric safety)
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=np.float64)

        # Objective: maximise sum(g)  → minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)

        # Bound each variable to be non‑negative, no explicit upper bound.
        bounds = [(0.0, None)] * n

        # Allocate a slice of the remaining budget for this LP.
        remaining = deadline - time.time()
        # Give a maximum of 30 s to any single LP (HiGHS respects the flag)
        lp_time_limit = min(30.0, remaining * 0.4)

        # Solve.
        lp_res = linprog(c,
                         A_ub=A,
                         b_ub=b_ub,
                         bounds=bounds,
                         method='highs',
                         options={'presolve': True,
                                  'time_limit': lp_time_limit,
                                  'dual_feasibility_tolerance': 1e-9,
                                  'primal_feasibility_tolerance': 1e-9,
                                  })

        # Discard matrix early to free memory.
        del A

        if lp_res.status != 0:
            # Not optimal (could be timeout or infeasibility) – stop LP phase.
            print(f"[LP {it+1}] linprog status {lp_res.status} – aborting LP phase.")
            break

        g0 = lp_res.x
        total = g0.sum()
        if total <= 0.0:
            print(f"[LP {it+1}] LP returned zero total mass – abort.")
            break
        g = g0 / total          # normalise to a probability vector

        # ------------------------------------------------------------------
        # 2. Mix the new g into the current f (convex combination).
        # ------------------------------------------------------------------
        t = t_schedule[min(it, len(t_schedule) - 1)]
        f_candidate = (1.0 - t) * f + t * g
        # maintain exact sum‑1
        f_candidate /= f_candidate.sum()

        new_val = _objective_surrogate(f_candidate)
        print(f"[LP {it+1}] mix t={t:.3f}  new surrogate = {new_val:.6f}")

        if new_val < best_val:
            best_val = new_val
            best_seq = f_candidate.copy()
            print(f"    *** Improved global best to {best_val:.6f}")

        # Accept the candidate only if it does not increase the max autocorr.
        # A slight tolerance is allowed to compensate for numeric noise.
        if new_val <= best_val * (1.0 + 1e-8):
            f = f_candidate
        else:
            # If mixing made things worse, reduce t for next round.
            print(f"    No improvement – keeping previous f.")
            # keep old f unchanged

    # ------------------------------------------------------------------
    # 3. Long exponentiated‑gradient polishing (use remaining budget).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(best_seq,
                                                 deadline,
                                                 max_iters=3_000_000,
                                                 eta0=6.0)
        if final_val < best_val:
            best_seq = final_seq
            best_val = final_val
            print(f"[Final EG] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Short permutation‑annealing tweak (few seconds).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 12.0
        seq_last, val_last = _perm_anneal(best_seq.copy(),
                                          short_deadline,
                                          max_iters=80_000)
        if val_last < best_val:
            best_seq = seq_last
            best_val = val_last
            print(f"[Final perm] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Evaluate with the provided `evaluate_sequence` and store.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:                    # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
