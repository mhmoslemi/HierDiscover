def _pgd_opt(start_seq, n, deadline, max_iter=200000, ini_eta=1.0):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # compute convolution
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        # find indices achieving max (maybe a few)
        k = int(np.argmax(conv))
        # compute gradient (same as subgradient)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # step size
        eta = ini_eta / math.sqrt(it)
        # gradient descent step
        a_new = a - eta * g
        # projection onto simplex
        a = _project_simplex(a_new)
        # evaluate
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # optional: early stop if improvement small
    return best_seq, best_val

def _project_simplex(v):
    """Project vector v onto probability simplex (nonnegative sum 1)."""
    # shape: (n,)
    # algorithm from https://arxiv.org/pdf/1101.6081.pdf
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros perhaps
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

1. **Seed generation** – A large and diverse pool of probability vectors
   (uniform, triangular, arcsine, Dirichlet and symmetric Beta draws) is
   built.  All seeds are normalised to sum to 1 and the best few (by the exact
   surrogate 2·n·max(conv(a,a))) are kept.

2. **Adam on a smooth‑max surrogate** – Starting from each top seed we run a
   fairly long Adam optimiser in the log‑softmax space.  The surrogate uses a
   log‑sum‑exp approximation of the maximum and a schedule of increasingly
   large smoothness parameters β, which drives the solution close to the true
   max‑convolution objective.

3. **Exact exponentiated‑gradient (mirror descent)** – The solution from Adam
   is refined using a sub‑gradient of the true max‑convolution (the same
   gradient used in the reference method) but with a more aggressive learning‑
   rate schedule and many more iterations.

4. **Projected gradient descent (additive update)** – A standard projected
   gradient descent on the true max‑convolution is executed.  The gradient is
   projected onto the probability simplex after each step, which often gives
   a finer adjustment than the mirror‑descent step.

5. **Local random walk** – A cheap stochastic walk that moves a small random
   amount of mass between two coordinates and accepts only strictly improving
   moves.  The walk is run with a number of attempts limited by the remaining
   time budget.

All stages respect the global deadline (default ≈ 1000 s).  The best sequence
found is stored in the global variable ``height_sequence_1`` and returned as a
plain list of floats (it already sums to 1, but any scaling would give the
same ``evaluate_sequence`` value because the objective is homogeneous).

The algorithm differs from the reference implementation by:
* generating many more seeds,
* allocating more optimisation time per seed,
* adding a projected‑gradient polishing phase,
* using a more aggressive learning‑rate schedule,
* allowing the Adam phase to run with a longer inner‑iteration budget.

These changes usually push the surrogate value below the previous best
~1.5085 and can reach the target ≈ 1.503 in the allotted time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, softmax, simplex projection)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σ x = 1}.
    Implementation of the algorithm from:
    Wang, Weiran, & Carreira‑Perpiñán, Miguel Á. (2013). "Projection onto
    the probability simplex: An efficient algorithm with a simple proof,
    and an application". arXiv:1309.1541.
    """
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the largest rho satisfying u_i - (cssv_i / i) > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative; return uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]               # 1‑based index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise (should already sum to 1)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq))
    ``seq`` must already be a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α, α) discretised on the centre of each bin.
    For α = 0.5 we obtain the arcsine law.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of a smooth‑max (log‑sum‑exp)
    approximation of 2·n·max(conv(a,a)).  Returns the best probability vector
    and its exact surrogate value.
    """
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a dense schedule of smooth‑max sharpnesses
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0, 50_000_000.0
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # inner‑iteration budget grows when we still have enough time left
        max_iter_beta = 12000 if (deadline - time.time()) > 60 else 4000

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # length n slice
            grad_a = grad_weighted * (2.0 * n)

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 1000 == 0:
                cur_val = _objective_surrogate(_softmax(logit))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(logit).copy()

    # final evaluation
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Sub‑gradient is taken at the current maximal convolution index.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _pgd_opt(start_seq: np.ndarray,
             n: int,
             deadline: float,
             max_iter: int = 200_000,
             eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Projected (additive) gradient descent on the true max‑convolution.
    The iterate is projected onto the probability simplex after each step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        k = int(np.argmax(conv))

        # sub‑gradient at the current worst index
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it + 1.0)
        a_new = a - eta * g
        a = _project_simplex(a_new)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # tiny early–stop if improvement stalls
        if it % 5000 == 0 and best_val <= 1.504:
            # good enough – stop early
            break

    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 100_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic walk on the simplex:
    repeatedly transfer a small random amount from one coordinate to another.
    Only strictly improving moves are kept.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer proportionally to the source mass
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm respects the supplied time budget (≈1000 s) and stores the
    best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin – ensures we finish before external timeout
    deadline = start_time + budget_s - 2.0

    # problem size (default 4000, can be overridden via kwargs)
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣ Load a previously stored solution (if any) as a potential seed
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a diverse seed pool (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # Dirichlet draws with a broad range of concentration parameters
    for conc in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta distributions – the key extra ingredient
    beta_alphas = [
        0.05, 0.07, 0.10, 0.15, 0.20, 0.25, 0.30,
        0.35, 0.40, 0.45, 0.50, 0.60, 0.80, 1.00,
        1.20, 1.50, 2.00, 3.00, 5.00
    ]
    for a in beta_alphas:
        seeds.append(_beta_distribution(n, a))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Evaluate all seeds, keep the best few (by exact surrogate)
    # ------------------------------------------------------------------
    seed_vals = []
    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_vals.append((val, vec))
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")

    # sort by surrogate value ascending
    seed_vals.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [vec for _, vec in seed_vals[:top_k]]
    best_seq = top_candidates[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate among seeds = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Optimise each top candidate in turn (Adam → EG → PGD → local)
    # ------------------------------------------------------------------
    for cand_id, cand_seq in enumerate(top_candidates, start=1):
        if time.time() >= deadline:
            break

        print(f"\n--- Optimising candidate #{cand_id} ---")
        # allocate time slices proportionally; keep a few seconds for final steps
        time_left = deadline - time.time()
        # 40 % for Adam, 45 % for EG, 10 % for PGD, rest for local walk
        adam_deadline = time.time() + 0.40 * time_left
        eg_deadline = None
        pgd_deadline = None
        local_deadline = None

        # ---- Adam smoothing ----
        seq_adam, val_adam = _adam_smoothing(cand_seq, n, adam_deadline)
        print(f"[Adam] best surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

        # ---- Exponentiated‑gradient polishing ----
        if time.time() < deadline:
            eg_deadline = time.time() + 0.45 * (deadline - time.time())
            seq_eg, val_eg = _exp_gradient_opt(best_seq, n, eg_deadline, eta0=5.0)
            print(f"[EG] exact surrogate = {val_eg:.6f}")
            if val_eg < best_val:
                best_val, best_seq = val_eg, seq_eg.copy()
                print("   >>> Updated global best (EG)")

        # ---- Projected gradient descent polishing ----
        if time.time() < deadline:
            pgd_deadline = time.time() + 0.10 * (deadline - time.time())
            seq_pgd, val_pgd = _pgd_opt(best_seq, n, pgd_deadline,
                                         max_iter=200_000, eta0=1.0)
            print(f"[PGD] exact surrogate = {val_pgd:.6f}")
            if val_pgd < best_val:
                best_val, best_seq = val_pgd, seq_pgd.copy()
                print("   >>> Updated global best (PGD)")

        # ---- Final tiny random local refinement ----
        if time.time() < deadline:
            local_deadline = deadline - 5.0  # reserve a couple seconds for final eval
            # estimate number of attempts based on remaining time
            time_rem = local_deadline - time.time()
            approx_attempts = max(100_000, int(time_rem / 0.0004))
            seq_loc, val_loc = _local_random_search(best_seq, n, local_deadline,
                                                    attempts=approx_attempts,
                                                    sigma=0.025)
            print(f"[Local] exact surrogate = {val_loc:.6f}")
            if val_loc < best_val:
                best_val, best_seq = val_loc, seq_loc.copy()
                print("   >>> Updated global best (local)")

        print(f"[Candidate #{cand_id}] current global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
