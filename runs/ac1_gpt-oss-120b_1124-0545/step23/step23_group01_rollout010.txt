def _adam_smooth_opt(start_seq, n, deadline, max_iters=5_000_000,
                     lr0=0.3, beta1=0.9, beta2=0.999, eps=1e-12,
                     alpha0=0.5, alpha_max=5000.0, alpha_ramp_iters=200_000):
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = _softmax(log_a)
        conv = _conv_fft(a, a)   # length 2n-1
        # compute surrogate
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # smoothing parameter
        if t <= alpha_ramp_iters:
            alpha = alpha0 + (alpha_max - alpha0) * t / alpha_ramp_iters
        else:
            alpha = alpha_max
        cmax = np.max(conv)
        # compute w
        exp_arg = alpha * (conv - cmax)
        # clip exponent to avoid overflow (though should be <=0)
        # Use np.exp directly
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / len(w))
        else:
            w = w / w_sum
        # gradient for smooth surrogate: grad = 2 * conv(w, a_rev)
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev)   # full conv length 3n-2
        grad = 2.0 * conv_w[n-1:n-1+n]
        # chain rule for softmax param
        dot = np.dot(grad, a)
        grad_v = a * (grad - dot)
        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)
        # bias correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

    return best_seq, best_val

"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Problem dimension** – fixed at n = 4000 (the size used in the
   reference solution).  The objective for a probability vector
   (∑a = 1) is 2·n·max(conv(a,a)).

2. **Initial pool** – a handful of diversified seeds:
   * the previous best vector (if it exists);
   * an exponential decay;
   * three random Dirichlet draws with a spiky concentration parameter.

3. **Fast Adam optimisation (sub‑gradient)** – each seed is refined
   by an Adam optimiser on the exact max‑convolution objective.
   The optimisation works on the log‑softmax parametrisation
   (`v = log(a)`, `a = softmax(v)`).  Only the active convolution
   entry (the current maximum) contributes to the gradient, which
   makes a single iteration very cheap (≈0.2 ms on a 4000‑vector).

4. **Long final Adam run** – the best seed found in step 3 is
   optimised for the remaining time budget with a much larger
   iteration budget.

5. **Exponentiated‑gradient polishing** – a classic EG pass
   (sub‑gradient descent on the non‑smooth objective) is applied to
   the best vector from the previous step.  This stage often yields
   the last few digits of improvement.

6. **Evaluation & storage** – the final vector is evaluated with the
   provided ``evaluate_sequence`` function, printed and stored in the
   global variable ``height_sequence_1`` for possible continuation
   runs.

All heavy work is done with FFT‑based convolutions (O(n log n)) and
the whole routine respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used for optimisation: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution objective
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 500_000,
              lr0: float = 0.2,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the non‑smooth max‑convolution
    objective. Returns the best probability vector seen and its
    surrogate value.
    """
    # initialise in log‑space
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        # convolution and location of the current maximum
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active entry w.r.t. a:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g_a = np.zeros_like(a)
        g_a[idx_i] = 2.0 * a[rev_idx]

        # chain‑rule: ∇_v = a * (g_a – (g_a·a))
        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # parameter update
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # evaluate surrogate and keep the best
        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (identical to the reference version)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

    start_time = time.time()

    # Problem dimension
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    candidates = []

    # ------------------------------------------------------------------

    best_overall_val = float("inf")

        print(f"  → surrogate after short Adam = {val:.6f}")

        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"  ** new global best (surrogate) = {best_overall_val:.6f}")

        remaining_seeds -= 1

    # ------------------------------------------------------------------
    # Long Adam optimisation on the current best
    # ------------------------------------------------------------------
    if best_overall_seq is not None:
        time_left = deadline - time.time()
        if time_left > 30.0:
            # heuristic: aim for ~0.8 ms per iteration → allocate as many as fit
            est_iter_time = 0.0008
            max_long_iters = int((time_left - 20.0) / est_iter_time)
            max_long_iters = max(max_long_iters, 200_000)   # ensure a decent budget
            print(f"\n--- Long Adam run (up to {max_long_iters} iters) ---")
            seq, val = _adam_opt(best_overall_seq, n, deadline,
                                 max_iters=max_long_iters,
                                 lr0=0.2, beta1=0.9, beta2=0.999, eps=1e-8)
            if val < best_overall_val:
                best_overall_val = val
                best_overall_seq = seq.copy()
                print(f"[Long Adam] improved surrogate = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with exponentiated‑gradient
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        best_overall_seq = _exp_decay(n, rate=0.001)

    if time.time() < deadline:
        print("\n--- Polishing with exponentiated‑gradient ---")
        polished_seq, polished_val = _exp_gradient_opt(
            best_overall_seq, n, deadline,
            max_iters=4_000_000, eta0=12.0)
        if polished_val < best_overall_val:
            best_overall_seq = polished_seq
            best_overall_val = polished_val
            print(f"[Polish] improved surrogate = {best_overall_val:.6f}")
        else:
            print(f"[Polish] no improvement (best surrogate unchanged)")

    # ------------------------------------------------------------------
    # Evaluate with the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_overall_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store for possible continuation runs
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
