def _smooth_adam_opt(start_seq, deadline, iters_per_beta=10000, beta_schedule=None):
    if beta_schedule is None: beta_schedule = [1,5,20,100,500,2000,8000,32000,128000,512000]
    n = len(start_seq)

    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1
            # compute a from log_a
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp
            # compute gradient of smoothed max:
            rev_a = a[::-1]
            # conv of rev_a with w yields length len= len(a)+len(w)-1 = n + (2n-1) -1 = 3n-2
            full = _conv_fft(rev_a, w)

            grad_a = 2.0 * full[n-1:n-1+n]
            grad_log = a * (grad_a - np.dot(grad_a, a))
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_log
            v = beta2 * v + (1 - beta2) * (grad_log ** 2)

            v_hat = v / (1 - beta2 ** t)

"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. **Rich seed pool** – build a large set of analytically‑generated
   candidates (uniform, exponential decay, power‑law, shifted power‑law,
   a “U‑shaped’’ sum of two power‑laws, beta‑shaped, Dirichlet draws,
   plus the previous solution).

2. **Coarse selection** – evaluate the cheap surrogate
   `2·n·max(conv(a,a))` (the vector is ℓ₁‑normalised, so this equals the
   true objective) for every seed and keep the 5 best.

3. **Refinement** – for each of the top seeds run two optimisation
   phases under the remaining time budget:

   * **Smooth Adam** – minimise a log‑sum‑exp smoothing of the max‑convolution.
     The smoothing parameter β is increased exponentially (up to 5·10⁵)
     and Adam’s adaptive steps are used.  The best iterate is retained.

   * **Exact exponentiated‑gradient (mirror‑descent)** – a sub‑gradient
     method on the true max‑convolution (the active index gives the
     sub‑gradient).  The step‑size decays like `η₀/√t`.

4. **Final polishing** – a long run of the exact exponentiated‑gradient
   on the current global best, followed by a brief permutation‑annealing
   (swap moves) to escape any remaining local minima.

All vectors are kept in the log‑softmax (exponentiated‑gradient) or
simple‑softmax (Adam) representation, guaranteeing non‑negativity and
unit ℓ₁‑sum throughout.  The program respects the supplied time budget
(≈1000 s) and finally stores the best sequence in the global variable
`height_sequence_1`.

import time
import math

# ----------------------------------------------------------------------
# Low‑level FFT convolution (real, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))   (seq must sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^{-p} , i = 0…n‑1, δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _u_power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """U‑shaped: sum of a shifted power‑law and its mirror."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p) + (n - 1 - i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `conc`."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

                     beta_schedule=None) -> tuple[np.ndarray, float]:

    if beta_schedule is None:

    v = np.zeros_like(log_a)

    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12

            # smooth max via log‑sum‑exp (temperature = 1/beta)
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            w = exp_beta / np.sum(exp_beta)        # soft‑max weights

            # gradient of the smooth max w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)   # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # convert to gradient on log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            # keep logs bounded – prevents overflow
            log_a = np.clip(log_a, -40.0, 40.0)

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient optimiser (mirror‑descent on the true max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # initialise log‑probabilities (log‑softmax)
    log_a = np.log(a + 1e-30)

    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and index of the active diagonal (the current max)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (average of all contributions)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing step‑size
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g

        # re‑exponentiate and renormalise (log‑softmax)
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Pure permutation optimiser – swaps two entries."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        accept = val < cur_val or (rng.random() < math.exp(-(val - cur_val) / T))
        if accept:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The best sequence is stored in the global
    variable ``height_sequence_1`` and also returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a couple of seconds as a safety margin
    deadline = start_time + budget_s - 2.0

    # problem size – 4000 is a good trade‑off between quality and speed
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution (if available) – useful for continuation
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                s = prev_vec.sum()
                if s > 0:
                    prev_vec = prev_vec / s
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build a very rich pool of seeds
    # ------------------------------------------------------------------
    seeds = []

    # basic analytic families
    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.0015))
    seeds.append(_exp_decay(n, rate=0.0035))
    seeds.append(_sym_exp_decay(n, rate=0.0025))

    # shifted power‑law family – dense grid
    p_vals = np.linspace(0.50, 0.62, 25)          # 0.50 … 0.62
    delta_vals = np.geomspace(0.01, 10.0, 20)    # 0.01 … 10 (log‑spaced)
    for p in p_vals:
        for delta in delta_vals:
            seeds.append(_power_shift(n, p, delta))
            seeds.append(_u_power_shift(n, p, delta))

    # beta‑shaped families (randomised)
    for _ in range(12):
        a = rng.uniform(0.2, 3.0)
        b = rng.uniform(0.2, 3.0)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws – both sparse and dense
    for conc in (0.02, 0.05, 0.1, 0.3, 0.7, 1.5, 3.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution and a few shuffled variants
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(4):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation, keep the best few
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Refinement of the top seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget (roughly equal parts)
        remaining = deadline - time.time()
        # give about 20 % of the remaining time to the current seed,
        # leaving the rest for later seeds and final polishing
        slice_time = remaining * 0.20 / max(1, (top_k - idx))
        sub_deadline = time.time() + slice_time
        print(f"\n[Seed {idx+1}] allocated time ≈ {slice_time:.1f}s")

        # ---- smoothed Adam optimisation (fast improvement) ----
        seq_adam, val_adam = _adam_smooth_opt(seed_vec, sub_deadline,
                                              iters_per_beta=12_000)

        # ---- exact EG polish (uses remaining time of the slice) ----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_adam, sub_deadline,
                                               max_iters=800_000,
                                               eta0=6.0)
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # pick the better of the two refinements
        if val_adam < val_eg:
            local_seq, local_val = seq_adam, val_adam
        else:
            local_seq, local_val = seq_eg, val_eg

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long EG polishing (use all remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=2_500_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation tweak (may shave a bit more)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + min(30.0, deadline - time.time())
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=100_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
