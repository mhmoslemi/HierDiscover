  inverse‑sqrt, shifted power‑law families, random Dirichlet, previous best).
- Picks the best seed (lowest surrogate 2·n·max(conv) on the ℓ₁‑normalised
  vector).
- Runs a long single‑run optimisation pipeline on that seed:
    * Exponentiated‑gradient on the exact max‑convolution (Adam‑like momentum).
    * Smooth‑max L‑BFGS with a fine β‑schedule.
    * Flattening gradient descent (quadratic loss on the whole convolution).
    * Simulated‑annealing polish.
- After polishing makes a few random perturbation restarts to escape
  possible local minima.
- Returns the best sequence found (stored globally as ``height_sequence_1``).

All stages respect the overall 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np

# optional imports
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ---------------------------------------------------------------------------
# FFT utilities --------------------------------------------------------------
# ---------------------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ---------------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1) -------------------
# ---------------------------------------------------------------------------

def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (requires sum(seq)=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ---------------------------------------------------------------------------
# Analytic seed families ----------------------------------------------------
# ---------------------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _inverse_sqrt(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i)
    return a / a.sum()


def _power_offset(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ 1/(i+offset)^p , i=0..n-1."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / ((i + offset) ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ---------------------------------------------------------------------------
# Optimisation primitives ----------------------------------------------------
# ---------------------------------------------------------------------------

def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            deadline: float,
                            max_iters: int = 2_500_000,
                            eta0: float = 12.0,
                            tol: float = 1e-10) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient on the exact max‑convolution.
    Uses all indices within a relative tolerance of the current maximum.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    # Adam‑like momentum
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # gradient = 2 * (a * indicator of active) convolved with a
        ind = np.zeros(2 * n - 1, dtype=np.float64)
        ind[active] = 1.0
        # correlation (a ⋆ ind) = conv(a, ind[::-1])
        grad = 2.0 * _conv_fft(a, ind[::-1])
        # extract the part that aligns with a (indices n-1 … n-1+n-1)
        g = grad[n - 1:n - 1 + n]

        # Adam‑like momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           deadline: float,
                           max_iter_per_beta: int = 4000) -> np.ndarray:
    """Smooth‑max (log‑sum‑exp) optimiser with L‑BFGS‑B."""
    if _opt is None:
        return start_vec

    beta_schedule = [
        0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
        80.0, 200.0, 500.0, 1_000.0, 2_000.0,
        5_000.0, 10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
    ]

    v = np.log(start_vec + 1e-30)  # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * len(a) * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[len(a) - 1:len(a) - 1 + len(a)]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(
                fun=fun,
                x0=v,
                method='L-BFGS-B',
                jac=True,
                callback=callback,
                options={'maxiter': max_iter_per_beta,
                         'disp': False,
                         'ftol': 1e-12}
            )
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


def _flatten_gradient_opt(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 1_200_000,
                         eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on Σ (conv‑avg)² to flatten the whole convolution."""
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)  # length of full convolution

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom  # = 1/denom when sum=1
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # Mirror‑descent step on simplex
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 500_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    """Simple SA with Gaussian perturbations and linear cooling."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=len(a))
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ---------------------------------------------------------------------------
# Full optimisation pipeline -------------------------------------------------
# ---------------------------------------------------------------------------

def _full_optimisation(start_vec: np.ndarray,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run EG → L‑BFGS → flatten → SA on a copy of *start_vec*."""
    # 1) exponentiated‑gradient (max‑convolution)
    seq_eg, val_eg = _exp_gradient_opt_multi(
        start_vec.copy(),
        deadline,
        max_iters=3_000_000,
        eta0=14.0,
        tol=1e-10,
    )

    # 2) smooth L‑BFGS
    if time.time() < deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(
            seq_eg,
            deadline,
            max_iter_per_beta=5000,
        )
        val_lbfgs = _objective_surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, val_eg

    # 3) flattening gradient
    if time.time() < deadline:
        seq_flat, val_flat = _flatten_gradient_opt(
            seq_lbfgs,
            deadline,
            max_iters=1_500_000,
            eta0=10.0,
        )
    else:
        seq_flat, val_flat = seq_lbfgs, val_lbfgs

    # 4) simulated annealing polish
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            seq_flat,
            deadline,
            rng,
            max_attempts=600_000,
            start_sigma=0.002,
            T_start=1e-2,
            T_end=1e-9,
        )
    else:
        seq_sa, val_sa = seq_flat, val_flat

    candidates = [
        (start_vec, _objective_surrogate(start_vec)),
        (seq_eg, val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_flat, val_flat),
        (seq_sa, val_sa),
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val


# ---------------------------------------------------------------------------
# Main entry point ---------------------------------------------------------
# ---------------------------------------------------------------------------

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine spends almost the whole budget on a single, carefully
    optimised run starting from the best analytic seed.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    n = 4000

    # -----------------------------------------------------------------------
    # 1) Build a focused seed set (analytic + random)
    # -----------------------------------------------------------------------
    seeds = []

    # analytic basics
    seeds.append(_uniform(n))
    seeds.append(_inverse_sqrt(n))

    # shifted power‑law family (p around 0.5, offset up to 5)
    for p in np.arange(0.35, 0.85 + 1e-12, 0.05):
        for off in [0.5, 1.0, 2.0, 4.0]:
            seeds.append(_power_offset(n, p, off))

    # random Dirichlet draws with a spread of concentrations
    for conc in [0.001, 0.005, 0.02, 0.1, 0.5]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous best (if available) – good continuation point
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape == (n,):
                seeds.append(prev_arr / prev_arr.sum())
        except Exception:
            pass

    # -----------------------------------------------------------------------
    # 2) Quick evaluation of seeds (all ℓ₁‑normalised)
    # -----------------------------------------------------------------------
    best_seed = None
    best_surrogate = float("inf")
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        if val < best_surrogate:
            best_surrogate = val
            best_seed = vec.copy()

    if best_seed is None:
        best_seed = _uniform(n)  # fallback

    print(f"\nInitial best surrogate from seeds: {best_surrogate:.6f}")

    # -----------------------------------------------------------------------
    # 3) Long single‑run optimisation on the best seed
    # -----------------------------------------------------------------------
    seq_opt, val_opt = _full_optimisation(best_seed, deadline, rng)
    print(f"After main optimisation: surrogate = {val_opt:.6f}")

    # -----------------------------------------------------------------------
    # 4) A few random‑restart polish rounds (tiny perturbations + short EG)
    # -----------------------------------------------------------------------
    for restart in range(3):
        if time.time() >= deadline:
            break
        # small random perturbation of the current best
        pert = seq_opt + 0.004 * rng.normal(size=n)
        pert = np.maximum(pert, 0.0)
        if pert.sum() == 0.0:
            continue
        pert /= pert.sum()

        # short EG run (few thousand iterations) just to escape flat regions
        sub_deadline = time.time() + min(30.0, deadline - time.time())
        seq_r, val_r = _exp_gradient_opt_multi(
            pert,
            sub_deadline,
            max_iters=200_000,
            eta0=8.0,
            tol=1e-9,
        )
        if val_r < val_opt:
            seq_opt, val_opt = seq_r, val_r
            print(f"   Restart {restart+1}: improved surrogate = {val_opt:.6f}")

    # -----------------------------------------------------------------------
    # 5) Final evaluation with the provided ``evaluate_sequence``
    # -----------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(seq_opt.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {val_opt:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = seq_opt.tolist()
    return seq_opt.tolist()
