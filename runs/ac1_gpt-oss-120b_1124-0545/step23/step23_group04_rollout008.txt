def _lp_refine(seq, deadline, max_active=200):
    # iterative refinement using linear programming
    cur_seq = seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    while time.time() < deadline:
        conv = _conv_fft(cur_seq, cur_seq)
        M = np.max(conv)
        # Identify active indices close to M (within tolerance)
        # tolerance set relative: tol = max(1e-12, M * 1e-10)
        active = np.where(conv >= M - tol)[0]
        if active.size == 0:
            break
        if active.size > max_active:
            # keep only those with highest conv values
            idxs = np.argsort(conv[active])[-max_active:]  # take top counts
            active = active[idxs]
        # Build LP matrices
        n = len(cur_seq)
        rows = []
        for k in active:
            row = np.zeros(n, dtype=np.float64)
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # fill
            for i in range(i_start, i_end + 1):
                j = k - i
                row[j] = cur_seq[i]
            rows.append(row)
        A = np.vstack(rows)
        b = np.full(A.shape[0], M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximize sum(b)
        bounds = [(0, None)] * n
        # solve
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        except Exception as e:
            print("LP solve failed:", e)
            break
        if not res.success:
            # cannot improve; break
            break
        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            break
        g = b_opt / sum_b
        # try mixing with step sizes
        improved = False
        for t in (0.5, 0.3, 0.1):
            cand = (1 - t) * cur_seq + t * g
            cand_val = _objective_surrogate(cand)
            if cand_val < cur_val:
                cur_seq = cand
                cur_val = cand_val
                improved = True
                break
        if not improved:
            # cannot improve via mixing, break
            break
    return cur_seq, cur_val

"""
Propose a step‑function (non‑negative sequence) that minimises the
provided ``evaluate_sequence`` function.

Algorithm Overview
------------------
1. **Seed Generation** – a large, diverse pool (uniform, arcsine,
   triangular, many symmetric Beta shapes, exponential‑decay, cosine
   shapes and Dirichlet draws).  Each seed is normalised to sum = 1 and
   its cheap surrogate ``2·n·max(conv(p,p))`` is evaluated.

2. **Selection** – keep the best few seeds.

3. **Adam smoothing** – optimise a smooth log‑sum‑exp approximation of the
   max‑convolution (the “soft‑max”).  The smoothness parameter β is raised
   gradually; Adam adapts the learning‑rate.

4. **Exact mirror‑descent (exponentiated‑gradient)** – minimise the true
   max‑convolution using the sub‑gradient at the active index.  Step‑size
   decays as η₀/√t.

5. **Simulated annealing** – a stochastic walk on the simplex that
   accepts occasional uphill moves; the temperature decays exponentially.
   This helps escape shallow local minima.

6. **Linear‑programming refinement** – for the current distribution *p*
   compute the active set of indices where the convolution attains its
   maximum M.  Solve a linear programme

        maximise   Σ b_i
        s.t.       (p * b)_k ≤ M   for every active k
                   b_i ≥ 0

   The optimiser gives a direction that cannot increase the current
   maximum.  Mix the normalised LP solution into *p* (convex combination)
   and keep the improvement.

7. **Final tiny random walk** – polish the result with a short local
   search.

All stages respect a global time‑budget (default 1000 s).  The best
sequence found is stored in ``height_sequence_1`` and returned as a plain
list of floats.
"""

import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax producing a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)).  The input is assumed to sum≈1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on the discrete set {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    return a / a.sum()


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()


def _cosine_shape(n: int) -> np.ndarray:
    """Cosine‑squared bump centred in the interval."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (i - (n - 1) / 2) / n) ** 2
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _adam_smoothing(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth (log‑sum‑exp) surrogate."""
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a reasonably long schedule – can be stopped early by the deadline
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
        1_000.0, 2_000.0, 5_000.0, 20_000.0,
        50_000.0, 100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        max_iter_beta = 8000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break
            a = _softmax(logit)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)   # outer factor 2·n

            # convert to gradient in log‑softmax space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.12
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # periodic exact evaluation
            if it % 600 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val
    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 5.0,
                     max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % 5000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val
    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         deadline: float,
                         sigma: float = 0.02,
                         init_temp: float = 1e-3,
                         cooling: float = 0.995) -> tuple[np.ndarray, float]:
    """Stochastic walk on the simplex with a cooling schedule."""
    n = len(start_seq)
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    T = init_temp
    steps = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * cur_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        new_val = _objective_surrogate(new_seq)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_seq = new_seq
            cur_val = new_val
            if cur_val < best_val:
                best_seq, best_val = cur_seq, cur_val
        steps += 1
        if steps % 2000 == 0:
            T *= cooling
            if T < 1e-12:
                T = 1e-12
    return best_seq, best_val


def _lp_refine(start_seq: np.ndarray,
               deadline: float,
               max_active: int = 200) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    Maximise Σ b_i while keeping (p * b)_k ≤ M for every k where the
    current convolution attains its maximum M.
    The optimiser b is normalised to a probability vector and mixed
    back into p.
    """
    if not _SCIPY_AVAILABLE:          # pragma: no cover
        return start_seq, _objective_surrogate(start_seq)

    n = len(start_seq)
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    while time.time() < deadline:
        conv = _conv_fft(cur, cur)
        M = np.max(conv)

        # active set – points whose value is within a tiny absolute tolerance
        tol = max(1e-14, M * 1e-12)
        active = np.where(conv >= M - tol)[0]
        if active.size == 0:
            break
        # limit the number of constraints for speed
        if active.size > max_active:
            # keep the indices with the largest values (they are already ≈M)
            idxs = np.argsort(conv[active])[-max_active:]
            active = active[idxs]

        # Build the inequality matrix A (shape |active| × n)
        rows = []
        for k in active:
            row = np.zeros(n, dtype=np.float64)
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # fill the row:   (p * b)_k = Σ_i p_i b_{k-i}
            for i in range(i_start, i_end + 1):
                j = k - i
                row[j] = cur[i]
            rows.append(row)
        A = np.vstack(rows)
        b_ub = np.full(A.shape[0], M, dtype=np.float64)

        # maximise Σ b_i  → minimise −Σ b_i
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception as exc:          # pragma: no cover
            print("LP solve exception:", exc)
            break

        if not res.success:
            # No feasible improvement
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0.0:
            break
        g = b_opt / sum_b                     # normalised direction

        improved = False
        # try a few mixing coefficients, favouring a sizeable step
        for t in (0.5, 0.3, 0.1):
            cand = (1.0 - t) * cur + t * g
            cand_val = _objective_surrogate(cand)
            if cand_val < cur_val:
                cur = cand
                cur_val = cand_val
                improved = True
                break
        if not improved:
            # mixing cannot improve – stop
            break
    return cur, cur_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """Tiny stochastic walk on the simplex (only improving moves are kept)."""
    n = len(start_seq)
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found within the allotted budget is stored globally as
    ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 0️⃣  Load previously stored solution (if it matches the length)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Assemble a very diverse seed pool
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]

    # Beta‑shaped seeds – a finer sweep than before
    for alpha in np.linspace(0.1, 3.0, 30):
        seeds.append(_beta_distribution(n, alpha))

    # Exponential‑decay shapes
    for rate in (0.001, 0.002, 0.005, 0.01, 0.02):
        seeds.append(_exp_decay(n, rate))

    # Cosine bump
    seeds.append(_cosine_shape(n))

    # Dirichlet draws with several concentrations
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣  Initial cheap evaluation of seeds (surrogate)
    # ------------------------------------------------------------------
    seed_infos = []
    print("\n--- Initial seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f" seed #{idx + 1:03d}: surrogate = {val:.6f}")

    seed_infos.sort(key=lambda x: x[0])
    top_k = 5
    top_seeds = [vec for _, vec in seed_infos[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Time allocation (fractions of the remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    adam_frac = 0.22
    eg_frac   = 0.44
    sa_frac   = 0.16
    lp_frac   = 0.12
    loc_frac  = 0.06

    # ------------------------------------------------------------------
    # 4️⃣  Adam smoothing (≈adam_frac of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = time.time() + remaining * adam_frac
        print("\n--- Adam smoothing (smooth max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_seq, best_val = seq_adam, val_adam
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 5️⃣  Exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = time.time() + remaining * eg_frac
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, eg_deadline,
                                           eta0=5.0, max_iters=8_000_000)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_seq, best_val = seq_eg, val_eg
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 6️⃣  Simulated annealing (stochastic escape)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + remaining * sa_frac
        print("\n--- Simulated annealing (stochastic walk) ---")
        seq_sa, val_sa = _simulated_annealing(best_seq, sa_deadline,
                                              sigma=0.025,
                                              init_temp=1e-3,
                                              cooling=0.996)
        print(f"[SA] exact surrogate = {val_sa:.6f}")
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print("   >>> Updated global best (SA)")

    # ------------------------------------------------------------------
    # 7️⃣  Linear‑programming refinement
    # ------------------------------------------------------------------
    if time.time() < deadline and _SCIPY_AVAILABLE:
        lp_deadline = time.time() + remaining * lp_frac
        print("\n--- Linear‑programming refinement (active‑set LP) ---")
        seq_lp, val_lp = _lp_refine(best_seq, lp_deadline, max_active=250)
        print(f"[LP] exact surrogate = {val_lp:.6f}")
        if val_lp < best_val:
            best_seq, best_val = seq_lp, val_lp
            print("   >>> Updated global best (LP)")

    # ------------------------------------------------------------------
    # 8️⃣  Final tiny random local refinement (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        local_deadline = deadline
        print("\n--- Final local random refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, local_deadline,
                                                attempts=250_000,
                                                sigma=0.015)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc, val_loc
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 9️⃣  Official evaluation with the supplied ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print("\n=== Finished ===")
        print(f"Surrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
