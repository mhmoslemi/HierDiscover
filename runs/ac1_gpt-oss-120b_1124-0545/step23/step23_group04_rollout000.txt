def propose_candidate(seed=..., budget_s=1000, n=4000):
    start_time = time.time()
    deadline = start_time + budget_s - 2

    rng = np.random.default_rng(seed)

    # load prev solution if any
    prev = globals().get('height_sequence_1')
    ...

    # generate seeds:
    seeds = []
    seeds.append(_uniform(n))
    seeds.append(_arcsine(n))
    seeds.append(_triangular(n))
    # Beta
    for alpha in [0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.7, 0.9, 1.2, 1.5, 2.0]:
        seeds.append(_beta_distribution(n, alpha))
    # Dirichlet draws
    for conc in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixture seeds:
    arcsine = _arcsine(n)
    uniform = _uniform(n)
    for lam in np.linspace(0, 1, 11):
        seeds.append(lam * arcsine + (1-lam) * uniform)

    # mixture arcsine + beta(0.2)
    beta_u = _beta_distribution(n, 0.2)
    for lam in np.linspace(0, 1, 11):
        seeds.append(lam * arcsine + (1-lam) * beta_u)

    # mixture arcsine + triangular
    triang = _triangular(n)
    for lam in np.linspace(0, 1, 11):
        seeds.append(lam * arcsine + (1-lam) * triang)

    if prev_vec is not None:
        seeds.append(prev_vec)

    # evaluate seeds surrogate; select top candidates
    seed_infos = [( _objective_surrogate(v/ v.sum()), v/ v.sum()) for v in seeds if time.time() < deadline]
    seed_infos.sort(key=lambda x: x[0])
    best_seq = seed_infos[0][1]
    best_val = seed_infos[0][0]

    # allocate time for detailed optimization of top few seeds
    top_k = min(5, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]

    # For each top seed, run adam smoothing + EG + local refine, track best overall
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        print(f"\n--- Optimising seed {idx+1}/{top_k} ---")
        # allocate a slice of time proportionally: allocate remaining time equally among seeds
        remaining_time = deadline - time.time()
        per_seed_budget = remaining_time / (top_k - idx)
        seed_deadline = time.time() + per_seed_budget * 0.9 # keep margin

        # Adam smoothing
        seq_adam, val_adam = _adam_smoothing(seed_vec, seed_deadline)
        # Use best from Adam as starting point for EG
        start_seq = seq_adam if val_adam < best_val else seed_vec
        start_val = min(val_adam, best_val)

        # EG polishing using remaining time portion
        eg_deadline = time.time() + per_seed_budget * 0.7
        seq_eg, val_eg = _exp_gradient_opt(start_seq, eg_deadline)
        # compare
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print(f"   >>> New global best after EG: {best_val:.6f}")

        # final local refinement
        local_deadline = deadline - 0.5
        seq_loc, val_loc = _local_random_search(best_seq, local_deadline)
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"   >>> New global best after local: {best_val:.6f}")

    # final evaluation via evaluate_sequence
    true_val = evaluate_sequence(best_seq.tolist())
    print(f"\n=== Finished ===\nBest surrogate = {best_val:.6f}, eval = {true_val:.6f}")
    globals()["height_sequence_1"] = best_seq.tolist()

"""
Optimise a non‑negative step‑function to minimise evaluate_sequence().
The routine builds a large, diverse seed pool (including mixtures of known
shapes), evaluates all seeds with the cheap surrogate 2·n·max(conv(a,a)),
selects the best few, and refines each of them with three stages:

1️⃣  Adam on a smooth (log‑sum‑exp) approximation of the max‑convolution.
    The β‑schedule is richer and the inner iteration limit is larger.

2️⃣  Exponentiated‑gradient (mirror‑descent) on the true max‑convolution.
    Runs until the per‑seed time budget expires.

3️⃣  A stochastic walk on the simplex (tiny random local search) to polish
    the final result.

All stages are time‑aware: the total budget (default 1000 s) is split among
the top seeds.  The best sequence found is stored in the global variable
``height_sequence_1`` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)).  Input must sum to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the discrete set {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth log‑sum‑exp surrogate of max(conv).
    Returns the best probability vector and its exact surrogate value.
    """
    n = len(start_seq)
    # initialise logits safely
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # richer β‑schedule reaching up to 1e9
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0, 50_000_000.0,
        100_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        max_iter_beta = 15000  # longer inner loops
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                # current vector on simplex
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights (log‑sum‑exp)
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            # convolution of rev_a with w gives gradient of each a_i
            full = _conv_fft(rev_a, w)
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # same length as a
            grad_a = grad_weighted * (2.0 * n)            # outer factor

            # convert to gradient in log‑softmax space (mirror descent)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam updates
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.12
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final evaluation
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 5.0,
                     max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Exact exponentiated‑gradient (mirror descent) on the true max‑convolution.
    Runs until ``deadline`` or ``max_iters``.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size diminishing like 1/sqrt(t)
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        if it % 5000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    # final check
    final_val = _objective_surrogate(a)
    if final_val < best_val:
        best_seq, best_val = a, final_val

    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 500_000,
                         sigma: float = 0.015) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex.
    Moves a small random amount of mass from one coordinate to another,
    accepting only improving moves.
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # draw a relative delta proportional to mass at i
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

def propose_candidate(seed: int = 123,

    """

    # --------------------------------------------------------------

    return best_seq.tolist()
