"""
Improved step‑function optimisation.

The algorithm proceeds in four stages:

0️⃣  Build a diverse pool of seed distributions (uniform, arcsine, β‑Beta,
    Dirichlet draws, triangular, power‑law, plus the previous solution if
    available) and pick the best according to the exact surrogate
    2·n·max(conv(a,a)).

1️⃣  Continuation Adam optimisation on a smooth log‑sum‑exp approximation
    of the max‑convolution.  The smoothness parameter β is increased
    gradually from 1 up to 1e9 while Adam‑type momentum updates are applied.
    This yields a high‑quality approximation of the sub‑gradient of the true
    max‑convolution.

2️⃣  High‑β Adam descent on the *exact* max‑convolution (i.e. the limit
    β→∞ of the previous stage).  The weight vector is essentially a
    one‑hot at the active index, so the update follows the true sub‑gradient.
    Adam‑style momentum greatly accelerates convergence.

3️⃣  Intensive greedy stochastic walk on the simplex with a simple
    simulated‑annealing schedule.  Randomly transfers a small amount of mass
    between two coordinates; moves that improve the surrogate are kept,
    occasional uphill moves are accepted according to a decaying temperature.

The best sequence found is stored globally as ``height_sequence_1`` and
returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate: 2·n·max(conv(seq,seq)).
    The input is assumed to sum to 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a *probability* vector summing to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _power_law(n: int, p: float = 0.5) -> np.ndarray:
    """a_i ∝ (i+1)^p, normalised."""
    a = np.arange(1, n + 1, dtype=np.float64) ** p
    a /= a.sum()
    return a

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_continuation_opt(start_seq: np.ndarray,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth log‑sum‑exp approximation of max(conv).
    The smoothness β is increased in a continuation schedule up to 1e9.
    Returns (best_seq, best_surrogate_value).
    """
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)          # initialise logits
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0

    best_seq = _softmax(logit).copy()
    best_val = _objective_surrogate(best_seq)

    # continuation schedule – more points in the high‑β region
    betas = [
        1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
        1_000.0, 2_000.0, 5_000.0,
        20_000.0, 50_000.0, 100_000.0,
        200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0,
        5_000_000.0, 10_000_000.0,
        20_000_000.0, 50_000_000.0,
        100_000_000.0, 200_000_000.0,
        500_000_000.0, 1_000_000_000.0,
    ]

    for beta in betas:
        if time.time() > deadline:
            break

        # allocate a modest number of inner iterations; the later betas get
        # fewer because they are more expensive (the softmax becomes near‑one‑hot)
        max_iter_beta = 8000 if beta < 1e6 else 3000

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            it += 1
            a = _softmax(logit)

            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad = grad_weighted * (2.0 * n)      # outer factor 2·n

            # Adam update in log‑softmax space
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.12
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # periodic exact evaluation
            if it % 4000 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final evaluation
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _adam_max_subgrad_opt(start_seq: np.ndarray,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the *exact* max‑convolution (β → ∞).
    Uses the sub‑gradient of the active index (or a tiny active set)
    and Adam‑style momentum.
    Returns (best_seq, best_surrogate_value).
    """
    n = len(start_seq)
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0

    best_seq = _softmax(logit).copy()
    best_val = _objective_surrogate(best_seq)

    # active‑set tolerance (absolute)
    act_tol = 1e-12

    while time.time() < deadline and it < 5_000_000:
        it += 1
        a = _softmax(logit)

        conv = _conv_fft(a, a)
        max_c = np.max(conv)

        # indices with value within act_tol of the maximum
        active = np.where(conv >= max_c - act_tol)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)], dtype=int)

        # weight vector: uniform over the active set (soft max with huge β)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # sub‑gradient for the (averaged) active constraints
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_weighted = 2.0 * full[n - 1:n - 1 + n]
        grad = grad_weighted * (2.0 * n)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing step size
        eta0 = 5.0
        eta = eta0 / math.sqrt(it)

        logit = logit - eta * m_hat / (np.sqrt(v_hat) + eps)

        if it % 5000 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    # final evaluation
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _local_random_anneal(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 300_000,
                         sigma: float = 0.015,
                         T_start: float = 1e-3,
                         T_end: float = 1e-7) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic walk with a simple exponential cooling schedule.
    Moves that improve the surrogate are always kept; uphill moves are
    accepted with probability exp(-(Δ)/T).  Returns (best_seq, best_val).
    """
    n = len(start_seq)
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    T = T_start
    step = 0
    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        step += 1

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # decide direction of transfer (donor -> receiver)
        if best_seq[i] > best_seq[j]:
            donor, recv = i, j
        else:
            donor, recv = j, i

        delta = sigma * best_seq[donor] * rng.random()
        if delta <= 0.0 or best_seq[donor] < delta:
            continue

        new_seq = best_seq.copy()
        new_seq[donor] -= delta
        new_seq[recv] += delta

        cur_val = _objective_surrogate(new_seq)
        if cur_val < best_val:
            best_seq = new_seq
            best_val = cur_val
        else:
            # occasional uphill acceptance
            prob = math.exp(-(cur_val - best_val) / T) if T > 0 else 0.0
            if rng.random() < prob:
                best_seq = new_seq
                best_val = cur_val

        # geometric cooling
        if step % 5000 == 0:
            T = max(T_end, T * 0.90)

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 6000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found within the allotted budget is stored in the global
    variable ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Load any previously stored solution (if length matches)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if isinstance(prev, (list, tuple, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            pass

    # --------------------------------------------------------------
    # 0️⃣  Assemble a diverse seed pool
    # --------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
        _power_law(n, p=0.5),
        _power_law(n, p=0.8),
    ]

    # Beta‑shaped seeds
    for α in (0.3, 0.5, 0.7, 0.9, 1.2, 1.5, 2.0, 3.0):
        seeds.append(_beta_distribution(n, α))

    # Dirichlet draws with several concentration parameters
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 1️⃣  Quick evaluation of seeds (exact surrogate)
    # --------------------------------------------------------------
    seed_infos = []
    print("\n--- Initial seed evaluation (exact surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        seed_infos.append((val, vec))
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")

    # pick the best few seeds
    seed_infos.sort(key=lambda x: x[0])
    top_k = min(5, len(seed_infos))
    top_seeds = [vec for _, vec in seed_infos[:top_k]]
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Adam continuation on smooth max
    # --------------------------------------------------------------
    if time.time() < deadline:
        cont_deadline = time.time() + (deadline - time.time()) * 0.35
        print("\n--- Adam continuation (smooth max) ---")
        seq_cont, val_cont = _adam_continuation_opt(best_seq, cont_deadline)
        print(f"[Cont] best surrogate = {val_cont:.6f}")
        if val_cont < best_val:
            best_seq, best_val = seq_cont, val_cont
            print("   >>> Updated global best (continuation)")

    # --------------------------------------------------------------
    # 3️⃣  Adam on exact max (sub‑gradient)
    # --------------------------------------------------------------
    if time.time() < deadline:
        max_deadline = deadline - 5.0                # reserve a few seconds
        print("\n--- Adam on exact max (sub‑gradient) ---")
        seq_max, val_max = _adam_max_subgrad_opt(best_seq, max_deadline)
        print(f"[Max] best surrogate = {val_max:.6f}")
        if val_max < best_val:
            best_seq, best_val = seq_max, val_max
            print("   >>> Updated global best (max‑subgrad)")

    # --------------------------------------------------------------
    # 4️⃣  Final stochastic local refinement (annealed walk)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final stochastic local refinement (annealing) ---")
        seq_loc, val_loc = _local_random_anneal(best_seq, deadline,
                                                attempts=300_000,
                                                sigma=0.015,
                                                T_start=1e-3,
                                                T_end=1e-8)
        print(f"[Local] best surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_seq, best_val = seq_loc, val_loc
            print("   >>> Updated global best (local)")

    # --------------------------------------------------------------
    # 5️⃣  Official evaluation with the supplied ``evaluate_sequence``
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
