Hybrid search for a non‑negative step‑function (probability vector) that
minimises ``evaluate_sequence``.

The algorithm consists of:

1.   Build a diverse pool of analytic and random seed vectors (uniform,
     exponential, power‑law, beta‑shaped, Dirichlet draws, piecewise‑linear).
     If a previous solution exists it is interpolated to the current length
     and added to the pool.
2.   Keep the best few (by the exact surrogate 2·n·max(conv)).
3.   Refine each seed with a continuation L‑BFGS run on a smooth
     log‑sum‑exp surrogate of the max‑convolution.  The schedule of the
     smoothing parameter β and the number of L‑BFGS iterations are
     more aggressive than the baseline.
4.   Polish the current global best with a long exponentiated‑gradient
     (mirror‑descent on the simplex) that directly uses the sub‑gradient of
     the true max‑convolution.
5.   Perform a mass‑transfer simulated‑annealing phase that moves small
     amounts of probability mass between random coordinates; this finds
     improvements that are not reachable by pure gradient steps.
6.   Finish with a brief permutation‑annealing tweak.
7.   Evaluate the final candidate with ``evaluate_sequence`` and store it.

The whole pipeline respects the global time budget (≈1000 s) and prints
progress information.

from scipy.optimize import minimize

# Global deadline – shared by inner optimisers to avoid over‑run.
# ----------------------------------------------------------------------
_global_deadline = None


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq))  (seq sums to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – each returns a normalised probability vector.
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a_vals = (i ** (a - 1.0)) * ((n - i + 1) ** (b - 1.0))
    a_vals = np.maximum(a_vals, 0.0)
    return a_vals / a_vals.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    # Choose break positions, assign random heights, linearly interpolate.
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    return a / a.sum()


# ----------------------------------------------------------------------
# L‑BFGS surrogate objective (log‑sum‑exp smoothing).
# ----------------------------------------------------------------------
class _LbfgsObjective:
    """Callable used by scipy.optimize.minimize – returns value & gradient."""

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, log_a: np.ndarray):
        # soft‑max to a probability vector
        max_log = np.max(log_a)
        exp_shift = np.exp(log_a - max_log)
        a = exp_shift / exp_shift.sum()

        # convolution a∗a
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        beta = self.beta

        # smooth max via log‑sum‑exp
        exp_beta = np.exp(beta * (conv - shift))
        sum_exp = np.sum(exp_beta)
        w = exp_beta / sum_exp                     # weights for the smoothed max

        # gradient w.r.t. a (2 * convolution of reversed a with w)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3·n‑2
        grad_a = 2.0 * full[self.n - 1:self.n - 1 + self.n]

        # gradient w.r.t. log‑softmax parameters
        grad_log = a * (grad_a - np.dot(grad_a, a))

        # smoothed objective value
        M_beta = shift + np.log(sum_exp) / beta
        obj = 2.0 * self.n * M_beta
        return obj, grad_log


def _lbfgs_opt(start_seq: np.ndarray,
               beta_schedule,
               max_iter_per_beta: int,
               deadline: float):
    """
    Continuation L‑BFGS: for each β run a limited‑iteration L‑BFGS on the
    smoothed surrogate.  Returns the best sequence (as a probability vector)
    discovered during the whole run together with its surrogate value.
    """
    log_a = np.log(start_seq + 1e-30)          # stay in log‑softmax space
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        obj = _LbfgsObjective(len(start_seq), beta)
        opts = {"maxiter": max_iter_per_beta, "disp": False}
        try:
            res = minimize(fun=obj,
                           x0=log_a,
                           method="L-BFGS-B",
                           jac=True,
                           options=opts)
        except Exception:
            break
        log_a = res.x
        # back‑project to probability vector
        exp_shift = np.exp(log_a - np.max(log_a))
        a = exp_shift / exp_shift.sum()
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser – works on the exact
# non‑smooth objective.  It runs until the deadline (or a maximal iteration
# count) and keeps the best visited point.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 6.0):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)
    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current max‑convolution index
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the active max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size schedule (diminishing)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        # keep the log‑parameters bounded
        log_a = np.clip(log_a, -40.0, 40.0)

        # project back onto the simplex (soft‑max)
        exp_shift = np.exp(log_a - np.max(log_a))
        a = exp_shift / exp_shift.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Mass‑transfer simulated annealing – moves a small amount of mass from a
# random source index to a random destination index.  Very cheap and often
# improves the max‑convolution beyond pure gradient steps.
# ----------------------------------------------------------------------
def _mass_transfer_anneal(start_seq: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000,
                         step_factor: float = 0.15):
    """
    Perform a stochastic local‑search where a random fraction of the mass
    at a source coordinate is transferred to a destination coordinate.
    Acceptance follows a linear temperature schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # choose a source that actually holds mass
        i = rng.integers(0, n)
        if cur_seq[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # transfer a small random portion of the source mass
        eps = cur_seq[i] * rng.random() * step_factor
        if eps <= 0.0:
            continue

        ns = cur_seq.copy()
        ns[i] -= eps
        ns[j] += eps

        # ensure numerical non‑negativity (tiny negatives can appear)
        ns[i] = max(ns[i], 0.0)

        val = _objective_surrogate(ns)

        # deterministic improvement -> accept
        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        else:
            # simulated‑annealing acceptance
            T = max(0.0, 1.0 - it / max_iters)   # linear cooling
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Very cheap permutation‑annealing (swap moves only) – final polish.
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 60_000):
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    T_start = cur_val
    T_end = cur_val * 1e-6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Uses a mixture of analytic seeds, continuation L‑BFGS, exponentiated‑gradient
    polishing and a mass‑transfer annealing phase.
    """
    global _global_deadline
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluate_sequence call
    _global_deadline = start_time + budget_s - 2.0
    deadline = _global_deadline

    # ------------------------------------------------------------------
    # Problem size – a moderately larger size gives a better theoretical floor.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Load previous solution (if any) and rescale it to the current length.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0:
                # linear interpolation to new length
                old_n = arr.shape[0]
                x_old = np.linspace(0, 1, old_n)
                x_new = np.linspace(0, 1, n)
                interp = np.interp(x_new, x_old, arr)
                prev_vec = np.maximum(interp, 0.0)
                prev_vec /= prev_vec.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of seed vectors.
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _exp_decay(n, 0.0015),
        _exp_decay(n, 0.0035),
        _sym_exp_decay(n, 0.0025),
        _power(n, 0.6),
        _power(n, 1.1),
        _sym_exp_decay(n, 0.0010),
        _sym_exp_decay(n, 0.0040),
    ]

    # Dirichlet draws with various concentration parameters.
    for conc in (0.02, 0.05, 0.15, 0.5, 1.0, 3.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random piecewise‑linear shapes.
    for nb in (5, 10, 15):
        for _ in range(2):
            seeds.append(_piecewise_linear(n, nb, rng))

    # Beta‑shaped families.
    for _ in range(6):
        a = rng.uniform(0.2, 2.0)
        b = rng.uniform(0.2, 2.0)
        seeds.append(_beta_shape(n, a, b))

    # Include the previous solution (interpolated) and a few random permutations.
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(3):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Fast surrogate evaluation – keep the best few candidates.
    # ------------------------------------------------------------------
    evals = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evals.append((vec, val))

    evals.sort(key=lambda x: x[1])
    top_k = min(6, len(evals))
    top_seeds = [vec for vec, _ in evals[:top_k]]

    print("\n--- Seed ranking (surrogate) ---")
    for i, (_, v) in enumerate(evals[:top_k]):
        print(f"  #{i + 1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Continuation L‑BFGS refinement on the top seeds.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # A richer β‑schedule and more iterations per β give a finer approximation.
    beta_schedule = [10, 40, 150, 600, 2500, 10000, 40000, 150000]
    max_iter_per_beta = 600   # roughly double the baseline

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate roughly equal time slices to each remaining seed
        share = remaining / max(1, (len(top_seeds) - idx))
        sub_deadline = time.time() + share
        print(f"\n[Seed {idx + 1}] allocated ~{share:.1f}s")

        # ----- continuation L‑BFGS -----
        seq_lbfgs, val_lbfgs = _lbfgs_opt(seed_vec,
                                          beta_schedule,
                                          max_iter_per_beta,
                                          sub_deadline)

        # ----- long EG polish (if time permits) -----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs,
                                               sub_deadline,
                                               max_iters=800_000,
                                               eta0=7.0)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # pick the better refinement
        if val_lbfgs < val_eg:
            local_seq, local_val = seq_lbfgs, val_lbfgs
        else:
            local_seq, local_val = seq_eg, val_eg

        print(f"[Seed {idx + 1}] local best surrogate = {local_val:.6f}")
        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long exponentiated‑gradient polishing on the remaining budget.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=2_500_000,
                                                 eta0=6.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Mass‑transfer annealing – tries many tiny redistributions.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # give roughly half of the remaining time to this phase
        anneal_deadline = time.time() + remaining * 0.5
        print("\n--- Mass‑transfer annealing (remaining budget) ---")
        seq_mt, val_mt = _mass_transfer_anneal(global_best_seq,
                                               anneal_deadline,
                                               max_iters=300_000,
                                               step_factor=0.12)
        if val_mt < global_best_val:
            global_best_seq = seq_mt
            global_best_val = val_mt
            print(f"[Mass‑transfer] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Very short permutation‑annealing tweak.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 12.0
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=60_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Evaluate with the provided function and store the result.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
