import time, math, numpy as np, sys
# assume evaluate_sequence is globally available

_global_deadline = None

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _conv_fft(x, y):
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx*fy, n=N)[:L]
    return conv

def _objective_surrogate(seq):
    # assume seq sums to 1
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _adam_optimize(start_seq, beta_schedule, steps_per_beta, deadline):
    """Adam on log‑softmax parameters to (approximately) minimise max(conv)."""
    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    # we will compute objective every so often.
    report_interval = max(1000, steps_per_beta // 10)

    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() >= deadline:
                return best_seq, best_val
            step += 1
            # softmax
            max_log = np.max(log_a)
            exp_shift = np.exp(log_a - max_log)
            a = exp_shift / exp_shift.sum()
            # convolution a*a
            conv = _conv_fft(a, a)
            shift = np.max(conv)
            # log‑sum‑exp smoothing
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp
            # grad w.r.t a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]   # slice of length n
            # grad w.r.t log‑softmax
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)
            m_hat = m / (1.0 - beta1 ** step)
            v_hat = v / (1.0 - beta2 ** step)
            log_a -= 0.02 * m_hat / (np.sqrt(v_hat) + eps)   # learning rate 0.02

            # keep logs bounded
            log_a = np.clip(log_a, -30.0, 30.0)

            if step % report_interval == 0:
                a_cur = np.exp(log_a - np.max(log_a))
                a_cur /= a_cur.sum()
                cur_val = _objective_surrogate(a_cur)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a_cur.copy()
    # final evaluation
    a_cur = np.exp(log_a - np.max(log_a))
    a_cur /= a_cur.sum()
    cur_val = _objective_surrogate(a_cur)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a_cur.copy()
    return best_seq, best_val

def _exp_gradient_opt(start_seq, deadline, max_iters=500_000, eta0=5.0):
    """Mirror‑descent on log‑space using exact max (sub‑gradient)."""
    a = start_seq.copy()
    n = len(a)
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (as in original code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        log_a = np.clip(log_a, -30.0, 30.0)

        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _random_dirichlet(n, conc, rng):
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _exp_decay(n, rate):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()

def _sym_exp_decay(n, rate):
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()

def _power_decay(n, p):
    idx = np.arange(1, n+1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()

def _beta_shape(n, alpha, beta):
    i = np.arange(1, n+1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()

        except Exception:

    # ------------------------------------------------------------------
    # 1. generate a diverse pool of seed vectors
    # ------------------------------------------------------------------
    seeds = [

"""
search for a non‑negative step‑function that minimises
evaluate_sequence (i.e. 2·n·max(conv(a,a))/(∑a)²).

the algorithm:
    1. generate a modest pool of analytic / random seed vectors
       (uniform, exponential decay, beta‑shapes, Dirichlet draws,
       plus the previous solution if any).
    2. evaluate the exact surrogate 2·n·max(conv) and keep the
       best few seeds.
    3. refine each kept seed with a continuation Adam optimiser
       that works on a log‑sum‑exp smooth approximation of the
       max‑convolution.  the β‑parameter is increased stepwise.
    4. the best refined candidate is polished with a long
       exact‑max exponentiated‑gradient run (sub‑gradient on the
       true max).
    5. finally evaluate with the provided ``evaluate_sequence`` and
       store the result in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global deadline – used by all inner optimisation loops.
# ----------------------------------------------------------------------
_global_deadline = None


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq))  (seq is a probability vector)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
#  Seed generators – always return a probability vector (sum = 1).
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()


def _power_decay(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parameters.
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   beta_schedule,
                   steps_per_beta: int,
                   deadline: float):
    """
    Continuation Adam: minimise the log‑sum‑exp smoothed max of a∗a.
    ``start_seq`` must already be a probability vector (sum = 1).
    Returns the best sequence found and its surrogate value.
    """
    n = len(start_seq)
    # work in the log‑softmax domain
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)          # 1st moment
    v = np.zeros_like(log_a)          # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    # reporting interval – roughly 10 % of the work per β
    report_interval = max(200, steps_per_beta // 10)

    for beta in beta_schedule:
        for _ in range(steps_per_beta):
            if time.time() >= deadline:
                return best_seq, best_val

            step += 1

            # soft‑max → probability vector a
            max_log = np.max(log_a)
            exp_shift = np.exp(log_a - max_log)
            a = exp_shift / exp_shift.sum()

            # convolution a∗a
            conv = _conv_fft(a, a)
            shift = np.max(conv)                     # the true max

            # smooth max by log‑sum‑exp
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp                    # weights that sum to 1

            # gradient wrt a  (2·conv(rev(a), w) )
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length 3·n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]       # size n

            # gradient wrt log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update on log_a
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)
            m_hat = m / (1.0 - beta1 ** step)
            v_hat = v / (1.0 - beta2 ** step)
            log_a -= 0.02 * m_hat / (np.sqrt(v_hat) + eps)   # learning‑rate = 0.02

            # keep the logs bounded
            log_a = np.clip(log_a, -30.0, 30.0)

            # occasional exact evaluation
            if step % report_interval == 0:
                a_cur = np.exp(log_a - np.max(log_a))
                a_cur /= a_cur.sum()
                cur_val = _objective_surrogate(a_cur)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a_cur.copy()

    # final exact evaluation
    a_cur = np.exp(log_a - np.max(log_a))
    a_cur /= a_cur.sum()
    cur_val = _objective_surrogate(a_cur)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a_cur.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact‑max exponentiated‑gradient (sub‑gradient) optimiser.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_500_000,
                     eta0: float = 4.0):
    """
    Mirror‑descent on the log‑variables using the true max‑convolution.
    The step size decays like η₀/√t.  Returns the best sequence seen.
    """
    a = start_seq.copy()
    n = len(a)
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active max entry (identical to original code)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        log_a = np.clip(log_a, -30.0, 30.0)

        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:

    ``evaluate_sequence``.  The pipeline follows the outline described

    """
    global _global_deadline
    rng = np.random.default_rng(seed)
    start_time = time.time()
    _global_deadline = start_time + budget_s - 1.0   # safety margin
    deadline = _global_deadline

    # ------------------------------------------------------------------
    # Problem size – 4000 points give a good balance of expressivity vs.
    # computational cost.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Pull the previously found vector (if any) to keep continuity.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                if prev_vec.sum() > 0.0:
                    prev_vec = prev_vec / prev_vec.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of seed vectors.
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _exp_decay(n, 0.0015),
        _exp_decay(n, 0.0030),
        _sym_exp_decay(n, 0.0025),
        _power_decay(n, 0.6),
        _power_decay(n, 1.2),
    ]

    # Dirichlet draws with a range of concentration parameters.
    for conc in (0.02, 0.05, 0.15, 0.5, 1.0, 3.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random beta‑shaped families.
    for _ in range(6):
        a = rng.uniform(0.2, 2.0)
        b = rng.uniform(0.2, 2.0)
        seeds.append(_beta_shape(n, a, b))

    # Include the previous solution and a few of its random permutations.
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(3):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few.
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() >= deadline:
            break
        v = np.maximum(vec, 0.0)
        if v.sum() == 0.0:
            continue
        v = v / v.sum()
        val = _objective_surrogate(v)
        evaluated.append((v, val))

    if not evaluated:
        raise RuntimeError("No valid initial seeds were generated before the deadline.")

    evaluated.sort(key=lambda p: p[1])
    top_k = min(5, len(evaluated))
    top_seeds = [seq for seq, _ in evaluated[:top_k]]

    print("\n--- Seed ranking (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Local refinement with continuation Adam.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    beta_schedule = [10, 80, 400, 2000, 10000, 50000]   # increasingly sharp
    steps_per_beta = 2000                              # modest budget per β

    for idx, seed_vec in enumerate(top_seeds):

        print(f"\n[Seed {idx + 1}] allocated ~{share:.1f}s (capped to {sub_deadline - time.time():.1f}s)")

        seq_refined, val_refined = _adam_optimize(seed_vec,
                                                  beta_schedule,
                                                  steps_per_beta,
                                                  sub_deadline)

        if val_refined < global_best_val:
            global_best_seq = seq_refined
            global_best_val = val_refined
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long polishing with the exact‑max exponentiated‑gradient.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final exact‑max polishing (remaining budget) ---")
        seq_final, val_final = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=1_500_000,
                                                 eta0=4.0)
        if val_final < global_best_val:
            global_best_seq = seq_final
            global_best_val = val_final
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. (Optional) a very short high‑β Adam polishing if we still have time.

    if time.time() < deadline:

        seq_extra, val_extra = _adam_optimize(global_best_seq,
                                               beta_schedule=[1e5, 2e5],
                                               steps_per_beta=1000,
                                               deadline=extra_deadline)
        if val_extra < global_best_val:
            global_best_seq = seq_extra
            global_best_val = val_extra
            print(f"[Extra Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate the final candidate with the provided function.
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # store the result for possible reuse in the next run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
