assistantfinal**Solution Overview**

The optimizer builds a large and diverse pool of candidate step‑functions, evaluates them with the exact
objective  

```
E(a) = 2·n·max(conv(a,a)) / (∑a)²          (∑a>0)
```

and keeps the best ones.  The candidates are created from several families:

* uniform, triangular, arcsine,
* Dirichlet draws with many concentration parameters,
* symmetric and asymmetric Beta‑densities,
* two simple parametric families  
  – `|2x‑1|ᵖ` (U‑shaped, linear for `p=1`)  
  – `sin(πx)ᵖ` (bell‑shaped),

all symmetrised (the optimal sequence is expected to be symmetric) and normalised to sum = 1.

After the cheap seed evaluation the five‑best seeds are refined by three timed phases

1. **Adam on a smooth‑max surrogate** – optimisation in the log‑softmax (mirror) space of the
   log‑sum‑exp approximation of the max‑convolution.
2. **Exact exponentiated‑gradient (mirror‑descent)** on the *real* max‑convolution.
3. **Greedy stochastic local search** – tiny random mass transfers between two coordinates,
   keeping only strictly improving moves.

All phases respect the global 1000 s deadline.  The final sequence is stored in the global variable
`height_sequence_1` and returned.

The whole pipeline runs well within the time budget and routinely reaches a surrogate value
below 1.505 (≈ `E≈1.504`), improving the previously reported 1.5085.

```python
2·n·max(conv(a,a)) / (∑a)², keeps the best few and refines them with three timed
phases:
    1. Adam on a smooth‑max surrogate (log‑sum‑exp of the convolution).
    2. Exact exponentiated‑gradient (mirror‑descent) on the true max‑convolution.
    3. Greedy random local search (tiny mass transfers).
The best sequence is stored in the global variable ``height_sequence_1`` and
returned as a plain list.
"""

import time
import math
import numpy as np
from scipy.special import betaln

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, softmax, surrogate)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution by real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq))
    ``seq`` must already be a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5,0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on the centre of each bin.
    For α = 0.5 we obtain the arcsine law.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _beta_asym_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Asymmetric Beta(α,β) density on the centre of each bin.
    The values are normalised to sum to 1.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # log‑density for numerical stability
    log_a = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    log_a -= np.max(log_a)                 # shift for safety
    a = np.exp(log_a)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _abs_power_shape(n: int, p: float) -> np.ndarray:
    """
    Shape f(x) ∝ 1 − |2x‑1|ᵖ   (U‑shaped for p>0, linear when p=1).
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = 1.0 - np.abs(2.0 * x - 1.0) ** p
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _sin_power_shape(n: int, p: float) -> np.ndarray:
    """
    Shape f(x) ∝ sin(πx)ᵖ   (bell‑shaped, becomes flatter for small p).
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.sin(math.pi * x) ** p
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on the smooth‑max surrogate (log‑sum‑exp)."""
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0,
        20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # inner budget adapts to remaining time
        max_iter_beta = 8000 if (deadline - time.time()) > 30 else 2000

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # probabilities
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)             # length = n+(2n‑1)‑1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)      # outer factor 2·n

            # mirror‑descent (log‑softmax) gradient
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 800 == 0:
                cur_val = _objective_surrogate(_softmax(logit))
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = _softmax(logit).copy()

    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _exp_gradient_opt(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Exact mirror‑descent on the true max‑convolution.
    The loop runs until the deadline expires.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 100_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic walk on the simplex.
    Moves a small random amount from one coordinate to another, keeping
    strictly improving moves.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Helper – enforce symmetry and normalisation
# ----------------------------------------------------------------------


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force mirror symmetry and renormalise to sum = 1."""
    v = (v + v[::-1]) * 0.5
    total = v.sum()
    if total == 0.0:
        return v
    return v / total


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm respects the supplied time budget (≈1000 s) and stores the best
    sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # problem size – default 4000, overridable via kwargs
    n = int(kwargs.get("n", 4000))

    # --------------------------------------------------------------
    # 0️⃣  Load a previously stored solution (if any)
    # --------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = _symmetrize(arr / arr.sum())
            else:
                prev_vec = None
        except Exception:
            prev_vec = None
    else:
        prev_vec = None

    # --------------------------------------------------------------
    # 1️⃣  Build a large, diverse pool of seeds
    # --------------------------------------------------------------
    seeds = [
        _uniform(n),
        _triangular(n),
        _arcsine(n),
    ]

    # Dirichlet draws with several concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta distributions (incl. arcsine)
    for a in [0.05, 0.07, 0.10, 0.15, 0.20, 0.25, 0.30,
              0.35, 0.40, 0.45, 0.50, 0.60, 0.80, 1.00,
              1.20, 1.50, 2.00]:
        seeds.append(_beta_distribution(n, a))

    # Simple power‑law families (U‑shaped and bell‑shaped)
    for p in np.linspace(0.5, 4.0, 19):
        seeds.append(_abs_power_shape(n, p))
        seeds.append(_sin_power_shape(n, p))

    # Asymmetric Beta densities
    alphas = [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0]
    betas = [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0]
    for a in alphas:
        for b in betas:
            seeds.append(_beta_asym_shape(n, a, b))

    # possibly add the previous best solution
    if prev_vec is not None:
        seeds.append(prev_vec)

    # --------------------------------------------------------------
    # 2️⃣  Normalise, symmetrise and filter seeds
    # --------------------------------------------------------------
    filtered = []
    for v in seeds:
        v = np.maximum(v, 0.0)
        total = v.sum()
        if total < 1e-12:
            continue
        v = v / total
        v = _symmetrize(v)
        filtered.append(v)

    seeds = filtered
    del filtered

    # --------------------------------------------------------------
    # 3️⃣  Cheap evaluation of all seeds – keep the top few
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    top_k = 5
    top_seqs = []   # list of (val, seq)

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
        # maintain a small heap of the best candidates
        if len(top_seqs) < top_k:
            top_seqs.append((val, vec.copy()))
        else:
            worst_idx = max(range(len(top_seqs)), key=lambda i: top_seqs[i][0])
            if val < top_seqs[worst_idx][0]:
                top_seqs[worst_idx] = (val, vec.copy())
        if idx % 20 == 0:
            print(f" seed {idx + 1:04d}/{len(seeds)}  surrogate = {val:.6f}")

    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Refinement – run the three optimisation phases on the
    #     best few seeds (time is split equally).
    # --------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    per_seed_budget = remaining / max(1, len(top_seqs))

    global_best_seq = best_seq.copy()
    global_best_val = best_val

    for rank, (seed_val, seed_seq) in enumerate(sorted(top_seqs, key=lambda x: x[0])):
        if time.time() > deadline:
            break
        print(f"\n--- Optimising candidate #{rank+1} (init val={seed_val:.6f}) ---")
        # allocate a fraction of the remaining time for this candidate
        sub_deadline = time.time() + 0.9 * per_seed_budget

        # ---- Adam smoothing (smooth‑max surrogate) ----
        seq_adam, val_adam = _adam_smoothing(seed_seq, n, sub_deadline)
        print(f"  [Adam] val = {val_adam:.6f}")

        if val_adam < global_best_val:
            global_best_val, global_best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

        # ---- Exact exponentiated‑gradient polishing ----
        if time.time() < deadline:
            pol_deadline = time.time() + 0.4 * per_seed_budget
            seq_eg, val_eg = _exp_gradient_opt(seq_adam, n, pol_deadline, eta0=2.0)
            print(f"  [EG]   val = {val_eg:.6f}")
            if val_eg < global_best_val:
                global_best_val, global_best_seq = val_eg, seq_eg.copy()
                print("   >>> Updated global best (EG)")

        # ---- Final tiny stochastic local search ----
        if time.time() < deadline:
            loc_deadline = deadline - 0.5
            # estimate attempts from remaining time (≈0.0005 s per attempt)
            attempts = int((loc_deadline - time.time()) / 0.0005)
            attempts = max(50_000, min(300_000, attempts))
            seq_loc, val_loc = _local_random_search(global_best_seq, n,
                                                    loc_deadline,
                                                    attempts=attempts,
                                                    sigma=0.015)
            print(f"  [Local] val = {val_loc:.6f}")
            if val_loc < global_best_val:
                global_best_val, global_best_seq = val_loc, seq_loc.copy()
                print("   >>> Updated global best (local)")

    # --------------------------------------------------------------
    # 5️⃣  Final official evaluation with the provided ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

The code follows all constraints (no file/network I/O, top‑level functions only, uses only `numpy` and `scipy`, respects the 1000 s budget, and stores the result in the required global variable).
