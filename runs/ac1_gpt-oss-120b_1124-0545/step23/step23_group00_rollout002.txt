"""
search_step_function.py

A multi‑stage, time‑bounded optimiser for the step‑function evaluation problem.

Goal
----
Minimise   2·n·max(conv(a,a)) / (sum(a)**2)  with a_i ≥ 0.
Because the objective is scale‑invariant we keep the vector on the probability
simplex (∑ a_i = 1).

Method
------
1. Initialise – reuse a previously found vector if present, otherwise start
   from a slowly decaying exponential (≈geometric) and normalise.
2. Adam optimiser on a *log‑softmax* parametrisation, minimising a smooth
   log‑sum‑exp surrogate of the max‑convolution.  The β‑parameter is raised
   geometrically from 0.5 to 1e7 over many iterations (default 600 k).  The
   Adam learning‑rate (α) is set to 0.8 to explore more aggressively.
3. Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
   Uses a decreasing step η_t = η₀/√t with a larger initial η₀ (default 50)
   and runs until the time budget expires.
4. Simulated‑annealing hill‑climbing: random swaps and tiny Gaussian
   perturbations, accepting worsening moves with a slowly cooling temperature.
   This polishes the solution found by the previous stages.
5. Random‑restart Adam refinements: while time permits a few quick Adam runs
   are launched from noisy, randomly permuted versions of the current best.
6. Final evaluation with the supplied ``evaluate_sequence`` function; the best
   vector is stored globally as ``height_sequence_1`` for possible continuation
   runs.

All loops are guarded by a deadline (budget – 5 s safety margin) so the
function always returns a valid sequence before the 1000 s limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)) when sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray, n: int, beta: float):
    """
    Gradient of the smoothed objective with respect to the log‑softmax
    parameters `v_local`.  Returns (objective, grad_v).
    """
    a = _softmax(v_local)                     # probability vector (sum = 1)
    conv = _conv_fft(a, a)                    # length = 2n‑1
    max_c = np.max(conv)

    # Stable log‑sum‑exp
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # Gradient w.r.t. a
    w = exp_beta / sum_exp                     # weights over convolution bins
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n‑2
    grad_a = 2.0 * full[n - 1 : n - 1 + n]     # slice back to length n

    # Chain rule for log‑softmax parametrisation
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 600_000,
              beta_start: float = 0.5,
              beta_end: float = 1e7,
              alpha: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑softmax parametrisation of the smoothed
    max‑convolution objective.  Returns the best probability vector observed
    and its exact surrogate value (2·n·max(conv)).
    """
    # initialise log‑softmax variables
    v = np.log(start_seq + 1e-30)
    m = np.zeros_like(v)       # first moment
    v2 = np.zeros_like(v)      # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric schedule for β
        t = (it - 1) / max_iters
        beta = beta_start * ((beta_end / beta_start) ** t)

        obj, grad = _smooth_obj_and_grad(v, n, beta)

        # Adam updates (in log‑space)
        m = beta1 * m + (1.0 - beta1) * grad
        v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v2_hat) + eps)
        v -= step

        a = _softmax(v)
        cur_val = _objective_surrogate(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 10_000_000,
                     eta0: float = 50.0) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective,
    performed in log‑space via a softmax update (multiplicative‑weights).
    The step size decays as η₀ / √t.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (swap / tiny Gaussian perturbation)
# ----------------------------------------------------------------------
def _hill_climb(start_seq: np.ndarray,
                n: int,
                deadline: float,
                rng: np.random.Generator,
                max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Local search that repeatedly swaps two positions or adds a tiny Gaussian
    perturbation.  Worsening moves are accepted with a simple linear cooling
    schedule.  Returns the best sequence observed.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    # temperature schedule (linear cooling)
    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap move (most common)
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                   # tiny Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq, best_val = ns.copy(), val
        else:
            # accept worsening move with probability exp(-(val‑cur_val)/T)
            if T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T):
                cur_seq = ns
                cur_val = val

    return best_seq, best_val

        print(f"   after HC    surrogate = {val_hc:.6f}")
        if val_hc < val_adam:
            seq_adam, val_adam = seq_hc, val_hc

    # --------------------------------------------------------------
    # 5. Short random‑restart Adam refinements (if time remains)
    # --------------------------------------------------------------
    while time.time() + 30.0 < deadline:
        print("\n[Stage 4] Random‑restart Adam")
        # create a noisy, randomly permuted copy of the current best
        perm = rng.permutation(n)
        noisy = seq_adam[perm] + 0.001 * rng.normal(size=n)
        noisy = np.maximum(noisy, 0.0)
        s = noisy.sum()
        if s == 0.0:
            continue
        noisy = noisy / s

        sub_deadline = time.time() + 25.0
        seq_rr, val_rr = _adam_opt(
            start_seq=noisy,
            n=n,
            deadline=sub_deadline,
            max_iters=120_000,
            beta_start=0.5,
            beta_end=2e5,
            alpha=0.8,
        )
        print(f"   restart surrogate = {val_rr:.6f}")
        if val_rr < val_adam:
            seq_adam, val_adam = seq_rr, val_rr
            print("   *** improvement found! ***")

    # --------------------------------------------------------------
    # 6. Final evaluation with the provided black‑box function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(seq_adam.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {val_adam:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = seq_adam.tolist()
    return seq_adam.tolist()
