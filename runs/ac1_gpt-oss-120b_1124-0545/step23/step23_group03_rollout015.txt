"""
Non‑negative step‑function optimiser (enhanced).

The optimiser builds a rich set of seed distributions (uniform,
triangular, arcsine, Dirichlet, symmetric Beta and power‑law shapes) and
searches the convex hull of these seeds by random mixtures.  The current
best candidate is polished several times by
* Adam on a smooth‑max surrogate (log‑sum‑exp) with an annealed β,
* exponentiated‑gradient (mirror‑descent) on the true max‑convolution,
* short random restarts of the EG step,
* a final stochastic local walk that only accepts improving mass transfers.
The whole routine respects the external 1000 s deadline; intermediate
bounds are printed and the best sequence is stored in the global variable
``height_sequence_1`` for the next invocation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution by real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2 * n * max(conv(seq, seq))
    ``seq`` must already be a probability vector (sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5, 0.5))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a /= a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric discrete triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on the centre of each bin.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric power‑law shape a_i ∝ (i+0.5)^{-p} + (n-0.5-i)^{-p}.
    For p=0 this collapses to uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    left = np.power(i, -p)
    right = np.power(n + 0.5 - i, -p)
    a = left + right
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a

def _adam_smoothing_time(start_seq: np.ndarray, n: int, max_seconds: float):
    """
    Adam optimisation on the smooth‑max surrogate.
    Runs for at most ``max_seconds`` seconds.
    Returns the best probability vector seen and its exact surrogate value.
    """
    deadline = time.time() + max_seconds
    # initialise log‑probabilities (logits)
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # how many inner iterations are feasible?
        # we allow a fixed budget per β unless we are very close to the deadline
        max_iter_beta = 4000 if (deadline - time.time()) > 5 else 500
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)             # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)      # outer factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 600 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


def _exp_gradient_opt_time(start_seq: np.ndarray, n: int, max_seconds: float, eta0: float = 5.0):
    """
    Mirror‑descent on the true max‑convolution.
    Runs until ``max_seconds`` have elapsed.
    Returns the best sequence seen and its exact surrogate value.
    """
    deadline = time.time() + max_seconds
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        # the factor 2 is because each product a_i a_j appears twice in full conv,
        # except at the centre when i=j, but the sub‑gradient formula above already
        # accounts for the correct factor.
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _local_random_search_time(start_seq: np.ndarray,
                              n: int,
                              max_seconds: float,
                              sigma: float = 0.02,
                              attempts: int = None):
    """
    Greedy stochastic walk on the simplex (only strictly improving swaps).
    The number of attempts is chosen automatically from the remaining time
    if ``attempts`` is ``None``.
    Returns the best sequence and its surrogate value.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    if attempts is None:
        # estimate a reasonable number of attempts from the time budget
        # one attempt ≈ 0.0006 s on a typical CPU
        est = int((max_seconds) / 0.0006)
        attempts = max(20_000, min(300_000, est))

    deadline = time.time() + max_seconds
    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric power‑law shape a_i ∝ (i+0.5)^{-p} + (n‑0.5‑i)^{-p}.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    left = np.power(i, -p)
    right = np.power(n + 0.5 - i, -p)
    a = left + right
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        return _uniform(n)
    a /= total
    return a

def _adam_smoothing_time(start_seq: np.ndarray, n: int, max_seconds: float):
    """
    Adam optimisation on the smooth‑max surrogate.
    Runs for at most ``max_seconds`` seconds.
    Returns the best probability vector seen and its exact surrogate value.
    """
    deadline = time.time() + max_seconds
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0, 500000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # inner budget per β
        max_iter_beta = 4000 if (deadline - time.time()) > 5 else 500
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current distribution
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 2n‑1 + n‑1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]
            grad_a = grad_weighted * (2.0 * n)        # outer factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 600 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val
    return best_seq, best_val


def _exp_gradient_opt_time(start_seq: np.ndarray, n: int, max_seconds: float,
                          eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
    Runs until ``max_seconds`` have elapsed.
    Returns the best sequence seen and its exact surrogate value.
    """
    deadline = time.time() + max_seconds
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of conv(k) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _local_random_search_time(start_seq: np.ndarray,
                              n: int,
                              max_seconds: float,
                              sigma: float = 0.02,
                              attempts: int = None):
    """
    Greedy stochastic walk on the simplex (strictly improving transfers).
    If ``attempts`` is ``None`` the routine picks a number of attempts
    based on the time budget.
    Returns the best sequence and its surrogate value.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    if attempts is None:
        # empirical cost per attempt ≈ 0.0006 s on a typical CPU
        est = int(max_seconds / 0.0006)
        attempts = max(20_000, min(300_000, est))

    deadline = time.time() + max_seconds
    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm runs for at most ``budget_s`` seconds, prints intermediate
    bounds and stores the best sequence in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin so that we always finish before the external timeout
    deadline = start_time + budget_s - 2.0

    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load a previously stored solution (if any)
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
            else:
                prev_vec = None
        except Exception:
            prev_vec = None
    else:
        prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣ Build a diverse seed pool (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _arcsine(n),
        _triangular(n),
    ]

    # Dirichlet draws with several concentration parameters
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Symmetric Beta distributions – main family
    beta_alphas = [
        0.05, 0.07, 0.10, 0.15, 0.20, 0.25, 0.30,
        0.35, 0.40, 0.45, 0.50, 0.60, 0.80, 1.00,
        1.20, 1.50, 2.00
    ]
    for a in beta_alphas:
        seeds.append(_beta_distribution(n, a))

    # Power‑law shapes (symmetric)
    power_ps = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]
    for p in power_ps:
        seeds.append(_power_law_distribution(n, p))

    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣ Initial evaluation of seeds – keep the best surrogate value
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Random convex combinations of seeds (mixture search)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        mix_deadline = time.time() + 0.25 * (deadline - time.time())
        print("\n--- Convex‑mixture search ---")
        mix_attempts = 0
        while time.time() < mix_deadline:
            mix_attempts += 1
            i, j = rng.integers(0, len(seeds), size=2)
            lam = rng.random()
            combo = lam * seeds[i] + (1.0 - lam) * seeds[j]

            # occasionally blend three seeds (≈1 % of the attempts)
            if rng.random() < 0.01:
                k = rng.integers(0, len(seeds))
                w = rng.dirichlet(np.ones(3))
                combo = (w[0] * seeds[i] +
                         w[1] * seeds[j] +
                         w[2] * seeds[k])

            tot = combo.sum()
            if tot == 0.0:
                continue
            combo = combo / tot
            val = _objective_surrogate(combo)
            if val < best_val:
                best_val, best_seq = val, combo.copy()
                print(f"  [Mix #{mix_attempts:05d}] new best = {best_val:.6f}")
        print(f"[Mix] best surrogate after mixtures = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Alternating polishing phases (Adam → EG → Adam …) until time out
    # ------------------------------------------------------------------
    phase = 0
    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()
        slice_time = max(5.0, 0.10 * remaining)   # at least 5 s per sub‑phase

        if phase % 2 == 0:
            # Adam smoothing on the current best
            print("\n--- Adam smoothing (smooth‑max) ---")
            seq_adam, val_adam = _adam_smoothing_time(best_seq, n, slice_time)
            if val_adam < best_val:
                best_val, best_seq = val_adam, seq_adam.copy()
                print(f"   >>> Improved by Adam: {best_val:.6f}")
        else:
            # Exponentiated‑gradient polishing (exact max)
            print("\n--- Exponentiated‑gradient polishing (exact max) ---")
            seq_eg, val_eg = _exp_gradient_opt_time(best_seq, n, slice_time,
                                                   eta0=10.0)
            if val_eg < best_val:
                best_val, best_seq = val_eg, seq_eg.copy()
                print(f"   >>> Improved by EG: {best_val:.6f}")

        phase += 1

    # ------------------------------------------------------------------
    # 5️⃣ Final random local refinement (strictly improving moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        time_left = deadline - time.time()
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search_time(best_seq, n, time_left,
                                                     sigma=0.02,
                                                     attempts=None)
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"   >>> Improved by local search: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
