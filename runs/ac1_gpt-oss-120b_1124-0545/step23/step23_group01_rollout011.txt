    beta_seq = [5, 20, 100, 500, 2000]
    v = np.log(start_seq + 1e-30)

            a = _softmax(v)

            cmax = np.max(conv)

                     beta_start=0.5, beta_end=100.0,
                     lr0=0.3, beta1=0.9, beta2=0.999, eps=1e-8):

    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    conv = _conv_fft(best_seq, best_seq)
    best_val = 2.0 * n * np.max(conv)
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = _softmax(log_a)
        # compute conv and smooth gradient
        conv = _conv_fft(a, a)
        conv_max = np.max(conv)

        # schedule beta
        beta = beta_start + (beta_end - beta_start) * (t - 1) / (max_iters - 1) if max_iters>1 else beta_end
        # compute weights w
        # stable: max of beta*conv is at conv_max => exponent 0.
        exp_arg = beta * (conv - conv_max)
        w = np.exp(exp_arg)      # max exp_arg = 0, so max w = 1, no overflow
        w_sum = np.sum(w)
        w_norm = w / w_sum

        # gradient wrt a
        a_rev = a[::-1]
        conv2 = _conv_fft(w_norm, a_rev)
        g = 2.0 * conv2[n-1:n-1+n]

        # gradient wrt v
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * grad_v
        v = beta2 * v + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

"""
Propose a non‑negative step‑function (probability vector) that minimizes
the evaluation function ``evaluate_sequence``.

Algorithm Overview
------------------
1. **Problem normalisation** – we work on the simplex
   ``∑ a_i = 1``; the objective then equals ``2·n·max(conv(a,a))`` and
   is invariant under scaling.

2. **Initial seed pool** – the previous best sequence (if any), a few
   exponential decays, a symmetric triangle, the uniform distribution,
   and a couple of random Dirichlet draws.

3. **Smooth‑max Adam optimisation** – instead of the non‑smooth
   ``max`` we minimise a smooth approximation
   ``f_β(a) = (1/β)·log∑_k exp(β·(a*a)[k])``.
   Its gradient w.r.t. the log‑softmax parameters ``v`` (where
   ``a = softmax(v)``) can be computed in O(n log n) via two FFT‑based
   convolutions.  Adam with a linear schedule for ``β`` is applied.
   A short “quick” optimisation is performed for every seed; the best
   result becomes the starting point for a longer “deep’’ optimisation.

4. **Final polishing** – the exact non‑smooth gradient (active index)
   is used in an exponentiated‑gradient (mirror‑descent) pass
   (identical to the reference method) to squeeze a few extra digits.

5. **Evaluation & storage** – the final vector is evaluated with the
   provided ``evaluate_sequence`` and stored in the global variable
   ``height_sequence_1`` for possible continuation runs.

The routine respects the 1000 s time budget and prints intermediate
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (both inputs are real)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple probability helpers
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used for optimisation: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families for initial guesses
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape, peak in the middle, zero at ends."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (original reference)
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 500_000,
              lr0: float = 0.2,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam optimiser using the true max entry (no smoothing)."""
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g_a = np.zeros_like(a)
        g_a[idx_i] = 2.0 * a[rev_idx]

        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        cur_val = _objective_surrogate(_softmax(log_a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = _softmax(log_a).copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (reference version)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on a smooth approximation of the max (new component)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200_000,
                     beta_start: float = 0.5,
                     beta_end: float = 200.0,
                     lr0: float = 0.3,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k])
    The parameters are transformed by a log‑softmax to enforce the simplex.
    """
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    a = _softmax(log_a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        # ---- convolution a*a -------------------------------------------------
        conv = _conv_fft(a, a)                     # length 2n‑1
        conv_max = np.max(conv)

        # ---- β schedule (linear) --------------------------------------------
        beta = beta_start + (beta_end - beta_start) * (t - 1) / (max_iters - 1)

        # ---- smooth weights w_k = exp(β·conv_k) / Σexp(β·conv) -----------------
        # use conv_max for numerical stability (max exponent becomes 0)
        exp_arg = beta * (conv - conv_max)         # max(exp_arg) = 0
        w = np.exp(exp_arg)                        # safe, never overflows
        w_sum = np.sum(w)
        w_norm = w / w_sum

        # ---- gradient wrt a --------------------------------------------------
        a_rev = a[::-1]
        conv2 = _conv_fft(w_norm, a_rev)           # length 3n‑2
        g = 2.0 * conv2[n - 1:n - 1 + n]           # size n

        # ---- chain‑rule to obtain gradient wrt log‑softmax variables ----------
        dot = np.dot(g, a)
        grad_v = a * (g - dot)

        # ---- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

        # ---- occasional true‑objective check ---------------------------------
        if (t % 2000 == 0) or (t == max_iters):
            cur_val = 2.0 * n * conv_max
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val

            if arr.shape == (n,):
                candidates.append(arr / arr.sum())
        except Exception:
            pass

    # exponential decays with various rates
    for rate in (0.0005, 0.001, 0.002, 0.005):
        candidates.append(_exp_decay(n, rate))

    candidates.append(_triangular(n))
    candidates.append(_uniform(n))

    # random Dirichlet draws (spiky distributions)
    for _ in range(2):
        cand = rng.dirichlet(alpha=np.full(n, 0.5))
        candidates.append(cand)

    # ------------------------------------------------------------------
    # Quick optimisation for every seed (low budget, low β)
    # ------------------------------------------------------------------
    best_overall_seq = None
    best_overall_val = float("inf")

    quick_iters = 30_000
    for idx, cand in enumerate(candidates, start=1):
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()

        time_left = deadline - time.time()
        if time_left < 30.0:
            print("[Time] Not enough time for more quick runs.")
            break

        print(f"[Quick] candidate {idx}/{len(candidates)} – "
              f"max iters = {quick_iters}")

        seq, val = _adam_smooth_opt(cand, n, deadline,
                                    max_iters=quick_iters,
                                    beta_start=0.5,
                                    beta_end=50.0,
                                    lr0=0.3)

        print(f"       → surrogate after quick = {val:.6f}")

        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"       ** new global best (quick) = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # Deep optimisation on the best candidate found so far
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        best_overall_seq = _exp_decay(n, rate=0.001)
        best_overall_seq = best_overall_seq / best_overall_seq.sum()

    time_left = deadline - time.time()
    if time_left > 60.0:
        deep_iters = int(min(250_000, time_left * 200))   # heuristic
        print("\n[Deep] running long smooth‑max optimisation "
              f"({deep_iters} iters, β up to 500)")

        seq, val = _adam_smooth_opt(best_overall_seq, n, deadline,
                                    max_iters=deep_iters,
                                    beta_start=10.0,
                                    beta_end=500.0,
                                    lr0=0.15)

        print(f"       → surrogate after deep = {val:.6f}")
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"       ** new global best (deep) = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with the exact max‑gradient (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        remaining_iters = int(min(3_000_000, (deadline - time.time()) * 3000))
        print("\n[Polish] exponentiated‑gradient pass")
        polished_seq, polished_val = _exp_gradient_opt(
            best_overall_seq, n, deadline,
            max_iters=remaining_iters,
            eta0=12.0)

        if polished_val < best_overall_val:
            best_overall_seq = polished_seq
            best_overall_val = polished_val
            print(f"[Polish] improved surrogate = {best_overall_val:.6f}")
        else:
            print("[Polish] no improvement (best unchanged)")

    # ------------------------------------------------------------------
    # Evaluate the true objective via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
        print(f"\n=== Final result ===\nSurrogate ≈ {best_overall_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:            # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
