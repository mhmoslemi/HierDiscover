"""
Step‑function optimiser for `evaluate_sequence`.

The optimiser works in several stages:

1. **Seed generation** – a diverse pool of analytically‑derived vectors
   (uniform, exponential, power‑law, symmetric variants, Dirichlet
   draws, and the previous best solution).  All vectors are ℓ₁‑normalised.

2. **Coarse screening** – cheap surrogate
   `2·n·max(conv(a,a))` (exact for ℓ₁‑normalised vectors) is used to keep
   the best few seeds.

3. **Continuation L‑BFGS‑B** – we rewrite the problem in the
   log‑softmax space `y → a = softmax(y)`.  For a given “temperature”
   β we minimise the smooth approximation
   `Lβ(y) = 2·n·∑ w_k·conv_k` with `w = softmax(β·conv)`.  β is increased
   geometrically (5 → 8·10⁴), each step solved by L‑BFGS‑B (few dozen
   iterations).  This yields a very accurate reduction of the
   ‖a∗a‖∞ term.

4. **Exact exponentiated‑gradient polishing** – a mirror‑descent (EG)
   run on the true max‑convolution objective uses the sub‑gradient of the
   active entry.  It finishes the optimisation with the remaining time.

5. **Optional permutation annealing** – a short simulated‑annealing
   phase that swaps entries (keeps the ℓ₁‑norm) to shave off the last
   digits.

All heavy work uses FFT‑based convolution (`O(n log n)`).  The
implementation adapts the time budget (default 1000 s) and stores the
best vector in the global variable `height_sequence_1`.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Low‑level FFT convolution (real, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Soft‑max utilities (log‑space parametrisation)
# ----------------------------------------------------------------------
_EPS = 1e-30


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a vector that sums to 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smoothed objective and its gradient (used by L‑BFGS‑B)
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(y: np.ndarray, n: int, beta: float):
    """
    Returns (objective, gradient) for the β‑smoothed problem.
    The optimisation variable is y (log‑softmax parameters).
    """
    a = _softmax(y)
    conv = _conv_fft(a, a)                     # length 2·n‑1
    shift = np.max(conv)                       # for numerical stability
    w = np.exp(beta * (conv - shift))
    w_sum = w.sum()
    if w_sum == 0.0:                           # extreme under‑flow => use hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    w /= w_sum                                 # soft‑max weights, sum == 1
    approx_max = np.dot(conv, w)               # smooth max
    obj = 2.0 * n * approx_max

    # gradient wrt a
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length 3·n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]       # size n

    # gradient wrt y  (soft‑max Jacobian)
    dot = np.dot(grad_a, a)
    grad_y = a * (grad_a - dot)
    return obj, grad_y


class _SmoothFunctor:
    """Callable object required by scipy.optimize (captures n and β)."""
    __slots__ = ("n", "beta")

    def __init__(self, n: int, beta: float):
        self.n = n
        self.beta = beta

    def __call__(self, y: np.ndarray):
        return _smooth_obj_and_grad(y, self.n, self.beta)


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^{‑p}, i = 0…n‑1, δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """Symmetric version of the shifted power law."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i) + delta
    a = dist ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient optimiser (sub‑gradient of the true max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_a: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns (best_vector, best_surrogate_value).
    """
    a = start_a.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space (log‑softmax)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and index of the active diagonal (the current max)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing stepsize
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g

        # re‑exponentiate and renormalise (log‑softmax)
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_a: np.ndarray,
                 deadline: float,
                 max_iters: int = 150_000):
    """Pure permutation optimiser – swaps two entries."""
    n = len(start_a)
    cur_a = start_a.copy()
    cur_val = _objective_surrogate(cur_a)
    best_a = cur_a.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_a.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or rng.random() < math.exp(-(val - cur_val) / T):
            cur_a = ns
            cur_val = val
            if val < best_val:
                best_a = ns.copy()
                best_val = val

    return best_a, best_val


# ----------------------------------------------------------------------
# L‑BFGS‑B continuation optimiser (smooth max → true max)
# ----------------------------------------------------------------------
def _lbfgsb_continuation(start_a: np.ndarray,
                         deadline: float,
                         beta_schedule=None):
    """
    Optimises the smooth approximation for an increasing sequence of β,
    each stage solved by L‑BFGS‑B.  Returns (best_vector, best_surrogate).
    """
    if beta_schedule is None:
        beta_schedule = [5, 20, 80, 320, 1280, 5120, 20480, 81920]

    n = len(start_a)
    y = np.log(start_a + _EPS)           # initialise log‑softmax parameters

    best_seq = start_a.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        max_iter = 60                     # a few dozen L‑BFGS‑B steps are enough
        functor = _SmoothFunctor(n, beta)

        # minimise; we ask for the gradient (jac=True)
        res = minimize(functor,
                       y,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': max_iter,
                                'disp': False,
                                'gtol': 1e-12})

        y = res.x                         # seed for the next β

        a = _softmax(y)
        val = _objective_surrogate(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # keep a tiny safety margin for the next stage
        y = np.log(best_seq + _EPS)

    return best_seq, best_val

    # shifted power‑law (non‑symmetric)
    for p in np.linspace(0.45, 0.65, 9):            # 0.45 … 0.65
        for delta in (0.05, 0.2, 0.5, 1.0, 2.0):
            seeds.append(_power_shift(n, p, delta))

    # symmetric shifted power‑law – our main hope
    for p in np.linspace(0.45, 0.65, 9):
        for delta in (0.05, 0.2, 0.5, 1.0, 2.0):
            seeds.append(_sym_power_shift(n, p, delta))

    # beta‑shaped families (random parameters)
    for _ in range(6):
        a = rng.uniform(0.2, 2.5)
        b = rng.uniform(0.2, 2.5)
        seeds.append(_beta_shape(n, a, b))

    # Dirichlet draws (different concentrations)
    for conc in (0.02, 0.08, 0.2, 0.6, 1.0, 2.5):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution and a few permutations of it
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(4):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation, keep the best few
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed screening (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Refinement of the top seeds (L‑BFGS‑B → exact EG)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget
        remaining = deadline - time.time()
        slice_time = remaining * 0.22 / max(1, (top_k - idx))
        sub_deadline = time.time() + slice_time
        print(f"\n[Seed {idx+1}] time slice ≈ {slice_time:.1f}s")

        # ---- L‑BFGS‑B continuation (smooth max) ----
        seq_lbfgs, val_lbfgs = _lbfgsb_continuation(seed_vec, sub_deadline)

        # ---- Exact EG polishing (uses whatever is left) ----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_lbfgs, sub_deadline,
                                               max_iters=800_000,
                                               eta0=6.0)
        else:
            seq_eg, val_eg = seq_lbfgs, val_lbfgs

        # choose the better refinement
        if val_lbfgs < val_eg:
            local_seq, local_val = seq_lbfgs, val_lbfgs
        else:
            local_seq, local_val = seq_eg, val_eg

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long EG polishing (use all remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=2_500_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation tweak (may shave a bit more)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + min(30.0, deadline - time.time())
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=100_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
