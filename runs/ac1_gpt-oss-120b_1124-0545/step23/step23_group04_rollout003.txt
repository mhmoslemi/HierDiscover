"""Candidate generator for step‑function minimisation.

The algorithm:
* Normalise to a probability vector (sum 1).
* Build a diverse seed pool (uniform, arcsine, triangular, several
  symmetric Beta(α,α) densities and a few Dirichlet draws).
* Pick the seed with smallest exact surrogate 2·n·max(conv).
* Run a mirror‑descent (exponentiated‑gradient) optimiser on the true
  max‑convolution using the sub‑gradient at the active index.  The
  learning‑rate η_t = η0/√t and a time budget guide the number of
  iterations.
* Finish with a cheap stochastic local walk that transfers a tiny
  amount of mass between two coordinates whenever this lowers the exact
  surrogate.
* The best sequence is finally handed to ``evaluate_sequence`` and stored
  as ``height_sequence_1``.
"""

import time, math, numpy as np

# Helper functions (FFT convolution, softmax, surrogate objective)
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s

def _objective_surrogate(p: np.ndarray) -> float:

    n = len(p)
    conv = _conv_fft(p, p)

    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)

def _exp_gradient_opt(start: np.ndarray,
                     deadline: float,
                     eta0: float = 3.0,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    n = len(start)
    a = start.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of max(conv) at index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        # occasional exact check
        if it % 2000 == 0:
            cur = _objective_surrogate(a)
            if cur < best_val:
                best_val, best_seq = cur, a.copy()

    final = _objective_surrogate(a)
    if final < best_val:
        best_val, best_seq = final, a.copy()
    return best_seq, best_val

def _local_random_search(start: np.ndarray,
                         deadline: float,
                         sigma: float = 0.015,
                         max_attempts: int = 300_000) -> tuple[np.ndarray, float]:
    """Tiny stochastic walk on the simplex that only accepts strictly improving moves."""
    n = len(start)
    rng = np.random.default_rng()
    best = start.copy()
    best_val = _objective_surrogate(best)

    attempts = 0
    while attempts < max_attempts and time.time() < deadline:
        attempts += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best[i] * rng.random()
        if delta <= 0.0:
            continue

        cand[j] += delta
        if cand[i] < 0.0:         # numerical safety
            continue
        val = _objective_surrogate(cand)
        if val < best_val:
            best, best_val = cand, val
    return best, best_val

Candidate generator for the “step‑function” optimisation challenge.

Algorithm
---------

1️⃣  **Seed pool** – Build a diverse set of probability vectors
    (uniform, arcsine, triangular, several symmetric Beta(α,α) densities
    and Dirichlet draws).  All vectors are normalised to sum = 1.

2️⃣  **Best seed** – Evaluate the exact surrogate
    `2·n·max(conv(p,p))` for every seed and keep the one with the smallest
    value.

3️⃣  **Exponentiated‑gradient (mirror descent)** – Refine the best seed on
    the *true* max‑convolution.  For a current vector `p` we locate the
    active index `k = argmax conv(p,p)` and use the sub‑gradient

4️⃣  **Local stochastic walk** – A tiny random walk on the simplex that
    transfers a small amount of mass between two coordinates and only

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)

    fy = np.fft.rfft(y, n=N)

    return e / s
