   which is fast for the lengths we use (n=4000/8000).

3. Refine the best few seeds
   – a short Adam optimiser on a smooth (log‑sum‑exp) approximation
   of the max‑convolution;
   – a long exact exponentiated‑gradient (mirror‑descent) that
   directly minimises the true surrogate.
   Time is split adaptively between the seeds.

4. Final polishing:
   * a long EG run on the current best,
   * a very short permutation‑annealing (swap moves only).

5. Evaluate the final candidate with the provided ``evaluate_sequence``
   and store it in the global variable ``height_sequence_1``.
--------------------------------------------------------------------
All functions are top‑level; no closures or lambdas are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT based linear convolution (real, full)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (seq assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """Monotone decreasing: a_i ∝ (i+δ)^{-p} , i=0…n‑1, δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """Symmetric U‑shape: a_i ∝ (i+δ)^{-p}+((n‑1‑i)+δ)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p) + ((n - 1 - i) + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) shape on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``conc`` (small → spiky)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimiser – exponentiated (mirror) gradient on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (log‑softmax space)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # initialise log‑probabilities (add tiny epsilon for safety)
    log_a = np.log(a + 1e-30)

    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active diagonal (the current max)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # diminishing stepsize η = η₀ / √t
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g

        # log‑softmax back to probabilities
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optimiser – Adam on a smooth (log‑sum‑exp) surrogate of the max
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 15_000) -> tuple[np.ndarray, float]:
    """Adam optimisation of a log‑sum‑exp smooth approximation of max(conv)."""
    if beta_schedule is None:
        # gradually sharpen the smoothing
        beta_schedule = [10, 50, 200, 800, 3_200, 12_800, 51_200, 200_000]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities from log‑softmax
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            conv = _conv_fft(a, a)                      # exact convolution
            shift = np.max(conv)                        # numerical stabiliser
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp                      # soft‑max over k

            # gradient of the smoothed max w.r.t a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                 # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain‑rule to the log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            log_a = np.clip(log_a, -30.0, 30.0)        # avoid overflow

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very simple permutation‑based simulated annealing (swap moves)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Swap‑only simulated annealing, useful for a final tiny tweak."""
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        accept = val < cur_val or (rng.random() < math.exp(-(val - cur_val) / T))
        if accept:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem sizes we will explore (4000 is well‑tested; 8000 gives a finer grid)
    # ------------------------------------------------------------------
    lengths = [4000, 8000]

    all_seeds = []  # list of (vector) – each already ℓ₁‑normalised

    # ------------------------------------------------------------------
    # 1. Build a very rich pool of analytic + random seeds
    # ------------------------------------------------------------------
    for n in lengths:
        # baseline
        all_seeds.append(_uniform(n))

        # monotone shifted power‑law (already used before)
        for p in np.linspace(0.45, 0.70, 7):          # 0.45 … 0.70
            for delta in (0.1, 0.5, 1.0, 2.0, 5.0):
                all_seeds.append(_power_shift(n, p, delta))

        # **new** symmetric shifted power‑law – the genuinely promising family
        for p in np.linspace(0.45, 0.70, 7):
            for delta in (0.1, 0.5, 1.0, 2.0, 5.0):
                all_seeds.append(_sym_power_shift(n, p, delta))

        # beta‑shapes – focus on strongly U‑shaped versions (α,β < 1)
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            for beta in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                all_seeds.append(_beta_shape(n, alpha, beta))

        # low‑concentration Dirichlet draws (push mass to the borders)
        for conc in (0.01, 0.02, 0.05):
            for _ in range(2):
                all_seeds.append(_random_dirichlet(n, conc, rng))

        # previous solution (if any) – helps continuity across runs
        prev = globals().get("height_sequence_1")
        if prev is not None:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                s = arr.sum()
                if s > 0:
                    arr = arr / s
                    all_seeds.append(arr)
                    all_seeds.append(rng.permutation(arr))

        if time.time() >= deadline:
            break   # safety – stop seed generation if we are out of time

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation, keep the best few seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in all_seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Refinement of the top seeds
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget (roughly equal parts)
        remaining = deadline - time.time()
        slice_time = remaining * 0.45 / max(1, (top_k - idx))
        sub_deadline = time.time() + slice_time
        print(f"\n[Seed {idx+1}] allocated time ≈ {slice_time:.1f}s")

        # ---- a quick Adam run on the smooth surrogate ----
        seq_adam, val_adam = _adam_smooth_opt(
            seed_vec,
            sub_deadline,
            iters_per_beta=12_000,
        )

        # ---- exact exponentiated‑gradient polishing (uses whatever time left) ----
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(
                seq_adam,
                sub_deadline,
                max_iters=900_000,
                eta0=6.0,
            )
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # keep the better of the two refinements
        if val_adam < val_eg:
            local_seq, local_val = seq_adam, val_adam
        else:
            local_seq, local_val = seq_eg, val_eg

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long EG polishing (use all remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(
            global_best_seq,
            deadline,
            max_iters=2_500_000,
            eta0=5.0,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation tweak (may shave a bit more)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + min(30.0, deadline - time.time())
        seq_last, val_last = _perm_anneal(
            global_best_seq.copy(),
            short_deadline,
            max_iters=100_000,
        )
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
