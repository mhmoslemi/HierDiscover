Simplex projection algorithm:

def simplex_projection(v):
    # from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _proj_subgrad_opt(start_seq, deadline, max_iters=1_000_000, eta0=0.5):
    a = start_seq copy
    best_seq = a copy; best_val = _objective_surrogate(a)
    eta_initial = eta0
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        # current conv and max idx
        conv = _conv_fft(a, a)
        max_val = conv.max()
        max_idx = int(conv.argmax())
        # compute subgradient g (dimension n)
        n = len(a)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n-1, max_idx)
        i_range = np.arange(i_start, i_end+1)
        rev = max_idx - i_range
        g[i_range] = 2 * a[rev]
        # stepsize
        eta = eta_initial / np.sqrt(it)
        # update
        a_temp = a - eta * g
        # project onto simplex
        a = _simplex_projection(a_temp)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # optional early stopping
    return best_seq, best_val

"""
Propose a step‑function (non‑negative heights) that minimises
`evaluate_sequence`.  

Algorithm Overview
------------------
1. **Seed generation** – a rich collection of analytic families
   (uniform, exponential, shifted power‑law, Gaussian, Beta,
   mixtures of two power‑laws) together with random Dirichlet draws.
   The parameter grids are deliberately fine around the known
   “sweet‑spot’’ (p≈0.55, small shift) to give many high‑quality
   starting points.

2. **Fast surrogate ranking** – each seed is ℓ₁‑normalised (∑a=1) and
   evaluated with the cheap surrogate `2·n·max(conv(a,a))`.  The
   best `top_k` seeds are retained.

3. **Local refinement of the shifted‑power‑law** – the best seed of
   that family is further optimised in the two‑parameter space
   (exponent p, shift δ) by a tiny coordinate‑descent search that
   respects the overall time budget.

4. **Two‑phase optimisation of each top seed**
   * a *smooth Adam* run on a log‑sum‑exp approximation of the max
     convolution (beta schedule up to 5·10⁴, many iterations);
   * a *projected sub‑gradient* descent on the exact max‑convolution,
     i.e. a mirror‑descent in the primal simplex (additive updates,
     projection onto the ℓ₁‑simplex after each step).

5. **Global polishing** – the current best sequence is polished with
   a long projected‑sub‑gradient run for the remaining budget.

6. **Very short random mass‑exchange annealing** – a few thousand
   cheap swaps of tiny probability mass are attempted to shave off
   the last bits.

All optimisation is performed on the ℓ₁‑normalised vectors, so the
surrogate value coincides with the true `evaluate_sequence` output
(the latter only clips and re‑normalises).  The code constantly checks
the time limit (default 1000 s) and stops gracefully, storing the
best sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT convolution (real, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq))  (seq is assumed to sum to 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1.0) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero or negative – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shift(n: int, p: float, delta: float) -> np.ndarray:
    """a_i ∝ (i+δ)^{-p}, i = 0…n‑1, δ>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + delta) ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shape(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_power_shift(
    n: int,
    p1: float,
    delta1: float,
    p2: float,
    delta2: float,
    weight: float,
) -> np.ndarray:
    """Convex combination of two shifted power‑law families."""
    a1 = _power_shift(n, p1, delta1)
    a2 = _power_shift(n, p2, delta2)
    a = weight * a1 + (1.0 - weight) * a2
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Adam optimiser on a smooth approximation of the max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth_opt(
    start_seq: np.ndarray,
    deadline: float,
    beta_schedule=None,
    iters_per_beta: int = 30_000,
) -> tuple[np.ndarray, float]:
    """Adam on a smooth (log‑sum‑exp) surrogate of the max‑convolution."""
    if beta_schedule is None:
        beta_schedule = [10, 50, 200, 800, 3_200, 12_800, 51_200, 200_000]

    n = start_seq.shape[0]
    log_a = np.log(start_seq + 1e-30)

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities (log‑softmax)
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            conv = _conv_fft(a, a)

            # log‑sum‑exp smooth max
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp  # soft‑max weights (length 2n‑1)

            # gradient wrt a (see derivation in previous solution)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)   # length = 3n‑2
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # gradient wrt log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            log_a = np.clip(log_a, -30.0, 30.0)   # keep numbers stable

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------
def _proj_subgrad_opt(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Primal mirror‑descent on the exact max‑convolution.
    Updates are additive (a ← a – η·g) followed by projection
    onto the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    n = a.shape[0]

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and index of current active diagonal
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        rev = max_idx - i_range
        g[i_range] = 2.0 * a[rev]

        # diminishing stepsize
        eta = eta0 / math.sqrt(it)

        # additive update and simplex projection
        a = _simplex_projection(a - eta * g)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny random “mass‑exchange’’ annealing (swap tiny ε between two positions)
# ----------------------------------------------------------------------
def _mass_exchange_anneal(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    eps_scale: float = 0.002,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    n = start_seq.shape[0]
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best = cur.copy()
    best_val = cur_val

    # temperature schedule (linear cooling)
    T0 = cur_val
    T1 = cur_val * 1e-6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T0 + (T1 - T0) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a tiny amount ε from i to j (or opposite)
        eps = eps_scale * rng.random() * min(cur[i], cur[j])
        if rng.random() < 0.5:
            ns = cur.copy()
            ns[i] -= eps
            ns[j] += eps
        else:
            ns = cur.copy()
            ns[i] += eps
            ns[j] -= eps

        # ensure non‑negativity
        if ns[i] < 0 or ns[j] < 0:
            continue

        # renormalise (actually sum unchanged, but projection protects against rounding)
        ns = _simplex_projection(ns)

        val = _objective_surrogate(ns)
        accept = val < cur_val or (rng.random() < math.exp(-(val - cur_val) / max(T, 1e-12)))
        if accept:
            cur = ns
            cur_val = val
            if val < best_val:
                best = ns.copy()
                best_val = val

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine builds a diverse set of analytic
    seeds, keeps the most promising ones, refines each with a
    smooth‑Adam stage followed by an exact projected‑subgradient stage,
    performs a tiny coordinate search on the shifted‑power‑law family,
    and finishes with a short mass‑exchange anneal.  The best sequence
    found is stored in the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin so we always finish before the external timeout
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – a compromise between resolution and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) – gives a good warm‑start
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1")
    prev_vec = None
    if prev_seq is not None:
        try:
            arr = np.asarray(prev_seq, dtype=np.float64)
            if arr.shape == (n,):
                prev_vec = np.maximum(arr, 0.0)
                total = prev_vec.sum()
                if total > 0:
                    prev_vec = prev_vec / total
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1. Build a very rich pool of seeds (analytic + random)
    # ------------------------------------------------------------------
    seeds = []
    seed_info = []   # (type, params) – helps later local refinement

    # basic families ----------------------------------------------------
    seeds.append(_uniform(n));                 seed_info.append(("uniform", None))
    seeds.append(_exp_decay(n, rate=0.0012));  seed_info.append(("exp_decay", 0.0012))
    seeds.append(_exp_decay(n, rate=0.0024));  seed_info.append(("exp_decay", 0.0024))
    seeds.append(_sym_exp_decay(n, rate=0.0020)); seed_info.append(("sym_exp", 0.0020))
    seeds.append(_power(n, p=0.70));           seed_info.append(("power", 0.70))
    seeds.append(_power(n, p=1.10));           seed_info.append(("power", 1.10))

    # shifted power‑law – fine grid around the known sweet‑spot
    for p in np.linspace(0.45, 0.65, 9):            # 0.45 … 0.65 step 0.025
        for delta in (0.1, 0.5, 1.0, 2.0, 5.0):
            seeds.append(_power_shift(n, p, delta))
            seed_info.append(("power_shift", (p, delta)))

    # Gaussian bumps (centre fixed, varied width)
    centre = (n - 1) / 2.0
    for sigma in (n * 0.05, n * 0.10, n * 0.20):
        seeds.append(_gaussian_shape(n, mu=centre, sigma=sigma))
        seed_info.append(("gauss", sigma))

    # Beta‑shaped families (random parameters)
    for _ in range(8):
        a = rng.uniform(0.2, 2.5)
        b = rng.uniform(0.2, 2.5)
        seeds.append(_beta_shape(n, a, b))
        seed_info.append(("beta", (a, b)))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.08, 0.2, 0.6, 1.0, 2.5):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
            seed_info.append(("dirichlet", conc))

    # Mixtures of two shifted power‑laws (adds extra flexibility)
    for _ in range(12):
        p1 = rng.uniform(0.45, 0.65)
        d1 = rng.uniform(0.1, 5.0)
        p2 = rng.uniform(0.45, 0.65)
        d2 = rng.uniform(0.1, 5.0)
        w = rng.random()
        seeds.append(_mix_power_shift(n, p1, d1, p2, d2, w))
        seed_info.append(("mix_power", (p1, d1, p2, d2, w)))

    # previous solution and a few permutations of it --------------------
    if prev_vec is not None:
        seeds.append(prev_vec);               seed_info.append(("prev", None))
        for _ in range(4):
            seeds.append(rng.permutation(prev_vec))
            seed_info.append(("prev_perm", None))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation, keep the best few
    # ------------------------------------------------------------------
    evaluated = []    # list of (vector, surrogate_val, info)
    for vec, info in zip(seeds, seed_info):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val, info))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_entries = evaluated[:top_k]
    top_seeds = [vec for vec, _, _ in top_entries]

    print("\n--- Seed ranking (surrogate) ---")
    for i, (_, v, _) in enumerate(top_entries):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Local refinement of the *best* shifted‑power‑law seed
    # ------------------------------------------------------------------
    # Find the best power‑shift entry (if any)
    best_ps_val = math.inf
    best_ps_params = None
    for vec, val, info in evaluated:
        typ, pars = info
        if typ == "power_shift":
            if val < best_ps_val:
                best_ps_val = val
                best_ps_params = pars
    if best_ps_params is not None:
        # tiny coordinate descent on (p, delta) – respects the global deadline
        p, delta = best_ps_params
        step_p = 0.02
        step_d = 0.5
        cur_best_vec = _power_shift(n, p, delta)
        cur_best_val = _objective_surrogate(cur_best_vec)
        print("\n[Power‑Shift local search] start: p={:.4f}, δ={:.4f}, val={:.6f}".format(p, delta, cur_best_val))
        while time.time() < deadline and (step_p > 1e-4 or step_d > 1e-3):
            improved = False
            for dp in (0.0, -step_p, step_p):
                for dd in (0.0, -step_d, step_d):
                    if dp == 0.0 and dd == 0.0:
                        continue
                    cand_p = p + dp
                    cand_d = delta + dd
                    if cand_p <= 0 or cand_d <= 0:
                        continue
                    cand_vec = _power_shift(n, cand_p, cand_d)
                    cand_val = _objective_surrogate(cand_vec)
                    if cand_val < cur_best_val:
                        p, delta = cand_p, cand_d
                        cur_best_vec = cand_vec
                        cur_best_val = cand_val
                        improved = True
            if not improved:
                step_p *= 0.5
                step_d *= 0.5
        # add the refined power‑shift vector as an extra seed
        seeds.append(cur_best_vec)
        seed_info.append(("power_shift_refined", (p, delta)))
        # make sure it participates in the top‑k selection
        evaluated.append((cur_best_vec, cur_best_val, ("power_shift_refined", (p, delta))))
        evaluated.sort(key=lambda x: x[1])
        top_entries = evaluated[:top_k]
        top_seeds = [vec for vec, _, _ in top_entries]
        print(f"[Power‑Shift] refined to p={p:.5f}, δ={delta:.5f}, val={cur_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refinement of the top seeds (Adam → exact EG)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # Allocate roughly 70 % of the remaining budget to per‑seed refinements
    while time.time() < deadline and (deadline - time.time()) > 1.0:
        remaining_total = deadline - time.time()
        refine_budget = remaining_total * 0.70
        if refine_budget < 10.0:
            break
        slice_per = refine_budget / max(1, len(top_seeds))
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + slice_per
            print(f"\n[Seed {idx+1}] allocated time ≈ {slice_per:.1f}s")

            # ---- Stage A: smooth Adam ----
            seq_adam, val_adam = _adam_smooth_opt(
                seed_vec,
                sub_deadline,
                iters_per_beta=15_000,
            )

            # ---- Stage B: exact projected sub‑gradient (if time permits) ----
            if time.time() < sub_deadline:
                seq_pg, val_pg = _proj_subgrad_opt(seq_adam, sub_deadline, max_iters=800_000, eta0=0.6)
            else:
                seq_pg, val_pg = seq_adam, val_adam

            # choose the better of the two
            if val_adam < val_pg:
                local_seq, local_val = seq_adam, val_adam
            else:
                local_seq, local_val = seq_pg, val_pg

            print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

            if local_val < global_best_val:
                global_best_seq = local_seq
                global_best_val = local_val
                print(f"   *** New global best = {global_best_val:.6f}")

        # after one full pass, break – the loop will re‑divide remaining time
        break

    # ------------------------------------------------------------------
    # 5. Final long projected‑subgradient polishing (use all remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long projected‑subgradient polishing ---")
        final_seq, final_val = _proj_subgrad_opt(
            global_best_seq,
            deadline,
            max_iters=2_500_000,
            eta0=0.5,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final PG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Very short random mass‑exchange tweak (may shave a bit more)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + min(30.0, deadline - time.time())
        seq_last, val_last = _mass_exchange_anneal(
            global_best_seq.copy(),
            short_deadline,
            max_iters=100_000,
        )
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final mass‑exchange] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
