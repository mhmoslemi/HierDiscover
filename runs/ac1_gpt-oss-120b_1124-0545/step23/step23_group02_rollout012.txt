def _grad_logsumexp(v, beta):
    a = _softmax(v)
    conv = _conv_fft(a, a)
    maxexp = None # not needed
    # compute softmax of beta*conv
    scaled = beta * conv
    # stable shift
    max_s = np.max(scaled)
    exp_scaled = np.exp(scaled - max_s)
    sum_exp = np.exp(-max_s) * np.sum(exp_scaled) # but ignore complex. Actually compute w = beta * exp(scaled) / sum(exp(scaled)).
    # Use stable.
    exp_scaled = np.exp(scaled - max_s)
    sum_exp = exp_scaled.sum()
    p = exp_scaled / sum_exp
    w = beta * p
    # Compute grad_a:
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_a = 2.0 * full[n-1:n-1+n]
    grad_v = a * (grad_a - np.dot(grad_a, a))
    loss = max_s + np.log(sum_exp)  # logsumexp = max_s + log(sum_exp)
    return loss, grad_v

"""
Improved step‑function optimiser.

The algorithm works on vectors a∈ℝⁿ₊ normalised to sum(a)=1,
i.e. a probability distribution.  For a given a the true objective
(used by ``evaluate_sequence``) equals

    f(a) = 2·n·max(conv(a,a))                (∑a=1)

Because f is invariant under scaling we can stay on the simplex.
The optimiser consists of three phases:

1. **Warm‑start** – take the previous best construction (if any) or a
   uniform distribution.

2. **Smooth max minimisation** – minimise a smooth approximation of the
   max, namely  L(v)=log∑exp(β·conv(softmax(v),softmax(v)))  with
   Adam on the log‑softmax parameters v.  The temperature β is
   increased gradually (5 → 20 → 80 → 200 → 500) to focus more and more
   on the true maximum while keeping the optimisation tractable.
   The gradient of L is derived analytically and computed with FFT‑based
   convolutions, giving O(n log n) per iteration.

3. **Exact max polishing** – after the smooth phase we run a classic
   exponentiated‑gradient sub‑gradient method on the exact max‑convolution
   (the same routine that was used in the previous submission) to squeeze
   the final value a little further.

All steps respect the 1000 s wall‑clock budget and store the best
sequence found in the global variable ``height_sequence_1``.  The code
requires only NumPy (SciPy is optional – used when available for the
Adam optimiser).  No file I/O or network access is performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used for a fast L‑BFGS fallback if desired
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Core numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (O(n log n)).
    Returns an array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; result sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate / true objective (sum(a) is assumed to be 1)
# ----------------------------------------------------------------------
def _objective_true(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a normalised vector."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Gradient of the smooth “log‑sum‑exp” approximation of the max‑conv.
# ----------------------------------------------------------------------
def _logsumexp_obj_and_grad(v: np.ndarray, beta: float):
    """
    Returns (loss, grad_v, conv) where

        loss = log∑_k exp(beta·conv_k(a)),
        grad_v = ∇_v loss,
        conv   = conv(a,a),

    with a = softmax(v).  The gradient w.r.t. v is obtained via the
    chain‑rule (soft‑max) and the analytic Jacobian of the convolution.
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)                     # length 2n‑1
    # ----- log‑sum‑exp (stable) --------------------------------------
    beta_conv = beta * conv
    shift = np.max(beta_conv)
    exp_shifted = np.exp(beta_conv - shift)
    sum_exp = exp_shifted.sum()
    loss = shift + np.log(sum_exp)             # = log‑sum‑exp

    # ----- gradient w.r.t. a (softmax space) -------------------------
    # p_i = exp(beta·conv_i) / Σ exp(beta·conv)
    p = exp_shifted / sum_exp
    w = beta * p                               # dL/d conv_i

    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length 3n‑2
    n = len(a)
    grad_a = 2.0 * full[n - 1:n - 1 + n]       # central slice, factor 2

    # ----- gradient w.r.t. v (log‑softmax parameters) ---------------
    grad_v = a * (grad_a - np.dot(grad_a, a))
    return loss, grad_v, conv


# ----------------------------------------------------------------------
# Adam optimiser on the smooth objective.
# ----------------------------------------------------------------------
def _adam_optimize_logsumexp(start_a: np.ndarray,
                             beta_schedule,
                             deadline: float,
                             eval_every: int = 100,
                             eta0: float = 0.4,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8):
    """
    Performs Adam on the log‑sum‑exp loss with a piece‑wise constant
    sequence of betas.  Returns the best vector (on the simplex) seen
    during the run together with its true objective value.
    """
    n = len(start_a)
    v = np.log(start_a + 1e-30)               # log‑softmax parameters
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)                     # second‑moment accumulator

    # overall best
    best_seq = start_a.copy()
    best_val = _objective_true(best_seq)

    step = 0
    beta_idx = 0
    cur_beta, next_change = beta_schedule[beta_idx]

    while time.time() < deadline:
        step += 1

        # switch beta when the schedule says so
        if step > next_change and beta_idx + 1 < len(beta_schedule):
            beta_idx += 1
            cur_beta, next_change = beta_schedule[beta_idx]

        loss, grad_v, conv = _logsumexp_obj_and_grad(v, cur_beta)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** step)
        v2_hat = v2 / (1.0 - beta2 ** step)

        eta = eta0 / math.sqrt(step)          # simple decay
        v -= eta * m_hat / (np.sqrt(v2_hat) + eps)

        # occasional evaluation of the true objective (cheap: we already have conv)
        if step % eval_every == 0:
            a = _softmax(v)
            cur_val = _objective_true(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Adam] step {step:7d}  β={cur_beta:6.1f}  val={cur_val:.6f}")

    # final evaluation (in case the last iteration was better)
    a = _softmax(v)
    cur_val = _objective_true(a)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a.copy()
        print(f"[Adam] final eval  β={cur_beta:6.1f}  val={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 800_000,
                            eta0: float = 8.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the exact max‑convolution.
    Uses all “active’’ indices whose value is within tol·max of the current
    maximum.  Returns the best sequence seen and its true objective.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_true(best_seq)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of the max (average over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑like momentum
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_true(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 50000 == 0:
                print(f"[EG] it {it:7d}  val={cur_val:.6f}")

    return best_seq, best_val

# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the 1000 s wall‑clock budget, stores the best found
    sequence in the global variable ``height_sequence_1`` and returns it.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0               # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Warm‑start: use previous best if it exists, otherwise uniform
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple)) and len(prev) == n:
        a0 = np.asarray(prev, dtype=np.float64)
        s = a0.sum()
        if s > 0:
            a0 = a0 / s
        else:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)      # uniform start

    # global best tracking
    best_seq = a0.copy()
    best_val = _objective_true(best_seq)
    print(f"[Init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth max minimisation with Adam
    # ------------------------------------------------------------------
    # schedule = list of (beta, step_at_which_this_beta_becomes_active)
    beta_schedule = [
        (5.0,   20000),   # beta=5  up to step 20k
        (20.0,  60000),   # beta=20 up to 60k
        (80.0, 150000),   # beta=80 up to 150k
        (200.0,300000),   # beta=200 up to 300k
        (500.0, 500000),  # beta=500 afterwards
    ]

    # run Adam (will stop automatically when the deadline is hit)
    seq_smooth, val_smooth = _adam_optimize_logsumexp(
        start_a=best_seq,
        beta_schedule=beta_schedule,
        deadline=deadline,
        eval_every=200,          # more frequent checks at the beginning
        eta0=0.5,
    )
    if val_smooth < best_val:
        best_val = val_smooth
        best_seq = seq_smooth.copy()
        print(f"[Phase 1] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact max polishing via exponentiated gradient
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Phase 2: Exact max polishing (EG) ---")
        seq_final, val_final = _exp_gradient_opt_multi(
            start_seq=best_seq,
            n=n,
            deadline=deadline,
            max_iters=4_000_000,
            eta0=6.0,
            tol=1e-9,
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = seq_final.copy()
            print(f"[Phase 2] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation via the provided evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print("\n=== Finished ===")
    print(f"Surrogate (2·n·max(conv))   = {2.0 * n * np.max(_conv_fft(best_seq, best_seq)):.6f}")
    print(f"evaluate_sequence() result = {true_val}")

    # store for possible warm‑starts in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
