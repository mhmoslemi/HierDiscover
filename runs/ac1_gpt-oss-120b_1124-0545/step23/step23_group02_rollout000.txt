"""
Propose a step‑function that improves the current best bound.

The algorithm works in three phases on a *fixed length* `n` (default 8000):

1. **Initialisation** – use the previously stored best sequence if it has the
   required length, otherwise fall back to the uniform distribution.
   The vector is normalised to sum‑to‑one (the objective is scale‑invariant).

2. **Projected sub‑gradient descent on the exact max–convolution** –  
   The objective `f(a) = max(conv(a,a))` is convex in the neighbourhood of a
   minimiser, and a sub‑gradient is easy to compute:
   for a maximising lag `k` the gradient is
        g_i = 2·a_{i‑k}   (zero out‑of‑range indices).
   We perform many SGD steps with a momentum term (Adam‑style) and after each
   step project back onto the probability simplex using the O(n log n) Euclidean
   projection of Duchi et al.  The step‑size decays as `η_t = η₀/√t`; the
   schedule and momentum are tuned for fast convergence.

3. **Fine‑grained simulated annealing** – once the time budget is almost
   exhausted a short SA run makes occasional random perturbations that can
   escape tiny plateaus.

All heavy work is done with FFT‑based convolutions (`_conv_fft`) which run in
`O(n log n)` and fit comfortably into the 1000 s wall‑clock limit.
The best sequence seen during the optimisation is stored globally as
`height_sequence_1` and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Core numeric utilities (FFT convolution, simplex projection, soft‑max)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {w≥0, sum w = 1}.
    Implementation of Duchi et al. (2008) – O(n log n).
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a normalised (sum=1) vector."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Sub‑gradient for the max‑convolution
# ----------------------------------------------------------------------
def _subgrad_max_conv(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Returns a sub‑gradient of f(a)=max(conv(a,a)) and the current max value.
    The sub‑gradient is taken w.r.t. a single maximising lag (the first one).
    """
    n = a.shape[0]
    conv = _conv_fft(a, a)
    max_val = float(conv.max())
    # index of a maximising lag (first occurrence)
    idx = int(conv.argmax())
    shift = idx - (n - 1)          # zero lag corresponds to index n‑1
    g = np.zeros_like(a)

    if shift >= 0:
        # contribution a[i‑shift] → g[i]; valid for i >= shift
        g[shift:] = a[: n - shift]
    else:
        # shift < 0 : contribution a[i‑shift] where i‑shift ≥ 0
        g[: n + shift] = a[-shift:]
    g *= 2.0                      # derivative of a_i a_{i+shift}
    return g, max_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with Adam‑style momentum
# ----------------------------------------------------------------------
def _sgd_max_conv(start: np.ndarray,
                  deadline: float,
                  eta0: float = 8.0,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-12,
                  max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Minimise max(conv) on the simplex using projected Adam.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start.copy()
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 1
    while time.time() < deadline and it <= max_iters:
        g, _ = _subgrad_max_conv(a)

        # Adam update (on the *raw* parameters, not on log‑softmax)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)          # classic decay
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _proj_simplex(a)                # stay on the simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing polish (Gaussian perturbations)
# ----------------------------------------------------------------------
def _simulated_annealing(start: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 120_000,
                         start_sigma: float = 0.0015,
                         T_start: float = 0.015,
                         T_end: float = 5e-9) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0
    n = a.shape[0]

    while time.time() < deadline and attempts < max_attempts:
        # linear schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val
