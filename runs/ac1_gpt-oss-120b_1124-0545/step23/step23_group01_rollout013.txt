Algorithm
---------
*   Work on a probability vector of length ``n = 4000`` (the objective

    landscape while the final stage follows the true max.
*   Gradients are obtained analytically.
    For the smoothed objective the weight vector
    ``w_k = exp(α·(c_k−max(c))) / ∑exp(…)`` (a softmax over the
    convolution entries) gives
    ``∇_a smooth_max = 2·(w ★ a)``,
    where ``★`` denotes the discrete cross‑correlation.
    The cross‑correlation is evaluated efficiently with a single FFT
    by convolving ``w`` with the reversed vector ``a[::-1]``.
*   After the annealing stages a final polishing pass with the exact
    sub‑gradient (α = None) and a long exponentiated‑gradient run is
    performed.
*   The best sequence found (according to the true surrogate
    ``2·n·max(conv)``) is stored in the global ``height_sequence_1`` and
    returned.

The routine respects the 1000 s budget, prints progress and finally
evaluates the true ``evaluate_sequence`` value.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (outputs sum‑to‑1 positive vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a probability vector: 2·n·max(conv(seq,seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Adam optimiser (smooth or exact)
# ----------------------------------------------------------------------
def _adam_opt_general(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300_000,
                      lr0: float = 0.2,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8,
                      alpha: float | None = None) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑softmax parameters.

    Parameters
    ----------
    start_seq : np.ndarray
        Initial probability vector (must sum to 1).
    n : int
        Length of the vector.
    deadline : float
        Unix timestamp after which the optimiser stops.
    max_iters : int
        Upper bound on Adam iterations.
    lr0, beta1, beta2, eps : Adam hyper‑parameters.
    alpha : float | None
        Smoothing parameter for the max:
        * None → exact max (sub‑gradient of the true max)
        * finite → log‑sum‑exp smoothing with temperature α.
    Returns
    -------
    best_seq, best_val : (np.ndarray, float)
        Best probability vector seen and its surrogate value.
    """
    # initialise log‑softmax variables
    log_a = np.log(start_seq + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)                     # current probability vector
        conv = _conv_fft(a, a)                  # full auto‑convolution (len 2n‑1)

        # surrogate (true) objective – used for bookkeeping
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        if alpha is None:                       # exact max → sub‑gradient
            k = int(np.argmax(conv))            # active index
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g_a = np.zeros_like(a)
            g_a[idx_i] = 2.0 * a[rev_idx]
        else:                                   # smoothed max (log‑sum‑exp)
            maxc = np.max(conv)
            # softmax weights over the convolution entries – numerically stable
            exp_term = np.exp(alpha * (conv - maxc))
            w = exp_term / exp_term.sum()       # shape (2n‑1,)
            # cross‑correlation w ★ a  = 2· (w * a_rev)[n‑1 : n‑1+n]
            a_rev = a[::-1]
            conv_rev = _conv_fft(w, a_rev)      # length 3n‑2
            g_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # chain rule for the log‑softmax parametrisation
        dot = np.dot(g_a, a)
        grad_v = a * (g_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_v
        v = beta2 * v + (1.0 - beta2) * (grad_v ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # parameter update
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        log_a -= step

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient polishing (exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient (mirror descent) on the true max‑objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g = np.zeros_like(a)
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    n = 4000

    # ------------------------------------------------------------------
    # Build a diversified pool of seeds
    # ------------------------------------------------------------------
    candidates = []

    # 1) previous best, if any
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                candidates.append(arr / arr.sum())
        except Exception:
            pass

    # 2) exponential decays with several rates
    for rate in (0.0004, 0.0006, 0.001, 0.002):
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-rate * i)
        candidates.append(a / a.sum())

    # 3) symmetric triangular shape
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    tri = 1.0 - np.abs(i - centre) / centre
    tri = np.maximum(tri, 0.0)
    candidates.append(tri / tri.sum())

    # 4) uniform distribution
    candidates.append(np.full(n, 1.0 / n))

    # 5) a few Dirichlet draws with various concentration parameters
    for alpha in (0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            cand = rng.dirichlet(alpha=np.full(n, alpha))
            candidates.append(cand)

    # ------------------------------------------------------------------
    # Optimisation schedule (smooth → exact)
    # ------------------------------------------------------------------
    stages = [
        (2.0, 0.5, 15_000),
        (5.0, 0.3, 20_000),
        (10.0, 0.2, 25_000),
        (30.0, 0.15, 30_000),
        (None, 0.10, 60_000),     # final exact‑max phase
    ]

    best_overall_seq = None
    best_overall_val = float("inf")

    for idx, seed_seq in enumerate(candidates, start=1):
        if time.time() > deadline - 20.0:
            break

        seq = np.maximum(seed_seq, 0.0)
        if seq.sum() == 0.0:
            continue
        seq = seq / seq.sum()

        cur_seq = seq.copy()
        print(f"[Seed {idx}/{len(candidates)}] starting optimisation")

        for stage_no, (alpha, lr0, max_it) in enumerate(stages, start=1):
            if time.time() > deadline - 5.0:
                break

            cur_seq, cur_val = _adam_opt_general(
                cur_seq,
                n,
                deadline,
                max_iters=max_it,
                lr0=lr0,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                alpha=alpha,
            )

            print(f"  stage {stage_no} (α={alpha}) → surrogate {cur_val:.6f}")

            if cur_val < best_overall_val:
                best_overall_val = cur_val
                best_overall_seq = cur_seq.copy()
                print(f"  ** new global best = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with a long exponentiated‑gradient run
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        best_overall_seq = np.full(n, 1.0 / n)

    if time.time() < deadline:
        print("\n--- Polishing with exponentiated‑gradient (exact max) ---")
        polished_seq, polished_val = _exp_gradient_opt(
            best_overall_seq,
            n,
            deadline,
            max_iters=4_000_000,
            eta0=12.0,
        )
        if polished_val < best_overall_val:
            best_overall_seq = polished_seq
            best_overall_val = polished_val
            print(f"[Polish] improved surrogate = {best_overall_val:.6f}")
        else:
            print(f"[Polish] no improvement (best unchanged)")

    # ------------------------------------------------------------------
    # Evaluate the true objective via the provided function
    # ------------------------------------------------------------------

        true_val = evaluate_sequence(best_overall_seq.tolist())
        print(
            f"\n=== Final result ===\nSurrogate ≈ {best_overall_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # Store the result for possible continuation runs
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
```

This implementation explores a richer set of initial vectors, uses a temperature‑annealed Adam optimiser that follows a smooth approximation of the max‑convolution, and finishes with a long exact‑max exponentiated‑gradient polishing. It stays within the 1000‑second budget, prints progress, and stores the best vector in ``height_sequence_1``.
