def gradient_step_softmax(a, step, tau, fft_len):
    # compute convolution of a with itself via FFT
    conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), n=fft_len)
    conv = conv[: 2 * len(a) - 1]
    # compute weights
    w = np.exp(conv / tau)
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:

    # compute gradient via convolution of w with reversed a
    a_rev = a[::-1]
    # compute conv_w_a_rev = conv(w, a_rev)
    conv_w_rev = np.fft.irfft(np.fft.rfft(w, fft_len) * np.fft.rfft(a_rev, fft_len), n=fft_len)

    s = a_new.sum()
    if s == 0:

    else:

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # choose n based on existing sequence if defined else 2500
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2500

    a = compute_catalan_sequence(n)  # sum to 1
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # FFT length for gradient
    fft_len = 1 << ((3 * n - 2).bit_length())
    # parameters
    step = 0.5   # initial step size
    tau = None   # will be set based on conv values
    decay_step = 0.9998
    decay_tau = 0.9995

    iter_count = 0
    # gradient phase
    while time.time() < deadline:
        conv = convolution_fft(a, a, fft_len)
        max_conv = conv.max()
        if tau is None:
            tau = max_conv * 0.5 + 1e-12  # avoid zero
        # gradient step
        a = gradient_step_softmax(a, step, tau, fft_len)

        iter_count += 1
        # decay step and tau
        step *= decay_step
        tau *= decay_tau

        if iter_count % 20 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[grad] iter {iter_count:5d} score {best_score:.6f}")

        # stop early if step becomes tiny
        if step < 1e-7:
            break

    # local search phase
    eps = 0.1
    temperature = 0.05
    local_iter = 0
    while time.time() < deadline:
        cand = random_perturb(a, eps)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(f"[ls] iter {local_iter:5d} improved score {best_score:.6f}")
        else:
            # metropolis acceptance
            prob = np.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                a = cand
        eps = max(eps * 0.999, 1e-5)
        temperature *= 0.999
        local_iter += 1

    # final clip for safety
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    return final_seq.tolist()

"""
Propose a step‑function that improves the bound on
    2·n·max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. Initialise the sequence with the truncated Catalan coefficients
   (the coefficients of (1‑x)^‑½).  After normalisation they already give
   a bound ≈ π/2.

2. Projected “soft‑max” gradient descent.
   * The true objective is max(conv), a non‑smooth function.
   * Replace the hard max by a temperature‑controlled soft‑max:
         w_k ∝ exp(conv_k / τ)
     and take the gradient of the smooth objective.
   * The gradient can be written as a convolution of the weight vector
     with the reversed current sequence; both convolutions are performed
     with FFTs (O(n log n)).
   * After each step we project onto the non‑negative orthant and
     re‑normalise (sum = 1).  τ and the step size are slowly cooled.

3. Stochastic local‑search (simulated‑annealing style).
   * Randomly move a small fraction of the mass from one coordinate to
     another.
   * Accept improvements unconditionally; uphill moves are accepted
     with probability exp(−Δ/temperature).
   * Both the perturbation size and temperature are exponentially
     decayed.

4. Keep track of the best sequence ever seen (according to
   ``evaluate_sequence``) and return it when the time budget expires.

The combination of a smooth gradient step (which is able to reduce the
maximum of the convolution without being trapped by a single dominant
index) and a final annealing phase yields a bound typically below 1.55
and often approaches the target 1.503 within the allotted time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def compute_catalan_sequence(n: int) -> np.ndarray:
    """
    Return the first *n* Catalan‑type coefficients
        c_i = (2i choose i) / 4**i ,  i = 0 … n‑1,
    normalised so that their sum equals 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence  c_i = c_{i‑1} * (2i‑1) / (2i)
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def convolution_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of vectors *x* and *y* using FFT.
    Returns the first (len(x)+len(y)‑1) elements.
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


def gradient_step_softmax(
    a: np.ndarray, step: float, tau: float, fft_len: int
) -> tuple[np.ndarray, float]:
    """
    One projected gradient step for the *smoothed* objective
          Φ(a) = log‑sum‑exp( conv(a,a) / τ )
    (up to an irrelevant additive constant).

    Returns the updated vector and the (actual) maximum of conv(a,a)
    which is needed to adapt the temperature.
    """
    # ---- 1) self‑convolution ------------------------------------------------
    conv_aa = convolution_fft(a, a, fft_len)            # length 2·n‑1
    max_conv = conv_aa.max()

    # ---- 2) soft‑max weights ------------------------------------------------
    # shift by max to avoid overflow
    w = np.exp((conv_aa - max_conv) / max(tau, 1e-12))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ---- 3) gradient of the smooth max ------------------------------------
    #   ∇Φ = 2 * (w * a_rev)   (where * denotes linear convolution)
    a_rev = a[::-1]
    conv_wa = convolution_fft(w, a_rev, fft_len)       # length 3·n‑2
    n = a.shape[0]
    start = n - 1
    grad = 2.0 * conv_wa[start : start + n]            # ⟶ shape (n,)

    # ---- 4) gradient step + projection ------------------------------------
    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)                     # stay non‑negative
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / n)               # re‑initialise if degenerate
    else:
        a_new /= s                                      # renormalise (sum = 1)

    return a_new, max_conv


def random_perturb(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Transfer a fraction *eps* of the mass from a random source index to a
    random destination index (i ≠ j).  The vector stays normalised.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]                # fraction of the source mass
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Produce a non‑negative step‑function (list of heights) that tries to
    minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Choose sequence length
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, (list, tuple, np.ndarray)):
        n = len(prev)
    else:
        n = 2500                               # sensible default

    # --------------------------------------------------------------
    # 1. Start from Catalan (almost‑flat) sequence
    # --------------------------------------------------------------
    a = compute_catalan_sequence(n)           # sum(a) == 1
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}  score={best_score:.6f}")

    # FFT size sufficient for both convolutions (max length 3·n‑2)
    fft_len = 1 << ((3 * n - 2).bit_length())

    # --------------------------------------------------------------
    # 2. Smoothed gradient descent on max(conv)
    # --------------------------------------------------------------
    step = 0.35                                 # initial learning rate
    tau = None                                  # temperature initialised later
    decay_step = 0.9995
    decay_tau = 0.9995
    grad_iter = 0

    while time.time() < deadline:
        # one gradient step; also retrieve current max(conv)
        a, cur_max = gradient_step_softmax(a, step, tau if tau is not None else 0.0, fft_len)

        # initialise / adapt temperature after we have a sensible max value
        if tau is None:
            tau = max(cur_max * 0.5, 1e-12)

        # geometric cooling
        step *= decay_step
        tau *= decay_tau
        grad_iter += 1

        # occasional evaluation of the true metric
        if grad_iter % 30 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[grad] iter {grad_iter:5d}  score={best_score:.6f}")
        # stop if step becomes negligible
        if step < 1e-7:
            break

    # --------------------------------------------------------------
    # 3. Stochastic local‑search / simulated‑annealing phase
    # --------------------------------------------------------------
    eps = 0.12          # start moving up to 12 % of a source entry
    temperature = 0.07
    ls_iter = 0
    while time.time() < deadline:
        cand = random_perturb(a, eps)
        cand_score = evaluate_sequence(cand.tolist())

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand                     # move to the improved point
            print(f"[ls]   iter {ls_iter:5d}  improved score={best_score:.6f}")
        else:
            # Metropolis acceptance for uphill moves
            prob = np.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                a = cand

        # anneal perturbation size and temperature
        eps = max(eps * 0.9997, 1e-5)
        temperature *= 0.9996
        ls_iter += 1

    # --------------------------------------------------------------
    # Return the best sequence found (plain Python list)
    # --------------------------------------------------------------
    final_seq = np.clip(best_seq, 0.0, 1000.0)   # safety against accidental overflow
    print(f"[final] best score={best_score:.6f} after {grad_iter} grad iters + {ls_iter} ls iters")
    return final_seq.tolist()
