def _project_simplex(v):
    # v is 1D np array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

Works for sum=1.

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Algorithm
---------
* The objective for a sum‑normalised sequence a (∑a = 1) is
      score = 2·n·max(conv(a,a)).
  Reducing the maximal entry of the linear convolution is therefore enough.

* We perform a continuous optimisation on the simplex using a smooth
  approximation of the max‑operator:
      f(a) = (1/β)·log Σ_k exp(β·(a*a)_k)   (β grows during the run)
  Its gradient is a weighted sum of the gradients of the individual
  convolution entries.  For a fixed weight vector w (∑w = 1) the gradient
  is   ∇f(a) = 2·( w ⋆ a )  (cross‑correlation).
  The correlation is evaluated efficiently with FFTs.

* Update rule: a ← a – η·∇f(a)  followed by Euclidean projection onto the
  probability simplex (non‑negative, sum = 1).  η is reduced as 1/√t.

* The optimisation runs from many random Dirichlet starts, occasionally
  jittering the current best solution to escape local minima.  The fast
  surrogate score (2·n·max(conv)) is used for cheap comparison; the
  official ``evaluate_sequence`` is called only when an improvement is
  observed.

* After the main loop a small random‑perturbation hill‑climb refines the
  best candidate.

The whole routine respects the 1000 s time budget, prints progress and
returns the best sequence found (as a plain Python list)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (real FFT, length 2·n‑1, returned as 1‑D array).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _fast_score(a: np.ndarray) -> float:
    """
    Fast surrogate for evaluate_sequence when sum(a) == 1:
        score = 2·n·max(conv(a,a))
    """
    n = a.shape[0]
    return 2.0 * n * _conv_full(a).max()


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute a soft‑max distribution over convolution entries.
    """
    max_c = conv.max()
    # stable exponentiation
    exp_term = np.exp(beta * (conv - max_c))
    return exp_term / exp_term.sum()


def _gradient_from_weights(weights: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate:
        ∇f(a) = 2·(weights ⋆ a)
    The cross‑correlation is performed via FFT.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed a
    size = 1 << ((len(weights) + n - 1).bit_length())
    W = np.fft.rfft(weights, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    corr = np.fft.irfft(W * A_rev, n=size)[: len(weights) + n - 1]

    # g_i = corr[(n‑1) – i]   (see derivation in analysis)
    g = corr[(n - 1) - np.arange(n)]
    return 2.0 * g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    Implementation of Duchi et al. (2008).
    """
    v = v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection (should not happen)
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _exp_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Multiplicative (exponentiated) update that keeps entries positive.
    a ← a * exp(‑η·grad) and normalises to sum 1.
    Small clipping of the gradient avoids overflow.
    """
    # Clip gradient to keep the exponential factor in a reasonable range
    clip_val = 20.0 / max(eta, 1e-12)
    grad = np.clip(grad, -clip_val, clip_val)

    factor = np.exp(-eta * grad)
    a_new = a * factor
    s = a_new.sum()
    if s == 0.0:
        # fallback to uniform if everything collapsed
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / s
