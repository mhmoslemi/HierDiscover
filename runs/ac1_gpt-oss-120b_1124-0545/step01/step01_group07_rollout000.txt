"""
Propose a candidate step‑function sequence that minimizes
`evaluate_sequence`.  

The algorithm has three stages:

1. **Broad parametric sweep** –  for several lengths (512 … 8192) a
   collection of analytically‑defined families (geometric, power‑law,
   sinusoidal, triangular, etc.) as well as a set of random vectors are
   generated.  Each candidate is normalised to a unit L1‑norm (the
   evaluation is homogeneous) and the best one is kept.

2. **Gradient‑based refinement** –  starting from the best candidate, we
   repeatedly reduce the peak of the auto‑convolution.  The peak occurs
   at one index `k`.  The sub‑gradient of `b_k = (a*a)[k]` with respect
   to `a` is simply `g_i = 2 * a_{k-i}` for all valid `i`.  Moving a
   small amount of mass from indices with large gradient to indices
   with small gradient reduces the peak.  After each move the vector is
   re‑normalised to keep the L1‑norm equal to one.

3. **Simulated‑annealing acceptance** –  to escape shallow local minima
   the algorithm occasionally accepts worsening moves with a
   temperature‑controlled probability.  The temperature falls linearly
   from `1.0` to `0.001`.  The whole search respects the supplied time
   budget (default 1000 s) and returns the best sequence found so far.

All helper functions are defined at module level, there are no closures
or lambda expressions, and the code only relies on NumPy (and the
already‑imported `evaluate_sequence`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalise_one(a: np.ndarray) -> np.ndarray:
    """
    Scale the vector to have L1‑norm exactly 1.
    This respects the homogeneity of the evaluation function and also
    guarantees the “sum too close to zero” guard is never triggered.
    """
    s = a.sum()
    if s == 0:
        # uniform tiny mass
        a = np.full_like(a, 1.0 / a.size)
        s = 1.0
    else:
        a = a / s
    # Clip to keep values comfortably below the 1000‑clip limit.
    a = np.clip(a, 0.0, 999.0)
    return a


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear (full) convolution a * a using a power‑of‑2 FFT.
    The result length is 2·len(a)−1.
    """
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, size)
    B = A * A
    conv = np.fft.irfft(B, size)[: 2 * n - 1]
    # Numerical noise may create tiny negative values – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _grad_at_peak(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the peak term (a*a)[k] with respect to a.
    For each valid i, grad[i] = 2 * a[k-i]; for indices that do not
    participate the gradient is zero.
    """
    n = a.size
    grad = np.zeros_like(a)
    # valid i satisfy 0 ≤ i ≤ n‑1 and 0 ≤ k‑i ≤ n‑1
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start > i_end:
        return grad
    idx = np.arange(i_start, i_end + 1, dtype=int)
    partner = k - idx
    grad[idx] = 2.0 * a[partner]
    return grad


def _candidate_shapes(n: int, rng: np.random.Generator):
    """
    Yield a collection of deterministic and random shapes of length `n`.
    Shapes are returned as NumPy arrays; normalisation is performed by
    the caller.
    """
    # deterministic families -------------------------------------------------
    # 1. constant
    yield np.ones(n, dtype=float)

    # 2. linear ramp
    yield np.arange(1, n + 1, dtype=float)          # increasing
    yield np.arange(n, 0, -1, dtype=float)          # decreasing

    # 3. quadratic
    yield np.square(np.arange(1, n + 1, dtype=float))
    yield np.square(np.arange(n, 0, -1, dtype=float))

    # 4. triangular “mountain”
    t = np.arange(1, n + 1, dtype=float)
    yield np.minimum(t, n - t + 1)

    # 5. valley (large at ends, small in centre)
    centre = (n - 1) / 2.0
    yield 1.0 / (1.0 + ((np.arange(n, dtype=float) - centre) ** 2))

    # 6. sine‑squared
    yield np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2

    # 7. exponential decays with random base
    for base in rng.uniform(0.4, 0.98, size=6):
        yield np.power(base, np.arange(n, dtype=float))

    # 8. power‑law decays with random exponent
    for p in rng.uniform(0.2, 3.0, size=6):
        i = np.arange(1, n + 1, dtype=float)
        yield 1.0 / (i ** p)

    # random mixtures --------------------------------------------------------
    for _ in range(10):
        # Random positive vector, then optionally smooth it a bit
        vec = rng.random(n)
        if rng.random() < 0.5:
            # apply a simple moving‑average filter to induce smoothness
            win = rng.integers(2, 8)
            cumsum = np.cumsum(np.insert(vec, 0, 0.0))
            vec = (cumsum[win:] - cumsum[:-win]) / win
            # pad back to original length
            pad = np.full(n - vec.size, vec[-1])
            vec = np.concatenate([vec, pad])
        yield vec

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative coefficient list that minimises the
    `evaluate_sequence` objective.  The function respects the time budget
    and returns the best sequence found (as a plain Python list).
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # tiny safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0. Warm‑start from any previously discovered sequence
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm_arr = _normalise_one(np.asarray(warm, dtype=float))
        score = evaluate_sequence(warm_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = warm_arr.tolist()
            print(f"[{time.time() - start_time:.1f}s] Warm‑start score={score:.6f}")

    # ------------------------------------------------------------------
    # 1. Broad parametric sweep over several lengths
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192]
    for n in lengths:
        if time.time() >= deadline:
            break
        for shape in _candidate_shapes(n, rng):
            if time.time() >= deadline:
                break
            seq = _normalise_one(shape.copy())
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"[{time.time() - start_time:.1f}s] New best (len={n}) score={score:.6f}")
            # early exit if we already beat the target
            if best_score <= 1.503:
                break

    # If the sweep never produced a candidate (very unlikely) fall back
    # to a random vector of moderate length.
    if best_seq is None:
        n = 1024
        seq = _normalise_one(rng.random(n))
        best_seq = seq.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time() - start_time:.1f}s] Fallback random start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Gradient‑based refinement (sub‑gradient descent + SA)
    # ------------------------------------------------------------------
    cur = np.asarray(best_seq, dtype=float)

    # Temperature schedule (linear annealing)
    T_start, T_end = 1.0, 0.001
    # Step‑size schedule (linear decay)
    step_start, step_end = 0.12, 0.001   # fraction of mass moved per iteration

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = T_start * (1 - elapsed / total) + T_end * (elapsed / total)
        step_frac = step_start * (1 - elapsed / total) + step_end * (elapsed / total)

        # ----- compute convolution and locate the peak -----
        conv = _full_convolution_fft(cur)
        k_max = int(np.argmax(conv))

        # ----- sub‑gradient of the peak term -----
        grad = _grad_at_peak(cur, k_max)

        # ----- choose source (high gradient) and target (low gradient) indices -----
        # source = index with maximal gradient
        src = int(np.argmax(grad))
        # target = random index among the lowest 15 % of gradient values
        low_thresh = np.percentile(grad, 15)
        low_cands = np.where(grad <= low_thresh)[0]
        if low_cands.size == 0:
            tgt = rng.integers(0, cur.size)
        else:
            tgt = int(rng.choice(low_cands))

        # ----- perform a mass transfer -----
        epsilon = cur[src] * step_frac
        if epsilon < 1e-14:   # source essentially empty – fall back to a tiny random tweak
            new = cur + rng.normal(scale=cur.mean() * 0.02, size=cur.size)
        else:
            new = cur.copy()
            new[src] -= epsilon
            new[tgt] += epsilon

        # Clip to stay non‑negative and renormalise L1‑norm to 1
        np.maximum(new, 0.0, out=new)
        new_sum = new.sum()
        if new_sum == 0:
            continue
        new = new / new_sum

        # ----- evaluation -----
        cand_score = evaluate_sequence(new.tolist())

        # ----- acceptance (SA) -----
        improved = cand_score < best_score
        if improved:
            best_score = cand_score
            best_seq = new.tolist()
            cur = new
            print(f"[{time.time() - start_time:.1f}s] Improved score={best_score:.6f}")
            if best_score <= 1.503:
                print("Target reached – stopping early.")
                break
        else:
            # accept with probability exp(-(Δ)/T)
            prob = np.exp(-(cand_score - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur = new

        # occasional random shake‑up to avoid stagnation (once every ~200 iters)
        if rng.integers(0, 200) == 0:
            noise = rng.normal(scale=cur.mean() * 0.01, size=cur.size)
            cur = np.clip(cur + noise, 0.0, None)
            cur /= cur.sum()

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    return best_seq
