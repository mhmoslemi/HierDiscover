"""
Sub‑gradient descent with occasional random perturbations.

The objective 2·n·max(conv(a,a))/(∑a)² is scale‑invariant, therefore we keep
the vector on the probability simplex (∑a = 1).  At each iteration we

  1. compute the autocorrelation b = a * a (FFT‑based O(n log n)),
  2. locate the (few) indices where b attains its maximum,
  3. compute a sub‑gradient of  max(b)  w.r.t. a,
  4. take a small step in the projected sub‑gradient direction
     (the projection onto the simplex is just a normalisation after clipping),
  5. with a small probability perform a random “mass‑transfer” move to
     escape flat plateaus (otherwise the sub‑gradient would be zero for a
     perfectly uniform distribution).

A simple restart heuristic re‑initialises the search when no improvement
has been seen for a while.  The code works for any length *n* (default 6000)
and finishes within the 1000 s budget, returning the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution via FFT of length ``size`` (must be ≥ len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    w = np.fft.irfft(fu * fv, n=size)
    return w[: len(u) + len(v) - 1]


def _max_convolution(a: np.ndarray, size1: int) -> np.ndarray:
    """Return a * a (full convolution) using the pre‑computed FFT size."""
    return _conv_fft(a, a, size1)


def _subgradient_step(
    a: np.ndarray,
    conv_ab: np.ndarray,
    mask: np.ndarray,
    size2: int,
    step: float,
) -> np.ndarray:
    """
    Perform a projected sub‑gradient step for the function max(conv(a,a)).
    ``conv_ab`` is a*a, ``mask`` marks the indices attaining (or almost attaining)
    the maximum, and ``size2`` is the FFT length needed for the
    convolution a_rev * mask.
    """
    n = a.shape[0]
    # reversed a
    a_rev = a[::-1]
    # convolution a_rev * mask  → length 3n‑2
    full = _conv_fft(a_rev, mask, size2)
    # extract the central n entries (the true gradient components)
    g = full[n - 1 : n - 1 + n]
    # sub‑gradient of the max entry
    d = -2.0 * g
    # project onto the hyperplane Σd = 0 (simple translation)
    d -= d.mean()
    # take the step and re‑project onto the simplex
    a_new = a + step * d
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        # fallback to uniform if everything vanished
        a_new = np.ones_like(a) / n
    else:
        a_new /= s
    return a_new


def _random_transfer(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a small amount of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    if max_transfer == 0.0:
        return a  # nothing to move
    delta = scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    The algorithm uses a projected sub‑gradient descent on the
    max‑autocorrelation functional and occasional random mass‑transfer moves.
    The best sequence (list of Python floats) found is returned.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a tiny safety margin
    deadline = start + budget_s - 0.5

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    # default length, can be overridden through kwargs
    n = int(kwargs.get("n", 6000))

    # --------------------------------------------------------------
    # initialise a starting distribution
    # --------------------------------------------------------------
    # If a previous construction is available, upscale it; otherwise use random.
    try:
        # ``height_sequence_1`` may be provided by the surrounding harness.
        # Upsample it linearly to the required length.
        seq0 = globals().get("height_sequence_1", None)
        if seq0 is not None:
            seq0 = np.asarray(seq0, dtype=float)
            if seq0.size != n:
                # linear interpolation to the new grid
                x_old = np.linspace(0, 1, seq0.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                a = np.interp(x_new, x_old, seq0)
                a = np.clip(a, 0.0, None)
                if a.sum() == 0.0:
                    a = np.random.rand(n)
                a /= a.sum()
            else:
                a = np.clip(seq0, 0.0, None)
                a /= a.sum()
        else:
            raise KeyError
    except Exception:
        a = np.random.rand(n)
        a /= a.sum()

    # --------------------------------------------------------------
    # pre‑compute FFT lengths (powers of two) – they stay constant
    # --------------------------------------------------------------
    size1 = _next_pow2(2 * n - 1)       # for a * a
    size2 = _next_pow2(3 * n - 2)       # for a_rev * mask

    # --------------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------------
    # initial score (objective = 2·n·max_b, because sum(a)=1)
    conv_full = _max_convolution(a, size1)
    max_b = conv_full.max()
    best_score = 2.0 * n * max_b
    best_seq = a.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # algorithmic parameters – tuned empirically
    init_step = 0.5                     # base step size for sub‑gradient
    rand_prob = 0.12                    # probability of a random transfer move
    rand_scale = 0.05                   # relative size of random transfer
    tolerance_factor = 1e-6             # for selecting “active” convolution peaks
    stagnation_restart = 45.0           # seconds without improvement → restart
    last_improve = start
    it = 0

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ----- 1. autocorrelation and current score -----
        conv_full = _max_convolution(a, size1)
        max_b = conv_full.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start
            print(f"[{it:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # ----- 2. decide which move to apply -----
        if np.random.rand() < rand_prob:
            # a cheap random mass transfer (exploration)
            a = _random_transfer(a, rand_scale)
        else:
            # ----- 3. sub‑gradient step aimed at reducing the peak -----
            # build mask of the (almost) maximal convolution entries
            tol = max(tolerance_factor * max_b, 1e-12)
            mask = (conv_full >= max_b - tol).astype(float)  # shape (2n‑1,)
            # adaptive step size (1/√iteration) – guarantees convergence for sub‑gradients
            step = init_step / np.sqrt(it + 1.0)
            a = _subgradient_step(a, conv_full, mask, size2, step)

        # ----- 4. occasional restart if we are stuck -----
        if (it % 30000 == 0) and (time.time() - last_improve > stagnation_restart):
            a = np.random.rand(n)
            a /= a.sum()
            print(f"[{it:7d}] restart due to stagnation")

        it += 1

    # --------------------------------------------------------------
    # final reporting
    # --------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iterations={it}, best score={best_score:.6f}, time={total:.1f}s")
    return best_seq.tolist()
