from scipy.linalg import toeplitz

def conv_matrix(f):
    n = len(f)
    c = np.concatenate([f, np.zeros(n-1, dtype=f.dtype)])
    r = np.concatenate([f[:1], np.zeros(n-1, dtype=f.dtype)])
    return toeplitz(c, r)
```

def solve_lp_for_g(f, M):
    n = len(f)
    A_ub = conv_matrix(f)  # shape (2n-1, n)
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)   # maximize sum(b) => minimize -sum(b)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    # normalize to sum 1
    g_sum = g.sum()
    if g_sum > 0:
        g = g / g_sum
    else:
        g = np.full_like(g, 1.0/n)
    return g
```

from scipy.linalg import toeplitz

def conv_matrix(f):
    n = len(f)
    c = np.concatenate([f, np.zeros(n-1, dtype=f.dtype)])
    r = np.concatenate([f[:1], np.zeros(n-1, dtype=f.dtype)])
    A = toeplitz(c, r)
    return A

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Determine sequence length
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2500  # fallback

    # initialize arcsine distribution
    i = np.arange(n, dtype=np.float64)
    # compute midpoints x = (i + 0.5)/(n)
    x = (i + 0.5) / n
    # arcsine density (Beta(0.5,0.5))
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.nan_to_num(a)   # handle potential zero at edges (should not be zero)
    a /= a.sum()
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] arcsine start score = {best_score:.6f}")

    # FFT size for convolutions
    fft_len = 1 << ((2*n - 1).bit_length())

    # Subgradient descent phase
    step = 0.3
    it = 0
    while time.time() < deadline - 2.0 and step > 1e-7:
        a = gradient_step(a, step, fft_len)
        if it % 5 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[grad] iter {it:5d} score {best_score:.6f}")
        step *= 0.9996
        it += 1

    # Alternating LP + mixing phase
    lp_iter = 0
    t = 0.15   # mixing weight
    while time.time() < deadline - 1.0:
        # compute current bound M
        conv_vals = np.convolve(a, a)
        M = conv_vals.max()
        # solve LP for g
        A_ub = conv_matrix(a)
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)   # maximize sum(g)
        bounds = [(0, None)] * n
        try:
            res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                         bounds=bounds, method='highs',
                                         options={'presolve': True, 'time_limit': 0.5})
        except Exception as e:
            print(f"[lp] iteration {lp_iter}: linprog failed: {e}")
            break
        if not res.success:
            print(f"[lp] iteration {lp_iter}: linprog not success: {res.message}")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g > 0:
            g = g / sum_g
        else:
            g = np.full_like(g, 1.0/n)

        # mix
        a = (1 - t) * a + t * g
        # renormalize (should already be approx 1)
        a /= a.sum()

        # evaluate
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[lp] iter {lp_iter:3d} improved score {best_score:.6f}")

        # decay mixing weight
        t = max(t * 0.99, 0.02)
        lp_iter += 1

    # Final local stochastic search
    eps = 0.08
    temperature = 0.05
    it = 0
    while time.time() < deadline:
        cand = random_perturb(a, eps)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand
            print(f"[pert] iter {it:5d} improved score {best_score:.6f}")
        else:
            # Accept uphill with probability
            prob = np.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                a = cand
        eps = max(eps * 0.9995, 1e-4)
        temperature *= 0.9995
        it += 1

    final_seq = np.clip(best_seq, 0.0, 1000.0)
    return final_seq.tolist()

"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

using a three‑stage optimisation:

1.  Initialise with the Catalan (almost flat) coefficients.
2.  Perform a few projected sub‑gradient steps on the supremum
    of the self‑convolution (fast FFT based).
3.  Repeatedly solve a linear programme

        maximise   Σ b_j
        subject to  (a * b)[k] ≤ M   for all k
                    b_j ≥ 0

   where *M* = max(conv(a,a)).  The LP finds a distribution *g*
   that puts as much mass as possible on indices where *a* is small.
   Mixing a←(1‑t)a + t g (with a slowly decreasing t) reduces the peak
   of the self‑convolution.  The process stops when the time budget is
   exhausted.

4.  A short stochastic local‑search (random mass transfer) finishes the
   search and occasionally accepts uphill moves.

The best sequence seen during the run is returned.
"""

import time
import math
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def compute_catalan_sequence(n: int) -> np.ndarray:
    """Return the first n Catalan‑type coefficients normalised to sum‑to‑1.

    c_i = (2i choose i) / 4**i  for i = 0 … n‑1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def max_convolution_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Maximum of the linear self‑convolution a*a, using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    conv = conv[: 2 * len(a) - 1]
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def gradient_step(a: np.ndarray, step: float, fft_len: int) -> np.ndarray:
    """Projected sub‑gradient step for max(conv(a,a))."""
    _, k = max_convolution_fft(a, fft_len)

    n = len(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    idx = np.arange(start, end + 1, dtype=np.int64)

    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[k - idx]            # sub‑gradient of the active term

    a_new = a - step * grad
    a_new = np.maximum(a_new, 0.0)          # projection onto ℝ₊
    s = a_new.sum()
    if s == 0.0:                            # safeguard against collapse
        a_new[:] = 1.0 / n
    else:
        a_new /= s
    return a_new


def random_perturb(a: np.ndarray, eps: float) -> np.ndarray:
    """Transfer a fraction *eps* of the mass from a random index i to j."""
    n = len(a)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


def conv_matrix_dense(f: np.ndarray) -> np.ndarray:
    """
    Build the dense (2·n‑1) × n matrix C with
        C[k, j] = f[k‑j]   if 0 ≤ k‑j < n, else 0.
    This matrix implements the linear map b ↦ f * b.
    """
    n = len(f)
    # First column: f followed by (n‑1) zeros
    c = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    # First row: f[0] followed by (n‑1) zeros
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    return scipy.linalg.toeplitz(c, r)


def solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP
        maximise Σ b_j
        subject to (f * b)[k] ≤ M  for all k,
                   b_j ≥ 0 .
    Returns the optimal b (normalised to sum‑to‑1) or None on failure.
    """
    n = len(f)
    A = conv_matrix_dense(f)                # shape (2n‑1, n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ b_j  →  minimise –Σ b_j
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # use HiGHS – it handles dense matrices efficiently
    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        g = res.x
        total = g.sum()
        if total > 0.0:
            g /= total
        else:
            g[:] = 1.0 / n
        return g
    return None

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Produce a non‑negative step‑function (list of heights) that tries to
    minimise the supplied ``evaluate_sequence`` metric.

    The method combines:
      * a Catalan‑type start,
      * projected sub‑gradient descent on the supremum of the
        self‑convolution,
      * an alternating LP / mixing phase that pushes mass onto
        low‑weight indices,
      * a final stochastic local‑search phase.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # reuse length of the previous best
    except NameError:
        n = 2400                            # reasonable fallback

    # ------------------------------------------------------------------
    # 1. Start from the Catalan (almost‑flat) sequence
    # ------------------------------------------------------------------
    a = compute_catalan_sequence(n)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] Catalan start score = {best_score:.6f}")

    # Fast FFT length for convolutions
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient descent on max(conv)
    # ------------------------------------------------------------------
    step = 0.25
    it = 0
    while time.time() < deadline - 3.0 and step > 1e-7:
        a = gradient_step(a, step, fft_len)
        if it % 7 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[grad] iter {it:5d}  score {best_score:.6f}")
        step *= 0.9996
        it += 1

    # ------------------------------------------------------------------
    # 3. Alternating LP + mixing phase
    # ------------------------------------------------------------------
    mixing = 0.15               # initial mixing weight
    lp_iter = 0
    while time.time() < deadline - 2.0:
        # current peak of the self‑convolution
        conv_self = np.convolve(a, a)
        M = conv_self.max()

        # solve LP to obtain a partner distribution g
        g = solve_lp_for_g(a, M)
        if g is None:
            print("[lp] LP failed – exiting LP phase")
            break

        # convex combination; after mixing we renormalise to sum‑to‑1
        a = (1.0 - mixing) * a + mixing * g
        a /= a.sum()

        # evaluate
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[lp] iter {lp_iter:3d}  improved score {best_score:.6f}")

        # slowly decay the mixing factor (allows finer adjustments later)
        mixing = max(mixing * 0.99, 0.02)
        lp_iter += 1

    # ------------------------------------------------------------------
    # 4. Random perturbation / simulated‑annealing finish
    # ------------------------------------------------------------------
    eps = 0.09
    temperature = 0.07
    it = 0
    while time.time() < deadline:
        cand = random_perturb(a, eps)
        cand_score = evaluate_sequence(cand.tolist())

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            a = cand                      # move to the better point
            print(f"[pert] iter {it:5d}  improved score {best_score:.6f}")
        else:
            # uphill acceptance with exponentially decreasing temperature
            prob = math.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                a = cand

        # decay move size and temperature
        eps = max(eps * 0.9997, 1e-4)
        temperature *= 0.9995
        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    final_seq = np.clip(best_seq, 0.0, 1000.0)   # safety clamp
    print(f"[final] best score = {best_score:.6f}  length = {len(final_seq)}")
    return final_seq.tolist()
