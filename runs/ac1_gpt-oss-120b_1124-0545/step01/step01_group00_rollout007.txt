```
def project_simplex(v):
    """Projects onto the probability simplex."""
    # ensure v is numpy array
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()   # next power of two

def conv_fft(x, y, fft_size=None):
    n = x.shape[0] + y.shape[0] - 1
    size = fft_size if fft_size is not None else 1 << (n-1).bit_length()
    X = np.fft.rfft(x, size)

    conv = np.fft.irfft(X*Y, size)[:n]
    return conv

def project_simplex(v):
    """Project onto probability simplex (sum=1, v>=0)."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

        v = beta2 * v + (1 - beta2) * (grad ** 2)
        t = iteration + 1
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional random perturbation

            # small jitter on a_new
            jitter = 0.001 * np.random.randn(n)
            a_new += jitter

        # project onto simplex
        a_new = project_simplex(a_new)

        # evaluate
        cand_score = compute_score(a_new)

        if cand_score < best_score - 1e-9:
            best_score = cand_score
            best_a = a_new.copy()
            last_improve = time.time()
            print(f"[{iteration:7d}] new best {best_score:.6f} time={elapsed:.1f}s beta={beta:.2f}")

        a = a_new

        # stagnation restart

            # restart from best with noise
            a = best_a + 0.01 * np.random.rand(n)
            a = project_simplex(a)
            m[:] = 0
            v[:] = 0
            last_improve = time.time()
            print(f"[{iteration:7d}] restart due to stagnation")

        iteration += 1

    total_time = time.time() - start
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_a.tolist()

"""
Gradient‑descent with a soft‑max approximation of the peak of the
self‑convolution, projected onto the probability simplex.
A simple Adam optimiser drives the sequence towards lower
max(conv(a,a)); the temperature (soft‑max β) and learning rate are
annealed.  Occasionally a tiny random jitter is added and, after a
stagnation period, the algorithm restarts from the current best with
added noise.  The best sequence found before the time budget expires
is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution (FFT‑based, reusable size)
# ----------------------------------------------------------------------


def _conv_fft(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """Return the linear convolution of two real vectors using FFT."""
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    conv = np.fft.irfft(X * Y, size)[: x.size + y.size - 1]
    return conv


def _prepare_fft_size(n: int) -> int:
    """Return the next power of two ≥ 2·n‑1 (suitable for FFT)."""
    conv_len = 2 * n - 1
    return 1 << (conv_len - 1).bit_length()


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (sum=1, v≥0)."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1  # last true index
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Score (exact evaluation, identical to evaluate_sequence for sum=1)
# ----------------------------------------------------------------------


def _score(a: np.ndarray, fft_size: int) -> float:
    """Exact objective 2·n·max(conv(a,a))/ (sum(a)^2).  Assumes a≥0."""
    n = a.shape[0]
    conv = _conv_fft(a, a, fft_size)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Random move (tiny mass transfer) – used for occasional jitter /
# stagnation restart.
# ----------------------------------------------------------------------


def random_move(a: np.ndarray, scale: float) -> np.ndarray:
    """Transfer a random amount (≤scale·mass) from one index to another."""
    n = a.size
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    delta = scale * max_transfer * np.random.rand()
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    # re‑normalise (projecting removes rounding errors)
    return project_simplex(a2)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    n: int = 2000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    evaluate_sequence().  A soft‑max smoothed objective is minimised
    by Adam‑style projected gradient descent, with annealed smoothing
    (β) and learning rate.  Random jitter and occasional restarts help
    escape plateaus.  The best sequence discovered within the time budget
    is returned.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a tiny safety margin
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # initial sequence (use provided global if it matches the length)
    # ------------------------------------------------------------------
    try:
        init = height_sequence_1  # type: ignore[name-defined]
        if isinstance(init, (list, tuple, np.ndarray)) and len(init) == n:
            a = np.asarray(init, dtype=float)
            a = project_simplex(a)  # ensure sum=1, non‑negative
        else:
            raise Exception()
    except Exception:
        a = np.random.rand(n)
        a = project_simplex(a)

    best_a = a.copy()
    fft_size = _prepare_fft_size(n)
    best_score = _score(best_a, fft_size)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros(n, dtype=float)
    v = np.zeros(n, dtype=float)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # schedule parameters
    beta_start, beta_end = 0.5, 15.0          # soft‑max temperature
    lr_start, lr_end = 0.20, 0.005           # learning rate (Adam step size)
    jitter_prob = 0.03                       # probability of a tiny random jitter
    jitter_scale = 0.001                     # magnitude of the jitter

    iteration = 0
    last_improve = start

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start
        # progress in [0,1] – we leave a small margin for the final seconds
        progress = min(elapsed / (budget_s - 10.0), 1.0)

        # annealed parameters
        beta = beta_start + (beta_end - beta_start) * progress
        lr = lr_start * (1.0 - progress) + lr_end * progress

        # ----- compute current convolution and soft‑max weights -----
        conv_aa = _conv_fft(a, a, fft_size)
        max_conv = conv_aa.max()
        # soft‑max (stable version)
        shifted = conv_aa - max_conv
        exp_weights = np.exp(beta * shifted)
        weight_sum = exp_weights.sum()
        p = exp_weights / weight_sum  # shape = 2n‑1

        # ∂M/∂a_i = 2 * Σ_j p_j * a_{j-i}
        grad_conv = _conv_fft(p, a[::-1], fft_size)        # length = 3n‑2

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        t = iteration + 1

        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # occasional tiny random jitter to improve exploration
        if np.random.rand() < jitter_prob:
            a_new += jitter_scale * np.random.randn(n)

        a_new = project_simplex(a_new)

        # ----- evaluate the true objective for the new point -----
        cand_score = _score(a_new, fft_size)

        # keep the best ever seen
        if cand_score + 1e-12 < best_score:
            best_score = cand_score
            best_a = a_new.copy()
            last_improve = time.time()
            print(
                f"[{iteration:7d}] new best {best_score:.6f} "
                f"time={elapsed:.1f}s beta={beta:.2f}"
            )

        a = a_new
        iteration += 1

        # ----- stagnation handling: restart from best + noise -----
        if time.time() - last_improve > 40.0:
            # add a small random perturbation to the best solution and continue
            a = best_a + 0.02 * np.random.rand(n)
            a = project_simplex(a)
            # reset Adam's moments (otherwise they would carry stale information)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[{iteration:7d}] restart after stagnation")

    total_time = time.time() - start
    print(
        f"[done] iterations={iteration}, best score={best_score:.6f}, "
        f"time={total_time:.1f}s"
    )
    return best_a.tolist()
