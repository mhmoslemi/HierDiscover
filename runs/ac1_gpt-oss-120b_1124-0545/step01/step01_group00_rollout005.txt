def project_simplex(v, s=1.0):
    # v: array
    if v.sum() == s and np.all(v >= 0):
        return v.copy()
    n = len(v)
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find rho = max {j | u_j - (cssv_j)/j > 0}
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def pair_move(a, step_scale):
    n = len(a)
    max_val, conv = conv_max(a)
    k = int(np.argmax(conv))
    # compute contributions w_i = a[i] * a[k - i]
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    w = a[idx] * a[k - idx]
    if w.sum() == 0:
        # fallback to random move
        return random_move(a, step_scale)
    probs = w / w.sum()
    i = np.random.choice(idx, p=probs)

    # choose destination d != i (and !=j)

"""
Improved search for a non‑negative step function minimizing
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm mixes several specialised moves:

* **random_move** – simple transfer of a tiny amount of mass.
* **targeted_move** – moves mass away from indices that contribute most to
  the current maximal convolution entry.
* **pair_move** – removes mass simultaneously from a pair (i, k‑i) that
  creates the peak and deposits it elsewhere.
* **gradient_step** – sub‑gradient descent on the active convolution index
  with an exact Euclidean projection onto the probability simplex
  (∑a = 1, a ≥ 0).

A short simulated‑annealing schedule controls a Metropolis acceptance
criterion and continuously shrinks the step size.  The search starts from
a small library of parametric shapes (sine‑powers, cosine‑powers,
exponential decays, triangular windows, etc.) together with the
previous best sequence (if the global ``height_sequence_1`` variable is
available).  The best of these seeds is used as the initial point.
Multiple restarts are triggered after periods of stagnation.

The implementation uses an FFT‑based convolution to obtain the maximal
auto‑convolution value efficiently (O(n log n) per evaluation) and a
fast O(n) projection onto the simplex.  The routine respects the 1000 s
budget and returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def conv_max(a: np.ndarray):
    """
    Compute the full convolution a*a using FFT and return its maximum value
    together with the full convolution array.
    """
    n = a.shape[0]
    L = 2 * n                     # zero‑pad length for a linear convolution
    # real FFT is enough because a is real
    A = np.fft.rfft(a, n=L)
    B = A * A                     # element‑wise product (â)^2
    conv = np.fft.irfft(B, n=L)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values
    conv = np.maximum(conv, 0.0)
    return conv.max(), conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective value for a (assumes a ≥ 0)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    max_b, _ = conv_max(a)
    return float(2.0 * len(a) * max_b / (s * s))


def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x | x ≥ 0, ∑x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # Find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are projected to zero (should not happen for s>0)
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Move primitives
# ----------------------------------------------------------------------


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """Transfer a tiny random amount of mass from i to j."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # Projection guarantees the simplex constraint
    a_new = project_simplex(a_new)
    return a_new


def targeted_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Move mass away from an index that contributes most to the current
    maximal convolution entry.
    """
    n = a.shape[0]
    max_val, conv = conv_max(a)
    k_max = int(np.argmax(conv))

    # contribution w_i = a_i * a_{k_max-i}
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1)
    w = a[idx] * a[k_max - idx]
    if w.sum() == 0.0:
        i = np.random.randint(n)
    else:
        probs = w / w.sum()
        i = np.random.choice(idx, p=probs)

    # pick a distinct destination j
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1

    max_transfer = a[i]
    if max_transfer == 0.0:
        return a.copy()
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = project_simplex(a_new)
    return a_new


def pair_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Remove mass from the pair (i, k−i) that creates the current peak and
    relocate it to a distant index.  This directly attacks the product term
    a_i·a_{k−i} that dominates the max convolution entry.
    """
    n = a.shape[0]
    max_val, conv = conv_max(a)
    k = int(np.argmax(conv))

    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    w = a[idx] * a[k - idx]
    if w.sum() == 0.0:
        return random_move(a, step_scale)

    probs = w / w.sum()
    i = np.random.choice(idx, p=probs)
    j = k - i
    # destination index (avoid i and j)
    d = np.random.randint(n)
    while d == i or d == j:
        d = np.random.randint(n)

    # transfer a fraction of the mass from both i and j
    delta_i = step_scale * a[i] * np.random.rand()
    delta_j = step_scale * a[j] * np.random.rand() if j != i else 0.0

    a_new = a.copy()
    a_new[i] -= delta_i
    a_new[j] -= delta_j
    a_new[d] += delta_i + delta_j
    a_new = np.maximum(a_new, 0.0)
    a_new = project_simplex(a_new)
    return a_new


def gradient_step(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Sub‑gradient step on the active convolution index with simplex projection.
    """
    n = a.shape[0]
    max_val, conv = conv_max(a)
    k = int(np.argmax(conv))

    # sub‑gradient of conv_k wrt a is g_i = 2·a_{k-i} for valid i
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]

    # move against the gradient
    a_new = a - step_scale * g
    a_new = np.maximum(a_new, 0.0)
    a_new = project_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
# Initial shape library
# ----------------------------------------------------------------------


def generate_initial_shapes(n: int):
    """Return a list of candidate start vectors constructed from simple analytic families."""
    shapes = []

    # Uniform
    shapes.append(np.ones(n) / n)

    # Sine power family
    x = np.arange(n)
    for p in np.linspace(0.5, 4.0, 7):
        a = np.sin(np.pi * (x + 0.5) / (n + 1)) ** p
        a = np.clip(a, 0.0, None)
        if a.sum() > 0:
            a /= a.sum()
            shapes.append(a)

    # Cosine power family (raised cosine)
    for p in np.linspace(0.5, 4.0, 7):
        a = (1 - np.cos(2 * np.pi * x / (n - 1))) ** p
        a = np.clip(a, 0.0, None)
        if a.sum() > 0:
            a /= a.sum()
            shapes.append(a)

    # Exponential decay from the edges (large at the ends, small in middle)
    d = np.minimum(x, n - 1 - x)               # distance to nearest edge
    for c in np.linspace(0.001, 0.015, 5):
        a = np.exp(-c * d)
        a = np.clip(a, 0.0, None)
        a /= a.sum()
        shapes.append(a)

    # Triangular windows of varying support width
    for w in np.linspace(0.3, 0.9, 5):
        support = max(2, int(w * n))
        a = np.zeros(n)
        start = (n - support) // 2
        a[start:start + support] = 1.0
        a /= a.sum()
        shapes.append(a)

    # Raised‑cosine windows (smooth triangular shape)
    for w in [0.4, 0.6, 0.8]:
        support = max(2, int(w * n))
        a = np.zeros(n)
        start = (n - support) // 2
        for i in range(support):
            a[start + i] = 0.5 * (1 - np.cos(2 * np.pi * i / (support - 1)))
        a /= a.sum()
        shapes.append(a)

    # Include the previous best if available
    try:
        h = np.asarray(height_sequence_1, dtype=float)
        if h.shape[0] != n:
            if h.shape[0] > n:
                h = h[:n]
            else:
                h = np.pad(h, (0, n - h.shape[0]), mode='constant')
        h = np.clip(h, 0.0, None)
        if h.sum() > 0:
            h /= h.sum()
            shapes.append(h)
    except NameError:
        pass

    return shapes


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a good height sequence within ``budget_s`` seconds.
    Returns the best sequence found as a plain list of Python floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # Initialise from a small library of analytic shapes
    # ------------------------------------------------------------------
    candidates = generate_initial_shapes(n)

    best_seq = None
    best_score = np.inf
    for cand in candidates:
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    # fall back to a random Dirichlet if something went wrong
    if best_seq is None:
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = compute_score(best_seq)

    print(f"[init] n={n}, initial best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing loop with multiple move types
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score

    # annealing schedule parameters
    init_temp = 1.0
    final_temp = 1e-3
    init_step = 0.25
    final_step = 5e-4

    iteration = 0
    last_improve = start_time

    while time.time() < deadline:
        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 10.0)          # in [0, 1]

        # (slowly) decreasing temperature and step size
        temp = init_temp * (1.0 - progress) + final_temp * progress
        step_scale = init_step * (1.0 - progress) + final_step * progress

        # choose which move to apply
        r = np.random.rand()
        if r < 0.35:
            cand = targeted_move(a, step_scale)
        elif r < 0.65:
            cand = random_move(a, step_scale)
        elif r < 0.85:
            cand = pair_move(a, step_scale)
        else:
            cand = gradient_step(a, step_scale * 0.5)

        cand_score = compute_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < np.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:7d}] new best = {best_score:.6f}  time={elapsed:.1f}s")

        # restart if stagnation > 45 s and we have spent at least 200 s
        if iteration % 4000 == 0 and (time.time() - last_improve) > 45.0 and elapsed > 200.0:
            a = np.random.rand(n)
            a /= a.sum()
            cur_score = compute_score(a)
            print(f"[{iteration:7d}] restart due to stagnation (elapsed {elapsed:.1f}s)")

        iteration += 1

    total = time.time() - start_time
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total:.1f}s")
    # Return plain Python list
    return best_seq.tolist()
```

This implementation follows the outlined strategy, starts from a diverse set of analytically‑motivated seeds (including the previously‑found best sequence), uses a richer repertoire of moves, projects every update onto the probability simplex, and employs a temperature‑controlled Metropolis rule with occasional restarts.  It stays within the 1000 s budget and returns the best sequence discovered.
