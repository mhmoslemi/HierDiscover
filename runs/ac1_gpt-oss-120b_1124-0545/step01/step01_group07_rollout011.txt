def project_simplex(v):
    # ensures sum(v)=1 and v >=0
    # Standard algorithm: sort v descending, find rho, compute theta.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate – fast parametric sweep + projected sub‑gradient descent.

The evaluation metric is homogeneous, therefore we may keep the
coefficients on the simplex (non‑negative, sum = 1).  The algorithm
consists of two stages.

Stage 1 – systematic search over simple parametric families
    * exponential decays   a_i = r^i          (r ∈ (0.5,0.9999))
    * reversed exponentials a_i = r^{n‑1‑i}
    * power‑law decays     a_i = 1/(i+1)^p    (p ∈ (0.1,4))
    * reversed power‑law
    * linear / quadratic / triangular / sinusoidal shapes
All candidates are evaluated with a fast FFT‑based convolution
(`conv_max_and_index`).  The best shape found is kept.

Stage 2 – refinement by projected sub‑gradient descent on the
          objective  2·n·max(a*a) / (sum a)^2 .
The gradient of the active convolution term (the one attaining the
maximum) is simply 2·shift_right(k, a).  After each gradient step the
vector is projected onto the probability simplex to preserve feasibility.
A simple back‑tracking line search guarantees monotone improvement, and
random perturbations are occasionally injected to escape flat regions.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best sequence found so far.  All numbers are non‑negative;
the scaling is irrelevant for the final score because the evaluation
function is homogeneous.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def conv_max_and_index(a: np.ndarray):
    """
    Returns (max_convolution_value, index_of_max) of the linear
    convolution a * a (full mode).  Uses FFT for O(n log n) speed.
    """
    n = a.shape[0]
    # smallest power of two >= 2·n‑1
    L = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def gradient_shift(a: np.ndarray, k: int):
    """
    Sub‑gradient of the active term b_k = Σ_i a_i·a_{k‑i}
    (the term attaining the maximum).  For non‑negative a the
    gradient coincides with 2·shift_right(k, a).

    Returns a vector of length len(a).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)                     # inclusive
    # indices i where 0 ≤ k‑i < n
    i = np.arange(i_start, i_end + 1, dtype=int)
    g[i] = 2.0 * a[k - i]
    return g


def simplex_projection(v: np.ndarray):
    """
    Projects vector v onto the probability simplex {x ≥ 0, Σx = 1}.
    Implementation of the algorithm by Wang & Carreira‑Perpiñán.
    """
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray):
    """
    Homogeneous objective: 2·n·max(a*a) / (sum a)^2.
    a is assumed non‑negative.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    max_b, _ = conv_max_and_index(a)
    return 2.0 * a.shape[0] * max_b / (s * s)

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):

    Search for a non‑negative coefficient list that minimizes
    `evaluate_sequence`.  The function respects the time budget; after the
    budget expires it returns the best sequence found so far.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1           # safety margin
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Warm‑start from the previously discovered construction, if any
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:

        seq = np.maximum(seq, 0.0)                  # just in case
        sc = evaluate_sequence(seq.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = seq.tolist()
            print(f"[{time.time() - start_time:.1f}s] Warm‑start from height_sequence_1 "
                  f"score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – systematic exploration of parametric families
    # --------------------------------------------------------------
    # target length – the previous best used 8192, which is a good trade‑off
    n = 8192

    def consider(arr: np.ndarray):
        """evaluate a candidate and possibly update the current best."""
        nonlocal best_seq, best_score
        if time.time() >= deadline:
            return
        arr = np.maximum(arr, 0.0)                   # enforce non‑negativity
        # fast homogeneous score; scaling does not matter
        sc = fast_score(arr)
        if sc < best_score:
            best_score = sc
            best_seq = arr.tolist()
            print(f"[{time.time() - start_time:.1f}s] New best (len={n}) score={best_score:.6f}")

    # Exponential decays (both directions)
    for r in np.linspace(0.5, 0.9999, 300):
        a = np.power(r, np.arange(n, dtype=float))
        consider(a)
        a_rev = np.power(r, np.arange(n - 1, -1, -1, dtype=float))
        consider(a_rev)

    # Power‑law decays (both directions)
    idx = np.arange(1, n + 1, dtype=float)
    for p in np.linspace(0.1, 4.0, 200):
        a = 1.0 / (idx ** p)
        consider(a)
        a_rev = a[::-1]
        consider(a_rev)

    # Linear, quadratic, triangular, sinusoidal (already used before)
    consider(np.arange(n, 0, -1, dtype=float))                     # decreasing linear
    consider(np.arange(1, n + 1, dtype=float))                     # increasing linear
    consider(np.square(np.arange(n, 0, -1, dtype=float)))          # decreasing quadratic
    consider(np.square(np.arange(1, n + 1, dtype=float)))          # increasing quadratic
    t = np.arange(1, n + 1, dtype=float)
    consider(np.minimum(t, n - t + 1))                             # mountain
    consider(np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2)  # sine‑squared

    # If no candidate has been accepted (unlikely), fall back to random
    if best_seq is None:
        cur = rng.random(n)
        best_seq = cur.tolist()
        best_score = fast_score(cur)
        print(f"[{time.time() - start_time:.1f}s] Random fallback score={best_score:.6f}")

    # --------------------------------------------------------------
    cur = np.array(best_seq, dtype=float)

    # Normalise to the simplex (sum = 1) – convenient for the gradient step
    cur = simplex_projection(cur / cur.sum())

    cur_score = fast_score(cur)
    best_score = cur_score
    best_seq = cur.tolist()

    it = 0
    max_iters = 5000
    while time.time() < deadline and it < max_iters:
        # active shift (where convolution attains its maximum)
        _, k_max = conv_max_and_index(cur)

        # gradient of the active term
        g = gradient_shift(cur, k_max)

        # back‑tracking line search (monotone improvement)
        step = 1.0
        improved = False
        cur_val = cur_score
        while step > 1e-7:
            cand = cur - step * g
            cand = np.maximum(cand, 0.0)
            cand = simplex_projection(cand)
            cand_val = fast_score(cand)
            if cand_val < cur_val:
                improved = True
                break
            step *= 0.5

        if not improved:
            # occasional random jitter to escape plateaus
            jitter = 0.01 * rng.random(cur.shape) * cur
            cand = cur + jitter
            cand = np.maximum(cand, 0.0)
            cand = simplex_projection(cand)
            cand_val = fast_score(cand)
            if cand_val < cur_val:
                improved = True

        if improved:
            cur = cand
            cur_score = cand_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.tolist()
                print(f"[{time.time() - start_time:.1f}s] Improved iter={it} "
                      f"score={best_score:.6f}")
                if best_score <= 1.503:
                    print("Target reached – terminating early.")
                    break
        else:
            # if truly stuck, restart from a small random perturbation
            cur = cur + 0.02 * rng.random(cur.shape) * cur
            cur = np.maximum(cur, 0.0)
            cur = simplex_projection(cur)
            cur_score = fast_score(cur)

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    return best_seq
