def lp_improve(seq, target_sum, t, time_left):
    # Compute conv maxima M
    conv = np.convolve(seq, seq)
    M = conv.max()
    # Solve LP
    n = len(seq)
    offsets = np.arange(n, dtype=int)
    data = [seq[:n - off] for off in offsets]  # each diag
    A = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        print("LP failed:", res.message)
        return seq, None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        # fallback
        return seq, None
    factor = target_sum / sum_b
    g = b * factor
    # Combine
    new_seq = (1 - t) * seq + t * g
    # Optional projection onto simplex just in case
    new_seq = _simplex_projection(new_seq, target_sum)
    return new_seq, M

def solve_lp(a, M):
    n = len(a)
    # Build sparse matrix A_ub
    offsets = np.arange(n)
    data = [a[:n - off] for off in offsets]
    A = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
    c = -np.ones(n)   # maximize sum b -> minimize -sum
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    return res

"""Propose candidate using alternating linear programming and simulated annealing."""

import time
import math
import random
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto simplex { w >= 0 , sum(w)=s }."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """Return sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = a.shape[0]
    # diagonal offsets 0..n-1, each diagonal length n-offset
    offsets = np.arange(n, dtype=int)
    data = [a[:n - off] for off in offsets]  # each is a view
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function height vector that minimises the
    evaluation function using a hybrid Linear‑Program → Simulated‑Annealing
    scheme.

    The routine first improves the starting sequence by repeatedly solving
    a linear programme that maximises the ℓ₁‑norm of a partner sequence
    under the current convolution bound.  The partner is then mixed back
    into the current sequence.  After a fixed number of LP steps the method
    switches to a cheap simulated‑annealing hill‑climber to polish the
    result.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size and target ℓ₁‑norm (the usual normalisation)
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        cur = np.array(height_sequence_1, dtype=float)
    else:
        n = 1500
        cur = np.full(n, 1.0)

    target_sum = math.sqrt(2.0 * n)

    # Normalise the starting vector to the target ℓ₁‑norm
    cur = cur * (target_sum / cur.sum())

    # ------------------------------------------------------------------
    # Keep track of the best sequence seen so far
    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, target_sum={target_sum:.4f}, initial score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – LP based improvement
    max_lp_iters = 30
    lp_iter = 0
    while time.time() < deadline and lp_iter < max_lp_iters:
        # recompute convolution and current bound
        conv = np.convolve(cur, cur)
        M = conv.max()

        # Solve LP: maximise sum(b) s.t. conv(cur,b) ≤ M, b≥0
        A = _build_conv_matrix(cur)
        c = -np.ones(n)                       # maximise sum → minimise -sum
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # use HiGHS – fast and supports sparse matrices
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[LP {lp_iter}] linprog failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            print(f"[LP {lp_iter}] trivial solution, aborting LP phase.")
            break

        # Scale b to have the exact target ℓ₁‑norm
        g = b * (target_sum / sum_b)

        # Mixing coefficient – start with 0.2 and decay linearly
        elapsed = time.time() - start_time
        t = max(0.05, 0.2 * (1.0 - (elapsed / (deadline - start_time))))

        # Convex combination and projection back onto the simplex
        new_cur = (1.0 - t) * cur + t * g
        new_cur = _simplex_projection(new_cur, target_sum)

        # Evaluate
        score = evaluate_sequence(new_cur.tolist())

        # Record improvement
        if score < best_score:
            best_score = score
            best_seq = new_cur.copy()
            print(f"[LP {lp_iter:02d}] New best score: {best_score:.6f} (t={t:.3f}, time {elapsed:.1f}s)")

        # Update current sequence (even if no improvement – the method is
        # monotone in the bound M, but keeping the iterate helps the search)
        cur = new_cur
        lp_iter += 1

    # ------------------------------------------------------------------
    # Phase 2 – cheap simulated‑annealing / random walk polish
    #   Use the remaining time budget.
    T_start = 1.0
    T_end = 1e-4
    sa_iter = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = (elapsed) / (deadline - start_time)
        T = T_start * ((T_end / T_start) ** frac)

        # Random transfer move on the simplex
        proposal = cur.copy()
        i, j = np.random.randint(n, size=2)
        if i != j:
            delta = np.random.normal(loc=0.0, scale=target_sum * 0.02 * T)
            # keep variables feasible
            delta = max(-proposal[i], min(delta, proposal[i]))
            proposal[i] -= delta
            proposal[j] += delta

        proposal = _simplex_projection(proposal, target_sum)

        # Sub‑gradient step targeting the current maximal convolution entry
        conv = np.convolve(proposal, proposal)
        k_max = int(np.argmax(conv))
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        gradient = np.zeros_like(proposal)
        gradient[mask] = proposal[partner[mask]]
        eta = target_sum * 0.1 * T
        proposal -= eta * gradient

"""Hybrid LP + Simulated‑Annealing optimizer for the step‑function problem.

The algorithm works in two phases:

1. **LP phase** – starting from the current sequence `a`, we maximise the
   ℓ₁‑norm of a partner sequence `b` under the constraint
   `conv(a, b) ≤ M`, where `M = max(conv(a, a))`.  The partner is then
   mixed back into `a` (`a ← (1‑t)·a + t·b`) and projected onto the
   simplex (ℓ₁‑norm fixed to `sqrt(2·n)`).  Repeating this step reduces
   the maximal convolution value.

2. **SA phase** – the remaining time budget is spent in a cheap
   simulated‑annealing walk that performs random mass transfers and a
   sub‑gradient step aimed at the presently maximal convolution entry.
   Improvements are accepted either deterministically or with a
   temperature‑dependent probability.

Both phases respect the non‑negativity constraint and the scale‑invariance
of the objective, and they keep the total ℓ₁‑norm exactly at the usual
normalisation `sqrt(2·n)`.  The best sequence encountered is returned as a
plain Python list.
"""

import time
import math
import random
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex { w ≥ 0 , sum(w)=s }."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                # all zero case (should not happen)
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Return a sparse matrix A such that (A @ b)[k] = (a * b)[k]
    (standard discrete convolution, zero padding outside the interval).
    """
    n = a.shape[0]
    # Offsets 0 … n‑1 give the required Toeplitz structure.
    offsets = np.arange(n, dtype=int)
    # Each diagonal consists of a[0:n‑offset]; scipy.diags pads with zeros
    # automatically when the supplied array is shorter than the diagonal.
    data = [a[: n - off] for off in offsets]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function height vector with a low
    evaluation score.  The search combines a linear‑programming based
    improvement loop with a final simulated‑annealing polishing phase.
    """
    np.random.seed(seed)
    random.seed(seed)

    # --------------------------------------------------------------
    # Problem size and initial vector (reuse previous construction if
    # available, otherwise fall back to a uniform start).
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        cur = np.asarray(height_sequence_1, dtype=float)
    else:
        n = 1500
        cur = np.full(n, 1.0, dtype=float)

    target_sum = math.sqrt(2.0 * n)

    # Normalise the ℓ₁‑norm to the usual value sqrt(2·n).
    cur = cur * (target_sum / cur.sum())

    # --------------------------------------------------------------
    # Keep track of the best sequence seen so far.
    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, target_sum={target_sum:.4f}, score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a tiny safety margin

    # ==============================================================
    # Phase 1 – Linear‑programming driven improvement
    # ==============================================================
    max_lp_iters = 30
    lp_iter = 0
    while time.time() < deadline and lp_iter < max_lp_iters:
        # Current bound M = max(conv(cur, cur))
        conv_cur = np.convolve(cur, cur)
        M = conv_cur.max()

        # Build the constraint matrix A (conv(cur, b) = A @ b)
        A = _build_conv_matrix(cur)

        # Linear programme: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n, dtype=float)                     # objective
        b_ub = np.full(2 * n - 1, M, dtype=float)       # upper bound vector
        bounds = [(0.0, None)] * n                       # b_i ≥ 0

        # Solve with HiGHS (supports sparse A_ub)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print(f"[LP {lp_iter}] linprog failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            print(f"[LP {lp_iter}] trivial solution, aborting LP phase.")
            break

        # Scale b to the exact ℓ₁‑norm required.
        g = b * (target_sum / sum_b)

        # Mixing coefficient – start relatively large and decay.
        elapsed = time.time() - start_time
        t = max(0.05, 0.20 * (1.0 - (elapsed / (deadline - start_time))))

        # Convex combination and projection back onto the simplex.
        new_cur = (1.0 - t) * cur + t * g
        new_cur = _simplex_projection(new_cur, target_sum)

        # Evaluate the new candidate.
        score = evaluate_sequence(new_cur.tolist())

        # Record improvement.
        if score < best_score:
            best_score = score
            best_seq = new_cur.copy()
            print(f"[LP {lp_iter:02d}] New best score: {best_score:.6f} "
                  f"(t={t:.3f}, time {elapsed:.1f}s)")

        # Continue the iteration with the new vector.
        cur = new_cur
        lp_iter += 1

    # ==============================================================
    # Phase 2 – Simulated‑Annealing polishing
    # ==============================================================
    T_start = 1.0
    T_end = 1e-4
    sa_iter = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = (elapsed) / (deadline - start_time)
        T = T_start * ((T_end / T_start) ** frac)   # exponential cooling

        # ---- random mass transfer on the simplex -----------------
        proposal = cur.copy()
        i, j = np.random.randint(n, size=2)
        if i != j:
            delta = np.random.normal(loc=0.0, scale=target_sum * 0.02 * T)
            delta = max(-proposal[i], min(delta, proposal[i]))  # keep ≥0
            proposal[i] -= delta
            proposal[j] += delta
        proposal = _simplex_projection(proposal, target_sum)

        # ---- sub‑gradient step targeting the current max entry -----
        conv_prop = np.convolve(proposal, proposal)
        k_max = int(np.argmax(conv_prop))
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        gradient = np.zeros_like(proposal)
        gradient[mask] = proposal[partner[mask]]
        eta = target_sum * 0.10 * T
        proposal -= eta * gradient
        proposal = _simplex_projection(proposal, target_sum)

        # ---- evaluate and accept ----------------------------------
        score = evaluate_sequence(proposal.tolist())
        accept = False
        if score < best_score:
            accept = True
        else:
            # uphill moves with probability exp(-(Δ) / T)
            prob = math.exp(-(score - best_score) / (T + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur = proposal
            if score < best_score:
                best_score = score
                best_seq = proposal.copy()
                print(f"[SA {sa_iter:06d}] New best score: {best_score:.6f} "
                      f"(time {elapsed:.1f}s)")

        sa_iter += 1

    # --------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} after "
          f"{lp_iter} LP iters and {sa_iter} SA iters")
    return best_seq.tolist()
