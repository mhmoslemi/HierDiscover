    # from https://gist.github.com/...
    n = v.shape[0]
    u = np.sort(v)[::-1]

    w = np.maximum(v - theta, 0)
    return w

    # Based on Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1

    cond = u - cssv / ind > 0

        # all v <=0?
        return np.full_like(v, 1.0 / n)

"""
Propose a candidate step‑function by minimizing the peak of its
self‑convolution.  The objective 2·n·max(conv(a,a))/sum(a)²
is invariant under scaling, so we work on the simplex
   Σ a_i = 1 , a_i ≥ 0.
A smooth surrogate of the max‑operator is the soft‑max
   w_k ∝ exp(α·(b_k−max b))   where b = a⊛a.
The gradient of Σ_k w_k b_k with respect to a is
   ∇ = 2·corr(a,w)          (correlation, not convolution).
We perform projected (sub‑)gradient descent on the simplex,
with an increasing temperature α (more focus on the true max) and a
decaying step size.  Random restarts and a final small random
perturbation phase are used to avoid local minima.
Only when a surrogate improvement is seen do we call the expensive
`evaluate_sequence` to obtain the exact score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the algorithm from Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(p: np.ndarray, fft_size: int) -> np.ndarray:
    """
    Linear self‑convolution a⊛a using FFT.
    Returns the full vector of length 2·len(p)−1.
    """
    p_fft = np.fft.rfft(p, n=fft_size)
    conv_full = np.fft.irfft(p_fft * p_fft, n=fft_size)
    return conv_full[: 2 * p.shape[0] - 1]


def _gradient(p: np.ndarray, w: np.ndarray, fft_size: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a⊛a)_k w.r.t. a.
    Implements   ∇ = 2·corr(a,w)   via FFT.
    """
    n = p.shape[0]
    # FFT of a (already zero‑padded to fft_size)
    p_fft = np.fft.rfft(p, n=fft_size)

    # correlation = convolution of a with reversed w
    w_rev = w[::-1]
    w_fft = np.fft.rfft(w_rev, n=fft_size)
    corr_full = np.fft.irfft(p_fft * w_fft, n=fft_size)

    start = len(w) - 1               # = 2·n‑2
    grad = 2.0 * corr_full[start : start + n]
    return grad


def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max over the convolution vector b.
    The maximum entry gets weight 1, others are down‑weighted by exp(α·Δ).
    """
    b_max = b.max()
    # stable exponentiation
    shifted = b - b_max                # ≤ 0, max becomes 0
    ex = np.exp(alpha * shifted)
    s = ex.sum()
    if s == 0.0:
        # extremely unlikely, fall back to uniform
        return np.full_like(b, 1.0 / b.size)
    return ex / s

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Return a list of non‑negative heights that (approximately) minimize
    evaluate_sequence.  The function runs for at most `budget_s` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if "height_sequence_1" in globals():

    best_a = np.full(n, 1.0 / n, dtype=float)

    best_surrogate = 2.0 * n * best_a.max() * n  # placeholder, will be overwritten

                best_score = sc

    # --------------------------------------------------------------

        concentration = np.random.uniform(0.2, 5.0)

    # Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------
    return best_a.tolist()
