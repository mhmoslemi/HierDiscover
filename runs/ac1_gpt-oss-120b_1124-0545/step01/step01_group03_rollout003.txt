import time, numpy as np, math
from scipy.optimize import linprog
from scipy.sparse import diags

def binomial_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0]=1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i-1)/(2.0*i)
    return a

def conv_max(a):
    # return max conv using fast FFT
    n=len(a)
    size=1<<((2*n-1).bit_length())
    f=np.fft.rfft(a, n=size)
    conv=np.fft.irfft(f*f, n=size)[:2*n-1]
    return conv.max()

def fast_score_simple(a):
    n=len(a)
    maxb = conv_max(a)
    sum_a = a.sum()
    return 2.0*n*maxb/(sum_a**2)
```

    iteration = 0

            break

"""
Propose a candidate step‑function sequence with a low evaluation score.

Algorithm
---------
1. Initialise a truncated central‑binomial sequence and re‑scale it so that
   its L1‑norm equals √(2·n).  For this normalisation the evaluation
   function reduces to the maximal entry of the (full) auto‑convolution.

2. Iteratively improve the sequence by solving a linear programme:
      maximise   Σ g_j
      subject to  (f * g)[k] ≤ M   for all k,
                  g_j ≥ 0,
   where f is the current sequence and M = max(f*f).  The constraints are
   linear in g because f is fixed.  After solving we re‑scale g to have the
   same L1‑norm as f and replace f by a convex combination
        f ← (1−t)·f + t·g
   (trying several step sizes t).  This guarantees a non‑increasing
   maximal convolution value.

3. After the LP loop we perform a light stochastic hill‑climb (tiny mass
   moves) to polish the result.

The whole process stops when the time budget (≈ 1000 s) is exhausted or
no further improvement is found.  The final list is returned and also
checked with the official `evaluate_sequence` function.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags


def binomial_sequence(length: int) -> np.ndarray:
    """
    Central binomial coefficients scaled by 4**i:
        a[i] = C(2i,i) / 4**i
    The recurrence a[i] = a[i-1] * (2*i-1) / (2*i) is used.
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def conv_max_fft(a: np.ndarray, fft_size: int | None = None) -> float:
    """
    Fast (FFT‑based) computation of max_{k} (a * a)[k].
    """
    n = a.shape[0]
    if fft_size is None:
        fft_size = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(a, n=fft_size)
    conv = np.fft.irfft(f * f, n=fft_size)[: 2 * n - 1]
    return float(conv.max())


def fast_score(a: np.ndarray, fft_size: int | None = None) -> float:
    """
    Compute the official objective 2·n·max(conv) / (sum(a)^2)
    using the fast FFT based convolution.
    """
    n = a.shape[0]
    max_b = conv_max_fft(a, fft_size)
    sum_a = float(a.sum())
    if sum_a < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)

    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1) Initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    n = 2000                                         # moderate size – tractable LP
    f = binomial_sequence(n)

    # Normalise to L1‑norm = √(2·n).  With this normalisation the official
    # objective equals the maximal convolution value.
    target_sum = math.sqrt(2.0 * n)
    f = f * (target_sum / f.sum())

    # Pre‑compute a reusable FFT length (next power of two ≥ 2·n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    M = conv_max_fft(f, fft_len)                     # current maximal convolution
    best_f = f.copy()
    best_M = M
    print(f"[init] n={n}, objective (max conv) = {M:.6f}")

    # ------------------------------------------------------------------
    # 2) LP‑based monotone improvement ---------------------------------
    # ------------------------------------------------------------------
    max_lp_iters = 30
    iteration = 0
    while time.time() < deadline and iteration < max_lp_iters:
        iteration += 1

        # Build the convolution matrix A such that A @ g = f * g (full mode)
        # A has shape (2n‑1, n) and constant diagonals equal to f[i].
        diag_data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
        offsets = np.arange(n, dtype=int)            # 0 … n‑1
        A = diags(diag_data, offsets, shape=(2 * n - 1, n), format='csc')

        # Right‑hand side: all entries must stay ≤ M
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # Linear programme: maximise Σ g_j  <=> minimise -Σ g_j
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # Solve using HiGHS (fast sparse LP solver)
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[iter {iteration}] LP failed: {res.message}")
            break

        g = np.maximum(res.x, 0.0)                  # numerical safety
        sum_g = g.sum()
        if sum_g == 0.0:
            print(f"[iter {iteration}] LP returned zero vector.")
            break

        # Rescale g to have the same L1‑norm as f
        g = g * (target_sum / sum_g)

        # Try several mixing coefficients to obtain a strict improvement.
        improved = False
        for t in (1.0, 0.5, 0.2, 0.1, 0.05, 0.02):
            f_new = (1.0 - t) * f + t * g
            # Optional clipping to respect the hard 1000‑cap of the evaluator
            np.clip(f_new, 0.0, 1000.0, out=f_new)

            M_new = conv_max_fft(f_new, fft_len)
            if M_new < best_M - 1e-9:               # strict improvement
                f = f_new
                M = M_new
                best_f = f.copy()
                best_M = M_new
                improved = True
                print(f"[iter {iteration}] t={t:.3f}, new objective={M_new:.6f}")
                break

        if not improved:
            # No improvement found with any step size – stop the LP loop.
            print(f"[iter {iteration}] No improving mix found; terminating LP phase.")
            break

        # Early exit if we have already crossed the target bound.
        if best_M < 1.503:
            print("[early stop] target objective reached.")
            break

    # ------------------------------------------------------------------
    # 3) Light stochastic hill‑climb (fine‑tuning) --------------------
    # ------------------------------------------------------------------
    # This follows the simple hill‑climb from the reference code but
    # works on the best sequence discovered by the LP driver.
    attempts = 0
    max_attempts = 200_000
    ft = best_f.copy()
    best_score = fast_score(ft, fft_len)

    while time.time() < deadline and attempts < max_attempts:
        attempts += 1

        i = np.random.randint(n)
        if ft[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # Move a random fraction (up to 5 %) of mass from i to j.
        frac = np.random.rand() * 0.05
        delta = frac * ft[i]
        new_seq = ft.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # Stability checks
        if new_seq[i] < 0.0:
            continue

        score = fast_score(new_seq, fft_len)
        if score < best_score:
            best_score = score
            best_f = new_seq.copy()
            ft = new_seq
            print(f"[hill climb] iter={attempts}, new_score={score:.6f}")

        # stop early if we beat the target
        if best_score < 1.503:
            print("[hill climb] target score reached.")
            break

    # ------------------------------------------------------------------
    # 4) Final validation ------------------------------------------------
    # ------------------------------------------------------------------
    final_sequence = best_f.tolist()
    try:
        official = evaluate_sequence(final_sequence)   # provided security‑checked evaluator
        print(f"[final] official evaluation = {official:.6f}")
    except Exception as exc:  # defensive
        print("[final] evaluation failed:", exc)
        official = None

    return final_sequence
