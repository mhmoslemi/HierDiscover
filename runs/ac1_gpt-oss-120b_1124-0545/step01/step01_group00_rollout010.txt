def gradient_softmax(a, beta):
    n = a.size
    b = linear_convolution(a)
    # b length = 2*n-1
    # compute exp(beta * (b - maxb)) to avoid overflow
    maxb = b.max()
    w = np.exp(beta * (b - maxb))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # compute gradient
    rev_a = a[::-1]
    size = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(rev_a, n=size)
    grad = np.fft.irfft(W * A_rev, n=size)
    # grad length = len(w) + n -1 >= n + (2n-1) -1 = 3n-2 for n>0; we need slice n-1 .. n-1+n
    start = n-1
    grad_slice = grad[start:start+n]
    return 2.0 * grad_slice

def exponentiated_update(a, grad, eta):
    # using log-space to avoid overflow/underflow
    log_a = np.log(a + 1e-30)
    log_a_new = log_a - eta * grad
    # shift to avoid large numbers: subtract max

    a_new /= a_new.sum()
    return a_new

def simplex_proj(v):
    v = np.asarray(v)
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid simulated‑annealing + exponentiated‑gradient search for the
convolution‑peak minimisation problem.

The routine keeps the current step‑function `a` on the probability simplex
(sum = 1, a_i ≥ 0).  Three elementary moves are mixed:

* **random_move** – a tiny random transfer of mass between two bins.
* **targeted_move** – moves mass away from the indices that most
  contribute to the current maximal convolution entry.
* **gradient_step** – a cheap “soft‑max” gradient descent step that
  flattens the convolution peak.  The gradient of the smoothed max is
  obtained efficiently with FFT‑based convolutions.  The update is an
  exponentiated‑gradient (mirror‑descent) step which automatically stays
  on the simplex.

A simple temperature schedule provides a Metropolis acceptance rule so
that occasional uphill moves are possible early on.  If no improvement
is observed for a while the search is restarted from a fresh random
distribution.  The best sequence found before the time budget expires is
returned as a plain list of floats.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def conv_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution of two real vectors using FFT.
    Returns a vector of length len(a) + len(b) - 1.
    """
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: n + m - 1]


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation used by the scoring harness.
    It is equivalent to   2·n·max(conv(a,a)) / (sum(a)²).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    b = conv_full(a, a)
    return float(2.0 * len(a) * b.max() / (s * s))


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Transfer a tiny amount of mass from a random source index to a random
    destination index (different from the source).  The transferred amount
    is at most step_scale·a[source].
    """
    n = len(a)
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)

    max_transfer = a[i]
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()                     # keep on simplex
    return a_new


def targeted_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Identify the convolution index `k_max` with the largest value.
    Choose a source index `i` with probability proportional to its
    contribution a[i]·a[k_max–i] (the terms that form that peak) and move
    a small amount of mass from `i` to a random other index.
    """
    n = len(a)
    b = conv_full(a, a)
    k_max = int(np.argmax(b))

    # contribution of each i to b[k_max]
    w = np.zeros(n)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)
    w[idx] = a[idx] * a[k_max - idx]

    if w.sum() == 0.0:
        src = np.random.randint(n)
    else:
        probs = w / w.sum()
        src = np.random.choice(n, p=probs)

    dst = np.random.randint(n - 1)
    if dst >= src:
        dst += 1

    max_transfer = a[src]
    if max_transfer == 0.0:
        return a.copy()

    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new /= a_new.sum()
    return a_new


def gradient_softmax(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the gradient of a smooth approximation of `max(conv(a,a))`.
    The smooth max uses a soft‑max with temperature 1/beta:
        M_β(a) = (1/β)·log Σ_k exp(β·b_k) ,  b = conv(a,a)
    Its gradient is
        ∇M_β = Σ_k p_k ∇b_k,   p_k = exp(β·b_k) / Σ_j exp(β·b_j).
    The implementation uses FFT‑based convolutions and runs in O(n log n).
    """
    n = a.shape[0]

    # full self‑convolution b = a * a
    b = conv_full(a, a)                     # length 2n‑1
    max_b = b.max()
    # soft‑max weights, numerically stabilised
    w = np.exp(beta * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum

    # gradient of each b_k: ∂b_k/∂a_i = 2·a_{k‑i} (zero when out of range)
    rev_a = a[::-1]                         # reversed a
    grad_full = conv_full(w, rev_a)         # length (2n‑1)+(n‑1) = 3n‑2
    start = n - 1
    grad = grad_full[start:start + n]       # slice of length n
    return 2.0 * grad                       # ∇M_β


def exponentiated_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Mirror‑descent (exponentiated‑gradient) update on the simplex.
    a_new_i ∝ a_i · exp(‑η·grad_i).  The result is renormalised to sum = 1.
    """
    # work in log‑space to avoid overflow/underflow
    log_a = np.log(a + 1e-30)
    log_new = log_a - eta * grad
    # shift for numerical stability (subtract max)
    log_new -= np.max(log_new)
    a_new = np.exp(log_new)
    total = a_new.sum()
    if total == 0.0:
        a_new = np.ones_like(a) / len(a)
    else:
        a_new /= total
    return a_new

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine mixes random, targeted and gradient
    moves while using a simple annealing schedule and occasional restarts.
    Returns the best sequence found as a list of plain Python floats.
    """

    start = time.time()
    deadline = start + budget_s - 0.5               # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction if available
    # ------------------------------------------------------------------
    try:

        import __main__ as main_mod
        init_seq = np.array(getattr(main_mod, "height_sequence_1"), dtype=float)
        if init_seq.size == n:
            a_cur = init_seq.copy()
            a_cur /= a_cur.sum()
        else:
            raise AttributeError
    except Exception:
        # otherwise start from a random Dirichlet‑like point
        a_cur = np.random.rand(n)
        a_cur /= a_cur.sum()

    best_seq = a_cur.copy()
    best_score = compute_score(best_seq)
    cur_score = best_score
    print(f"[init] n={n}, initial_score={cur_score:.6f}")

    # ----- annealing / step‑size schedules ---------------------------------
    init_temp = 1.0
    final_temp = 0.001
    init_eta = 0.50        # for exponentiated‑gradient
    final_eta = 0.001
    init_beta = 0.2        # softness of the max‑approximation
    final_beta = 8.0
    init_step = 0.10       # random / targeted move magnitude
    final_step = 0.001

    iteration = 0
    last_improve = start

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start
        progress = min(1.0, max(0.0, elapsed / (budget_s - 10.0)))

        # schedule parameters

        eta = init_eta * (1.0 - progress) + final_eta * progress
        beta = init_beta * (1.0 - progress) + final_beta * progress
        step_scale = init_step * (1.0 - progress) + final_step * progress

            a_cur /= a_cur.sum()
            cur_score = compute_score(a_cur)
            print(f"[{iteration:6d}] restart due to stagnation")

        iteration += 1

    total_time = time.time() - start
    print(f"[done] iterations={iteration}, best_score={best_score:.6f}, time={total_time:.1f}s")
