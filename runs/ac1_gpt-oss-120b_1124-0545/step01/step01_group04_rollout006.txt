"""
Propose a step‑function that minimizes
    2·n·max(conv(a,a)) / (sum(a)^2)   (evaluate_sequence)

Approach
--------
* Initialise with the Catalan ( (1‑x)^‑½ ) coefficients, which already give a
  good bound (~π/2 ≈ 1.571).

* Perform a *smoothed* projected gradient descent.
  – Compute the full linear convolution a*a via FFT.
  – Replace the non‑smooth max by a soft‑max:
        m_β(a) = (1/β)·log Σ_k exp(β·conv_k)
    which approaches the true max as β → ∞.
  – The gradient of m_β is a weighted average of the sub‑gradients of each
    shift; the weights are p_k ∝ exp(β·conv_k).  Using the FFT we obtain the
    gradient efficiently as a correlation:  grad = 2·corr(p, a).

  – After a gradient step we project onto the feasible set
    (non‑negative entries, sum = 1).

  – β is increased slowly (continuation) so the optimisation starts
    with a smooth landscape and finishes with a near‑exact max.

* When the time budget is nearly exhausted we run a short stochastic
  local‑search phase (random mass transfers) to polish the best solution.

The algorithm stays within the 1000 s budget, uses only NumPy and the
provided `evaluate_sequence`, and returns the best sequence found.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions (top‑level)
# ----------------------------------------------------------------------


def compute_catalan_sequence(n: int) -> np.ndarray:
    """
    First n Catalan‑type coefficients  c_i = (2i choose i) / 4**i
    (coefficients of (1‑x)^‑½) normalised to sum to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def random_perturb(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Transfer a fraction *eps* of the mass at a random index to another
    random index (i != j).  The vector stays normalised.
    """
    n = a.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


def smoothed_max_gradient(a: np.ndarray, beta: float, fft_len: int):
    """
    Compute the value of the smoothed max  m_β(a)  and its gradient.
    Returns (m_beta, grad) where grad has the same shape as a.
    """
    # FFT of a
    f_a = np.fft.rfft(a, n=fft_len)

    # linear convolution a*a (full length 2n‑1)
    conv_full = np.fft.irfft(f_a * f_a, n=fft_len)[: 2 * a.size - 1]

    # soft‑max (log‑sum‑exp) – shifted by the exact maximum for numerical stability
    maxb = conv_full.max()
    exp_term = np.exp(beta * (conv_full - maxb))
    p = exp_term / exp_term.sum()                     # normalized weights, sum(p)=1

    # Gradient:  grad_i = 2 * Σ_k p_k a_{k‑i}   (linear correlation)
    # Compute by FFT:   grad = 2·corr(p, a)
    f_p = np.fft.rfft(p, n=fft_len)
    grad_full = np.fft.irfft(np.conj(f_p) * f_a, n=fft_len)[: a.size]
    grad = 2.0 * grad_full

    # The smoothed objective value (use the true max for reporting)
    m_beta = maxb
    return m_beta, grad
