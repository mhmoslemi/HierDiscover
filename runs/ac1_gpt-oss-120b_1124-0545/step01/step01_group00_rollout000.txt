def lp_improve(f, M, t_mix=0.5):
    # Build A scaled
    n = len(f)
    rows = 2*n - 1
    A = np.zeros((rows, n))
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        A[k, lo:hi+1] = f[k - idx]
    # scale
    A_scaled = A / M
    b_ub = np.ones(rows)
    c = -np.ones(n)  # maximize sum(g)
    res = linprog(c, A_ub=A_scaled, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return f, M, False
    g = res.x
    sum_g = g.sum()

    # Mix

    # Ensure nonneg (should be)
    f_new = np.clip(f_new, 0, None)
    f_new /= f_new.sum()
    return f_new, None, True

    # Phase 1: Linear programming improvement
    lp_deadline = start + min(200.0, budget_s * 0.2)  # allocate up to 20% of time or 200s
    t_mix = 0.5
    iteration_lp = 0
    while time.time() < lp_deadline:
        conv = np.convolve(a, a)
        M = conv.max()
        # solve LP
        rows = 2 * n - 1
        A = np.zeros((rows, n))
        for k in range(rows):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # indices from lo to hi inclusive
            idx = np.arange(lo, hi + 1)
            A[k, lo:hi+1] = a[k - idx]
        # scale for numerical stability
        if M <= 0:
            break
        A_scaled = A / M
        b_ub = np.ones(rows)
        c = -np.ones(n)
        # Solve LP
        try:
            res = linprog(c, A_ub=A_scaled, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        except Exception as e:
            print(f"[lp] linprog failed: {e}")
            break

        if not res.success:
            print(f"[lp] LP not successful: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # normalize g to sum=1
        g = g / sum_g

        # mix with current a
        a_new = (1 - t_mix) * a + t_mix * g
        a_new = np.clip(a_new, 0, None)
        a_new /= a_new.sum()
        new_score = compute_score_fast(a_new)

        if new_score < best_score - 1e-9:
            best_seq = a_new.copy()
            best_score = new_score
            a = a_new
            print(f"[lp {iteration_lp}] improved score={best_score:.6f}")
        else:
            # if not improved, reduce mixing factor
            t_mix *= 0.7
            if t_mix < 1e-4:
                break
        iteration_lp += 1

    # Phase 2: Simulated annealing / local refinement
    # Use same functions as earlier but maybe adjust parameters
    # Initialize parameters
    a_curr = best_seq.copy()
    best_seq = a_curr.copy()
    best_score = compute_score_fast(best_seq)
    # SA parameters
    init_temp = 1.0
    final_temp = 0.001
    init_step = 0.2
    final_step = 0.001
    iteration = 0
    last_improve = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        progress = (elapsed) / (budget_s - 10.0)  # safety margin
        temp = init_temp * (1.0 - progress) + final_temp * progress
        step_scale = init_step * (1.0 - progress) + final_step * progress

        # choose move
        r = np.random.rand()
        if r < 0.45:
            cand = targeted_move(a_curr, step_scale)
        elif r < 0.85:
            cand = random_move(a_curr, step_scale)
        else:
            cand = gradient_step(a_curr, step_scale * 0.5)

        cand_score = compute_score_fast(cand)
        # acceptance
        if cand_score < best_score or np.random.rand() < np.exp(-(cand_score - best_score) / max(temp, 1e-12)):
            a_curr = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:6d}] SA new best = {best_score:.6f} elapsed={elapsed:.1f}s")
        # restart if stagnant
        if iteration % 4000 == 0 and (time.time() - last_improve) > 30.0:
            a_curr = np.random.rand(n)
            a_curr /= a_curr.sum()
            print(f"[{iteration:6d}] SA restart due to stagnation")
        iteration += 1

    total_time = time.time() - start
    print(f"[done] total_time={total_time:.1f}s best score={best_score:.6f}")
    return best_seq.tolist()

"""
Hybrid linear‑programming + simulated‑annealing optimizer.

The search proceeds in two phases:

1. **LP improvement** –  For the current distribution `f` we solve the
   linear program (maximise ∑g subject to f∗g ≤ M, g ≥ 0) where
   `M = max(f∗f)`.  The optimiser pushes mass towards indices that are
   *under‑used* by the current autocorrelation.  The resulting `g` is
   normalised to sum = 1 and mixed with `f` (mixing factor `t`).  This
   step often yields a noticeable drop in the peak of the convolution.

2. **Refinement** –  A simulated‑annealing / hill‑climbing loop similar to
   the previous submission, but seeded with the best‑found sequence from
   phase 1.  It uses three elementary moves (targeted, random transfer,
   cheap gradient step) and an exponential‐cooling schedule.  Occasional
   restarts avoid stagnation.

Both phases respect the non‑negativity and normalisation constraints,
so the final list can be passed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# optional import – the LP phase is skipped if SciPy is unavailable
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a non‑negative vector ``a``.
    The vector is assumed to be normalised (sum may be ≠ 1, the formula
    accounts for it).  This matches ``evaluate_sequence`` up to the
    clipping performed there.
    """
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    max_b = np.convolve(a, a).max()
    return float(2.0 * len(a) * max_b / (s * s))


def _random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one index to another."""
    n = len(a)
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = np.clip(a_new, 0.0, None)
    total = a_new.sum()
    if total == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= total
    return a_new


def _targeted_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Move mass away from an index that contributes most to the current
    maximal autocorrelation entry.
    """
    n = len(a)
    conv = np.convolve(a, a)
    k_max = int(conv.argmax())

    # contribution of each index to the peak at k_max
    w = np.zeros(n)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1)
    w[idx] = a[idx] * a[k_max - idx]

    if w.sum() == 0.0:
        i = np.random.randint(n)
    else:
        probs = w / w.sum()
        i = np.random.choice(n, p=probs)

    # choose a destination different from i
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1

    max_transfer = a[i]
    if max_transfer == 0.0:
        return a.copy()
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new = np.clip(a_new, 0.0, None)
    total = a_new.sum()
    if total == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= total
    return a_new


def _gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    Cheap gradient‑like move that flattens the peak of the convolution.
    The direction d = –g + mean(g) keeps the sum unchanged.
    """
    n = len(a)
    conv = np.convolve(a, a)
    k = int(conv.argmax())

    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]               # ∂b_k/∂a_i = 2·a_{k−i}
    d = -g + g.mean()
    a_new = a + step * d
    a_new = np.clip(a_new, 0.0, None)
    total = a_new.sum()
    if total == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= total
    return a_new


def _lp_improve(f: np.ndarray, mix: float) -> tuple[np.ndarray, bool]:
    """
    One linear‑programming improvement step.

    Parameters
    ----------
    f : np.ndarray
        Current non‑negative distribution (sum ≈ 1).
    mix : float
        Mixing coefficient t ∈ (0, 1] used in f←(1‑t)f + t·ĝ,
        where ĝ is the LP solution normalised to sum = 1.

    Returns
    -------
    f_new : np.ndarray
        Updated distribution (may be identical to the input).
    improved : bool
        True if the new distribution strictly reduces the objective.
    """
    n = len(f)
    rows = 2 * n - 1

    # Build the Toeplitz‑like constraint matrix A where
    #   (A @ g)[k] = (f * g)[k]   (full convolution)
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            A[k, lo:hi + 1] = f[k - idx]

    M = np.convolve(f, f).max()
    if M <= 0.0:                         # degenerate – nothing to improve
        return f, False

    # Scale constraints to avoid tiny numbers (numerical stability)
    A_scaled = A / M
    b_ub = np.ones(rows, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)    # maximise ∑g  →  minimise -∑g

    try:
        res = linprog(c, A_ub=A_scaled, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
    except Exception:                    # pragma: no cover
        return f, False

    if not res.success:
        return f, False

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f, False

    # Normalise g to have sum = 1 (the same normalisation as f)
    g_hat = g / sum_g

    # Convex combination
    f_new = (1.0 - mix) * f + mix * g_hat
    f_new = np.clip(f_new, 0.0, None)
    total = f_new.sum()
    if total == 0.0:
        f_new = np.ones_like(f) / n
    else:
        f_new /= total

    improved = _score(f_new) < _score(f) - 1e-12
    return f_new, improved


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # initialise – use the known good construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if init.size != n:                     # truncate / pad if needed
            if init.size > n:
                init = init[:n]
            else:
                pad = np.zeros(n - init.size, dtype=np.float64)
                init = np.concatenate([init, pad])
        a = np.maximum(init, 0.0)
        s = a.sum()
        a = a / s if s > 0 else np.ones(n) / n
    else:
        a = np.random.rand(n)
        a /= a.sum()

    best_seq = a.copy()
    best_score = _score(best_seq)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – linear‑programming driven improvement
    # ------------------------------------------------------------------
    if _SCIPY_AVAILABLE:
        lp_deadline = start + min(200.0, budget_s * 0.20)   # ≤ 20 % of the budget
        mix = 0.5                                            # initial mixing factor
        lp_iter = 0
        while time.time() < lp_deadline:
            a_new, improved = _lp_improve(a, mix)
            if improved:
                a = a_new
                cur_score = _score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[lp {lp_iter:04d}] improved score={best_score:.6f}")
                # keep mixing factor steady if we see progress
            else:
                # no progress – shrink the mixing factor to make finer moves
                mix *= 0.7
                if mix < 1e-4:
                    break
            lp_iter += 1
    else:                                                   # pragma: no cover
        print("[info] SciPy not available – skipping LP phase.")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing / hill climbing
    # ------------------------------------------------------------------
    a_curr = best_seq.copy()
    best_score = _score(best_seq)

    # annealing schedule
    init_temp = 1.0
    final_temp = 0.001
    init_step = 0.20
    final_step = 0.001

    iteration = 0
    last_improve = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / (budget_s - 10.0)          # keep a tiny safety margin
        # linear interpolation of temperature and step size
        temp = init_temp * (1.0 - progress) + final_temp * progress
        step_scale = init_step * (1.0 - progress) + final_step * progress

        # choose a move
        r = np.random.rand()
        if r < 0.45:
            cand = _targeted_move(a_curr, step_scale)
        elif r < 0.85:
            cand = _random_move(a_curr, step_scale)
        else:
            cand = _gradient_step(a_curr, step_scale * 0.5)

        cand_score = _score(cand)

        # Metropolis acceptance
        if cand_score < best_score or np.random.rand() < math.exp(
                -(cand_score - best_score) / max(temp, 1e-12)):
            a_curr = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:07d}] new best = {best_score:.6f}  t={elapsed:.1f}s")

        # restart if we have been stagnant for a while
        if iteration % 4000 == 0 and (time.time() - last_improve) > 30.0:
            a_curr = np.random.rand(n)
            a_curr /= a_curr.sum()
            print(f"[{iteration:07d}] restart (stagnation)")

        iteration += 1

    total_time = time.time() - start
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
