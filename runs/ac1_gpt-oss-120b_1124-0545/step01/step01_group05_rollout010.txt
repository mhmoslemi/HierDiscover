def _project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _project_simplex(v): ... (maybe not needed)

def _conv_full(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv

    conv = _conv_full(a)
    idx = int(np.argmax(conv))
    max_val = float(conv[idx])
    return max_val, idx

def _fast_score(a):
    n = a.shape[0]
    maxconv, _ = _conv_max_and_idx(a)
    return 2.0 * n * maxconv

def _make_exponential(n, lam):
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a

def _make_power_law(n, p):
    i = np.arange(1, n+1, dtype=float)
    a = i ** p
    a /= a.sum()
    return a

def _make_two_block(n, m, high_frac):
    high = high_frac / m
    low = (1.0 - high_frac) / (n - m)
    a = np.empty(n, dtype=float)
    a[:m] = high
    a[m:] = low
    return a

def _make_mix_exponential(n, w, lam1, lam2):
    i = np.arange(n, dtype=float)
    a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
    a /= a.sum()
    return a

def _make_dirichlet(n, alpha):
    a = np.random.dirichlet([alpha] * n)
    return a

def _refine_candidate(a0, deadline):
    """
    Subgradient-like multiplicative refinement.
    The function stops when the wall‑clock time reaches `deadline`.
    Returns the best sequence found and its fast score.
    """
    a = a0.astype(float).copy()
    # ensure non‑negativity and normalise
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total == 0.0:
        a = np.full_like(a, 1.0 / len(a))
    else:
        a /= total

    best = a.copy()
    best_score = _fast_score(best)
    it = 0
    step0 = 0.6  # initial learning‑rate constant

    while time.time() < deadline:
        it += 1
        # convolution and the index of the current maximal entry
        _, k = _conv_max_and_idx(a)
        n = len(a)

        # gradient of the active convolution entry w.r.t a
        g = np.zeros_like(a)
        if k < n:
            g[: n - k] += a[k:]
        if k > 0:
            g[k:] += a[: n - k]

        # diminishing multiplicative step
        eta = step0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # renormalise onto the probability simplex (simply divide by the sum)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        score = _fast_score(a)
        if score < best_score:
            best_score = score
            best = a.copy()

            a[j] = max(0.0, a[j] + delta)
            s = a.sum()
            if s == 0.0:

            else:

"""
Propose a step‑function with a low value of

    2·n·max(conv(a,a)) / (sum(a)**2)

using a two‑phase scheme:

1. Randomly sample a seed from a few simple parametric families
   (exponential, power‑law, two‑block, mixture of exponentials,
   Dirichlet).

2. Refine the seed with a cheap sub‑gradient style multiplicative update
   that directly reduces the currently maximal convolution entry.
   The update preserves non‑negativity, after which the vector is
   projected back onto the probability simplex (simply re‑normalised).

The method respects the 1000 s budget, prints progress and returns the
best candidate as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------
# Convolution helpers
# --------------------------------------------------

def _conv_full(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a using an FFT of appropriate size."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _conv_max_and_idx(a: np.ndarray) -> tuple[float, int]:
    """Maximum value of a*a and the index where it is attained."""
    conv = _conv_full(a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _fast_score(a: np.ndarray) -> float:
    """Fast surrogate of evaluate_sequence when sum(a)=1:
    2·n·max(conv(a,a))."""
    n = a.shape[0]
    maxconv, _ = _conv_max_and_idx(a)
    return 2.0 * n * maxconv


# --------------------------------------------------
# Simple parametric families (used for random restarts)
# --------------------------------------------------

def _make_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) (lam ≥ 0)."""
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a


def _make_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** p
    a /= a.sum()
    return a


def _make_two_block(n: int, m: int, high_frac: float) -> np.ndarray:
    """First `m` entries share `high_frac` of the mass, the rest share the rest."""
    high = high_frac / m
    low = (1.0 - high_frac) / (n - m)
    a = np.empty(n, dtype=float)
    a[:m] = high
    a[m:] = low
    return a


def _make_mix_exponential(n: int, w: float, lam1: float, lam2: float) -> np.ndarray:
    """Mixture of two exponentials."""
    i = np.arange(n, dtype=float)
    a = w * np.exp(-lam1 * i) + (1.0 - w) * np.exp(-lam2 * i)
    a /= a.sum()
    return a


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Random Dirichlet vector."""
    return np.random.dirichlet([alpha] * n)


# --------------------------------------------------
# Gradient of a single convolution entry
# --------------------------------------------------

def _gradient_at_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of conv_k = (a*a)[k] with respect to a.
    The formula follows from
        conv_k = Σ_i a_i·a_{k-i}
    and the convention that out‑of‑range indices are zero.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo > hi:
        return g

    idx = np.arange(lo, hi + 1, dtype=int)
    partner = k - idx                     # partner indices are also in [0,n)

    # contribution from the term a_i·a_{k-i}
    g[idx] += a[partner]
    # contribution from the symmetric term a_{k-i}·a_i
    g[partner] += a[idx]

    return g


# --------------------------------------------------
# Multiplicative sub‑gradient refinement
# --------------------------------------------------

def _refine_candidate(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Starting from `a0` (already normalised to sum=1), repeatedly
    1) locate the current maximal convolution entry,
    2) compute its sub‑gradient,
    3) apply a diminishing multiplicative step,
    4) renormalise onto the simplex.

    The routine stops when the wall‑clock time reaches `deadline`
    and returns the best sequence found together with its fast score.
    """
    a = np.clip(a0, 0.0, None)
    total = a.sum()
    if total == 0.0:
        a = np.full_like(a, 1.0 / len(a))
    else:
        a /= total

    best = a.copy()
    best_score = _fast_score(best)

    it = 0
    step0 = 0.6                     # initial learning‑rate factor

    while time.time() < deadline:
        it += 1

        # 1) current maximal convolution entry
        _, k = _conv_max_and_idx(a)

        # 2) sub‑gradient of that entry
        g = _gradient_at_shift(a, k)

        # 3) multiplicative update with a diminishing step size
        eta = step0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # 4) re‑normalise (projection onto the probability simplex)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / len(a))
        else:
            a /= s

        # keep the best local result
        score = _fast_score(a)
        if score < best_score:
            best_score = score
            best = a.copy()

        # occasional tiny random jitter to escape flat regions
        if it % 400 == 0:
            i, j = np.random.choice(len(a), 2, replace=False)
            delta = a[i] * np.random.uniform(-0.2, 0.2)
            a[i] = max(0.0, a[i] - delta)
            a[j] = max(0.0, a[j] + delta)
            a /= a.sum()

    return best, best_score
