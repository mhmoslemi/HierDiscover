def propose_candidate(seed=0, budget_s=1000):

    # Initialize candidate from height_sequence_1 if exists else uniform
    if 'height_sequence_1' in globals():
        best_seq = np.array(height_sequence_1, dtype=float)
        # ensure sum = target_sum via scaling
        best_seq = best_seq / best_seq.sum() * target_sum
    else:
        best_seq = np.full(n, target_sum / n)

    best_score = evaluate_sequence(best_seq.tolist())

    # keep second best perhaps? Let's use a list of candidates. 
    candidates = []
    # generate some random restarts
    for _ in range(5):
        a = np.random.random(n)
        a = a / a.sum() * target_sum
        candidates.append(a)

    # add best_seq to candidates
    candidates.append(best_seq)

    # Set Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # momentum per candidate: store m, v arrays same size.
    a_states = [{'a': a, 'm': np.zeros_like(a), 'v': np.zeros_like(a), 'score': evaluate_sequence(a.tolist()), 't':0} for a in candidates]

    # time management
    start = time.time()
    end_time = start + budget_s - 2.0

    iteration=0
    best_seq = best_score = None? Already variable.

    while time.time() < end_time:
        # choose candidate: either best or random
        # With probability 0.7 use best, else random
        if np.random.rand() < 0.7:
            # use best candidate (lowest score)
            idx = np.argmin([c['score'] for c in a_states])
        else:
            idx = np.random.randint(len(a_states))
        state = a_states[idx]
        a = state['a']

        # compute conv
        conv = np.fft.ifft(np.fft.fft(a, n*2) * np.fft.fft(a, n*2)).real  # length >=2n
        # conv length is maybe 2n-1. pick max
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # compute gradient = 2 * a_rev shifted
        gradient = np.zeros_like(a)
        # compute partner = max_idx - np.arange(n)
        partner = max_idx - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        gradient[mask] = 2.0 * a[partner[mask]]

        # Adam update
        state['t'] += 1
        t = state['t']
        lr = target_sum * 0.1 / (np.sqrt(t) + 1)   # decreasing lr over time
        state['m'] = beta1 * state['m'] + (1 - beta1) * gradient
        state['v'] = beta2 * state['v'] + (1 - beta2) * (gradient ** 2)
        m_hat = state['m'] / (1 - beta1 ** t)

        a_new = _simplex_projection(a_new, target_sum)

        # random jitter: with probability p_jitter, mix in random noise

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, math, random, numpy as np

    # Set seeds
    np.random.seed(seed)
    random.seed(seed)

    # Determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        init_seq = np.array(height_sequence_1, dtype=float)
    else:
        n = 1500
        init_seq = np.full(n, 1.0)

    target_sum = math.sqrt(2.0 * n)

    # Normalise init_seq to target_sum
    if init_seq.sum() > 0:
        init_seq = init_seq / init_seq.sum() * target_sum
    else:
        init_seq = np.full(n, target_sum / n)

    # Helper: projection onto weighted simplex (via y = w*b)
    def _weighted_simplex_projection(y, w, s):
        # Convert to normalized simplex on y (already weighted)
        return _simplex_projection(y, s)

    # Phase 1: blockwise Adam descent
    phase1_ratio = 0.55  # proportion of total time
    start_time = time.time()
    phase1_deadline = start_time + budget_s * phase1_ratio

    # Choose block size B
    # Heuristic: choose B so that number of blocks ~ 120-200
    approx_blocks = 150
    B = max(1, n // approx_blocks)
    # Ensure B not too small
    if B < 5:
        B = 5
    # Build block lengths
    full_blocks = n // B
    remainder = n % B
    lengths = [B] * full_blocks
    if remainder:
        lengths.append(remainder)
    lengths = np.array(lengths, dtype=int)
    m = len(lengths)

    # block id mapping
    block_ids = np.repeat(np.arange(m), lengths)

    # Initialise y (weighted values)
    # Compute per-block average from init_seq
    block_sums = np.bincount(block_ids, weights=init_seq, minlength=m)
    b_init = block_sums / lengths  # average per element inside block
    y = b_init * lengths  # weighted vector such that a = repeat(b_init)

    # Evaluate initial
    def _expand_y_to_a(y_vec):
        b_vals = y_vec / lengths
        a_vec = np.repeat(b_vals, lengths)
        return a_vec

    a_curr = _expand_y_to_a(y)
    best_seq = a_curr.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # Adam state
    m_vec = np.zeros(m)
    v_vec = np.zeros(m)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    # Determine FFT length
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    while time.time() < phase1_deadline:
        a_curr = _expand_y_to_a(y)

        # convolution via FFT
        fa = np.fft.rfft(a_curr, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        # find max index and value
        k_max = int(np.argmax(conv))
        # compute gradient of max entry w.r.t a
        partner = k_max - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        grad_a = np.zeros_like(a_curr)
        grad_a[mask] = 2.0 * a_curr[partner[mask]]

        # accumulate gradient per block
        grad_block = np.bincount(block_ids, weights=grad_a, minlength=m)
        # gradient w.r.t y (since y = b*len, b = y/len)
        grad_y = grad_block / lengths

        # Adam update
        t_iter += 1
        lr = target_sum * 0.2 / (np.sqrt(t_iter) + 1.0)
        m_vec = beta1 * m_vec + (1 - beta1) * grad_y
        v_vec = beta2 * v_vec + (1 - beta2) * (grad_y ** 2)
        m_hat = m_vec / (1 - beta1 ** t_iter)
        v_hat = v_vec / (1 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y_new = y - step

        # occasional random jitter to escape
        if np.random.rand() < 0.02:
            jitter = np.random.normal(scale=target_sum * 0.01, size=m)
            y_new += jitter

        # project onto simplex (weighted sum constraint)
        y = _simplex_projection(y_new, target_sum)

        # evaluate
        a_curr = _expand_y_to_a(y)
        try:
            score = evaluate_sequence(a_curr.tolist())
        except Exception:
            continue

        if score < best_score:
            best_score = score
            best_seq = a_curr.copy()
            print(f"[Phase1] iter {t_iter:6d} new best {best_score:.6f}")

    # Phase 2: fine‑grained simulated annealing on the full vector
    # Use the best_seq from Phase1 as start
    a_full = best_seq.copy()
    n = a_full.size
    target_sum = math.sqrt(2.0 * n)
    best_score = evaluate_sequence(a_full.tolist())
    # Simple random start also considered
    rand_seq = np.random.exponential(scale=1.0, size=n)
    rand_seq = rand_seq / rand_seq.sum() * target_sum
    rand_score = evaluate_sequence(rand_seq.tolist())
    if rand_score < best_score:
        a_full = rand_seq.copy()
        best_score = rand_score

    # Parameters for simulated annealing
    deadline = start_time + budget_s - 2.0   # safety margin
    iteration = 0
    T_start = 1.0
    T_end = 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = (elapsed) / (budget_s - 2.0)
        T = T_start * ((T_end / T_start) ** frac)

        # random transfer between two positions
        proposal = a_full.copy()
        i, j = np.random.randint(n, size=2)
        if i != j:
            delta = np.random.normal(loc=0.0, scale=target_sum * 0.03 * T)
            # bound delta not to create negatives
            delta = max(-proposal[i], min(delta, proposal[i]))
            proposal[i] -= delta
            proposal[j] += delta

        # projection onto simplex to keep sum
        proposal = _simplex_projection(proposal, target_sum)

        # sub‑gradient step using the current worst convolution entry
        fa = np.fft.rfft(proposal, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        k_max = int(np.argmax(conv))
        partner = k_max - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(proposal)
        grad[mask] = 2.0 * proposal[partner[mask]]
        eta = target_sum * 0.15 * T
        proposal -= eta * grad
        proposal = _simplex_projection(proposal, target_sum)

        # evaluate
        try:
            score = evaluate_sequence(proposal.tolist())
        except Exception:
            continue

        # accept if improvement or with annealed probability
        if score < best_score or np.random.rand() < math.exp(-(score - best_score) / (T + 1e-12)):
            a_full = proposal.copy()
            if score < best_score:
                best_score = score
                best_seq = proposal.copy()
                print(f"[Phase2] iter {iteration:6d} new best {best_score:.6f}")

        iteration += 1

    # final return
    return best_seq.tolist()

"""
Propose a sequence of non‑negative heights that minimises

    E(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm works in two stages:

1. **Coarse block‑wise Adam descent** – the vector is represented by a
   small number of constant blocks.  The optimisation is performed on the
   block‑weights (the “y” vector) using an Adam‑style update on the
   sub‑gradient of the current worst convolution entry.  The weighted
   simplex projection guarantees the ℓ₁‑norm stays equal to
   √(2·n).

2. **Fine‑grained simulated‑annealing** – the best block‑wise solution
   is expanded to the full sequence and refined with a temperature‑scaled
   random walk plus a sub‑gradient step, similar to the previous best
   approach but starting from a much better initial point.

Both stages respect the non‑negativity and ℓ₁‑norm constraints, use
FFT‑based convolutions for speed, and keep the best sequence seen
throughout the allotted time budget.

import math
import random
import numpy as np


# ----------------------------------------------------------------------
# Helper: simplex projection (unweighted)
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex { w >= 0 , sum(w) = s }."""
    n = v.size
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
def _convolution(a: np.ndarray) -> np.ndarray:
    """Return the full discrete convolution a * a using FFT."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def _max_conv_index(conv: np.ndarray) -> int:
    """Index of the maximal entry of a convolution."""
    return int(np.argmax(conv))


def _subgradient_of_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of the entry conv[k_max] w.r.t. a (factor 2 included)."""
    n = a.shape[0]
    partner = k_max - np.arange(n)
    mask = (partner >= 0) & (partner < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[partner[mask]]
    return grad


# ----------------------------------------------------------------------
# Block structure helpers
def _create_block_structure(n: int, block_size: int):
    """Return (lengths, block_ids) for a uniform block partition."""
    full = n // block_size
    rem = n % block_size
    lengths = [block_size] * full
    if rem:
        lengths.append(rem)
    lengths = np.array(lengths, dtype=int)
    block_ids = np.repeat(np.arange(len(lengths)), lengths)
    return lengths, block_ids


def _expand_y_to_seq(y: np.ndarray, lengths: np.ndarray) -> np.ndarray:
    """Expand weighted vector y (y_i = block_value_i * length_i) to full sequence."""
    b = y / lengths
    return np.repeat(b, lengths)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function height vector that minimises the evaluation
    function.  The method first performs a block‑wise Adam descent and then
    refines the result with a simulated‑annealing style random walk.
    """
    # ------------------------------------------------------------------
    # reproducibility
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # problem size
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        init_seq = np.asarray(height_sequence_1, dtype=float)
    else:
        n = 1500
        init_seq = np.full(n, 1.0)

    target_sum = math.sqrt(2.0 * n)

    # normalise initial sequence to the required ℓ₁‑norm
    total = init_seq.sum()
    if total > 0:
        init_seq = init_seq / total * target_sum
    else:
        init_seq = np.full(n, target_sum / n)

    # ------------------------------------------------------------------
    # ----------- Phase 1 : block‑wise Adam descent --------------------
    start_time = time.time()
    phase1_frac = 0.55                       # 55 % of total budget
    phase1_deadline = start_time + budget_s * phase1_frac

    # pick a block size giving ≈150 blocks (adjusted for small n)
    approx_blocks = 150
    block_size = max(5, n // approx_blocks)
    lengths, block_ids = _create_block_structure(n, block_size)
    m = len(lengths)                         # number of blocks

    # initialise weighted vector y from the init sequence
    block_sums = np.bincount(block_ids, weights=init_seq, minlength=m)
    b_init = block_sums / lengths            # average inside each block
    y = b_init * lengths                      # weighted representation

    # Adam bookkeeping
    m_vec = np.zeros(m)
    v_vec = np.zeros(m)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_iter = 0
    last_improve = 0

    # best seen so far
    a_current = _expand_y_to_seq(y, lengths)
    best_seq = a_current.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[Phase1] start best score = {best_score:.6f}")

    # FFT length (constant for this n)
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    while time.time() < phase1_deadline:
        a_current = _expand_y_to_seq(y, lengths)

        # convolution and index of maximal entry
        fa = np.fft.rfft(a_current, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        partner = k_max - np.arange(n)
        mask = (partner >= 0) & (partner < n)
        grad_a = np.zeros_like(a_current)
        grad_a[mask] = 2.0 * a_current[partner[mask]]

        # accumulate gradient per block
        grad_block = np.bincount(block_ids, weights=grad_a, minlength=m)
        grad_y = grad_block / lengths

        # Adam step
        t_iter += 1
        lr = target_sum * 0.2 / (np.sqrt(t_iter) + 1.0)
        m_vec = beta1 * m_vec + (1 - beta1) * grad_y
        v_vec = beta2 * v_vec + (1 - beta2) * (grad_y ** 2)
        m_hat = m_vec / (1 - beta1 ** t_iter)
        v_hat = v_vec / (1 - beta2 ** t_iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y_new = y - step

        # occasional random jitter to escape narrow basins
        if np.random.rand() < 0.02:
            jitter = np.random.normal(scale=target_sum * 0.01, size=m)
            y_new += jitter

        # enforce the weighted simplex constraint (sum(y) = target_sum)
        y = _simplex_projection(y_new, target_sum)

        # evaluate
        a_current = _expand_y_to_seq(y, lengths)
        try:
            score = evaluate_sequence(a_current.tolist())
        except Exception:
            continue

        if score < best_score:
            best_score = score
            best_seq = a_current.copy()
            last_improve = t_iter
            print(f"[Phase1] iter {t_iter:6d}   new best = {best_score:.6f}")

        # if we have stagnated for a while, inject a small random perturbation
        if t_iter - last_improve > 2000:
            y += np.random.normal(scale=target_sum * 0.05, size=m)
            y = _simplex_projection(y, target_sum)
            last_improve = t_iter

    # ------------------------------------------------------------------
    # ----------- Phase 2 : fine‑grained simulated annealing -----------
    # Use the best block‑wise result as starting point.
    a_full = best_seq.copy()
    n = a_full.size               # keep the same n
    target_sum = math.sqrt(2.0 * n)

    best_score = evaluate_sequence(a_full.tolist())
    # occasional completely random restart (helps escape deep local minima)
    rand_seq = np.random.exponential(scale=1.0, size=n)
    rand_seq = rand_seq / rand_seq.sum() * target_sum
    rand_score = evaluate_sequence(rand_seq.tolist())
    if rand_score < best_score:
        a_full = rand_seq.copy()
        best_score = rand_score

    # parameters for the annealing schedule
    deadline = start_time + budget_s - 2.0   # safety margin
    iteration = 0
    T_start, T_end = 1.0, 1e-5

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s - 2.0)
        T = T_start * ((T_end / T_start) ** frac)

        # ---- random mass transfer between two positions -----------------
        proposal = a_full.copy()
        i, j = np.random.randint(n, size=2)
        if i != j:
            delta = np.random.normal(loc=0.0, scale=target_sum * 0.03 * T)
            delta = max(-proposal[i], min(delta, proposal[i]))
            proposal[i] -= delta
            proposal[j] += delta

        # ---- keep the ℓ₁‑norm fixed ------------------------------------
        proposal = _simplex_projection(proposal, target_sum)

        # ---- sub‑gradient step on the current worst convolution entry --
        fa = np.fft.rfft(proposal, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        k_max = int(np.argmax(conv))
        grad = _subgradient_of_max(proposal, k_max)
        eta = target_sum * 0.15 * T
        proposal -= eta * grad
        proposal = _simplex_projection(proposal, target_sum)

        # ---- evaluation & acceptance -----------------------------------
        try:
            score = evaluate_sequence(proposal.tolist())
        except Exception:
            continue

        # accept if better, otherwise with a temperature‑scaled probability
        if score < best_score or np.random.rand() < math.exp(-(score - best_score) / (T + 1e-12)):
            a_full = proposal.copy()
            if score < best_score:
                best_score = score
                best_seq = proposal.copy()
                print(f"[Phase2] iter {iteration:6d}   new best = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    return best_seq.tolist()
