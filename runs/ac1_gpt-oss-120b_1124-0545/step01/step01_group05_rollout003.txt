        a = a * np.exp(-eta * g)

        a_sum = a.sum()

            a /= a_sum

            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best_a = a.copy()

Algorithm: Projected exponentiated‑gradient minimisation of the L∞ norm of the
convolution a∗a (equivalently minimise 2·n·max(conv(a,a))).  The non‑negative
vector a is kept on the probability simplex (sum = 1).  A smooth surrogate for
the max, the log‑sum‑exp with a temperature β, is used; its gradient can be
computed analytically and efficiently by an FFT‑based convolution.  During the
run β is increased geometrically so that the surrogate becomes tighter, while
the step size η is slowly decreased.  The best candidate found (according to
the official `evaluate_sequence`) is kept and finally refined with a few
random local perturbations.

The routine stops after the given time budget (default 1000 s) and returns the
best sequence as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Fast convolution utilities (real FFT, linear convolution)
# --------------------------------------------------------------

def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _conv_fft_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of x and y (both real) using a real FFT."""
    nx = x.shape[0]
    ny = y.shape[0]
    size = 1 << ((nx + ny - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)[: nx + ny - 1]
    return conv


# --------------------------------------------------------------
# Scoring helpers
# --------------------------------------------------------------

def _fast_score(a: np.ndarray) -> float:
    """Fast surrogate for evaluate_sequence when sum(a)=1:
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_full_fft(a)
    return float(2.0 * n * conv.max())
