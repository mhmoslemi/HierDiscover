"""
Optimisation of step‑function heights using an expanded family of analytic
shapes, random re‑ordering of the entries, and a deterministic mass‑transfer
local search that directly targets the entries most responsible for the
current maximal convolution bin.  The algorithm respects the 1000 s budget
and returns the best sequence found (as a plain ``list[float]``).
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Utility functions (all top‑level – required by the rules)
# ----------------------------------------------------------------------


def _normalise_seq(a: np.ndarray) -> np.ndarray:
    """
    Scale a sequence without changing the evaluation score.
    The scaling forces the maximum entry below 1.0 and guarantees that
    the total sum is at least 0.01 (otherwise ``evaluate_sequence`` returns
    ``inf``).  The function works in-place on a copy of *a*.
    """
    a = np.array(a, dtype=float, copy=True)

    # 1) bring the maximum under 0.9 (far below the 1000‑clip limit)
    mx = a.max()
    if mx > 0.0:
        a *= 0.9 / mx

    # 2) make sure the total sum is not too tiny
    s = a.sum()
    if s < 0.01:
        if s > 0.0:
            a *= 0.01 / s
        else:                      # all zeros – give every entry a tiny amount
            a += 0.01 / a.size

    return a


def _interleave_high_low(v: np.ndarray) -> np.ndarray:
    """
    Produce an ordering that alternates large and small values:
    sort ``v`` descending and then place the first half on even positions
    and the remainder on odd positions.  Useful as a cheap heuristic for
    reducing the autocorrelation peak.
    """
    s = np.sort(v)[::-1]           # descending
    out = np.empty_like(v)
    half = (len(v) + 1) // 2
    out[0::2][:half] = s[:half]
    out[1::2][:len(v) - half] = s[half:]
    return out

    The returned arrays are *unscaled*; they will be normalised later.
    """
    shapes = []

    # 1) constant
    shapes.append(np.ones(n, dtype=float))

    # 2) linear ramps
    shapes.append(np.arange(n, 0, -1, dtype=float))
    shapes.append(np.arange(1, n + 1, dtype=float))

    # 3) quadratic ramps
    shapes.append(np.square(np.arange(n, 0, -1, dtype=float)))
    shapes.append(np.square(np.arange(1, n + 1, dtype=float)))

    # 4) triangular “mountain”
    t = np.arange(1, n + 1, dtype=float)
    shapes.append(np.minimum(t, n - t + 1))

    # 5) valley – large at the ends, small in the centre (inverse quadratic)
    centre = (n - 1) / 2.0
    shapes.append(1.0 / (1.0 + ((np.arange(n, dtype=float) - centre) ** 2)))

    # 6) sine‑squared (smooth hump)
    shapes.append(np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2)

    # 7) geometric decays (different bases)
    for r in np.linspace(0.5, 0.99, 9):
        shapes.append(np.power(r, np.arange(n, dtype=float)))

    # 8) power‑law decays
    for p in np.linspace(0.2, 3.0, 7):
        i = np.arange(1, n + 1, dtype=float)
        shapes.append(1.0 / (i ** p))

    # 9) Gaussians of various widths
    for sigma in (n / 8.0, n / 6.0, n / 4.0):
        x = np.arange(n, dtype=float)
        g = np.exp(-((x - n / 2.0) ** 2) / (2.0 * sigma ** 2))
        shapes.append(g)

    # 10) exponential valleys (large at both ends, small in the middle)
    for r in np.linspace(0.5, 0.99, 5):
        left = r ** np.arange(n // 2, dtype=float)
        right = left[::-1]
        if n % 2 == 0:
            shapes.append(np.concatenate([left, right]))
        else:
            shapes.append(np.concatenate([left, [0.5], right]))

    # 11) power‑law valleys (large at ends, small at centre)
    for p in np.linspace(0.5, 3.0, 5):
        left = (np.arange(1, n // 2 + 1, dtype=float)) ** p
        right = left[::-1]
        if n % 2 == 0:
            shapes.append(np.concatenate([left, right]))
        else:
            shapes.append(np.concatenate([left, [0.0], right]))

    return shapes


def _max_bin_contributions(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """
    For every index ``i`` compute how much ``a[i]`` contributes to the
    convolution bins that currently attain ``max_val``.
    The contribution for a particular bin ``k`` is ``a[k-i]`` (when valid).
    The function returns a vector ``c`` where ``c[i] = Σ_{k∈max} a[k-i]``.
    """
    n = len(a)
    contrib = np.zeros(n, dtype=float)

    # Small tolerance to capture a few neighbouring bins that are numerically equal.
    tol = max_val * 1e-12
    max_bins = np.where(conv >= max_val - tol)[0]

    for k in max_bins:
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        i_vals = np.arange(start, end + 1)
        j_vals = k - i_vals               # the partner index in the pair (i, j)
        contrib[i_vals] += a[j_vals]

    return contrib


def _mass_transfer_move(a: np.ndarray, conv: np.ndarray, max_val: float,
                       eps_factor: float = 0.02) -> np.ndarray:
    """
    Perform a deterministic “mass‑transfer” step: move a small amount of mass
    from the entry that contributes the most to the offending bin(s) to the
    entry that contributes the least.  ``eps_factor`` is a relative step size.
    """
    contrib = _max_bin_contributions(a, conv, max_val)
    i_max = int(np.argmax(contrib))
    i_min = int(np.argmin(contrib))

    # Amount to transfer – never larger than what is available at the source.
    delta = a[i_max] * eps_factor
    delta = min(delta, a[i_max])          # guarantee non‑negativity

    if delta <= 0.0:
        # fallback to a tiny random jitter if nothing can be moved
        new = a.copy()
        idx = np.random.randint(len(a))
        jitter = np.max(a) * 1e-6
        new[idx] = max(0.0, new[idx] + jitter)
        return new

    new = a.copy()
    new[i_max] -= delta
    new[i_min] += delta
    return new

def propose_candidate(seed: int = 123, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative coefficient list that minimises
    ``evaluate_sequence``.  The function respects the given time budget
    (default 1000 s) and returns the best sequence found so far.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1        # safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from the previously discovered construction, if any
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = _normalise_seq(np.asarray(hs1, dtype=float))
        scr = evaluate_sequence(seq.tolist())
        if scr < best_score:
            best_score, best_seq = scr, seq.tolist()
            print(f"[{time.time() - start_time:.1f}s] Warm‑start from height_sequence_1 "
                  f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – systematic scan of parametric families + cheap permutations
    # ------------------------------------------------------------------
    lengths = [64, 128, 256, 512, 1024, 2048, 4096, 8192, 16384]
    for n in lengths:
        if time.time() >= deadline:
            break
        for shape in _candidate_shapes(n):
            if time.time() >= deadline:
                break

            # plain shape
            seq = _normalise_seq(shape)
            scr = evaluate_sequence(seq.tolist())
            if scr < best_score:
                best_score, best_seq = scr, seq.tolist()
                print(f"[{time.time() - start_time:.1f}s] New best ({n} terms) score={best_score:.6f}")

            # interleaved high‑low ordering – often reduces the convolution peak
            seq2 = _normalise_seq(_interleave_high_low(shape))
            scr2 = evaluate_sequence(seq2.tolist())
            if scr2 < best_score:
                best_score, best_seq = scr2, seq2.tolist()
                print(f"[{time.time() - start_time:.1f}s] New best (interleaved, {n}) score={best_score:.6f}")

            # a few random permutations of the same multiset
            for _ in range(2):
                perm = rng.permutation(shape.shape[0])
                seq3 = _normalise_seq(shape[perm])
                scr3 = evaluate_sequence(seq3.tolist())
                if scr3 < best_score:
                    best_score, best_seq = scr3, seq3.tolist()
                    print(f"[{time.time() - start_time:.1f}s] New best (shuffle, {n}) score={best_score:.6f}")

            # early exit if we already pass the target
            if best_score <= 1.503:
                break

    # ------------------------------------------------------------------
    # Safeguard: fall back to a random vector if nothing was accepted
    # ------------------------------------------------------------------
    if best_seq is None:
        cur = rng.random(1024)
        cur = _normalise_seq(cur)
        best_seq = cur.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"[{time.time() - start_time:.1f}s] Fallback random initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deterministic mass‑transfer + stochastic annealing
    # ------------------------------------------------------------------
    cur = np.asarray(best_seq, dtype=float)

    # Linear temperature schedule (1 → 0.001)
    T_start, T_end = 1.0, 0.001

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = T_start * (1.0 - elapsed / total) + T_end * (elapsed / total)

        conv = np.convolve(cur, cur)
        max_val = conv.max()

        # ----- deterministic mass‑transfer move -----
        if rng.random() < 0.6:
            cand = _mass_transfer_move(cur, conv, max_val,
                                       eps_factor=0.02 * (1.0 - elapsed / total))
        else:
            # ----- stochastic perturbation of a random subset -----
            m = max(1, int(0.015 * len(cur)))          # 1.5 % of entries
            idx = rng.choice(len(cur), size=m, replace=False)
            delta = rng.standard_normal(m) * np.mean(cur) * 0.05
            cand = cur.copy()
            cand[idx] += delta
            cand = np.clip(cand, 0.0, None)

        # Ensure we stay away from the “sum too small” guard.
        if cand.sum() < 0.01:
            cand += (0.01 - cand.sum()) / cand.size

        cand_score = evaluate_sequence(cand.tolist())

        # ----- acceptance logic -----
        if cand_score < best_score:
            best_score, best_seq = cand_score, cand.tolist()
            cur = cand
            print(f"[{time.time() - start_time:.1f}s] Improved score={best_score:.6f}")
            if best_score <= 1.503:
                print("Target reached – terminating search.")
                break
        else:
            # Simulated‑annealing acceptance of uphill moves
            prob = np.exp(-(cand_score - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur = cand

        # Occasionally inject a block‑reversal move (helps escape local minima)
        if rng.random() < 0.05:
            start = rng.integers(0, len(cur))
            length = rng.integers(2, min(200, len(cur) - start) + 1)
            rev = cur.copy()
            rev[start:start + length] = rev[start:start + length][::-1]
            rev_score = evaluate_sequence(rev.tolist())
            if rev_score < best_score:
                best_score, best_seq = rev_score, rev.tolist()
                cur = rev
                print(f"[{time.time() - start_time:.1f}s] Block‑reverse improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    return best_seq
