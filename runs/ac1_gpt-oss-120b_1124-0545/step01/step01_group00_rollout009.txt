def project_simplex(v):
    # Ensures by sorting descending etc.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The evaluation is scale‑invariant, so we work with a simplex
∑a_i = 1.

The algorithm proceeds in three stages:

1. **Good analytic start** – we try a few analytically motivated
   families (inverse‑sqrt “arcsine”, mixed inverse‑sqrt powers,
   uniform) and keep the best one.

2. **Projected gradient descent on the L∞‑peak** – the current maximal
   convolution entry `k_max` determines a sub‑gradient
   g_i = 2·a_{k_max‑i} (zero outside the valid range).  A step
   `a ← a – η·g` is taken and the result is projected back onto the
   probability simplex (non‑negative, sum = 1).  A simple
   Metropolis‐style acceptance rule with a decreasing temperature
   allows occasional uphill moves.

3. **Exploration moves** – with a small probability we perform a cheap
   random mass transfer, which helps to escape tiny local minima.

The whole routine respects the 1000 s budget, prints progress, and
returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level)
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective (scale‑invariant) for a non‑negative array a."""
    s = a.sum()
    if s < 1e-12:                 # defensive, should never happen
        return np.inf
    b = np.convolve(a, a)        # linear (zero‑padded) convolution
    max_b = b.max()
    return float(2.0 * len(a) * max_b / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                # sorted descending
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / (np.arange(n) + 1.0)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:                    # all entries are zero?
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from one random index to another."""
    n = a.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    max_transfer = a[i]
    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # Project back onto simplex (takes care of any rounding error)
    return project_simplex(a_new)


def gradient_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    One step of projected sub‑gradient descent w.r.t. the current
    maximal convolution entry.
    """
    n = a.shape[0]
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    # Build sub‑gradient g_i = 2 * a_{k_max‑i} (if the index exists)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)
    g = np.zeros_like(a)
    # The counterpart indices are k_max - idx
    g[idx] = 2.0 * a[k_max - idx]

    # Take a step and project onto the simplex
    a_new = a - step_scale * g
    a_new = np.clip(a_new, 0.0, None)
    a_new = project_simplex(a_new)
    return a_new


def initialise_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a short list of analytically motivated candidates.
    Returns a list of normalized (sum=1) arrays.
    """
    candidates = []

    # 1. Uniform
    candidates.append(np.ones(n) / n)

    # 2. Inverse‑sqrt product (arcsine‑like)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    candidates.append(a / a.sum())

    # 3. Mixed inverse‑sqrt sum with several exponents
    for p in (0.3, 0.5, 0.7, 1.0):
        a = (i + 1) ** (-p) + (n - i) ** (-p)
        candidates.append(a / a.sum())

    # 4. Pure inverse‑sqrt product without the 0.5 offset
    a = 1.0 / np.sqrt((i + 1) * (n - i))
    candidates.append(a / a.sum())

    return candidates


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function of length ``n`` (default 2000) whose
    evaluation value is as low as possible.  The routine starts from an
    analytically good shape, then performs projected gradient descent
    with occasional random perturbations.  The best sequence found
    before the time budget expires is returned.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a tiny safety margin
    deadline = start + budget_s - 0.5

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # 1 – choose a strong initial point from a small analytic pool
    # ------------------------------------------------------------------
    cand_pool = initialise_candidates(n)
    best_seq = None
    best_score = np.inf
    for idx, a in enumerate(cand_pool):
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
        print(f"[init] candidate {idx:02d} score = {sc:.6f}")

    a = best_seq.copy()
    print(f"[init] chosen start score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2 – optimisation loop (projected sub‑gradient + occasional random)
    # ------------------------------------------------------------------
    # annealing / step‑size parameters
    init_temp = 1.0
    final_temp = 0.001
    init_step = 0.5          # relatively large at the beginning
    final_step = 1e-4

    iteration = 0
    last_improve = start

    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / (budget_s - 10.0)   # in [0,1] (smooth)

        # linear schedules
        temp = init_temp * (1.0 - progress) + final_temp * progress
        step_scale = init_step * (1.0 - progress) + final_step * progress

        # decide move type
        if np.random.rand() < 0.85:          # mainly gradient steps
            cand = gradient_move(a, step_scale)
        else:                                # occasional random mass swap
            cand = random_move(a, step_scale * 0.5)

        cand_score = compute_score(cand)

        # Metropolis acceptance (lower is better)
        if cand_score < best_score or np.random.rand() < np.exp(-(cand_score - best_score) / max(temp, 1e-12)):
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:06d}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # restart if stuck for a while
        if iteration % 20000 == 0 and (time.time() - last_improve) > 30.0:
            # re‑initialise around the current best with a small random jitter
            jitter = np.random.rand(n) * 1e-3
            a = best_seq + jitter
            a = project_simplex(a)
            print(f"[{iteration:06d}] restart (stagnation)")

        iteration += 1

    total_time = time.time() - start
    print(f"[done] iterations={iteration}, best score={best_score:.6f}, time={total_time:.1f}s")
    return best_seq.tolist()
