def build_conv_matrix(f):
    n = len(f)
    rows = 2*n-1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]
    return A

    import time, numpy as np, cvxpy as cp

    start = time.time()
    deadline = start + budget_s - 5

    n = 800 (or maybe 600)
    # initial f: binomial normalized to sum sqrt(2n)
    f = binomial_sequence(n)
    f = f * (np.sqrt(2*n) / np.sum(f))

    best_f = f.copy()
    best_score = fast_score(f)

    # compute matrix A for f (needs to update each iteration because f changes)
    # We'll recompute each iteration.

    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        iteration += 1

        conv_f_f = np.convolve(f, f)
        M = np.max(conv_f_f)

        # Build matrix A for constraints conv(f,b) <= M
        A = np.zeros((2*n-1, n))
        for i in range(n):
            A[i:i+n, i] = f[i]

        b_var = cp.Variable(n, nonneg=True)
        constraints = [A @ b_var <= M]
        obj = cp.Maximize(cp.sum(b_var))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver='ECOS', warm_start=True, verbose=False)

        if b_var.value is None:
            print("[LP] iteration failed to solve")
            break
        b_opt = np.array(b_var.value).flatten()
        if np.isnan(b_opt).any():
            print("[LP] solution contains NaN")
            break

        sum_b = np.sum(b_opt)
        if sum_b <= 1e-12:
            print("[LP] sum_b approx zero")
            break

        # scale b to match normalization sum sqrt(2n)
        scale = np.sqrt(2*n) / sum_b
        g = b_opt * scale

        # convex combine f and g
        t = 0.5  # mixing factor; could be adaptive
        new_f = (1 - t) * f + t * g

        new_score = fast_score(new_f)

        print(f"[final] official={official:.6f}")

        print("[final] error", e)

"""
Iterative linear-programming + local search for minimizing the evaluation 
function.  Starting from the truncated binomial (which attains the 
asymptotic bound π/2) we solve a small LP that maximises the L1‑norm 
subject to the current ℓ∞‑convolution bound.  The solution is rescaled 
to the required normalisation and convex‑combined with the current 
candidate.  Repeating this step rapidly decreases the maximal 
convolution value.  Finally a short stochastic hill‑climb is performed 
to fine‑tune the result.
"""

import time
import numpy as np
import cvxpy as cp

def binomial_sequence(length: int) -> np.ndarray:
    """Return a_i = C(2*i,i) / 4**i for i=0..length‑1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i-1] * (2*i-1) / (2.0*i)
    return a

def fast_score(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no safety checks)."""
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[:2 * n - 1]
    max_b = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Construct the (2n‑1)×n matrix A with A[k,i] = f[i] if i≤k≤i+n‑1."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f[i]
    return A

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    reference π/2 bound.  The algorithm consists of:

    1. Initialise a normalised binomial sequence.
    2. Repeatedly:
         * compute the current ℓ∞ convolution bound M,
         * solve a linear programme   max Σ b   subject to   (f⋆b) ≤ M,
         * rescale the LP solution to the required ℓ₁ norm,
         * convex‑combine it with the current candidate.
       This monotonically decreases M.
    3. A final light stochastic hill‑climb for polishing.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ---- 1) start from the truncated binomial, normalised to ∥f∥₁ = √(2n) ----
    n = 800                     # length – chosen to keep the LP cheap yet effective
    f = binomial_sequence(n)
    target_norm = np.sqrt(2.0 * n)
    f = f * (target_norm / f.sum())   # now ∑f = √(2n)
    best_seq = f.copy()
    best_val = fast_score(best_seq)
    print(f"[init] n={n}, fast_score={best_val:.6f}")

    # ---- 2) LP‑based improvement loop ---------------------------------------
    max_iters = 30           # LP solves are relatively expensive, keep it modest
    t_mix = 0.5              # convex‑combination weight
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv_ff = np.convolve(f, f)
        M = float(conv_ff.max())
        # build linear‑program matrix
        A = build_conv_matrix(f)
        # LP: maximise sum(b)  s.t. A @ b ≤ M,  b ≥ 0
        b = cp.Variable(n, nonneg=True)
        constraints = [A @ b <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        # ECOS is a lightweight interior‑point solver available in cvxpy
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)

        if b.value is None:
            print(f"[lp] iteration {it}: solver failed, status={prob.status}")
            break
        b_opt = np.array(b.value).ravel()
        # Safety: clip tiny negatives caused by numeric noise
        b_opt = np.maximum(b_opt, 0.0)

        sum_b = b_opt.sum()
        if sum_b <= 1e-12:
            print(f"[lp] iteration {it}: degenerate solution (sum≈0)")
            break

        # rescale to the required ℓ₁ norm
        scale = target_norm / sum_b
        g = b_opt * scale

        # convex combination
        new_f = (1.0 - t_mix) * f + t_mix * g

        new_val = fast_score(new_f)
        print(f"[lp] iter={it}, M={M:.6f}, sum_b={sum_b:.4f}, new_score={new_val:.6f}")

        # keep the best ever seen
        if new_val < best_val:
            best_val = new_val
            best_seq = new_f.copy()
            print(f"[lp]   ** new best {best_val:.6f}")

        # update current candidate for next round
        f = new_f

    # ---- 3) final stochastic fine‑tuning ------------------------------------
    attempts = 0
    max_attempts = 100_000
    while time.time() < deadline and attempts < max_attempts:
        attempts += 1
        # random move: shift a small fraction of mass
        i = np.random.randint(n)
        if best_seq[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = 0.02 * best_seq[i]          # 2 % of the source entry
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = fast_score(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq
            print(f"[hill] iter={attempts}, improved to {best_val:.6f}")

    # ---- 4) final official evaluation ---------------------------------------
    final_list = best_seq.tolist()
    try:
        official = evaluate_sequence(final_list)
        print(f"[final] official evaluate_sequence = {official:.6f}")
    except Exception as exc:   # pragma: no cover – defensive
        print("[final] evaluation failed:", exc)

    return final_list

"""
Iterative LP + hill‑climb optimizer.

We start from the truncated binomial sequence (which gives the asymptotic
π/2 bound).  The sequence is normalised so that its ℓ₁‑norm equals
√(2 n); for such a normalisation the evaluation score is exactly the
maximum value of the self‑convolution.

At each iteration we solve a linear programme

    maximise   Σ b_j
    subject to  (f ⋆ b)_k ≤ M   for all k
                b_j ≥ 0

where M = max(f ⋆ f) is the current ℓ∞‑convolution bound.
The LP finds a distribution b with a larger ℓ₁‑norm while keeping the
same convolution bound.  After rescaling b to the required ℓ₁‑norm we
convex‑combine it with the current f; the combination strictly reduces
M.  Repeating the step rapidly drives the score below the π/2 value.

When the time budget is nearly exhausted we switch to a tiny stochastic
hill‑climb that moves a small fraction of mass from one index to another.
The best sequence seen is returned and finally checked with the official
`evaluate_sequence` routine.
"""

import time
import numpy as np
import cvxpy as cp

def binomial_sequence(length: int) -> np.ndarray:
    """Return a_i = C(2*i,i) / 4**i  for i = 0,…,length‑1."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a

def fast_score(seq: np.ndarray) -> float:
    """Fast counterpart of evaluate_sequence – uses FFT, no checks."""
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())

    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    max_b = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 1e-12:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

    return A

    # ----------------------------------------------------------------------
    # 2) LP‑based improvement loop (monotonically reduces max convolution)
    # ----------------------------------------------------------------------
    max_iters = 30               # LP solves are the costly part
    t_mix = 0.5                  # convex‑combination weight
