def _conv_fft(a):
    n = len(a)
    N = 1 << ((2*n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=N)
    # elementwise product
    conv = np.fft.irfft(fft_a * fft_a, n=N)
    # conv length N, but only first 2*n-1 relevant
    return conv[:2*n-1]

def _gradient_at_k(a, k):
    n = len(a)
    g = np.zeros_like(a)
    # valid i range: i from max(0, k-(n-1)) to min(k, n-1)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)    # inclusive
    # For i in [i_start, i_end], j = k - i
    # if 0 <= j < n
    # assign g[i] = 2 * a[j]
    # vectorized:
    i_range = np.arange(i_start, i_end+1)
    j = k - i_range
    g[i_range] = 2.0 * a[j]
    return g

            L(a) = max_k (a∗a)_k .

            ∂L/∂a_i = 2·a_{k*‑i}  (if k*‑i is a valid index, otherwise 0).

    Random block reversals are occasionally injected (1 % chance) to

found at any moment (its exact score computed by `evaluate_sequence`) is

The algorithm discovers a near‑geometric decay with c≈2, giving a score

def _normalise_seq(a: np.ndarray) -> np.ndarray:
    """
    Scale the array so that its maximum is 0.9 (well below the 1000 cap)
    and guarantee a total sum of at least 0.01 (otherwise evaluate_sequence
    returns ∞).  The scaling does not affect the final score because the
    evaluation is homogeneous.
    """
    a = np.asarray(a, dtype=float)
    max_v = a.max()
    if max_v > 0.0:
        a = a * (0.9 / max_v)          # keep values < 1
    total = a.sum()
    if total < 0.01:                   # avoid the “sum too close to zero” guard
        if total > 0.0:
            a = a * (0.01 / total)
        else:
            # all zeros – give a tiny uniform bump
            a = a + (0.01 / a.size)
    return a


def _candidate_shapes_ext(n: int):
    """
    Generate a list of candidate shape vectors of length `n`.
    The returned arrays are raw (un‑normalised) float ndarrays; they are
    normalised later before evaluation.
    """
    cand = []

    # 1. constant
    cand.append(np.ones(n, dtype=float))

    # 2. linear (decreasing / increasing)
    cand.append(np.arange(n, 0, -1, dtype=float))
    cand.append(np.arange(1, n + 1, dtype=float))

    # 3. quadratic (decreasing / increasing)
    cand.append(np.square(np.arange(n, 0, -1, dtype=float)))
    cand.append(np.square(np.arange(1, n + 1, dtype=float)))

    # 4. triangular “mountain”
    t = np.arange(1, n + 1, dtype=float)
    cand.append(np.minimum(t, n - t + 1))

    # 5. valley (large at the ends, small in the centre)
    centre = (n - 1) / 2.0
    cand.append(1.0 / (1.0 + ((np.arange(n, dtype=float) - centre) ** 2)))

    # 6. sine‑squared shape
    cand.append(np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2)

    # 7. classical exponential decays (already used before)
    for r in (0.95, 0.9, 0.8, 0.7, 0.6, 0.5):
        cand.append(np.power(r, np.arange(n, dtype=float)))

    # 8. **new** near‑unit exponential decays  a_i = exp(-c·i / n)
    for c in (0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0):
        r = np.exp(-c / n)
        cand.append(np.power(r, np.arange(n, dtype=float)))

    # 9. power‑law decays
    for p in (0.5, 1.0, 1.5, 2.0):
        i = np.arange(1, n + 1, dtype=float)
        cand.append(1.0 / (i ** p))

    # 10. beta‑like shapes  a_i = (i+1)^α·(n-i)^β
    for alpha in (0.5, 1.0, 2.0):
        for beta in (0.5, 1.0, 2.0):
            i = np.arange(1, n + 1, dtype=float)
            cand.append((i ** alpha) * ((n - i + 1) ** beta))

    return cand


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full (linear) convolution a∗a using an FFT of length
    a power‑of‑two.  The returning array has length 2·len(a)‑1.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())   # smallest power‑of‑two ≥ 2n‑1
    fft_a = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fft_a * fft_a, n=N)
    return conv[: 2 * n - 1]


def _gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the objective L(a)=max_k (a∗a)_k at the active index k.
    It returns the vector g with entries
        g_i = 2·a_{k‑i}  if 0 ≤ k‑i < n,
        g_i = 0        otherwise.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    i = np.arange(lo, hi + 1, dtype=int)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _subgradient_optimize(start_seq: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator,
                          target_score: float = 1.48) -> np.ndarray:
    """
    Perform sub‑gradient descent on the max‑convolution objective.
    The input `start_seq` must already be normalised (non‑negative,
    Σa_i = 1).  The routine returns the best sequence found before the
    deadline.
    """
    a = start_seq.copy()
    n = a.shape[0]

    best_a = a.copy()
    # initial exact score (using the official evaluator)
    best_score = evaluate_sequence(best_a.tolist())

    # step‑size schedule
    start_eta, end_eta = 0.5, 0.001
    iteration = 0

    while time.time() < deadline:
        # ---- evaluate current point ------------------------------------
        conv = _conv_fft(a)
        b_max = conv.max()
        cur_score = 2.0 * n * b_max               # sum(a)=1 ⇒ exact score
        # keep the exact evaluator's result for the best ever seen
        exact = evaluate_sequence(a.tolist())
        if exact < best_score:
            best_score = exact
            best_a = a.copy()
            print(f"[{time.time() - (deadline - 1000):.1f}s] "
                  f"New global best score={best_score:.6f}")

            if best_score <= target_score:
                print("Target achieved – terminating sub‑gradient phase.")
                break

        # ---- sub‑gradient step ----------------------------------------
        k_star = int(conv.argmax())
        g = _gradient_at_max(a, k_star)
        g_max = g.max()
        if g_max == 0.0:          # unlikely, but protect against division by zero
            break

        # time‑dependent step size
        elapsed = time.time() - (deadline - 1000)
        total = deadline - (deadline - 1000)
        eta = start_eta * (1.0 - elapsed / total) + end_eta * (elapsed / total)

        # multiplicative (exponentiated‑gradient) update
        factor = np.exp(-eta * g / g_max)   # g/g_max ∈ [0,1]
        a_new = a * factor
        a_new_sum = a_new.sum()
        if a_new_sum == 0.0:      # protect against degenerate zero vector
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum   # simplex projection (L1‑normalisation)

        # occasional random block reversal to escape plateaus
        if rng.random() < 0.01:
            start_idx = rng.integers(0, n)
            block_len = rng.integers(1, min(256, n - start_idx) + 1)
            a_new[start_idx:start_idx + block_len] = \
                a_new[start_idx:start_idx + block_len][::-1]

        a = a_new
        iteration += 1

    # Return the best sequence discovered (as a plain Python list)
    return best_a.tolist()

            break
        print(f"[{time.time() - start_time:.1f}s] Exploring length {n}")
        for raw in _candidate_shapes_ext(n):
            if time.time() >= deadline:
                break
            seq = _normalise_seq(raw)
            # quick homogeneous score (no need to call the full evaluator yet)
            conv = _conv_fft(seq)
            approx_score = 2.0 * n * conv.max()
            # only call the expensive evaluator if the approximation looks promising
            if approx_score < best_score * 1.05:
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = seq.tolist()
                    print(f"[{time.time() - start_time:.1f}s] "
                          f"New best (len={n}) score={best_score:.6f}")
                if best_score <= 1.48:   # early exit if already very good
                    break

    # ------------------------------------------------------------------
    # Guard: if nothing was accepted, start from a random vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 8192
        best_seq = rng.random(n).tolist()
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient refinement on the best length
    # ------------------------------------------------------------------
    cur = np.asarray(best_seq, dtype=float)
    # re‑normalise to sum = 1 (the scaling does not affect the score)
    cur = cur / cur.sum()

    refined = _subgradient_optimize(cur, deadline, rng)

    # final verification – keep the absolute best
    final_score = evaluate_sequence(refined)
    if final_score < best_score:
        best_score = final_score
        best_seq = refined

    print(f"[{time.time() - start_time:.1f}s] Finished – best score={best_score:.6f}")
    # Ensure the returned list obeys the API guarantees (non‑negative floats)
    return [float(x) for x in best_seq]
