"""
Propose a step‑function (non‑negative vector) that minimises
    max(a * a)   (with sum(a)=sqrt(2*n))

The algorithm is a smooth‑max gradient descent on the simplex.

Key ideas
---------
* Keep the ℓ₁–norm fixed to sqrt(2*n) (the usual normalisation).
* Replace the non‑smooth objective max(conv) by a smooth
  approximation   M_α(a)= (1/α)·log Σ_k exp(α·conv_k).
  For α→∞ the smooth objective tends to the true max.
* Compute the gradient of M_α analytically:
        ∂M/∂a = 2·(w * a_rev)[n‑1 : n‑1+n] ,
  where w are the soft‑max weights of conv and a_rev = a[::-1].
* Use an Adam optimiser with a linearly decaying learning rate.
* After each update project the vector onto the probability simplex
  {x≥0, Σx = sqrt(2·n)}.
* Occasionally (probability 0.02) restart from a fresh random point
  to escape possible local minima.
* The current best vector (and its max‑convolution value) is kept
  and returned after the time budget is exhausted.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {w ≥ 0, Σw = s}."""
    n = v.size
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, s / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray, nfft: int) -> np.ndarray:
    """Linear convolution via FFT (real input, real output)."""
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    z = np.fft.irfft(X * Y, n=nfft)
    return z[: x.size + y.size - 1]

    deadline = start_time + budget_s - 5.0   # keep a safety margin
    iteration = 0
    last_improve_iter = 0

    # ------------------------------------------------------------------
    while time.time() < deadline:
        now = time.time()
        elapsed = now - start_time
        frac = elapsed / (deadline - start_time)

        # ---- annealing schedule for the smooth‑max temperature ----
        alpha = start_alpha * ((end_alpha / start_alpha) ** frac)

        # ---- forward convolution a*a --------------------------------
        conv = _conv_fft(a, a, nfft_conv)            # length 2n‑1
        conv_max = np.max(conv)

        # ---- soft‑max weights (stable) -----------------------------
        #   w_i = exp(alpha·(conv_i - conv_max))   --> max weight = 1
        w = np.exp(alpha * (conv - conv_max))
        w_sum = np.sum(w)
        if w_sum == 0:
            # numerical safety – fall back to uniform weights
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth‑max ----------------------------
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev, nfft_grad)   # length = 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # slice of length n

        # ---- Adam update -------------------------------------------
        t += 1
        lr = base_lr * (1.0 - frac) + 1e-6           # decay to a tiny step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

        a_new = a - step
        # project back onto the simplex (keeps sum exact)
        a_new = _simplex_projection(a_new, target_sum)

        # ---- occasional random restart to improve exploration -------
        if np.random.rand() < 0.02:
            a_new = np.random.exponential(scale=1.0, size=n)
            a_new = a_new / a_new.sum() * target_sum
            a_new = _simplex_projection(a_new, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

        # ---- evaluate candidate (using our own max‑conv) ------------
        cur_score = float(np.max(_conv_fft(a_new, a_new, nfft_conv)))

        # ---- keep the best found so far -----------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            last_improve_iter = iteration
            print(f"[{iteration:6d}] new best = {best_score:.6f} (t={elapsed:.1f}s)")

        # ---- accept the new point for the next iteration -----------
        a = a_new
        iteration += 1

        # ---- simple restart if we are stuck too long ---------------
        if iteration - last_improve_iter > 5000 and frac > 0.3:
            # re‑initialise from a fresh random point
            a = np.random.exponential(scale=1.0, size=n)
            a = a / a.sum() * target_sum
            a = _simplex_projection(a, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve_iter = iteration

    # ------------------------------------------------------------------
    # Final sanity check – make sure the returned object is a plain list
    print(f"[final] best score = {best_score:.6f} after {iteration} iterations")
    return best_a.tolist()
