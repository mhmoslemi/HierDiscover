"""
Propose a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the linear convolution.  The algorithm is a hybrid
exponentiated‑gradient / Adam optimiser that works directly on the
probability‑simplex (the sequence is normalised to sum to 1).  It
combines three ideas that improve over the reference implementation:

1. **Exact sub‑gradient after a warm‑up phase** – we start with a
   smoothed max (log‑sum‑exp) to obtain a good descent direction, then
   switch to the true sub‑gradient of the max‑convolution (a one‑hot
   weight vector).  This avoids the bias of a permanently softened
   objective.

2. **Adam‑style adaptive steps in the log‑domain** – we maintain first
   and second moment estimates of the gradient and update the vector
   multiplicatively (mirror‑descent w.r.t. KL).  The adaptive step size
   reaches higher precision and escapes shallow local minima more
   reliably than the fixed‑step scheme of the reference code.

3. **Multiple restarts + occasional jitter** – we run a few short
   optimisations from diverse initial points (the previous best,
   uniform, and random Dirichlet draws).  Every 1000 iterations a tiny
   random perturbation is added to the current iterate to encourage
   exploration.

All operations are vectorised and use FFT‑based convolution
(`_conv_fft`) which is very fast for the lengths we consider (≈ 4000).
The procedure respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions ------------------------------------------------------
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using the FFT (full length).

    Parameters
    ----------
    x, y : np.ndarray
        Input 1‑D arrays.

    Returns
    -------
    np.ndarray
        Full convolution of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _evaluate_simplex(a: np.ndarray) -> float:
    """
    Evaluation under the assumption sum(a) == 1.
    Returns 2 * n * max(conv(a, a)).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     cur_max: float, beta: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp smoothed max.
    Returns full gradient (length n) for the update a ← a * exp(-step·grad).
    """
    w = np.exp(beta * (conv - cur_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.zeros_like(a)
    w /= w_sum
    grad_full = _conv_fft(a[::-1], w)
    grad = 2.0 * grad_full[len(a) - 1:len(a) - 1 + len(a)]
    return grad


def _exact_max_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of the true max‑convolution (sub‑gradient using a one‑hot weight).
    """
    idx = int(np.argmax(conv))
    w = np.zeros_like(conv)
    w[idx] = 1.0
    grad_full = _conv_fft(a[::-1], w)
    grad = 2.0 * grad_full[len(a) - 1:len(a) - 1 + len(a)]
    return grad


def _smoothness_gradient(a: np.ndarray, mu: float) -> np.ndarray:
    """
    Gradient of μ·∑(a_{i+1}−a_i)^2, encouraging a smooth shape.
    """
    if mu == 0.0:
        return np.zeros_like(a)
    n = len(a)
    lap = np.empty_like(a)
    # interior points
    lap[1:-1] = 2.0 * a[1:-1] - a[:-2] - a[2:]
    # boundaries
    lap[0] = a[0] - a[1]
    lap[-1] = a[-1] - a[-2]
    return mu * lap


# ----------------------------------------------------------------------
# Core optimiser -------------------------------------------------------
# ----------------------------------------------------------------------


def _optimise_one(init_a: np.ndarray,
                  n: int,
                  deadline: float,
                  best_val_ref: float,
                  best_seq_ref: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Perform Adam‑style exponentiated‑gradient optimisation starting from
    ``init_a`` (already normalised to sum = 1).  Returns the best value
    and corresponding sequence found during this run.
    """
    # Adam state
    m = np.zeros_like(init_a)
    v = np.zeros_like(init_a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5               # will be divided by sqrt(t+1)
    max_step = 5.0               # clip exponent step to keep numbers stable
    warmup_ratio = 0.2           # fraction of iterations using smoothed max
    mu_smooth = 0.01 / n         # strength of the smoothness regulariser

    a = init_a.copy()
    best_val_local = best_val_ref
    best_seq_local = best_seq_ref

    # very rough estimate of how many iterations we can afford:
    # we assume ~300 µs per iteration (FFT+overheads) → about 3000 it/s.
    # The calling code already limits the overall time, so we simply loop
    # until the deadline is reached.
    t = 0
    while True:
        if time.time() > deadline:
            break

        # ------------------------------------------------------------------
        # 1) evaluate current objective
        # ------------------------------------------------------------------
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max   # sum(a) == 1 by construction

        if cur_val < best_val_local:
            best_val_local = cur_val
            best_seq_local = a.copy()
            print(f"[run] iter {t:5d} new best = {cur_val:.6f}")

        # ------------------------------------------------------------------
        # 2) compute gradient (smooth → exact switch)
        # ------------------------------------------------------------------
        warmup_iters = int(warmup_ratio * (t + 1))  # grows slowly
        if t < warmup_iters:
            # smoothed max (log‑sum‑exp)
            beta = 5.0 * (1.02 ** t)
            grad = _smooth_gradient(a, conv, cur_max, beta)
        else:
            # exact sub‑gradient of the max
            grad = _exact_max_gradient(a, conv)

        # add smoothness regulariser
        if mu_smooth > 0.0:
            grad += _smoothness_gradient(a, mu_smooth)

        # ------------------------------------------------------------------
        # 3) Adam update in the log‑domain (exponentiated‑gradient)
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (t + 1))
        v_hat = v / (1.0 - beta2 ** (t + 1))

        lr = base_lr / np.sqrt(t + 1.0)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -max_step, max_step)

        # multiplicative (mirror) update
        a = a * np.exp(-step)

        # re‑normalise to the simplex (sum = 1)
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.ones_like(a) / n
        else:
            a = a / s

        # ------------------------------------------------------------------
        # 4) occasional jitter to escape plateaus
        # ------------------------------------------------------------------
        if t % 1000 == 0 and t > 0:
            jitter = np.random.normal(scale=0.01, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        t += 1

    return best_val_local, best_seq_local

# Public entry point ----------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises the evaluation function.  The routine respects the
    ``budget_s`` time limit, runs several independent restarts (including
    the previously discovered best construction, a uniform start and a
    few random Dirichlet draws) and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # Determine the working length n.
    # Prefer the length of the previously found construction if it exists.
    # ------------------------------------------------------------------
    try:
        init_global = np.asarray(height_sequence_1, dtype=np.float64)
        if init_global.ndim != 1 or init_global.size == 0:
            raise ValueError
        n = init_global.size
    except Exception:
        # fall back to a modest default length
        n = 4000
        init_global = np.ones(n, dtype=np.float64) / n

    # normalise any initial vector to a probability simplex
    init_global = init_global / np.maximum(init_global.sum(), 1e-12)

    # ------------------------------------------------------------------
    # Prepare a collection of diverse starting points.
    # ------------------------------------------------------------------
    starts = [init_global]                                      # 0 – previous best
    starts.append(np.ones(n, dtype=np.float64) / n)            # 1 – uniform
    for _ in range(3):                                         # 2‑4 – random Dirichlet
        starts.append(np.random.dirichlet(np.ones(n)))

    best_val = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Run a short optimisation from each start.
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(starts):
        if time.time() > deadline:
            break
        print(f"[restart {idx}] length={n}")

        # Run the optimiser; it will stop automatically when the deadline is hit.
        val, seq = _optimise_one(a0, n, deadline, best_val, best_seq if best_seq is not None else a0)

        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"[global] new overall best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) / n

    # Return as a plain Python list (evaluate_sequence will clip/normalise again)
    return best_seq.tolist()
