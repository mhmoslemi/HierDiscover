"""
Propose a candidate step‐function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in two stages.

* **Stage 1 – combinatorial exploration**  
  Randomly pick a subset `S` of indices (size `m` roughly 60‑90 % of `n`) and
  assign the weight `1/m` to each chosen index (the rest stay zero).  
  For a random subset the convolution of the resulting binary sequence is
  close to uniform, giving a maximum far below the uniform benchmark (`2.0`).

* **Stage 2 – continuous refinement**  
  Starting from the best discrete candidate, a cheap simulated‑annealing /
  hill‑climbing loop (targeted moves, random moves, a gradient‑like step)
  further reduces the peak.  The loop respects the non‑negative simplex
  constraints (`a_i ≥ 0`, `∑a_i = 1`) and stops when the time budget expires.

The whole search respects the 1000 s budget, prints intermediate progress
and finally returns the best sequence found as a plain list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact evaluation of the objective for a non‑negative vector a (sum ≈ 1)."""
    s = a.sum()
    if s < 1e-12:                     # defensive – should never happen
        return np.inf
    # convolution (linear, not circular)
    b = np.convolve(a, a)
    max_b = b.max()
    # denominator is s**2 ; we keep it for completeness
    return float(2.0 * len(a) * max_b / (s * s))


def random_subset_sequence(n: int, m: int) -> np.ndarray:
    """
    Create a vector of length n with exactly m non‑zero entries,
    each equal to 1/m (hence sum = 1).
    """
    a = np.zeros(n, dtype=float)
    # choose m distinct positions without replacement
    idx = np.random.choice(n, size=m, replace=False)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Move operators for the continuous refinement stage
# ----------------------------------------------------------------------


def random_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """Transfer a tiny amount of mass from a random index to another."""
    n = len(a)
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)

    max_transfer = a[i]
    if max_transfer == 0.0:
        return a.copy()

    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


def targeted_move(a: np.ndarray, step_scale: float) -> np.ndarray:
    """
    Move mass away from indices that contribute most to the current
    maximal convolution entry.
    """
    n = len(a)
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    # contribution w_i = a_i * a_{k_max-i}
    w = np.zeros(n)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1)
    w[idx] = a[idx] * a[k_max - idx]

    if w.sum() == 0.0:
        i = np.random.randint(n)
    else:
        probs = w / w.sum()
        i = np.random.choice(n, p=probs)

    # pick a destination different from i
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1

    max_transfer = a[i]
    if max_transfer == 0.0:
        return a.copy()

    delta = step_scale * max_transfer * np.random.rand()
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


def gradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    Cheap sub‑gradient step that flattens the peak of the convolution.
    The direction d = -g + mean(g) leaves the sum unchanged.
    """
    n = len(a)
    b = np.convolve(a, a)
    k = int(np.argmax(b))

    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]          # ∂b_k/∂a_i = 2·a_{k-i}

    d = -g + g.mean()
    a_new = a + step * d
    a_new = np.clip(a_new, 0.0, None)

    total = a_new.sum()
    if total == 0.0:
        a_new = np.ones_like(a) / n
    else:
        a_new /= total
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value height sequence within ``budget_s`` seconds.
    The routine first explores random binary‑mask candidates, then refines the
    best one with a small simulated‑annealing loop.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    # leave a small safety margin
    deadline = start + budget_s - 0.5

    n = int(kwargs.get("n", 2000))
    if n < 2:
        raise ValueError("Sequence length must be at least 2")

    # ------------------------------------------------------------------
    # Phase 1 – random subset search
    # ------------------------------------------------------------------
    # we try several densities (fraction of non‑zero entries)
    densities = [0.55, 0.65, 0.75, 0.85, 0.95]
    m_candidates = [max(20, int(d * n)) for d in densities]

    best_seq = None
    best_score = np.inf
    iter_cnt = 0

    # allocate roughly 60 % of the total time to the discrete phase
    phase1_end = start + (budget_s - 5) * 0.6

    while time.time() < min(deadline, phase1_end):
        m = int(np.random.choice(m_candidates))
        a = random_subset_sequence(n, m)
        score = compute_score(a)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{iter_cnt:6d}] new best (phase 1) = {best_score:.6f}  m={m}  time={time.time() - start:.1f}s")
        iter_cnt += 1

    # ------------------------------------------------------------------
    # Phase 2 – continuous refinement (simulated annealing)
    # ------------------------------------------------------------------
    # Ensure we have a starting point
    if best_seq is None:
        a = np.random.rand(n)
        a /= a.sum()
        best_seq = a.copy()
        best_score = compute_score(best_seq)

    a = best_seq.copy()
    best_seq = a.copy()
    best_score = compute_score(best_seq)

    iteration = 0
    last_improve = time.time()

    # simple annealing schedule
    init_temp = 1.0
    final_temp = 0.001
    init_step = 0.25
    final_step = 0.001

    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / (budget_s - 5.0)          # in [0,1]

        temp = init_temp * (1.0 - progress) + final_temp * progress
        step_scale = init_step * (1.0 - progress) + final_step * progress

        # pick a move
        r = np.random.rand()
        if r < 0.35:
            cand = targeted_move(a, step_scale)
        elif r < 0.70:
            cand = random_move(a, step_scale)
        else:
            cand = gradient_step(a, step_scale * 0.5)

        cand_score = compute_score(cand)

        # Metropolis acceptance
        if cand_score < best_score or np.random.rand() < np.exp(-(cand_score - best_score) / max(temp, 1e-12)):
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{iteration:6d}] new best (phase 2) = {best_score:.6f}  time={elapsed:.1f}s")

        # occasional restart if we are stuck for a while
        if iteration % 5000 == 0 and (time.time() - last_improve) > 30.0:
            a = np.random.rand(n)
            a /= a.sum()
            print(f"[{iteration:6d}] restart (stagnation)")

        iteration += 1

    total_time = time.time() - start
    print(f"[done] final best score = {best_score:.6f}  iterations={iteration}  time={total_time:.1f}s")
    return best_seq.tolist()
