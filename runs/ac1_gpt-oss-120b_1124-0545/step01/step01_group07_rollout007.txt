def compute_score(a):
    # a as np.ndarray, non-negative
    s = a.sum()
    if s < 1e-12:
        return np.inf, None
    a = a / s  # enforce sum=1
    # compute convolution via FFT

    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2*n-1]
    max_b = c.max()
    score = 2 * n * max_b
    return score, c

"""
Propose candidate step‑function heights.

The algorithm proceeds in three phases:

1. **Parametric sweep** – a collection of analytic families (product‑power,
   sum‑power, Gaussian valley, mixtures) is generated for a fixed length
   (default 8192).  Each shape is normalised to sum = 1 and evaluated.  The
   best shape becomes the warm‑start for the next phase.

2. **Random Dirichlet sampling** – a number of random probability vectors
   (Gamma/Dirichlet draws with various concentration parameters) are drawn,
   normalised and evaluated.  This often discovers structures that are not
   covered by the analytic families.

3. **Smooth‑max gradient descent / simulated‑annealing** – starting from the
   best candidate so far we minimise a smooth approximation of the max‑convolution
   (log‑sum‑exp).  The gradient can be computed efficiently via FFTs.  The
   step‑size and the smoothing parameter β are gradually annealed; worse
   moves are occasionally accepted with a temperature schedule, allowing the
   search to escape shallow local minima.

All operations respect the time budget (default 1000 s).  The returned list
contains non‑negative numbers; the evaluation function is homogeneous, so the
vector is normalised to sum = 1 before the final return (scaling does not affect
the score).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _normalise(a: np.ndarray) -> np.ndarray:
    """
    Project onto the feasible set:
    * non‑negative,
    * sum ≥ 0.01 (the evaluator would otherwise return ∞),
    * then normalise to sum = 1 (the scale does not influence the score).
    """
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        # fallback to uniform distribution
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total
    # protect against too‑small total (should not happen after the division)
    if a.sum() < 0.01:
        a += (0.01 - a.sum()) / a.size
        a = a / a.sum()
    return a


def _compute_score(a: np.ndarray, size_fft: int = None) -> tuple[float, np.ndarray]:
    """
    Compute the exact evaluation score for a given vector `a`.
    Returns (score, convolution_array).  The vector does *not* have to be
    normalised – the function takes care of the denominator.
    """
    n = a.shape[0]
    if size_fft is None:
        size_fft = 1 << ((2 * n - 1).bit_length())
    # real FFT based linear convolution (full length)
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf"), conv
    score = float(2.0 * n * max_b / (sum_a ** 2))
    return score, conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute the soft‑max (log‑sum‑exp) weights for a convolution array.
    The weights sum to 1 and are numerically stable.
    """
    # subtract the max for stability
    c_max = conv.max()
    w = np.exp(beta * (conv - c_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        # fallback to uniform weights (should not happen)
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    return w


def _gradient_softmax(a: np.ndarray, conv: np.ndarray, beta: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of the smooth max objective
        f(a) = (1/β) log Σ_k exp(β * (a*a)[k])
    with respect to `a`.  The formula reduces to

        ∇f_i = 2 * Σ_k w_k * a_{k-i},

    where `w_k` are the soft‑max weights.  The sum can be obtained by a
    convolution of `a` with the reversed weight vector, which we evaluate via FFT.
    """
    n = a.shape[0]
    w = _softmax_weights(conv, beta)          # shape (2n-1,)
    w_rev = w[::-1]                           # reverse for the correlation
    # FFTs
    fa = np.fft.rfft(a, size_fft)
    fw = np.fft.rfft(w_rev, size_fft)
    grad_full = np.fft.irfft(fa * fw, size_fft)  # length = size_fft
    w_len = 2 * n - 1
    # indices that pick out Σ_k w_k * a_{k-i}
    idx = w_len - 1 - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Search for a non‑negative coefficient list minimising `evaluate_sequence`."""
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Settings
    # ------------------------------------------------------------------
    n = 8192                         # length of the step function
    size_fft = 1 << ((2 * n - 1).bit_length())  # power‑of‑two for FFT

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction (if any)
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = _normalise(np.asarray(hs1, dtype=float))
        score, _ = _compute_score(seq, size_fft)
        best_seq = seq.tolist()
        best_score = score
        print(f"[{time.time() - start_time:.1f}s] Warm‑start from height_sequence_1 "
              f"score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – systematic exploration of analytic families
    # ------------------------------------------------------------------
    # 1. product‑power (valley) family: a_i ∝ ((i+1)*(n-i))^{-p}
    idx = np.arange(n, dtype=float)
    prod_term = (idx + 1) * (n - idx)                       # (i+1)*(n-i)

    for p in np.linspace(0.0, 2.0, 21):                    # 0 → 2  (incl. uniform)
        shape = np.power(prod_term, -p, where=prod_term != 0)
        shape = _normalise(shape)
        score, _ = _compute_score(shape, size_fft)
        if score < best_score:
            best_score = score
            best_seq = shape.tolist()
            print(f"[{time.time() - start_time:.1f}s] "
                  f"Prod‑power p={p:.2f} → score={best_score:.6f}")

    # 2. sum‑power family: a_i ∝ (i+1)^{-p} + (n-i)^{-p}
    for p in np.linspace(0.0, 2.0, 21):
        shape = np.power(idx + 1, -p, where=idx + 1 != 0) + \
                np.power(n - idx, -p, where=n - idx != 0)
        shape = _normalise(shape)
        score, _ = _compute_score(shape, size_fft)
        if score < best_score:
            best_score = score
            best_seq = shape.tolist()
            print(f"[{time.time() - start_time:.1f}s] "
                  f"Sum‑power p={p:.2f} → score={best_score:.6f}")

    # 3. Gaussian‑valley family: a_i ∝ 1 / (1 + ((i-centre)/σ)^2)
    centre = (n - 1) / 2.0
    for sigma_frac in np.linspace(0.05, 0.30, 11):          # σ as fraction of n
        sigma = sigma_frac * n
        shape = 1.0 / (1.0 + ((idx - centre) / sigma) ** 2)
        shape = _normalise(shape)
        score, _ = _compute_score(shape, size_fft)
        if score < best_score:
            best_score = score
            best_seq = shape.tolist()
            print(f"[{time.time() - start_time:.1f}s] "
                  f"Gaussian σ={sigma_frac:.2f} → score={best_score:.6f}")

    # 4. Mixture of uniform and product‑power
    uniform = np.ones(n, dtype=float)
    for p in np.linspace(0.0, 2.0, 11):
        valley = np.power(prod_term, -p, where=prod_term != 0)
        valley = _normalise(valley)
        for w in np.linspace(0.0, 1.0, 11):
            mix = (1.0 - w) * uniform + w * valley
            mix = _normalise(mix)
            score, _ = _compute_score(mix, size_fft)
            if score < best_score:
                best_score = score
                best_seq = mix.tolist()
                print(f"[{time.time() - start_time:.1f}s] "
                      f"Mix p={p:.2f} w={w:.2f} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet / Gamma draws
    # ------------------------------------------------------------------
    alphas = [0.1, 0.3, 0.5, 0.8, 1.0, 2.0, 5.0]          # concentration parameters
    draws_per_alpha = 250
    for alpha in alphas:
        if time.time() >= deadline:
            break
        for _ in range(draws_per_alpha):
            # gamma draws give a Dirichlet(α,…,α) vector after normalisation
            vec = rng.gamma(alpha, 1.0, size=n)
            if vec.sum() == 0.0:
                continue
            vec = _normalise(vec)
            score, _ = _compute_score(vec, size_fft)
            if score < best_score:
                best_score = score
                best_seq = vec.tolist()
                print(f"[{time.time() - start_time:.1f}s] "
                      f"Dirichlet α={alpha:.2f} → score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max gradient descent with simulated‑annealing
    # ------------------------------------------------------------------
    cur = np.asarray(best_seq, dtype=float)  # current point (already normalised)
    cur = _normalise(cur)

    # schedule parameters
    start_temp, end_temp = 0.8, 0.001
    beta = 5.0            # start smoothness (soft‑max)
    beta_target = 400.0
    lr_start, lr_end = 0.08, 0.001

    # main optimisation loop
    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        # linear temperature for SA
        T = start_temp * (1 - elapsed / total) + end_temp * (elapsed / total)

        # compute convolution and gradient for the current vector
        score_cur, conv_cur = _compute_score(cur, size_fft)
        grad = _gradient_softmax(cur, conv_cur, beta, size_fft)

        # anneal learning‑rate and β
        lr = lr_start * (1 - elapsed / total) + lr_end * (elapsed / total)
        beta = min(beta_target, beta * 1.001)

        # gradient step (projected)
        nxt = cur - lr * grad
        nxt = _normalise(nxt)

        # evaluate the candidate
        score_nxt, _ = _compute_score(nxt, size_fft)

        # acceptance test
        if score_nxt < best_score:
            best_score = score_nxt
            best_seq = nxt.tolist()
            cur = nxt
            print(f"[{time.time() - start_time:.1f}s] NEW BEST (grad) score={best_score:.6f}")
            if best_score <= 1.503:
                print("[+] Target reached – terminating early.")
                break
        else:
            # simulated‑annealing probability
            prob = np.exp(-(score_nxt - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur = nxt

        # occasional random perturbation to escape flat regions
        if rng.random() < 0.02:
            # random jitter on a small fraction of entries
            m = max(1, int(0.005 * n))
            idxs = rng.choice(n, size=m, replace=False)
            jitter = rng.standard_normal(m) * cur.mean() * 0.05
            cur[idxs] = np.clip(cur[idxs] + jitter, 0.0, None)
            cur = _normalise(cur)

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain list of floats)
    # ------------------------------------------------------------------
    return best_seq
```
