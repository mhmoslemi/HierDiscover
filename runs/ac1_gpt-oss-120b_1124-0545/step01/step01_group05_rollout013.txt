def project_simplex(v):
    # v is 1D numpy array
    # returns projection onto simplex (nonnegative, sum=1).
    # if sum(v) == 0, return uniform? but we ensure positive.
    # Implementation: sort descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

import time
import numpy as np

def _conv_full(a, size):
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2 * len(a) - 1]
    return conv

def _conv_max(a, size):
    conv = _conv_full(a, size)
    return float(conv.max())

def _fast_score(a, size):
    n = a.shape[0]
    return 2.0 * n * _conv_max(a, size)

def _project_simplex(v):
    """Simplex projection: min_{x >= 0, sum x = 1} ||x - v||_2."""
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components <= 0 ; return uniform distribution
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _gradient_step(a, step, size):
    """Perform one projected gradient descent step on max(conv(a,a))."""
    n = a.shape[0]
    # current convolution and argmax
    conv = _conv_full(a, size)
    k = int(conv.argmax())
    # gradient of the max entry (2*a_{k-i})
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k - idx]
    # take step and project onto simplex
    v = a - step * g
    a_new = _project_simplex(v)
    # compute new convolution max for return
    max_new = _conv_max(a_new, size)
    return a_new, max_new

def _make_exponential(n, lam):
    i = np.arange(n, dtype=float)
    a = np.exp(-lam * i)
    a /= a.sum()
    return a

def _make_two_block(n, m, high_frac):
    if m <= 0 or m >= n:

def _conv_full(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT size."""
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * len(a) - 1]
    return conv


def _conv_max(a: np.ndarray, size: int) -> float:
    """Maximum entry of the self‑convolution."""
    return float(_conv_full(a, size).max())


def _fast_score(a: np.ndarray, size: int) -> float:
    """Fast surrogate: 2·n·max(conv(a,a)) (sequence is already normalised)."""
    n = a.shape[0]
    return 2.0 * n * _conv_max(a, size)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries ≤ 0 → return the uniform distribution
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_step(a: np.ndarray, step: float, size: int):
    """
    One projected‑gradient step that directly reduces the current
    maximal convolution value.
    Returns the new vector and the new max‑convolution entry.
    """
    n = a.shape[0]
    conv = _conv_full(a, size)
    k = int(conv.argmax())                     # index of the worst lag

    # gradient of the k‑th convolution entry:  d/d a_i (c_k) = 2·a_{k-i}
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    g = np.zeros_like(a)
    g[idx] = 2.0 * a[k - idx]

    # take a (sub)gradient step and project back onto the simplex
    v = a - step * g
    a_new = _project_simplex(v)

    max_new = _conv_max(a_new, size)
    return a_new, max_new
