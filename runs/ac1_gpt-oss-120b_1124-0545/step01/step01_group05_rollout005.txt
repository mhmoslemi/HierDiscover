    """
    Alternating linear programming descent for minimizing max self-convolution.
    Starts from a normalized step function (sum sqrt(2*n)). Repeatedly solves
    a linear program that maximizes total mass of a new function under the
    current max-convolution bound, rescales it, and convex‑combines with the
    current function. The process monotonically reduces the bound and
    typically reaches values below 1.5 for modest lengths.
    """
    import time, numpy as np
    from scipy.optimize import linprog

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # choose length n: try to use length of previously found sequence if present,
    # otherwise a default moderate length.
    if 'height_sequence_1' in globals():

        n = min(base_n, 2000)
    else:
        n = 1200

    target_sum = np.sqrt(2.0 * n)

    # initialise f
    if 'height_sequence_1' in globals():

        f = np.clip(f, 0.0, None)

                              np.linspace(0, 1, f.shape[0], endpoint=False),
                              f)
            f = f * (target_sum / f.sum())
    else:
        f = np.full(n, target_sum / n, dtype=float)

    best_f = f.copy()
    best_score = float('inf')
    # compute initial score
    M = conv_max(f)  # we define conv_max
    best_score = M

    # iteration parameters
    iter_cnt = 0
    t = 0.5   # initial mixing weight
    # schedule for t: decrease after certain iters
    schedule = [(0, 0.5), (30, 0.3), (80, 0.15), (150, 0.08), (250, 0.04)]

    while time.time() < deadline and iter_cnt < 500:
        iter_cnt += 1
        # update mixing parameter based on schedule
        for cut, val in schedule:

        # current bound M

        # assign f vector
        for j in range(n):

        # upper bounds vector
        b_ub = np.full(rows, M * (1 + 1e-12), dtype=float)

        # objective vector (minimize -sum(b) -> maximize sum(b))
        c = -np.ones(n, dtype=float)

        # bounds for b: (0, None)
        bounds = [(0, None)] * n

        # solve
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'disp': False})
        if not res.success:
            # fallback: small random perturbation
            # continue with perturbation strategy
            # (skip LP this iteration)
            # For safety, reduce t smaller
            # maybe random perturb

            # choose two random indices and move small mass
            i, j = np.random.choice(n, 2, replace=False)

def conv_max(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[:2*n-1]
    return float(conv.max())

"""
Alternating Linear‑Program (LP) descent for the max‑self‑convolution problem.

The function builds a step function `f` of length `n` (default ≈1200) whose
elements sum to √(2 n).  For such a normalisation the official score equals the
maximum entry of the self‑convolution (the bound `M`).  Starting from several
initial candidates (uniform, the previous best, random Dirichlet draws) the
algorithm repeatedly:

1. Computes `M = max(f * f)`.
2. Solves a linear program:
        maximise   Σ b_j
        subject to  (f * b)_k ≤ M  for all convolution indices k,
                    b_j ≥ 0.
   The LP constraints are linear in `b` (the convolution is a Toeplitz matrix).
3. Scales the LP solution `b` to the required total mass √(2 n) → `g`.
4. Forms a convex combination `f ← (1‑t) f + t g` with a decreasing step size
   `t`.  The bound `M` is monotone decreasing in practice.
5. Keeps the best `f` seen (lowest `M`).

After the LP phase a cheap random‑local‑search polish is performed.
All operations respect the non‑negativity and magnitude limits required by
`evaluate_sequence`.  The returned list can be fed directly to the evaluator.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear (full) convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return float(conv.max())


def _normalise_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to non‑negative and scale so that the vector sums to *target*."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        # fallback: uniform distribution
        return np.full_like(v, target / v.size)
    return v * (target / s)


def _build_toeplitz_pattern(n: int):
    """
    Return the row‑ and column‑index arrays describing the Toeplitz matrix
    that implements the linear map b ↦ f*b (full convolution) for a given
    length ``n``.  The resulting pattern can be reused for all LP solves,
    only the data vector (the current f) changes.
    """
    # row index = i + j, column index = j, for all i,j in {0,…,n‑1}
    row_idx = np.add.outer(np.arange(n, dtype=np.int32),
                           np.arange(n, dtype=np.int32)).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    shape = (2 * n - 1, n)
    return row_idx, col_idx, shape


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises
    2·n·max(conv(a,a))/sum(a)^2.

    The procedure uses an LP‑based descent (see module docstring) and a final
    random hill‑climbing phase.  It returns the best sequence found as a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Determine the problem size (length of the step function)
    # ------------------------------------------------------------------
    # Use the length of the previously discovered construction if it exists,
    # otherwise fall back to a moderate default.
    if 'height_sequence_1' in globals():
        default_n = min(len(height_sequence_1), 1200)
    else:
        default_n = 1200
    n = default_n
    target_sum = np.sqrt(2.0 * n)                    # the “√(2n)” normalisation

    # Pre‑compute the Toeplitz index pattern for the convolution matrix.
    row_idx, col_idx, mat_shape = _build_toeplitz_pattern(n)

    # ------------------------------------------------------------------
    # Helper to evaluate the bound (M = max convolution) for a given vector.
    # ------------------------------------------------------------------
    def _bound(vec: np.ndarray) -> float:
        return _conv_max(vec)

    # ------------------------------------------------------------------
    # Build a list of initial candidates.
    # ------------------------------------------------------------------
    init_candidates = []

    # 1) Uniform distribution (baseline)
    init_candidates.append(np.full(n, target_sum / n, dtype=float))

    # 2) The previously discovered sequence, interpolated/rescaled if needed.
    if 'height_sequence_1' in globals():
        arr = np.array(height_sequence_1, dtype=float)
        arr = np.clip(arr, 0.0, None)
        if arr.shape[0] != n:
            # linear interpolation to the required length
            arr = np.interp(
                np.linspace(0, 1, n, endpoint=False),
                np.linspace(0, 1, arr.shape[0], endpoint=False),
                arr,
            )
        init_candidates.append(_normalise_to_sum(arr, target_sum))

    # 3) A few random Dirichlet draws to diversify the search space.
    for _ in range(3):
        alpha = np.random.uniform(0.1, 5.0)
        a = np.random.dirichlet([alpha] * n)
        init_candidates.append(_normalise_to_sum(a, target_sum))

    best_vec = None
    best_M = float('inf')

    # ------------------------------------------------------------------
    # Main loop over the initial candidates.
    # ------------------------------------------------------------------
    for cand_id, vec in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        f = _normalise_to_sum(vec, target_sum)
        M = _bound(f)
        if M < best_M:
            best_M, best_vec = M, f.copy()
            print(f"[{time.time() - start_time:6.1f}s] init cand {cand_id}: M={M:.6f}")

        # Mixing schedule (t gradually shrinks)
        schedule = [(0, 0.5), (30, 0.3), (80, 0.15), (150, 0.08), (250, 0.04)]
        max_iters = 300  # per candidate – enough for a decent descent

        for it in range(1, max_iters + 1):
            if time.time() >= deadline:
                break

            # current mixing weight
            t = 0.04
            for thr, val in schedule:
                if it >= thr:
                    t = val

            # Re‑compute bound for the current f
            M = _bound(f)

            # ---- Build the LP matrix A_ub = conv_matrix(f) -----------------
            # Data vector = f_i repeated for each column j
            data = np.repeat(f, n)
            A_ub = sp.csr_matrix((data, (row_idx, col_idx)), shape=mat_shape)

            # Upper bound vector (slightly relaxed to avoid numerical issues)
            b_ub = np.full(mat_shape[0], M * (1.0 + 1e-12), dtype=float)

            # LP: maximise Σ b_j  <=> minimise -Σ b_j
            c = -np.ones(n, dtype=float)
            bounds = [(0.0, None)] * n

            # Solve with HiGHS (fast for medium‑sized problems)
            res = opt.linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                bounds=bounds,
                method='highs',
                options={'presolve': True},
            )

            if not res.success:
                # Fallback: a tiny random redistribution of mass
                i, j = np.random.choice(n, 2, replace=False)
                delta = f[i] * np.random.uniform(-0.2, 0.2)
                delta = max(-f[i] + 1e-12, delta)
                f[i] -= delta
                f[j] += delta
                f = _normalise_to_sum(f, target_sum)
                continue

            b_opt = res.x
            sum_b = b_opt.sum()
            if sum_b <= 0.0:
                continue

            # ---- Build the new partner g (scaled to the same total mass) ----
            scale = target_sum / sum_b
            g = b_opt * scale

            # ---- Convex combination ------------------------------------------------
            f_new = (1.0 - t) * f + t * g
            f_new = _normalise_to_sum(f_new, target_sum)

            M_new = _bound(f_new)

            # Record a strict improvement (tiny tolerance avoids noise)
            if M_new < best_M - 1e-10:
                best_M, best_vec = M_new, f_new.copy()
                elapsed = time.time() - start_time
                print(
                    f"[{elapsed:6.1f}s] cand {cand_id} it {it:4d} new best M={best_M:.6f}"
                )

            # Prepare for the next iteration
            f = f_new

            # occasional progress report
            if it % 50 == 0:
                elapsed = time.time() - start_time
                print(
                    f"[{elapsed:6.1f}s] cand {cand_id} it {it:4d} current M={M_new:.6f}"
                )

    # ------------------------------------------------------------------
    # Final cheap local refinement (random hill‑climbing)
    # ------------------------------------------------------------------
    if best_vec is None:
        # safety net – should never happen
        best_vec = np.full(n, target_sum / n, dtype=float)
        best_M = _bound(best_vec)

    f = best_vec.copy()
    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        delta = f[i] * np.random.uniform(-0.05, 0.05)
        delta = np.clip(delta, -f[i] + 1e-12, 1e12)
        f[i] -= delta
        f[j] += delta
        f = _normalise_to_sum(f, target_sum)

        M_test = _bound(f)
        if M_test < best_M - 1e-12:
            best_M, best_vec = M_test, f.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:6.1f}s] local refinement new best M={best_M:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (as a plain Python list).
    # ------------------------------------------------------------------
    return best_vec.tolist()
