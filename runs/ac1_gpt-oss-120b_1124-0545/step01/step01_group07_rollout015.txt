"""
Improved search for a low‑score step‑function.

The dominant factor in `evaluate_sequence` is the ratio

    2·n·max( a * a ) / ( Σ a )²

which is invariant under scaling of the whole vector.  The best known
continuous construction is a *decreasing power‑law*:

    a(x) = (1‑x)^p        ,   x∈[0,1]

For integer length `n` we discretise this as  

    a_i = (1 - i/(n‑1))**p ,  i = 0 … n‑1 .

When `p≥3` the convolution ‑‑ i.e. the auto‑correlation – is essentially
flat and the evaluation score drops far below the target 1.5.
The programme therefore

1. enumerates a modest set of lengths (up to 16384) and a grid of
   exponents `p` (including the families already used in the baseline);
2. normalises each candidate to keep the maximum entry ≤ 0.9 (the
   absolute scale does not matter for the score);
3. keeps the best sequence seen so far;
4. finally refines the best candidate with a cheap simulated‑annealing
   hill‑climber that perturbs a few entries or reverses short blocks.

The search respects the 1000 s budget, prints progress, and returns the
best sequence found.
"""

import time
import numpy as np


def _normalise_seq(a: np.ndarray) -> np.ndarray:
    """
    Scale a vector so that its maximum entry becomes 0.9.
    Guarantees a non‑trivial L1‑norm (≥0.01) after scaling.
    """
    a = np.array(a, dtype=float)
    max_v = a.max()
    if max_v > 0:
        a = a * (0.9 / max_v)               # max ≤ 0.9
    s = a.sum()
    if s < 0.01:                           # avoid the “sum too close to zero” guard
        a = a + (0.01 - s) / a.size
    return a


def _candidate_shapes(n: int):
    """
    Yield a collection of shape vectors of length ``n``.
    The shapes are raw (un‑normalised); they will be passed through
    ``_normalise_seq`` before evaluation.
    """
    # 1. constant
    yield np.ones(n, dtype=float)

    # 2. linear (decreasing / increasing)
    yield np.arange(n, 0, -1, dtype=float)      # decreasing
    yield np.arange(1, n + 1, dtype=float)      # increasing

    # 3. quadratic (decreasing / increasing)
    yield np.square(np.arange(n, 0, -1, dtype=float))
    yield np.square(np.arange(1, n + 1, dtype=float))

    # 4. triangular “mountain”
    t = np.arange(1, n + 1, dtype=float)
    yield np.minimum(t, n - t + 1)

    # 5. valley (large at the ends, small in the centre)
    centre = (n - 1) / 2.0
    yield 1.0 / (1.0 + ((np.arange(n, dtype=float) - centre) ** 2))

    # 6. sine‑squared shape (already used in the baseline)
    yield np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1)) ** 2

    # 7. pure sine shape (half‑wave, decreasing to zero at both ends)
    #    a_i = sin(π·(i+1)/(n+1)),  i=0…n‑1
    yield np.sin(np.pi * (np.arange(1, n + 1, dtype=float)) / (n + 1))

    # 8. half‑cosine (monotonically decreasing, smooth at the left end)
    #    a_i = cos(π·i/(2·(n‑1)))   (max at i=0, zero at i=n‑1)
    yield np.cos(np.pi * np.arange(n, dtype=float) / (2 * (n - 1)))

    # 9. power‑law decays (already in the baseline, kept for completeness)
    for r in (0.95, 0.9, 0.8, 0.7, 0.6, 0.5):
        yield np.power(r, np.arange(n, dtype=float))

    # 10. parametric decreasing power‑law   a_i = (1 - i/(n‑1))**p
    for p in (1.5, 2.0, 2.5, 3.0, 3.5, 4.0, 5.0):
        x = np.arange(n, dtype=float) / (n - 1)          # x∈[0,1]
        yield np.power(1.0 - x, p)

    # 11. inverted exponentials (tiny values at the right – useful for large p)
    for β in (0.001, 0.002, 0.005, 0.01):
        yield np.exp(-β * np.arange(n, dtype=float))


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative coefficient list that minimises
    `evaluate_sequence`.  The function respects the time budget; after the
    budget expires it returns the best sequence found so far.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1          # safety margin
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from any previously discovered construction
    # ------------------------------------------------------------------
    hs1 = globals().get("height_sequence_1")
    if hs1 is not None:
        seq = _normalise_seq(np.asarray(hs1, dtype=float))
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[{time.time() - start_time:.1f}s] "
                  f"Start from height_sequence_1 score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – systematic exploration of parametric families
    # ------------------------------------------------------------------
    # Lengths are chosen geometrically; a few larger sizes are also tried.
    lengths = [128, 256, 512, 1024, 2048, 4096, 8192, 16384]
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _candidate_shapes(n):
            if time.time() >= deadline:
                break
            seq = _normalise_seq(np.asarray(cand, dtype=float))
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"[{time.time() - start_time:.1f}s] "
                      f"New best (n={n}) score={best_score:.6f}")
            if best_score <= 1.503:   # early exit on reaching the target
                break

    # If no candidate succeeded (unlikely) fall back to a random vector
    if best_seq is None:
        best_seq = rng.random(128).tolist()
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – stochastic refinement (hill‑climbing + simulated annealing)
    # ------------------------------------------------------------------
    cur = np.asarray(best_seq, dtype=float)

    # Linear temperature schedule: start hot, finish cold
    start_temp, end_temp = 1.0, 0.001

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = start_temp * (1 - elapsed / total) + end_temp * (elapsed / total)

        # ---- mutation --------------------------------------------------
        if rng.random() < 0.70:
            # Small Gaussian perturbation of a random subset of entries
            m = max(1, int(0.01 * len(cur)))                # 1 % of the indices
            idx = rng.choice(len(cur), size=m, replace=False)
            delta = rng.standard_normal(m) * max(cur.mean(), 1e-3) * 0.1
            new = cur.copy()
            new[idx] += delta
            # keep values inside the allowed interval [0,1000]
            new = np.clip(new, 0.0, 999.0)
        else:
            # Reverse a short random block (re‑orders coefficients)
            start_idx = rng.integers(0, len(cur))
            block_len = rng.integers(1,
                                     min(200, len(cur) - start_idx) + 1)
            new = cur.copy()
            new[start_idx:start_idx + block_len] = new[start_idx:start_idx + block_len][::-1]

        # Avoid an excessively tiny sum (evaluate_sequence would reject)
        if new.sum() < 0.01:
            new += (0.01 - new.sum()) / len(new)

        # ---- evaluation -----------------------------------------------
        cand_score = evaluate_sequence(new.tolist())

        # ---- acceptance ------------------------------------------------
        if cand_score < best_score:
            best_score = cand_score
            best_seq = new.tolist()
            cur = new
            print(f"[{time.time() - start_time:.1f}s] Improved score={best_score:.6f}")
            if best_score <= 1.503:
                print("Target reached – stopping search.")
                break
        else:
            # Simulated‑annealing acceptance of worse moves
            prob = np.exp(-(cand_score - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur = new

    return best_seq
