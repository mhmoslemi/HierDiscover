def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5  # safety

    # Choose length
    n = 4000 # maybe 5000 as before

    # initialize using 1/sqrt shape
    seq = 1.0/np.sqrt(np.arange(1, n+1))
    seq = seq / seq.sum()  # sum=1

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, score={best_score}")

    # parameters
    step = 0.1  # step size
    step_decay = 0.99998
    beta = 1.0
    beta_inc = 1.001
    max_beta = 1000.0

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        conv = np.convolve(seq, seq)
        max_conv = conv.max()

        # compute weights
        w = np.exp(beta * (conv - max_conv))
        # compute gradient of weighted sum
        # correlation
        corr = np.correlate(w, seq, mode='full')
        # gradient for each i
        grad = 2.0 * corr[(n-1) - np.arange(n)]

        # make gradient orthogonal to sum constraint
        grad = grad - grad.mean()

        # step update
        new_seq = seq - step * grad
        new_seq = np.maximum(new_seq, 0.0)
        # renorm to sum=1
        total = new_seq.sum()
        if total == 0:
            # fallback to initial
            new_seq = seq.copy()
            total = new_seq.sum()
        new_seq = new_seq / total

        seq = new_seq

        # optionally random move every few iters
        if iteration % 200 == 0:
            # swap small mass
            i, j = np.random.choice(n, 2, replace=False)
            eps = step * seq[i]
            eps = min(eps, seq[i])
            cand = seq.copy()
            cand[i] -= eps
            cand[j] += eps
            cand = np.maximum(cand,0)
            cand = cand / cand.sum()
            seq = cand

        # update step and beta
        step = max(step * step_decay, 1e-7)
        beta = min(beta * beta_inc, max_beta)

        # evaluate
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

    print(f"[final] iters={iteration}, best={best_score:.6f}")
    return best_seq.tolist()

"""
Propose a step‑function (non‑negative sequence) that minimizes
the provided evaluation function.

Method
------
1.  Initialise a reasonably large sequence (n≈3000‑5000) with the
    classical  a_i ∝ 1/√(i+1) shape.
2.  Quickly sweep a one‑parameter family  a_i ∝ 1/√(i+α)  and a small
    exponent neighbourhood  β≈0.5 .  The best candidate from this
    cheap grid search becomes the starting point for the main
    optimisation.
3.  Main optimisation = a smoothed projected gradient descent.
    The gradient is taken from a *soft‑max* of the convolution
    entries (instead of the hard max used in the reference code)
    which provides a more informative descent direction.
    After each step the sequence is projected onto the affine
    subspace Σ a_i = 1 and clipped to non‑negative values.
4.  Every few hundred iterations an LP‑based “direction” is
    computed: using the currently active (largest) convolution
    constraints we maximise Σ g_i subject to   f∗g ≤ max(f∗f).
    The resulting g is mixed with the current f (small step size)
    – this reproduces the active‑set idea from the paper and
    helps escape plateaux.
5.  Random tiny swaps are performed periodically to add stochastic
   ity.
6.  The best sequence seen during the whole run (according to
    ``evaluate_sequence``) is returned.

The routine respects the 1000 s time budget, prints progress and
always returns a valid (non‑negative, sum>0) list.
"""

import time
import numpy as np
from scipy.optimize import linprog  # available in the execution environment


def _root_sequence(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+1)  (sum normalised to 1)."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _shifted_power_sequence(n: int, offset: float, beta: float) -> np.ndarray:
    """
    a_i ∝ 1 / (i + offset) ** beta   (i starts at 0),
    normalised to sum‑1.  ``offset`` > 0 avoids division by zero.
    """
    i = np.arange(n, dtype=float)
    # add a tiny epsilon to keep the denominator >0 even if offset==0
    a = 1.0 / np.power(i + offset + 1e-12, beta)
    a_sum = a.sum()
    if a_sum == 0.0:
        a[:] = 1.0 / n
        a_sum = 1.0
    a /= a_sum
    return a


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex Σ v_i = 1, v_i ≥ 0."""
    # standard O(n) algorithm (see Wang & Carreira‑Perpinan 2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – fall back to the uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return a soft‑max weighting of the convolution entries."""
    # conv.max() is subtracted for numerical stability
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))
    return w


def _gradient_from_weights(seq: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum  Σ w_k (seq * seq)_k  w.r.t. ``seq``.
    Using the identity  ∂/∂a_j Σ w_k (a*a)_k = 2 Σ_k w_k a_{k-j}.
    It is efficiently computed via a correlation.
    """
    n = seq.size
    # correlation w * seq  (full length = len(w)+len(seq)-1)
    corr = np.correlate(weights, seq, mode="full")
    # gradient component j = 2 * corr[n-1-j]  (see derivation above)
    grad = 2.0 * corr[(n - 1) - np.arange(n)]
    return grad


def _lp_direction(f: np.ndarray, max_conv: float, active_idx: np.ndarray) -> np.ndarray:
    """
    Solve the linear programme described in the statement:
        maximise Σ g_i
        s.t.   Σ_j f_j * g_{k-j} ≤ max_conv   for every k in ``active_idx``
               g_i ≥ 0.
    Returns a feasible direction ``g`` (normalised to sum‑1).  If the LP
    fails for any reason, ``None`` is returned.
    """
    n = f.size
    m = len(active_idx)

    # Build the constraint matrix A (shape m × n)
    A = np.zeros((m, n), dtype=float)
    for row, k in enumerate(active_idx):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        cols = np.arange(lo, hi + 1, dtype=int)
        A[row, cols] = f[k - cols]

    b = np.full(m, max_conv, dtype=float)

    # linprog minimises c^T x, so use -1 to maximise Σ g_i
    c = -np.ones(n, dtype=float)

    # Solve with HiGHS (fast interior‑point method).  Bounds are (0, ∞).
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": 0.5},
    )
    if not res.success:
        return None
    g = res.x
    if g.sum() == 0.0:
        return None
    # Normalise to a probability distribution (sum = 1)
    g /= g.sum()
    return g


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence that minimises ``evaluate_sequence``.
    The procedure respects the given time budget and always returns
    a valid (non‑negative, sum>0) sequence.
    """
    import numpy as np

    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------
    # 1. Choose a moderate size – large enough to approach the
    #    continuous optimum, small enough for fast iterations.
    # ------------------------------------------------------------
    n = 3000  # works well within the typical time limits

    # ------------------------------------------------------------
    # 2. Fast parameter sweep (offset, beta) to get a good start.
    # ------------------------------------------------------------
    best_seq = _root_sequence(n)
    best_score = float("inf")
    # offsets in [0.0, 5.0] – 0 means the pure 1/√(i+1) shape
    offsets = np.linspace(0.0, 5.0, 51)
    betas = np.linspace(0.45, 0.55, 11)

    for off in offsets:
        for bet in betas:
            if time.time() >= deadline:
                break
            seq = _shifted_power_sequence(n, offset=off + 1.0, beta=bet)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(
                    f"[grid] offset={off:.2f} beta={bet:.3f} "
                    f"score={best_score:.6f}"
                )
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------
    # 3. Main iterative optimisation.
    # ------------------------------------------------------------
    seq = best_seq.copy()
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())

    # hyper‑parameters
    step = 0.2               # initial projected‑gradient step size
    step_decay = 0.99995     # geometric decay
    beta = 4.0               # start with moderate smoothing
    beta_growth = 1.001      # slowly sharpen the soft‑max
    beta_max = 500.0
    lp_interval = 250        # how often to call the LP sub‑routine
    lp_mix = 0.07            # mixing factor when applying the LP direction
    random_swap_every = 70   # frequency of a tiny random swap

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----- convolution, soft‑max weights and gradient ----------
        conv = np.convolve(seq, seq)
        max_conv = conv.max()

        # smooth max with a soft‑max (gradient from many top entries)
        weights = _softmax_weights(conv, beta)

        grad = _gradient_from_weights(seq, weights)

        # make the step tangent to the hyperplane Σ a_i = 1
        grad -= grad.mean()

        # projected gradient step
        new_seq = seq - step * grad
        new_seq = np.maximum(new_seq, 0.0)
        # re‑project onto the probability simplex
        new_seq = _project_onto_simplex(new_seq)

        seq = new_seq

        # ----- occasional random tiny swap (helps escape flat spots) -----
        if iteration % random_swap_every == 0:
            i, j = np.random.choice(n, 2, replace=False)
            eps = step * seq[i]
            eps = min(eps, seq[i])
            cand = seq.copy()
            cand[i] -= eps
            cand[j] += eps
            cand = np.maximum(cand, 0.0)
            cand = _project_onto_simplex(cand)
            seq = cand

        # ----- LP‑based improvement step (active‑set idea) -----
        if iteration % lp_interval == 0:
            # determine the most critical convolution indices
            # keep every index whose value is within 0.2 % of the max
            thresh = max_conv * (1.0 - 2e-3)
            active = np.where(conv >= thresh)[0]
            # keep the number reasonable (max ~30)
            if len(active) > 30:
                # keep only the largest 30 entries
                idx_sorted = np.argsort(conv)[-30:]
                active = idx_sorted
            g_dir = _lp_direction(seq, max_conv, active)
            if g_dir is not None:
                # mix a small amount of the LP direction into the current iterate
                seq = (1.0 - lp_mix) * seq + lp_mix * g_dir
                # re‑normalise (the mixing already preserves the sum, but do it for safety)
                seq = _project_onto_simplex(seq)

        # ----- update meta‑parameters -----
        step = max(step * step_decay, 1e-6)
        beta = min(beta * beta_growth, beta_max)

        # ----- evaluate -----
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[improve] iter={iteration} score={best_score:.6f}")

    # final report
    total_iters = iteration
    print(f"[final] iterations={total_iters} best_score={best_score:.6f}")

    # make sure the returned list conforms to the tester expectations
    return best_seq.tolist()
