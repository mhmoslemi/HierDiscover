def compute_conv(seq):

def compute_gradient(seq, alpha):
    conv = compute_conv(seq)
    max_c = conv.max()
    # compute w = exp(alpha * (conv - max_c)), subtract max for stability
    w = np.exp(alpha * (conv - max_c))  # now max w is 1.
    sum_w = w.sum()
    rev_seq = seq[::-1]
    # compute h = np.convolve(w, rev_seq)
    h = np.convolve(w, rev_seq)
    # gradient before scaling: grad_i = 2 * h[n-1 + i] / sum_w
    grad = 2.0 * h[(len(seq)-1):(len(seq)-1+len(seq))] / sum_w
    # ensure shape n
    # Project onto sum=0 hyperplane:
    grad -= grad.mean()
    return grad

def compute_gradient(seq, alpha):
    # Returns gradient of smoothed max conv
    n = seq.shape[0]
    conv = full_convolution_fft(seq, seq)
    max_conv = conv.max()
    # w_i = exp(alpha * (conv_i - max_conv))
    # shift - max to avoid overflow
    w = np.exp(alpha * (conv - max_conv))
    sum_w = w.sum()
    rev_seq = seq[::-1]
    # conv of w with rev_seq
    h = full_convolution_fft(w, rev_seq)   # length len(w)+len(rev_seq)-1 = (2n-1)+(n)-1 = 3n-2
    # extract indices n-1 to 2n-2 inclusive => shape n
    start = n - 1
    grad = 2.0 * h[start:start+n] / sum_w
    # Project onto sum=0 hyperplane
    grad -= grad.mean()
    return grad

def optimize_seq(n, init_seq=None, max_iters=2000, time_limit=deadline, report_interval=100):
   if init_seq is None: seq = _initial_root_sequence(n)
   else seq = init_seq.copy()
   # Add small noise
   seq += np.random.uniform(-0.001,0.001,size=n)
   seq = np.maximum(seq,0)
   if seq.sum() == 0: seq = _initial_root_sequence(n)
   seq /= seq.sum()
   m = np.zeros_like(seq)
   v = np.zeros_like(seq)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8
   lr = 0.1
   alpha = 1.0
   max_alpha = 2000.0  # increase gradually
   best_seq_local = seq.copy()
   best_score_local = evaluate_score(seq)
   t = 0
   while t < max_iters and time.time() < time_limit:
       t += 1
       grad = compute_gradient(seq, alpha)
       # Adam update
       m = beta1 * m + (1 - beta1) * grad
       v = beta2 * v + (1 - beta2) * (grad**2)
       m_hat = m / (1 - beta1**t)
       v_hat = v / (1 - beta2**t)
       step = lr * m_hat / (np.sqrt(v_hat) + eps)
       seq_new = seq - step
       # enforce non-negativity
       seq_new = np.maximum(seq_new, 0.0)
       total = seq_new.sum()
       if total > 0:
           seq_new /= total
       else:
           seq_new = _initial_root_sequence(n)  # fallback
       seq = seq_new
       # increase alpha
       alpha = min(alpha * 1.025, max_alpha)
       if t % report_interval == 0:
           cur_score = evaluate_score(seq)
           if cur_score < best_score_local:
               best_score_local = cur_score
               best_seq_local = seq.copy()
               print(f"[n={n}] iter={t}, score={best_score_local:.6f}, alpha={alpha:.2f}")
   return best_seq_local, best_score_local

def project_simplex(v):
    # Sort v descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a sequence that minimizes the evaluation function.

Algorithm
---------

*  Initialise a non‑negative ’root’ sequence a_i ∝ 1/√(i+1) (or a
   randomised power‑law variant) and normalise it to sum 1.

*  Perform a projected Adam optimisation that minimises a smooth
   approximation of  max(conv(seq,seq)).
   The smooth loss is  L_α(seq)= (1/α)·log ∑_k exp(α·c_k)  where
   c_k = (seq∗seq)_k .  Its gradient can be written in closed form as a
   convolution, which we evaluate with FFT (O(n log n)).  The
   temperature‑parameter α is increased slowly; for small α the loss
   flattens the whole convolution, for large α it focuses on the peak.

*  After each Adam step the trial point is projected onto the probability
   simplex (non‑negative, sum = 1) – this is the exact feasible set of the
   original problem.

*  Occasionally a tiny random mass transfer between two coordinates is
   performed to help escape shallow local minima.

*  The optimisation is run for many short independent runs with
   different lengths (2000…8000) and random initialisations; the best
   sequence found before the time budget expires is returned.

The whole routine respects the 1000 s budget and uses only numpy (no
file/network access).  The reported score is 2·n·max(conv)/sum² which is
identical to ``evaluate_sequence`` for feasible inputs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0,  Σ_i x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive -> return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (exact, no circular wrap)."""
    n = a.size + b.size - 1
    # next power of two for speed / to avoid circular effects
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    fc = fa * fb
    conv = np.fft.irfft(fc, size)[:n]
    return conv


def _initial_root_sequence(n: int) -> np.ndarray:
    """Return the classic  a_i ∝ 1/√(i+1)  sequence, normalised to sum = 1."""
    a = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def _random_power_law_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a random power‑law sequence  a_i ∝ (i+δ)^{-p}  with
    p∈[0.45,0.7] and shift δ∈[0,0.3·n].
    """
    p = rng.uniform(0.45, 0.70)
    delta = rng.uniform(0.0, 0.30 * n)
    i = np.arange(n, dtype=float) + 1.0  # avoid zero denominator
    a = 1.0 / np.power(i + delta, p)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = _initial_root_sequence(n)
    else:
        a /= a.sum()
    # small Gaussian jitter
    a += rng.normal(scale=0.001, size=n)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = _initial_root_sequence(n)
    else:
        a /= a.sum()
    return a


def _evaluate_score(seq: np.ndarray) -> float:
    """
    Compute the exact score  2·n·max(conv(seq,seq)) / (sum(seq)²) .
    The sum is 1 for all sequences we keep, so the denominator is 1.
    """
    n = seq.size
    conv = _convolve_fft(seq, seq)
    return 2.0 * n * conv.max()


def _gradient_smooth_max(seq: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed max loss:
        L_α(seq) = (1/α)·log∑_k exp(α·c_k)   ,  c = seq∗seq .
    Returns a gradient vector whose components sum to zero (projection
    onto the hyperplane Σ_i g_i = 0).  The formula uses two FFT‑based
    convolutions and is O(n log n).
    """
    n = seq.size
    # full self‑convolution
    conv = _convolve_fft(seq, seq)

    # soft‑max weighting (stable because we shift by the maximum)
    max_c = conv.max()
    w = np.exp(alpha * (conv - max_c))
    sum_w = w.sum()

    # gradient before projection:
    #   ∂c_k/∂seq_i = 2·seq_{k-i}  (zero when out of range)
    #   g_i = 2·Σ_k w_k·seq_{k-i} / Σ_w
    rev_seq = seq[::-1]
    tmp = _convolve_fft(w, rev_seq)          # length = (2n-1)+n-1 = 3n-2
    start = n - 1
    grad = 2.0 * tmp[start:start + n] / sum_w

    # make it tangent to Σ_i seq_i = const  (i.e. Σ_i g_i = 0)
    grad -= grad.mean()
    return grad


# ----------------------------------------------------------------------
# Core optimiser for a single length
# ----------------------------------------------------------------------


def _optimise_one_length(
    n: int,
    init_seq: np.ndarray,
    max_iters: int,
    time_limit: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation that minimises the smooth‑max loss.
    Returns the best sequence encountered and its (exact) score.
    """
    seq = init_seq.copy()
    # Adam state
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15                 # base learning rate (tuned empirically)
    alpha = 1.0               # start of the smooth‑max temperature
    alpha_max = 2000.0        # final temperature
    alpha_inc = 1.025         # multiplicative increase per iteration

    best_seq = seq.copy()
    best_score = _evaluate_score(best_seq)

    # small counter to abort a run that makes no progress
    no_improve_counter = 0
    last_improve_iter = 0

    for it in range(1, max_iters + 1):
        if time.time() >= time_limit:
            break

        # ---- gradient of the smoothed max ----
        grad = _gradient_smooth_max(seq, alpha)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # projected onto the simplex (exact feasibility)
        seq = _project_onto_simplex(seq - step)

        # gradually raise the temperature – focuses the loss more on the true max
        if alpha < alpha_max:
            alpha = min(alpha * alpha_inc, alpha_max)

        # occasional tiny random transfer (helps to escape plateaus)
        if it % 50 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                eps_transfer = lr * seq[i]   # a small fraction of mass
                eps_transfer = min(eps_transfer, seq[i])
                cand = seq.copy()
                cand[i] -= eps_transfer
                cand[j] += eps_transfer
                cand = _project_onto_simplex(cand)
                seq = cand

        # evaluate exact score
        cur_score = _evaluate_score(seq)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            no_improve_counter = 0
            last_improve_iter = it
        else:
            no_improve_counter += 1

        # early termination if we have been stuck for many iterations
        if no_improve_counter > 500 and it - last_improve_iter > 500:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the testing harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The routine runs many short Adam‑optimisation trials for several
    lengths (2000‑8000) and returns the best sequence found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 4.0

    # lengths we will try – a mix of moderate and relatively large sizes
    lengths = [2000, 3000, 4000, 5000, 6000, 8000]

    global_best_seq = None
    global_best_score = np.inf
    run_id = 0

    while time.time() < deadline:
        # ----- pick a length (biased towards the larger ones) -----
        n = int(rng.choice(lengths))
        # ----- initialise -------------------------------------------------
        if rng.random() < 0.5:
            seq0 = _initial_root_sequence(n)
        else:
            seq0 = _random_power_law_sequence(n, rng)

        # random jitter to diversify starts
        seq0 += rng.normal(scale=0.001, size=n)
        seq0 = np.maximum(seq0, 0.0)
        if seq0.sum() == 0:
            seq0 = _initial_root_sequence(n)
        else:
            seq0 /= seq0.sum()

        # allocate a modest iteration budget; the optimiser stops early on its own
        max_iters = 3000

        # run optimisation for this length
        seq_opt, score_opt = _optimise_one_length(
            n=n,
            init_seq=seq0,
            max_iters=max_iters,
            time_limit=deadline,
            rng=rng,
        )

        if score_opt < global_best_score:
            global_best_score = score_opt
            global_best_seq = seq_opt.copy()
            print(
                f"[global improve] run={run_id}, n={n}, score={global_best_score:.6f}"
            )
        run_id += 1

    # final report
    if global_best_seq is None:
        # very unlikely; fall back to a default root sequence
        global_best_seq = _initial_root_sequence(5000)
        global_best_score = _evaluate_score(global_best_seq)

    print(f"[final] best_score={global_best_score:.6f}, length={global_best_seq.size}")
    # convert to plain Python list as required by the harness
    return global_best_seq.tolist()
