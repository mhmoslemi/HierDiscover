"""
Algorithm:
1. Initialise a binomial sequence (central binomial coefficients) of length n.
   This gives a baseline score of π/2.
2. Repeatedly solve a linear programme that, for the current sequence f,
   finds a non‑negative vector g maximizing its ℓ1‑norm while guaranteeing
   that the convolution (f∗g) does not exceed the current maximal
   self‑convolution M = max(f∗f).  The resulting g can carry considerably
   more mass because it concentrates the mass on indices where f is small.
3. Update the sequence by a small convex combination f←(1‑t)f + t g,
   renormalise to keep the total sum unchanged, and recompute the score.
   The mixture keeps the maximal convolution close to M while the
   ℓ1‑norm grows, which strictly lowers the evaluation value.
4. Iterate until time runs out or the score stops improving.
5. Return the best sequence found (as a plain Python list); a final call
   to the provided `evaluate_sequence` logs the official score.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def binomial_sequence(length: int) -> np.ndarray:
    """Return a truncated central‑binomial coefficient sequence a_i = C(2i,i)/4^i."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a

def conv_max(seq: np.ndarray) -> float:
    """Fast computation of max(f∗f) using an FFT‑based convolution."""
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    return float(conv.max())

def eval_score(seq: np.ndarray) -> float:
    """Return the exact evaluation value 2·n·max(conv)/(sum(seq)^2)."""
    n = len(seq)
    S = float(seq.sum())
    if S < 1e-12:
        return float("inf")
    M = conv_max(seq)
    return 2.0 * n * M / (S * S)

def build_constraint_matrix(a: np.ndarray) -> sp.csr_matrix:
    """Build the (n×n) matrix H with H[k,j] = a[k‑j] for j ≤ k, 0 otherwise."""
    n = a.shape[0]
    # Pre‑allocate dense matrix then convert to CSR – simple and fast for n≈3000.
    H = np.zeros((n, n), dtype=np.float64)
    for j in range(n):
        # rows k = j … n‑1 receive a[0 … n‑1‑j]
        H[j:, j] = a[: n - j]
    return sp.csr_matrix(H)

    """Search for a step‑function that beats the π/2 bound."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # -----------------------------------------------------------------
    n = 3000                     # 3000 gives a good trade‑off between quality and speed
    f = binomial_sequence(n)
    # Normalise to a convenient ℓ1‑norm; the scale does not affect the score.
    target_sum = np.sqrt(2.0 * n)
    f = f * (target_sum / f.sum())

    best_seq = f.copy()
    best_val = eval_score(best_seq)
    print(f"[init] n={n}, score={best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. iterative improvement using a linear programme and a small
    #    convex‑combination step.
    # -----------------------------------------------------------------
    t = 0.15                     # initial mixing weight
    iteration = 0
    while time.time() < deadline and iteration < 30:
        iteration += 1

        M = conv_max(f)
        S = f.sum()

        H = build_constraint_matrix(f)

        # Linear programme: maximise 1ᵀg  s.t.  H⋅g ≤ M , g ≥ 0
        c = -np.ones(n)                       # linprog minimises, so minimise –sum(g)

        res = opt.linprog(c, A_ub=H, b_ub=b, bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'time_limit': max(0.0, deadline - lp_start - 1.0)})
        if not res.success:
            print(f"[lp] iteration {iteration} failed – {res.message}")
            break

        g = res.x
        # -----------------------------------------------------------------
        # 3. Mix the new direction into the current sequence.
        #    t is gradually reduced; this keeps the maximal convolution
        #    close to the previous bound while the ℓ1‑norm grows.
        # -----------------------------------------------------------------
        new_f = (1.0 - t) * f + t * g
        new_val = eval_score(new_f)

        print(f"[iter{iteration:02d}] M={M:.6f}, sum={S:.3f}, t={t:.3f}, score={new_val:.6f}")

        if new_val < best_val - 1e-9:
            best_val = new_val
            best_seq = new_f.copy()
            f = new_f
            # shrink the step size a little to keep the process stable
            t = max(0.02, t * 0.85)
        else:
            # no improvement – try a smaller step next round
            t = max(0.02, t * 0.5)
            if t < 0.02:
                print("[stop] step size too small, terminating early")
                break

    # 4. final evaluation with the official checker
    # -----------------------------------------------------------------
    final = best_seq.tolist()
    try:
        official = evaluate_sequence(final)
        print(f"[final] official evaluation = {official:.6f}")
    except Exception as exc:  # defensive

    H = np.zeros((n, n), dtype=np.float64)          # dense intermediate is fast for n≈3000
    for j in range(n):
        H[j:, j] = f[: n - j]                       # column j gets the prefix of f
    return sp.csr_matrix(H)


# --------------------------------------------------------------
# Main entry point required by the framework
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a step‑function that beats the π/2 bound."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # -----------------------------------------------------------------
    # 1. initialise with the analytical construction
    # -----------------------------------------------------------------
    n = 3000                                 # a comfortable size for the LP
    f = binomial_sequence(n)
    # scale to a convenient ℓ1‑norm; the scale does not influence the score
    target_sum = np.sqrt(2.0 * n)
    f = f * (target_sum / f.sum())

    best_seq = f.copy()
    best_val = eval_score(best_seq)
    print(f"[init] n={n}, score={best_val:.6f}")

    # -----------------------------------------------------------------
    # 2. iterative improvement via LP + small convex combination
    # -----------------------------------------------------------------
    t = 0.15                 # initial mixing weight (will be reduced gradually)
    iteration = 0
    while time.time() < deadline and iteration < 30:
        iteration += 1

        # current maximal self‑convolution
        M = conv_max(f)
        S = f.sum()

        # build constraint matrix for (f * g) ≤ M
        H = build_constraint_matrix(f)

        # linear programme: maximise Σ g_j   ⇔   minimise –Σ g_j
        c = -np.ones(n)
        b = np.full(n, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        lp_start = time.time()
        res = opt.linprog(
            c,
            A_ub=H,
            b_ub=b,
            bounds=bounds,
            method='highs',
            options={
                'presolve': True,
                # leave a second for the outer loop and a safety margin
                'time_limit': max(0.0, deadline - lp_start - 1.0),
            },
        )
        if not res.success:
            print(f"[lp] iteration {iteration} failed – {res.message}")
            break

        g = res.x

        # -----------------------------------------------------------------
        # 3. convex mixture and score evaluation
        # -----------------------------------------------------------------
        new_f = (1.0 - t) * f + t * g
        new_val = eval_score(new_f)

        print(
            f"[iter{iteration:02d}] M={M:.6f}, sum={S:.3f}, t={t:.3f}, "
            f"score={new_val:.6f}"
        )

        if new_val < best_val - 1e-9:
            # improvement – keep the new point and shrink the step a bit
            best_val = new_val
            best_seq = new_f.copy()
            f = new_f
            t = max(0.02, t * 0.85)
        else:
            # no improvement – try a smaller step next round
            t = max(0.02, t * 0.5)
            if t <= 0.02:
                print("[stop] step size too small, terminating early")
                break

    # -----------------------------------------------------------------
    # 4. final validation with the official evaluator
    # -----------------------------------------------------------------
    final = best_seq.tolist()
    try:
        official = evaluate_sequence(final)
        print(f"[final] official evaluation = {official:.6f}")
    except Exception as exc:   # defensive
        print("[final] evaluation failed:", exc)

    return final
