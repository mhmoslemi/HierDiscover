"""
Propose a step‑function sequence with a low evaluation score.

The algorithm starts from the classic central‑binomial (Catalan) sequence,
which gives the known bound ≈π/2 ≃ 1.571.  To improve it we repeatedly
flatten the convolution peak:

* Compute the linear convolution a*a (FFT based) and locate the shift(s)
  where the maximum is attained.
* For those active shifts compute the gradient of the peak
  (∂/∂a_i  max_k (a∗a)[k]) ≈ Σ_{k active} (a_{i‑k}+a_{i+k}).
  The gradient is large for indices that contribute a lot to the peak
  (typically the centre) and small near the edges.
* Move a small fraction of mass from a high‑gradient index to a low‑gradient
  one.  Because the move reduces the peak while keeping the ℓ₁‑norm
  unchanged, the evaluation 2·n·max/(∑a)² strictly improves.
* Accept the move only if the fast‑score improves; otherwise discard it.
* Occasionally perform a tiny random perturbation to avoid stagnation.

The whole loop runs until the time budget is exhausted (default 1000 s) or
a safe iteration limit is hit.  The best sequence found is finally passed
to the official ``evaluate_sequence`` for the official score.

The method keeps the whole process vectorised (FFT convolutions) and
requires only a few hundred thousand cheap iterations, comfortably fitting
the 1000 s limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Catalan‑type binomial sequence a_i = C(2i,i) / 4**i
# ----------------------------------------------------------------------
def binomial_sequence(length: int) -> np.ndarray:
    """Return the truncated central‑binomial sequence of given length."""
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Fast convolution / score (identical to evaluate_sequence but without checks)
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv) / (sum(seq)**2) using FFT (no security checks)."""
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    max_b = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 1e-12:               # should never happen
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Gradient of the active convolution peak (vectorised via FFT)
# ----------------------------------------------------------------------
def peak_gradient(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_{k∈active} (a_{i‑k}+a_{i+k}) for all i.
    Uses two FFT‑based convolutions; O(n log n).
    """
    n = len(a)
    # binary mask h[k] = 1 if shift k is active (k∈[0,n‑1])
    h = np.zeros(n, dtype=np.float64)
    h[active_shifts] = 1.0

    size = 1 << ((2 * n - 1).bit_length())

    # term1 : Σ_k a[i‑k] * h[k]  (i ≥ k)
    f_a = np.fft.rfft(a, n=size)
    f_h = np.fft.rfft(h, n=size)
    conv1 = np.fft.irfft(f_a * f_h, n=size)[: 2 * n - 1]
    term1 = conv1[:n]                     # only indices 0…n‑1 are needed

    # term2 : Σ_k a[i+k] * h[k]  (i+k < n)
    a_rev = a[::-1]
    f_rev = np.fft.rfft(a_rev, n=size)
    conv2 = np.fft.irfft(f_rev * f_h, n=size)[: 2 * n - 1]
    # conv2[t] = Σ_k a_rev[t‑k] h[k]; we need t = n‑1‑i
    term2 = conv2[n - 1 :: -1][:n]        # reverse‑slice, length n

    return term1 + term2   # g_i

    """

    reference bound 1.503.  The routine performs a gradient‑guided
    mass‑redistribution hill‑climb starting from a binomial sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1) initialise – length can be tuned; 5000 gives a comfortable trade‑off
    # ------------------------------------------------------------------
    n = 5000
    if "height_sequence_1" in globals():
        # try to reuse a previously found construction if it exists
        a0 = np.array(height_sequence_1, dtype=np.float64)
        if len(a0) != n:
            a0 = binomial_sequence(n)
    else:
        a0 = binomial_sequence(n)
    a = a0.copy()

    best_seq = a.copy()
    best_val = fast_score(best_seq)
    print(f"[init] length={n}, fast_score={best_val:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    attempts = 0

    tol_rel = 1e-12                       # tolerance for defining the active set

    while time.time() < deadline and attempts < max_attempts:
        attempts += 1

        # ---- compute convolution and locate the active shifts -------------
        size = 1 << ((2 * n - 1).bit_length())
        f = np.fft.rfft(a, n=size)
        conv_full = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
        max_conv = float(conv_full.max())

        # map indices > n‑1 to their symmetric counterpart (the convolution is
        # symmetric).  Afterwards we keep only 0…n‑1.
        idxs = np.where(conv_full >= max_conv - max_conv * tol_rel)[0]
        # mirror indices larger than n‑1

        active = np.unique(np.minimum(idxs, mirrored))
        active = active[active <= n - 1]          # keep only one side

        # ---- compute gradient of the peak ---------------------------------
        g = peak_gradient(a, active)

        # ---- pick source (high gradient & non‑zero) and target (low gradient) --

        src_candidates = np.where(a > 1e-12)[0]

        src_scores = g[src_candidates] * a[src_candidates]
        src = src_candidates[np.argmax(src_scores)]

        # target prefers low gradient; we also prefer indices with small current mass
        tgt = int(np.argmin(g))

        # avoid degenerate move (same index)

        delta = move_frac * a[src]            # move 2 % of source mass
        if delta <= 0.0:
            continue

        # ---- try the move -------------------------------------------------
        new_a = a.copy()
        new_a[src] -= delta
        new_a[tgt] += delta

        # safety: keep non‑negative (should already hold)
        if new_a[src] < 0.0:
            continue

        new_val = fast_score(new_a)
        if new_val < best_val:
            best_val = new_val
            best_seq = new_a.copy()
            a = new_a                         # continue from the improved point
            print(f"[improve] iter={attempts}, new_score={best_val:.6f}")

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------

                    a = tmp

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    try:

        print(f"[final] official evaluate_sequence = {official:.6f}")
    except Exception as exc:  # defensive – should never happen
        print("[final] evaluation failed:", exc)

    return final_list
