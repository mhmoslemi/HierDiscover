def proj_simplex(v):
    # Returns projection onto the simplex {x >=0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function height sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three phases

1. **Initialisation** – start from the previous best construction if
   ``height_sequence_1`` is present, otherwise from a random simplex point.

2. **Smoothed‑gradient descent** – minimise a smooth approximation of the
   max‑convolution using a soft‑max (temperature β).  Each iteration performs
   a projected gradient step on the probability simplex.  β and the step size
   are annealed from a small to a large value, allowing global exploration
   early and fine tuning later.

3. **Linear‑programming refinement** – once the gradient stage has spent
   ~70 % of the time budget, a short alternating LP (the “alternating
   optimisation” described in the problem statement) is performed.
   For a fixed sequence ``f`` we solve

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   ∀k
                    g_i ≥ 0

   where ``M = max(f*f)``.  The optimal ``g`` is mixed back into ``f`` with a
   small step size.  A few LP iterations usually lower the peak further.

The whole search respects the 1000 s wall‑clock limit, prints progress,
and finally returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """
    Exact evaluation (without the security wrapper) for a non‑negative array.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * len(a) * b.max() / (s * s))


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Reference:  Duchi et al., “Efficient Projections onto the L1‑Ball”.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical fallback
        return np.maximum(v, 0.0) / max(np.sum(np.maximum(v, 0.0)), 1e-12)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 2 * n - 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    # numerical noise may create tiny negative values
    conv[conv < 0] = 0.0
    return conv


def softmax_gradient_step(a: np.ndarray, step: float, beta: float) -> np.ndarray:
    """
    One projected gradient step on the smoothed objective

        Jβ(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

    The gradient is 2·corr(p, a) where
    p_k ∝ exp(β·(a*a)_k).
    """
    conv = conv_fft(a)                     # length 2n‑1
    # soft‑max probabilities (log‑sum‑exp trick)
    scaled = beta * conv
    shift = np.max(scaled)
    exp_vals = np.exp(scaled - shift)
    p = exp_vals / exp_vals.sum()          # Σ p = 1
    # gradient = 2·correlation(p, a)  (valid mode → length n)
    grad = 2.0 * np.correlate(p, a, mode='valid')
    a_new = a - step * grad
    return proj_simplex(a_new)


# ----------------------------------------------------------------------
# LP refinement (alternating optimisation)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:      # pragma: no cover
    _HAS_CVX = False


def lp_refine(f: np.ndarray, t: float, max_iter: int, deadline: float) -> np.ndarray:
    """
    Perform a handful of alternating LP refinements on ``f``.
    Each iteration solves:
        maximise Σ g_i   s.t. (f * g)_k ≤ M  ∀k , g ≥ 0
    where M = max(f*f).  The new point is
        f ← (1‑t)·f + t·(g / Σg) .
    The process stops when time runs out or when the LP fails.
    """
    if not _HAS_CVX:
        return f  # cannot improve without cvxpy

    n = f.shape[0]
    # Pre‑allocate the convolution‑matrix template (Toeplitz‑like)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current maximal autocorrelation
        conv_f = conv_fft(f)
        M = conv_f.max()

        # Build the linear map  g ↦ f*g
        # Column i of A contains f[i] on rows i … i+n‑1
        # (other entries stay zero)
        A.fill(0.0)
        for i in range(n):
            A[i:i + n, i] = f[i]

        # LP: maximise sum(g) subject to A·g ≤ M, g ≥ 0
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
        except Exception:      # pragma: no cover
            break

        if g.value is None:
            break

        g_val = np.maximum(g.value, 0.0)
        sum_g = g_val.sum()
        if sum_g == 0.0:
            break
        g_norm = g_val / sum_g
        # convex combination
        f = (1.0 - t) * f + t * g_norm
    return f

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑evaluation step‑function within ``budget_s`` seconds.
    The algorithm mixes a soft‑max gradient descent (majority of the time)
    with a short alternating‑LP refinement at the end.
    Returns the best sequence found as a list of Python floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    n = int(kwargs.get("n", 2000))

    # ------------------------------------------------------------------
    # initialise a (use previous best if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.shape[0] >= n:
            a = prev[:n].copy()
        else:
            extra = np.random.rand(n - prev.shape[0])
            a = np.concatenate([prev, extra])
    else:
        a = np.random.rand(n)

    a = np.maximum(a, 0.0)
    a = proj_simplex(a)          # ensure ∑a = 1

    best_seq = a.copy()
    best_score = compute_score(best_seq)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # PARAMETERS (gradient phase)
    # ------------------------------------------------------------------
    init_step = 0.20
    final_step = 0.001
    beta_start = 5.0
    beta_end = 200.0

    # allocate ~70 % of the budget to gradient descent
    gradient_deadline = start + 0.70 * budget_s

    it = 0
    last_improve = start

    # ------------------------------------------------------------------
    # PHASE 1 – smoothed gradient descent (simulated annealing flavour)
    # ------------------------------------------------------------------
    while time.time() < gradient_deadline:
        elapsed = time.time() - start
        progress = elapsed / budget_s
        step = init_step * (1.0 - progress) + final_step * progress
        beta = beta_start * (1.0 - progress) + beta_end * progress

        # Occasionally inject a random move to increase diversity
        if np.random.rand() < 0.03:
            # simple random transfer
            i, j = np.random.choice(n, size=2, replace=False)
            delta = step * a[i]
            a_tmp = a.copy()
            a_tmp[i] -= delta
            a_tmp[j] += delta
            a_tmp = proj_simplex(a_tmp)
            cand = a_tmp
        else:
            cand = softmax_gradient_step(a, step, beta)

        cand_score = compute_score(cand)

        # Metropolis‑like acceptance (allow rare uphill moves)
        if cand_score < best_score or np.random.rand() < 0.001:
            a = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                last_improve = time.time()
                print(f"[{it:7d}] new best = {best_score:.6f}  t={elapsed:.1f}s")

        # simple stagnation restart
        if it % 20000 == 0 and (time.time() - last_improve) > 30.0:
            a = np.random.rand(n)
            a = proj_simplex(a)
            print(f"[{it:7d}] restart due to stagnation")
            last_improve = time.time()

        it += 1

    # ------------------------------------------------------------------
    # PHASE 2 – alternating LP refinement (if time permits)
    # ------------------------------------------------------------------
    if _HAS_CVX and time.time() < deadline:
        # a small mixing parameter – larger values would move too aggressively
        t_mix = 0.15
        # allow at most 8 LP iterations (each may take a few seconds)
        max_lp_iters = 8
        a_lp = best_seq.copy()
        a_lp = lp_refine(a_lp, t_mix, max_lp_iters, deadline)

        # evaluate the refined candidate
        lp_score = compute_score(a_lp)
        if lp_score < best_score:
            best_score = lp_score
            best_seq = a_lp.copy()
            print(f"[LP] improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final reporting
    # ------------------------------------------------------------------
    total = time.time() - start
    print(f"[done] iters={it}, best score={best_score:.6f}, time={total:.1f}s")
    # convert to plain Python list of floats for the harness
    return best_seq.tolist()
