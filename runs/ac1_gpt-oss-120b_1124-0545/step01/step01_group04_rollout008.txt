```
def solve_lp_for_b(f, M):
    n = len(f)
    A = build_constraint_matrix(f)
    # b_ub vector
    ub = np.full(2*n-1, M)
    c = -np.ones(n)   # minimize -sum b => maximize sum b

    # using bounds (0, None)
    res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method='highs', options={'presolve': True})
    if not res.success:
        print("[lp] linprog failed:", res.message)
        return None
    b = res.x
    # ensure non-negative
    b = np.maximum(b, 0.0)
    # normalise sum to 1
    s = b.sum()
    if s == 0:
        return None
    return b / s
```

```
"""
Search for a non‑negative step function that minimizes the
``evaluate_sequence`` metric.

Algorithm
---------

1. Initialise the coefficients with the (truncated) Catalan sequence,
   which already gives a bound ≈π/2≈1.571.

2. Repeatedly improve the shape by solving a linear programme:
   given the current vector ``f`` and its maximal self‑convolution ``M``,
   maximise Σ b_j under the constraints
       b_j ≥ 0,
       (f * b)_k ≤ M   for every shift k.
   The optimiser (SciPy ``linprog``) returns a vector ``b`` that can carry
   more total mass than ``f`` while respecting the same convolution bound.
   Normalising ``b`` to unit sum and mixing it with ``f`` (``f←(1‑t)f+t b``)
   strictly reduces the bound after a few iterations.

3. Finish with a cheap stochastic local‑search phase (small random mass
   transfers with simulated‑annealing acceptance) to polish the result.

The whole search stops when the supplied time budget expires (≈1000 s)
and returns the best sequence seen.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def compute_catalan_sequence(n: int) -> np.ndarray:
    """Return the first *n* Catalan‑type coefficients
    c_i = (2i choose i) / 4**i, normalised to sum to 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a /= a.sum()
    return a


def max_convolution_fft(a: np.ndarray, fft_len: int) -> tuple[float, int]:
    """Return (max value, index) of the linear convolution a*a using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    conv = conv[: 2 * len(a) - 1]          # keep only linear part
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def build_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A of shape (2·n‑1, n) such that
    (A @ b)[k] = (f * b)[k] holds for every shift k.
    Column ``j`` contains the vector ``f`` shifted down by ``j`` rows.
    """
    n = len(f)
    # diags[d] = f[:n-d] will be placed on the diagonal with offset = d
    diags = [f[: n - d] for d in range(n)]
    offsets = np.arange(n, dtype=int)          # 0,1,…,n‑1
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def solve_lp_for_b(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear programme
        maximise   Σ b_j
        subject to b ≥ 0,
                   (f * b)_k ≤ M   for all k.
    Returns a normalised vector ``b`` with Σ b = 1, or ``None`` on failure.
    """
    n = len(f)
    A = build_constraint_matrix(f)
    ub = np.full(2 * n - 1, M, dtype=np.float64)    # (f*b)_k ≤ M
    c = -np.ones(n, dtype=np.float64)               # maximise Σ b ⇒ minimise –Σ b
    # bounds for each variable: (0, ∞)
    bounds = [(0, None) for _ in range(n)]

    # Solve with the HiGHS interior‑point solver (fast for moderate size)
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method="highs")
    if not res.success:
        print("[lp] linprog failed:", res.message)
        return None

    b = np.maximum(res.x, 0.0)
    total = b.sum()
    if total == 0.0:
        return None
    return b / total


def random_perturb(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Transfer a fraction ``eps`` of the mass from a random index i
    to a different random index j (i≠j). Result stays normalised.
    """
    n = len(a)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new

    """
    Transfer a fraction ``eps`` of the mass from a random index i to a
    different random index j (i≠j).  Result stays normalised.
    """
    n = len(a)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return a
    delta = eps * a[i]
    if delta <= 0.0:
        return a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point for the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimise
    ``evaluate_sequence``.  The function respects the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a thin margin for return

    # ------------------------------------------------------------------
    # Choose a length and an initial vector
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=np.float64)
        n = len(init)
        a = init / init.sum()
        print(f"[init] start from previous best, length {n}")
    except NameError:
        n = 1200
        a = compute_catalan_sequence(n)
        print(f"[init] Catalan start, length {n}")

    # ------------------------------------------------------------------
    # Fast‑convolution settings
    # ------------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Track the best sequence discovered so far
    # ------------------------------------------------------------------
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Linear‑programming / mixing iterations
    # ------------------------------------------------------------------
    mix_t = 0.3               # initial mixing coefficient
    lp_iter = 0
    while time.time() < deadline - 5.0:   # reserve a few seconds for polishing
        M, _ = max_convolution_fft(a, fft_len)

        b = solve_lp_for_b(a, M)
        if b is None:
            print("[lp] LP failed – exiting LP loop")
            break

        # Mix the current vector with the LP‑derived partner
        a_candidate = (1.0 - mix_t) * a + mix_t * b
        a_candidate /= a_candidate.sum()          # normalization (should stay 1)

        cur_score = evaluate_sequence(a_candidate.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            print(f"[lp] iter {lp_iter:4d} – improved score {best_score:.6f}")

        # Accept the candidate for the next round (the bound never increases
        # theoretically, but we keep the latest vector regardless)
        a = a_candidate

        # Decay the mixing factor so later updates become more conservative
        mix_t = max(mix_t * 0.95, 0.02)
        lp_iter += 1

    # ------------------------------------------------------------------
    # 2. Stochastic local‑search polishing (as in the reference code)
    # ------------------------------------------------------------------
    eps = 0.08
    temperature = 0.12
    it = 0
    while time.time() < deadline:
        cand = random_perturb(best_seq, eps)
        cand_score = evaluate_sequence(cand.tolist())

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[pert] iter {it:5d} – improved score {best_score:.6f}")
        else:
            # Occasionally accept uphill moves (simulated‑annealing)
            prob = np.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                best_seq = cand.copy()

        eps = max(eps * 0.9997, 1e-4)
        temperature *= 0.9995
        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence (as plain Python list)
    # ------------------------------------------------------------------
    final_seq = np.clip(best_seq, 0.0, 1000.0)   # safety against overflow
    return final_seq.tolist()
```
